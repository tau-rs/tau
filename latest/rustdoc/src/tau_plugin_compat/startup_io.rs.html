1//! Per-plugin startup-IO path helpers for Layer 4 tests.
2//!
3//! The `tau-sandbox-native::light::BASELINE_SYSTEM_READ_PATHS` runtime
4//! constant covers paths that every Rust binary needs (libc/dyld bootstrap,
5//! `/proc/self`, `/sys/fs/cgroup`, etc.). This module supplies the
6//! long-tail of plugin-specific paths beyond that baseline — for example,
7//! an HTTP plugin that reads a distribution-specific TLS root cert bundle
8//! that isn't covered by the `/etc` baseline entry.
9//!
10//! Tests in `tau-plugin-compat/tests/layer4_native.rs` call
11//! `startup_io_paths_for(plugin_bin)` and add the returned paths as an
12//! additional `cap_fs_read` entry in the test's `SandboxPlan`.
13
14/// Return plugin-specific filesystem paths needed at startup that aren't
15/// already covered by `tau-sandbox-native`'s
16/// `BASELINE_SYSTEM_READ_PATHS`.
17///
18/// `plugin_bin` is the plugin binary's basename (e.g. `"shell-plugin"`,
19/// `"anthropic-plugin"`). Unknown plugins return an empty slice — the
20/// caller is responsible for either providing a binding here or relying
21/// solely on the runtime baseline.
22///
23/// Empty arms for shell + fs-read in PR 1 reflect that those plugins
24/// don't touch any plugin-specific paths beyond the runtime baseline
25/// (T1 findings 2026-05-09). HTTP plugin arms (anthropic, ollama, openai)
26/// are populated in PR 2 (`feat/layer4-startup-io-http`).
27pub fn startup_io_paths_for(plugin_bin: &str) -> &'static [&'static str] {
28    match plugin_bin {
29        // PR 1 — simple plugins. No plugin-specific paths needed beyond
30        // the runtime baseline (per T1 findings).
31        "shell-plugin" => &[],
32        "fs-read-plugin" => &[],
33        // PR 2 — HTTP plugins. Populated in feat/layer4-startup-io-http.
34        "anthropic-plugin" => &[],
35        "ollama-plugin" => &[],
36        "openai-plugin" => &[],
37        // Unknown plugin: caller bears responsibility.
38        _ => &[],
39    }
40}

50            "shell does not need plugin-specific startup paths in PR 1"

52    }

54    #[test]

56        assert!(
57            startup_io_paths_for("fs-read-plugin").is_empty(),
58            "fs-read does not need plugin-specific startup paths in PR 1"
59        );
60    }
61
62    #[test]
63    fn unknown_plugin_returns_empty() {
64        assert!(startup_io_paths_for("nonexistent-plugin").is_empty());
65    }
