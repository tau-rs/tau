1//! Workflow runner: dispatches each step in order.
2//!
3//! For v1, the runner is linear. Each step's output feeds future steps'
4//! `${steps.<id>.output}` templates. A failed step aborts the run.
5//! Persistence is append-only JSONL (see `crate::persistence`).
6

28/// Per-run options.
29///
30/// # Examples
31///
32/// Build a fresh `RunOpts` for a first run with no prior completed steps:
33///
34/// ```
35/// use tau_workflow::RunOpts;
36///
37/// let opts = RunOpts {
38///     input: "analyse the Q3 report".into(),
39///     run_id: None,
40///     completed: vec![],
41///     agents: std::collections::BTreeMap::new(),
42/// };
43///
44/// assert_eq!(opts.input, "analyse the Q3 report");
45/// assert!(opts.run_id.is_none());
46/// assert!(opts.completed.is_empty());
47/// ```
48#[derive(Debug, Clone)]
49pub struct RunOpts {
50    /// Caller-supplied input string available as `${input}`.
51    pub input: String,
52    /// Optional pre-existing run id (used by `--resume`). When `None`,
53    /// a fresh ULID is allocated.
54    pub run_id: Option<String>,
55    /// Already-completed step records (from replay). The runner skips
56    /// any step whose id appears here with status Ok.
57    pub completed: Vec<StepRecord>,
58    /// Agent definitions resolved from tau.toml by the caller.
59    pub agents: BTreeMap<String, (AgentDefinition, PackageManifest)>,
60}

62/// Outcome of a single run invocation.
63#[derive(Debug, Clone)]
64pub struct RunOutcome {
65    /// The run id (ULID).
66    pub run_id: String,
67    /// Path to the JSONL log on disk.
68    pub log_path: PathBuf,
69    /// Whether every step completed with status Ok.
70    pub success: bool,
71    /// The last step's output (or empty on failure).
72    pub last_output: String,
73}

76    /// Construct a runner.
77    ///
78    /// # Examples
79    ///
80    /// (`no_run` because `Runtime::builder().build()` requires a registered
81    /// LLM backend and project config that are only available at runtime.)
82    ///
83    /// ```no_run
84    /// use tau_workflow::Runner;
85    /// use tau_runtime_tokio::Runtime;
86    /// use std::sync::Arc;
87    /// use std::path::PathBuf;
88    ///
89    /// # tokio_test::block_on(async {
90    /// let runtime = Arc::new(
91    ///     Runtime::builder().build().expect("runtime should build"),
92    /// );
93    /// let runner = Runner::new(runtime, PathBuf::from("/workspace"));
94    /// // runner is ready to execute workflows against the given scope root.
95    /// # });
96    /// ```
97    pub fn new(runtime: Arc<Runtime>, scope_root: PathBuf) -> Self {
98        Self {
99            runtime,
100            scope_root,
101        }
102    }
103
104    /// Run (or resume) a workflow.
105    pub async fn run(
106        &self,
107        workflow: &Workflow,
108        opts: RunOpts,
109    ) -> Result<RunOutcome, WorkflowError> {
110        let run_id = opts
111            .run_id
112            .clone()
113            .unwrap_or_else(|| ulid::Ulid::new().to_string());
114        let log_path = run_log_path(&self.scope_root, &workflow.name, &run_id);
115        let mut log = RunLog::open_for_write(&log_path).await?;
116
117        // Seed prior_outputs from completed records (resume path).
118        let mut prior_outputs: BTreeMap<String, String> = BTreeMap::new();
119        let mut completed_ids: BTreeMap<String, ()> = BTreeMap::new();
120        for record in &opts.completed {
121            if record.status == StepStatus::Ok {
122                prior_outputs.insert(record.step_id.clone(), record.output.clone());
123                completed_ids.insert(record.step_id.clone(), ());
124            }
125        }
126
127        let mut last_output = String::new();
128
129        for (idx, step) in workflow.steps.iter().enumerate() {
130            if completed_ids.contains_key(&step.id) {
131                last_output = prior_outputs.get(&step.id).cloned().unwrap_or_default();
132                continue;
133            }
134
135            let started_at = Utc::now();
136            let (input_str, output_result) = match &step.kind {
137                StepKind::AgentRun { agent, input } => {
138                    let resolved_input = resolve_template(
139                        input,
140                        &opts.input,
141                        &prior_outputs,
142                        &workflow.name,
143                        &step.id,
144                    )?;
145                    let (agent_def, manifest) =
146                        opts.agents
147                            .get(agent)
148                            .ok_or_else(|| WorkflowError::AgentNotFound {
149                                workflow: workflow.name.clone(),
150                                step_id: step.id.clone(),
151                                agent: agent.clone(),
152                            })?;
153                    let initial_message = build_user_message(resolved_input.clone());
154                    // Host-shell contract (`stream.rs::clock_ref`): inject the
155                    // production clock + randomness or `run_streaming_inner`
156                    // panics on the first port use.
157                    let mut run_opts = tau_runtime_tokio::RunOptions::default();
158                    run_opts.clock = Some(std::sync::Arc::new(tau_runtime_tokio::TokioClock));
159                    run_opts.random = Some(std::sync::Arc::new(tau_runtime_tokio::OsRandom));
160                    let result = self
161                        .runtime
162                        .run(
163                            agent_def.clone(),
164                            manifest.clone(),
165                            initial_message,
166                            run_opts,
167                        )
168                        .await;
169                    (resolved_input, agent_outcome_to_string(result))
170                }
171                StepKind::ToolCall { tool, args } => {
172                    // Resolve any template strings inside args (string values only).
173                    let resolved_args =
174                        resolve_args(args, &opts.input, &prior_outputs, &workflow.name, &step.id)?;
175                    let default_agent_id = workflow.default_agent.as_ref().ok_or_else(|| {
176                        WorkflowError::ParseFailed {
177                            path: workflow.source_path.clone(),
178                            message: "tool.call step requires [workflow].default-agent".into(),
179                        }
180                    })?;
181                    let (agent_def, manifest) =
182                        opts.agents.get(default_agent_id).ok_or_else(|| {
183                            WorkflowError::AgentNotFound {
184                                workflow: workflow.name.clone(),
185                                step_id: step.id.clone(),
186                                agent: default_agent_id.clone(),
187                            }
188                        })?;
189                    // Convert serde_json::Value → tau_domain::Value for invoke_tool.
190                    let tau_args = json_to_tau_value(&resolved_args);
191                    // Inherent `invoke_tool` on the kernel Runtime — returns
192                    // tau_runtime_core::error::RuntimeError. We map below into
193                    // `tool_outcome_to_string` which accepts either flavor.
194                    let result = self
195                        .runtime
196                        .invoke_tool(agent_def, manifest, tool, tau_args)
197                        .await;
198                    let input_repr = serde_json::to_string(&resolved_args).unwrap_or_default();
199                    (input_repr, tool_outcome_to_string(result))
200                }
201            };
202
203            let ended_at = Utc::now();
204            let duration_ms = (ended_at - started_at).num_milliseconds().max(0) as u64;
205
206            let (output, status, error, detail) = match output_result {
207                Ok(out) => (out, StepStatus::Ok, None, None),
208                Err((err_kind, err_detail)) => (
209                    String::new(),
210                    StepStatus::Failed,
211                    Some(err_kind),
212                    Some(err_detail),
213                ),
214            };
215
216            let record = StepRecord {
217                ts: ended_at,
218                run_id: run_id.clone(),
219                step_id: step.id.clone(),
220                step_index: idx,
221                kind: match &step.kind {
222                    StepKind::AgentRun { .. } => "agent.run".into(),
223                    StepKind::ToolCall { .. } => "tool.call".into(),
224                },
225                input: input_str,
226                output: output.clone(),
227                started_at,
228                ended_at,
229                duration_ms,
230                status,
231                error,
232                detail,
233            };
234            log.append(&record).await?;
235
236            if status == StepStatus::Failed {
237                return Ok(RunOutcome {
238                    run_id,
239                    log_path,
240                    success: false,
241                    last_output: output,
242                });
243            }
244
245            prior_outputs.insert(step.id.clone(), output.clone());
246            last_output = output;
247        }
248
249        Ok(RunOutcome {
250            run_id,
251            log_path,
252            success: true,
253            last_output,
254        })
255    }
256}

258/// Construct a `Message` representing the user's initial input.
259///
260/// Uses `Message::new` with `Address::User` sender and a fresh
261/// `Address::Agent(AgentInstanceId::new())` recipient — matching the
262/// same pattern used by `tau-cli/src/cmd/run.rs`. The kernel replaces
263/// the recipient placeholder with its own `AgentInstanceId` once the
264/// loop starts.
265fn build_user_message(text: String) -> Message {
266    Message::new(
267        Address::User,
268        Address::Agent(AgentInstanceId::new()),
269        MessagePayload::Text { content: text },
270    )
271}

273/// Extract the final assistant text from a `RunOutcome`.
274///
275/// For `RunOutcome::Completed`, returns the text content of `final_message`.
276/// For `RunOutcome::Failed`, returns an error tuple with kind `"agent_failed"`
277/// and the status debug string.
278fn agent_outcome_to_string(
279    result: Result<tau_runtime_tokio::RunOutcome, tau_runtime_tokio::error::CoreRuntimeError>,
280) -> Result<String, (String, String)> {
281    match result {
282        Ok(tau_runtime_tokio::RunOutcome::Completed { final_message, .. }) => {
283            Ok(extract_text_from_payload(&final_message.payload))
284        }
285        Ok(tau_runtime_tokio::RunOutcome::Failed { status, .. }) => {
286            Err(("agent_failed".into(), format!("{status:?}")))
287        }
288        // #[non_exhaustive] forward-compat wildcard
289        Ok(_) => Err((
290            "unknown_outcome".into(),
291            "unknown RunOutcome variant".into(),
292        )),
293        Err(e) => Err(("runtime_error".into(), format!("{e}"))),
294    }
295}

297/// Project a [`MessagePayload`] to a text string. Non-text payloads
298/// fall back to a `Debug`-formatted preview, matching `tau-cli`'s
299/// `format_message_text` helper.
300fn extract_text_from_payload(payload: &MessagePayload) -> String {
301    match payload {
302        MessagePayload::Text { content } => content.clone(),
303        other => format!("{other:?}"),
304    }
305}

307/// Convert a `tau_ports::ToolResult` to a string for the JSONL output column.
308///
309/// On `is_error = true`, the step is surfaced as failed so the run aborts.
310/// On success, the content blocks are serialized as a JSON string.
311fn tool_outcome_to_string(
312    result: Result<tau_ports::ToolResult, tau_runtime_tokio::error::CoreRuntimeError>,
313) -> Result<String, (String, String)> {
314    match result {
315        Ok(tool_result) => {
316            if tool_result.is_error {
317                // Tool returned an error result. Surface as a failed step.
318                return Err((
319                    "tool_error".into(),
320                    serde_json::to_string(&tool_result).unwrap_or_default(),
321                ));
322            }
323            // Success — render the content as a JSON string for the log.
324            Ok(serde_json::to_string(&tool_result).unwrap_or_default())
325        }
326        Err(e) => Err(("runtime_error".into(), format!("{e}"))),
327    }
328}

330/// Convert `serde_json::Value` → `tau_domain::Value`.
331///
332/// `tau_domain::Value` has no `From<serde_json::Value>` impl, so we
333/// pattern-match each variant manually.
334///
335/// - `serde_json::Number`: integers fit in `i64` → `Value::Integer`;
336///   everything else → `Value::Float`.
337/// - `serde_json::Null` → `Value::Null`.
338/// - Array/Object recurse.
339fn json_to_tau_value(value: &serde_json::Value) -> tau_domain::Value {
340    match value {
341        serde_json::Value::Null => tau_domain::Value::Null,
342        serde_json::Value::Bool(b) => tau_domain::Value::Bool(*b),
343        serde_json::Value::Number(n) => {
344            if let Some(i) = n.as_i64() {
345                tau_domain::Value::Integer(i)
346            } else {
347                tau_domain::Value::Float(n.as_f64().unwrap_or(f64::NAN))
348            }
349        }
350        serde_json::Value::String(s) => tau_domain::Value::String(s.clone()),
351        serde_json::Value::Array(arr) => {
352            tau_domain::Value::Array(arr.iter().map(json_to_tau_value).collect())
353        }
354        serde_json::Value::Object(map) => {
355            let mut out = std::collections::BTreeMap::new();
356            for (k, v) in map {
357                out.insert(k.clone(), json_to_tau_value(v));
358            }
359            tau_domain::Value::Object(out)
360        }
361    }
362}

364fn resolve_args(
365    args: &serde_json::Value,
366    input: &str,
367    prior: &BTreeMap<String, String>,
368    workflow: &str,
369    step_id: &str,
370) -> Result<serde_json::Value, WorkflowError> {
371    match args {
372        serde_json::Value::String(s) => {
373            let resolved = resolve_template(s, input, prior, workflow, step_id)?;
374            Ok(serde_json::Value::String(resolved))
375        }
376        serde_json::Value::Array(arr) => {
377            let resolved: Result<Vec<_>, _> = arr
378                .iter()
379                .map(|v| resolve_args(v, input, prior, workflow, step_id))
380                .collect();
381            Ok(serde_json::Value::Array(resolved?))
382        }
383        serde_json::Value::Object(map) => {
384            let mut out = serde_json::Map::with_capacity(map.len());
385            for (k, v) in map {
386                out.insert(k.clone(), resolve_args(v, input, prior, workflow, step_id)?);
387            }
388            Ok(serde_json::Value::Object(out))
389        }
390        other => Ok(other.clone()),
391    }
392}

394/// Compare a workflow's current step ids against a log's recorded step
395/// ids. Returns Err(DriftDetected) when they diverge; returns Ok(()) when
396/// the log's ids are a prefix of the workflow's (the resume case).
397///
398/// "Prefix match" means: every record in `logged_records` (in order)
399/// matches the corresponding step in `workflow.steps`. Trailing steps in
400/// the workflow that aren't yet in the log are fine (that's the work
401/// the resume will do).
402///
403/// # Examples
404///
405/// A partial log (prefix match) is accepted for `--resume`:
406///
407/// ```
408/// use tau_workflow::{check_drift, Workflow, WorkflowError};
409/// use tau_workflow::{StepRecord, StepStatus};
410/// use std::path::Path;
411/// use chrono::Utc;
412///
413/// let src = r#"
414/// [[steps]]
415/// id = "step-a"
416/// kind = "agent.run"
417/// agent = "helper"
418/// input = "${input}"
419///
420/// [[steps]]
421/// id = "step-b"
422/// kind = "agent.run"
423/// agent = "helper"
424/// input = "${steps.step-a.output}"
425/// "#;
426/// let workflow = Workflow::from_str(src, Path::new("workflows/demo.toml"))
427///     .expect("workflow TOML should parse");
428///
429/// // Only step-a has been logged so far; step-b is still pending.
430/// let now = Utc::now();
431/// let logged = vec![StepRecord {
432///     ts: now,
433///     run_id: "01HKZTEST".into(),
434///     step_id: "step-a".into(),
435///     step_index: 0,
436///     kind: "agent.run".into(),
437///     input: "hello".into(),
438///     output: "world".into(),
439///     started_at: now,
440///     ended_at: now,
441///     duration_ms: 10,
442///     status: StepStatus::Ok,
443///     error: None,
444///     detail: None,
445/// }];
446///
447/// assert!(check_drift(&workflow, &logged).is_ok(), "prefix match should be accepted");
448///
449/// // A mismatched step id triggers DriftDetected.
450/// let mut drifted = logged.clone();
451/// drifted[0].step_id = "WRONG".into();
452/// assert!(matches!(
453///     check_drift(&workflow, &drifted),
454///     Err(WorkflowError::DriftDetected { .. })
455/// ));
456/// ```
457pub fn check_drift(
458    workflow: &Workflow,
459    logged_records: &[StepRecord],
460) -> Result<(), WorkflowError> {
461    if logged_records.len() > workflow.steps.len() {
462        return Err(WorkflowError::DriftDetected {
463            logged: logged_records.iter().map(|r| r.step_id.clone()).collect(),
464            current: workflow.steps.iter().map(|s| s.id.clone()).collect(),
465        });
466    }
467    for (idx, record) in logged_records.iter().enumerate() {
468        if workflow.steps[idx].id != record.step_id {
469            return Err(WorkflowError::DriftDetected {
470                logged: logged_records.iter().map(|r| r.step_id.clone()).collect(),
471                current: workflow.steps.iter().map(|s| s.id.clone()).collect(),
472            });
473        }
474    }
475    Ok(())
476}

483    fn step(id: &str, agent: &str) -> crate::model::Step {
484        crate::model::Step {

503    fn record(idx: usize, step_id: &str) -> StepRecord {
504        let now = Utc::now();
505        StepRecord {
506            ts: now,
507            run_id: "01HK".into(),
508            step_id: step_id.into(),
509            step_index: idx,
510            kind: "agent.run".into(),
511            input: String::new(),
512            output: String::new(),
513            started_at: now,
514            ended_at: now,
515            duration_ms: 0,
516            status: StepStatus::Ok,
517            error: None,
518            detail: None,
519        }
520    }
