1//! Workflow step input templating.
2//!
3//! Recognizes two reference forms:
4//! - `${input}` → the workflow's user-supplied input string.
5//! - `${steps.<id>.output}` → the prior step's output, by id.
6//!
7//! Both are resolved at step-dispatch time, after that step's preceding
8//! steps have completed. Forward references (a step referencing a later
9//! step) are detected at workflow-parse time in the runner before any
10//! step runs — see runner.rs.
11
12use std::collections::BTreeMap;
13
14use crate::error::WorkflowError;
15
16/// Resolve `${...}` references in `template` against `input` + prior step
17/// outputs. Unknown references produce `WorkflowError::TemplateUnresolved`
18/// with `workflow` and `step_id` populated by the caller (we don't know
19/// them here).
20///
21/// Escape sequence: `$${` resolves to a literal `${`.
22///
23/// # Examples
24///
25/// Substitute `${input}`:
26///
27/// ```
28/// use tau_workflow::resolve_template;
29/// use std::collections::BTreeMap;
30///
31/// let result = resolve_template(
32///     "Hello, ${input}!",
33///     "world",
34///     &BTreeMap::new(),
35///     "my-workflow",
36///     "greeting-step",
37/// ).expect("template should resolve");
38///
39/// assert_eq!(result, "Hello, world!");
40/// ```
41///
42/// Chain prior step outputs using `${steps.<id>.output}`:
43///
44/// ```
45/// use tau_workflow::resolve_template;
46/// use std::collections::BTreeMap;
47///
48/// let mut prior = BTreeMap::new();
49/// prior.insert("gather".to_string(), "market analysis".to_string());
50///
51/// let result = resolve_template(
52///     "Summarise: ${steps.gather.output}",
53///     "",
54///     &prior,
55///     "pipeline",
56///     "summarise-step",
57/// ).expect("template should resolve");
58///
59/// assert_eq!(result, "Summarise: market analysis");
60/// ```
61///
62/// An unknown reference returns `WorkflowError::TemplateUnresolved`:
63///
64/// ```
65/// use tau_workflow::{resolve_template, WorkflowError};
66/// use std::collections::BTreeMap;
67///
68/// let err = resolve_template(
69///     "${steps.missing.output}",
70///     "x",
71///     &BTreeMap::new(),
72///     "pipeline",
73///     "step-b",
74/// ).expect_err("unknown reference should fail");
75///
76/// assert!(matches!(err, WorkflowError::TemplateUnresolved { .. }));
77/// ```
78pub fn resolve(
79    template: &str,
80    input: &str,
81    prior_outputs: &BTreeMap<String, String>,
82    workflow: &str,
83    step_id: &str,
84) -> Result<String, WorkflowError> {
85    let mut out = String::with_capacity(template.len());
86    let mut chars = template.char_indices().peekable();
87
88    while let Some((_, c)) = chars.next() {
89        if c == '$' {
90            // Lookahead for `$${` (escape) or `${...}` (reference).
91            if let Some(&(_, '$')) = chars.peek() {
92                chars.next(); // consume the second $
93                if let Some(&(_, '{')) = chars.peek() {
94                    chars.next(); // consume {
95                    out.push_str("${");
96                    continue;
97                } else {
98                    out.push_str("$$");
99                    continue;
100                }
101            }
102            if let Some(&(_, '{')) = chars.peek() {
103                chars.next(); // consume {
104                let mut key = String::new();
105                let mut closed = false;
106                for (_, ch) in chars.by_ref() {
107                    if ch == '}' {
108                        closed = true;
109                        break;
110                    }
111                    key.push(ch);
112                }
113                if !closed {
114                    return Err(WorkflowError::TemplateUnresolved {
115                        workflow: workflow.into(),
116                        step_id: step_id.into(),
117                        missing: format!("unterminated ${{{key}"),
118                    });
119                }
120                let value = resolve_key(&key, input, prior_outputs).ok_or_else(|| {
121                    WorkflowError::TemplateUnresolved {
122                        workflow: workflow.into(),
123                        step_id: step_id.into(),
124                        missing: key.clone(),
125                    }
126                })?;
127                out.push_str(value);
128                continue;
129            }
130        }
131        out.push(c);
132    }
133    Ok(out)
134}
135
136fn resolve_key<'a>(
137    key: &str,
138    input: &'a str,
139    prior_outputs: &'a BTreeMap<String, String>,
140) -> Option<&'a str> {
141    if key == "input" {
142        return Some(input);
143    }
144    // steps.<id>.output
145    let stripped = key.strip_prefix("steps.")?;
146    let id = stripped.strip_suffix(".output")?;
147    prior_outputs.get(id).map(String::as_str)
148}

150#[cfg(test)]
151mod tests {
152    use super::*;
153
154    fn empty_outputs() -> BTreeMap<String, String> {
155        BTreeMap::new()
156    }
157
158    fn one_output(id: &str, val: &str) -> BTreeMap<String, String> {
159        let mut m = BTreeMap::new();
160        m.insert(id.to_string(), val.to_string());
161        m
162    }
163
164    #[test]
165    fn resolves_input_reference() {
166        let out = resolve("hello ${input}!", "world", &empty_outputs(), "wf", "s").unwrap();
167        assert_eq!(out, "hello world!");
168    }
169
170    #[test]
171    fn resolves_step_output_reference() {
172        let outputs = one_output("a", "alpha");
173        let out = resolve("got ${steps.a.output}", "in", &outputs, "wf", "s").unwrap();
174        assert_eq!(out, "got alpha");
175    }
176
177    #[test]
178    fn unresolved_step_reference_errors() {
179        let err = resolve("${steps.nope.output}", "x", &empty_outputs(), "wf", "s").unwrap_err();
180        let msg = format!("{err}");
181        assert!(msg.contains("nope"), "got {msg}");
182    }
183
184    #[test]
185    fn passes_through_plain_text() {
186        let out = resolve("no templates here", "x", &empty_outputs(), "wf", "s").unwrap();
187        assert_eq!(out, "no templates here");
188    }
189
190    #[test]
191    fn escapes_double_dollar() {
192        let out = resolve("price: $${input}", "10", &empty_outputs(), "wf", "s").unwrap();
193        assert_eq!(out, "price: ${input}");
194    }
195
196    #[test]
197    fn unterminated_reference_errors() {
198        let err = resolve("${input", "x", &empty_outputs(), "wf", "s").unwrap_err();
199        assert!(format!("{err}").contains("unterminated"), "got {err}");
200    }
201}
