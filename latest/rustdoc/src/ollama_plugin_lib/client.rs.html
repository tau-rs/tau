1//! Ollama HTTP client with retry + optional bearer-token auth.
2//!
3//! `post_chat` is the only outbound entrypoint. Owns a `reqwest::Client`
4//! with TLS + timeout configured by the plugin's `Configure::from_config`.
5//!
6//! Retry policy per spec §4.1:
7//! - 2xx → return immediately.
8//! - 429, 503 → retry with exponential backoff; honor `Retry-After`
9//!   when `retry.respect_retry_after` is true. Exhausted →
10//!   `ClientError::Exhausted`. **503 is the load-bearing case for
11//!   Ollama** (returned during model load).
12//! - 4xx other than 429 → return immediately (caller maps to LlmError).
13//! - 5xx other than 503 → retry (treated as transient).
14//! - Network timeout → retry.
15//! - Other transport error (DNS, TLS, connection refused) →
16//!   `ClientError::Transport`, no retry.
17

26/// HTTP client that knows how to call Ollama's `/api/chat` endpoint.
27pub(crate) struct OllamaClient {
28    inner: reqwest::Client,
29    base_url: String,
30    bearer_token: Option<SecretString>,
31    retry: RetryConfig,
32}

34impl OllamaClient {

51    /// `POST /api/chat` with retry. The body is a `serde_json::Value`
52    /// produced by `request::build_chat_body`. The `stream` flag is
53    /// already encoded inside `body.stream`; no additional `Accept`
54    /// header is needed (Ollama emits NDJSON when `body.stream: true`).
55    pub(crate) async fn post_chat(
56        &self,
57        body: &serde_json::Value,
58        _stream: bool,
59    ) -> Result<Response, ClientError> {
60        let url = format!("{}/api/chat", self.base_url.trim_end_matches('/'));
61        let mut attempt: u32 = 0;
62
63        loop {
64            attempt += 1;
65
66            let mut req = self
67                .inner
68                .post(&url)
69                .header("content-type", "application/json")
70                .json(body);
71            if let Some(token) = &self.bearer_token {
72                req = req.header("authorization", format!("Bearer {}", token.expose_secret()));
73            }
74
75            let send_result = req.send().await;
76            match self.classify(send_result, attempt) {
77                Decision::Return(resp) => return Ok(resp),
78                Decision::Error(err) => return Err(err),
79                Decision::Retry { delay_ms, status } => {
80                    if attempt >= self.retry.max_attempts {
81                        return Err(ClientError::Exhausted {
82                            status,
83                            attempts: attempt,
84                        });
85                    }
86                    tracing::warn!(
87                        target: "ollama_plugin::retry",
88                        attempt,
89                        max = self.retry.max_attempts,
90                        delay_ms,
91                        status = status.as_u16(),
92                        "retrying transient error",
93                    );
94                    sleep(Duration::from_millis(delay_ms)).await;
95                }
96            }
97        }
98    }
99
100    fn classify(&self, res: reqwest::Result<Response>, attempt: u32) -> Decision {
101        match res {
102            Ok(resp) => {
103                let status = resp.status();
104                if status.is_success() {
105                    Decision::Return(resp)
106                } else if is_retryable_status(status) {
107                    let delay_ms = self.retry_delay(&resp, attempt);
108                    Decision::Retry { delay_ms, status }
109                } else {
110                    // Non-retryable 4xx (or unexpected): caller maps.
111                    Decision::Return(resp)
112                }
113            }
114            Err(e) if e.is_timeout() => {
115                let delay_ms = self.backoff_only(attempt);
116                // No status; use 408 (Request Timeout) as a placeholder
117                // for the Exhausted variant when timeouts pile up.
118                Decision::Retry {
119                    delay_ms,
120                    status: StatusCode::REQUEST_TIMEOUT,
121                }
122            }
123            Err(e) => Decision::Error(ClientError::Transport(e)),
124        }
125    }
126
127    fn retry_delay(&self, resp: &Response, attempt: u32) -> u64 {
128        if self.retry.respect_retry_after {
129            if let Some(secs) = resp
130                .headers()
131                .get("retry-after")
132                .and_then(|h| h.to_str().ok())
133                .and_then(|s| s.parse::<u64>().ok())
134            {
135                return secs * 1000;
136            }
137        }
138        self.backoff_only(attempt)
139    }
140
141    fn backoff_only(&self, attempt: u32) -> u64 {
142        // base * 2^(attempt-1), capped at 60s.
143        let shift = (attempt - 1).min(6);
144        let delay = self.retry.base_delay_ms.saturating_mul(1u64 << shift);
145        delay.min(60_000)
146    }
147}
148
149fn is_retryable_status(status: StatusCode) -> bool {
150    matches!(status.as_u16(), 429 | 503) || (status.as_u16() >= 500 && status.as_u16() != 501)
151}
152
153enum Decision {
154    Return(Response),
155    Retry { delay_ms: u64, status: StatusCode },
156    Error(ClientError),
157}

167    /// Spawn a tiny HTTP/1.1 server on 127.0.0.1:0 that replies with
168    /// pre-canned responses for the first N connections, then becomes
169    /// silent. Returns the listener URL + a counter of accepted
170    /// connections + a vector of received request bytes (for asserting
171    /// header presence).
172    async fn spawn_canned_server(
173        responses: Vec<&'static str>,
174    ) -> (
175        String,
176        Arc<AtomicUsize>,
177        Arc<tokio::sync::Mutex<Vec<Vec<u8>>>>,
178    ) {
179        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
180        let port = listener.local_addr().unwrap().port();
181        let url = format!("http://127.0.0.1:{port}");
182        let counter = Arc::new(AtomicUsize::new(0));
183        let received = Arc::new(tokio::sync::Mutex::new(Vec::<Vec<u8>>::new()));
184        let counter_clone = counter.clone();
185        let received_clone = received.clone();
186
187        tokio::spawn(async move {
188            for canned in responses {
189                let (mut sock, _) = listener.accept().await.unwrap();
190                counter_clone.fetch_add(1, Ordering::Relaxed);
191                let mut buf = [0u8; 4096];
192                let n = sock.read(&mut buf).await.unwrap_or(0);
193                received_clone.lock().await.push(buf[..n].to_vec());
194                sock.write_all(canned.as_bytes()).await.unwrap();
195                sock.flush().await.unwrap();
196                drop(sock);
197            }
198        });
199
200        (url, counter, received)
201    }

212    fn service_unavailable_body(retry_after: Option<&str>) -> String {
213        let mut s = String::from("HTTP/1.1 503 Service Unavailable\r\n");
214        if let Some(ra) = retry_after {
215            s.push_str(&format!("retry-after: {ra}\r\n"));
216        }
217        s.push_str(
218            "content-type: application/json\r\n\
219             content-length: 0\r\n\
220             connection: close\r\n\
221             \r\n",
222        );
223        s
224    }

226    fn rate_limited_body(retry_after: Option<&str>) -> String {
227        let mut s = String::from("HTTP/1.1 429 Too Many Requests\r\n");
228        if let Some(ra) = retry_after {
229            s.push_str(&format!("retry-after: {ra}\r\n"));
230        }
231        s.push_str(
232            "content-type: application/json\r\n\
233             content-length: 0\r\n\
234             connection: close\r\n\
235             \r\n",
236        );
237        s
238    }

240    fn build_client(
241        base_url: String,
242        bearer_token: Option<SecretString>,
243        max_attempts: u32,
244        base_delay_ms: u64,
245    ) -> OllamaClient {
246        OllamaClient::new(
247            reqwest::Client::builder()
248                .timeout(Duration::from_secs(5))
249                .build()
250                .unwrap(),
251            base_url,

256                respect_retry_after: true,
257            },

261    #[tokio::test]
262    async fn post_chat_happy_path_no_bearer_token() {
263        let (url, counter, received) = spawn_canned_server(vec![ok_body()]).await;
264        let client = build_client(url, None, 3, 0);
265        let resp = client
266            .post_chat(&serde_json::json!({"model": "m", "messages": []}), false)
267            .await
268            .unwrap();
269        assert!(resp.status().is_success());
270        assert_eq!(counter.load(Ordering::Relaxed), 1);
271
272        // Critical: NO authorization header should have been sent.
273        let raw = received.lock().await;
274        let req_str = std::str::from_utf8(&raw[0]).unwrap_or("");
275        assert!(
276            !req_str.to_ascii_lowercase().contains("authorization"),
277            "expected NO Authorization header on local Ollama; got: {req_str}"
278        );
279    }
