1//! `ServerContract` — what tau pins for an MCP server.
2//!
3//! Captures the server's `initialize` response + `tools/list` snapshot
4//! at build time. PR-4's lowering pass canonical-hashes a `ServerContract`
5//! and stores `(url, contract_hash, expanded_tools)` in the lockfile.
6
7use alloc::string::String;
8use alloc::vec::Vec;
9use serde::{Deserialize, Serialize};
10use tau_domain::Capability;
11
12use crate::protocol::initialize::ServerInfo;
13use crate::protocol::tools::{McpTool, McpToolInputSchema};
14
15/// Frozen server contract.
16///
17/// One contract per MCP server URL. `tau build` constructs this from
18/// the live (or pinned) handshake; `canonical_hash` produces the
19/// `Hash256` that participates in the IR module hash.
20#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
21pub struct ServerContract {
22    /// MCP protocol version the server advertised at `initialize`.
23    pub protocol_version: String,
24    /// Server's reported info.
25    pub server_info: ServerInfo,
26    /// The full `tools/list` snapshot, in server order.
27    pub tools: Vec<ContractTool>,
28}
29
30/// One tool from the server's `tools/list` plus its declared caps.
31///
32/// `caps` is the **server-declared** capability set for this tool. PR-4
33/// intersects it with the author's per-server envelope before storing
34/// in the IR's `ToolImpl::Mcp::capability_subset`.
35#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
36pub struct ContractTool {
37    /// Server-side tool name.
38    pub name: String,
39    /// Server-supplied description (passed through to the LLM).
40    #[serde(default, skip_serializing_if = "Option::is_none")]
41    pub description: Option<String>,
42    /// Server-supplied input JSON schema.
43    pub input_schema: McpToolInputSchema,
44    /// Server-declared capabilities (per-tool).
45    ///
46    /// Note: MCP spec does NOT currently standardize a "capability
47    /// declaration" field on tools/list entries. tau extracts caps from
48    /// a tau-specific extension field; if the server doesn't ship it,
49    /// caps default to the empty vector and the author's envelope is
50    /// the upper bound (per the spec's "envelope ∩ contract" rule).
51    ///
52    /// β.3.1 may evolve this once the MCP spec lands per-tool caps.
53    #[serde(default)]
54    pub caps: Vec<Capability>,
55}
56
57impl ServerContract {
58    /// Build a `ServerContract` from a handshake-completed pair of
59    /// (`InitializeResponse`, `ToolsListResponse`). PR-2 + PR-3 wire
60    /// this from the live transport.
61    ///
62    /// `caps_extractor` lets the caller pull caps from a tau-specific
63    /// extension field on each `McpTool`; if the extension is absent
64    /// (most off-the-shelf servers), the closure returns `Vec::new()`
65    /// and the author's envelope is the upper bound.
66    pub fn from_handshake<F>(
67        init: crate::protocol::initialize::InitializeResponse,
68        tools_list: crate::protocol::tools::ToolsListResponse,
69        mut caps_extractor: F,
70    ) -> Self
71    where
72        F: FnMut(&McpTool) -> Vec<Capability>,
73    {
74        let tools = tools_list
75            .tools
76            .into_iter()
77            .map(|t| ContractTool {
78                caps: caps_extractor(&t),
79                name: t.name,
80                description: t.description,
81                input_schema: t.input_schema,
82            })
83            .collect();
84        ServerContract {
85            protocol_version: init.protocol_version,
86            server_info: init.server_info,
87            tools,
88        }
89    }
90}

92#[cfg(test)]
93mod tests {
94    use super::*;
95    use alloc::collections::BTreeMap;
96    use alloc::string::ToString;
97    use alloc::vec;
98    use serde_json::json;
99
100    fn weather_contract() -> ServerContract {
101        ServerContract {
102            protocol_version: "2025-03-26".to_string(),
103            server_info: ServerInfo {
104                name: "weather".to_string(),
105                version: "1.0".to_string(),
106                additional: BTreeMap::new(),
107            },
108            tools: vec![ContractTool {
109                name: "get_forecast".to_string(),
110                description: Some("Get weather forecast".to_string()),
111                input_schema: McpToolInputSchema(json!({
112                    "type":"object",
113                    "properties":{"lat":{"type":"number"},"lon":{"type":"number"}}
114                })),
115                caps: vec![],
116            }],
117        }
118    }
119
120    #[test]
121    fn server_contract_round_trips() {
122        let c = weather_contract();
123        let bytes = serde_json::to_vec(&c).expect("serialize");
124        let decoded: ServerContract = serde_json::from_slice(&bytes).expect("deserialize");
125        assert_eq!(c, decoded);
126    }
127
128    #[test]
129    fn from_handshake_constructs_contract() {
130        use crate::protocol::initialize::InitializeResponse;
131        use crate::protocol::tools::ToolsListResponse;
132
133        let init = InitializeResponse {
134            protocol_version: "2025-03-26".to_string(),
135            server_info: ServerInfo {
136                name: "weather".to_string(),
137                version: "1.0".to_string(),
138                additional: BTreeMap::new(),
139            },
140            capabilities: None,
141        };
142        let tools = ToolsListResponse {
143            tools: vec![McpTool {
144                name: "get_forecast".to_string(),
145                description: None,
146                input_schema: McpToolInputSchema(json!({"type":"object"})),
147            }],
148            next_cursor: None,
149        };
150        let c = ServerContract::from_handshake(init, tools, |_| vec![]);
151        assert_eq!(c.tools.len(), 1);
152        assert_eq!(c.tools[0].name, "get_forecast");
153        assert!(c.tools[0].caps.is_empty());
154    }
