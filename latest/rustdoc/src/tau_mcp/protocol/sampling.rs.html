1//! `sampling/createMessage` — server-initiated request asking host to
2//! invoke an LLM.
3//!
4//! Per the β.3 design doc §8.3 and §9: v0 routes this through the
5//! agent's `LlmBackend` filtered by the `sampling.models` allowlist.
6//! `modelPreferences` is parsed but ignored in v0 (β.3.1 adds it).
7
8use alloc::string::String;
9use alloc::vec::Vec;
10use serde::{Deserialize, Serialize};
11use serde_json::Value;
12
13/// `sampling/createMessage` request — server asks host for an LLM
14/// completion.
15#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
16pub struct SamplingCreateMessageRequest {
17    /// Chat-style message history.
18    pub messages: Vec<SamplingMessage>,
19    /// Model hints (intelligence / speed / cost weighting). v0 ignores;
20    /// β.3.1 honors.
21    #[serde(default, skip_serializing_if = "Option::is_none")]
22    #[serde(rename = "modelPreferences")]
23    pub model_preferences: Option<ModelPreferences>,
24    /// Optional system prompt the host should pass to the LLM.
25    #[serde(default, skip_serializing_if = "Option::is_none")]
26    #[serde(rename = "systemPrompt")]
27    pub system_prompt: Option<String>,
28    /// Optional inclusion of host-context in the request.
29    #[serde(default, skip_serializing_if = "Option::is_none")]
30    #[serde(rename = "includeContext")]
31    pub include_context: Option<String>,
32    /// Maximum tokens hint.
33    #[serde(default, skip_serializing_if = "Option::is_none")]
34    #[serde(rename = "maxTokens")]
35    pub max_tokens: Option<u32>,
36    /// Other parameters (temperature, stopSequences, etc.); preserved
37    /// across (de)serialization. BTreeMap keeps key order stable for
38    /// canonical hashing.
39    #[serde(
40        flatten,
41        default,
42        skip_serializing_if = "alloc::collections::BTreeMap::is_empty"
43    )]
44    pub additional: alloc::collections::BTreeMap<String, Value>,
45}
46
47/// A message in a sampling request.
48#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
49pub struct SamplingMessage {
50    /// Role (`"user"` | `"assistant"` per spec; tau forwards through).
51    pub role: String,
52    /// Content block — v0 supports text only on inbound sampling.
53    pub content: SamplingContent,
54}
55
56/// Content block of a sampling message.
57#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
58#[serde(tag = "type", rename_all = "lowercase")]
59pub enum SamplingContent {
60    /// Plain text.
61    Text {
62        /// The text.
63        text: String,
64    },
65    /// Image (base64 data + mime).
66    Image {
67        /// Base64-encoded image bytes.
68        data: String,
69        /// MIME type.
70        #[serde(rename = "mimeType")]
71        mime_type: String,
72    },
73}
74
75/// Model-preference hints (parsed in v0; not yet honored).
76#[derive(Debug, Clone, PartialEq, Default, Serialize, Deserialize)]
77pub struct ModelPreferences {
78    /// Server's hint for intelligence (0.0–1.0).
79    #[serde(default, skip_serializing_if = "Option::is_none")]
80    #[serde(rename = "intelligencePriority")]
81    pub intelligence_priority: Option<f32>,
82    /// Server's hint for speed (0.0–1.0).
83    #[serde(default, skip_serializing_if = "Option::is_none")]
84    #[serde(rename = "speedPriority")]
85    pub speed_priority: Option<f32>,
86    /// Server's hint for cost (0.0–1.0).
87    #[serde(default, skip_serializing_if = "Option::is_none")]
88    #[serde(rename = "costPriority")]
89    pub cost_priority: Option<f32>,
90    /// Server's hint for specific model names (free-form).
91    #[serde(default, skip_serializing_if = "Vec::is_empty")]
92    pub hints: Vec<ModelHint>,
93}
94
95/// One model-name hint.
96#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
97pub struct ModelHint {
98    /// Suggested model name.
99    pub name: String,
100}
101
102/// `sampling/createMessage` response — the host's LLM completion.
103#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
104pub struct SamplingCreateMessageResponse {
105    /// Role of the response (always `"assistant"` per spec).
106    pub role: String,
107    /// Completion content.
108    pub content: SamplingContent,
109    /// Model name actually used.
110    pub model: String,
111    /// Stop reason (`"endTurn"` | `"stopSequence"` | `"maxTokens"`).
112    #[serde(default, skip_serializing_if = "Option::is_none")]
113    #[serde(rename = "stopReason")]
114    pub stop_reason: Option<String>,
115}

117#[cfg(test)]
118mod tests {
119    use super::*;
120    use alloc::string::ToString;
121    use alloc::vec;
122    use serde_json::json;
123
124    #[test]
125    fn sampling_request_round_trips() {
126        let req = SamplingCreateMessageRequest {
127            messages: vec![SamplingMessage {
128                role: "user".to_string(),
129                content: SamplingContent::Text {
130                    text: "summarize".to_string(),
131                },
132            }],
133            model_preferences: Some(ModelPreferences {
134                intelligence_priority: Some(0.9),
135                speed_priority: Some(0.1),
136                cost_priority: None,
137                hints: vec![ModelHint {
138                    name: "claude-haiku".to_string(),
139                }],
140            }),
141            system_prompt: None,
142            include_context: None,
143            max_tokens: Some(512),
144            additional: alloc::collections::BTreeMap::new(),
145        };
146        let bytes = serde_json::to_vec(&req).expect("serialize");
147        let decoded: SamplingCreateMessageRequest =
148            serde_json::from_slice(&bytes).expect("deserialize");
149        assert_eq!(req, decoded);
150        // Suppress unused-import warning for `json!` macro now that
151        // `additional` no longer needs Value construction here.
152        let _ = json!({});
153    }
154
155    #[test]
156    fn sampling_response_round_trips() {
157        let resp = SamplingCreateMessageResponse {
158            role: "assistant".to_string(),
159            content: SamplingContent::Text {
160                text: "summary".to_string(),
161            },
162            model: "claude-haiku-4-5".to_string(),
163            stop_reason: Some("endTurn".to_string()),
164        };
165        let bytes = serde_json::to_vec(&resp).expect("serialize");
166        let decoded: SamplingCreateMessageResponse =
167            serde_json::from_slice(&bytes).expect("deserialize");
168        assert_eq!(resp, decoded);
169    }
