1//! `initialize` method — first request the host sends.
2//!
3//! The host advertises its protocol version + client info; the server
4//! responds with its protocol version + server info + capabilities.
5
6use alloc::collections::BTreeMap;
7use alloc::string::String;
8use serde::{Deserialize, Serialize};
9use serde_json::Value;
10
11/// `initialize` request params.
12#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
13pub struct InitializeRequest {
14    /// MCP protocol version the host speaks (tau v0 sends
15    /// `"2025-03-26"`).
16    #[serde(rename = "protocolVersion")]
17    pub protocol_version: String,
18    /// Client (host) info.
19    #[serde(rename = "clientInfo")]
20    pub client_info: ClientInfo,
21    /// Client capabilities (per MCP spec — free-form map).
22    #[serde(default, skip_serializing_if = "Option::is_none")]
23    pub capabilities: Option<Value>,
24}
25
26/// `initialize` response result.
27#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
28pub struct InitializeResponse {
29    /// MCP protocol version the server speaks.
30    #[serde(rename = "protocolVersion")]
31    pub protocol_version: String,
32    /// Server info.
33    #[serde(rename = "serverInfo")]
34    pub server_info: ServerInfo,
35    /// Server capabilities (per MCP spec — free-form map).
36    #[serde(default, skip_serializing_if = "Option::is_none")]
37    pub capabilities: Option<Value>,
38}
39
40/// Host-side client info (tau ships `name="tau"`, version = crate version).
41#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
42pub struct ClientInfo {
43    /// Client name (`"tau"` for tau-mcp).
44    pub name: String,
45    /// Client version (tau crate version string).
46    pub version: String,
47    /// Additional fields the server may report; preserved across
48    /// (de)serialization.
49    #[serde(flatten, default, skip_serializing_if = "BTreeMap::is_empty")]
50    pub additional: BTreeMap<String, Value>,
51}
52
53/// Server-side info reported by the MCP server.
54#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
55pub struct ServerInfo {
56    /// Server name (e.g. `"weather-mcp"`).
57    pub name: String,
58    /// Server version string.
59    pub version: String,
60    /// Additional fields the server may report; preserved across
61    /// (de)serialization.
62    #[serde(flatten, default, skip_serializing_if = "BTreeMap::is_empty")]
63    pub additional: BTreeMap<String, Value>,
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69    use alloc::string::ToString;
70    use serde_json::json;
71
72    #[test]
73    fn initialize_request_round_trips() {
74        let req = InitializeRequest {
75            protocol_version: "2025-03-26".to_string(),
76            client_info: ClientInfo {
77                name: "tau".to_string(),
78                version: "0.0.0".to_string(),
79                additional: BTreeMap::new(),
80            },
81            capabilities: Some(json!({"roots":{"listChanged":false}})),
82        };
83        let bytes = serde_json::to_vec(&req).expect("serialize");
84        let decoded: InitializeRequest = serde_json::from_slice(&bytes).expect("deserialize");
85        assert_eq!(req, decoded);
86    }
87
88    #[test]
89    fn initialize_response_round_trips() {
90        let resp = InitializeResponse {
91            protocol_version: "2025-03-26".to_string(),
92            server_info: ServerInfo {
93                name: "weather".to_string(),
94                version: "1.2.3".to_string(),
95                additional: BTreeMap::new(),
96            },
97            capabilities: Some(json!({"tools":{"listChanged":false}})),
98        };
99        let bytes = serde_json::to_vec(&resp).expect("serialize");
100        let decoded: InitializeResponse = serde_json::from_slice(&bytes).expect("deserialize");
101        assert_eq!(resp, decoded);
102    }
103
104    #[test]
105    fn server_info_preserves_additional_fields() {
106        // Real servers report extra fields tau doesn't know about; they
107        // must round-trip without loss.
108        let wire = json!({
109            "name":"weather","version":"1.0","author":"NWS","website":"https://weather.gov"
110        });
111        let info: ServerInfo = serde_json::from_value(wire.clone()).expect("decode");
112        assert_eq!(info.name, "weather");
113        assert_eq!(
114            info.additional.get("author").and_then(Value::as_str),
115            Some("NWS")
116        );
117        let reencoded = serde_json::to_value(&info).expect("encode");
118        assert_eq!(reencoded, wire);
119    }
120}
