1//! JSON-RPC 2.0 envelopes used by MCP.
2//!
3//! MCP uses JSON-RPC 2.0 (https://www.jsonrpc.org/specification) as its
4//! wire format, with `jsonrpc: "2.0"` on every envelope. Three envelope
5//! kinds:
6//!
7//! - [`JsonRpcRequest`] — a method call expecting a response.
8//! - [`JsonRpcResponse`] — a response (success `result` or error
9//!   `JsonRpcError`).
10//! - [`JsonRpcNotification`] — a fire-and-forget message (no `id`,
11//!   no response).
12//!
13//! [`JsonRpcMessage`] is the discriminated-union shape used over the
14//! wire — a single `serde_json::Value::Object` is parsed once and
15//! routed to the right variant by the presence/absence of `id` and
16//! `result`/`error`.
17
18use alloc::string::String;
19use serde::{Deserialize, Serialize};
20use serde_json::Value;
21
22/// JSON-RPC version this implementation speaks.
23pub const JSONRPC_VERSION: &str = "2.0";
24
25/// JSON-RPC 2.0 request-id. Per spec: number or string (or null for
26/// notifications — see [`JsonRpcNotification`] for those).
27#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
28#[serde(untagged)]
29pub enum RequestId {
30    /// Integer id (the common case in MCP).
31    Number(i64),
32    /// String id.
33    String(String),
34}
35
36/// A JSON-RPC 2.0 request envelope.
37#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
38pub struct JsonRpcRequest {
39    /// Always `"2.0"`.
40    pub jsonrpc: String,
41    /// Request id (echoed in the response).
42    pub id: RequestId,
43    /// Method name (e.g. `"tools/call"`).
44    pub method: String,
45    /// Method-specific parameters.
46    #[serde(default, skip_serializing_if = "Option::is_none")]
47    pub params: Option<Value>,
48}
49
50/// A JSON-RPC 2.0 response envelope (success).
51#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
52pub struct JsonRpcResponse {
53    /// Always `"2.0"`.
54    pub jsonrpc: String,
55    /// Echoed request id.
56    pub id: RequestId,
57    /// Success result; `None` if this is an error response.
58    #[serde(skip_serializing_if = "Option::is_none")]
59    pub result: Option<Value>,
60    /// Error payload; `None` on success.
61    #[serde(skip_serializing_if = "Option::is_none")]
62    pub error: Option<JsonRpcError>,
63}
64
65/// JSON-RPC 2.0 error payload.
66#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
67pub struct JsonRpcError {
68    /// Error code (negative integers per JSON-RPC spec; MCP defines its
69    /// own range for protocol-level errors).
70    pub code: i32,
71    /// Human-readable error message.
72    pub message: String,
73    /// Optional additional payload.
74    #[serde(default, skip_serializing_if = "Option::is_none")]
75    pub data: Option<Value>,
76}
77
78/// A JSON-RPC 2.0 notification envelope (no id, no response expected).
79#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
80pub struct JsonRpcNotification {
81    /// Always `"2.0"`.
82    pub jsonrpc: String,
83    /// Method name (e.g. `"notifications/progress"`).
84    pub method: String,
85    /// Method-specific parameters.
86    #[serde(default, skip_serializing_if = "Option::is_none")]
87    pub params: Option<Value>,
88}
89
90/// Discriminated-union envelope over the wire.
91///
92/// A peer receives bytes, parses them once into `serde_json::Value`,
93/// and routes by the presence/absence of `id` and `result`/`error`.
94/// The `#[serde(untagged)]` attribute lets serde do this routing
95/// automatically; the variant order matters — Request is tried first
96/// (has `id` AND `method`), then Response (has `id` AND
97/// `result`/`error`), then Notification (no `id`).
98#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
99#[serde(untagged)]
100pub enum JsonRpcMessage {
101    /// A method-call request.
102    Request(JsonRpcRequest),
103    /// A response to a prior request.
104    Response(JsonRpcResponse),
105    /// A fire-and-forget notification.
106    Notification(JsonRpcNotification),
107}

109#[cfg(test)]
110mod tests {
111    use super::*;
112    use alloc::string::ToString;
113    use alloc::vec;
114    use serde_json::json;
115
116    #[test]
117    fn request_round_trips() {
118        let req = JsonRpcRequest {
119            jsonrpc: JSONRPC_VERSION.to_string(),
120            id: RequestId::Number(7),
121            method: "tools/call".to_string(),
122            params: Some(json!({"name":"get_forecast","arguments":{"lat":40.7,"lon":-74.0}})),
123        };
124        let bytes = serde_json::to_vec(&req).expect("serialize");
125        let decoded: JsonRpcRequest = serde_json::from_slice(&bytes).expect("deserialize");
126        assert_eq!(req, decoded);
127    }
128
129    #[test]
130    fn response_round_trips() {
131        let resp = JsonRpcResponse {
132            jsonrpc: JSONRPC_VERSION.to_string(),
133            id: RequestId::Number(7),
134            result: Some(json!({"content":[{"type":"text","text":"Sunny, 72°F"}]})),
135            error: None,
136        };
137        let bytes = serde_json::to_vec(&resp).expect("serialize");
138        let decoded: JsonRpcResponse = serde_json::from_slice(&bytes).expect("deserialize");
139        assert_eq!(resp, decoded);
140    }
141
142    #[test]
143    fn notification_round_trips() {
144        let n = JsonRpcNotification {
145            jsonrpc: JSONRPC_VERSION.to_string(),
146            method: "notifications/progress".to_string(),
147            params: Some(json!({"progressToken":"call-7","progress":50,"total":100})),
148        };
149        let bytes = serde_json::to_vec(&n).expect("serialize");
150        let decoded: JsonRpcNotification = serde_json::from_slice(&bytes).expect("deserialize");
151        assert_eq!(n, decoded);
152    }
153
154    #[test]
155    fn untagged_routing_request() {
156        let wire = json!({
157            "jsonrpc":"2.0","id":7,"method":"tools/call",
158            "params":{"name":"get_forecast"}
159        });
160        let msg: JsonRpcMessage = serde_json::from_value(wire).expect("route");
161        assert!(matches!(msg, JsonRpcMessage::Request(_)));
162    }
163
164    #[test]
165    fn untagged_routing_response_success() {
166        let wire = json!({"jsonrpc":"2.0","id":7,"result":{"content":[]}});
167        let msg: JsonRpcMessage = serde_json::from_value(wire).expect("route");
168        assert!(matches!(msg, JsonRpcMessage::Response(_)));
169    }
170
171    #[test]
172    fn untagged_routing_response_error() {
173        let wire =
174            json!({"jsonrpc":"2.0","id":7,"error":{"code":-32601,"message":"method not found"}});
175        let msg: JsonRpcMessage = serde_json::from_value(wire).expect("route");
176        assert!(matches!(msg, JsonRpcMessage::Response(_)));
177    }
