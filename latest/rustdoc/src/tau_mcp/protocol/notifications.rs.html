1//! Bidirectional notifications + cancellation.
2//!
3//! Per the β.3 design doc §4 (v0 scope) and §8.4 (cancellation
4//! propagation). Notifications are fire-and-forget (no `id`, no
5//! response). Cancellation is also a notification per MCP spec.
6
7use alloc::string::String;
8use serde::{Deserialize, Serialize};
9use serde_json::Value;
10
11use crate::protocol::jsonrpc::RequestId;
12
13/// `notifications/progress` — host or server reporting progress on an
14/// in-flight request.
15#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
16pub struct ProgressNotification {
17    /// Progress token (mirrors the request's `_meta.progressToken` if
18    /// the caller asked for progress; otherwise free-form).
19    #[serde(rename = "progressToken")]
20    pub progress_token: Value,
21    /// Current progress (units defined by the producer).
22    pub progress: f64,
23    /// Optional total to compute percentage.
24    #[serde(default, skip_serializing_if = "Option::is_none")]
25    pub total: Option<f64>,
26}
27
28/// `notifications/cancelled` — caller is aborting an in-flight request.
29#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
30pub struct CancelledNotification {
31    /// The request id being cancelled.
32    #[serde(rename = "requestId")]
33    pub request_id: RequestId,
34    /// Optional reason.
35    #[serde(default, skip_serializing_if = "Option::is_none")]
36    pub reason: Option<String>,
37}
38
39/// `notifications/initialized` — host signals it has finished processing
40/// the `initialize` response.
41#[derive(Debug, Clone, PartialEq, Default, Serialize, Deserialize)]
42pub struct InitializedNotification {}
43
44/// `notifications/message` (logging) — server emits a log line.
45#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
46pub struct LogNotification {
47    /// Log level (`"debug"` | `"info"` | `"warn"` | `"error"`).
48    pub level: String,
49    /// Logger name.
50    #[serde(default, skip_serializing_if = "Option::is_none")]
51    pub logger: Option<String>,
52    /// Free-form structured payload (server-defined).
53    pub data: Value,
54}

56#[cfg(test)]
57mod tests {
58    use super::*;
59    use alloc::string::ToString;
60    use serde_json::json;
61
62    #[test]
63    fn progress_round_trips() {
64        let n = ProgressNotification {
65            progress_token: json!("call-7"),
66            progress: 50.0,
67            total: Some(100.0),
68        };
69        let bytes = serde_json::to_vec(&n).expect("serialize");
70        let decoded: ProgressNotification = serde_json::from_slice(&bytes).expect("deserialize");
71        assert_eq!(n, decoded);
72    }
73
74    #[test]
75    fn cancelled_round_trips() {
76        let n = CancelledNotification {
77            request_id: RequestId::Number(7),
78            reason: Some("user abort".to_string()),
79        };
80        let bytes = serde_json::to_vec(&n).expect("serialize");
81        let decoded: CancelledNotification = serde_json::from_slice(&bytes).expect("deserialize");
82        assert_eq!(n, decoded);
83    }
84
85    #[test]
86    fn log_round_trips() {
87        let n = LogNotification {
88            level: "info".to_string(),
89            logger: Some("weather".to_string()),
90            data: json!({"msg":"forecast fetched","duration_ms":42}),
91        };
92        let bytes = serde_json::to_vec(&n).expect("serialize");
93        let decoded: LogNotification = serde_json::from_slice(&bytes).expect("deserialize");
94        assert_eq!(n, decoded);
95    }
