19/// Error returned by an inbound handler to refuse a server request.
20///
21/// Surfaces as an MCP `JsonRpcError` payload to the server with code
22/// = `-32000` (custom error range).
23#[derive(Debug, Clone, Error)]
24#[non_exhaustive]
25pub enum InboundError {
26    /// Server requested sampling but the host has no models allowlisted.
27    #[error("sampling refused: allowlist is empty")]
28    SamplingNotAllowed,
29    /// Server requested sampling with a model that's not in the
30    /// allowlist.
31    #[error("sampling refused: model {requested:?} not in allowlist")]
32    SamplingModelRefused {
33        /// The model the server asked for.
34        requested: String,
35    },
36    /// Server requested roots but the host's roots list is empty
37    /// (semantically the same as "no fs visibility granted").
38    #[error("roots returned []: no roots declared")]
39    RootsEmpty,
40    /// Backend invocation (LLM call) failed.
41    #[error("backend error: {0}")]
42    Backend(String),
43}

45/// Type alias for a boxed-future returning a result.
46pub type BoxFuture<'a, T> = Pin<Box<dyn Future<Output = T> + Send + 'a>>;
47
48/// Host-side handlers for inbound (server-initiated) MCP requests.
49///
50/// One impl per contracted MCP server. Concrete impl lives in PR-5
51/// (`tau-cli::cmd::run::ir_dispatcher::WiredHostHandlers` or similar)
52/// and composes the agent's `LlmBackend` + per-server allowlist.
53pub trait HostHandlers: Send + Sync {
54    /// Handle a `sampling/createMessage` request from the server.
55    fn sampling<'a>(
56        &'a self,
57        req: SamplingCreateMessageRequest,
58    ) -> BoxFuture<'a, Result<SamplingCreateMessageResponse, InboundError>>;
59
60    /// Handle a `roots/list` request from the server.
61    fn roots<'a>(&'a self) -> BoxFuture<'a, Result<Vec<Root>, InboundError>>;
62}
63
64/// Default-deny baseline impl: refuses every inbound request.
65///
66/// Suitable as a starting point for tests that don't need to exercise
67/// inbound handlers. PR-5's production impl follows the same trait
68/// shape but composes real backends.
69#[derive(Debug, Default, Clone, Copy)]
70pub struct DefaultDenyHandlers;
71
72impl HostHandlers for DefaultDenyHandlers {
73    fn sampling<'a>(
74        &'a self,
75        _req: SamplingCreateMessageRequest,
76    ) -> BoxFuture<'a, Result<SamplingCreateMessageResponse, InboundError>> {
77        Box::pin(async { Err(InboundError::SamplingNotAllowed) })
78    }
79
80    fn roots<'a>(&'a self) -> BoxFuture<'a, Result<Vec<Root>, InboundError>> {
81        // Default-deny for roots returns an EMPTY list, not an error —
82        // per the spec, `roots/list` returning `[]` is a valid response
83        // meaning "host grants no fs visibility." Servers must accept
84        // that gracefully.
85        Box::pin(async { Ok(Vec::new()) })
86    }
87}

96    fn sample_request() -> SamplingCreateMessageRequest {
97        SamplingCreateMessageRequest {
98            messages: vec![SamplingMessage {

104            model_preferences: Some(ModelPreferences::default()),
105            system_prompt: None,
106            include_context: None,
107            max_tokens: None,
108            additional: alloc::collections::BTreeMap::new(),

110    }

112    #[tokio::test(flavor = "current_thread")]
113    async fn default_deny_sampling_refuses() {
114        let h = DefaultDenyHandlers;
115        let r = h.sampling(sample_request()).await;
116        assert!(matches!(r, Err(InboundError::SamplingNotAllowed)));
117    }
118
119    #[tokio::test(flavor = "current_thread")]
120    async fn default_deny_roots_returns_empty() {
121        let h = DefaultDenyHandlers;
122        let r = h.roots().await.expect("ok");
123        assert!(r.is_empty());
