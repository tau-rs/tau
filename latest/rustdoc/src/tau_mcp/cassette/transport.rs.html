23/// Live cassette-as-transport.
24pub struct CassetteTransport {
25    replayer: Mutex<Replayer>,
26    inbound_rx: futures::lock::Mutex<UnboundedReceiver<JsonRpcMessage>>,
27    inbound_tx: UnboundedSender<JsonRpcMessage>,
28}
29
30impl CassetteTransport {
31    /// Construct from raw JSONL bytes.
32    pub fn from_jsonl_bytes(bytes: &[u8]) -> Result<Arc<Self>, ReplayError> {
33        let replayer = Replayer::from_jsonl_bytes(bytes)?;
34        let (tx, rx) = mpsc::unbounded();
35        Ok(Arc::new(Self {
36            replayer: Mutex::new(replayer),
37            inbound_rx: futures::lock::Mutex::new(rx),
38            inbound_tx: tx,
39        }))
40    }
41
42    /// Push any queued outbounds (notifications + server-initiated
43    /// requests) into the inbound channel for the host to consume.
44    fn drain_pending(&self) -> Result<(), McpError> {
45        let mut replayer = self
46            .replayer
47            .lock()
48            .map_err(|_| McpError::Transport("replayer mutex poisoned".to_string()))?;
49        while let Some(rec) = replayer.next_pending_outbound() {
50            let msg = cassette_record_to_jsonrpc(&rec)?;
51            if self.inbound_tx.unbounded_send(msg).is_err() {
52                return Err(McpError::Transport("inbound channel closed".to_string()));
53            }
54        }
55        Ok(())
56    }
57}
58
59impl Transport for CassetteTransport {
60    fn send_message<'a>(
61        &'a self,
62        msg: &'a JsonRpcMessage,
63    ) -> Pin<Box<dyn Future<Output = Result<(), McpError>> + Send + 'a>> {
64        Box::pin(async move {
65            match msg {
66                JsonRpcMessage::Request(req) => {
67                    // Match the request in the cassette; queue its
68                    // response + any preceding notifications/server-
69                    // initiated-requests for the host to read next.
70                    let response = {
71                        let mut replayer = self.replayer.lock().map_err(|_| {
72                            McpError::Transport("replayer mutex poisoned".to_string())
73                        })?;
74                        let method = req.method.clone();
75                        let args = req.params.clone().unwrap_or(serde_json::Value::Null);
76                        replayer
77                            .match_request(&method, &args)
78                            .map_err(|e| McpError::Transport(format!("cassette: {e}")))?
79                    };
80                    // Drain pending_outbound (notifications +
81                    // server-initiated requests recorded BETWEEN this
82                    // request and its response).
83                    self.drain_pending()?;
84                    // Push the matched response.
85                    let resp_msg = cassette_record_to_jsonrpc(&response)?;
86                    self.inbound_tx
87                        .unbounded_send(resp_msg)
88                        .map_err(|_| McpError::Transport("inbound channel closed".to_string()))?;
89                    Ok(())
90                }
91                JsonRpcMessage::Response(_) | JsonRpcMessage::Notification(_) => {
92                    // Host responding to a server-initiated request or
93                    // emitting a notification. The cassette's
94                    // pending_outbound queue was already drained when
95                    // the prior request matched; nothing more to do.
96                    Ok(())
97                }
98            }
99        })
100    }
101
102    fn next_message<'a>(
103        &'a self,
104    ) -> Pin<Box<dyn Future<Output = Result<Option<JsonRpcMessage>, McpError>> + Send + 'a>> {
105        Box::pin(async move {
106            let mut rx = self.inbound_rx.lock().await;
107            Ok(rx.next().await)
108        })
109    }
110}

112/// Convert a `CassetteMessage` (Direction::Out only) into a `JsonRpcMessage`
113/// for the host to consume.
114fn cassette_record_to_jsonrpc(rec: &CassetteMessage) -> Result<JsonRpcMessage, McpError> {
115    if rec.dir != Direction::Out {
116        return Err(McpError::Transport(format!(
117            "cassette record direction must be Out for replay; got {:?}",
118            rec.dir
119        )));
120    }
121    match rec.kind {
122        MessageKind::Response => {
123            // result/error live in rec.payload (already split or as
124            // {result: ..., error: ...}? — cassette format §11 says
125            // payload IS the result/error directly for responses).
126            // Parse defensively: if payload has top-level "error", treat
127            // as error; else as result.
128            let id = rec
129                .id
130                .clone()
131                .ok_or_else(|| McpError::Transport("cassette response without id".to_string()))?;
132            // The cassette stores the result directly as `payload` per
133            // spec §11. We construct a JsonRpcResponse.
134            // Per §11 example: payload for response is the inner result.
135            let resp = JsonRpcResponse {
136                jsonrpc: crate::protocol::jsonrpc::JSONRPC_VERSION.to_string(),
137                id,
138                result: Some(rec.payload.clone()),
139                error: None,
140            };
141            Ok(JsonRpcMessage::Response(resp))
142        }
143        MessageKind::Notification => {
144            let method = rec.method.clone().ok_or_else(|| {
145                McpError::Transport("cassette notification without method".to_string())
146            })?;
147            let n = JsonRpcNotification {
148                jsonrpc: crate::protocol::jsonrpc::JSONRPC_VERSION.to_string(),
149                method,
150                params: Some(rec.payload.clone()),
151            };
152            Ok(JsonRpcMessage::Notification(n))
153        }
154        MessageKind::Request => {
155            // Server-initiated request — same shape as a regular
156            // JsonRpcRequest. Host responds to it via send_message.
157            let id = rec.id.clone().ok_or_else(|| {
158                McpError::Transport("cassette server-initiated request without id".to_string())
159            })?;
160            let method = rec.method.clone().ok_or_else(|| {
161                McpError::Transport("cassette server-initiated request without method".to_string())
162            })?;
163            let req = JsonRpcRequest {
164                jsonrpc: crate::protocol::jsonrpc::JSONRPC_VERSION.to_string(),
165                id,
166                method,
167                params: Some(rec.payload.clone()),
168            };
169            Ok(JsonRpcMessage::Request(req))
170        }
171    }
