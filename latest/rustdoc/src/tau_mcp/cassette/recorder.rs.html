1//! In-memory cassette recorder.
2//!
3//! Captures `CassetteMessage` records as they're produced (PR-3 wires
4//! this into the host loop at the handler-dispatch boundary). The
5//! `Recorder` itself is transport-agnostic — it gets called with
6//! already-parsed `JsonRpcMessage` values; the transport layer is
7//! responsible for handing them to the recorder before they're framed
8//! / after they're parsed.
9//!
10//! File-I/O sink (`save_to_file`) requires `std`; the in-memory record
11//! API is `no_std`-compatible.
12
13#[cfg(feature = "with-std-adapters")]
14use alloc::string::String;
15use alloc::vec::Vec;
16
17use crate::cassette::message::{
18    CassetteHeader, CassetteMessage, Direction, MessageKind, CASSETTE_VERSION,
19};
20use crate::protocol::jsonrpc::{JsonRpcMessage, RequestId};
21
22/// In-memory cassette accumulator.
23#[derive(Debug, Default, Clone)]
24pub struct Recorder {
25    messages: Vec<CassetteMessage>,
26}
27
28impl Recorder {
29    /// Construct an empty recorder.
30    pub fn new() -> Self {
31        Self::default()
32    }
33
34    /// Record one JSON-RPC message with its direction.
35    pub fn record(&mut self, dir: Direction, msg: &JsonRpcMessage) {
36        let record = match msg {
37            JsonRpcMessage::Request(r) => CassetteMessage {
38                dir,
39                kind: MessageKind::Request,
40                id: Some(r.id.clone()),
41                method: Some(r.method.clone()),
42                payload: r.params.clone().unwrap_or(serde_json::Value::Null),
43            },
44            JsonRpcMessage::Response(r) => CassetteMessage {
45                dir,
46                kind: MessageKind::Response,
47                id: Some(r.id.clone()),
48                method: None,
49                payload: if let Some(e) = &r.error {
50                    serde_json::json!({"error": e})
51                } else {
52                    r.result.clone().unwrap_or(serde_json::Value::Null)
53                },
54            },
55            JsonRpcMessage::Notification(n) => CassetteMessage {
56                dir,
57                kind: MessageKind::Notification,
58                id: None,
59                method: Some(n.method.clone()),
60                payload: n.params.clone().unwrap_or(serde_json::Value::Null),
61            },
62        };
63        self.messages.push(record);
64    }
65
66    /// Return the recorded messages.
67    pub fn messages(&self) -> &[CassetteMessage] {
68        &self.messages
69    }
70
71    /// Serialize the cassette to JSONL bytes (header line + one line
72    /// per recorded message). Pure-allocator API; no I/O.
73    pub fn to_jsonl_bytes(&self) -> Result<Vec<u8>, serde_json::Error> {
74        let mut out: Vec<u8> = Vec::new();
75        let header = CassetteHeader {
76            version: CASSETTE_VERSION,
77        };
78        out.extend_from_slice(&serde_json::to_vec(&header)?);
79        out.push(b'\n');
80        for m in &self.messages {
81            out.extend_from_slice(&serde_json::to_vec(m)?);
82            out.push(b'\n');
83        }
84        Ok(out)
85    }
86
87    /// Save the cassette to a file. Requires std.
88    #[cfg(feature = "with-std-adapters")]
89    pub fn save_to_file<P: AsRef<std::path::Path>>(&self, path: P) -> Result<(), String> {
90        let bytes = self.to_jsonl_bytes().map_err(|e| alloc::format!("{e}"))?;
91        std::fs::write(path, bytes).map_err(|e| alloc::format!("{e}"))?;
92        Ok(())
93    }
94
95    /// Return how many records are stored, by request id, useful for
96    /// asserting in tests.
97    pub fn count_for(&self, id: &RequestId) -> usize {
98        self.messages
99            .iter()
100            .filter(|m| m.id.as_ref() == Some(id))
101            .count()
102    }
103}

105#[cfg(test)]
106mod tests {
107    use super::*;
108    use crate::protocol::jsonrpc::{JsonRpcRequest, JsonRpcResponse, JSONRPC_VERSION};
109    use alloc::string::ToString;
110    use serde_json::json;
111
112    #[test]
113    fn records_request_and_response() {
114        let mut r = Recorder::new();
115        r.record(
116            Direction::In,
117            &JsonRpcMessage::Request(JsonRpcRequest {
118                jsonrpc: JSONRPC_VERSION.to_string(),
119                id: RequestId::Number(1),
120                method: "initialize".to_string(),
121                params: Some(json!({"protocolVersion":"2025-03-26"})),
122            }),
123        );
124        r.record(
125            Direction::Out,
126            &JsonRpcMessage::Response(JsonRpcResponse {
127                jsonrpc: JSONRPC_VERSION.to_string(),
128                id: RequestId::Number(1),
129                result: Some(json!({"protocolVersion":"2025-03-26"})),
130                error: None,
131            }),
132        );
133        assert_eq!(r.messages().len(), 2);
134        assert_eq!(r.count_for(&RequestId::Number(1)), 2);
135    }
136
137    #[test]
138    fn jsonl_bytes_start_with_version_header() {
139        let r = Recorder::new();
140        let bytes = r.to_jsonl_bytes().expect("serialize");
141        assert!(bytes.starts_with(br#"{"version":1}"#));
142    }
143
144    #[test]
145    fn jsonl_bytes_per_message_line() {
146        let mut r = Recorder::new();
147        r.record(
148            Direction::In,
149            &JsonRpcMessage::Request(JsonRpcRequest {
150                jsonrpc: JSONRPC_VERSION.to_string(),
151                id: RequestId::Number(1),
152                method: "x".to_string(),
153                params: None,
154            }),
155        );
156        let bytes = r.to_jsonl_bytes().expect("serialize");
157        let s = core::str::from_utf8(&bytes).expect("utf8");
158        let line_count = s.lines().count();
159        assert_eq!(line_count, 2, "header + 1 record");
160    }
161}
