1//! `embed_c` — render the C HOST glue that loads + drives the EPIC 5.1
2//! wasm-guest component via the wasmtime C API.
3//!
4//! The four `tau:host/host` imports (wit/tau-host.wit) are a frozen contract,
5//! so the header declares them directly; the cap-derived WASI imports in the
6//! generated world are satisfied by `wasmtime_wasi` and need no host bodies.
7//! Bodies are `TODO(EPIC 7.1)` stubs. Pure string renderer — no tool runs at
8//! emit time. The embedded `tau.wit` lets a product regenerate typed bindings
9//! with `wit-bindgen` / the wasmtime C API if it wants more than this stub.
10
11use std::collections::BTreeMap;
12use std::path::PathBuf;
13
14/// Inputs for [`render_embed_c`].
15pub struct EmbedCInput<'a> {
16    /// Sanitized project stem: uppercased for the include guard, lowercased in
17    /// docs/prose. File names are fixed (`tau_embed.{h,c}`).
18    pub base_name: &'a str,
19    /// Lowercase-hex IR module hash (README provenance).
20    pub ir_hash: &'a str,
21    /// Cap-derived WIT world text, embedded verbatim.
22    pub wit: &'a str,
23}
24
25/// Render the C host scaffold as relative-path -> contents under `embed-c/`.
26pub fn render_embed_c(input: EmbedCInput) -> BTreeMap<PathBuf, String> {
27    let mut out = BTreeMap::new();
28    let guard = format!(
29        "TAU_EMBED_{}_H",
30        input
31            .base_name
32            .chars()
33            .map(|c| if c.is_ascii_alphanumeric() {
34                c.to_ascii_uppercase()
35            } else {
36                '_'
37            })
38            .collect::<String>()
39    );
40
41    out.insert(
42        PathBuf::from("embed-c/tau_embed.h"),
43        format!(
44            r#"/* Generated by `tau embed --host c` (EPIC 5.2) — do not edit. */
45#ifndef {guard}
46#define {guard}
47
48#include <stdint.h>
49
50/* The four `tau:host/host` imports (wit/tau-host.wit) the guest requires.
51 * Wire these to your product in tau_embed.c (EPIC 7.1). `request_json` /
52 * `event_json` are NUL-terminated serialized JSON; returned strings are
53 * heap-allocated and owned by the caller. */
54char    *tau_host_complete(const char *request_json);   /* -> CompletionResponse JSON */
55uint64_t tau_host_now_millis(void);
56uint64_t tau_host_next_u64(void);
57void     tau_host_emit_event(const char *event_json);
58
59/* Load the bundled wasm-guest component, wire the imports above, and drive its
60 * `run(prompt)` export. Returns 0 on success. RunEvents stream through
61 * tau_host_emit_event. `component_path` is the `.wasm` from `tau build
62 * --target wasm-guest`. */
63int tau_embed_run(const char *component_path, const char *prompt);
64
65#endif /* {guard} */
66"#,
67        ),
68    );
69
70    out.insert(
71        PathBuf::from("embed-c/tau_embed.c"),
72        r#"/* Generated by `tau embed --host c` (EPIC 5.2) — a scaffold.
73 *
74 * Drives a tau wasm-guest component through the wasmtime C API. The host-import
75 * bodies and the instantiate/call wiring are TODO(EPIC 7.1) stubs: this file
76 * compiles the shape of the glue, not a working runtime. Link against
77 * libwasmtime and fill the TODOs. */
78#include "tau_embed.h"
79
80#include <stdlib.h>
81#include <string.h>
82
83/* wasmtime C API — provided by libwasmtime (https://docs.wasmtime.dev/c-api). */
84#include <wasmtime.h>

87    (void)request_json;
88    /* TODO(EPIC 7.1): bridge to an LLM backend; return CompletionResponse JSON. */
89    return NULL;

93    /* TODO(EPIC 7.1): return a wall-clock millisecond count. */
94    return 0;

98    /* TODO(EPIC 7.1): return the next u64 from your RandomSource. */
99    return 0;

103    (void)event_json;

107int tau_embed_run(const char *component_path, const char *prompt) {
108    (void)component_path;
109    (void)prompt;
110    /* TODO(EPIC 7.1): with the wasmtime C API —
111     *   1. wasm_engine_new / wasmtime_store_new (+ wasmtime_wasi for the WASI
112     *      imports listed in tau.wit),
113     *   2. wasmtime_component_from_file(component_path),
114     *   3. wasmtime_component_linker_define the four tau:host/host imports to
115     *      the tau_host_* functions above,
116     *   4. instantiate and call the `run(prompt)` export.
117     * See tau.wit for the exact import/export set. */
118    return -1;
119}
120"#
121        .to_string(),
122    );
123
124    out.insert(PathBuf::from("embed-c/tau.wit"), input.wit.to_string());
125
126    out.insert(
127        PathBuf::from("embed-c/README.md"),
128        format!(
129            "# tau embed --host c ({base} host scaffold)\n\n\
130             Generated by `tau embed --host c` (EPIC 5.2). Baked IR hash: \
131             `{hash}`.\n\n\
132             C host glue that loads a `tau build --target wasm-guest` component \
133             through the [wasmtime C API](https://docs.wasmtime.dev/c-api) and \
134             drives its `run(prompt)` export. `tau_embed.h` declares the four \
135             frozen `tau:host/host` imports plus `tau_embed_run`; fill in the \
136             `TODO(EPIC 7.1)` bodies in `tau_embed.c` and link `libwasmtime`.\n\n\
137             `tau.wit` is the component's WIT world, embedded for reference. To \
138             regenerate typed bindings instead of hand-writing them, run \
139             `wit-bindgen c tau.wit` (guest side) or use the wasmtime C API's \
140             component bindings (host side).\n",
141            base = input.base_name,
142            hash = input.ir_hash,
143        ),
144    );
145
146    out
147}

149#[cfg(test)]
150mod tests {
151    use super::*;
152
153    #[test]
154    fn render_embed_c_emits_wasmtime_host_stub_with_wit_imports() {
155        let wit = "package tau:generated@0.1.0;\n\nworld runner {\n    \
156            import tau:host/host@0.1.0;\n    export run: func(prompt: string) \
157            -> result<string, string>;\n}\n";
158        let out = render_embed_c(EmbedCInput {
159            base_name: "trivial",
160            ir_hash: "abc123",
161            wit,
162        });
163
164        for f in ["tau_embed.h", "tau_embed.c", "tau.wit", "README.md"] {
165            assert!(
166                out.contains_key(&PathBuf::from(format!("embed-c/{f}"))),
167                "missing {f}"
168            );
169        }
170
171        let h = &out[&PathBuf::from("embed-c/tau_embed.h")];
172        // Include guard from base_name; the four frozen tau:host/host imports.
173        assert!(h.contains("#ifndef TAU_EMBED_TRIVIAL_H"), "{h}");
174        for sym in [
175            "tau_host_complete",
176            "tau_host_now_millis",
177            "tau_host_next_u64",
178            "tau_host_emit_event",
179        ] {
180            assert!(h.contains(sym), "header missing {sym}");
181        }
182        assert!(h.contains("tau_embed_run"), "{h}");
183
184        let c = &out[&PathBuf::from("embed-c/tau_embed.c")];
185        assert!(c.contains("#include \"tau_embed.h\""), "{c}");
186        assert!(c.contains("wasmtime"), "must use the wasmtime C API: {c}");
187        assert!(
188            c.contains("/* TODO(EPIC 7.1)"),
189            "port bodies must be TODO stubs: {c}"
190        );
191
192        assert_eq!(out[&PathBuf::from("embed-c/tau.wit")], wit);
193        assert!(out[&PathBuf::from("embed-c/README.md")].contains("wit-bindgen"));
194    }
