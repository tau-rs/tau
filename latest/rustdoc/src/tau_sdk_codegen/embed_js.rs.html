94const README: &str = r#"# @tau/embed-js
95
96Host-embedding glue for running a `tau build wasm` component from
97JavaScript/TypeScript (Phase 2 §5.2), in the main thread or a Web Worker.
98Normalizes the wire-level `RunEvent` stream into an idiomatic, kebab-tagged
99TypeScript union (see `src/RunEvent.ts`).
100
101## Build
102
103```sh
104npm install
105npm run build -- path/to/component.wasm
106```
107
108This runs `jco transpile --instantiation async --name component --out-dir
109src/generated <wasm>`, producing the JS bindings (`component.js` plus the
110sibling `component.core*.wasm` core modules) that `src/index.ts` imports from
111`./generated`. (The `--` passes the component path through to `jco`; npm no
112longer forwards `--flag=value` into scripts as `npm_config_*`.)
113
114## Usage
115
116```ts
117import { loadTau } from "@tau/embed-js";
118
119const tau = await loadTau({
120  // Bridges the WIT `complete` host import to an LLM backend. `requestJson`
121  // is a serialized `tau_ports::llm::CompletionRequest`; returns a
122  // serialized `CompletionResponse` JSON string. SYNCHRONOUS — see below.
123  complete: (requestJson) => lookUpCassetteResponse(requestJson),
124});
125for await (const event of tau.run({ prompt: "hello" })) {
126  if (event.type === "text-delta") process.stdout.write(event.delta);
127}
128```
129
130`complete` is a **synchronous** host import: `tau:host/host`'s `complete` is
131a sync WIT function and this package transpiles the component in jco's
132default sync mode, so the guest blocks on the return value. A backend that
133must await I/O — a live network LLM — cannot be bridged as-is; supply a
134`complete` backed by preloaded/cassette responses. Live async inference
135needs a `jco --async-mode jspi` build and is a documented follow-up. Omit
136`complete` and the guest throws a clear "not configured" error the moment it
137calls the LLM. `nowMillis`/`nextU64` default to
138`Date.now()`/`crypto.getRandomValues`; override both for deterministic
139(e.g. conformance/cassette) runs.
140
141For a Web Worker host, use `loadTauInWorker` instead — same `TauComponent`
142surface, but the component runs off the main thread. Host imports can't be
143bridged across `postMessage` (functions aren't structured-cloneable), so its
144`complete` always throws until a dedicated RPC bridge is added.
145

152- `src/index.ts` — `loadTau` / `loadTauInWorker` + `TauComponent`, and the

155- `src/worker.ts` — the Web Worker host driven by `loadTauInWorker`.
156- `src/generated/` — jco's build output (gitignored; not committed).

275    all_messages: unknown[];

306/// Normalize a raw, externally-tagged RunEvent JSON value into the
307/// idiomatic kebab-tagged union.
308export function normalize(raw: unknown): RunEvent {
309  if (typeof raw === "string") {
310    switch (raw) {
311      case "RunStarted":
312        return { type: "run-started" };
313      case "InferenceCallStarted":
314        return { type: "inference-call-started" };
315      default:
316        throw new Error(`unknown unit RunEvent variant: ${raw}`);
317    }
318  }
319
320  const obj = raw as Record<string, unknown>;
321
322  if ("ContextStepRan" in obj) {
323    const v = obj.ContextStepRan as { step: string; tokens_in: number; tokens_out: number };
324    return { type: "context-step-ran", step: v.step, tokensIn: v.tokens_in, tokensOut: v.tokens_out };
325  }
326  if ("InferenceCallCompleted" in obj) {
327    const v = obj.InferenceCallCompleted as {
328      stop_reason: string;
329      tokens_in: number;
330      tokens_out: number;
331    };
332    return {
333      type: "inference-call-completed",
334      stopReason: toStopReason(v.stop_reason),
335      tokensIn: v.tokens_in,
336      tokensOut: v.tokens_out,
337    };
338  }
339  if ("TextDelta" in obj) {
340    const v = obj.TextDelta as { delta: string };
341    return { type: "text-delta", delta: v.delta };
342  }
343  if ("ToolCallStarted" in obj) {
344    const v = obj.ToolCallStarted as { id: string; name: string; args: unknown };
345    return { type: "tool-call-started", id: v.id, name: v.name, args: v.args };
346  }
347  if ("ToolCallCompleted" in obj) {
348    const v = obj.ToolCallCompleted as {
349      id: string;
350      name: string;
351      result: { Ok?: unknown; Err?: string };
352    };
353    return { type: "tool-call-completed", id: v.id, name: v.name, result: toToolResult(v.result) };
354  }
355  if ("TurnCompleted" in obj) {
356    const v = obj.TurnCompleted as {
357      stop_reason: string;
358      turn: number;
359      usage?: { input_tokens: number; output_tokens: number; total_tokens?: number | null } | null;
360    };
361    return {
362      type: "turn-completed",
363      stopReason: toStopReason(v.stop_reason),
364      usage: v.usage ? toTokenUsage(v.usage) : undefined,
365      turn: v.turn,
366    };
367  }
368  if ("RunCompleted" in obj) {
369    const v = obj.RunCompleted as { outcome: Parameters<typeof toRunOutcome>[0] };
370    return { type: "run-completed", outcome: toRunOutcome(v.outcome) };
371  }
372  if ("FatalError" in obj) {
373    const v = obj.FatalError as {
374      kind: string;
375      detail: string;
376      context_json?: string | null;
377      tool_error_variant?: string | null;
378    };
379    return {
380      type: "fatal-error",
381      kind: v.kind,
382      detail: v.detail,
383      contextJson: v.context_json ?? undefined,
384      toolErrorVariant: v.tool_error_variant ?? undefined,
385    };
386  }
387
388  throw new Error(`unrecognized RunEvent payload: ${JSON.stringify(raw)}`);
389}

468interface AsyncQueue<T> extends AsyncIterable<T> {
469  push(value: T): void;
470  close(): void;
471}
472
473function makeQueue<T>(): AsyncQueue<T> {
474  const items: T[] = [];
475  const waiters: Array<(result: IteratorResult<T>) => void> = [];
476  let closed = false;
477
478  return {
479    push(value: T) {
480      const waiter = waiters.shift();
481      if (waiter) waiter({ value, done: false });
482      else items.push(value);
483    },
484    close() {
485      closed = true;
486      while (waiters.length > 0) {
487        waiters.shift()!({ value: undefined as unknown as T, done: true });
488      }
489    },
490    [Symbol.asyncIterator](): AsyncIterator<T> {
491      return {
492        next(): Promise<IteratorResult<T>> {
493          if (items.length > 0) {
494            return Promise.resolve({ value: items.shift() as T, done: false });
495          }
496          if (closed) {
497            return Promise.resolve({ value: undefined as unknown as T, done: true });
498          }
499          return new Promise((resolve) => waiters.push(resolve));
500        },
501      };
502    },
503  };
504}

520  const buf = new Uint32Array(2);
521  return () => {
522    crypto.getRandomValues(buf);

570/** Load this package's bundled tau wasm component (jco-transpiled into
571 * ./generated by `npm run build`) and run it on the calling thread. There is
572 * no wasm argument: `--instantiation async` bakes the component into the
573 * generated module + sibling core wasm, which `instantiate` imports. */
574export async function loadTau(
575  overrides: HostImportOverrides = {},
576): Promise<TauComponent> {
577  return {
578    run(input: RunInput): AsyncIterable<RunEvent> {
579      const queue = makeQueue<RunEvent>();
580      const emitEvent = (json: string) => queue.push(normalize(JSON.parse(json)));
581      const hostImports = resolveHostImports(emitEvent, overrides);
582      instantiate(hostImports)
583        .then((root) => root.run(input.prompt))
584        .catch((err: unknown) => queue.push(toFatalErrorEvent(err)))
585        .finally(() => queue.close());
586      return queue;
587    },
588  };
589}

591/** Load the bundled component inside a dedicated Web Worker.
592 *
593 * Note: host imports cannot be bridged from the main thread — functions are
594 * not structured-cloneable via `postMessage`. The worker uses its own
595 * defaults: `complete` always throws a clear "not configured" error (so this
596 * path is for components that never call the LLM, or a future MessageChannel
597 * RPC bridge), and `nowMillis`/`nextU64` are non-deterministic.
598 * Single-threaded `loadTau` is the only path that supports a real
599 * `complete`. */
600export async function loadTauInWorker(): Promise<TauComponent> {
601  return {
602    run(input: RunInput): AsyncIterable<RunEvent> {
603      const queue = makeQueue<RunEvent>();
604      const worker = new Worker(new URL("./worker.ts", import.meta.url), { type: "module" });
605      const onMessage = (ev: MessageEvent) => {
606        const msg = ev.data as { kind: "event"; json: string } | { kind: "done" };
607        if (msg.kind === "event") {
608          queue.push(normalize(JSON.parse(msg.json)));
609        } else {
610          queue.close();
611          worker.removeEventListener("message", onMessage);
612          worker.terminate();
613        }
614      };
615      worker.addEventListener("message", onMessage);
616      worker.postMessage({ input });
617      return queue;
618    },
619  };
620}

623const WORKER_TS: &str = r#"// Web Worker host driven by loadTauInWorker (./index.ts). Instantiates the
624// jco-transpiled component (./generated/component.js, `--instantiation
625// async`) off the main thread against all four `tau:host/host` imports
626// (wit/tau-host.wit), then forwards normalized events back via postMessage.
627//
628// Host imports cannot be bridged from the main thread here (functions are
629// not structured-cloneable via postMessage), so `complete` always throws a
630// clear "not configured" error and `nowMillis`/`nextU64` use worker-local
631// defaults — see loadTauInWorker's doc comment in index.ts.
632
633/// <reference lib="webworker" />
634/// <reference path="./generated.d.ts" />
635
636import type { RunInput } from "./index";
637
638type RunMessage = { input: RunInput };
639type WorkerMessage = { kind: "event"; json: string } | { kind: "done" };
640
641interface HostImports {
642  complete: (requestJson: string) => string;
643  nowMillis: () => bigint;
644  nextU64: () => bigint;
645  emitEvent: (json: string) => void;
646}
647
648interface GeneratedRoot {
649  run(prompt: string): string;
650}
651
652interface GeneratedModule {
653  instantiate(
654    getCoreModule: undefined,
655    imports: { "tau:host/host": HostImports },
656  ): GeneratedRoot | Promise<GeneratedRoot>;
657}
658
659async function instantiate(hostImports: HostImports): Promise<GeneratedRoot> {
660  const mod = (await import("./generated/component.js")) as unknown as GeneratedModule;
661  return await mod.instantiate(undefined, { "tau:host/host": hostImports });
662}
663
664function defaultNextU64(): () => bigint {
665  const buf = new Uint32Array(2);
666  return () => {
667    crypto.getRandomValues(buf);
668    return (BigInt(buf[0]) << 32n) | BigInt(buf[1]);
669  };
670}
671
672self.addEventListener("message", async (ev: MessageEvent<RunMessage>) => {
673  const { input } = ev.data;
674  const post = (msg: WorkerMessage) => (self as unknown as Worker).postMessage(msg);
675  const hostImports: HostImports = {
676    complete: () => {
677      throw new Error(
678        "@tau/embed-js worker: no `complete` host import configured — " +
679          "see loadTauInWorker's doc comment in index.ts.",
680      );
681    },
682    nowMillis: () => BigInt(Date.now()),
683    nextU64: defaultNextU64(),
684    emitEvent: (json: string) => post({ kind: "event", json }),
685  };
686  try {
687    const root = await instantiate(hostImports);
688    root.run(input.prompt);
689  } catch (err) {
690    post({
691      kind: "event",
692      json: JSON.stringify({
693        FatalError: {
694          kind: "EmbedJsHostError",
695          detail: err instanceof Error ? err.message : String(err),
696        },
697      }),
698    });
699  } finally {
700    post({ kind: "done" });
701  }
702});
