1//! `embed_rust` — render the native HOST crate that drives the EPIC 5.1
2//! rust-lib (Variant B) artifact.
3//!
4//! Sibling to `emit_rust_lib` (which emits the *library* crate baking the IR).
5//! This emits the *host* crate that links that library, supplies port impls,
6//! and calls `run_ir(TAU_IR, …)`. Bodies are `todo!()` stubs: real port impls
7//! are the product's job (EPIC 7.1). Pure string renderer — never compiled here.
8
9use std::collections::BTreeMap;
10use std::path::PathBuf;
11
12/// Inputs for [`render_embed_rust`].
13pub struct EmbedRustInput<'a> {
14    /// Sanitized project stem used as the host crate name.
15    pub crate_name: &'a str,
16    /// Name of the EPIC 5.1 rust-lib crate this host links (path dep `..`).
17    pub lib_crate_name: &'a str,
18    /// Lowercase-hex IR module hash (README provenance).
19    pub ir_hash: &'a str,
20    /// Cap-derived WIT world text, embedded verbatim for reference.
21    pub wit: &'a str,
22    /// Pinned `tau-runtime-core` / `tau-ir` dependency version.
23    pub tau_version: &'a str,
24}
25
26/// Render the rust HOST scaffold as crate-relative-path -> contents under
27/// `embed-rust/`. A product drops this beside the rust-lib crate, fills in the
28/// `StubDispatcher` port bodies (EPIC 7.1), and runs it to drive the workflow.
29pub fn render_embed_rust(input: EmbedRustInput) -> BTreeMap<PathBuf, String> {
30    let mut out = BTreeMap::new();
31
32    out.insert(
33        PathBuf::from("embed-rust/Cargo.toml"),
34        format!(
35            r#"[package]
36name = "{name}"
37version = "0.1.0"
38edition = "2021"
39publish = false
40
41# Generated by `tau embed --host rust` (EPIC 5.2). Native host scaffold: links
42# the sibling rust-lib crate, supplies port impls (EPIC 7.1), drives run_ir.
43[dependencies]
44{lib} = {{ path = ".." }}
45tau-runtime-core = {{ version = "{ver}" }}
46tau-ir = {{ version = "{ver}" }}
47serde_json = {{ version = "1" }}
48tokio = {{ version = "1", features = ["macros", "rt-multi-thread"] }}
49"#,
50            name = input.crate_name,
51            lib = input.lib_crate_name,
52            ver = input.tau_version,
53        ),
54    );
55
56    out.insert(
57        PathBuf::from("embed-rust/src/main.rs"),
58        format!(
59            r#"//! Generated by `tau embed --host rust` (EPIC 5.2) — a scaffold.
60//!
61//! Links the rust-lib crate (`{lib}`), decodes its baked `TAU_IR`, and drives
62//! `run_ir` with a dispatcher whose port bodies are `todo!()` stubs. Fill them
63//! in (EPIC 7.1) to bridge tools + inference to your product, then run.
64//!
65//! Capabilities the dispatcher must service are the WIT imports in `tau.wit`.
66
67use std::sync::Arc;
68
69use tau_ir::{{from_canonical_bytes, ToolId}};
70use serde_json::Value;
71use tau_runtime_core::builder::DynLlmBackend;
72use tau_runtime_core::error::RuntimeError;
73use tau_runtime_core::interpreter::tool_dispatch::{{ToolDispatcher, ToolInvocationResult}};
74use {lib}::{{run_ir, TAU_IR}};
75
76/// Product port surface. Every method is a `todo!()` stub — EPIC 7.1 supplies
77/// the real bodies (tool execution + LLM backend resolution). The default trait
78/// methods (clock/random/assets/…) are inherited; override as your host needs.
79struct StubDispatcher;
80
81impl ToolDispatcher for StubDispatcher {{
82    fn invoke<'a>(
83        &'a self,
84        _tool_id: &'a ToolId,
85        _args: &'a Value,
86    ) -> std::pin::Pin<
87        Box<dyn std::future::Future<Output = Result<ToolInvocationResult, RuntimeError>> + Send + 'a>,
88    > {{
89        todo!("EPIC 7.1: execute the tool identified by tool_id")
90    }}
91
92    fn llm_backend_for(&self, _backend: &str) -> Result<Arc<dyn DynLlmBackend>, RuntimeError> {{
93        todo!("EPIC 7.1: resolve the named LLM backend")
94    }}
95}}
96
97#[tokio::main]
98async fn main() -> Result<(), Box<dyn std::error::Error>> {{
99    let module = Arc::new(from_canonical_bytes(TAU_IR)?);
100    let entry = module.workflow.agents.keys().next().expect("IR module has at least one agent").clone();
101    let outcome = run_ir(module, &entry, Arc::new(StubDispatcher), Vec::new()).await?;
102    println!("{{outcome:?}}");
103    Ok(())
104}}
105"#,
106            lib = input.lib_crate_name,
107        ),
108    );
109
110    out.insert(PathBuf::from("embed-rust/tau.wit"), input.wit.to_string());
111
112    out.insert(
113        PathBuf::from("embed-rust/README.md"),
114        format!(
115            "# {name} (rust host scaffold)\n\n\
116             Generated by `tau embed --host rust` (EPIC 5.2). Baked IR hash: \
117             `{hash}`.\n\n\
118             Native host for the sibling rust-lib crate (`{lib}`). Link it, fill \
119             in `StubDispatcher`'s `todo!()` port bodies (tool execution + LLM \
120             backend — EPIC 7.1), and `cargo run`. The capabilities your \
121             dispatcher must service are the WIT imports in `tau.wit`.\n",
122            name = input.crate_name,
123            hash = input.ir_hash,
124            lib = input.lib_crate_name,
125        ),
126    );
127
128    out
129}

131#[cfg(test)]
132mod tests {
133    use super::*;
134    use std::path::PathBuf;
135
136    #[test]
137    fn render_embed_rust_emits_host_crate_that_drives_run_ir() {
138        let out = render_embed_rust(EmbedRustInput {
139            crate_name: "trivial_host",
140            lib_crate_name: "trivial",
141            ir_hash: "abc123",
142            wit:
143                "package tau:generated@0.1.0;\nworld runner {\n    import tau:host/host@0.1.0;\n}\n",
144            tau_version: "0.0.0",
145        });
146
147        for f in ["Cargo.toml", "src/main.rs", "tau.wit", "README.md"] {
148            assert!(
149                out.contains_key(&PathBuf::from(format!("embed-rust/{f}"))),
150                "missing embed-rust/{f}"
151            );
152        }
153
154        let main = &out[&PathBuf::from("embed-rust/src/main.rs")];
155        // Links the 5.1 rust-lib crate and drives run_ir with a stub dispatcher.
156        assert!(main.contains("use trivial::{run_ir, TAU_IR}"), "{main}");
157        assert!(
158            main.contains("impl ToolDispatcher for StubDispatcher"),
159            "{main}"
160        );
161        assert!(
162            main.contains("todo!("),
163            "port bodies must be todo!() stubs: {main}"
164        );
165        assert!(main.contains("run_ir("), "{main}");
166
167        let cargo = &out[&PathBuf::from("embed-rust/Cargo.toml")];
168        assert!(cargo.contains(r#"trivial = { path = ".." }"#), "{cargo}");
169        assert!(
170            cargo.contains(r#"tau-runtime-core = { version = "0.0.0""#),
171            "{cargo}"
172        );
173        assert!(cargo.contains("serde_json"), "emitted Cargo.toml must depend on serde_json (main.rs uses serde_json::Value): {cargo}");
174
175        assert_eq!(
176            out[&PathBuf::from("embed-rust/tau.wit")],
177            "package tau:generated@0.1.0;\nworld runner {\n    import tau:host/host@0.1.0;\n}\n"
178        );
179        assert!(out[&PathBuf::from("embed-rust/README.md")].contains("abc123"));
180    }
181}
