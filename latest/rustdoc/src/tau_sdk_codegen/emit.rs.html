1//! Orchestrates rendering + writing the generated SDK packages.
2
3use std::collections::BTreeMap;
4use std::path::{Path, PathBuf};
5
6use crate::emit_python;
7use crate::emit_ts;
8use crate::error::CodegenError;
9use crate::schema::SchemaModel;
10
11/// Render every generated file, keyed by repo-relative path.
12pub fn render_all(repo_root: &Path) -> Result<BTreeMap<PathBuf, String>, CodegenError> {
13    let schema = SchemaModel::load(repo_root)?;
14    let mut all = BTreeMap::new();
15    for (rel, contents) in emit_python::render_package(&schema) {
16        all.insert(PathBuf::from("sdk/python").join(rel), contents);
17    }
18    for (rel, contents) in emit_ts::render_package(&schema) {
19        all.insert(PathBuf::from("sdk/ts").join(rel), contents);
20    }
21    Ok(all)
22}
23
24/// Render all SDK files (schema read from `repo_root`) and write them under
25/// `out_root`, i.e. `out_root/sdk/python/...` and `out_root/sdk/ts/...`.
26pub fn generate_into(repo_root: &Path, out_root: &Path) -> Result<(), CodegenError> {
27    for (rel, contents) in render_all(repo_root)? {
28        let path = out_root.join(&rel);
29        if let Some(parent) = path.parent() {
30            std::fs::create_dir_all(parent)?;
31        }
32        std::fs::write(&path, contents)?;
33    }
34    Ok(())
35}
36
37/// Generate all SDK packages in-place under `repo_root` (writes the committed tree).
38pub fn generate(repo_root: &Path) -> Result<(), CodegenError> {
39    generate_into(repo_root, repo_root)
40}
