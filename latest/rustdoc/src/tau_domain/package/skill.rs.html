1//! Skill-specific manifest block + SKILL.md parsing.
2//!
3//! ROADMAP §16 (Skills as first-class packages, Constitution G10).
4//! v1 design ratified in
5//! `docs/superpowers/specs/2026-05-12-skills-1-manifest-design.md`.
6//!
7//! A tau skill package is a directory containing both an
8//! Anthropic-format `SKILL.md` (content) and a tau-format `tau.toml`
9//! (packaging). This module owns:
10//!
11//! - [`SkillManifest`] — the typed `[skill]` block parsed from
12//!   `tau.toml`.
13//! - [`SkillFrontmatter`] + [`SkillContent`] — the parsed YAML
14//!   frontmatter + Markdown body of `SKILL.md`.
15//! - [`parse_skill_md`] — the frontmatter splitter + YAML parser.
16//! - [`SKILL_DIR_VAR`] — the public string constant for the
17//!   `${SKILL_DIR}` interpolation variable. Substitution itself
18//!   is Skills-4's responsibility (runtime invocation); Skills-1
19//!   just establishes the variable as a recognized symbolic form.
20//!
21//! See `docs/decisions/0025-skills-foundation.md` for the ADR.
22

28/// Public string constant for the `${SKILL_DIR}` interpolation
29/// variable that resolves at runtime to the absolute path of the
30/// installed skill's directory.
31///
32/// Parallel to the conventional `${SCOPE}` and `${PROJECT}` variables
33/// (also symbolic in v1; substitution lives outside `tau-domain`).
34///
35/// Used in capability `paths` entries and `SKILL.md` body references.
36/// Validated for syntactic recognition at install time (Skills-2);
37/// substituted at spawn time (Skills-4).
38pub const SKILL_DIR_VAR: &str = "${SKILL_DIR}";

40/// Skill-specific manifest block (parsed from `[skill]` in `tau.toml`).
41///
42/// `content` defaults to `"SKILL.md"` (the canonical Anthropic skill
43/// content filename); `requires_tools` and `requires_skills` default
44/// to empty lists.
45///
46/// `SkillManifest` is `#[non_exhaustive]` — construction outside the crate
47/// requires the `serde` feature (TOML deserialization). The canonical path
48/// to a `SkillManifest` is through `UncheckedManifest::skill` after parsing.
49///
50/// # Example
51///
52/// ```no_run
53/// use tau_domain::{UncheckedManifest, SkillManifest};
54/// // Obtained by deserializing a tau.toml with a [skill] block:
55/// // let u: UncheckedManifest = toml::from_str(toml_str)?;
56/// // let skill: &SkillManifest = u.skill.as_ref().unwrap();
57/// # let _ = std::any::type_name::<SkillManifest>();
58/// ```
59#[non_exhaustive]
60#[derive(Debug, Clone, PartialEq)]
61#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
62pub struct SkillManifest {
63    /// Path to the SKILL.md content file, relative to the package
64    /// root. Defaults to `"SKILL.md"`.
65    #[cfg_attr(feature = "serde", serde(default = "default_skill_content"))]
66    pub content: String,
67
68    /// Tool dependencies (same shape as the top-level
69    /// `[[requires.tools]]`).
70    #[cfg_attr(feature = "serde", serde(default))]
71    pub requires_tools: Vec<PackageDep>,
72
73    /// Sub-skill dependencies. Resolved at install time
74    /// (Skills-2); the runtime side wires through `agent.<kind>.spawn`
75    /// in Skills-4.
76    #[cfg_attr(feature = "serde", serde(default))]
77    pub requires_skills: Vec<PackageDep>,
78}
79
80#[cfg(feature = "serde")]
81fn default_skill_content() -> String {
82    use alloc::string::ToString;
83    "SKILL.md".to_string()
84}

86/// Parsed YAML frontmatter from a `SKILL.md` file.
87///
88/// Both `name` and `description` are required by the Anthropic skill
89/// format. Other frontmatter fields are tolerated and discarded
90/// in v1 — Skills-5 may surface them if Agent Skills spec compliance
91/// requires it.
92///
93/// `SkillFrontmatter` is `#[non_exhaustive]` — struct-literal construction
94/// is blocked across crate boundaries. Instances come from
95/// [`parse_skill_md`] (requires the `serde` feature).
96///
97/// # Example
98///
99/// ```no_run
100/// use tau_domain::{SkillFrontmatter, SkillContent};
101/// // Obtained by calling parse_skill_md (requires serde feature):
102/// // let content: SkillContent = tau_domain::parse_skill_md(raw_md)?;
103/// // assert_eq!(content.frontmatter.name, "critic");
104/// # let _ = std::any::type_name::<SkillFrontmatter>();
105/// ```
106#[non_exhaustive]
107#[derive(Debug, Clone, PartialEq, Eq)]
108#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
109pub struct SkillFrontmatter {
110    /// The skill's canonical name. Must equal the `name` field of
111    /// the containing package's `tau.toml`; mismatch is rejected at
112    /// install time (Skills-2).
113    pub name: String,
114
115    /// Short human-readable description.
116    pub description: String,
117}
118
119/// Parsed `SKILL.md` content: frontmatter + body.
120///
121/// `body` is the verbatim text between the closing `---` frontmatter
122/// delimiter and end of file. Becomes the spawned child agent's
123/// `system_prompt` at runtime (Skills-4).
124///
125/// `SkillContent` is `#[non_exhaustive]` — struct-literal construction
126/// is blocked across crate boundaries. Instances come from
127/// [`parse_skill_md`] (requires the `serde` feature).
128///
129/// # Example
130///
131/// ```no_run
132/// use tau_domain::SkillContent;
133/// // Obtained by calling parse_skill_md (requires serde feature):
134/// // let content: SkillContent = tau_domain::parse_skill_md(raw_md)?;
135/// // assert_eq!(&content.body, "\nYou are a strict editor.\n");
136/// # let _ = std::any::type_name::<SkillContent>();
137/// ```
138#[non_exhaustive]
139#[derive(Debug, Clone, PartialEq, Eq)]
140#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
141pub struct SkillContent {
142    /// The frontmatter (parsed from the YAML block).
143    pub frontmatter: SkillFrontmatter,
144    /// The Markdown body (becomes the spawned child's
145    /// `system_prompt`).
146    pub body: String,
147}
148
149/// Errors raised by [`parse_skill_md`].
150///
151/// # Example
152///
153/// ```
154/// use tau_domain::SkillContentError;
155///
156/// // MissingFrontmatterOpener is returned when the file doesn't begin with `---`.
157/// let err = SkillContentError::MissingFrontmatterOpener;
158/// assert!(err.to_string().contains("---"));
159///
160/// // MissingName is returned when the YAML block has no `name` field.
161/// let err2 = SkillContentError::MissingName;
162/// assert!(err2.to_string().contains("name"));
163/// ```
164#[derive(Debug, thiserror::Error, PartialEq, Eq)]
165#[non_exhaustive]
166pub enum SkillContentError {
167    /// File did not begin with a `---` frontmatter opener (on its
168    /// own line, optionally with surrounding whitespace).
169    #[error("missing leading `---` frontmatter delimiter")]
170    MissingFrontmatterOpener,
171
172    /// File began with `---` but no closing `---` was found.
173    #[error("missing closing `---` frontmatter delimiter")]
174    MissingFrontmatterCloser,
175
176    /// YAML parse failure inside the frontmatter block.
177    #[error("frontmatter YAML parse error: {0}")]
178    YamlParse(String),
179
180    /// Frontmatter parsed but is missing the required `name` field.
181    #[error("frontmatter missing required field `name`")]
182    MissingName,
183
184    /// Frontmatter parsed but is missing the required `description`
185    /// field.
186    #[error("frontmatter missing required field `description`")]
187    MissingDescription,
188}

190/// Parse a `SKILL.md` file's text into [`SkillContent`].
191///
192/// Format:
193/// ```text
194/// ---
195/// name: foo
196/// description: ...
197/// ---
198///
199/// Markdown body...
200/// ```
201///
202/// The body is everything after the closing `---` line.
203#[cfg(feature = "skill-md")]
204pub fn parse_skill_md(input: &str) -> Result<SkillContent, SkillContentError> {
205    use alloc::string::ToString;
206
207    // Helper: strip an optional trailing `\r` from a line slice.
208    fn trim_cr(s: &str) -> &str {
209        s.strip_suffix('\r').unwrap_or(s)
210    }
211
212    let mut lines = input.split_inclusive('\n');
213
214    // First line must be `---` (with optional whitespace + CR).
215    let first = lines
216        .next()
217        .ok_or(SkillContentError::MissingFrontmatterOpener)?;
218    let first_stripped = first.strip_suffix('\n').unwrap_or(first);
219    let first_stripped = trim_cr(first_stripped);
220    if first_stripped.trim() != "---" {
221        return Err(SkillContentError::MissingFrontmatterOpener);
222    }
223
224    // Collect lines until the closing `---`.
225    let mut yaml_buf = String::new();
226    let mut closer_found = false;
227    let mut consumed = first.len();
228    for line in lines.by_ref() {
229        consumed += line.len();
230        let line_stripped = line.strip_suffix('\n').unwrap_or(line);
231        let line_stripped = trim_cr(line_stripped);
232        if line_stripped.trim() == "---" {
233            closer_found = true;
234            break;
235        }
236        yaml_buf.push_str(line);
237    }
238    if !closer_found {
239        return Err(SkillContentError::MissingFrontmatterCloser);
240    }
241
242    // Parse YAML body into a generic map first so we can produce
243    // field-specific errors before serde_yaml's generic missing-field
244    // message.
245    let map: serde_yaml::Mapping =
246        serde_yaml::from_str(&yaml_buf).map_err(|e| SkillContentError::YamlParse(e.to_string()))?;
247
248    let name = map
249        .get(serde_yaml::Value::String("name".into()))
250        .and_then(|v| v.as_str().map(String::from))
251        .ok_or(SkillContentError::MissingName)?;
252    let description = map
253        .get(serde_yaml::Value::String("description".into()))
254        .and_then(|v| v.as_str().map(String::from))
255        .ok_or(SkillContentError::MissingDescription)?;
256
257    let frontmatter = SkillFrontmatter { name, description };
258
259    // Body is everything after the closing `---` line.
260    let body = input[consumed..].to_string();
261
262    Ok(SkillContent { frontmatter, body })
263}

265#[cfg(all(test, feature = "skill-md"))]
266mod parse_tests {
267    use super::*;
268
269    #[test]
270    fn parses_valid_skill_md() {
271        let input =
272            "---\nname: critic\ndescription: Reviews drafts.\n---\n\nYou are a strict editor.\n";
273        let parsed = parse_skill_md(input).unwrap();
274        assert_eq!(parsed.frontmatter.name, "critic");
275        assert_eq!(parsed.frontmatter.description, "Reviews drafts.");
276        assert_eq!(parsed.body, "\nYou are a strict editor.\n");
277    }
278
279    #[test]
280    fn rejects_missing_opener() {
281        let input = "no frontmatter here\nname: critic\n";
282        assert_eq!(
283            parse_skill_md(input).unwrap_err(),
284            SkillContentError::MissingFrontmatterOpener
285        );
286    }
287
288    #[test]
289    fn rejects_missing_closer() {
290        let input = "---\nname: critic\ndescription: stuck\n";
291        assert_eq!(
292            parse_skill_md(input).unwrap_err(),
293            SkillContentError::MissingFrontmatterCloser
294        );
295    }
296
297    #[test]
298    fn rejects_malformed_yaml() {
299        let input = "---\nname: critic\ndescription: : :\n  - bad indent\n---\nbody\n";
300        let err = parse_skill_md(input).unwrap_err();
301        assert!(
302            matches!(err, SkillContentError::YamlParse(_)),
303            "expected YamlParse, got {err:?}"
304        );
305    }
306
307    #[test]
308    fn rejects_missing_name() {
309        let input = "---\ndescription: Reviews drafts.\n---\nbody\n";
310        assert_eq!(
311            parse_skill_md(input).unwrap_err(),
312            SkillContentError::MissingName
313        );
314    }
315
316    #[test]
317    fn rejects_missing_description() {
318        let input = "---\nname: critic\n---\nbody\n";
319        assert_eq!(
320            parse_skill_md(input).unwrap_err(),
321            SkillContentError::MissingDescription
322        );
323    }
324
325    #[test]
326    fn tolerates_extra_frontmatter_fields() {
327        // Future-compat: extra YAML keys are accepted and ignored.
328        let input = "---\nname: critic\ndescription: x\ntags: [editing]\nversion: 0.1\n---\nbody\n";
329        let parsed = parse_skill_md(input).unwrap();
330        assert_eq!(parsed.frontmatter.name, "critic");
331    }
332
333    #[test]
334    fn tolerates_crlf_line_endings() {
335        let input = "---\r\nname: critic\r\ndescription: x\r\n---\r\nbody\r\n";
336        let parsed = parse_skill_md(input).unwrap();
337        assert_eq!(parsed.frontmatter.name, "critic");
338    }
339}

341#[cfg(all(test, feature = "serde"))]
342mod skill_dir_var_tests {
343    use super::*;
344    use crate::package::manifest::UncheckedManifest;
345
346    #[test]
347    fn skill_dir_var_constant_is_the_canonical_string() {
348        assert_eq!(SKILL_DIR_VAR, "${SKILL_DIR}");
349    }
350
351    #[test]
352    fn skill_dir_var_in_capability_path_round_trips_verbatim() {
353        // Skills-1 is purely symbolic — the ${SKILL_DIR} token survives
354        // serde round-trip without expansion. Substitution is Skills-4.
355        let toml_src = r#"
356name = "critic"
357version = "0.1.0"
358description = "Reviews drafts."
359authors = []
360source = "https://example.com/critic.git"
361kind = "skill"
362dependencies = []
363
364[[capabilities]]
365kind = "fs.read"
366paths = ["${SKILL_DIR}/references/**", "${SKILL_DIR}/templates/**"]
367
368[skill]
369"#;
370        let u: UncheckedManifest = toml::from_str(toml_src).expect("parse");
371        let cap = &u.capabilities[0];
372        match cap {
373            crate::package::capability::Capability::Filesystem(
374                crate::package::capability::FsCapability::Read { paths },
375            ) => {
376                assert_eq!(paths.len(), 2);
377                assert!(paths[0].contains(SKILL_DIR_VAR));
378                assert!(paths[1].contains(SKILL_DIR_VAR));
379                // Verbatim — no expansion happened.
380                assert_eq!(paths[0], "${SKILL_DIR}/references/**");
381            }
382            other => panic!("expected fs.read, got {other:?}"),
383        }
384    }
