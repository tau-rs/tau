15/// Concrete method set: `None` (the top element, "all methods") expands to the
16/// full nine-verb universe so the sound per-method coverage check below can
17/// iterate a finite set.
18fn method_set(m: &Option<BTreeSet<HttpMethod>>) -> BTreeSet<HttpMethod> {
19    match m {
20        None => HttpMethod::ALL.iter().copied().collect(),
21        Some(s) => s.clone(),
22    }
23}

25/// Meet (intersection) of two method grants, preserving `None` (= all) as the
26/// top element — so `meet(a, a)` structurally equals `canon(a)` for a `None`
27/// grant (the lattice idempotence law).
28fn method_meet(
29    a: &Option<BTreeSet<HttpMethod>>,
30    b: &Option<BTreeSet<HttpMethod>>,
31) -> Option<BTreeSet<HttpMethod>> {
32    match (a, b) {
33        (None, x) | (x, None) => x.clone(),
34        (Some(sa), Some(sb)) => Some(sa.intersection(sb).copied().collect()),
35    }
36}

38/// A method grant denotes ⊥ iff it is `Some(∅)` (deny all). `None` (= all
39/// methods) is never empty.
40fn method_is_empty(m: &Option<BTreeSet<HttpMethod>>) -> bool {
41    matches!(m, Some(s) if s.is_empty())
42}

44/// A single child capability that exceeds the parent ceiling. The caller
45/// (1.4 / 1.5) prepends agent/link framing; this type names only *what*
46/// exceeded.
47#[non_exhaustive]
48#[derive(Debug, Clone, PartialEq, Eq)]
49pub struct CeilingViolation {
50    /// Capability kind that violated (`"fs.read"`, `"agent.spawn"`, `"custom"`, …).
51    pub kind: String,
52    /// The exact child entry that exceeded (a path / host / command / kind /
53    /// skill / mode token), or the kind name when the kind itself is unmatched.
54    pub offender: String,
55    /// Human-readable reason.
56    pub reason: String,
57}
58
59/// `child ⊆ parent` over full capability sets, matched by kind (`Custom` by
60/// `name`). Returns the first violation. A child kind with no matching parent
61/// kind is a violation (deny-by-default ceiling of ∅).
62pub fn capability_subset(
63    child: &[Capability],
64    parent: &[Capability],
65) -> Result<(), CeilingViolation> {
66    for c in child {
67        let kind = kind_str(c);
68        let matching: Vec<&Capability> = parent.iter().filter(|p| same_kind(c, p)).collect();
69        if matching.is_empty() {
70            return Err(CeilingViolation {
71                kind: kind.to_string(),
72                offender: kind.to_string(),
73                reason: "kind not in ceiling".to_string(),
74            });
75        }
76        if let Err((offender, reason)) = cap_subset_against(c, &matching) {
77            return Err(CeilingViolation {
78                kind: kind.to_string(),
79                offender,
80                reason,
81            });
82        }
83    }
84    Ok(())
85}

87/// Stable kind discriminator. `Custom` → `"custom"`; unknown future variants
88/// → `"unknown"` (which never matches a parent → deny-by-default).
89fn kind_str(cap: &Capability) -> &'static str {
90    match cap {
91        Capability::Filesystem(FsCapability::Read { .. }) => "fs.read",
92        Capability::Filesystem(FsCapability::Write { .. }) => "fs.write",
93        Capability::Filesystem(FsCapability::Exec { .. }) => "fs.exec",
94        Capability::Network(NetCapability::Http { .. }) => "net.http",
95        Capability::Process(ProcessCapability::Spawn { .. }) => "process.spawn",
96        Capability::Agent(AgentCapability::Spawn { .. }) => "agent.spawn",
97        Capability::Skill(SkillCapability::Spawn { .. }) => "skill.spawn",
98        Capability::TaskList { .. } => "task_list",
99        Capability::Plan { .. } => "plan",
100        Capability::Custom { .. } => "custom",
101        // Forward-compat escape hatch: opaque like `Custom`, refined by
102        // (kind + params) in `same_kind`/`cap_subset_against` (fail-closed).
103        Capability::Forward { .. } => "forward",
104    }
105}

107/// Same kind for matching purposes. `Custom` matches `Custom` only when the
108/// `name` matches. `"unknown"` never matches (fail-closed).
109fn same_kind(a: &Capability, b: &Capability) -> bool {
110    match (a, b) {
111        (Capability::Custom { name: na, .. }, Capability::Custom { name: nb, .. }) => na == nb,
112        (Capability::Custom { .. }, _) | (_, Capability::Custom { .. }) => false,
113        (Capability::Forward { kind: ka, .. }, Capability::Forward { kind: kb, .. }) => ka == kb,
114        (Capability::Forward { .. }, _) | (_, Capability::Forward { .. }) => false,
115        _ => {
116            let k = kind_str(a);
117            k != "unknown" && k == kind_str(b)
118        }
119    }
120}

122/// Whether two caps fold into one in `canon_caps`. Single-dimension kinds
123/// (read/exec paths, token sets, modes) merge by kind because unioning their
124/// one dimension is language-preserving. The two-dimensional kinds — `net.http`
125/// (hosts × methods) and `fs.write` (paths × max_bytes) — must NOT fold, since
126/// unioning each dimension independently is a bounding-box that widens the
127/// grant; distinct rectangles stay separate and are absorbed by 2-D containment
128/// in `canon_caps`. `Custom` folds only on identical (name+params).
129fn mergeable(a: &Capability, b: &Capability) -> bool {
130    match (a, b) {
131        (
132            Capability::Custom {
133                name: na,
134                params: pa,
135            },
136            Capability::Custom {
137                name: nb,
138                params: pb,
139            },
140        ) => na == nb && pa == pb,
141        (
142            Capability::Forward {
143                kind: ka,
144                params: pa,
145            },
146            Capability::Forward {
147                kind: kb,
148                params: pb,
149            },
150        ) => ka == kb && pa == pb,
151        // Multi-dimensional rectangles never fold (would bounding-box).
152        (
153            Capability::Network(NetCapability::Http { .. }),
154            Capability::Network(NetCapability::Http { .. }),
155        )
156        | (
157            Capability::Filesystem(FsCapability::Write { .. }),
158            Capability::Filesystem(FsCapability::Write { .. }),
159        ) => false,
160        _ => same_kind(a, b),
161    }
162}

170    }

173/// Compare one child cap against all parent caps of the same kind. Returns
174/// `Err((offender, reason))` on violation.
175fn cap_subset_against(child: &Capability, parents: &[&Capability]) -> Result<(), (String, String)> {
176    match child {
177        Capability::Filesystem(FsCapability::Read { paths, .. })
178        | Capability::Filesystem(FsCapability::Exec { paths, .. }) => {
179            let pp = gather_paths(parents);
180            crate::package::capability::lattice::glob::glob_subset_set(paths, &pp)
181                .map_err(|o| (o, "not a subset of any allowed path".into()))
182        }
183        Capability::Filesystem(FsCapability::Write {
184            paths, max_bytes, ..
185        }) => {
186            // Sound joint coverage: each child path must be writable up to the
187            // child's max_bytes by a SINGLE parent entry (path + byte-limit from
188            // the same grant). Restrict to parent entries whose limit covers the
189            // child's (None = unlimited), then require path coverage against
190            // exactly those entries' paths.
191            let eligible: Vec<String> = parents
192                .iter()
193                .filter_map(|p| match p {
194                    Capability::Filesystem(FsCapability::Write {
195                        paths: pp,
196                        max_bytes: pb,
197                    }) => {
198                        let covers = match (pb, max_bytes) {
199                            (None, _) => true,        // parent unlimited covers any child
200                            (Some(_), None) => false, // child unlimited, parent capped
201                            (Some(pbv), Some(cbv)) => pbv >= cbv,
202                        };
203                        if covers {
204                            Some(pp.clone())
205                        } else {
206                            None
207                        }
208                    }
209                    _ => None,
210                })
211                .flatten()
212                .collect();
213            crate::package::capability::lattice::glob::glob_subset_set(paths, &eligible)
214                .map_err(|o| (o, "not writable within max_bytes in ceiling".to_string()))
215        }
216        Capability::Network(NetCapability::Http { hosts, methods }) => {
217            // Sound joint coverage: for each method the child requests, its hosts
218            // must be granted by parent entries that ALSO grant that method — a
219            // single grant must cover both dimensions, not the bounding box of
220            // independently-unioned hosts × methods. `methods = None` (all) is
221            // expanded to the finite nine-verb universe (see `method_set`).
222            use crate::package::capability::lattice::host;
223            for m in method_set(methods) {
224                let union = host::host_union(parents.iter().filter_map(|p| match p {
225                    Capability::Network(NetCapability::Http {
226                        hosts: ph,
227                        methods: pm,
228                    }) if method_set(pm).contains(&m) => Some(ph),
229                    _ => None,
230                }));
231                if !union.subsumes(hosts) {
232                    let offender = host::host_offender(hosts, &union);
233                    let reason = format!(
234                        "host {offender} not granted for method {} in ceiling",
235                        m.as_str()
236                    );
237                    return Err((offender, reason));
238                }
239            }
240            Ok(())
241        }
242        Capability::Process(ProcessCapability::Spawn { commands, .. }) => {
243            let pc = gather_commands(parents);
244            string_set_subset(commands, &pc).map_err(|o| (o, "command not in ceiling".into()))
245        }
246        Capability::Agent(AgentCapability::Spawn { allowed_kinds, .. }) => {
247            let pk = gather_agent_kinds(parents);
248            string_set_subset(allowed_kinds, &pk)
249                .map_err(|o| (o, "agent kind not in ceiling".into()))
250        }
251        Capability::Skill(SkillCapability::Spawn { allowed_skills, .. }) => {
252            let ps = gather_skills(parents);
253            string_set_subset(allowed_skills, &ps).map_err(|o| (o, "skill not in ceiling".into()))
254        }
255        Capability::TaskList { mode } => mode_subset(mode, parents, true),
256        Capability::Plan { mode } => mode_subset(mode, parents, false),
257        Capability::Custom { name, params } => {
258            let ok = parents.iter().any(|p| {
259                matches!(p, Capability::Custom { name: pn, params: pp } if pn == name && pp == params)
260            });
261            if ok {
262                Ok(())
263            } else {
264                Err((name.clone(), "custom params do not match ceiling".into()))
265            }
266        }
267        // Forward-compat escape hatch: opaque, so subset only holds against an
268        // identical (kind + params) parent Forward — fail-closed otherwise.
269        Capability::Forward { kind, params } => {
270            let ok = parents.iter().any(|p| {
271                matches!(p, Capability::Forward { kind: pk, params: pp } if pk == kind && pp == params)
272            });
273            if ok {
274                Ok(())
275            } else {
276                Err((kind.clone(), "forward params do not match ceiling".into()))
277            }
278        }
279    }
280}

282fn mode_subset(
283    mode: &str,
284    parents: &[&Capability],
285    allow_manage: bool,
286) -> Result<(), (String, String)> {
287    let child_rank = mode_rank(mode, allow_manage)
288        .ok_or_else(|| (format!("mode={mode}"), "unknown mode".to_string()))?;
289    let parent_rank = parents
290        .iter()
291        .filter_map(|p| match p {
292            Capability::TaskList { mode } | Capability::Plan { mode } => {
293                mode_rank(mode, allow_manage)
294            }
295            _ => None,
296        })
297        .max();
298    match parent_rank {
299        Some(pr) if child_rank <= pr => Ok(()),
300        Some(_) => Err((format!("mode={mode}"), "mode exceeds ceiling".to_string())),
301        None => Err((format!("mode={mode}"), "ceiling mode unknown".to_string())),
302    }
303}

305/// Callers MUST pass a same-kind-filtered parent slice (guaranteed by `same_kind` in
306/// `capability_subset`); this gathers Read/Write/Exec paths indiscriminately.
307fn gather_paths(parents: &[&Capability]) -> Vec<String> {
308    parents
309        .iter()
310        .filter_map(|p| match p {
311            Capability::Filesystem(FsCapability::Read { paths, .. })
312            | Capability::Filesystem(FsCapability::Write { paths, .. })
313            | Capability::Filesystem(FsCapability::Exec { paths, .. }) => Some(paths.clone()),
314            _ => None,
315        })
316        .flatten()
317        .collect()
318}

359/// Exact-set inclusion: every `child` entry equals some `parent` entry.
360/// `Err(offender)` names the first child entry not present in `parent`.
361fn string_set_subset(child: &[String], parent: &[String]) -> Result<(), String> {
362    for c in child {
363        if !parent.iter().any(|p| p == c) {
364            return Err(c.clone());
365        }
366    }
367    Ok(())
368}

373/// Greatest lower bound of two capability sets. Total on G2.
374pub fn meet(a: &[Capability], b: &[Capability]) -> Vec<Capability> {
375    let mut out: Vec<Capability> = Vec::new();
376    for ca in a {
377        for cb in b {
378            if same_kind(ca, cb) {
379                if let Some(m) = meet_pair(ca, cb) {
380                    push_cap(&mut out, m);
381                }
382            }
383        }
384    }
385    canon_caps(&out)
386}

388fn meet_pair(a: &Capability, b: &Capability) -> Option<Capability> {
389    use Capability::*;
390    match (a, b) {
391        (
392            Filesystem(FsCapability::Read { paths: pa }),
393            Filesystem(FsCapability::Read { paths: pb }),
394        ) => {
395            let m = glob_meet(pa, pb);
396            (!m.is_empty()).then_some(Filesystem(FsCapability::Read { paths: m }))
397        }
398        (
399            Filesystem(FsCapability::Exec { paths: pa }),
400            Filesystem(FsCapability::Exec { paths: pb }),
401        ) => {
402            let m = glob_meet(pa, pb);
403            (!m.is_empty()).then_some(Filesystem(FsCapability::Exec { paths: m }))
404        }
405        (
406            Filesystem(FsCapability::Write {
407                paths: pa,
408                max_bytes: ma,
409            }),
410            Filesystem(FsCapability::Write {
411                paths: pb,
412                max_bytes: mb,
413            }),
414        ) => {
415            let m = glob_meet(pa, pb);
416            if m.is_empty() {
417                return None;
418            }
419            let max_bytes = min_max_bytes(*ma, *mb);
420            Some(Filesystem(FsCapability::Write {
421                paths: m,
422                max_bytes,
423            }))
424        }
425        (
426            Network(NetCapability::Http {
427                hosts: ha,
428                methods: mea,
429            }),
430            Network(NetCapability::Http {
431                hosts: hb,
432                methods: meb,
433            }),
434        ) => {
435            let hosts = host_meet(ha, hb);
436            let methods = method_meet(mea, meb);
437            (!host_is_empty(&hosts) && !method_is_empty(&methods))
438                .then_some(Network(NetCapability::Http { hosts, methods }))
439        }
440        (
441            Process(ProcessCapability::Spawn { commands: a }),
442            Process(ProcessCapability::Spawn { commands: b }),
443        ) => {
444            let c = str_intersect(a, b);
445            (!c.is_empty()).then_some(Process(ProcessCapability::Spawn { commands: c }))
446        }
447        (
448            Agent(AgentCapability::Spawn { allowed_kinds: a }),
449            Agent(AgentCapability::Spawn { allowed_kinds: b }),
450        ) => {
451            let k = str_intersect(a, b);
452            (!k.is_empty()).then_some(Agent(AgentCapability::Spawn { allowed_kinds: k }))
453        }
454        (
455            Skill(SkillCapability::Spawn { allowed_skills: a }),
456            Skill(SkillCapability::Spawn { allowed_skills: b }),
457        ) => {
458            let s = str_intersect(a, b);
459            (!s.is_empty()).then_some(Skill(SkillCapability::Spawn { allowed_skills: s }))
460        }
461        (TaskList { mode: a }, TaskList { mode: b }) => {
462            min_mode(a, b, true).map(|mode| TaskList { mode })
463        }
464        (Plan { mode: a }, Plan { mode: b }) => min_mode(a, b, false).map(|mode| Plan { mode }),
465        // Fully-qualified `Capability::Custom` (not the bare `use`d form) so the
466        // escape-hatch registry scanner doesn't mistake these match patterns for
467        // an undocumented variant declaration once rustfmt puts them on their own
468        // line. The documented declaration lives on the enum in capability.rs.
469        (
470            Capability::Custom {
471                name: na,
472                params: pa,
473            },
474            Capability::Custom {
475                name: nb,
476                params: pb,
477            },
478        ) if na == nb && pa == pb => Some(Capability::Custom {
479            name: na.clone(),
480            params: pa.clone(),
481        }),
482        _ => None,
483    }
484}

486fn str_intersect(a: &[String], b: &[String]) -> Vec<String> {
487    let mut out: Vec<String> = a.iter().filter(|x| b.contains(x)).cloned().collect();
488    out.sort();
489    out.dedup();
490    out
491}

493fn min_max_bytes(a: Option<u64>, b: Option<u64>) -> Option<u64> {
494    match (a, b) {
495        (None, x) | (x, None) => x,
496        (Some(x), Some(y)) => Some(x.min(y)),
497    }

500fn min_mode(a: &str, b: &str, allow_manage: bool) -> Option<String> {
501    let ra = mode_rank(a, allow_manage)?;
502    let rb = mode_rank(b, allow_manage)?;
503    let lo = ra.min(rb);
504    Some(
505        match lo {
506            0 => "read",
507            1 => "write",
508            _ => "manage",
509        }
510        .to_string(),
511    )

520/// Canonical form of a capability set. Single-dimension same-kind caps are
521/// merged (union is language-preserving); two-dimensional kinds (`net.http`,
522/// `fs.write`) are kept as separate rectangles and absorbed by 2-D containment
523/// rather than bounding-box-merged. Internal glob/host/token lists are
524/// canonicalized, empty (⊥) caps dropped, and the result is a deterministically
525/// ordered antichain — so `meet` never widens and the lattice law holds as
526/// structural equality.
527pub fn canon_caps(caps: &[Capability]) -> Vec<Capability> {
528    // 1. fold single-dimension same-kind caps; multi-dimensional rectangles and
529    //    distinct-params customs stay separate (see `mergeable`).
530    let mut merged: Vec<Capability> = Vec::new();
531    for c in caps {
532        if let Some(existing) = merged.iter_mut().find(|m| mergeable(m, c)) {
533            union_into(existing, c);
534        } else {
535            merged.push(c.clone());
536        }
537    }
538    // 2. canonicalize each cap's internal lists; drop empties (⊥).
539    let canon: Vec<Capability> = merged.iter().filter_map(canon_one).collect();
540    // 3. absorb: drop any cap that is a subset of another (2-D containment for
541    //    http/write rectangles; a no-op for the folded single-entry kinds). The
542    //    `j < i` guard keeps exactly one survivor for an equal pair.
543    let mut kept: Vec<Capability> = Vec::new();
544    'outer: for (i, ci) in canon.iter().enumerate() {
545        for (j, cj) in canon.iter().enumerate() {
546            if i != j && cap_contained(ci, cj) && !(cap_contained(cj, ci) && j < i) {
547                continue 'outer;
548            }
549        }
550        kept.push(ci.clone());
551    }
552    // 4. deterministic total order.
553    kept.sort_by(|x, y| {
554        kind_str(x)
555            .cmp(kind_str(y))
556            .then_with(|| render_cap(x).cmp(&render_cap(y)))
557    });
558    kept
559}

561/// `inner ⊆ outer` for two single capabilities, via the sound per-kind subset
562/// (2-D rectangle containment for http/write). Used by `canon_caps`'s absorb.
563fn cap_contained(inner: &Capability, outer: &Capability) -> bool {
564    capability_subset(core::slice::from_ref(inner), core::slice::from_ref(outer)).is_ok()
565}

567/// Union `src`'s grant list into `dst` for a **single-dimension** kind (same
568/// kind guaranteed by `mergeable`). The two-dimensional kinds (`net.http`,
569/// `fs.write`) are intentionally ABSENT: unioning each of their dimensions
570/// independently is a bounding-box that widens the grant, so `mergeable` never
571/// routes them here — they stay as separate rectangles and are absorbed by 2-D
572/// containment in `canon_caps`. Adding an http/write arm here would reintroduce
573/// the unsound widening.
574fn union_into(dst: &mut Capability, src: &Capability) {
575    use Capability::*;
576    match (dst, src) {
577        (
578            Filesystem(FsCapability::Read { paths: d }),
579            Filesystem(FsCapability::Read { paths: s }),
580        )
581        | (
582            Filesystem(FsCapability::Exec { paths: d }),
583            Filesystem(FsCapability::Exec { paths: s }),
584        ) => {
585            d.extend(s.iter().cloned());
586        }
587        (
588            Process(ProcessCapability::Spawn { commands: d }),
589            Process(ProcessCapability::Spawn { commands: s }),
590        ) => {
591            d.extend(s.iter().cloned());
592        }
593        (
594            Agent(AgentCapability::Spawn { allowed_kinds: d }),
595            Agent(AgentCapability::Spawn { allowed_kinds: s }),
596        ) => {
597            d.extend(s.iter().cloned());
598        }
599        (
600            Skill(SkillCapability::Spawn { allowed_skills: d }),
601            Skill(SkillCapability::Spawn { allowed_skills: s }),
602        ) => {
603            d.extend(s.iter().cloned());
604        }
605        (TaskList { mode: d }, TaskList { mode: s }) if mode_rank(s, true) > mode_rank(d, true) => {
606            *d = s.clone();
607        }
608        (Plan { mode: d }, Plan { mode: s }) if mode_rank(s, false) > mode_rank(d, false) => {
609            *d = s.clone();
610        }
611        // custom (same name+params per mergeable) and the lower-mode cases keep dst.
612        _ => {}
613    }
614}

616/// Canonicalize one merged cap's lists; `None` if its grant list is empty (⊥).
617fn canon_one(c: &Capability) -> Option<Capability> {
618    use Capability::*;
619    fn sorted(v: &[String]) -> Vec<String> {
620        let mut o = v.to_vec();
621        o.sort();
622        o.dedup();
623        o
624    }
625    let cc = match c {
626        Filesystem(FsCapability::Read { paths }) => Filesystem(FsCapability::Read {
627            paths: glob_canon(paths),
628        }),
629        Filesystem(FsCapability::Exec { paths }) => Filesystem(FsCapability::Exec {
630            paths: glob_canon(paths),
631        }),
632        Filesystem(FsCapability::Write { paths, max_bytes }) => Filesystem(FsCapability::Write {
633            paths: glob_canon(paths),
634            max_bytes: *max_bytes,
635        }),
636        Network(NetCapability::Http { hosts, methods }) => Network(NetCapability::Http {
637            hosts: host_canon(hosts),
638            // `Option<BTreeSet<HttpMethod>>` is already canonical: `None` (all)
639            // stays `None`; a `BTreeSet` is sorted+deduped.
640            methods: methods.clone(),
641        }),
642        Process(ProcessCapability::Spawn { commands }) => Process(ProcessCapability::Spawn {
643            commands: sorted(commands),
644        }),
645        Agent(AgentCapability::Spawn { allowed_kinds }) => Agent(AgentCapability::Spawn {
646            allowed_kinds: sorted(allowed_kinds),
647        }),
648        Skill(SkillCapability::Spawn { allowed_skills }) => Skill(SkillCapability::Spawn {
649            allowed_skills: sorted(allowed_skills),
650        }),
651        other => other.clone(),
652    };
653    let empty = match &cc {
654        Filesystem(
655            FsCapability::Read { paths }
656            | FsCapability::Exec { paths }
657            | FsCapability::Write { paths, .. },
658        ) => paths.is_empty(),
659        Network(NetCapability::Http { hosts, methods }) => {
660            host_is_empty(hosts) || method_is_empty(methods)
661        }
662        Process(ProcessCapability::Spawn { commands }) => commands.is_empty(),
663        Agent(AgentCapability::Spawn { allowed_kinds }) => allowed_kinds.is_empty(),
664        Skill(SkillCapability::Spawn { allowed_skills }) => allowed_skills.is_empty(),
665        _ => false,
666    };
667    if empty {
668        None
669    } else {
670        Some(cc)
671    }
672}

674fn render_cap(c: &Capability) -> String {
675    // stable string key for ordering; not a wire format
676    let mut s = String::from(kind_str(c));
677    match c {
678        Capability::Filesystem(FsCapability::Read { paths })
679        | Capability::Filesystem(FsCapability::Exec { paths }) => {
680            for p in paths {
681                s.push('|');
682                s.push_str(p);
683            }
684        }
685        Capability::Filesystem(FsCapability::Write { paths, max_bytes }) => {
686            for p in paths {
687                s.push('|');
688                s.push_str(p);
689            }
690            match max_bytes {
691                Some(n) => {
692                    s.push('~');
693                    s.push_str(&n.to_string());
694                }
695                None => s.push_str("~*"),
696            }
697        }
698        Capability::Network(NetCapability::Http { hosts, methods }) => {
699            match hosts {
700                HostSet::Any => s.push_str("|any"),
701                HostSet::Exact(hs) => {
702                    for h in hs {
703                        s.push('|');
704                        s.push_str(h.as_str());
705                    }
706                }
707            }
708            match methods {
709                None => s.push_str("#*"),
710                Some(ms) => {
711                    for m in ms {
712                        s.push('#');
713                        s.push_str(m.as_str());
714                    }
715                }
716            }
717        }
718        Capability::Custom { name, .. } => {
719            s.push('|');
720            s.push_str(name);
721        }
722        _ => {}
723    }
724    s
725}

928    // small G2 path generator over a fixed alphabet
929    fn path_strat() -> impl Strategy<Value = String> {
930        let seg = prop_oneof![Just("a"), Just("b"), Just("c"), Just("*")];
931        prop::collection::vec(seg, 1..4).prop_flat_map(|segs| {
932            prop_oneof![Just(false), Just(true)].prop_map(move |dbl| {
933                let mut p = String::new();
934                for s in &segs {
935                    p.push('/');
936                    p.push_str(s);
937                }
938                if dbl {
939                    p.push_str("/**");
940                }
941                p
942            })
943        })
944    }
945    // One cap of a randomly-chosen kind (Read paths or Process.Spawn commands),
946    // so property tests exercise cross-kind AND multi-same-kind (0..3 caps can
947    // yield two Reads → the merge path in canon_caps).
948    fn cap_strat() -> impl Strategy<Value = Capability> {
949        prop_oneof![
950            prop::collection::vec(path_strat(), 1..3)
951                .prop_map(|p| { read(&p.iter().map(|s| s.as_str()).collect::<Vec<_>>()) }),
952            prop::collection::vec(prop_oneof![Just("ls"), Just("cat"), Just("rm")], 1..3).prop_map(
953                |c| {
954                    Capability::Process(ProcessCapability::Spawn {
955                        commands: c.iter().map(|s| s.to_string()).collect(),
956                    })
957                }
958            ),
959            // net.http: two-dimensional (hosts × methods). Exercises BOTH top
960            // elements (`HostSet::Any`, `methods = None`) and finite `Exact` /
961            // `Some` sets, so the antichain / joint-coverage soundness is tested
962            // against the widest grants too.
963            (
964                prop_oneof![
965                    Just(HostSet::Any),
966                    prop::collection::vec(
967                        prop_oneof![Just("a.example.com"), Just("b.example.com")],
968                        1..3
969                    )
970                    .prop_map(|hs| HostSet::Exact(
971                        hs.iter().map(|h| HostName::parse(h).unwrap()).collect()
972                    )),
973                ],
974                prop_oneof![
975                    Just(None),
976                    prop::collection::vec(
977                        prop_oneof![Just(HttpMethod::Get), Just(HttpMethod::Post)],
978                        1..3
979                    )
980                    .prop_map(|ms| Some(ms.into_iter().collect())),
981                ],
982            )
983                .prop_map(|(hosts, methods)| {
984                    Capability::Network(NetCapability::Http { hosts, methods })
985                }),
986            // fs.write: two-dimensional (paths × max_bytes), exercises the
987            // write antichain / max_bytes-filtered subset.
988            (
989                prop::collection::vec(path_strat(), 1..3),
990                prop_oneof![Just(None), Just(Some(100u64)), Just(Some(1000u64))],
991            )
992                .prop_map(|(paths, max_bytes)| {
993                    Capability::Filesystem(FsCapability::Write {
994                        paths: paths.iter().map(|s| s.to_string()).collect(),
995                        max_bytes,
996                    })
997                }),
998        ]
999    }
1000    // 0..4 so a single set can carry TWO http (or two write) entries → the
1001    // multi-entry rectangle-antichain path in canon/meet/subset gets sampled.
1002    fn caps_strat() -> impl Strategy<Value = Vec<Capability>> {
1003        prop::collection::vec(cap_strat(), 0..4)
1004    }

1023            // Lattice equality is language-equivalence (mutual subset), NOT Rust
1024            // structural ==: a union of 2-D rectangles (net.http hosts×methods,
1025            // fs.write paths×max_bytes) has no unique minimal cover, so meet(a,b)
1026            // and canon_caps(a) can denote the same language via different covers

1030            let m = meet(&a, &b);
1031            let meet_lang_eq_a =
1032                capability_subset(&m, &a).is_ok() && capability_subset(&a, &m).is_ok();
