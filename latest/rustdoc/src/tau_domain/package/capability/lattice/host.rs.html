1//! Host dimension of the `net.http` capability lattice.
2//!
3//! Since ADR-0064 (HostSet), a host grant is `HostSet::Any` (the top element)
4//! or `HostSet::Exact(set)` of validated exact hostnames — there is no glob
5//! grammar (suffix wildcards are deferred). The host lattice operations are
6//! therefore plain set operations with a top:
7//!
8//! - subset: `parent.subsumes(child)` (defined on [`HostSet`]).
9//! - join (union): [`host_union`] — `Any` absorbs; else set union.
10//! - meet (glb): [`host_meet`] — `Any ∩ x = x`; else set intersection.
11//!
12//! `canon` is the identity (a `BTreeSet` is already sorted+deduped and there
13//! is no glob absorption), kept for symmetry with the path dimension's
14//! `glob_canon`.
15
16use crate::package::host::{HostName, HostSet};
17use alloc::collections::BTreeSet;
18
19/// Join (union) of a set of host grants. `HostSet::Any` is the top element
20/// (absorbs everything); otherwise the union of the exact host sets.
21pub fn host_union<'a>(sets: impl IntoIterator<Item = &'a HostSet>) -> HostSet {
22    let mut acc: BTreeSet<HostName> = BTreeSet::new();
23    for s in sets {
24        match s {
25            HostSet::Any => return HostSet::Any,
26            HostSet::Exact(hs) => acc.extend(hs.iter().cloned()),
27        }
28    }
29    HostSet::Exact(acc)
30}
31
32/// Meet (greatest lower bound / intersection). `Any ∩ x = x`; otherwise the
33/// intersection of the two exact host sets.
34pub fn host_meet(a: &HostSet, b: &HostSet) -> HostSet {
35    match (a, b) {
36        (HostSet::Any, x) | (x, HostSet::Any) => x.clone(),
37        (HostSet::Exact(sa), HostSet::Exact(sb)) => {
38            HostSet::Exact(sa.intersection(sb).cloned().collect())
39        }
40    }
41}
42
43/// `HostSet` is already canonical (the `Exact` `BTreeSet` is sorted+deduped
44/// and there is no glob absorption), so this is the identity — kept for
45/// symmetry with `glob_canon`.
46pub fn host_canon(hosts: &HostSet) -> HostSet {
47    hosts.clone()
48}
49
50/// A host grant denotes ⊥ (grants nothing) iff it is `Exact(∅)`. `Any` is
51/// never empty.
52pub fn host_is_empty(hosts: &HostSet) -> bool {
53    matches!(hosts, HostSet::Exact(s) if s.is_empty())
54}
55
56/// Name a host in `child` that the `union` grant does not cover, for error
57/// reporting. An uncovered `Any` child reports `"any"`.
58pub fn host_offender(child: &HostSet, union: &HostSet) -> alloc::string::String {
59    use alloc::string::ToString;
60    match child {
61        HostSet::Any => "any".to_string(),
62        HostSet::Exact(hs) => hs
63            .iter()
64            .find(|h| !union.subsumes(&HostSet::Exact(core::iter::once((*h).clone()).collect())))
65            .map(|h| h.as_str().to_string())
66            .unwrap_or_else(|| "host".to_string()),
67    }
68}

70#[cfg(test)]
71mod tests {
72    use super::*;
73
74    fn exact(hs: &[&str]) -> HostSet {
75        HostSet::Exact(hs.iter().map(|h| HostName::parse(h).unwrap()).collect())
76    }
77
78    #[test]
79    fn union_any_absorbs() {
80        assert_eq!(
81            host_union([&exact(&["a.com"]), &HostSet::Any]),
82            HostSet::Any
83        );
84    }
85
86    #[test]
87    fn union_of_exacts() {
88        assert_eq!(
89            host_union([&exact(&["a.com"]), &exact(&["b.com"])]),
90            exact(&["a.com", "b.com"])
91        );
92    }
93
94    #[test]
95    fn meet_any_is_identity() {
96        assert_eq!(
97            host_meet(&HostSet::Any, &exact(&["a.com"])),
98            exact(&["a.com"])
99        );
100        assert_eq!(host_meet(&HostSet::Any, &HostSet::Any), HostSet::Any);
101    }
102
103    #[test]
104    fn meet_intersects_exacts() {
105        assert_eq!(
106            host_meet(&exact(&["a.com", "b.com"]), &exact(&["b.com", "c.com"])),
107            exact(&["b.com"])
108        );
109    }
110
111    #[test]
112    fn empty_only_for_exact_empty() {
113        assert!(host_is_empty(&exact(&[])));
114        assert!(!host_is_empty(&exact(&["a.com"])));
115        assert!(!host_is_empty(&HostSet::Any));
116    }
117
118    #[test]
119    fn offender_names_uncovered_exact_host() {
120        assert_eq!(
121            host_offender(&exact(&["a.com", "b.com"]), &exact(&["a.com"])),
122            "b.com"
123        );
124        assert_eq!(host_offender(&HostSet::Any, &exact(&["a.com"])), "any");
125    }
126}
