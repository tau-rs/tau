1//! Per-kind agent definitions (ADR-0024): a named, spawnable agent *kind*
2//! carrying its own capability set.
3//!
4//! This is the static `kind → capabilities` map whose absence deferred
5//! build-time `agent ⊇ spawn` enforcement in EPIC 1 story 1.5 ("no static
6//! kind→agent map"). EPIC 4.4 uses it to check, at `tau check` time, that a
7//! spawned kind's caps ⊆ the spawning agent's / dynamic region's effective
8//! caps via the sound `capability_subset` lattice primitive.
9
10use alloc::string::String;
11use alloc::vec::Vec;
12
13use crate::Capability;
14
15/// A named spawnable agent kind and its capability set.
16///
17/// The `name` matches the string used in `Agent(Spawn { allowed_kinds })`
18/// and in a dynamic region's `spawns` list. `capabilities` is the kind's
19/// grant, authored as raw caps in `[agent.kinds.<name>]`.
20#[non_exhaustive]
21#[derive(Debug, Clone, PartialEq)]
22pub struct AgentKind {
23    /// The kind name (referenced by spawn allow-lists and dynamic regions).
24    pub name: String,
25    /// The kind's capability grant.
26    pub capabilities: Vec<Capability>,
27}
28
29impl AgentKind {
30    /// Construct a kind from a name and its capability grant.
31    pub fn new(name: String, capabilities: Vec<Capability>) -> Self {
32        Self { name, capabilities }
33    }
34}
