1//! Test fixtures for `tau-domain` types.
2//!
3//! Gated behind the `test-fixtures` feature (off by default). Downstream
4//! crates depend via:
5//!
6//! ```toml
7//! [dev-dependencies]
8//! tau-domain = { workspace = true, features = ["test-fixtures"] }
9//! ```
10//!
11//! All helpers are deterministic where possible; UUID-based ones
12//! (`any_message`, `any_agent_definition`-derived IDs) generate fresh
13//! v7 UUIDs each call.

120/// Build a `Capability::Filesystem(FsCapability::Read)` granting read access
121/// to the given path glob patterns.
122pub fn cap_fs_read(paths: &[&str]) -> Capability {
123    Capability::Filesystem(FsCapability::Read {
124        paths: paths.iter().map(|s| s.to_string()).collect(),
125    })
126}
127
128/// Build a `Capability::Filesystem(FsCapability::Write)` granting write access

131    Capability::Filesystem(FsCapability::Write {
132        paths: paths.iter().map(|s| s.to_string()).collect(),
133        max_bytes,

135}

145/// Build a `Capability::Network(NetCapability::Http)` granting HTTP access to
146/// the given hosts with the given HTTP methods. Passing `["any"]` yields the
147/// any-host escape hatch ([`HostSet::Any`]); any other slice yields a
148/// [`HostSet::Exact`] over the parsed hostnames.
149pub fn cap_net_http(hosts: &[&str], methods: &[&str]) -> Capability {
150    let hosts = if hosts == ["any"] {
151        HostSet::Any
152    } else {
153        HostSet::Exact(
154            hosts
155                .iter()
156                .map(|h| HostName::parse(h).expect("valid host"))
157                .collect(),
158        )
159    };
160    let methods = if methods.is_empty() {
161        None
162    } else {
163        Some(
164            methods
165                .iter()
166                .map(|m| HttpMethod::parse(m).expect("valid method"))
167                .collect(),
168        )
169    };
170    Capability::Network(NetCapability::Http { hosts, methods })
171}
