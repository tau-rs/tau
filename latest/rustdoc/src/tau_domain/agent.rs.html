13/// Agent lifecycle status. Carries diagnostic data only on `Failed`;
14/// transition rules live in tau-runtime.
15///
16/// State graph (informational; not enforced here):
17/// `Declared → Installed → Ready → Running ↔ Stopped`,
18/// with `Failed` reachable from any non-terminal state.
19///
20/// # Example
21///
22/// ```
23/// use tau_domain::{AgentStatus, FailureKind};
24///
25/// // The `Failed` variant is variant-level `#[non_exhaustive]`, so
26/// // external callers use [`AgentStatus::failed`] instead of
27/// // struct-literal construction.
28/// let s = AgentStatus::failed(
29///     FailureKind::BackendError,
30///     Some("connection refused: api.openai.com".into()),
31/// );
32/// match s {
33///     AgentStatus::Failed { kind: FailureKind::BackendError, .. } => {
34///         // retry with backoff
35///     }
36///     _ => panic!(),
37/// }
38/// ```
39#[non_exhaustive]
40#[derive(Debug, Clone, PartialEq, Eq)]
41#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
42#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
43pub enum AgentStatus {
44    /// Manifest seen, package not yet installed.
45    Declared,
46    /// Package installed on disk, ready to instantiate.
47    Installed,
48    /// Instance created, idle.
49    Ready,
50    /// Actively processing a message.
51    Running,
52    /// Intentionally halted.
53    Stopped,
54    /// The agent failed. `kind` enables typed retry/restart logic;
55    /// `detail` carries human-readable specifics.
56    #[non_exhaustive]
57    Failed {
58        /// Typed failure category for restart logic.
59        kind: FailureKind,
60        /// Human-readable detail (e.g. `"panic at src/foo.rs:42"`).
61        detail: Option<String>,
62    },
63}
64
65impl AgentStatus {
66    /// Construct an [`AgentStatus::Failed`] with the given `kind` and
67    /// optional `detail`.
68    ///
69    /// `Failed` is variant-level `#[non_exhaustive]` (E0639), which
70    /// blocks struct-literal construction from outside `tau-domain`.
71    /// Callers (notably tau-runtime when reporting agent-level
72    /// failures from `Runtime::run`) use this constructor.
73    ///
74    /// # Example
75    ///
76    /// ```
77    /// use tau_domain::{AgentStatus, FailureKind};
78    ///
79    /// let s = AgentStatus::failed(FailureKind::BackendError, Some("HTTP 503".into()));
80    /// match s {
81    ///     AgentStatus::Failed { kind: FailureKind::BackendError, .. } => {}
82    ///     _ => panic!(),
83    /// }
84    /// ```
85    pub fn failed(kind: FailureKind, detail: Option<String>) -> Self {
86        AgentStatus::Failed { kind, detail }
87    }
88}

90/// Categorical failure kinds. New typed kinds are added additively;
91/// `InternalError` is the catch-all escape hatch.
92///
93/// See: [escape-hatches.md#failurekind-internalerror](../docs/explanation/escape-hatches.md#failurekind-internalerror).
94#[non_exhaustive]
95#[derive(Debug, Clone, Copy, PartialEq, Eq)]
96#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
97#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
98pub enum FailureKind {
99    /// Agent process crashed unexpectedly (panic, signal, abort).
100    Crashed,
101    /// Configured LLM backend returned an error or was unreachable.
102    BackendError,
103    /// A capability check denied an operation the agent attempted.
104    PolicyDenied,
105    /// Agent exceeded a resource limit (memory, message rate, timeout).
106    OutOfResources,
107    /// Catch-all for failures that don't match the named kinds.
108    /// See: [escape-hatches.md#failurekind-internalerror](../docs/explanation/escape-hatches.md#failurekind-internalerror).
109    InternalError,
110}

134/// Static description of an agent. Holds what the runtime needs to
135/// instantiate one; richer config lives in skills / plugin packages
136/// per G2.
137///
138/// # Example
139///
140/// ```
141/// use tau_domain::{AgentDefinition, AgentId, PackageId, PackageName, Version};
142/// use std::str::FromStr;
143///
144/// // `PackageId` is `#[non_exhaustive]` — use `::new` instead of
145/// // struct-literal construction across crate boundaries.
146/// let pkg = PackageId::new(
147///     PackageName::from_str("research-pkg").unwrap(),
148///     Version::parse("0.1.0").unwrap(),
149/// );
150/// let def = AgentDefinition::new(
151///     AgentId::from_str("researcher").unwrap(),
152///     "Researcher".into(),
153///     pkg,
154///     PackageName::from_str("claude-anthropic").unwrap(),
155/// );
156/// assert_eq!(def.id.as_str(), "researcher");
157/// ```
158#[non_exhaustive]
159#[derive(Debug, Clone, PartialEq)]
160#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
161pub struct AgentDefinition {
162    /// Canonical identifier for this definition.
163    pub id: AgentId,
164    /// Human-readable display name (free-form, unvalidated).
165    pub display_name: String,
166    /// Which package this agent ships from.
167    pub package: PackageId,
168    /// Reference to an installed LLM-backend plugin package.
169    /// Required at v0.1 (see ADR-0002 escape clause).
170    pub llm_backend: PackageName,
171    /// Resolved vendor model id used to build the LLM request.
172    /// Defaults to empty string; set via [`AgentDefinition::with_model`].
173    pub model: String,
174    /// Optional system prompt.
175    pub system_prompt: Option<String>,
176    /// Free-form per-agent config (validated by plugins, not by tau-domain).
177    pub config: BTreeMap<String, Value>,
178}
179
180impl AgentDefinition {
181    /// Construct an `AgentDefinition` with empty `system_prompt` and
182    /// `config`. Use the `with_*` builders to fill them in.
183    pub fn new(
184        id: AgentId,
185        display_name: String,
186        package: PackageId,
187        llm_backend: PackageName,
188    ) -> Self {
189        Self {
190            id,
191            display_name,
192            package,
193            llm_backend,
194            model: String::new(),
195            system_prompt: None,
196            config: BTreeMap::new(),
197        }
198    }
199
200    /// Set the resolved vendor model id used to build the LLM request.
201    pub fn with_model(mut self, model: String) -> Self {
202        self.model = model;
203        self
204    }
205
206    /// Set `system_prompt`.
207    ///
208    /// # Example
209    ///
210    /// ```
211    /// use tau_domain::{AgentDefinition, AgentId, PackageId, PackageName, Version};
212    /// use std::str::FromStr;
213    ///
214    /// let def = AgentDefinition::new(
215    ///     AgentId::from_str("assistant").expect("valid id"),
216    ///     "Assistant".into(),
217    ///     PackageId::new(
218    ///         PackageName::from_str("assistant-pkg").expect("valid name"),
219    ///         Version::parse("0.1.0").expect("valid version"),
220    ///     ),
221    ///     PackageName::from_str("claude-anthropic").expect("valid name"),
222    /// )
223    /// .with_system_prompt("You are a helpful assistant.".into());
224    /// assert_eq!(
225    ///     def.system_prompt.as_deref(),
226    ///     Some("You are a helpful assistant."),
227    /// );
228    /// ```
229    pub fn with_system_prompt(mut self, prompt: String) -> Self {
230        self.system_prompt = Some(prompt);
231        self
232    }
233
234    /// Set `config`.
235    ///
236    /// # Example
237    ///
238    /// ```
239    /// use tau_domain::{AgentDefinition, AgentId, PackageId, PackageName, Value, Version};
240    /// use std::collections::BTreeMap;
241    /// use std::str::FromStr;
242    ///
243    /// let mut cfg = BTreeMap::new();
244    /// cfg.insert("model".into(), Value::String("claude-opus-4".into()));
245    ///
246    /// let def = AgentDefinition::new(
247    ///     AgentId::from_str("analyst").expect("valid id"),
248    ///     "Analyst".into(),
249    ///     PackageId::new(
250    ///         PackageName::from_str("analyst-pkg").expect("valid name"),
251    ///         Version::parse("0.1.0").expect("valid version"),
252    ///     ),
253    ///     PackageName::from_str("claude-anthropic").expect("valid name"),
254    /// )
255    /// .with_config(cfg);
256    /// assert_eq!(
257    ///     def.config.get("model").and_then(|v| v.as_string()),
258    ///     Some("claude-opus-4"),
259    /// );
260    /// ```
261    pub fn with_config(mut self, config: BTreeMap<String, Value>) -> Self {
262        self.config = config;
263        self
264    }
265}
