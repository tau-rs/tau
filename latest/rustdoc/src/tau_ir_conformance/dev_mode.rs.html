36// ---------------------------------------------------------------------------
37// SequencedLlm — pops scripted responses in order
38// ---------------------------------------------------------------------------
39
40/// A `LlmBackend` that pops scripted `CompletionResponse` values from a
41/// queue in the order they were added. Used to replay a `mock_llm.jsonl`
42/// script against the IR interpreter.
43///
44/// Shared with `bundle_mode` so DevMode and BundleMode drive the IR
45/// interpreter through the exact same in-process scripted backend.
46pub(crate) struct SequencedLlm {
47    name: String,
48    queue: Mutex<VecDeque<CompletionResponse>>,
49}
50
51impl SequencedLlm {
52    pub(crate) fn new(name: impl Into<String>, responses: Vec<CompletionResponse>) -> Self {
53        Self {
54            name: name.into(),
55            queue: Mutex::new(responses.into()),
56        }
57    }
58}
59
60impl LlmBackend for SequencedLlm {
61    fn name(&self) -> &str {
62        &self.name
63    }
64
65    async fn complete(&self, _req: CompletionRequest) -> Result<CompletionResponse, LlmError> {
66        self.queue
67            .lock()
68            .expect("SequencedLlm mutex poisoned")
69            .pop_front()
70            .ok_or_else(|| LlmError::Internal {
71                message: "SequencedLlm: no more scripted responses".into(),
72            })
73    }
74
75    async fn stream(&self, req: CompletionRequest) -> Result<CompletionStream, LlmError> {
76        let resp = LlmBackend::complete(self, req).await?;
77        Ok(batch_to_stream(resp))
78    }
79}

82// RecordingDispatcher — records tool invocations + owns LLM backend

84
85/// Recorded side-effect entry for one tool invocation.
86pub(crate) struct ToolCallRecord {
87    pub(crate) tool_name: String,
88    pub(crate) args_canonical: Vec<u8>,
89}
90
91/// A `ToolDispatcher` that records every tool invocation and returns a
92/// canned successful response (`{"ok": true}`). The recording is then
93/// harvested to build the `ConformanceReport`.
94///
95/// Shared with `bundle_mode` so DevMode and BundleMode produce
96/// byte-identical recordings for the same fixture.
97pub(crate) struct RecordingDispatcher {
98    backend: Arc<dyn DynLlmBackend>,
99    records: Arc<Mutex<Vec<ToolCallRecord>>>,
100    /// Map from ToolId → tool name (used to look up the human name from the IR).
101    tool_names: std::collections::BTreeMap<String, String>,
102}
103
104impl RecordingDispatcher {
105    pub(crate) fn new(
106        backend: Arc<dyn DynLlmBackend>,
107        tool_names: std::collections::BTreeMap<String, String>,
108    ) -> Self {
109        Self {
110            backend,
111            records: Arc::new(Mutex::new(Vec::new())),
112            tool_names,
113        }
114    }
115
116    pub(crate) fn records(&self) -> Arc<Mutex<Vec<ToolCallRecord>>> {
117        self.records.clone()
118    }
119}
120
121impl ToolDispatcher for RecordingDispatcher {
122    fn invoke<'a>(
123        &'a self,
124        tool_id: &'a tau_ir::ToolId,
125        args: &'a JsonValue,
126    ) -> Pin<Box<dyn Future<Output = Result<ToolInvocationResult, RuntimeError>> + Send + 'a>> {
127        // Resolve the tool name from the id (fall back to the raw id string).
128        let tool_name = self
129            .tool_names
130            .get(&tool_id.0)
131            .cloned()
132            .unwrap_or_else(|| tool_id.0.clone());
133
134        // Canonical args bytes: deterministic JSON serialization.
135        let args_canonical = serde_json::to_vec(args).unwrap_or_default();
136
137        let records = self.records.clone();
138
139        Box::pin(async move {
140            records
141                .lock()
142                .expect("records mutex poisoned")
143                .push(ToolCallRecord {
144                    tool_name,
145                    args_canonical,
146                });
147
148            Ok(ToolInvocationResult {
149                body: Some(serde_json::json!({"ok": true})),
150                error: None,
151            })
152        })
153    }
154
155    fn llm_backend_for(&self, _backend: &str) -> Result<Arc<dyn DynLlmBackend>, RuntimeError> {
156        // Single-backend recording dispatcher: every agent/judge resolves to
157        // the one mock backend regardless of the name baked into its model_ref.
158        Ok(self.backend.clone())
159    }
160
161    fn deterministic_registry(
162        &self,
163    ) -> Option<
164        std::sync::Arc<dyn tau_runtime_core::interpreter::deterministic::DeterministicRegistry>,
165    > {
166        Some(crate::fixture_deterministic_registry())
167    }
168}

170// ---------------------------------------------------------------------------
171// mock_llm.jsonl parser
172// ---------------------------------------------------------------------------
173
174/// One line of `mock_llm.jsonl`. Represents one scripted LLM response.
175///
176/// ```json
177/// {"turn": 0, "response": {"tool_uses": [...], "stop_reason": "tool_use"}}
178/// {"turn": 1, "response": {"text": "ok", "stop_reason": "end_turn"}}
179/// ```
180#[derive(serde::Deserialize)]
181struct MockLlmLine {
182    response: MockLlmResponse,
183}
184
185#[derive(serde::Deserialize)]
186struct MockLlmResponse {
187    #[serde(default)]
188    text: String,
189    #[serde(default)]
190    tool_uses: Vec<MockToolUse>,
191    stop_reason: String,
192}
193
194#[derive(serde::Deserialize)]
195struct MockToolUse {
196    id: String,
197    name: String,
198    #[serde(default)]
199    input: JsonValue,
200}
201
202/// Parse `mock_llm.jsonl` into a vec of `CompletionResponse` values in
203/// turn order (sorted by the `turn` field, then extracted in order).
204///
205/// Shared with `bundle_mode` so both modes consume the same script bytes
206/// and produce the same scripted-response sequence.
207pub(crate) fn parse_mock_llm(jsonl: &str) -> Vec<CompletionResponse> {
208    let lines: Vec<MockLlmLine> = jsonl
209        .lines()
210        .filter(|l| !l.trim().is_empty())
211        .filter_map(|l| serde_json::from_str(l).ok())
212        .collect();
213
214    // Turn order is preserved by file order; the `turn` field in each
215    // line is advisory only. Process lines in the order they appear.
216    lines
217        .into_iter()
218        .map(|line| {
219            let resp = line.response;
220            let tool_uses: Vec<ToolUse> = resp
221                .tool_uses
222                .into_iter()
223                .map(|tu| {
224                    let input: tau_domain::Value =
225                        serde_json::from_value(tu.input).unwrap_or(tau_domain::Value::Null);
226                    tau_ports::fixtures::make_tool_use(tu.id, tu.name, input)
227                })
228                .collect();
229
230            let stop_reason = match resp.stop_reason.as_str() {
231                "tool_use" => StopReason::ToolUse,
232                _ => StopReason::EndTurn,
233            };
234
235            tau_ports::fixtures::make_completion_response(resp.text, tool_uses, stop_reason, None)
236        })
237        .collect()
238}

244/// Dev-mode runner.
245///
246/// For each fixture:
247/// 1. Reads `workflow.toml`; parses into a `ProjectConfig`.
248/// 2. Lowers to an `IrModule` against the host target triple.
249/// 3. Reads `mock_llm.jsonl`; builds a `SequencedLlm` backend.
250/// 4. Drives `run_ir` with a `RecordingDispatcher`.
251/// 5. Harvests the recordings into a `ConformanceReport`.
252///
253/// Panics if any setup step fails — fixture misconfiguration should
254/// be caught immediately, not swallowed into a `ConformanceReport`.
255pub struct DevMode;
256
257#[async_trait(?Send)]
258impl ExecutionMode for DevMode {
259    async fn run(&self, fixture_dir: &Path) -> ConformanceReport {
260        // All synchronous setup (lowering, parse) runs in a separate sync
261        // block so that the non-Send/non-Sync Caches closures are dropped
262        // before the first `.await`, satisfying the `Send` bound on the
263        // async trait future. Returns Ok(...) on successful lowering or
264        // Err(refusal_string) when the build was refused at lowering —
265        // BundleMode mirrors this exactly so the two modes report
266        // build-refused fixtures symmetrically (see ConformanceReport).
267        let lowered: Result<(IrModule, Vec<CompletionResponse>, String), String> = {
268            // 1. Load workflow.toml.
269            let workflow_toml = std::fs::read_to_string(fixture_dir.join("workflow.toml"))
270                .expect("fixture must contain workflow.toml");
271            // Parse + validate: project-config validation errors (e.g.
272            // `DeliverableNoProducer`) surface here as `Err`, not at the
273            // `lower_project` stage — capture them symmetrically with
274            // BundleMode so build-refused fixtures compare correctly.
275            let config = match ProjectConfig::parse_str(&workflow_toml) {
276                Ok(c) => c,
277                Err(e) => {
278                    // Move the mock_llm.jsonl read out of scope (it is only
279                    // consumed when the run actually executes), then early-
280                    // exit with the refusal string — no lowering needed.
281                    return ConformanceReport::build_refused(format!("{e}"));
282                }
283            };
284
285            // 2. Lower to IrModule. Caches are stack-only closures that don't
286            //    cross the await point — they are dropped at the end of this block.
287            //    Uses the SHA-256-of-name native-tool cache symmetric with
288            //    `tau-cli::cmd::build::lower_ir` so DevMode and BundleMode
289            //    hash identical IR bytes for the same source workflow.
290            let target = target_registry::list_available()
291                .next()
292                .expect("at least one target triple available")
293                .triple;
294            let module_result = {
295                let caches = Caches {
296                    native_tool: &|name: &str| Some(crate::sha256_name(name)),
297                    mcp_contract: &|_| None,
298                    skill: &|_| None,
299                    // Conformance fixtures use inline prompts only.
300                    prompt_file: &|_| Ok(Vec::new()),
301                };
302                // Conformance fixtures carry no file prompts, so the asset map
303                // is always empty; keep the module only (D6-B).
304                lower_project(&config, &target, &caches).map(|out| out.module)
305            }; // caches dropped here
306
307            match module_result {
308                Ok(module) => {
309                    // 3. Load mock_llm.jsonl.
310                    let mock_llm_jsonl =
311                        std::fs::read_to_string(fixture_dir.join("mock_llm.jsonl"))
312                            .expect("fixture must contain mock_llm.jsonl");
313                    let responses = parse_mock_llm(&mock_llm_jsonl);
314
315                    // Determine entry agent: first in BTreeMap (alphabetical) order.
316                    let entry = module
317                        .workflow
318                        .agents
319                        .keys()
320                        .next()
321                        .expect("fixture must declare at least one agent")
322                        .clone();
323                    Ok((module, responses, entry.0))
324                }
325                Err(e) => Err(format!("{e}")),
326            }
327        };
328
329        match lowered {
330            Ok((module, responses, entry_id)) => {
331                let module = Arc::new(module);
332                // Pipeline workflows are engine-sequenced: there is no
333                // single entry agent loop. Branch to the pipeline driver
334                // when the lowered module declares one; otherwise keep the
335                // single-entry `run_ir` path unchanged (fixtures 01-07).
336                if module.workflow.pipeline.is_some() {
337                    drive_pipeline(module, CONFORMANCE_PIPELINE_INPUT.to_string(), responses).await
338                } else {
339                    let entry = tau_ir::AgentId(entry_id);
340                    drive_module(module, &entry, responses).await
341                }
342            }
343            Err(refusal) => ConformanceReport::build_refused(refusal),
344        }
345    }
346}

348/// Fixed run input fed to a pipeline's first step (`${input}`). Held
349/// constant so DevMode and BundleMode render the same step templates and
350/// produce byte-identical reports under `assert_conform`.
351pub(crate) const CONFORMANCE_PIPELINE_INPUT: &str = "conformance-input";
352
353/// Drive the IR interpreter for a (decoded) module with a scripted LLM
354/// backend, recording side effects into a `ConformanceReport`.
355///
356/// Shared by DevMode (in-process lowering) and BundleMode (round-tripped
357/// through a built bundle). Both modes feed in identical
358/// `(module, entry, responses)` inputs for the same fixture, so the
359/// emitted reports compare byte-for-byte under `assert_conform`.
360pub(crate) async fn drive_module(
361    module: Arc<IrModule>,
362    entry: &tau_ir::AgentId,
363    responses: Vec<CompletionResponse>,
364) -> ConformanceReport {
365    // Build a name→name map for RecordingDispatcher (ToolId == tool name in v0).
366    let tool_names: std::collections::BTreeMap<String, String> = module
367        .workflow
368        .tools
369        .keys()
370        .map(|id| (id.0.clone(), id.0.clone()))
371        .collect();
372
373    // Build SequencedLlm + RecordingDispatcher.
374    let backend: Arc<dyn DynLlmBackend> = Arc::new(SequencedLlm::new("mock-llm", responses));
375    let dispatcher = Arc::new(RecordingDispatcher::new(backend, tool_names));
376    let records_handle = dispatcher.records();
377
378    // Run the interpreter. The Arc is passed directly — no clone needed.
379    let outcome: RunOutcome = run_ir(module, entry, dispatcher, Vec::new())
380        .await
381        .expect("run_ir must not return an Err for a valid conformance fixture");
382
383    // Build ConformanceReport from recorded side effects.
384    let mut report = ConformanceReport::new(outcome.clone());
385
386    // Record tool calls.
387    let records = records_handle.lock().expect("records mutex poisoned");
388    for rec in records.iter() {
389        report.record_tool_call(rec.tool_name.clone(), rec.args_canonical.clone());
390    }
391    drop(records);
392
393    // Record messages from outcome (all_messages multiset).
394    let messages = match &outcome {
395        RunOutcome::Completed { all_messages, .. } => all_messages.clone(),
396        RunOutcome::Failed { all_messages, .. } => all_messages.clone(),
397        _ => Vec::new(),
398    };
399    for msg in &messages {
400        // Canonicalize the message body before recording so two runs
401        // with different per-run UUIDs / timestamps produce identical
402        // multiset keys. See `canonical_message_bytes` for the exact
403        // shape — payload + address discriminants only.
404        let canonical = crate::canonical_message_bytes(msg);
405        report.record_message(canonical);
406    }
407
408    report
409}

411/// Fixed run id fed to `run_pipeline_suspendable`'s `SuspendConfig` — the
412/// `--resume` handle a real CLI invocation would generate per-run. Held
413/// constant (like [`CONFORMANCE_PIPELINE_INPUT`]) so DevMode and BundleMode
414/// persist/load suspensions under the same key and produce byte-identical
415/// `SuspendedSummary`s under `assert_conform`.
416const CONFORMANCE_RUN_ID: &str = "conformance";

418/// Drive an `IrModule`'s engine-sequenced pipeline with a scripted LLM
419/// backend, recording side effects into a `ConformanceReport`.
420///
421/// Used instead of [`drive_module`] when the lowered module declares a
422/// `workflow.pipeline` (the single-entry agent loop does not exist for
423/// pipeline workflows — the engine sequences the steps). Shared by
424/// DevMode (in-process lowering) and BundleMode (round-tripped through a
425/// built bundle); both feed identical `(module, responses)` for the same
426/// fixture so the emitted reports compare under `assert_conform`.
427///
428/// Drives `run_pipeline_suspendable` (not the non-suspend `run_pipeline`
429/// convenience wrapper) wired to a fresh in-memory `MockSuspensionStore`
430/// under the fixed [`CONFORMANCE_RUN_ID`], with `resume: None` — every
431/// conformance run starts fresh; resume-after-suspend is exercised by
432/// `tau-cli`'s integration tests (Task 6), not this cross-mode harness.
433/// This makes a `Suspend` step a first-class comparable terminal state
434/// instead of the old `run_pipeline` behavior (which mapped a `Suspend`
435/// step to `Err(SuspendUnsupported)` → `RunOutcome::Failed`, so two modes
436/// would only "conform" by both erroring — never actually comparing the
437/// pause point).
438///
439/// - **Tool calls** are still captured: each pipeline agent runs through
440///   the SAME shared `RecordingDispatcher`, so any tool the agents invoke
441///   is recorded exactly as in `drive_module`, on every branch below.
442/// - **Messages** are NOT observable here — the pipeline executor does not
443///   surface them — so `message_added` stays empty. Cross-mode
444///   conformance still holds because BOTH modes drive the same
445///   `run_pipeline_suspendable` and observe the same (empty) message
446///   multiset.
447/// - **`PipelineOutcome::Completed`** synthesizes `RunOutcome::Completed`
448///   exactly as the old `run_pipeline` path did.
449/// - **`PipelineOutcome::Suspended`** loads the just-persisted
450///   `PipelineSuspension` back out of the mock store (its `step_cursor`
451///   and `outputs` snapshot are not on `PipelineOutcome` itself — only
452///   `resume_signal` is) and returns [`ConformanceReport::suspended`].
453/// - **`Err(RuntimeError)`** (a real pipeline-step failure, not a suspend)
454///   maps to `RunOutcome::Failed` exactly as the old `run_pipeline` path
455///   did.
456pub(crate) async fn drive_pipeline(
457    module: Arc<IrModule>,
458    input: String,
459    responses: Vec<CompletionResponse>,
460) -> ConformanceReport {
461    use tau_ports::fixtures::MockSuspensionStore;
462    use tau_ports::orchestration::SuspensionStore;
463    use tau_runtime_core::interpreter::pipeline::{
464        run_pipeline_suspendable, PipelineOutcome, SuspendConfig,
465    };
466
467    // Build a name→name map for RecordingDispatcher (ToolId == tool name in v0).
468    let tool_names: std::collections::BTreeMap<String, String> = module
469        .workflow
470        .tools
471        .keys()
472        .map(|id| (id.0.clone(), id.0.clone()))
473        .collect();
474
475    let backend: Arc<dyn DynLlmBackend> = Arc::new(SequencedLlm::new("mock-llm", responses));
476    let dispatcher = Arc::new(RecordingDispatcher::new(backend, tool_names));
477    let records_handle = dispatcher.records();
478
479    // The id of the LAST pipeline step in execution order — its stored
480    // output is the run's final result. Captured before `module` is moved
481    // into `run_pipeline_suspendable`. Mirrors production
482    // `render_pipeline_result` (tau-cli::cmd::run), which keys off
483    // `pipeline.steps.last()` rather than the alphabetically-last id
484    // surfaced by `template_map().into_values()`: the BTreeMap projection
485    // only coincidentally yields the last-executed step when step ids sort
486    // in execution order (fixture 08: `gather` < `writer`). A
487    // reverse-alphabetical fixture would silently harvest the wrong step.
488    let last_step_id: Option<String> = module
489        .workflow
490        .pipeline
491        .as_ref()
492        .and_then(|p| p.steps.last())
493        .map(|step| step.id.0.clone());
494
495    let suspension_store = Arc::new(MockSuspensionStore::new());
496    let suspend_config = SuspendConfig {
497        run_id: CONFORMANCE_RUN_ID.to_string(),
498        store: suspension_store.clone(),
499    };
500
501    match run_pipeline_suspendable(module, input, dispatcher, suspend_config, None).await {
502        Ok(PipelineOutcome::Completed(store)) => {
503            // Synthesize a Completed outcome carrying the last step's
504            // output text as the final message (the pipeline executor
505            // does not return a RunOutcome of its own). The message
506            // multiset stays empty — the pipeline executor surfaces no
507            // message history — but tool calls recorded by the shared
508            // dispatcher are harvested below.
509            //
510            // A `Value::String` renders as its inner text; any structured
511            // value renders as compact JSON — symmetric with
512            // `OutputStore::template_map` and production
513            // `render_pipeline_result`.
514            let final_text = last_step_id
515                .as_deref()
516                .and_then(|id| store.get(id))
517                .map(|value| match value {
518                    serde_json::Value::String(s) => s.clone(),
519                    other => other.to_string(),
520                })
521                .unwrap_or_default();
522            let final_message = pipeline_final_message(&final_text);
523            let outcome = RunOutcome::Completed {
524                final_message,
525                all_messages: Vec::new(),
526                total_turns: 0,
527                token_usage: Default::default(),
528            };
529            let mut report = ConformanceReport::new(outcome);
530            let records = records_handle.lock().expect("records mutex poisoned");
531            for rec in records.iter() {
532                report.record_tool_call(rec.tool_name.clone(), rec.args_canonical.clone());
533            }
534            report
535        }
536        Ok(PipelineOutcome::Suspended { resume_signal, .. }) => {
537            // `PipelineOutcome::Suspended` itself carries only
538            // `run_id`/`resume_signal`/`step_id` — the `step_cursor` and
539            // `outputs` snapshot were durably persisted to the
540            // `SuspensionStore` by `run_pipeline_suspendable` before it
541            // returned, so load them back out to build the comparable
542            // `SuspendedSummary`.
543            let persisted = suspension_store
544                .load_suspension(&CONFORMANCE_RUN_ID.to_string())
545                .expect("MockSuspensionStore::load_suspension never errors")
546                .expect(
547                    "PipelineOutcome::Suspended implies a suspension was persisted under \
548                     CONFORMANCE_RUN_ID",
549                );
550            let summary = crate::SuspendedSummary {
551                resume_signal,
552                step_cursor: persisted.step_cursor,
553                outputs_canonical: serde_json::to_vec(&persisted.outputs)
554                    .expect("BTreeMap<String, serde_json::Value> always serializes"),
555            };
556            let mut report = ConformanceReport::suspended(summary);
557            let records = records_handle.lock().expect("records mutex poisoned");
558            for rec in records.iter() {
559                report.record_tool_call(rec.tool_name.clone(), rec.args_canonical.clone());
560            }
561            report
562        }
563        // A pipeline-step failure surfaces as Err here. Map it to a
564        // Failed outcome so both modes report the failure symmetrically
565        // (assert_conform compares the RunOutcome *discriminant*; the `e`
566        // detail is not part of the compared key, matching how the
567        // single-agent path discards per-run nondeterminism).
568        Err(_e) => ConformanceReport::new(RunOutcome::Failed {
569            status: tau_domain::AgentStatus::Stopped,
570            all_messages: Vec::new(),
571            total_turns: 0,
572            token_usage: Default::default(),
573        }),
574    }
575}

577/// Build the synthetic `final_message` for a completed pipeline run from
578/// the last step's output text. Uses the same `Message::new` /
579/// `Address::User → Address::System` shape so the value is canonicalizable
580/// by `canonical_message_bytes` if a caller ever records it; the pipeline
581/// path leaves `message_added` empty, so this only populates the outcome's
582/// `final_message` slot.
583fn pipeline_final_message(text: &str) -> tau_domain::Message {
584    use tau_domain::{Address, Message, MessagePayload};
585    Message::new(
586        Address::System,
587        Address::User,
588        MessagePayload::Text {
589            content: text.to_string(),
590        },
591    )
592}
