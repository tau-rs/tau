1//! Bundle-mode runner: build a bundle from the fixture, decode its
2//! embedded IR payload, then drive the v0 interpreter against the
3//! deserialized `IrModule`.
4//!
5//! Symmetric with [`crate::dev_mode::DevMode`] by construction: both
6//! modes ultimately call [`crate::dev_mode::drive_module`] with the
7//! same `(module, entry, responses)` triple for the same fixture. The
8//! only difference is the IR-module sourcing path:
9//!
10//! - DevMode lowers `workflow.toml` in-process via
11//!   `tau_ir_lower::lower_project`.
12//! - BundleMode lowers the same `workflow.toml` via
13//!   `tau_pkg::bundle::build` (which calls the exact same `lower_project`
14//!   internally via [`build_ir_payload`]), then decodes the bundle's
15//!   `ir_payload.canonical_ir_bytes_hex` back to an `IrModule` via
16//!   `tau_ir::from_canonical_bytes`. The round-trip is verified by
17//!   asserting `module.canonical_hash() == ir_payload.canonical_ir_hash`.
18//!
19//! The conformance suite is in-process: BundleMode does NOT spawn any
20//! real plugin processes. Tool invocations are recorded by the same
21//! `RecordingDispatcher` DevMode uses (canned `{"ok": true}` response).
22

36/// Bundle-mode runner: builds a bundle from the fixture, decodes the
37/// embedded IR, and drives the interpreter through the shared
38/// dispatcher.
39///
40/// See the module-level docs for the round-trip invariant.
41pub struct BundleMode;
42
43#[async_trait(?Send)]
44impl ExecutionMode for BundleMode {
45    async fn run(&self, fixture_dir: &Path) -> ConformanceReport {
46        // All non-`Send` setup (the `Caches` closures, tempdir handles)
47        // is confined to this synchronous block so the async future
48        // remains `Send`-friendly (the `?Send` on ExecutionMode is for
49        // run_ir's RefCell internals later — see lib.rs).
50        let prepared: Result<(IrModule, Vec<tau_ports::llm::CompletionResponse>, String), String> = {
51            // 1. Load workflow.toml + mock_llm.jsonl. Missing files are a
52            //    fixture misconfiguration, not a build refusal — panic so
53            //    the failure is loud and obvious in test output.
54            let workflow_toml = std::fs::read_to_string(fixture_dir.join("workflow.toml"))
55                .expect("fixture must contain workflow.toml");
56            let mock_llm_jsonl = std::fs::read_to_string(fixture_dir.join("mock_llm.jsonl"))
57                .expect("fixture must contain mock_llm.jsonl");
58            let responses = parse_mock_llm(&mock_llm_jsonl);
59
60            // 2. Stage a scratch project root: tau.toml = the fixture's
61            //    workflow.toml verbatim + a synthesised empty tau.lock so
62            //    `tau_pkg::bundle::build`'s install-state check passes.
63            //    Conformance fixtures don't ship installed packages, so
64            //    the lockfile is empty and the agent-home-package check
65            //    (which only fires for capability-override agents) stays
66            //    quiet.
67            let scratch = match tempfile::tempdir() {
68                Ok(t) => t,
69                Err(e) => {
70                    return ConformanceReport::build_refused(format!(
71                        "bundle mode: tempdir create failed: {e}"
72                    ));
73                }
74            };
75            if let Err(e) = std::fs::write(scratch.path().join("tau.toml"), &workflow_toml) {
76                return ConformanceReport::build_refused(format!(
77                    "bundle mode: writing scratch tau.toml failed: {e}"
78                ));
79            }
80            // Empty lockfile (schema_version 6 — the current version
81            // tau_pkg::lockfile::LockFile expects).
82            let empty_lock = "schema_version = 6\n\
83                              generated_by_tau_version = \"0.1.0\"\n\
84                              generated_at = \"2024-01-01T00:00:00Z\"\n";
85            if let Err(e) = std::fs::write(scratch.path().join("tau.lock"), empty_lock) {
86                return ConformanceReport::build_refused(format!(
87                    "bundle mode: writing scratch tau.lock failed: {e}"
88                ));
89            }
90
91            // 3. Pre-lower the IR in-process (same SHA-256-of-name cache
92            //    DevMode uses, mirrored from `tau-cli::cmd::build::lower_ir`)
93            //    so the IR payload is populated. Errors here are the build-
94            //    refused signal: report symmetric with DevMode.
95            let target = target_registry::list_available()
96                .next()
97                .expect("at least one target triple available")
98                .triple;
99            let ir_payload = match build_ir_payload(&workflow_toml, &target) {
100                Ok(payload) => Some(payload),
101                Err(refusal) => return ConformanceReport::build_refused(refusal),
102            };
103
104            // 4. Build the bundle. The build path validates the project
105            //    config + lockfile + install state and stamps the IR
106            //    payload into the manifest. v0 fixtures never have
107            //    capability overrides, so step-5 home-package resolution
108            //    is skipped — keeping the scratch dir's empty lockfile
109            //    enough to succeed.
110            let opts = BuildOptions {
111                project_root: scratch.path().to_path_buf(),
112                target,
113                output_path: None,
114                agent_filter: None,
115                ir_payload,
116                governance: None,
117                // Conformance fixtures use inline prompts only.
118                assets: Vec::new(),
119            };
120            let artifact = match build_bundle(opts) {
121                Ok(a) => a,
122                Err(e) => {
123                    return ConformanceReport::build_refused(format!(
124                        "bundle mode: tau_pkg::bundle::build failed: {e}"
125                    ));
126                }
127            };
128
129            // 5. Parse the bundle back and pull out the IR payload. A v2
130            //    bundle MUST carry an `ir_payload`; if it doesn't, the
131            //    build path silently dropped it (would mean we passed
132            //    `None` above — already a bug) — surface as a build
133            //    refusal so the test fails informatively.
134            let bundle_str = match std::fs::read_to_string(&artifact.path) {
135                Ok(s) => s,
136                Err(e) => {
137                    return ConformanceReport::build_refused(format!(
138                        "bundle mode: reading built bundle failed: {e}"
139                    ));
140                }
141            };
142            let manifest = match BundleManifest::parse_str(&bundle_str) {
143                Ok(m) => m,
144                Err(e) => {
145                    return ConformanceReport::build_refused(format!(
146                        "bundle mode: parsing built bundle failed: {e}"
147                    ));
148                }
149            };
150            let ir_payload = match manifest.ir_payload.as_ref() {
151                Some(p) => p,
152                None => {
153                    return ConformanceReport::build_refused(
154                        "bundle mode: built bundle has no ir_payload".to_string(),
155                    );
156                }
157            };
158
159            // 6. Decode the canonical IR bytes back to an IrModule and
160            //    verify the round-trip hash. A mismatch here means
161            //    canonical-encoding drift between the build path and
162            //    `tau_ir::from_canonical_bytes` — a serious bug, surface
163            //    as a build refusal so the test fails informatively.
164            let ir_bytes = match ir_payload.canonical_ir_bytes() {
165                Ok(b) => b,
166                Err(e) => {
167                    return ConformanceReport::build_refused(format!(
168                        "bundle mode: hex decoding canonical_ir_bytes_hex failed: {e}"
169                    ));
170                }
171            };
172            let module = match tau_ir::from_canonical_bytes(&ir_bytes) {
173                Ok(m) => m,
174                Err(e) => {
175                    return ConformanceReport::build_refused(format!(
176                        "bundle mode: decoding canonical IR bytes failed: {e}"
177                    ));
178                }
179            };
180            let recomputed_hash = tau_ir::compute_hash(&module);
181            let recorded_hash = match ir_payload.canonical_ir_hash_bytes() {
182                Ok(h) => h,
183                Err(e) => {
184                    return ConformanceReport::build_refused(format!(
185                        "bundle mode: hex decoding canonical_ir_hash failed: {e}"
186                    ));
187                }
188            };
189            if recomputed_hash != recorded_hash {
190                return ConformanceReport::build_refused(format!(
191                    "bundle mode: IR canonical-hash round-trip mismatch: recorded={}, recomputed={}",
192                    hex_lower(&recorded_hash),
193                    hex_lower(&recomputed_hash),
194                ));
195            }
196
197            // 7. Pick the entry agent (first BTreeMap key — alphabetical
198            //    order), same convention as DevMode + the CLI dispatcher.
199            let entry = module
200                .workflow
201                .agents
202                .keys()
203                .next()
204                .expect("decoded IR module must declare at least one agent")
205                .clone();
206
207            // `scratch` and `caches` closures drop here. The IR bytes
208            // have already been round-tripped through the bundle, so the
209            // tempdir's lifetime doesn't matter past this point.
210            Ok((module, responses, entry.0))
211        };
212
213        match prepared {
214            Ok((module, responses, entry_id)) => {
215                let module = std::sync::Arc::new(module);
216                // Mirror DevMode: pipeline workflows are engine-sequenced
217                // (no single entry agent loop). Both modes drive the same
218                // in-process `run_pipeline`, so the round-tripped bundle
219                // module produces the same report as the in-process lower.
220                if module.workflow.pipeline.is_some() {
221                    drive_pipeline(module, CONFORMANCE_PIPELINE_INPUT.to_string(), responses).await
222                } else {
223                    let entry = tau_ir::AgentId(entry_id);
224                    drive_module(module, &entry, responses).await
225                }
226            }
227            Err(refusal) => ConformanceReport::build_refused(refusal),
228        }
229    }
230}

232/// Lower a workflow.toml to an `IrPayload`, mirroring the cache shape
233/// `tau-cli::cmd::build::lower_ir` uses. Errors surface as the human-
234/// readable `Display` form of `IrError` so DevMode and BundleMode emit
235/// byte-identical build-refused diagnostics for the same fixture.
236fn build_ir_payload(
237    workflow_toml: &str,
238    target: &tau_ports::target::TargetTriple,
239) -> Result<IrPayload, String> {
240    // Project-config validation errors (e.g. `DeliverableNoProducer`) are
241    // surfaced here. The error string is formatted WITHOUT a "bundle mode:"
242    // prefix so both DevMode and BundleMode emit the same diagnostic and
243    // `assert_conform` can compare them byte-for-byte (D-3b refusal symmetry).
244    let config = ProjectConfig::parse_str(workflow_toml).map_err(|e| format!("{e}"))?;
245
246    let caches = Caches {
247        native_tool: &|name: &str| Some(crate::sha256_name(name)),
248        mcp_contract: &|_| None,
249        skill: &|_| None,
250        // Conformance fixtures use inline prompts only (never invoked).
251        prompt_file: &|_| Ok(Vec::new()),
252    };
253    let module = lower_project(&config, target, &caches)
254        .map_err(|e| format!("{e}"))?
255        .module;
256    let bytes = tau_ir::to_canonical_bytes(&module);
257    let hash_bytes = tau_ir::compute_hash(&module);
258    Ok(IrPayload {
259        ir_format: module.ir_format.0.clone(),
260        canonical_ir_hash: hex_lower(&hash_bytes),
261        canonical_ir_bytes_hex: hex_lower(&bytes),
262    })
263}

265/// Encode a byte slice as lowercase hex — symmetric with
266/// `tau-cli::cmd::build::hex_lower` so DevMode-built and BundleMode-
267/// built IR payloads encode their bytes identically.
268fn hex_lower(bytes: &[u8]) -> String {
269    const HEX: &[u8] = b"0123456789abcdef";
270    let mut out = String::with_capacity(bytes.len() * 2);
271    for b in bytes {
272        out.push(HEX[(b >> 4) as usize] as char);
273        out.push(HEX[(b & 0x0f) as usize] as char);
274    }
275    out
276}

279mod tests {
280    use super::*;

282    /// A fixture whose tool declares an `agent.spawn` capability — that
283    /// shape is NOT in `linux-native-strict`'s `required_shapes`
284    /// (`fs_rw_exec_net`), so `tau_ir_lower::capability_fit::check`
285    /// refuses lowering with `IrError::CapabilityFitFailed`. BundleMode
286    /// must surface this as a build-refused report instead of crashing.
287    ///
288    /// This is the D-3b symmetry test: any fixture lowering refuses MUST
289    /// produce a `build_refused`-shaped report in BOTH modes — DevMode
290    /// is covered by its own panic-replacement in `dev_mode.rs`.
291    #[tokio::test(flavor = "current_thread")]
292    async fn bundle_mode_reports_build_refused_on_capability_fit_failure() {
293        let scratch = tempfile::tempdir().unwrap();
294        // workflow.toml: one agent + one tool declaring an agent.spawn
295        // capability. The first available target (linux-native-strict)
296        // supports fs.read/fs.write/process.exec/net.http only, so the
297        // capability_fit step refuses.
298        std::fs::write(
299            scratch.path().join("workflow.toml"),
300            r#"
301packages = ["mock-llm"]
302
303[project]
304name = "refused"
305
306[models.mock-1]
307backend = "mock-llm"
308model = "mock-1"
309
310[agents.solo]
311display_name = "Solo"
312package      = "p@^0.1"
313model        = "mock-1"
314tool_refs    = ["spawner"]
315max_turns    = 1
316
317[tools.spawner]
318native      = "Spawner"
319description = "spawns a sub-agent"
320[[tools.spawner.capabilities]]
321kind = "agent.spawn"
322allowed_kinds = ["critic"]
323"#,
324        )
325        .unwrap();
326        // mock_llm.jsonl unused (build refuses before drive_module).
327        std::fs::write(scratch.path().join("mock_llm.jsonl"), "").unwrap();
328
329        let report = BundleMode.run(scratch.path()).await;
330        let refused = report
331            .build_refused
332            .as_ref()
333            .expect("expected build_refused report; got an executed-run report");
334        // The diagnostic must reference the IrError surface so any
335        // assert_conform asymmetry would name the actual refusal cause.
336        assert!(
337            refused.to_lowercase().contains("capability")
338                || refused.to_lowercase().contains("agentspawn")
339                || refused.contains("AgentSpawn"),
340            "diagnostic should name the capability-fit refusal; got: {refused}"
341        );
342        // The multiset fields stay empty when build_refused is Some.
343        assert!(report.tool_calls.is_empty());
344        assert!(report.message_added.is_empty());
345    }
