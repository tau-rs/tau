22/// Deterministic SHA-256 stand-in for a native tool's symbolic name —
23/// the same shape `tau-cli::cmd::build::sha256_name` uses to populate the
24/// `lower_ir` cache. Shared by DevMode and BundleMode here so both modes
25/// hash identical IR bytes for the same workflow.
26pub(crate) fn sha256_name(name: &str) -> [u8; 32] {
27    use sha2::{Digest, Sha256};
28    let mut h = Sha256::new();
29    h.update(name.as_bytes());
30    h.finalize().into()
31}

33/// Canonical multiset-key bytes for a `tau_domain::Message`.
34///
35/// Strips fields that are inherently per-run nondeterministic so the
36/// dev-mode and bundle-mode runs of the same fixture produce identical
37/// multisets:
38///
39/// - `id` and `parent_id` (fresh `MessageId` UUIDs minted per `Message::new`).
40/// - `created_at` (per-run `SystemTime::now()`).
41/// - `AgentInstanceId` UUIDs inside `Address::Agent(...)` — normalized to
42///   the bare discriminant `"Agent"`. Tool / User / System addresses
43///   keep their full identity (those ARE the load-bearing parts of
44///   D-7a's side-effect record).
45///
46/// Returns JSON bytes of the form
47/// `{"sender": "...", "recipient": "...", "payload": <json>}` so two
48/// distinct payload shapes still produce distinct keys.
49pub(crate) fn canonical_message_bytes(msg: &tau_domain::Message) -> Vec<u8> {
50    use tau_domain::Address;
51    fn addr_tag(a: &Address) -> serde_json::Value {
52        match a {
53            // Strip the AgentInstanceId UUID — per-run nondeterminism.
54            Address::Agent(_) => serde_json::Value::String("Agent".into()),
55            Address::Tool(name) => serde_json::json!({"Tool": name}),
56            Address::User => serde_json::Value::String("User".into()),
57            Address::System => serde_json::Value::String("System".into()),
58            _ => serde_json::Value::String("Unknown".into()),
59        }
60    }
61    let v = serde_json::json!({
62        "sender": addr_tag(&msg.sender),
63        "recipient": addr_tag(&msg.recipient),
64        "payload": serde_json::to_value(&msg.payload).unwrap_or(serde_json::Value::Null),
65    });
66    serde_json::to_vec(&v).unwrap_or_default()
67}

69/// Discriminant of a `RunOutcome` (Completed / Failed / future variants)
70/// as a stable string. Used by `assert_conform` to compare outcomes
71/// without touching the per-run nondeterministic message UUIDs +
72/// timestamps embedded in the variant payloads.
73pub(crate) fn run_outcome_kind(outcome: &Option<RunOutcome>) -> &'static str {
74    match outcome {
75        None => "None",
76        Some(RunOutcome::Completed { .. }) => "Completed",
77        Some(RunOutcome::Failed { .. }) => "Failed",
78        // Future variants — RunOutcome is #[non_exhaustive].
79        Some(_) => "Unknown",
80    }
81}

83/// Side-effect summary produced by a single execution.
84///
85/// Used by [`assert_conform`] to compare dev-mode and bundle-mode runs
86/// per D-7a (multiset side-effect equivalence).
87#[derive(Debug, Clone, PartialEq)]
88pub struct ConformanceReport {
89    /// Final outcome (Completed / Failed / ...).
90    ///
91    /// `Some(_)` when the interpreter actually ran; `None` when the
92    /// build refused before the interpreter could be driven (in which
93    /// case `build_refused` is `Some(_)` instead).
94    pub run_outcome: Option<RunOutcome>,
95    /// Multiset of (tool_name, args_canonical_bytes) → count.
96    ///
97    /// Keyed by `(tool_name, serde_json::to_vec(args))` so two identical
98    /// invocations collapse to one entry with count incremented. Note: this
99    /// is insertion-order-dependent for JSON objects; β.2.6.1 will switch
100    /// to `tau_ir::to_canonical_bytes` for stable ordering across modes.
101    pub tool_calls: BTreeMap<(String, Vec<u8>), u32>,
102    /// Multiset of message bodies keyed by canonical bytes → count.
103    pub message_added: BTreeMap<Vec<u8>, u32>,
104    /// Build-refused marker (D-3b).
105    ///
106    /// `None` ⇒ lowering succeeded and the run was executed; the
107    /// multiset fields reflect the recorded side effects.
108    ///
109    /// `Some(diagnostic)` ⇒ lowering / IR-bundle decode refused this
110    /// fixture before the interpreter could run. The `String` is the
111    /// `Display`-formatted `IrError` (or bundle-build error). When this
112    /// field is `Some`, the multiset fields are empty placeholders and
113    /// [`assert_conform`] compares only the build-refused strings.
114    pub build_refused: Option<String>,
115}
116
117impl ConformanceReport {
118    /// Construct an empty report with the given outcome.
119    pub fn new(run_outcome: RunOutcome) -> Self {
120        Self {
121            run_outcome: Some(run_outcome),
122            tool_calls: BTreeMap::new(),
123            message_added: BTreeMap::new(),
124            build_refused: None,
125        }
126    }
127
128    /// Construct a build-refused report. Use this when lowering (DevMode)
129    /// or bundle build / IR-payload decode (BundleMode) refused before
130    /// the interpreter could be driven.
131    pub fn build_refused(diagnostic: impl Into<String>) -> Self {
132        Self {
133            run_outcome: None,
134            tool_calls: BTreeMap::new(),
135            message_added: BTreeMap::new(),
136            build_refused: Some(diagnostic.into()),
137        }
138    }
139
140    /// Record a tool invocation.
141    ///
142    /// `args` is the canonical JSON bytes of the tool's input.
143    pub fn record_tool_call(&mut self, tool_name: impl Into<String>, args_canonical: Vec<u8>) {
144        *self
145            .tool_calls
146            .entry((tool_name.into(), args_canonical))
147            .or_insert(0) += 1;
148    }
149
150    /// Record a message body (canonical bytes).
151    pub fn record_message(&mut self, canonical_bytes: Vec<u8>) {
152        *self.message_added.entry(canonical_bytes).or_insert(0) += 1;
153    }
154}

156/// Assert that two reports are equivalent per D-7a.
157///
158/// Comparison semantics:
159///
160/// - When BOTH reports carry `build_refused = Some(_)`: compares only
161///   the refusal strings. The two modes must produce the same
162///   `IrError::Display` (or equivalent bundle-build error message),
163///   matching the D-3b "refusal symmetry" rule — both surfaces must
164///   refuse the same fixture for the same reason.
165/// - When NEITHER carries `build_refused`: compares `run_outcome`,
166///   `tool_calls`, and `message_added` byte-for-byte.
167/// - When ONE carries `build_refused` and the other does not: panics
168///   immediately — the two surfaces disagree about whether the fixture
169///   is buildable, which is a conformance bug.
170pub fn assert_conform(dev: &ConformanceReport, bundle: &ConformanceReport) {
171    match (&dev.build_refused, &bundle.build_refused) {
172        (Some(d), Some(b)) => {
173            assert_eq!(
174                d, b,
175                "build-refused diagnostic mismatch: dev refused with {d:?}, bundle refused with {b:?}"
176            );
177        }
178        (Some(d), None) => {
179            panic!(
180                "build-refused asymmetry: dev refused ({d:?}) but bundle did not refuse \
181                 (bundle outcome: {:?})",
182                bundle.run_outcome
183            );
184        }
185        (None, Some(b)) => {
186            panic!(
187                "build-refused asymmetry: bundle refused ({b:?}) but dev did not refuse \
188                 (dev outcome: {:?})",
189                dev.run_outcome
190            );
191        }
192        (None, None) => {
193            // Compare RunOutcome by *discriminant* only — variant
194            // payloads embed per-run nondeterministic message UUIDs
195            // and SystemTime::now() values that would never compare
196            // equal across two runs. D-7a's side-effect equivalence is
197            // captured by `tool_calls` and `message_added` (which key
198            // off canonical message bytes that strip those nondeterminisms).
199            assert_eq!(
200                run_outcome_kind(&dev.run_outcome),
201                run_outcome_kind(&bundle.run_outcome),
202                "RunOutcome kind mismatch: dev={:?}, bundle={:?}",
203                dev.run_outcome,
204                bundle.run_outcome,
205            );
206            assert_eq!(
207                dev.tool_calls, bundle.tool_calls,
208                "tool-call multiset mismatch"
209            );
210            assert_eq!(
211                dev.message_added, bundle.message_added,
212                "message-added multiset mismatch"
213            );
214        }
215    }
216}

218/// Trait the runner calls to execute a fixture under one mode.
219///
220/// `DevMode` runs the IR interpreter directly with in-process tool
221/// dispatch. `BundleMode` (β.2.6.1) builds a bundle and routes through
222/// the bundle's wasm-gated dispatch path.
223///
224/// `?Send` because `tau_runtime_core::interpreter::run_ir` uses `RefCell`
225/// internally and produces a non-`Send` future. Tests must use
226/// `#[tokio::test(flavor = "current_thread")]` or a `LocalSet`.
227#[async_trait::async_trait(?Send)]
228pub trait ExecutionMode {
229    /// Run the fixture at `fixture_dir` and return a side-effect report.
230    async fn run(&self, fixture_dir: &Path) -> ConformanceReport;
231}

237use std::sync::Arc;
238
239use tau_runtime_core::error::RuntimeError;
240use tau_runtime_core::interpreter::deterministic::DeterministicRegistry;

242/// Type alias for the closure stored per registered function.
243type RegistryFn =
244    Arc<dyn Fn(&serde_json::Value) -> Result<serde_json::Value, RuntimeError> + Send + Sync>;
245
246/// A `DeterministicRegistry` backed by a `BTreeMap<String, Fn>`.
247///
248/// The conformance suite uses this to wire scripted deterministic
249/// functions into `RecordingDispatcher::deterministic_registry()`.
250/// Fixture authors call [`MapBackedDeterministicRegistry::with`] to
251/// register named functions.
252#[derive(Default)]
253pub struct MapBackedDeterministicRegistry {
254    fns: BTreeMap<String, RegistryFn>,
255}
256
257impl MapBackedDeterministicRegistry {
258    /// Register a function under `fn_name`. The function must be pure
259    /// (no I/O, no global mutation).
260    pub fn with<F>(mut self, fn_name: impl Into<String>, f: F) -> Self
261    where
262        F: Fn(&serde_json::Value) -> Result<serde_json::Value, RuntimeError>
263            + Send
264            + Sync
265            + 'static,
266    {
267        self.fns.insert(fn_name.into(), Arc::new(f));
268        self
269    }
270}
271
272impl DeterministicRegistry for MapBackedDeterministicRegistry {
273    fn invoke(
274        &self,
275        fn_name: &str,
276        args: &serde_json::Value,
277    ) -> Result<serde_json::Value, RuntimeError> {
278        let f = self
279            .fns
280            .get(fn_name)
281            .ok_or_else(|| RuntimeError::Internal {
282                message: format!("MapBackedDeterministicRegistry: unknown fn {fn_name:?}"),
283            })?;
284        f(args)
285    }
286}

288/// The canonical conformance-fixture registry used by the test suite.
289///
290/// Registers:
291///
292/// - `parse_celsius`: `{"raw": "<digits>"}` → `{"celsius": <int>}`.
293///   Used by fixture 05 (`05_deterministic_step`).
294/// - `FN_BUILTIN_NON_EMPTY` (`__tau::goal::non_empty`): the built-in `non_empty`
295///   goal predicate. Used by fixture 09 (`09_deliverables_happy`).
296///
297/// conformance-local predicate registry — production predicates live in
298/// tau-cli and are unit-tested there (Task 22); this exercises the
299/// interpreter's check-dispatch + retry e2e. The same registry answers both
300/// deterministic-step fns and built-in goal-predicate fns, since the
301/// interpreter resolves both through `DeterministicRegistry::invoke`.
302pub fn fixture_deterministic_registry() -> Arc<MapBackedDeterministicRegistry> {
303    use tau_runtime_core::vocabulary::FN_BUILTIN_NON_EMPTY;
304
305    Arc::new(
306        MapBackedDeterministicRegistry::default()
307            .with("parse_celsius", |args: &serde_json::Value| {
308                let raw = args.get("raw").and_then(|v| v.as_str()).ok_or_else(|| {
309                    RuntimeError::Internal {
310                        message: "parse_celsius: `raw` must be a string".into(),
311                    }
312                })?;
313                let celsius: i64 = raw.trim().parse().map_err(|e| RuntimeError::Internal {
314                    message: format!("parse_celsius: not an integer: {e}"),
315                })?;
316                Ok(serde_json::json!({ "celsius": celsius }))
317            })
318            .with(FN_BUILTIN_NON_EMPTY, |args: &serde_json::Value| {
319                // `non_empty`: the locus resolved to present, non-blank content.
320                // The interpreter injects `present` (bool) and `content`
321                // (string|null) into the args object (see evaluate_goal).
322                let present = args["present"].as_bool().unwrap_or(false);
323                let content = args["content"].as_str().unwrap_or("");
324                Ok(serde_json::json!(present && !content.trim().is_empty()))
325            }),
326    )
