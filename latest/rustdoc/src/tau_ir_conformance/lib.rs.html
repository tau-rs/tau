22/// Deterministic SHA-256 stand-in for a native tool's symbolic name —
23/// the same shape `tau-cli::cmd::build::sha256_name` uses to populate the
24/// `lower_ir` cache. Shared by DevMode and BundleMode here so both modes
25/// hash identical IR bytes for the same workflow.
26pub(crate) fn sha256_name(name: &str) -> [u8; 32] {
27    use sha2::{Digest, Sha256};
28    let mut h = Sha256::new();
29    h.update(name.as_bytes());
30    h.finalize().into()
31}

33/// Canonical multiset-key bytes for a `tau_domain::Message`.
34///
35/// Strips fields that are inherently per-run nondeterministic so the
36/// dev-mode and bundle-mode runs of the same fixture produce identical
37/// multisets:
38///
39/// - `id` and `parent_id` (fresh `MessageId` UUIDs minted per `Message::new`).
40/// - `created_at` (per-run `SystemTime::now()`).
41/// - `AgentInstanceId` UUIDs inside `Address::Agent(...)` — normalized to
42///   the bare discriminant `"Agent"`. Tool / User / System addresses
43///   keep their full identity (those ARE the load-bearing parts of
44///   D-7a's side-effect record).
45///
46/// Returns JSON bytes of the form
47/// `{"sender": "...", "recipient": "...", "payload": <json>}` so two
48/// distinct payload shapes still produce distinct keys.
49pub(crate) fn canonical_message_bytes(msg: &tau_domain::Message) -> Vec<u8> {
50    use tau_domain::Address;
51    fn addr_tag(a: &Address) -> serde_json::Value {
52        match a {
53            // Strip the AgentInstanceId UUID — per-run nondeterminism.
54            Address::Agent(_) => serde_json::Value::String("Agent".into()),
55            Address::Tool(name) => serde_json::json!({"Tool": name}),
56            Address::User => serde_json::Value::String("User".into()),
57            Address::System => serde_json::Value::String("System".into()),
58            _ => serde_json::Value::String("Unknown".into()),
59        }
60    }
61    let v = serde_json::json!({
62        "sender": addr_tag(&msg.sender),
63        "recipient": addr_tag(&msg.recipient),
64        "payload": serde_json::to_value(&msg.payload).unwrap_or(serde_json::Value::Null),
65    });
66    serde_json::to_vec(&v).unwrap_or_default()
67}

69/// Discriminant of a `RunOutcome` (Completed / Failed / future variants)
70/// as a stable string. Used by `assert_conform` to compare outcomes
71/// without touching the per-run nondeterministic message UUIDs +
72/// timestamps embedded in the variant payloads.
73pub(crate) fn run_outcome_kind(outcome: &Option<RunOutcome>) -> &'static str {
74    match outcome {
75        None => "None",
76        Some(RunOutcome::Completed { .. }) => "Completed",
77        Some(RunOutcome::Failed { .. }) => "Failed",
78        // Future variants — RunOutcome is #[non_exhaustive].
79        Some(_) => "Unknown",
80    }
81}

83/// Summary of a pipeline paused at a top-level `Suspend` step (EPIC 4.3).
84///
85/// Captures the observable shape of a [`tau_ports::orchestration::PipelineSuspension`]
86/// persisted by `run_pipeline_suspendable`, trimmed to the fields that must
87/// agree across DevMode and BundleMode: the signal a `--resume --signal`
88/// must supply, the top-level step index paused at, and a deterministic
89/// snapshot of the accumulated pipeline outputs at pause time.
90///
91/// `outputs_canonical` is `serde_json::to_vec` of the persisted
92/// `BTreeMap<String, serde_json::Value>` — `BTreeMap`'s iteration order is
93/// sorted by key, so the encoding is deterministic and comparable
94/// byte-for-byte across modes without a separate canonicalization step.
95#[derive(Debug, Clone, PartialEq)]
96pub struct SuspendedSummary {
97    /// The signal name a `--resume --signal <name>` must supply to continue.
98    pub resume_signal: String,
99    /// Index of the `Suspend` step in the top-level pipeline slice.
100    pub step_cursor: usize,
101    /// Deterministic `serde_json::to_vec` encoding of the persisted
102    /// `PipelineSuspension.outputs` snapshot (sorted-key `BTreeMap`).
103    pub outputs_canonical: Vec<u8>,
104}

106/// Side-effect summary produced by a single execution.
107///
108/// Used by [`assert_conform`] to compare dev-mode and bundle-mode runs
109/// per D-7a (multiset side-effect equivalence).
110#[derive(Debug, Clone, PartialEq)]
111pub struct ConformanceReport {
112    /// Final outcome (Completed / Failed / ...).
113    ///
114    /// `Some(_)` when the interpreter actually ran; `None` when the
115    /// build refused before the interpreter could be driven (in which
116    /// case `build_refused` is `Some(_)` instead).
117    pub run_outcome: Option<RunOutcome>,
118    /// Multiset of (tool_name, args_canonical_bytes) → count.
119    ///
120    /// Keyed by `(tool_name, serde_json::to_vec(args))` so two identical
121    /// invocations collapse to one entry with count incremented. Note: this
122    /// is insertion-order-dependent for JSON objects; β.2.6.1 will switch
123    /// to `tau_ir::to_canonical_bytes` for stable ordering across modes.
124    pub tool_calls: BTreeMap<(String, Vec<u8>), u32>,
125    /// Multiset of message bodies keyed by canonical bytes → count.
126    pub message_added: BTreeMap<Vec<u8>, u32>,
127    /// Build-refused marker (D-3b).
128    ///
129    /// `None` ⇒ lowering succeeded and the run was executed; the
130    /// multiset fields reflect the recorded side effects.
131    ///
132    /// `Some(diagnostic)` ⇒ lowering / IR-bundle decode refused this
133    /// fixture before the interpreter could run. The `String` is the
134    /// `Display`-formatted `IrError` (or bundle-build error). When this
135    /// field is `Some`, the multiset fields are empty placeholders and
136    /// [`assert_conform`] compares only the build-refused strings.
137    pub build_refused: Option<String>,
138    /// Suspended-at-pause marker (EPIC 4.3).
139    ///
140    /// `None` ⇒ the run either completed/failed normally or was build-
141    /// refused. `Some(summary)` ⇒ the pipeline paused at a top-level
142    /// `Suspend` step; `run_outcome` is `None` and `build_refused` is
143    /// `None` in that case (a suspension is a third terminal state,
144    /// distinct from both). [`assert_conform`] treats this as a first-class
145    /// terminal state: both modes must agree on whether a fixture suspends,
146    /// and if so, on the persisted [`SuspendedSummary`].
147    pub suspended: Option<SuspendedSummary>,
148}
149
150impl ConformanceReport {
151    /// Construct an empty report with the given outcome.
152    pub fn new(run_outcome: RunOutcome) -> Self {
153        Self {
154            run_outcome: Some(run_outcome),
155            tool_calls: BTreeMap::new(),
156            message_added: BTreeMap::new(),
157            build_refused: None,
158            suspended: None,
159        }
160    }
161
162    /// Construct a build-refused report. Use this when lowering (DevMode)
163    /// or bundle build / IR-payload decode (BundleMode) refused before
164    /// the interpreter could be driven.
165    pub fn build_refused(diagnostic: impl Into<String>) -> Self {
166        Self {
167            run_outcome: None,
168            tool_calls: BTreeMap::new(),
169            message_added: BTreeMap::new(),
170            build_refused: Some(diagnostic.into()),
171            suspended: None,
172        }
173    }
174
175    /// Construct a suspended report. Use this when `run_pipeline_suspendable`
176    /// returns `PipelineOutcome::Suspended` — the pipeline paused at a
177    /// top-level `Suspend` step rather than completing or failing.
178    /// `tool_calls`/`message_added` still get populated by the caller via
179    /// [`ConformanceReport::record_tool_call`] / [`ConformanceReport::record_message`]
180    /// for whatever side effects occurred before the pause.
181    pub fn suspended(summary: SuspendedSummary) -> Self {
182        Self {
183            run_outcome: None,
184            tool_calls: BTreeMap::new(),
185            message_added: BTreeMap::new(),
186            build_refused: None,
187            suspended: Some(summary),
188        }
189    }
190
191    /// Record a tool invocation.
192    ///
193    /// `args` is the canonical JSON bytes of the tool's input.
194    pub fn record_tool_call(&mut self, tool_name: impl Into<String>, args_canonical: Vec<u8>) {
195        *self
196            .tool_calls
197            .entry((tool_name.into(), args_canonical))
198            .or_insert(0) += 1;
199    }
200
201    /// Record a message body (canonical bytes).
202    pub fn record_message(&mut self, canonical_bytes: Vec<u8>) {
203        *self.message_added.entry(canonical_bytes).or_insert(0) += 1;
204    }
205}

207/// Assert that two reports are equivalent per D-7a.
208///
209/// Comparison semantics:
210///
211/// - When BOTH reports carry `build_refused = Some(_)`: compares only
212///   the refusal strings. The two modes must produce the same
213///   `IrError::Display` (or equivalent bundle-build error message),
214///   matching the D-3b "refusal symmetry" rule — both surfaces must
215///   refuse the same fixture for the same reason.
216/// - When NEITHER carries `build_refused`: a suspended run (EPIC 4.3) is a
217///   third terminal state, checked before falling back to `run_outcome`:
218///   - BOTH suspended: compares the two [`SuspendedSummary`]s for equality,
219///     plus the `tool_calls`/`message_added` multisets (any side effects
220///     recorded before the pause).
221///   - ONE suspended, the other not: panics — the two surfaces disagree
222///     about whether the fixture pauses, which is a conformance bug.
223///   - NEITHER suspended: compares `run_outcome`, `tool_calls`, and
224///     `message_added` byte-for-byte (the pre-EPIC-4.3 behavior).
225/// - When ONE carries `build_refused` and the other does not: panics
226///   immediately — the two surfaces disagree about whether the fixture
227///   is buildable, which is a conformance bug.
228pub fn assert_conform(dev: &ConformanceReport, bundle: &ConformanceReport) {
229    match (&dev.build_refused, &bundle.build_refused) {
230        (Some(d), Some(b)) => {
231            assert_eq!(
232                d, b,
233                "build-refused diagnostic mismatch: dev refused with {d:?}, bundle refused with {b:?}"
234            );
235        }
236        (Some(d), None) => {
237            panic!(
238                "build-refused asymmetry: dev refused ({d:?}) but bundle did not refuse \
239                 (bundle outcome: {:?})",
240                bundle.run_outcome
241            );
242        }
243        (None, Some(b)) => {
244            panic!(
245                "build-refused asymmetry: bundle refused ({b:?}) but dev did not refuse \
246                 (dev outcome: {:?})",
247                dev.run_outcome
248            );
249        }
250        // Neither surface refused the build — a suspended pause point
251        // (EPIC 4.3) is checked next, ahead of `run_outcome`, since a
252        // suspended report always carries `run_outcome: None`.
253        (None, None) => match (&dev.suspended, &bundle.suspended) {
254            (Some(ds), Some(bs)) => {
255                assert_eq!(
256                    ds, bs,
257                    "suspended-summary mismatch: dev suspended with {ds:?}, bundle suspended with {bs:?}"
258                );
259                assert_eq!(
260                    dev.tool_calls, bundle.tool_calls,
261                    "tool-call multiset mismatch (suspended run)"
262                );
263                assert_eq!(
264                    dev.message_added, bundle.message_added,
265                    "message-added multiset mismatch (suspended run)"
266                );
267            }
268            (Some(ds), None) => {
269                panic!(
270                    "suspend asymmetry: dev suspended ({ds:?}) but bundle did not suspend \
271                     (bundle outcome: {:?})",
272                    bundle.run_outcome
273                );
274            }
275            (None, Some(bs)) => {
276                panic!(
277                    "suspend asymmetry: bundle suspended ({bs:?}) but dev did not suspend \
278                     (dev outcome: {:?})",
279                    dev.run_outcome
280                );
281            }
282            (None, None) => {
283                // Compare RunOutcome by *discriminant* only — variant
284                // payloads embed per-run nondeterministic message UUIDs
285                // and SystemTime::now() values that would never compare
286                // equal across two runs. D-7a's side-effect equivalence is
287                // captured by `tool_calls` and `message_added` (which key
288                // off canonical message bytes that strip those nondeterminisms).
289                assert_eq!(
290                    run_outcome_kind(&dev.run_outcome),
291                    run_outcome_kind(&bundle.run_outcome),
292                    "RunOutcome kind mismatch: dev={:?}, bundle={:?}",
293                    dev.run_outcome,
294                    bundle.run_outcome,
295                );
296                assert_eq!(
297                    dev.tool_calls, bundle.tool_calls,
298                    "tool-call multiset mismatch"
299                );
300                assert_eq!(
301                    dev.message_added, bundle.message_added,
302                    "message-added multiset mismatch"
303                );
304            }
305        },
306    }
307}

309/// Trait the runner calls to execute a fixture under one mode.
310///
311/// `DevMode` runs the IR interpreter directly with in-process tool
312/// dispatch. `BundleMode` (β.2.6.1) builds a bundle and routes through
313/// the bundle's wasm-gated dispatch path.
314///
315/// `?Send` because `tau_runtime_core::interpreter::run_ir` uses `RefCell`
316/// internally and produces a non-`Send` future. Tests must use
317/// `#[tokio::test(flavor = "current_thread")]` or a `LocalSet`.
318#[async_trait::async_trait(?Send)]
319pub trait ExecutionMode {
320    /// Run the fixture at `fixture_dir` and return a side-effect report.
321    async fn run(&self, fixture_dir: &Path) -> ConformanceReport;
322}

330use tau_runtime_core::error::RuntimeError;
331use tau_runtime_core::interpreter::deterministic::DeterministicRegistry;

333/// Type alias for the closure stored per registered function.
334type RegistryFn =
335    Arc<dyn Fn(&serde_json::Value) -> Result<serde_json::Value, RuntimeError> + Send + Sync>;
336
337/// A `DeterministicRegistry` backed by a `BTreeMap<String, Fn>`.
338///
339/// The conformance suite uses this to wire scripted deterministic
340/// functions into `RecordingDispatcher::deterministic_registry()`.
341/// Fixture authors call [`MapBackedDeterministicRegistry::with`] to
342/// register named functions.
343#[derive(Default)]
344pub struct MapBackedDeterministicRegistry {
345    fns: BTreeMap<String, RegistryFn>,
346}
347
348impl MapBackedDeterministicRegistry {
349    /// Register a function under `fn_name`. The function must be pure
350    /// (no I/O, no global mutation).
351    pub fn with<F>(mut self, fn_name: impl Into<String>, f: F) -> Self
352    where
353        F: Fn(&serde_json::Value) -> Result<serde_json::Value, RuntimeError>
354            + Send
355            + Sync
356            + 'static,
357    {
358        self.fns.insert(fn_name.into(), Arc::new(f));
359        self
360    }
361}
362
363impl DeterministicRegistry for MapBackedDeterministicRegistry {
364    fn invoke(
365        &self,
366        fn_name: &str,
367        args: &serde_json::Value,
368    ) -> Result<serde_json::Value, RuntimeError> {
369        let f = self
370            .fns
371            .get(fn_name)
372            .ok_or_else(|| RuntimeError::Internal {
373                message: format!("MapBackedDeterministicRegistry: unknown fn {fn_name:?}"),
374            })?;
375        f(args)
376    }
377}

379/// The canonical conformance-fixture registry used by the test suite.
380///
381/// Registers:
382///
383/// - `parse_celsius`: `{"raw": "<digits>"}` → `{"celsius": <int>}`.
384///   Used by fixture 05 (`05_deterministic_step`).
385/// - `FN_BUILTIN_NON_EMPTY` (`__tau::goal::non_empty`): the built-in `non_empty`
386///   goal predicate. Used by fixture 09 (`09_deliverables_happy`).
387///
388/// conformance-local predicate registry — production predicates live in
389/// tau-cli and are unit-tested there (Task 22); this exercises the
390/// interpreter's check-dispatch + retry e2e. The same registry answers both
391/// deterministic-step fns and built-in goal-predicate fns, since the
392/// interpreter resolves both through `DeterministicRegistry::invoke`.
393pub fn fixture_deterministic_registry() -> Arc<MapBackedDeterministicRegistry> {
394    use tau_runtime_core::vocabulary::{FN_BUILTIN_MATCHES, FN_BUILTIN_NON_EMPTY};
395
396    Arc::new(
397        MapBackedDeterministicRegistry::default()
398            .with("parse_celsius", |args: &serde_json::Value| {
399                let raw = args.get("raw").and_then(|v| v.as_str()).ok_or_else(|| {
400                    RuntimeError::Internal {
401                        message: "parse_celsius: `raw` must be a string".into(),
402                    }
403                })?;
404                let celsius: i64 = raw.trim().parse().map_err(|e| RuntimeError::Internal {
405                    message: format!("parse_celsius: not an integer: {e}"),
406                })?;
407                Ok(serde_json::json!({ "celsius": celsius }))
408            })
409            .with(FN_BUILTIN_NON_EMPTY, |args: &serde_json::Value| {
410                // `non_empty`: the locus resolved to present, non-blank content.
411                // The interpreter injects `present` (bool) and `content`
412                // (string|null) into the args object (see evaluate_goal).
413                let present = args["present"].as_bool().unwrap_or(false);
414                let content = args["content"].as_str().unwrap_or("");
415                Ok(serde_json::json!(present && !content.trim().is_empty()))
416            })
417            .with(FN_BUILTIN_MATCHES, |args: &serde_json::Value| {
418                // `matches`: present content matches the `pattern` regex. The
419                // interpreter injects `present`/`content`; `predicate_call`
420                // injects `pattern` (see evaluate_goal). Mirrors production
421                // `GoalPredicate::Matches` semantics with a real regex engine.
422                let present = args["present"].as_bool().unwrap_or(false);
423                let content = args["content"].as_str().unwrap_or("");
424                let pattern = args["pattern"]
425                    .as_str()
426                    .ok_or_else(|| RuntimeError::Internal {
427                        message: "matches: `pattern` must be a string".into(),
428                    })?;
429                let re = regex::Regex::new(pattern).map_err(|e| RuntimeError::Internal {
430                    message: format!("matches: invalid regex {pattern:?}: {e}"),
431                })?;
432                Ok(serde_json::json!(present && re.is_match(content)))
433            }),
434    )
