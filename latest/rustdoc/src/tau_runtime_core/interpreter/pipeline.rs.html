32/// Max `Parallel` branches in flight at once (nginx `worker_connections`
33/// analogue). Bounded cooperative fork-join — see ADR-0059.
34const PARALLEL_CAP: usize = 8;

36/// Terminal outcome of driving a pipeline via [`run_pipeline_suspendable`].
37#[derive(Debug)]
38pub enum PipelineOutcome {
39    /// Ran to the end; carries the accumulated outputs.
40    Completed(OutputStore),
41    /// Paused at a top-level `Suspend`. State was persisted via the
42    /// `SuspensionStore` passed in `SuspendConfig` before returning.
43    Suspended {
44        /// The run id the suspension was persisted under.
45        run_id: String,
46        /// The signal name a `--resume --signal <name>` must supply.
47        resume_signal: String,
48        /// The pipeline-step id of the `Suspend` block that paused.
49        step_id: String,
50    },
51}

53/// How a `run_steps` slice terminated. Only the top-level slice (the one
54/// `run_pipeline_suspendable` invokes directly, with `suspend: Some(_)`) can
55/// terminate `Suspended`; nested slices (`Branch`/`Loop`/`Parallel` bodies)
56/// always pass `suspend: None` and must complete or error.
57enum StepsFlow {
58    /// The slice ran every step.
59    Completed,
60    /// The slice paused at a `Suspend` step.
61    Suspended {
62        /// The pipeline-step id of the `Suspend` block.
63        step_id: String,
64        /// The signal name the resumer must supply.
65        resume_signal: String,
66    },
67}

69/// Durable-suspend wiring for the top-level pipeline driver. Carries the
70/// run id (the suspension key + the `--resume` handle) and the store a
71/// `Suspend` step's snapshot is persisted to.
72pub struct SuspendConfig {
73    /// Run id used as the suspension key + `--resume` handle.
74    pub run_id: String,
75    /// Where a pause is durably persisted.
76    pub store: Arc<dyn tau_ports::orchestration::SuspensionStore>,
77}

79/// Rehydrated state for a resumed run: the `OutputStore` snapshot restored
80/// from the prior suspension, and the top-level step index to resume at
81/// (`step_cursor + 1`, so the `Suspend` step itself is not re-run).
82pub struct ResumeState {
83    /// The `OutputStore` snapshot restored from the suspension.
84    pub store: OutputStore,
85    /// Top-level step index to resume at (`step_cursor + 1`).
86    pub start_at: usize,
87    /// Per-check attempt counts restored from the prior suspension, so a
88    /// check's `max_attempts` budget accumulates across resume boundaries
89    /// instead of resetting each resume. Fresh runs pass an empty map.
90    pub attempts: BTreeMap<String, u32>,
91}

93/// Internal: passed down to `run_steps` for the top-level slice only. A
94/// nested slice (Branch/Loop/Parallel body) receives `None`, which is how
95/// `run_steps` tells a real `Suspend` step from one that is out of place.
96struct SuspendCtx<'a> {
97    run_id: &'a str,
98    ir_digest: &'a str,
99    store: &'a dyn tau_ports::orchestration::SuspensionStore,
100}

102/// Drive an `IrModule`'s pipeline, pausing durably at a top-level `Suspend`
103/// step rather than erroring.
104///
105/// The module's `workflow.pipeline` must be `Some` — callers branch on
106/// that before dispatching here (see `tau run`). Each step renders its
107/// `input` template against the run `input` and prior outputs, runs the
108/// step, and records the step's output keyed by its pipeline-step id so
109/// later steps can reference it via `${steps.<id>.output}`.
110///
111/// [`StepRun::Agent`], [`StepRun::Tool`], [`StepRun::Deterministic`],
112/// [`StepRun::Check`], [`StepRun::Branch`], [`StepRun::Loop`],
113/// [`StepRun::Parallel`], and [`StepRun::Suspend`] steps are all supported. A
114/// `Check` step evaluates a postcondition (goal or deliverable) against the
115/// accumulated outputs. A `Branch` evaluates its condition and recurses into
116/// the chosen arm (`then`/`otherwise`) against the same shared store; it
117/// stores no output of its own. A `Loop` runs its `body` up to `max_iters`
118/// times, checking `until` after each pass and threading a failed verdict's
119/// rationale into the next pass as feedback; exhausting `max_iters` without
120/// `until` holding is a hard error ([`RuntimeError::LoopExhausted`]). Like
121/// `Branch`, a `Loop` stores no output of its own. A `Parallel` forks each
122/// branch over an isolated read-only snapshot of the store (branches cannot
123/// see each other's outputs), drives them with bounded concurrency
124/// (`PARALLEL_CAP` in flight at once), and merges each branch's produced
125/// outputs back into the shared store in index order once all branches
126/// complete; like `Branch` and `Loop`, it stores no output of its own. A
127/// top-level `Suspend` step pauses the run: computes the module's
128/// canonical-IR digest once (for drift-detection on a future resume), seeds
129/// the run's `OutputStore` and start index either fresh (`resume: None`) or
130/// from a prior pause (`resume: Some(ResumeState)` — restore-and-continue:
131/// the store is rehydrated from the persisted snapshot and the walk resumes
132/// at `step_cursor + 1`, so the prefix up to and including the `Suspend`
133/// step is never re-run), and — on hitting `Suspend` — durably persists a
134/// [`tau_ports::orchestration::PipelineSuspension`] via the `SuspensionStore`
135/// in `SuspendConfig` and returns [`PipelineOutcome::Suspended`] rather than
136/// erroring. Only a top-level `Suspend` may pause; one nested inside a
137/// `Branch`/`Loop`/`Parallel` body is a build-time typecheck error (an
138/// [`RuntimeError::Internal`] here would mean that check regressed).
139///
140/// # Failure handling: abort, or rewind-to-gate retry
141///
142/// On a failed check the [`RetryPolicy`](tau_ir::check::RetryPolicy)
143/// decides what happens. If `on_fail == Abort`, or the attempt count has
144/// reached `max_attempts`, the pipeline aborts with
145/// [`RuntimeError::CheckFailed`]. Otherwise the loop emits
146/// [`EV_CHECK_RETRY`], stashes the judge's rationale as `feedback` keyed by
147/// the check's `gate` step and the check's id, and rewinds the loop index to
148/// that `gate` step so the forward slice re-runs. On the next pass through the
149/// gate's agent step every outstanding rationale registered for *that* gate is
150/// injected as a labelled prior turn (`"Previous attempt rejected: (check
151/// '<id>') <rationale>"`) so the agent sees *why* it was rejected. Keying by
152/// gate + check id keeps concurrently-failing checks distinct (a pass clears
153/// only its own check's entry; a fresh failure updates only its own) and
154/// confines each check's feedback to its own gate step.
155pub async fn run_pipeline_suspendable<D>(
156    module: Arc<IrModule>,
157    input: String,
158    dispatcher: Arc<D>,
159    suspend: SuspendConfig,
160    resume: Option<ResumeState>,
161) -> Result<PipelineOutcome, RuntimeError>
162where
163    D: ToolDispatcher + Send + Sync + 'static,
164{
165    let pipeline = module
166        .workflow
167        .pipeline
168        .clone()
169        .ok_or_else(|| RuntimeError::Internal {
170            message: "run_pipeline called on a module without a pipeline".to_string(),
171        })?;
172
173    // Canonical-IR digest, computed once, for drift-detection on resume.
174    let ir_digest = {
175        let bytes = tau_ir::to_canonical_bytes(&module);
176        tau_ir::asset::asset_hash(&bytes)
177    };
178
179    let (mut store, start_at, initial_attempts) = match resume {
180        Some(r) => (r.store, r.start_at, r.attempts),
181        None => (OutputStore::new(), 0, BTreeMap::new()),
182    };
183
184    let ctx = SuspendCtx {
185        run_id: &suspend.run_id,
186        ir_digest: &ir_digest,
187        store: suspend.store.as_ref(),
188    };
189    let flow = run_steps(
190        &module,
191        &pipeline.steps,
192        &input,
193        &mut store,
194        &dispatcher,
195        None,
196        Some(&ctx),
197        start_at,
198        initial_attempts,
199    )
200    .await?;
201    Ok(match flow {
202        StepsFlow::Completed => PipelineOutcome::Completed(store),
203        StepsFlow::Suspended {
204            step_id,
205            resume_signal,
206        } => PipelineOutcome::Suspended {
207            run_id: suspend.run_id,
208            resume_signal,
209            step_id,
210        },
211    })
212}

214/// Non-suspend convenience wrapper over [`run_pipeline_suspendable`] for
215/// callers that cannot pause (e.g. the bundle-run path in v1). Wires a
216/// no-op `SuspensionStore` and runs fresh (never resumes); a pipeline that
217/// hits a `Suspend` step errors [`RuntimeError::SuspendUnsupported`] instead
218/// of pausing. Callers that support HITL suspend/resume use
219/// `run_pipeline_suspendable` directly.
220pub async fn run_pipeline<D>(
221    module: Arc<IrModule>,
222    input: String,
223    dispatcher: Arc<D>,
224) -> Result<OutputStore, RuntimeError>
225where
226    D: ToolDispatcher + Send + Sync + 'static,
227{
228    struct NoopSuspensions;
229    impl tau_ports::orchestration::SuspensionStore for NoopSuspensions {
230        fn persist_suspension(
231            &self,
232            _: &tau_ports::orchestration::PipelineSuspension,
233        ) -> Result<(), tau_ports::orchestration::CheckpointError> {
234            Ok(())
235        }
236        fn load_suspension(
237            &self,
238            _: &tau_ports::orchestration::RunId,
239        ) -> Result<
240            Option<tau_ports::orchestration::PipelineSuspension>,
241            tau_ports::orchestration::CheckpointError,
242        > {
243            Ok(None)
244        }
245    }
246    let outcome = run_pipeline_suspendable(
247        module,
248        input,
249        dispatcher,
250        SuspendConfig {
251            run_id: String::new(),
252            store: Arc::new(NoopSuspensions),
253        },
254        None,
255    )
256    .await?;
257    match outcome {
258        PipelineOutcome::Completed(store) => Ok(store),
259        PipelineOutcome::Suspended {
260            step_id,
261            resume_signal,
262            ..
263        } => Err(RuntimeError::SuspendUnsupported {
264            step: step_id,
265            resume_signal,
266        }),
267    }
268}

270/// Execute a slice of pipeline steps against a shared [`OutputStore`].
271///
272/// Extracted from [`run_pipeline`] so nested blocks (Branch/Parallel/Loop)
273/// can drive their own step slices with the same gate-rewind, feedback, and
274/// check-dispatch semantics. `initial_feedback` seeds the per-slice feedback
275/// carried into the first gate agent step (used when a nested slice re-runs
276/// with a prior rejection rationale); top-level callers pass `None`.
277///
278/// See [`run_pipeline`]'s docs for the abort / rewind-to-gate retry model —
279/// the loop body lives here.
280#[allow(clippy::too_many_arguments)]
281async fn run_steps<D>(
282    module: &Arc<IrModule>,
283    steps: &[tau_ir::pipeline::PipelineStep],
284    input: &str,
285    store: &mut OutputStore,
286    dispatcher: &Arc<D>,
287    initial_feedback: Option<String>,
288    suspend: Option<&SuspendCtx<'_>>,
289    start_at: usize,
290    initial_attempts: BTreeMap<String, u32>,
291) -> Result<StepsFlow, RuntimeError>
292where
293    D: ToolDispatcher + Send + Sync + 'static,
294{
295    // Per-check attempt counter (1-based on first eval), keyed by check id.
296    // Seeded from `initial_attempts` on a resumed top-level slice so a check's
297    // `max_attempts` budget accumulates across resume boundaries; every
298    // nested slice (Branch/Loop/Parallel body) passes an empty map.
299    let mut attempts: BTreeMap<String, u32> = initial_attempts;
300    // Rejection rationales of failed checks that triggered a rewind, keyed by
301    // the gate step they rewind to, then by check id (issue #471). Injecting
302    // only the entries whose key matches the current step id scopes each
303    // check's feedback to its own gate agent step (never leaking into
304    // unrelated agent steps in the rewound slice). Multiple checks gating one
305    // step keep distinct entries — a pass clears only its own check's entry, a
306    // fresh failure updates only its own — instead of last-writer-wins over a
307    // single shared slot.
308    let mut feedback: BTreeMap<String, BTreeMap<String, String>> = BTreeMap::new();
309    // Loop prior-iteration rationale (from `Loop.until`), which has no
310    // associated check or gate. Injected once, at the first agent step of this
311    // slice, then consumed. Top-level/branch/parallel callers pass `None`.
312    let mut pending_initial_feedback: Option<String> = initial_feedback;
313    // gate-id -> pipeline-step index, so a failed retryable check can rewind
314    // `i` to its gate without re-scanning the pipeline each time.
315    let gate_index: BTreeMap<&str, usize> = steps
316        .iter()
317        .enumerate()
318        .map(|(idx, step)| (step.id.0.as_str(), idx))
319        .collect();
320
321    // Index loop (not `for`) so a failed retryable check can rewind `i` to a
322    // gate step. On the happy path `i` only ever advances by one. `start_at`
323    // is nonzero only on a resumed top-level slice (skips the already-run
324    // prefix up to and including the prior `Suspend` step); every recursive
325    // call passes `0`.
326    let mut i = start_at;
327    while i < steps.len() {
328        let step = &steps[i];
329
330        // Check steps have their own span/event vocabulary and store no
331        // output, so dispatch them before the Agent/Tool/Deterministic path.
332        if let StepRun::Check(check_id) = &step.run {
333            let check =
334                module
335                    .workflow
336                    .checks
337                    .get(check_id)
338                    .ok_or_else(|| RuntimeError::Internal {
339                        message: format!("unknown check {}", check_id.0),
340                    })?;
341
342            let check_span = info_span!(SPAN_PIPELINE_CHECK, id = check_id.0.as_str());
343
344            let (verdict, kind) = match &check.verify {
345                CheckVerify::Goal {
346                    evaluates,
347                    predicate,
348                } => {
349                    let reg = dispatcher.deterministic_registry().ok_or_else(|| {
350                        RuntimeError::Internal {
351                            message: format!("check {} needs a deterministic registry", check_id.0),
352                        }
353                    })?;
354                    // Bind the reader to a local so the `Arc` temporary lives
355                    // across the borrow passed to `evaluate_goal`.
356                    let reader = dispatcher.artifact_reader();
357                    let verdict = evaluate_goal(
358                        evaluates,
359                        predicate,
360                        store,
361                        reader.as_ref().map(|a| a.as_ref()),
362                        reg.as_ref(),
363                    )?;
364                    (verdict, "goal")
365                }
366                CheckVerify::Deliverable {
367                    locus,
368                    must_satisfy,
369                    judge,
370                } => {
371                    let reader = dispatcher.artifact_reader();
372                    let verdict = Box::pin(evaluate_deliverable(
373                        module.clone(),
374                        locus,
375                        must_satisfy,
376                        judge,
377                        store,
378                        reader.as_ref().map(|a| a.as_ref()),
379                        dispatcher.clone(),
380                    ))
381                    .instrument(check_span.clone())
382                    .await?;
383                    (verdict, "deliverable")
384                }
385            };
386
387            // Real (1-based) attempt count for this check.
388            let attempt = attempts.get(&check_id.0).copied().unwrap_or(0) + 1;
389            attempts.insert(check_id.0.clone(), attempt);
390
391            tracing::info!(
392                parent: &check_span,
393                name = EV_CHECK_EVALUATED,
394                id = check_id.0.as_str(),
395                kind = kind,
396                verdict = if verdict.met { "pass" } else { "fail" },
397                attempt = attempt,
398            );
399
400            if verdict.met {
401                // Passed: clear only THIS check's feedback under its gate,
402                // leaving any sibling checks' outstanding rationales intact.
403                // Checks store no output.
404                if let Some(inner) = feedback.get_mut(&check.retry.gate.0) {
405                    inner.remove(&check_id.0);
406                    if inner.is_empty() {
407                        feedback.remove(&check.retry.gate.0);
408                    }
409                }
410                i += 1;
411                continue;
412            }
413
414            // Failed. Abort if the policy says so, or if we've exhausted the
415            // attempt budget.
416            if check.retry.on_fail == OnFail::Abort || attempt >= check.retry.max_attempts {
417                return Err(RuntimeError::CheckFailed {
418                    id: check_id.0.clone(),
419                    kind: String::from(kind),
420                    rationale: verdict.rationale,
421                    attempt,
422                });
423            }
424
425            // Retryable with attempts remaining: rewind to the gate step.
426            let gate_idx = *gate_index.get(check.retry.gate.0.as_str()).ok_or_else(|| {
427                // Build-time integrity checks (tau-ir typecheck) guarantee the
428                // gate names an existing pipeline step, so reaching here is an
429                // invariant violation — surface it loudly rather than aborting
430                // the run as a plain check failure.
431                RuntimeError::Internal {
432                    message: format!(
433                        "check {} rewinds to unknown gate step {}",
434                        check_id.0, check.retry.gate.0
435                    ),
436                }
437            })?;
438
439            tracing::info!(
440                parent: &check_span,
441                name = EV_CHECK_RETRY,
442                id = check_id.0.as_str(),
443                rewind_to = check.retry.gate.0.as_str(),
444                next_attempt = attempt + 1,
445            );
446
447            feedback
448                .entry(check.retry.gate.0.clone())
449                .or_default()
450                .insert(check_id.0.clone(), verdict.rationale);
451            i = gate_idx;
452            continue;
453        }
454
455        // `Branch` blocks have their own early dispatch, mirroring `Check`
456        // above: they store no output of their own (the chosen arm's steps
457        // record their own outputs into the shared `store` as they run).
458        if let StepRun::Branch {
459            on,
460            then,
461            otherwise,
462        } = &step.run
463        {
464            let reg =
465                dispatcher
466                    .deterministic_registry()
467                    .ok_or_else(|| RuntimeError::Internal {
468                        message: format!("branch {} needs a deterministic registry", step.id.0),
469                    })?;
470            let reader = dispatcher.artifact_reader();
471            let verdict = crate::interpreter::check::eval_condition(
472                on,
473                store,
474                reader.as_ref().map(|a| a.as_ref()),
475                reg.as_ref(),
476            )?;
477            let arm = if verdict.met { then } else { otherwise };
478            match Box::pin(run_steps(
479                module,
480                arm,
481                input,
482                store,
483                dispatcher,
484                None,
485                None,
486                0,
487                BTreeMap::new(),
488            ))
489            .await?
490            {
491                StepsFlow::Completed => {}
492                StepsFlow::Suspended { .. } => {
493                    return Err(RuntimeError::Internal {
494                        message: "suspend escaped a nested slice (typecheck should reject)"
495                            .to_string(),
496                    })
497                }
498            }
499            i += 1;
500            continue;
501        }
502
503        // `Loop` blocks have their own early dispatch, mirroring `Check` and
504        // `Branch` above: they store no output of their own (the body's
505        // steps record their own outputs into the shared `store` as they
506        // run, each pass overwriting the prior pass's). Runs `body` up to
507        // `max_iters` times, checking `until` after each pass; converges as
508        // soon as `until` holds, otherwise threads the failed verdict's
509        // rationale into the next pass as feedback (same
510        // "Previous attempt rejected: <rationale>" idiom `Check`'s
511        // rewind-to-gate retry uses). Exhausting `max_iters` without `until`
512        // holding is a hard error (ADR-0058 / ADR-0059) — a bounded loop
513        // that cannot reach its goal is a failure, not a silent success.
514        if let StepRun::Loop {
515            body,
516            until,
517            max_iters,
518        } = &step.run
519        {
520            let mut loop_feedback: Option<String> = None;
521            let mut converged = false;
522            for _iter in 0..*max_iters {
523                match Box::pin(run_steps(
524                    module,
525                    body,
526                    input,
527                    store,
528                    dispatcher,
529                    loop_feedback.take(),
530                    None,
531                    0,
532                    BTreeMap::new(),
533                ))
534                .await?
535                {
536                    StepsFlow::Completed => {}
537                    StepsFlow::Suspended { .. } => {
538                        return Err(RuntimeError::Internal {
539                            message: "suspend escaped a nested slice (typecheck should reject)"
540                                .to_string(),
541                        })
542                    }
543                }
544                let reg =
545                    dispatcher
546                        .deterministic_registry()
547                        .ok_or_else(|| RuntimeError::Internal {
548                            message: format!("loop {} needs a deterministic registry", step.id.0),
549                        })?;
550                let reader = dispatcher.artifact_reader();
551                let verdict = crate::interpreter::check::eval_condition(
552                    until,
553                    store,
554                    reader.as_ref().map(|a| a.as_ref()),
555                    reg.as_ref(),
556                )?;
557                if verdict.met {
558                    converged = true;
559                    break;
560                }
561                loop_feedback = Some(verdict.rationale);
562            }
563            if !converged {
564                return Err(RuntimeError::LoopExhausted {
565                    step: step.id.0.clone(),
566                    max_iters: *max_iters,
567                });
568            }
569            i += 1;
570            continue;
571        }
572
573        // `Suspend` blocks have their own early dispatch, mirroring `Branch`,
574        // `Loop`, and `Parallel` above. Only the top-level slice carries a
575        // `SuspendCtx` (nested Branch/Loop/Parallel bodies pass `None` —
576        // typecheck rejects a `Suspend` nested inside those, so `None` here
577        // is an invariant violation, not a normal case). With a ctx: persist
578        // the pause (run id, resume signal, step cursor, IR digest, and a
579        // snapshot of the store so far) and unwind to the caller as
580        // `StepsFlow::Suspended` rather than erroring.
581        if let StepRun::Suspend { resume_signal } = &step.run {
582            match suspend {
583                Some(ctx) => {
584                    ctx.store
585                        .persist_suspension(&tau_ports::orchestration::PipelineSuspension {
586                            run_id: ctx.run_id.to_string(),
587                            resume_signal: resume_signal.clone(),
588                            step_cursor: i,
589                            step_id: step.id.0.clone(),
590                            ir_digest: ctx.ir_digest.to_string(),
591                            outputs: store.snapshot(),
592                            attempts: attempts.clone(),
593                        })
594                        .map_err(|e| RuntimeError::Internal {
595                            message: format!("persist suspension: {e}"),
596                        })?;
597                    return Ok(StepsFlow::Suspended {
598                        step_id: step.id.0.clone(),
599                        resume_signal: resume_signal.clone(),
600                    });
601                }
602                None => {
603                    return Err(RuntimeError::SuspendUnsupported {
604                        step: step.id.0.clone(),
605                        resume_signal: resume_signal.clone(),
606                    })
607                }
608            }
609        }
610
611        // `Parallel` blocks have their own early dispatch, mirroring `Branch`
612        // and `Loop` above: they store no output of their own. Each branch
613        // forks a read-only snapshot of `store` (branches are read-isolated
614        // from each other — a branch cannot see a sibling's outputs, only
615        // the pre-fork state), runs to completion, and returns its produced
616        // store. Branches are driven with bounded concurrency
617        // (`PARALLEL_CAP` in flight at once, nginx `worker_connections`
618        // style) and their outputs are merged back into the shared store in
619        // index order once all branches complete, so the merge is
620        // deterministic regardless of which branch's future resolves first.
621        if let StepRun::Parallel { branches } = &step.run {
622            use futures_util::stream::{self, StreamExt};
623            let futs = branches.iter().enumerate().map(|(idx, branch)| {
624                let module = module.clone();
625                let dispatcher = dispatcher.clone();
626                let input = input.to_string();
627                let mut snap = store.clone();
628                async move {
629                    match run_steps(
630                        &module,
631                        branch,
632                        &input,
633                        &mut snap,
634                        &dispatcher,
635                        None,
636                        None,
637                        0,
638                        BTreeMap::new(),
639                    )
640                    .await?
641                    {
642                        StepsFlow::Completed => {}
643                        StepsFlow::Suspended { .. } => {
644                            return Err(RuntimeError::Internal {
645                                message: "suspend escaped a nested slice (typecheck should reject)"
646                                    .to_string(),
647                            })
648                        }
649                    }
650                    Ok::<(usize, OutputStore), RuntimeError>((idx, snap))
651                }
652            });
653            let mut results: alloc::vec::Vec<(usize, OutputStore)> = stream::iter(futs)
654                .buffered(PARALLEL_CAP)
655                .collect::<alloc::vec::Vec<_>>()
656                .await
657                .into_iter()
658                .collect::<Result<_, _>>()?;
659            // `buffered` preserves input order, but sort by index defensively
660            // so the merge is deterministic regardless of driver scheduling.
661            results.sort_by_key(|(idx, _)| *idx);
662            for (_, produced) in results {
663                store.merge(produced);
664            }
665            i += 1;
666            continue;
667        }
668
669        // NOTE: we do NOT call `.entered()` here — `EnteredSpan` mutates a
670        // thread-local span stack, and tokio's multi-thread scheduler can
671        // move this task to a different worker thread at any `.await`,
672        // leaving the guard on the wrong thread and mis-parenting child
673        // spans/events. Instead every event uses `parent: &step_span`
674        // explicitly, and every awaited future is wrapped with
675        // `.instrument(step_span.clone())`. See stream.rs:273-283 for the
676        // same idiom applied to the runtime turn span.
677        let step_span = info_span!(SPAN_PIPELINE_STEP, id = step.id.0.as_str());
678        tracing::info!(parent: &step_span, name = EV_PIPELINE_STEP_STARTED, id = step.id.0.as_str());
679
680        let rendered = tau_ir::template::resolve(&step.input, input, &store.template_map())
681            .map_err(|e| RuntimeError::Internal {
682                message: format!("pipeline step {}: {e}", step.id.0),
683            })?;
684
685        let output: Value = match &step.run {
686            StepRun::Agent(agent_id) => {
687                let agent = module
688                    .workflow
689                    .agents
690                    .get(agent_id)
691                    .ok_or_else(|| RuntimeError::AgentNotFound {
692                        agent: agent_id.0.clone(),
693                    })?
694                    .clone();
695                // When a prior check rewound to this gate, inject its
696                // rationale as a prior turn so the agent sees *why* it was
697                // rejected. `split_history` (agent_loop.rs) treats all-but-last
698                // as history and the last as the live turn, so each feedback
699                // message becomes a prior turn and `rendered` stays the live
700                // prompt.
701                let mut initial: alloc::vec::Vec<Message> = alloc::vec::Vec::new();
702                // Loop prior-iteration feedback (no check/gate): inject once,
703                // at the first agent step of the slice, then consume it.
704                if let Some(fb) = pending_initial_feedback.take() {
705                    initial.push(user_message(&format!("Previous attempt rejected: {fb}")));
706                }
707                // Check feedback gating THIS step: one labeled prior turn per
708                // outstanding check, so several checks on one gate stay
709                // distinct (issue #471). BTreeMap iteration is sorted by check
710                // id, so ordering is deterministic.
711                if let Some(inner) = feedback.get(&step.id.0) {
712                    for (cid, fb) in inner {
713                        initial.push(user_message(&format!(
714                            "Previous attempt rejected: (check '{cid}') {fb}"
715                        )));
716                    }
717                }
718                initial.push(user_message(&rendered));
719                let outcome = Box::pin(run_agent(
720                    module.clone(),
721                    &agent,
722                    dispatcher.clone(),
723                    initial,
724                ))
725                .instrument(step_span.clone())
726                .await?;
727                match outcome {
728                    RunOutcome::Failed { status, .. } => {
729                        return Err(RuntimeError::Internal {
730                            message: format!(
731                                "pipeline step {} (agent {}) failed: {status:?}",
732                                step.id.0, agent_id.0
733                            ),
734                        })
735                    }
736                    _ => Value::String(last_assistant_text(&outcome)),
737                }
738            }
739            StepRun::Tool(tool_id) => {
740                let args = rendered_to_args(&rendered);
741                let result = dispatcher
742                    .invoke(tool_id, &args)
743                    .instrument(step_span.clone())
744                    .await?;
745                match (result.body, result.error) {
746                    (Some(body), _) => body,
747                    (None, Some(err)) => {
748                        return Err(RuntimeError::Internal {
749                            message: alloc::format!(
750                                "pipeline step {} (tool {}) errored: {err}",
751                                step.id.0,
752                                tool_id.0
753                            ),
754                        })
755                    }
756                    (None, None) => Value::Null,
757                }
758            }
759            StepRun::Deterministic(step_node_id) => {
760                let registry =
761                    dispatcher
762                        .deterministic_registry()
763                        .ok_or_else(|| RuntimeError::Internal {
764                            message: alloc::format!(
765                                "pipeline step {} needs a deterministic registry, none provided",
766                                step.id.0
767                            ),
768                        })?;
769                let node = module.workflow.steps.get(step_node_id).ok_or_else(|| {
770                    RuntimeError::Internal {
771                        message: alloc::format!("unknown deterministic step {}", step_node_id.0),
772                    }
773                })?;
774                let args = rendered_to_args(&rendered);
775                crate::interpreter::deterministic::run_step(node, registry.as_ref(), &args)?
776            }
777            // `Check` steps are dispatched at the top of the loop and never
778            // reach this `match` (they store no output of their own).
779            StepRun::Check(_) => unreachable!("check steps are handled before this match"),
780            // `Branch`, `Parallel`, `Loop`, and `Suspend` are all
781            // early-dispatched above (each either stores no output of its
782            // own, or — for `Suspend` — returns `StepsFlow::Suspended`/errors
783            // before reaching here), so none of these ever reach this
784            // `match`.
785            StepRun::Branch { .. }
786            | StepRun::Parallel { .. }
787            | StepRun::Loop { .. }
788            | StepRun::Suspend { .. } => {
789                unreachable!("control-flow blocks are early-dispatched")
790            }
791        };
792
793        store.insert(step.id.0.clone(), output);
794        tracing::info!(parent: &step_span, name = EV_PIPELINE_STEP_COMPLETED, id = step.id.0.as_str());
795
796        i += 1;
797    }
798
799    Ok(StepsFlow::Completed)
800}

802/// Turn a rendered template string into the `Value` a tool/deterministic
803/// step receives: parse it as JSON if it parses, else wrap as a string.
804///
805/// Footgun: a rendered string that happens to be a bare JSON scalar
806/// (`42`, `true`, `null`) parses to that scalar rather than wrapping as a
807/// string. Author tool/deterministic `input` templates accordingly.
808fn rendered_to_args(rendered: &str) -> Value {
809    serde_json::from_str::<Value>(rendered).unwrap_or_else(|_| Value::String(rendered.to_string()))
810}

812/// Build a user-turn [`Message`] carrying `content` as its text payload.
813///
814/// Mirrors the initial-message idiom in `tau-cli`'s `run` command: the
815/// recipient is a freshly-minted [`AgentInstanceId`] placeholder that the
816/// kernel replaces when it assigns the loop's own instance id.
817///
818/// Exposed as `pub(crate)` so `check.rs` can reuse it without a separate
819/// definition (single source of truth).
820pub(crate) fn user_message(content: &str) -> Message {
821    // Use no_std-safe constructors (std-gated `Message::new` and
822    // `AgentInstanceId::new` aren't available in the wasm-interpreter
823    // build). The kernel replaces the agent instance id when it builds
824    // the inner agent loop, so the placeholder zeros are ephemeral.
825    use tau_domain::MessageId;
826    Message::new_with(
827        MessageId::from_parts(0, [0u8; 10]),
828        chrono::DateTime::<chrono::Utc>::from_timestamp_millis(0).unwrap(),
829        Address::User,
830        Address::Agent(AgentInstanceId::from_parts(0, [0u8; 10])),
831        MessagePayload::Text {
832            content: content.to_string(),
833        },
834    )
835}
