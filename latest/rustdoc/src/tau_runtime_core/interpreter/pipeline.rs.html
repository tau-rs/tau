32/// Max `Parallel` branches in flight at once (nginx `worker_connections`
33/// analogue). Bounded cooperative fork-join — see ADR-0059.
34const PARALLEL_CAP: usize = 8;

36/// Terminal outcome of driving a pipeline via [`run_pipeline_suspendable`].
37#[derive(Debug)]
38pub enum PipelineOutcome {
39    /// Ran to the end; carries the accumulated outputs.
40    Completed(OutputStore),
41    /// Paused at a top-level `Suspend`. State was persisted via the
42    /// `SuspensionStore` passed in `SuspendConfig` before returning.
43    Suspended {
44        /// The run id the suspension was persisted under.
45        run_id: String,
46        /// The signal name a `--resume --signal <name>` must supply.
47        resume_signal: String,
48        /// The pipeline-step id of the `Suspend` block that paused.
49        step_id: String,
50    },
51}

53/// How a `run_steps` slice terminated. Only the top-level slice (the one
54/// `run_pipeline_suspendable` invokes directly, with `suspend: Some(_)`) can
55/// terminate `Suspended`; nested slices (`Branch`/`Loop`/`Parallel` bodies)
56/// always pass `suspend: None` and must complete or error.
57enum StepsFlow {
58    /// The slice ran every step.
59    Completed,
60    /// The slice paused at a `Suspend` step.
61    Suspended {
62        /// The pipeline-step id of the `Suspend` block.
63        step_id: String,
64        /// The signal name the resumer must supply.
65        resume_signal: String,
66    },
67}

69/// Durable-suspend wiring for the top-level pipeline driver. Carries the
70/// run id (the suspension key + the `--resume` handle) and the store a
71/// `Suspend` step's snapshot is persisted to.
72pub struct SuspendConfig {
73    /// Run id used as the suspension key + `--resume` handle.
74    pub run_id: String,
75    /// Where a pause is durably persisted.
76    pub store: Arc<dyn tau_ports::orchestration::SuspensionStore>,
77}

79/// Rehydrated state for a resumed run: the `OutputStore` snapshot restored
80/// from the prior suspension, and the top-level step index to resume at
81/// (`step_cursor + 1`, so the `Suspend` step itself is not re-run).
82pub struct ResumeState {
83    /// The `OutputStore` snapshot restored from the suspension.
84    pub store: OutputStore,
85    /// Top-level step index to resume at (`step_cursor + 1`).
86    pub start_at: usize,
87    /// Per-check attempt counts restored from the prior suspension, so a
88    /// check's `max_attempts` budget accumulates across resume boundaries
89    /// instead of resetting each resume. Fresh runs pass an empty map.
90    pub attempts: BTreeMap<String, u32>,
91}

93/// Internal: passed down to `run_steps` for the top-level slice only. A
94/// nested slice (Branch/Loop/Parallel body) receives `None`, which is how
95/// `run_steps` tells a real `Suspend` step from one that is out of place.
96struct SuspendCtx<'a> {
97    run_id: &'a str,
98    ir_digest: &'a str,
99    store: &'a dyn tau_ports::orchestration::SuspensionStore,
100}

102/// Drive an `IrModule`'s pipeline, pausing durably at a top-level `Suspend`
103/// step rather than erroring.
104///
105/// The module's `workflow.pipeline` must be `Some` — callers branch on
106/// that before dispatching here (see `tau run`). Each step renders its
107/// `input` template against the run `input` and prior outputs, runs the
108/// step, and records the step's output keyed by its pipeline-step id so
109/// later steps can reference it via `${steps.<id>.output}`.
110///
111/// [`StepRun::Agent`], [`StepRun::Tool`], [`StepRun::Deterministic`],
112/// [`StepRun::Check`], [`StepRun::Branch`], [`StepRun::Loop`],
113/// [`StepRun::Parallel`], and [`StepRun::Suspend`] steps are all supported.
114/// [`StepRun::Dynamic`] is recognized but not yet executed: it errors
115/// [`RuntimeError::DynamicRegionRequiresRuntimeGate`] pending the EPIC 4.5
116/// runtime gate (membership, attenuation, bounds counters). A
117/// `Check` step evaluates a postcondition (goal or deliverable) against the
118/// accumulated outputs. A `Branch` evaluates its condition and recurses into
119/// the chosen arm (`then`/`otherwise`) against the same shared store; it
120/// stores no output of its own. A `Loop` runs its `body` up to `max_iters`
121/// times, checking `until` after each pass and threading a failed verdict's
122/// rationale into the next pass as feedback; exhausting `max_iters` without
123/// `until` holding is a hard error ([`RuntimeError::LoopExhausted`]). Like
124/// `Branch`, a `Loop` stores no output of its own. A `Parallel` forks each
125/// branch over an isolated read-only snapshot of the store (branches cannot
126/// see each other's outputs), drives them with bounded concurrency
127/// (`PARALLEL_CAP` in flight at once), and merges each branch's produced
128/// outputs back into the shared store in index order once all branches
129/// complete; like `Branch` and `Loop`, it stores no output of its own. A
130/// top-level `Suspend` step pauses the run: computes the module's
131/// canonical-IR digest once (for drift-detection on a future resume), seeds
132/// the run's `OutputStore` and start index either fresh (`resume: None`) or
133/// from a prior pause (`resume: Some(ResumeState)` — restore-and-continue:
134/// the store is rehydrated from the persisted snapshot and the walk resumes
135/// at `step_cursor + 1`, so the prefix up to and including the `Suspend`
136/// step is never re-run), and — on hitting `Suspend` — durably persists a
137/// [`tau_ports::orchestration::PipelineSuspension`] via the `SuspensionStore`
138/// in `SuspendConfig` and returns [`PipelineOutcome::Suspended`] rather than
139/// erroring. Only a top-level `Suspend` may pause; one nested inside a
140/// `Branch`/`Loop`/`Parallel` body is a build-time typecheck error (an
141/// [`RuntimeError::Internal`] here would mean that check regressed).
142///
143/// # Failure handling: abort, or rewind-to-gate retry
144///
145/// On a failed check the [`RetryPolicy`](tau_ir::check::RetryPolicy)
146/// decides what happens. If `on_fail == Abort`, or the attempt count has
147/// reached `max_attempts`, the pipeline aborts with
148/// [`RuntimeError::CheckFailed`]. Otherwise the loop emits
149/// [`EV_CHECK_RETRY`], stashes the judge's rationale as `feedback` keyed by
150/// the check's `gate` step and the check's id, and rewinds the loop index to
151/// that `gate` step so the forward slice re-runs. On the next pass through the
152/// gate's agent step every outstanding rationale registered for *that* gate is
153/// injected as a labelled prior turn (`"Previous attempt rejected: (check
154/// '<id>') <rationale>"`) so the agent sees *why* it was rejected. Keying by
155/// gate + check id keeps concurrently-failing checks distinct (a pass clears
156/// only its own check's entry; a fresh failure updates only its own) and
157/// confines each check's feedback to its own gate step.
158pub async fn run_pipeline_suspendable<D>(
159    module: Arc<IrModule>,
160    input: String,
161    dispatcher: Arc<D>,
162    suspend: SuspendConfig,
163    resume: Option<ResumeState>,
164) -> Result<PipelineOutcome, RuntimeError>
165where
166    D: ToolDispatcher + Send + Sync + 'static,
167{
168    let pipeline = module
169        .workflow
170        .pipeline
171        .clone()
172        .ok_or_else(|| RuntimeError::Internal {
173            message: "run_pipeline called on a module without a pipeline".to_string(),
174        })?;
175
176    // Canonical-IR digest, computed once, for drift-detection on resume.
177    let ir_digest = {
178        let bytes = tau_ir::to_canonical_bytes(&module);
179        tau_ir::asset::asset_hash(&bytes)
180    };
181
182    let (mut store, start_at, initial_attempts) = match resume {
183        Some(r) => (r.store, r.start_at, r.attempts),
184        None => (OutputStore::new(), 0, BTreeMap::new()),
185    };
186
187    let ctx = SuspendCtx {
188        run_id: &suspend.run_id,
189        ir_digest: &ir_digest,
190        store: suspend.store.as_ref(),
191    };
192    let flow = run_steps(
193        &module,
194        &pipeline.steps,
195        &input,
196        &mut store,
197        &dispatcher,
198        None,
199        Some(&ctx),
200        start_at,
201        initial_attempts,
202    )
203    .await?;
204    Ok(match flow {
205        StepsFlow::Completed => PipelineOutcome::Completed(store),
206        StepsFlow::Suspended {
207            step_id,
208            resume_signal,
209        } => PipelineOutcome::Suspended {
210            run_id: suspend.run_id,
211            resume_signal,
212            step_id,
213        },
214    })
215}

217/// Non-suspend convenience wrapper over [`run_pipeline_suspendable`] for
218/// callers that cannot pause (e.g. the bundle-run path in v1). Wires a
219/// no-op `SuspensionStore` and runs fresh (never resumes); a pipeline that
220/// hits a `Suspend` step errors [`RuntimeError::SuspendUnsupported`] instead
221/// of pausing. Callers that support HITL suspend/resume use
222/// `run_pipeline_suspendable` directly.
223pub async fn run_pipeline<D>(
224    module: Arc<IrModule>,
225    input: String,
226    dispatcher: Arc<D>,
227) -> Result<OutputStore, RuntimeError>
228where
229    D: ToolDispatcher + Send + Sync + 'static,
230{
231    struct NoopSuspensions;
232    impl tau_ports::orchestration::SuspensionStore for NoopSuspensions {
233        fn persist_suspension(
234            &self,
235            _: &tau_ports::orchestration::PipelineSuspension,
236        ) -> Result<(), tau_ports::orchestration::CheckpointError> {
237            Ok(())
238        }
239        fn load_suspension(
240            &self,
241            _: &tau_ports::orchestration::RunId,
242        ) -> Result<
243            Option<tau_ports::orchestration::PipelineSuspension>,
244            tau_ports::orchestration::CheckpointError,
245        > {
246            Ok(None)
247        }
248    }
249    let outcome = run_pipeline_suspendable(
250        module,
251        input,
252        dispatcher,
253        SuspendConfig {
254            run_id: String::new(),
255            store: Arc::new(NoopSuspensions),
256        },
257        None,
258    )
259    .await?;
260    match outcome {
261        PipelineOutcome::Completed(store) => Ok(store),
262        PipelineOutcome::Suspended {
263            step_id,
264            resume_signal,
265            ..
266        } => Err(RuntimeError::SuspendUnsupported {
267            step: step_id,
268            resume_signal,
269        }),
270    }
271}

273/// Execute a slice of pipeline steps against a shared [`OutputStore`].
274///
275/// Extracted from [`run_pipeline`] so nested blocks (Branch/Parallel/Loop)
276/// can drive their own step slices with the same gate-rewind, feedback, and
277/// check-dispatch semantics. `initial_feedback` seeds the per-slice feedback
278/// carried into the first gate agent step (used when a nested slice re-runs
279/// with a prior rejection rationale); top-level callers pass `None`.
280///
281/// See [`run_pipeline`]'s docs for the abort / rewind-to-gate retry model —
282/// the loop body lives here.
283#[allow(clippy::too_many_arguments)]
284async fn run_steps<D>(
285    module: &Arc<IrModule>,
286    steps: &[tau_ir::pipeline::PipelineStep],
287    input: &str,
288    store: &mut OutputStore,
289    dispatcher: &Arc<D>,
290    initial_feedback: Option<String>,
291    suspend: Option<&SuspendCtx<'_>>,
292    start_at: usize,
293    initial_attempts: BTreeMap<String, u32>,
294) -> Result<StepsFlow, RuntimeError>
295where
296    D: ToolDispatcher + Send + Sync + 'static,
297{
298    // Per-check attempt counter (1-based on first eval), keyed by check id.
299    // Seeded from `initial_attempts` on a resumed top-level slice so a check's
300    // `max_attempts` budget accumulates across resume boundaries; every
301    // nested slice (Branch/Loop/Parallel body) passes an empty map.
302    let mut attempts: BTreeMap<String, u32> = initial_attempts;
303    // Rejection rationales of failed checks that triggered a rewind, keyed by
304    // the gate step they rewind to, then by check id (issue #471). Injecting
305    // only the entries whose key matches the current step id scopes each
306    // check's feedback to its own gate agent step (never leaking into
307    // unrelated agent steps in the rewound slice). Multiple checks gating one
308    // step keep distinct entries — a pass clears only its own check's entry, a
309    // fresh failure updates only its own — instead of last-writer-wins over a
310    // single shared slot.
311    let mut feedback: BTreeMap<String, BTreeMap<String, String>> = BTreeMap::new();
312    // Loop prior-iteration rationale (from `Loop.until`), which has no
313    // associated check or gate. Injected once, at the first agent step of this
314    // slice, then consumed. Top-level/branch/parallel callers pass `None`.
315    let mut pending_initial_feedback: Option<String> = initial_feedback;
316    // gate-id -> pipeline-step index, so a failed retryable check can rewind
317    // `i` to its gate without re-scanning the pipeline each time.
318    let gate_index: BTreeMap<&str, usize> = steps
319        .iter()
320        .enumerate()
321        .map(|(idx, step)| (step.id.0.as_str(), idx))
322        .collect();
323
324    // Index loop (not `for`) so a failed retryable check can rewind `i` to a
325    // gate step. On the happy path `i` only ever advances by one. `start_at`
326    // is nonzero only on a resumed top-level slice (skips the already-run
327    // prefix up to and including the prior `Suspend` step); every recursive
328    // call passes `0`.
329    let mut i = start_at;
330    while i < steps.len() {
331        let step = &steps[i];
332
333        // Check steps have their own span/event vocabulary and store no
334        // output, so dispatch them before the Agent/Tool/Deterministic path.
335        if let StepRun::Check(check_id) = &step.run {
336            let check =
337                module
338                    .workflow
339                    .checks
340                    .get(check_id)
341                    .ok_or_else(|| RuntimeError::Internal {
342                        message: format!("unknown check {}", check_id.0),
343                    })?;
344
345            let check_span = info_span!(SPAN_PIPELINE_CHECK, id = check_id.0.as_str());
346
347            let (verdict, kind) = match &check.verify {
348                CheckVerify::Goal {
349                    evaluates,
350                    predicate,
351                } => {
352                    let reg = dispatcher.deterministic_registry().ok_or_else(|| {
353                        RuntimeError::Internal {
354                            message: format!("check {} needs a deterministic registry", check_id.0),
355                        }
356                    })?;
357                    // Bind the reader to a local so the `Arc` temporary lives
358                    // across the borrow passed to `evaluate_goal`.
359                    let reader = dispatcher.artifact_reader();
360                    let verdict = evaluate_goal(
361                        evaluates,
362                        predicate,
363                        store,
364                        reader.as_ref().map(|a| a.as_ref()),
365                        reg.as_ref(),
366                    )?;
367                    (verdict, "goal")
368                }
369                CheckVerify::Deliverable {
370                    locus,
371                    must_satisfy,
372                    judge,
373                } => {
374                    let reader = dispatcher.artifact_reader();
375                    let verdict = Box::pin(evaluate_deliverable(
376                        module.clone(),
377                        locus,
378                        must_satisfy,
379                        judge,
380                        store,
381                        reader.as_ref().map(|a| a.as_ref()),
382                        dispatcher.clone(),
383                    ))
384                    .instrument(check_span.clone())
385                    .await?;
386                    (verdict, "deliverable")
387                }
388            };
389
390            // Real (1-based) attempt count for this check.
391            let attempt = attempts.get(&check_id.0).copied().unwrap_or(0) + 1;
392            attempts.insert(check_id.0.clone(), attempt);
393
394            tracing::info!(
395                parent: &check_span,
396                name = EV_CHECK_EVALUATED,
397                id = check_id.0.as_str(),
398                kind = kind,
399                verdict = if verdict.met { "pass" } else { "fail" },
400                attempt = attempt,
401            );
402
403            if verdict.met {
404                // Passed: clear only THIS check's feedback under its gate,
405                // leaving any sibling checks' outstanding rationales intact.
406                // Checks store no output.
407                if let Some(inner) = feedback.get_mut(&check.retry.gate.0) {
408                    inner.remove(&check_id.0);
409                    if inner.is_empty() {
410                        feedback.remove(&check.retry.gate.0);
411                    }
412                }
413                i += 1;
414                continue;
415            }
416
417            // Failed. Abort if the policy says so, or if we've exhausted the
418            // attempt budget.
419            if check.retry.on_fail == OnFail::Abort || attempt >= check.retry.max_attempts {
420                return Err(RuntimeError::CheckFailed {
421                    id: check_id.0.clone(),
422                    kind: String::from(kind),
423                    rationale: verdict.rationale,
424                    attempt,
425                });
426            }
427
428            // Retryable with attempts remaining: rewind to the gate step.
429            let gate_idx = *gate_index.get(check.retry.gate.0.as_str()).ok_or_else(|| {
430                // Build-time integrity checks (tau-ir typecheck) guarantee the
431                // gate names an existing pipeline step, so reaching here is an
432                // invariant violation — surface it loudly rather than aborting
433                // the run as a plain check failure.
434                RuntimeError::Internal {
435                    message: format!(
436                        "check {} rewinds to unknown gate step {}",
437                        check_id.0, check.retry.gate.0
438                    ),
439                }
440            })?;
441
442            tracing::info!(
443                parent: &check_span,
444                name = EV_CHECK_RETRY,
445                id = check_id.0.as_str(),
446                rewind_to = check.retry.gate.0.as_str(),
447                next_attempt = attempt + 1,
448            );
449
450            feedback
451                .entry(check.retry.gate.0.clone())
452                .or_default()
453                .insert(check_id.0.clone(), verdict.rationale);
454            i = gate_idx;
455            continue;
456        }
457
458        // `Branch` blocks have their own early dispatch, mirroring `Check`
459        // above: they store no output of their own (the chosen arm's steps
460        // record their own outputs into the shared `store` as they run).
461        if let StepRun::Branch {
462            on,
463            then,
464            otherwise,
465        } = &step.run
466        {
467            let reg =
468                dispatcher
469                    .deterministic_registry()
470                    .ok_or_else(|| RuntimeError::Internal {
471                        message: format!("branch {} needs a deterministic registry", step.id.0),
472                    })?;
473            let reader = dispatcher.artifact_reader();
474            let verdict = crate::interpreter::check::eval_condition(
475                on,
476                store,
477                reader.as_ref().map(|a| a.as_ref()),
478                reg.as_ref(),
479            )?;
480            let arm = if verdict.met { then } else { otherwise };
481            match Box::pin(run_steps(
482                module,
483                arm,
484                input,
485                store,
486                dispatcher,
487                None,
488                None,
489                0,
490                BTreeMap::new(),
491            ))
492            .await?
493            {
494                StepsFlow::Completed => {}
495                StepsFlow::Suspended { .. } => {
496                    return Err(RuntimeError::Internal {
497                        message: "suspend escaped a nested slice (typecheck should reject)"
498                            .to_string(),
499                    })
500                }
501            }
502            i += 1;
503            continue;
504        }
505
506        // `Loop` blocks have their own early dispatch, mirroring `Check` and
507        // `Branch` above: they store no output of their own (the body's
508        // steps record their own outputs into the shared `store` as they
509        // run, each pass overwriting the prior pass's). Runs `body` up to
510        // `max_iters` times, checking `until` after each pass; converges as
511        // soon as `until` holds, otherwise threads the failed verdict's
512        // rationale into the next pass as feedback (same
513        // "Previous attempt rejected: <rationale>" idiom `Check`'s
514        // rewind-to-gate retry uses). Exhausting `max_iters` without `until`
515        // holding is a hard error (ADR-0058 / ADR-0059) — a bounded loop
516        // that cannot reach its goal is a failure, not a silent success.
517        if let StepRun::Loop {
518            body,
519            until,
520            max_iters,
521        } = &step.run
522        {
523            let mut loop_feedback: Option<String> = None;
524            let mut converged = false;
525            for _iter in 0..*max_iters {
526                match Box::pin(run_steps(
527                    module,
528                    body,
529                    input,
530                    store,
531                    dispatcher,
532                    loop_feedback.take(),
533                    None,
534                    0,
535                    BTreeMap::new(),
536                ))
537                .await?
538                {
539                    StepsFlow::Completed => {}
540                    StepsFlow::Suspended { .. } => {
541                        return Err(RuntimeError::Internal {
542                            message: "suspend escaped a nested slice (typecheck should reject)"
543                                .to_string(),
544                        })
545                    }
546                }
547                let reg =
548                    dispatcher
549                        .deterministic_registry()
550                        .ok_or_else(|| RuntimeError::Internal {
551                            message: format!("loop {} needs a deterministic registry", step.id.0),
552                        })?;
553                let reader = dispatcher.artifact_reader();
554                let verdict = crate::interpreter::check::eval_condition(
555                    until,
556                    store,
557                    reader.as_ref().map(|a| a.as_ref()),
558                    reg.as_ref(),
559                )?;
560                if verdict.met {
561                    converged = true;
562                    break;
563                }
564                loop_feedback = Some(verdict.rationale);
565            }
566            if !converged {
567                return Err(RuntimeError::LoopExhausted {
568                    step: step.id.0.clone(),
569                    max_iters: *max_iters,
570                });
571            }
572            i += 1;
573            continue;
574        }
575
576        // `Suspend` blocks have their own early dispatch, mirroring `Branch`,
577        // `Loop`, and `Parallel` above. Only the top-level slice carries a
578        // `SuspendCtx` (nested Branch/Loop/Parallel bodies pass `None` —
579        // typecheck rejects a `Suspend` nested inside those, so `None` here
580        // is an invariant violation, not a normal case). With a ctx: persist
581        // the pause (run id, resume signal, step cursor, IR digest, and a
582        // snapshot of the store so far) and unwind to the caller as
583        // `StepsFlow::Suspended` rather than erroring.
584        if let StepRun::Suspend { resume_signal } = &step.run {
585            match suspend {
586                Some(ctx) => {
587                    ctx.store
588                        .persist_suspension(&tau_ports::orchestration::PipelineSuspension {
589                            run_id: ctx.run_id.to_string(),
590                            resume_signal: resume_signal.clone(),
591                            step_cursor: i,
592                            step_id: step.id.0.clone(),
593                            ir_digest: ctx.ir_digest.to_string(),
594                            outputs: store.snapshot(),
595                            attempts: attempts.clone(),
596                        })
597                        .map_err(|e| RuntimeError::Internal {
598                            message: format!("persist suspension: {e}"),
599                        })?;
600                    return Ok(StepsFlow::Suspended {
601                        step_id: step.id.0.clone(),
602                        resume_signal: resume_signal.clone(),
603                    });
604                }
605                None => {
606                    return Err(RuntimeError::SuspendUnsupported {
607                        step: step.id.0.clone(),
608                        resume_signal: resume_signal.clone(),
609                    })
610                }
611            }
612        }
613
614        // `Dynamic` regions have their own early dispatch, mirroring
615        // `Branch`/`Loop`/`Suspend` above. Real execution (membership,
616        // attenuation, bounds counters) lands in EPIC 4.5; until then the
617        // interpreter meets a `Dynamic` region with a named error rather
618        // than executing or silently skipping it.
619        if let StepRun::Dynamic { .. } = &step.run {
620            return Err(RuntimeError::DynamicRegionRequiresRuntimeGate {
621                step_id: step.id.0.clone(),
622            });
623        }
624
625        // `Parallel` blocks have their own early dispatch, mirroring `Branch`
626        // and `Loop` above: they store no output of their own. Each branch
627        // forks a read-only snapshot of `store` (branches are read-isolated
628        // from each other — a branch cannot see a sibling's outputs, only
629        // the pre-fork state), runs to completion, and returns its produced
630        // store. Branches are driven with bounded concurrency
631        // (`PARALLEL_CAP` in flight at once, nginx `worker_connections`
632        // style) and their outputs are merged back into the shared store in
633        // index order once all branches complete, so the merge is
634        // deterministic regardless of which branch's future resolves first.
635        if let StepRun::Parallel { branches } = &step.run {
636            use futures_util::stream::{self, StreamExt};
637            let futs = branches.iter().enumerate().map(|(idx, branch)| {
638                let module = module.clone();
639                let dispatcher = dispatcher.clone();
640                let input = input.to_string();
641                let mut snap = store.clone();
642                async move {
643                    match run_steps(
644                        &module,
645                        branch,
646                        &input,
647                        &mut snap,
648                        &dispatcher,
649                        None,
650                        None,
651                        0,
652                        BTreeMap::new(),
653                    )
654                    .await?
655                    {
656                        StepsFlow::Completed => {}
657                        StepsFlow::Suspended { .. } => {
658                            return Err(RuntimeError::Internal {
659                                message: "suspend escaped a nested slice (typecheck should reject)"
660                                    .to_string(),
661                            })
662                        }
663                    }
664                    Ok::<(usize, OutputStore), RuntimeError>((idx, snap))
665                }
666            });
667            let mut results: alloc::vec::Vec<(usize, OutputStore)> = stream::iter(futs)
668                .buffered(PARALLEL_CAP)
669                .collect::<alloc::vec::Vec<_>>()
670                .await
671                .into_iter()
672                .collect::<Result<_, _>>()?;
673            // `buffered` preserves input order, but sort by index defensively
674            // so the merge is deterministic regardless of driver scheduling.
675            results.sort_by_key(|(idx, _)| *idx);
676            for (_, produced) in results {
677                store.merge(produced);
678            }
679            i += 1;
680            continue;
681        }
682
683        // NOTE: we do NOT call `.entered()` here — `EnteredSpan` mutates a
684        // thread-local span stack, and tokio's multi-thread scheduler can
685        // move this task to a different worker thread at any `.await`,
686        // leaving the guard on the wrong thread and mis-parenting child
687        // spans/events. Instead every event uses `parent: &step_span`
688        // explicitly, and every awaited future is wrapped with
689        // `.instrument(step_span.clone())`. See stream.rs:273-283 for the
690        // same idiom applied to the runtime turn span.
691        let step_span = info_span!(SPAN_PIPELINE_STEP, id = step.id.0.as_str());
692        tracing::info!(parent: &step_span, name = EV_PIPELINE_STEP_STARTED, id = step.id.0.as_str());
693
694        let rendered = tau_ir::template::resolve(&step.input, input, &store.template_map())
695            .map_err(|e| RuntimeError::Internal {
696                message: format!("pipeline step {}: {e}", step.id.0),
697            })?;
698
699        let output: Value = match &step.run {
700            StepRun::Agent(agent_id) => {
701                let agent = module
702                    .workflow
703                    .agents
704                    .get(agent_id)
705                    .ok_or_else(|| RuntimeError::AgentNotFound {
706                        agent: agent_id.0.clone(),
707                    })?
708                    .clone();
709                // When a prior check rewound to this gate, inject its
710                // rationale as a prior turn so the agent sees *why* it was
711                // rejected. `split_history` (agent_loop.rs) treats all-but-last
712                // as history and the last as the live turn, so each feedback
713                // message becomes a prior turn and `rendered` stays the live
714                // prompt.
715                let mut initial: alloc::vec::Vec<Message> = alloc::vec::Vec::new();
716                // Loop prior-iteration feedback (no check/gate): inject once,
717                // at the first agent step of the slice, then consume it.
718                if let Some(fb) = pending_initial_feedback.take() {
719                    initial.push(user_message(&format!("Previous attempt rejected: {fb}")));
720                }
721                // Check feedback gating THIS step: one labeled prior turn per
722                // outstanding check, so several checks on one gate stay
723                // distinct (issue #471). BTreeMap iteration is sorted by check
724                // id, so ordering is deterministic.
725                if let Some(inner) = feedback.get(&step.id.0) {
726                    for (cid, fb) in inner {
727                        initial.push(user_message(&format!(
728                            "Previous attempt rejected: (check '{cid}') {fb}"
729                        )));
730                    }
731                }
732                initial.push(user_message(&rendered));
733                let outcome = Box::pin(run_agent(
734                    module.clone(),
735                    &agent,
736                    dispatcher.clone(),
737                    initial,
738                ))
739                .instrument(step_span.clone())
740                .await?;
741                match outcome {
742                    RunOutcome::Failed { status, .. } => {
743                        return Err(RuntimeError::Internal {
744                            message: format!(
745                                "pipeline step {} (agent {}) failed: {status:?}",
746                                step.id.0, agent_id.0
747                            ),
748                        })
749                    }
750                    _ => Value::String(last_assistant_text(&outcome)),
751                }
752            }
753            StepRun::Tool(tool_id) => {
754                let args = rendered_to_args(&rendered);
755                let result = dispatcher
756                    .invoke(tool_id, &args)
757                    .instrument(step_span.clone())
758                    .await?;
759                match (result.body, result.error) {
760                    (Some(body), _) => body,
761                    (None, Some(err)) => {
762                        return Err(RuntimeError::Internal {
763                            message: alloc::format!(
764                                "pipeline step {} (tool {}) errored: {err}",
765                                step.id.0,
766                                tool_id.0
767                            ),
768                        })
769                    }
770                    (None, None) => Value::Null,
771                }
772            }
773            StepRun::Deterministic(step_node_id) => {
774                let registry =
775                    dispatcher
776                        .deterministic_registry()
777                        .ok_or_else(|| RuntimeError::Internal {
778                            message: alloc::format!(
779                                "pipeline step {} needs a deterministic registry, none provided",
780                                step.id.0
781                            ),
782                        })?;
783                let node = module.workflow.steps.get(step_node_id).ok_or_else(|| {
784                    RuntimeError::Internal {
785                        message: alloc::format!("unknown deterministic step {}", step_node_id.0),
786                    }
787                })?;
788                let args = rendered_to_args(&rendered);
789                crate::interpreter::deterministic::run_step(node, registry.as_ref(), &args)?
790            }
791            // `Check` steps are dispatched at the top of the loop and never
792            // reach this `match` (they store no output of their own).
793            StepRun::Check(_) => unreachable!("check steps are handled before this match"),
794            // `Branch`, `Parallel`, `Loop`, `Suspend`, and `Dynamic` are all
795            // early-dispatched above (each either stores no output of its
796            // own, or — for `Suspend`/`Dynamic` — returns
797            // `StepsFlow::Suspended`/errors before reaching here), so none of
798            // these ever reach this `match`.
799            StepRun::Branch { .. }
800            | StepRun::Parallel { .. }
801            | StepRun::Loop { .. }
802            | StepRun::Suspend { .. }
803            | StepRun::Dynamic { .. } => {
804                unreachable!("control-flow blocks are early-dispatched")
805            }
806        };
807
808        store.insert(step.id.0.clone(), output);
809        tracing::info!(parent: &step_span, name = EV_PIPELINE_STEP_COMPLETED, id = step.id.0.as_str());
810
811        i += 1;
812    }
813
814    Ok(StepsFlow::Completed)
815}

817/// Turn a rendered template string into the `Value` a tool/deterministic
818/// step receives: parse it as JSON if it parses, else wrap as a string.
819///
820/// Footgun: a rendered string that happens to be a bare JSON scalar
821/// (`42`, `true`, `null`) parses to that scalar rather than wrapping as a
822/// string. Author tool/deterministic `input` templates accordingly.
823fn rendered_to_args(rendered: &str) -> Value {
824    serde_json::from_str::<Value>(rendered).unwrap_or_else(|_| Value::String(rendered.to_string()))
825}

827/// Build a user-turn [`Message`] carrying `content` as its text payload.
828///
829/// Mirrors the initial-message idiom in `tau-cli`'s `run` command: the
830/// recipient is a freshly-minted [`AgentInstanceId`] placeholder that the
831/// kernel replaces when it assigns the loop's own instance id.
832///
833/// Exposed as `pub(crate)` so `check.rs` can reuse it without a separate
834/// definition (single source of truth).
835pub(crate) fn user_message(content: &str) -> Message {
836    // Use no_std-safe constructors (std-gated `Message::new` and
837    // `AgentInstanceId::new` aren't available in the wasm-interpreter
838    // build). The kernel replaces the agent instance id when it builds
839    // the inner agent loop, so the placeholder zeros are ephemeral.
840    use tau_domain::MessageId;
841    Message::new_with(
842        MessageId::from_parts(0, [0u8; 10]),
843        chrono::DateTime::<chrono::Utc>::from_timestamp_millis(0).unwrap(),
844        Address::User,
845        Address::Agent(AgentInstanceId::from_parts(0, [0u8; 10])),
846        MessagePayload::Text {
847            content: content.to_string(),
848        },
849    )
850}
