38/// Terminal outcome of driving a pipeline via [`run_pipeline_suspendable`].
39#[derive(Debug)]
40pub enum PipelineOutcome {
41    /// Ran to the end; carries the accumulated outputs.
42    Completed(OutputStore),
43    /// Paused at a top-level `Suspend`. State was persisted via the
44    /// `SuspensionStore` passed in `SuspendConfig` before returning.
45    Suspended {
46        /// The run id the suspension was persisted under.
47        run_id: String,
48        /// The signal name a `--resume --signal <name>` must supply.
49        resume_signal: String,
50        /// The pipeline-step id of the `Suspend` block that paused.
51        step_id: String,
52    },
53}

55/// How a `run_steps` slice terminated. Only the top-level slice (the one
56/// `run_pipeline_suspendable` invokes directly, with `suspend: Some(_)`) can
57/// terminate `Suspended`; nested slices (`Branch`/`Loop`/`Parallel` bodies)
58/// always pass `suspend: None` and must complete or error.
59enum StepsFlow {
60    /// The slice ran every step.
61    Completed,
62    /// The slice paused at a `Suspend` step.
63    Suspended {
64        /// The pipeline-step id of the `Suspend` block.
65        step_id: String,
66        /// The signal name the resumer must supply.
67        resume_signal: String,
68    },
69}

71/// Durable-suspend wiring for the top-level pipeline driver. Carries the
72/// run id (the suspension key + the `--resume` handle) and the store a
73/// `Suspend` step's snapshot is persisted to.
74pub struct SuspendConfig {
75    /// Run id used as the suspension key + `--resume` handle.
76    pub run_id: String,
77    /// Where a pause is durably persisted.
78    pub store: Arc<dyn tau_ports::orchestration::SuspensionStore>,
79}

81/// Rehydrated state for a resumed run: the `OutputStore` snapshot restored
82/// from the prior suspension, and the top-level step index to resume at
83/// (`step_cursor + 1`, so the `Suspend` step itself is not re-run).
84pub struct ResumeState {
85    /// The `OutputStore` snapshot restored from the suspension.
86    pub store: OutputStore,
87    /// Top-level step index to resume at (`step_cursor + 1`).
88    pub start_at: usize,
89    /// Per-check attempt counts restored from the prior suspension, so a
90    /// check's `max_attempts` budget accumulates across resume boundaries
91    /// instead of resetting each resume. Fresh runs pass an empty map.
92    pub attempts: BTreeMap<String, u32>,
93}

95/// Internal: passed down to `run_steps` for the top-level slice only. A
96/// nested slice (Branch/Loop/Parallel body) receives `None`, which is how
97/// `run_steps` tells a real `Suspend` step from one that is out of place.
98struct SuspendCtx<'a> {
99    run_id: &'a str,
100    ir_digest: &'a str,
101    store: &'a dyn tau_ports::orchestration::SuspensionStore,
102}

104/// Drive an `IrModule`'s pipeline, pausing durably at a top-level `Suspend`
105/// step rather than erroring.
106///
107/// The module's `workflow.pipeline` must be `Some` — callers branch on
108/// that before dispatching here (see `tau run`). Each step renders its
109/// `input` template against the run `input` and prior outputs, runs the
110/// step, and records the step's output keyed by its pipeline-step id so
111/// later steps can reference it via `${steps.<id>.output}`.
112///
113/// [`StepRun::Agent`], [`StepRun::Tool`], [`StepRun::Deterministic`],
114/// [`StepRun::Check`], [`StepRun::Branch`], [`StepRun::Loop`],
115/// [`StepRun::Parallel`], [`StepRun::Suspend`], and [`StepRun::Dynamic`]
116/// steps are all supported. A `Dynamic` region runs its `owner` coordinator
117/// agent with one [`crate::interpreter::dynamic::SpawnTool`] registered per
118/// offered kind (`agent.<kind>.spawn`); each spawn is admitted against the
119/// region's `max_spawns`/`max_concurrency` bounds counters and attenuated to
120/// the meet of the region's `envelope` and the kind's declared capabilities
121/// before the child agent runs — a spawn past bounds is soft-denied (the
122/// coordinator sees a denial `ToolResult`, the run itself still completes).
123/// A `Check` step evaluates a postcondition (goal or deliverable) against the
124/// accumulated outputs. A `Branch` evaluates its condition and recurses into
125/// the chosen arm (`then`/`otherwise`) against the same shared store; it
126/// stores no output of its own. A `Loop` runs its `body` up to `max_iters`
127/// times, checking `until` after each pass and threading a failed verdict's
128/// rationale into the next pass as feedback; exhausting `max_iters` without
129/// `until` holding is a hard error ([`RuntimeError::LoopExhausted`]). Like
130/// `Branch`, a `Loop` stores no output of its own. A `Parallel` forks each
131/// branch over an isolated read-only snapshot of the store (branches cannot
132/// see each other's outputs), drives them with bounded concurrency
133/// (`PARALLEL_CAP` in flight at once), and merges each branch's produced
134/// outputs back into the shared store in index order once all branches
135/// complete; like `Branch` and `Loop`, it stores no output of its own. A
136/// top-level `Suspend` step pauses the run: computes the module's
137/// canonical-IR digest once (for drift-detection on a future resume), seeds
138/// the run's `OutputStore` and start index either fresh (`resume: None`) or
139/// from a prior pause (`resume: Some(ResumeState)` — restore-and-continue:
140/// the store is rehydrated from the persisted snapshot and the walk resumes
141/// at `step_cursor + 1`, so the prefix up to and including the `Suspend`
142/// step is never re-run), and — on hitting `Suspend` — durably persists a
143/// [`tau_ports::orchestration::PipelineSuspension`] via the `SuspensionStore`
144/// in `SuspendConfig` and returns [`PipelineOutcome::Suspended`] rather than
145/// erroring. Only a top-level `Suspend` may pause; one nested inside a
146/// `Branch`/`Loop`/`Parallel` body is a build-time typecheck error (an
147/// [`RuntimeError::Internal`] here would mean that check regressed).
148///
149/// # Failure handling: abort, or rewind-to-gate retry
150///
151/// On a failed check the [`RetryPolicy`](tau_ir::check::RetryPolicy)
152/// decides what happens. If `on_fail == Abort`, or the attempt count has
153/// reached `max_attempts`, the pipeline aborts with
154/// [`RuntimeError::CheckFailed`]. Otherwise the loop emits
155/// [`EV_CHECK_RETRY`], stashes the judge's rationale as `feedback` keyed by
156/// the check's `gate` step and the check's id, and rewinds the loop index to
157/// that `gate` step so the forward slice re-runs. On the next pass through the
158/// gate's agent step every outstanding rationale registered for *that* gate is
159/// injected as a labelled prior turn (`"Previous attempt rejected: (check
160/// '<id>') <rationale>"`) so the agent sees *why* it was rejected. Keying by
161/// gate + check id keeps concurrently-failing checks distinct (a pass clears
162/// only its own check's entry; a fresh failure updates only its own) and
163/// confines each check's feedback to its own gate step.
164pub async fn run_pipeline_suspendable<D>(
165    module: Arc<IrModule>,
166    input: String,
167    dispatcher: Arc<D>,
168    suspend: SuspendConfig,
169    resume: Option<ResumeState>,
170) -> Result<PipelineOutcome, RuntimeError>
171where
172    D: ToolDispatcher + Send + Sync + 'static,
173{
174    let pipeline = module
175        .workflow
176        .pipeline
177        .clone()
178        .ok_or_else(|| RuntimeError::Internal {
179            message: "run_pipeline called on a module without a pipeline".to_string(),
180        })?;
181
182    // Canonical-IR digest, computed once, for drift-detection on resume.
183    let ir_digest = {
184        let bytes = tau_ir::to_canonical_bytes(&module);
185        tau_ir::asset::asset_hash(&bytes)
186    };
187
188    let (mut store, start_at, initial_attempts) = match resume {
189        Some(r) => (r.store, r.start_at, r.attempts),
190        None => (OutputStore::new(), 0, BTreeMap::new()),
191    };
192
193    let ctx = SuspendCtx {
194        run_id: &suspend.run_id,
195        ir_digest: &ir_digest,
196        store: suspend.store.as_ref(),
197    };
198    // Run-scoped `Dynamic`-step entry counter (#731). Bumped once per
199    // execution of a `Dynamic` step — including each `Loop` pass and each
200    // check rewind — and folded into every child agent id so ids stay unique
201    // across a run, which per-region `RegionCounters` alone cannot achieve
202    // (it is rebuilt per execution). Owned here, borrowed by every nested
203    // `run_steps` slice; `&AtomicU64` is `Copy + Sync`, so the `Parallel`
204    // arm's concurrent branches share the one counter without a lock (there
205    // is no `Mutex` on the `no_std` guest profile).
206    let region_entries = AtomicU64::new(0);
207    let flow = run_steps(
208        &module,
209        &pipeline.steps,
210        &input,
211        &mut store,
212        &dispatcher,
213        None,
214        Some(&ctx),
215        start_at,
216        initial_attempts,
217        &region_entries,
218    )
219    .await?;
220    Ok(match flow {
221        StepsFlow::Completed => PipelineOutcome::Completed(store),
222        StepsFlow::Suspended {
223            step_id,
224            resume_signal,
225        } => PipelineOutcome::Suspended {
226            run_id: suspend.run_id,
227            resume_signal,
228            step_id,
229        },
230    })
231}

233/// Non-suspend convenience wrapper over [`run_pipeline_suspendable`] for
234/// callers that cannot pause (e.g. the bundle-run path in v1). Wires a
235/// no-op `SuspensionStore` and runs fresh (never resumes); a pipeline that
236/// hits a `Suspend` step errors [`RuntimeError::SuspendUnsupported`] instead
237/// of pausing. Callers that support HITL suspend/resume use
238/// `run_pipeline_suspendable` directly.
239pub async fn run_pipeline<D>(
240    module: Arc<IrModule>,
241    input: String,
242    dispatcher: Arc<D>,
243) -> Result<OutputStore, RuntimeError>
244where
245    D: ToolDispatcher + Send + Sync + 'static,
246{
247    struct NoopSuspensions;
248    impl tau_ports::orchestration::SuspensionStore for NoopSuspensions {
249        fn persist_suspension(
250            &self,
251            _: &tau_ports::orchestration::PipelineSuspension,
252        ) -> Result<(), tau_ports::orchestration::CheckpointError> {
253            Ok(())
254        }
255        fn load_suspension(
256            &self,
257            _: &tau_ports::orchestration::RunId,
258        ) -> Result<
259            Option<tau_ports::orchestration::PipelineSuspension>,
260            tau_ports::orchestration::CheckpointError,
261        > {
262            Ok(None)
263        }
264    }
265    let outcome = run_pipeline_suspendable(
266        module,
267        input,
268        dispatcher,
269        SuspendConfig {
270            run_id: String::new(),
271            store: Arc::new(NoopSuspensions),
272        },
273        None,
274    )
275    .await?;
276    match outcome {
277        PipelineOutcome::Completed(store) => Ok(store),
278        PipelineOutcome::Suspended {
279            step_id,
280            resume_signal,
281            ..
282        } => Err(RuntimeError::SuspendUnsupported {
283            step: step_id,
284            resume_signal,
285        }),
286    }
287}

289/// Execute a slice of pipeline steps against a shared [`OutputStore`].
290///
291/// Extracted from [`run_pipeline`] so nested blocks (Branch/Parallel/Loop)
292/// can drive their own step slices with the same gate-rewind, feedback, and
293/// check-dispatch semantics. `initial_feedback` seeds the per-slice feedback
294/// carried into the first gate agent step (used when a nested slice re-runs
295/// with a prior rejection rationale); top-level callers pass `None`.
296///
297/// See [`run_pipeline`]'s docs for the abort / rewind-to-gate retry model —
298/// the loop body lives here.
299///
300/// `region_entries` is the run-scoped `Dynamic`-step entry counter; every
301/// nested slice borrows the top-level one so dynamic-region child ids stay
302/// unique across the whole run (#731).
303#[allow(clippy::too_many_arguments)]
304async fn run_steps<D>(
305    module: &Arc<IrModule>,
306    steps: &[tau_ir::pipeline::PipelineStep],
307    input: &str,
308    store: &mut OutputStore,
309    dispatcher: &Arc<D>,
310    initial_feedback: Option<String>,
311    suspend: Option<&SuspendCtx<'_>>,
312    start_at: usize,
313    initial_attempts: BTreeMap<String, u32>,
314    region_entries: &AtomicU64,
315) -> Result<StepsFlow, RuntimeError>
316where
317    D: ToolDispatcher + Send + Sync + 'static,
318{
319    // Per-check attempt counter (1-based on first eval), keyed by check id.
320    // Seeded from `initial_attempts` on a resumed top-level slice so a check's
321    // `max_attempts` budget accumulates across resume boundaries; every
322    // nested slice (Branch/Loop/Parallel body) passes an empty map.
323    let mut attempts: BTreeMap<String, u32> = initial_attempts;
324    // Rejection rationales of failed checks that triggered a rewind, keyed by
325    // the gate step they rewind to, then by check id (issue #471). Injecting
326    // only the entries whose key matches the current step id scopes each
327    // check's feedback to its own gate agent step (never leaking into
328    // unrelated agent steps in the rewound slice). Multiple checks gating one
329    // step keep distinct entries — a pass clears only its own check's entry, a
330    // fresh failure updates only its own — instead of last-writer-wins over a
331    // single shared slot.
332    let mut feedback: BTreeMap<String, BTreeMap<String, String>> = BTreeMap::new();
333    // Loop prior-iteration rationale (from `Loop.until`), which has no
334    // associated check or gate. Injected once, at the first agent step of this
335    // slice, then consumed. Top-level/branch/parallel callers pass `None`.
336    let mut pending_initial_feedback: Option<String> = initial_feedback;
337    // gate-id -> pipeline-step index, so a failed retryable check can rewind
338    // `i` to its gate without re-scanning the pipeline each time.
339    let gate_index: BTreeMap<&str, usize> = steps
340        .iter()
341        .enumerate()
342        .map(|(idx, step)| (step.id.0.as_str(), idx))
343        .collect();
344
345    // Index loop (not `for`) so a failed retryable check can rewind `i` to a
346    // gate step. On the happy path `i` only ever advances by one. `start_at`
347    // is nonzero only on a resumed top-level slice (skips the already-run
348    // prefix up to and including the prior `Suspend` step); every recursive
349    // call passes `0`.
350    let mut i = start_at;
351    while i < steps.len() {
352        let step = &steps[i];
353
354        // Check steps have their own span/event vocabulary and store no
355        // output, so dispatch them before the Agent/Tool/Deterministic path.
356        if let StepRun::Check(check_id) = &step.run {
357            let check =
358                module
359                    .workflow
360                    .checks
361                    .get(check_id)
362                    .ok_or_else(|| RuntimeError::Internal {
363                        message: format!("unknown check {}", check_id.0),
364                    })?;
365
366            let check_span = info_span!(SPAN_PIPELINE_CHECK, id = check_id.0.as_str());
367
368            let (verdict, kind) = match &check.verify {
369                CheckVerify::Goal {
370                    evaluates,
371                    predicate,
372                } => {
373                    let reg = dispatcher.deterministic_registry().ok_or_else(|| {
374                        RuntimeError::Internal {
375                            message: format!("check {} needs a deterministic registry", check_id.0),
376                        }
377                    })?;
378                    // Bind the reader to a local so the `Arc` temporary lives
379                    // across the borrow passed to `evaluate_goal`.
380                    let reader = dispatcher.artifact_reader();
381                    let verdict = evaluate_goal(
382                        evaluates,
383                        predicate,
384                        store,
385                        reader.as_ref().map(|a| a.as_ref()),
386                        reg.as_ref(),
387                    )?;
388                    (verdict, "goal")
389                }
390                CheckVerify::Deliverable {
391                    locus,
392                    must_satisfy,
393                    judge,
394                } => {
395                    let reader = dispatcher.artifact_reader();
396                    let verdict = Box::pin(evaluate_deliverable(
397                        module.clone(),
398                        locus,
399                        must_satisfy,
400                        judge,
401                        store,
402                        reader.as_ref().map(|a| a.as_ref()),
403                        dispatcher.clone(),
404                    ))
405                    .instrument(check_span.clone())
406                    .await?;
407                    (verdict, "deliverable")
408                }
409            };
410
411            // Real (1-based) attempt count for this check.
412            let attempt = attempts.get(&check_id.0).copied().unwrap_or(0) + 1;
413            attempts.insert(check_id.0.clone(), attempt);
414
415            tracing::info!(
416                parent: &check_span,
417                name = EV_CHECK_EVALUATED,
418                id = check_id.0.as_str(),
419                kind = kind,
420                verdict = if verdict.met { "pass" } else { "fail" },
421                attempt = attempt,
422            );
423
424            if verdict.met {
425                // Passed: clear only THIS check's feedback under its gate,
426                // leaving any sibling checks' outstanding rationales intact.
427                // Checks store no output.
428                if let Some(inner) = feedback.get_mut(&check.retry.gate.0) {
429                    inner.remove(&check_id.0);
430                    if inner.is_empty() {
431                        feedback.remove(&check.retry.gate.0);
432                    }
433                }
434                i += 1;
435                continue;
436            }
437
438            // Failed. Abort if the policy says so, or if we've exhausted the
439            // attempt budget.
440            if check.retry.on_fail == OnFail::Abort || attempt >= check.retry.max_attempts {
441                return Err(RuntimeError::CheckFailed {
442                    id: check_id.0.clone(),
443                    kind: String::from(kind),
444                    rationale: verdict.rationale,
445                    attempt,
446                });
447            }
448
449            // Retryable with attempts remaining: rewind to the gate step.
450            let gate_idx = *gate_index.get(check.retry.gate.0.as_str()).ok_or_else(|| {
451                // Build-time integrity checks (tau-ir typecheck) guarantee the
452                // gate names an existing pipeline step, so reaching here is an
453                // invariant violation — surface it loudly rather than aborting
454                // the run as a plain check failure.
455                RuntimeError::Internal {
456                    message: format!(
457                        "check {} rewinds to unknown gate step {}",
458                        check_id.0, check.retry.gate.0
459                    ),
460                }
461            })?;
462
463            tracing::info!(
464                parent: &check_span,
465                name = EV_CHECK_RETRY,
466                id = check_id.0.as_str(),
467                rewind_to = check.retry.gate.0.as_str(),
468                next_attempt = attempt + 1,
469            );
470
471            feedback
472                .entry(check.retry.gate.0.clone())
473                .or_default()
474                .insert(check_id.0.clone(), verdict.rationale);
475            i = gate_idx;
476            continue;
477        }
478
479        // `Branch` blocks have their own early dispatch, mirroring `Check`
480        // above: they store no output of their own (the chosen arm's steps
481        // record their own outputs into the shared `store` as they run).
482        if let StepRun::Branch {
483            on,
484            then,
485            otherwise,
486        } = &step.run
487        {
488            let reg =
489                dispatcher
490                    .deterministic_registry()
491                    .ok_or_else(|| RuntimeError::Internal {
492                        message: format!("branch {} needs a deterministic registry", step.id.0),
493                    })?;
494            let reader = dispatcher.artifact_reader();
495            let verdict = crate::interpreter::check::eval_condition(
496                on,
497                store,
498                reader.as_ref().map(|a| a.as_ref()),
499                reg.as_ref(),
500            )?;
501            let arm = if verdict.met { then } else { otherwise };
502            match Box::pin(run_steps(
503                module,
504                arm,
505                input,
506                store,
507                dispatcher,
508                None,
509                None,
510                0,
511                BTreeMap::new(),
512                region_entries,
513            ))
514            .await?
515            {
516                StepsFlow::Completed => {}
517                StepsFlow::Suspended { .. } => {
518                    return Err(RuntimeError::Internal {
519                        message: "suspend escaped a nested slice (typecheck should reject)"
520                            .to_string(),
521                    })
522                }
523            }
524            i += 1;
525            continue;
526        }
527
528        // `Loop` blocks have their own early dispatch, mirroring `Check` and
529        // `Branch` above: they store no output of their own (the body's
530        // steps record their own outputs into the shared `store` as they
531        // run, each pass overwriting the prior pass's). Runs `body` up to
532        // `max_iters` times, checking `until` after each pass; converges as
533        // soon as `until` holds, otherwise threads the failed verdict's
534        // rationale into the next pass as feedback (same
535        // "Previous attempt rejected: <rationale>" idiom `Check`'s
536        // rewind-to-gate retry uses). Exhausting `max_iters` without `until`
537        // holding is a hard error (ADR-0058 / ADR-0059) — a bounded loop
538        // that cannot reach its goal is a failure, not a silent success.
539        if let StepRun::Loop {
540            body,
541            until,
542            max_iters,
543        } = &step.run
544        {
545            let mut loop_feedback: Option<String> = None;
546            let mut converged = false;
547            for _iter in 0..*max_iters {
548                match Box::pin(run_steps(
549                    module,
550                    body,
551                    input,
552                    store,
553                    dispatcher,
554                    loop_feedback.take(),
555                    None,
556                    0,
557                    BTreeMap::new(),
558                    region_entries,
559                ))
560                .await?
561                {
562                    StepsFlow::Completed => {}
563                    StepsFlow::Suspended { .. } => {
564                        return Err(RuntimeError::Internal {
565                            message: "suspend escaped a nested slice (typecheck should reject)"
566                                .to_string(),
567                        })
568                    }
569                }
570                let reg =
571                    dispatcher
572                        .deterministic_registry()
573                        .ok_or_else(|| RuntimeError::Internal {
574                            message: format!("loop {} needs a deterministic registry", step.id.0),
575                        })?;
576                let reader = dispatcher.artifact_reader();
577                let verdict = crate::interpreter::check::eval_condition(
578                    until,
579                    store,
580                    reader.as_ref().map(|a| a.as_ref()),
581                    reg.as_ref(),
582                )?;
583                if verdict.met {
584                    converged = true;
585                    break;
586                }
587                loop_feedback = Some(verdict.rationale);
588            }
589            if !converged {
590                return Err(RuntimeError::LoopExhausted {
591                    step: step.id.0.clone(),
592                    max_iters: *max_iters,
593                });
594            }
595            i += 1;
596            continue;
597        }
598
599        // `Suspend` blocks have their own early dispatch, mirroring `Branch`,
600        // `Loop`, and `Parallel` above. Only the top-level slice carries a
601        // `SuspendCtx` (nested Branch/Loop/Parallel bodies pass `None` —
602        // typecheck rejects a `Suspend` nested inside those, so `None` here
603        // is an invariant violation, not a normal case). With a ctx: persist
604        // the pause (run id, resume signal, step cursor, IR digest, and a
605        // snapshot of the store so far) and unwind to the caller as
606        // `StepsFlow::Suspended` rather than erroring.
607        if let StepRun::Suspend { resume_signal } = &step.run {
608            match suspend {
609                Some(ctx) => {
610                    ctx.store
611                        .persist_suspension(&tau_ports::orchestration::PipelineSuspension {
612                            run_id: ctx.run_id.to_string(),
613                            resume_signal: resume_signal.clone(),
614                            step_cursor: i,
615                            step_id: step.id.0.clone(),
616                            ir_digest: ctx.ir_digest.to_string(),
617                            outputs: store.snapshot(),
618                            attempts: attempts.clone(),
619                        })
620                        .map_err(|e| RuntimeError::Internal {
621                            message: format!("persist suspension: {e}"),
622                        })?;
623                    return Ok(StepsFlow::Suspended {
624                        step_id: step.id.0.clone(),
625                        resume_signal: resume_signal.clone(),
626                    });
627                }
628                None => {
629                    return Err(RuntimeError::SuspendUnsupported {
630                        step: step.id.0.clone(),
631                        resume_signal: resume_signal.clone(),
632                    })
633                }
634            }
635        }
636
637        // `Parallel` blocks have their own early dispatch, mirroring `Branch`
638        // and `Loop` above: they store no output of their own. Each branch
639        // forks a read-only snapshot of `store` (branches are read-isolated
640        // from each other — a branch cannot see a sibling's outputs, only
641        // the pre-fork state), runs to completion, and returns its produced
642        // store. Branches are driven with bounded concurrency
643        // (`PARALLEL_CAP` in flight at once, nginx `worker_connections`
644        // style) and their outputs are merged back into the shared store in
645        // index order once all branches complete, so the merge is
646        // deterministic regardless of which branch's future resolves first.
647        if let StepRun::Parallel { branches } = &step.run {
648            use futures_util::stream::{self, StreamExt};
649            let futs = branches.iter().enumerate().map(|(idx, branch)| {
650                let module = module.clone();
651                let dispatcher = dispatcher.clone();
652                let input = input.to_string();
653                let mut snap = store.clone();
654                async move {
655                    match run_steps(
656                        &module,
657                        branch,
658                        &input,
659                        &mut snap,
660                        &dispatcher,
661                        None,
662                        None,
663                        0,
664                        BTreeMap::new(),
665                        region_entries,
666                    )
667                    .await?
668                    {
669                        StepsFlow::Completed => {}
670                        StepsFlow::Suspended { .. } => {
671                            return Err(RuntimeError::Internal {
672                                message: "suspend escaped a nested slice (typecheck should reject)"
673                                    .to_string(),
674                            })
675                        }
676                    }
677                    Ok::<(usize, OutputStore), RuntimeError>((idx, snap))
678                }
679            });
680            let mut results: alloc::vec::Vec<(usize, OutputStore)> = stream::iter(futs)
681                .buffered(PARALLEL_CAP)
682                .collect::<alloc::vec::Vec<_>>()
683                .await
684                .into_iter()
685                .collect::<Result<_, _>>()?;
686            // `buffered` preserves input order, but sort by index defensively
687            // so the merge is deterministic regardless of driver scheduling.
688            results.sort_by_key(|(idx, _)| *idx);
689            for (_, produced) in results {
690                store.merge(produced);
691            }
692            i += 1;
693            continue;
694        }
695
696        // NOTE: we do NOT call `.entered()` here — `EnteredSpan` mutates a
697        // thread-local span stack, and tokio's multi-thread scheduler can
698        // move this task to a different worker thread at any `.await`,
699        // leaving the guard on the wrong thread and mis-parenting child
700        // spans/events. Instead every event uses `parent: &step_span`
701        // explicitly, and every awaited future is wrapped with
702        // `.instrument(step_span.clone())`. See stream.rs:273-283 for the
703        // same idiom applied to the runtime turn span.
704        let step_span = info_span!(SPAN_PIPELINE_STEP, id = step.id.0.as_str());
705        tracing::info!(parent: &step_span, name = EV_PIPELINE_STEP_STARTED, id = step.id.0.as_str());
706
707        let rendered = tau_ir::template::resolve(&step.input, input, &store.template_map())
708            .map_err(|e| RuntimeError::Internal {
709                message: format!("pipeline step {}: {e}", step.id.0),
710            })?;
711
712        let output: Value = match &step.run {
713            StepRun::Agent(agent_id) => {
714                let agent = module
715                    .workflow
716                    .agents
717                    .get(agent_id)
718                    .ok_or_else(|| RuntimeError::AgentNotFound {
719                        agent: agent_id.0.clone(),
720                    })?
721                    .clone();
722                // When a prior check rewound to this gate, inject its
723                // rationale as a prior turn so the agent sees *why* it was
724                // rejected. `split_history` (agent_loop.rs) treats all-but-last
725                // as history and the last as the live turn, so each feedback
726                // message becomes a prior turn and `rendered` stays the live
727                // prompt.
728                let mut initial: alloc::vec::Vec<Message> = alloc::vec::Vec::new();
729                // Loop prior-iteration feedback (no check/gate): inject once,
730                // at the first agent step of the slice, then consume it.
731                if let Some(fb) = pending_initial_feedback.take() {
732                    initial.push(user_message(&format!("Previous attempt rejected: {fb}")));
733                }
734                // Check feedback gating THIS step: one labeled prior turn per
735                // outstanding check, so several checks on one gate stay
736                // distinct (issue #471). BTreeMap iteration is sorted by check
737                // id, so ordering is deterministic.
738                if let Some(inner) = feedback.get(&step.id.0) {
739                    for (cid, fb) in inner {
740                        initial.push(user_message(&format!(
741                            "Previous attempt rejected: (check '{cid}') {fb}"
742                        )));
743                    }
744                }
745                initial.push(user_message(&rendered));
746                let outcome = Box::pin(run_agent(
747                    module.clone(),
748                    &agent,
749                    dispatcher.clone(),
750                    initial,
751                ))
752                .instrument(step_span.clone())
753                .await?;
754                match outcome {
755                    RunOutcome::Failed { status, .. } => {
756                        return Err(RuntimeError::Internal {
757                            message: format!(
758                                "pipeline step {} (agent {}) failed: {status:?}",
759                                step.id.0, agent_id.0
760                            ),
761                        })
762                    }
763                    _ => Value::String(last_assistant_text(&outcome)),
764                }
765            }
766            // EPIC 4.5: run the region's coordinator (`owner`) with one
767            // `SpawnTool` registered per offered kind. Mirrors the `Agent`
768            // arm immediately above for `user_message` construction, span
769            // instrumentation, and `RunOutcome::Failed` error mapping — the
770            // only difference is the coordinator's tool registry gains the
771            // per-kind spawn tools. `retry_from`/`retry.gate` accepts ANY
772            // step id (tau-pkg validation only requires gate_index <=
773            // producer_index and that some agent step exists in the rewound
774            // range) — a `Dynamic` step id is a legal rewind gate, so both
775            // `pending_initial_feedback` (loop prior-iteration rationale)
776            // and `feedback.get(&step.id.0)` (check-retry rationale) are
777            // injected here exactly like the `Agent` arm, so a check that
778            // rewinds to a `Dynamic` gate doesn't burn its whole
779            // `max_attempts` budget re-running the coordinator with an
780            // identical prompt and no rationale.
781            StepRun::Dynamic {
782                owner,
783                envelope,
784                spawns,
785                max_spawns,
786                max_concurrency,
787            } => {
788                let agent = module
789                    .workflow
790                    .agents
791                    .get(owner)
792                    .ok_or_else(|| RuntimeError::Internal {
793                        message: format!(
794                            "dynamic region '{}' owner '{}' not in workflow.agents (typecheck should reject)",
795                            step.id.0, owner.0
796                        ),
797                    })?
798                    .clone();
799                let counters = Arc::new(RegionCounters::new(*max_spawns, *max_concurrency));
800                // One bump per *execution* of this step, taken before the
801                // coordinator runs. `counters` is rebuilt here too, so its
802                // admission index restarts at 0 on a `Loop` pass or a check
803                // rewind; `entry` is what keeps the resulting child ids
804                // distinct across those re-entries (#731).
805                let entry = region_entries.fetch_add(1, Ordering::SeqCst);
806                let spawn_tools: alloc::vec::Vec<SpawnTool<D>> = spawns
807                    .iter()
808                    .map(|s| {
809                        SpawnTool::new(
810                            s.clone(),
811                            envelope.clone(),
812                            counters.clone(),
813                            step.id.0.clone(),
814                            entry,
815                            module.clone(),
816                            dispatcher.clone(),
817                        )
818                    })
819                    .collect();
820                let mut initial: alloc::vec::Vec<Message> = alloc::vec::Vec::new();
821                if let Some(fb) = pending_initial_feedback.take() {
822                    initial.push(user_message(&format!("Previous attempt rejected: {fb}")));
823                }
824                if let Some(inner) = feedback.get(&step.id.0) {
825                    for (cid, fb) in inner {
826                        initial.push(user_message(&format!(
827                            "Previous attempt rejected: (check '{cid}') {fb}"
828                        )));
829                    }
830                }
831                initial.push(user_message(&rendered));
832                let outcome = Box::pin(run_agent_with_spawn_tools(
833                    module.clone(),
834                    &agent,
835                    dispatcher.clone(),
836                    initial,
837                    spawn_tools,
838                ))
839                .instrument(step_span.clone())
840                .await?;
841                match outcome {
842                    RunOutcome::Failed { status, .. } => {
843                        return Err(RuntimeError::Internal {
844                            message: format!(
845                                "pipeline step {} (dynamic region owner {}) failed: {status:?}",
846                                step.id.0, owner.0
847                            ),
848                        })
849                    }
850                    _ => Value::String(last_assistant_text(&outcome)),
851                }
852            }
853            StepRun::Tool(tool_id) => {
854                let args = rendered_to_args(&rendered);
855                let result = dispatcher
856                    .invoke(tool_id, &args)
857                    .instrument(step_span.clone())
858                    .await?;
859                match (result.body, result.error) {
860                    (Some(body), _) => body,
861                    (None, Some(err)) => {
862                        return Err(RuntimeError::Internal {
863                            message: alloc::format!(
864                                "pipeline step {} (tool {}) errored: {err}",
865                                step.id.0,
866                                tool_id.0
867                            ),
868                        })
869                    }
870                    (None, None) => Value::Null,
871                }
872            }
873            StepRun::Deterministic(step_node_id) => {
874                let registry =
875                    dispatcher
876                        .deterministic_registry()
877                        .ok_or_else(|| RuntimeError::Internal {
878                            message: alloc::format!(
879                                "pipeline step {} needs a deterministic registry, none provided",
880                                step.id.0
881                            ),
882                        })?;
883                let node = module.workflow.steps.get(step_node_id).ok_or_else(|| {
884                    RuntimeError::Internal {
885                        message: alloc::format!("unknown deterministic step {}", step_node_id.0),
886                    }
887                })?;
888                let args = rendered_to_args(&rendered);
889                crate::interpreter::deterministic::run_step(node, registry.as_ref(), &args)?
890            }
891            // `Check` steps are dispatched at the top of the loop and never
892            // reach this `match` (they store no output of their own).
893            StepRun::Check(_) => unreachable!("check steps are handled before this match"),
894            // `Branch`, `Parallel`, `Loop`, and `Suspend` are all
895            // early-dispatched above (each either stores no output of its
896            // own, or — for `Suspend` — returns `StepsFlow::Suspended` before
897            // reaching here), so none of these ever reach this `match`.
898            // `Dynamic` is NOT early-dispatched — it's a real arm above,
899            // like `Agent` — so it is deliberately absent from this group.
900            StepRun::Branch { .. }
901            | StepRun::Parallel { .. }
902            | StepRun::Loop { .. }
903            | StepRun::Suspend { .. } => {
904                unreachable!("control-flow blocks are early-dispatched")
905            }
906        };
907
908        store.insert(step.id.0.clone(), output);
909        tracing::info!(parent: &step_span, name = EV_PIPELINE_STEP_COMPLETED, id = step.id.0.as_str());
910
911        i += 1;
912    }
913
914    Ok(StepsFlow::Completed)
915}

917/// Turn a rendered template string into the `Value` a tool/deterministic
918/// step receives: parse it as JSON if it parses, else wrap as a string.
919///
920/// Footgun: a rendered string that happens to be a bare JSON scalar
921/// (`42`, `true`, `null`) parses to that scalar rather than wrapping as a
922/// string. Author tool/deterministic `input` templates accordingly.
923fn rendered_to_args(rendered: &str) -> Value {
924    serde_json::from_str::<Value>(rendered).unwrap_or_else(|_| Value::String(rendered.to_string()))
925}

927/// Build a user-turn [`Message`] carrying `content` as its text payload.
928///
929/// Mirrors the initial-message idiom in `tau-cli`'s `run` command: the
930/// recipient is a freshly-minted [`AgentInstanceId`] placeholder that the
931/// kernel replaces when it assigns the loop's own instance id.
932///
933/// Exposed as `pub(crate)` so `check.rs` can reuse it without a separate
934/// definition (single source of truth).
935pub(crate) fn user_message(content: &str) -> Message {
936    // Use no_std-safe constructors (std-gated `Message::new` and
937    // `AgentInstanceId::new` aren't available in the wasm-interpreter
938    // build). The kernel replaces the agent instance id when it builds
939    // the inner agent loop, so the placeholder zeros are ephemeral.
940    use tau_domain::MessageId;
941    Message::new_with(
942        MessageId::from_parts(0, [0u8; 10]),
943        chrono::DateTime::<chrono::Utc>::from_timestamp_millis(0).unwrap(),
944        Address::User,
945        Address::Agent(AgentInstanceId::from_parts(0, [0u8; 10])),
946        MessagePayload::Text {
947            content: content.to_string(),
948        },
949    )
950}
