1//! Per-agent loop driver.
2//!
3//! Routes through the existing `Runtime::run_with_history` (kernel
4//! agent loop) by constructing a `Runtime` configured with the agent
5//! node's tools, prompt, model, and budget. The ToolDispatcher trait
6//! call (task 4.3) is what each tool reaches when invoked — the
7//! `Runtime`'s tool registry is wired with a thin wrapper that delegates
8//! to the dispatcher.

35/// Convert a `tau_domain::Value` to `serde_json::Value` via serde round-trip.
36///
37/// `tau_domain::Value` and `serde_json::Value` are distinct types; serde
38/// is the lowest-friction bridge between them.
39fn domain_value_to_json(v: &tau_domain::Value) -> serde_json::Value {
40    serde_json::to_value(v).unwrap_or(serde_json::Value::Null)
41}
42
43/// Convert a `serde_json::Value` to `tau_domain::Value` via serde round-trip.
44fn json_to_domain_value(v: serde_json::Value) -> tau_domain::Value {
45    serde_json::from_value(v).unwrap_or(tau_domain::Value::Null)
46}

52/// Extract the body text of the last `Assistant`-authored message in a
53/// `RunOutcome`'s `all_messages`. Returns an empty `String` when no
54/// assistant message exists (e.g. immediate `Failed` outcomes).
55///
56/// Used by `DispatcherTool::invoke`'s `Subflow` arm to convert a child
57/// agent's terminal state into a tool-result body for the parent.
58///
59/// # Important
60///
61/// Both `RunOutcome::Completed` and `RunOutcome::Failed` carry `all_messages`
62/// and this function returns the same shaped `String` for both. Callers MUST
63/// match on the outcome variant separately to decide whether to surface the
64/// result as a success or error — this function alone does not distinguish
65/// between them.
66pub(crate) fn last_assistant_text(outcome: &RunOutcome) -> String {
67    // Note: `RunOutcome` is `#[non_exhaustive]` but we're in the defining
68    // crate, so the compiler sees all variants. No `_` arm needed here;
69    // adding one would generate an `unreachable_patterns` warning. Future
70    // variants must be handled by updating this match.
71    let messages = match outcome {
72        RunOutcome::Completed { all_messages, .. } => all_messages,
73        RunOutcome::Failed { all_messages, .. } => all_messages,
74    };
75    for msg in messages.iter().rev() {
76        if matches!(msg.sender, tau_domain::Address::Agent(_)) {
77            if let tau_domain::MessagePayload::Text { content } = &msg.payload {
78                return content.clone();
79            }
80        }
81    }
82    String::new()
83}

89/// A `tau_ports::Tool` whose `invoke` branches on the IR `ToolImpl`:
90///
91/// - `Native` / `Mcp` → forwarded to the user's `ToolDispatcher`.
92/// - `Subflow { target }` → recursively `run_ir` against the target
93///   agent in the same module. The recursion is broken with `Box::pin`
94///   to satisfy Rust's no-self-referential-async-fn rule.
95/// - `Step { id }` → looked up in `module.workflow.steps` and called
96///   against the dispatcher's `DeterministicRegistry`.
97///
98/// One wrapper instance is created per tool in `agent.tool_refs`. Each
99/// instance holds an `Arc<D>` (the dispatcher) and the `ToolId` it should
100/// forward invocations to.
101struct DispatcherTool<D> {
102    /// Tool name as seen by the LLM (from the IR `ToolSpec`).
103    tool_name: String,
104    /// `ToolId` forwarded to the dispatcher on each invoke (Native/Mcp).
105    tool_id: ToolId,
106    /// LLM-facing spec (constructed via serde to bypass `#[non_exhaustive]`).
107    spec: ToolSpec,
108    /// IR module, shared so subflow recursion + step lookup work.
109    module: alloc::sync::Arc<tau_ir::IrModule>,
110    /// Clone of the IR tool's `ToolImpl`. Cheap to clone (a few hashes
111    /// or a `String`).
112    tool_impl: tau_ir::ToolImpl,
113    /// Shared dispatcher handle.
114    dispatcher: Arc<D>,
115}

117// `Tool::capabilities()` is intentionally NOT overridden (issue #581): the
118// dispatch-site gate compares a tool's required caps against the running
119// agent's grant, and the IR has no per-agent grant — the interpreter's
120// synthesised manifest grants `[]`, so forwarding the IR tool node's
121// declared caps here would policy-deny every capability-bearing tool on
122// every bundle run. IR-declared caps are enforced at build time (governance
123// ceiling + D-3b capability-fit), at the host boundary (OS sandbox / wasm
124// WIT world + WasiCtx), and at subflow frames (`AttenuatedDispatcher`).
125// Contract pinned by `tests/ir_dispatch_gate_inert.rs`.
126impl<D> tau_ports::tool::Tool for DispatcherTool<D>
127where
128    D: ToolDispatcher + Send + Sync + 'static,
129{
130    type Session = ();
131
132    fn name(&self) -> &str {
133        &self.tool_name
134    }
135
136    fn schema(&self) -> ToolSpec {
137        self.spec.clone()
138    }
139
140    async fn init(&self, _ctx: SessionContext) -> Result<Self::Session, ToolError> {
141        Ok(())
142    }
143
144    async fn invoke(
145        &self,
146        _session: &mut Self::Session,
147        args: tau_domain::Value,
148    ) -> Result<ToolResult, ToolError> {
149        // Convert tau_domain::Value → serde_json::Value once, then branch
150        // on the IR ToolImpl. The branches do NOT share extraction logic
151        // because Subflow returns a domain-side `RunOutcome` and Step
152        // returns a raw `serde_json::Value` from the registry — both end
153        // up as ToolResult Text content but the source paths differ.
154        let json_args = domain_value_to_json(&args);
155
156        match &self.tool_impl {
157            ToolImpl::Native { .. } | ToolImpl::Mcp { .. } => {
158                // Existing path: forward to the user's ToolDispatcher.
159                //
160                // Extract text symmetrically with `ForwardingDispatcher`'s body-shape
161                // lift in `crates/tau-cli/src/cmd/ir_dispatcher.rs`. The dispatcher
162                // try-parses the joined tool text as JSON and falls back to
163                // `Value::String(text)` on parse failure, so plain text reaches
164                // `body` as `Value::String("hello")`. `Display` on
165                // `serde_json::Value::String("hello")` emits the literal quoted
166                // form `"hello"` — corrupting every plain-text result the LLM
167                // sees. Special-case `Value::String` to extract the raw `String`
168                // (no quoting); fall through to `Display` for structured shapes
169                // (objects/arrays/numbers/bools/null) so they serialise as
170                // compact JSON. This makes the inverse direction lossless:
171                //   plain text  → Value::String  → raw text   (no extra quotes)
172                //   JSON object → Value::Object  → compact JSON
173                let result: ToolInvocationResult = self
174                    .dispatcher
175                    .invoke(&self.tool_id, &json_args)
176                    .await
177                    .map_err(|e| ToolError::Internal {
178                        message: alloc::format!("dispatcher error: {e}"),
179                    })?;
180
181                if let Some(err_msg) = result.error {
182                    return Ok(ToolResult::new(
183                        alloc::vec![ToolContent::Text { text: err_msg }],
184                        true,
185                    ));
186                }
187
188                let text = match result.body {
189                    Some(serde_json::Value::String(s)) => s,
190                    Some(v) => alloc::format!("{v}"),
191                    None => alloc::string::String::new(),
192                };
193                Ok(ToolResult::new(
194                    alloc::vec![ToolContent::Text { text }],
195                    false,
196                ))
197            }
198
199            ToolImpl::Subflow { target } => {
200                // Box::pin breaks the async self-reference (run_ir →
201                // run_agent → DispatcherTool::invoke → run_ir). Without it
202                // the async-fn future type would be infinitely recursive.
203                let module = self.module.clone();
204                let target = target.clone();
205
206                // Attenuation frame: this subflow tool's declared capabilities
207                // are the cap_subset the child (and its descendants) run
208                // under. Absent tool entry (shouldn't happen post-typecheck)
209                // falls back to an empty grant — deny-by-default.
210                let cap_subset = module
211                    .workflow
212                    .tools
213                    .get(&self.tool_id)
214                    .map(|t| t.capabilities.clone())
215                    .unwrap_or_default();
216                // Kept as a concrete `Arc<AttenuatedDispatcher>` (not
217                // widened to `Arc<dyn ToolDispatcher + Send + Sync>`) because
218                // `run_ir<D>` requires `D: Sized`; the inner-dispatcher
219                // coercion to the trait object happens at the `new(...)`
220                // call site instead, via `self.dispatcher.clone()`'s target
221                // type (`Arc<dyn ToolDispatcher + Send + Sync>`).
222                let child_dispatcher =
223                    Arc::new(crate::interpreter::attenuate::AttenuatedDispatcher::new(
224                        cap_subset,
225                        self.tool_id.clone(),
226                        target.0.clone(),
227                        module.clone(),
228                        self.dispatcher.clone(), // Arc<D> → Arc<dyn ToolDispatcher + Send + Sync>
229                    ));
230
231                let outcome = alloc::boxed::Box::pin(crate::interpreter::run_ir(
232                    module,
233                    &target,
234                    child_dispatcher,
235                    Vec::new(),
236                ))
237                .await
238                .map_err(|e| ToolError::Internal {
239                    message: alloc::format!("subflow recursion error: {e}"),
240                })?;
241
242                // Propagate child-agent failures as `is_error: true` tool
243                // results so the parent agent sees them and can react.
244                // A silently-swallowed Failed outcome would look like a
245                // successful empty-string result to the parent — wrong.
246                match &outcome {
247                    RunOutcome::Failed { .. } => {
248                        // Child agent failed — propagate as an error tool result so
249                        // the parent agent sees an `is_error: true` and can react.
250                        // The child's last assistant text (if any) is preserved as
251                        // diagnostic content.
252                        let last_text = last_assistant_text(&outcome);
253                        let body = if last_text.is_empty() {
254                            alloc::format!(
255                                "subflow agent {:?} failed with no further detail",
256                                target
257                            )
258                        } else {
259                            alloc::format!("subflow agent {:?} failed: {}", target, last_text)
260                        };
261                        return Ok(ToolResult::new(
262                            alloc::vec![ToolContent::Text { text: body }],
263                            true,
264                        ));
265                    }
266                    RunOutcome::Completed { .. } => {}
267                }
268
269                // Convert RunOutcome → tool result body. v0 contract:
270                // emit the last `Assistant`-sent text from `all_messages`
271                // as the tool result body. Empty string if none.
272                let text = last_assistant_text(&outcome);
273                Ok(ToolResult::new(
274                    alloc::vec![ToolContent::Text { text }],
275                    false,
276                ))
277            }
278
279            ToolImpl::Step { id } => {
280                // Step accounting (e.g. conformance test recording) lives in the
281                // DeterministicRegistry implementation, NOT in the dispatcher's
282                // `invoke` path — calling dispatcher.invoke for accounting would
283                // fire a real plugin lookup in production (ForwardingDispatcher) for
284                // a tool that doesn't exist there. Recording belongs in the registry.
285                //
286                // TODO (Phase 4): to count step invocations in conformance tests,
287                // make `MapBackedDeterministicRegistry` push records into a shared
288                // `Arc<Mutex<Vec<ToolCallRecord>>>` and expose that handle via
289                // `RecordingDispatcher::deterministic_registry()`.
290
291                // Look up the step in the IR's workflow.steps table.
292                let step =
293                    self.module
294                        .workflow
295                        .steps
296                        .get(id)
297                        .ok_or_else(|| ToolError::Internal {
298                            message: alloc::format!(
299                                "step tool {:?} references unknown step {:?} \
300                             (typecheck should have caught this — possible IR corruption)",
301                                self.tool_id,
302                                id,
303                            ),
304                        })?;
305
306                let registry = self.dispatcher.deterministic_registry().ok_or_else(|| {
307                    ToolError::Internal {
308                        message: alloc::format!(
309                            "agent invoked step tool {:?} but the dispatcher \
310                             did not provide a DeterministicRegistry",
311                            self.tool_id,
312                        ),
313                    }
314                })?;
315
316                let result = registry
317                    .invoke(&step.fn_ref.name, &json_args)
318                    .map_err(|e| ToolError::Internal {
319                        message: alloc::format!(
320                            "deterministic step {:?} (fn={:?}) failed: {e}",
321                            self.tool_id,
322                            step.fn_ref.name,
323                        ),
324                    })?;
325
326                // Same body-shape extraction as the Native/Mcp arm so the
327                // LLM sees identically-shaped tool results regardless of
328                // ToolImpl variant.
329                let text = match result {
330                    serde_json::Value::String(s) => s,
331                    v => alloc::format!("{v}"),
332                };
333                Ok(ToolResult::new(
334                    alloc::vec![ToolContent::Text { text }],
335                    false,
336                ))
337            }
338        }
339    }
340
341    async fn teardown(&self, _session: Self::Session) -> Result<(), ToolError> {
342        Ok(())
343    }
344}

346// ---------------------------------------------------------------------------
347// ToolSpec constructor (bypasses #[non_exhaustive] via serde)
348// ---------------------------------------------------------------------------
349
350/// Construct a `ToolSpec` from its fields.
351///
352/// `ToolSpec` is `#[non_exhaustive]` and has no public constructor outside
353/// `tau-ports`. We construct it by deserializing from a `serde_json::Value`
354/// that carries the three fields — this is the canonical escape hatch for
355/// `#[non_exhaustive]` types that have serde derives.
356fn make_tool_spec(name: &str, description: &str, input_schema: &serde_json::Value) -> ToolSpec {
357    // `ToolSpec.input_schema` is `tau_domain::Value` (not serde_json::Value).
358    // We round-trip through JSON to get the domain value.
359    let input_schema_domain = json_to_domain_value(input_schema.clone());
360    let json = serde_json::json!({
361        "name": name,
362        "description": description,
363        "input_schema": input_schema_domain,
364    });
365    serde_json::from_value(json).unwrap_or_else(|_| {
366        // Absolute fallback: empty-schema spec so the tool still registers.
367        let fallback = serde_json::json!({
368            "name": name,
369            "description": description,
370            "input_schema": tau_domain::Value::Object(Default::default()),
371        });
372        serde_json::from_value(fallback).expect("fallback ToolSpec must deserialize")
373    })
374}

376// ---------------------------------------------------------------------------
377// Synthesise a minimal PackageManifest from thin air (via serde)
378// ---------------------------------------------------------------------------
379
380/// Build a stub `PackageManifest` for IR agents (no plugin, no deps,
381/// no capabilities).
382///
383/// The manifest is needed as a passthrough container for the kernel's
384/// capability-check plumbing; IR agents declare their capabilities via
385/// the IR capability table, not via this manifest.
386///
387/// `UncheckedManifest` is `#[non_exhaustive]`, so we construct it via
388/// serde JSON deserialization — the same escape hatch used for other
389/// non-exhaustive domain types.
390fn stub_manifest() -> PackageManifest {
391    // `UncheckedManifest` is `#[non_exhaustive]` (no external struct-literal
392    // construction allowed), so we go through its serde `Deserialize` impl.
393    //
394    // Root cause of the original wasm panic: `PackageSource::from_str` for
395    // `https://…` URLs requires the `package-source` feature (which enables
396    // `url::Url`). Without it, `source` deserialization returns `MalformedUrl`
397    // and `.expect(…)` panics. Fix: use an scp-style address
398    // (`example.com:example/ir.git`) — always parseable, no feature gate.
399    let json = serde_json::json!({
400        "name": "ir-agent",
401        "version": "0.0.0",
402        "description": "Synthesised manifest for IR interpreter agent",
403        "authors": [],
404        "source": "example.com:example/ir.git",
405        "kind": "tool",
406        "dependencies": [],
407        "capabilities": [],
408        "sandbox": {}
409    });
410    let unchecked: UncheckedManifest =
411        serde_json::from_value(json).expect("stub manifest JSON must deserialize");
412    unchecked
413        .validate()
414        .expect("stub manifest should always validate")
415}

417// ---------------------------------------------------------------------------
418// run_agent — the main entry
419// ---------------------------------------------------------------------------
420
421/// Construct everything needed to drive one `Agent` node through the kernel
422/// agent loop, returning the pieces instead of running them.
423///
424/// This is the single shared construction path for both the batch entry
425/// ([`run_agent`], via `Runtime::run_with_history`) and the streaming entry
426/// ([`run_agent_streaming`], via `Runtime::run_streaming_with_history`). The
427/// two MUST NOT drift: any change to backend wiring, tool registration,
428/// `AgentDefinition` synthesis, `RunOptions` (clock/random injection + the
429/// β.4 context pipeline), or history splitting belongs here so both callers
430/// inherit it identically.
431///
432/// Behaviour is identical to the former inline body of `run_agent`; this is a
433/// pure extraction (steps 1–7), with the final `rt.run_with_history(...)`
434/// call left to the caller.
435#[allow(clippy::type_complexity)]
436fn prepare_agent_run<D>(
437    module: alloc::sync::Arc<IrModule>,
438    agent: &Agent,
439    dispatcher: Arc<D>,
440    initial_messages: Vec<Message>,
441) -> Result<
442    (
443        Runtime,
444        AgentDefinition,
445        PackageManifest,
446        Vec<Message>,
447        Message,
448        RunOptions,
449    ),
450    RuntimeError,
451>
452where
453    D: ToolDispatcher + Send + Sync + 'static,
454{
455    // 1. Obtain the LLM backend from the dispatcher, keyed by the IR backend name.
456    let backend = dispatcher.llm_backend_for(&agent.model_ref.backend)?;
457    let backend_name = String::from(backend.name());
458
459    // 2. Build the RuntimeBuilder with the backend.
460    let mut builder = Runtime::builder().with_dyn_llm_backend(backend);
461
462    // 3. Register each tool in agent.tool_refs as a dispatcher-delegating wrapper.
463    for tool_id in &agent.tool_refs {
464        let ir_tool = module
465            .workflow
466            .tools
467            .get(tool_id)
468            .ok_or_else(|| RuntimeError::Internal {
469                message: alloc::format!(
470                    "agent {:?} references tool {:?} not in workflow.tools",
471                    agent.id.0,
472                    tool_id.0,
473                ),
474            })?;
475
476        let spec = make_tool_spec(
477            &ir_tool.spec.name,
478            &ir_tool.spec.description,
479            &ir_tool.spec.input_schema,
480        );
481
482        builder = builder.with_tool(DispatcherTool {
483            tool_name: ir_tool.spec.name.clone(),
484            tool_id: tool_id.clone(),
485            spec,
486            module: module.clone(),
487            tool_impl: ir_tool.impl_.clone(),
488            dispatcher: dispatcher.clone(),
489        });
490    }
491
492    // 4. Build the Runtime.
493    let rt = builder.build().map_err(|e| RuntimeError::Internal {
494        message: alloc::format!("failed to build Runtime for IR agent {:?}: {e}", agent.id.0),
495    })?;
496
497    // 5. Synthesise the AgentDefinition from the IR agent.
498    //    PackageName allows [a-z0-9-] starting with a letter — same as AgentId.
499    //    We fall back to "ir-agent" if the agent id doesn't conform.
500    let pkg_name = PackageName::from_str(&agent.id.0)
501        .unwrap_or_else(|_| PackageName::from_str("ir-agent").expect("ir-agent is always valid"));
502
503    let llm_backend_pkg_name = PackageName::from_str(&backend_name)
504        .unwrap_or_else(|_| PackageName::from_str("ir-agent").expect("ir-agent is always valid"));
505
506    let domain_agent_id = DomainAgentId::from_str(&agent.id.0)
507        .unwrap_or_else(|_| DomainAgentId::from_str("ir-agent").expect("ir-agent is always valid"));
508
509    // Resolve the agent's system prompt (D6-B). `Inline` is the literal text.
510    // `Asset` references the bundle's content-addressed asset store, supplied
511    // by the host via `ToolDispatcher::assets()`; look up the blob by hash and
512    // decode its bytes as UTF-8. A missing store or missing/!utf-8 blob is a
513    // structured error, not a panic — never trust input shape (the host's
514    // load-time closed-world check should make the missing cases unreachable).
515    let system_prompt = match &agent.prompt {
516        tau_ir::prompt::PromptSource::Inline(s) => s.clone(),
517        tau_ir::prompt::PromptSource::Asset(a) => {
518            let store = dispatcher.assets().ok_or_else(|| RuntimeError::Internal {
519                message: alloc::format!(
520                    "agent {:?}: prompt references asset {:?} but the host supplied no asset store",
521                    agent.id.0,
522                    a.asset,
523                ),
524            })?;
525            let blob = store.get(&a.asset).ok_or_else(|| RuntimeError::Internal {
526                message: alloc::format!(
527                    "agent {:?}: prompt asset {:?} not found in the bundle asset store",
528                    agent.id.0,
529                    a.asset,
530                ),
531            })?;
532            alloc::string::String::from_utf8(blob.bytes.clone()).map_err(|_| {
533                RuntimeError::Internal {
534                    message: alloc::format!(
535                        "agent {:?}: prompt asset {:?} is not valid UTF-8",
536                        agent.id.0,
537                        a.asset,
538                    ),
539                }
540            })?
541        }
542    };
543
544    let agent_def = AgentDefinition::new(
545        domain_agent_id,
546        agent.id.0.clone(),
547        PackageId::new(
548            pkg_name,
549            Version::parse("0.0.0").expect("0.0.0 is always valid"),
550        ),
551        llm_backend_pkg_name,
552    )
553    .with_system_prompt(system_prompt)
554    .with_model(agent.model_ref.model_id.clone());
555
556    let manifest = stub_manifest();
557
558    // 6. Build RunOptions from the agent's budget.
559    let mut run_options = RunOptions::default();
560    if let Some(max_turns) = agent.budget.max_turns {
561        run_options.max_turns = max_turns;
562    }
563    // Prefer the host shell's clock/random, supplied through the
564    // dispatcher (e.g. tau-cli's `ForwardingDispatcher` returns
565    // `TokioClock` / `OsRandom`). This is how a production
566    // (non-`test-fixtures`) build of `run_ir` / `run_pipeline` obtains a
567    // real wall-clock and entropy source — the kernel is `no_std` and
568    // cannot mint them itself. Dispatchers that don't override these
569    // return `None` (the trait default), in which case the test-fixtures
570    // injection below applies, or the panic guard fires.
571    if run_options.clock.is_none() {
572        run_options.clock = dispatcher.clock();
573    }
574    if run_options.random.is_none() {
575        run_options.random = dispatcher.random();
576    }
577    // Inject test-fixture clock/random when neither the host shell nor the
578    // dispatcher provided them (e.g. tau-ir-conformance drives run_ir
579    // directly without the tokio shell drive entry). The test-fixtures
580    // feature is the sanctioned escape hatch; matching the
581    // spawn_root_agent_inner pattern in run.rs.
582    #[cfg(feature = "test-fixtures")]
583    {
584        if run_options.clock.is_none() {
585            run_options.clock = Some(alloc::sync::Arc::new(tau_ports::MockClock::new()));
586        }
587        if run_options.random.is_none() {
588            run_options.random = Some(alloc::sync::Arc::new(
589                tau_ports::DeterministicRandom::seeded(0),
590            ));
591        }
592    }
593    #[cfg(not(feature = "test-fixtures"))]
594    {
595        if run_options.clock.is_none() {
596            panic!("prepare_agent_run: clock must be supplied unless test-fixtures is enabled");
597        }
598        if run_options.random.is_none() {
599            panic!("prepare_agent_run: random must be supplied unless test-fixtures is enabled");
600        }
601    }
602
603    // 6b. β.4: build the per-turn context pipeline from the IR agent's
604    //     config. `run_agent` is the single shared agent-loop entry for
605    //     all three drive paths (run_ir, run_pipeline, check), so wiring
606    //     here covers `tau dev`, `tau run --bundle`, pipeline steps, and
607    //     deliverable checks. Custom nodes resolve via the dispatcher's
608    //     optional registry; builtins resolve regardless. Agents without
609    //     a `context` config keep the empty default pipeline.
610    if let Some(ctx_cfg) = agent.context.as_ref() {
611        let registry = dispatcher.context_transformer_registry();
612        run_options.context_pipeline =
613            crate::context::build_context_pipeline(ctx_cfg, registry.as_deref())?;
614    }
615
616    // 6c. ADR-0053: wire durable checkpoint/resume for agents that opt in.
617    //     Gated on `agent.durable` so only the declaring agent checkpoints —
618    //     a non-durable judge / subflow / pipeline-step agent sharing this
619    //     code path is left untouched. The host dispatcher supplies the
620    //     store + run id (and any resume checkpoint); a dispatcher without a
621    //     store leaves the agent running as ordinary (non-durable).
622    if agent.durable.is_some() {
623        if let Some(handles) = dispatcher.checkpointing() {
624            run_options.checkpoint_store = Some(handles.store);
625            run_options.run_id = Some(handles.run_id);
626            run_options.resume_from = handles.resume;
627            run_options.durable_granularity = Some(handles.checkpoint);
628        }
629    }
630
631    // 7. Split initial_messages into history + initial_message.
632    //    The kernel requires exactly one initial_message; if the caller
633    //    provided none, synthesise a placeholder so the run loop has
634    //    something to send to the LLM.
635    let (history, initial_message) = split_history(initial_messages).unwrap_or_else(|| {
636        // Use the no_std-safe constructor so this path compiles without the
637        // `std` feature (required for the wasm guest that calls run_ir_streaming
638        // with `initial_messages = Vec::new()`).
639        let clock = run_options.clock.as_ref().expect("clock checked above");
640        let random = run_options.random.as_ref().expect("random checked above");
641        let placeholder = Message::new_with(
642            crate::ids::message_id(clock, random),
643            crate::ids::now_utc(clock),
644            Address::User,
645            Address::System,
646            MessagePayload::Text {
647                content: String::new(),
648            },
649        );
650        (Vec::new(), placeholder)
651    });
652
653    // 8. Hand the constructed pieces back to the caller, which drives the
654    //    kernel agent loop (batch via `run_with_history`, streaming via
655    //    `run_streaming_with_history`).
656    Ok((
657        rt,
658        agent_def,
659        manifest,
660        history,
661        initial_message,
662        run_options,
663    ))
664}

666/// Execute one `Agent` node end-to-end through the existing kernel agent loop.
667///
668/// Constructs a `Runtime` from the dispatcher's LLM backend + the agent's
669/// tool refs (each wrapped as a thin dispatcher-delegating `Tool`), then
670/// calls `run_with_history` with a synthesised `AgentDefinition` and
671/// `PackageManifest`. Construction is shared with [`run_agent_streaming`] via
672/// [`prepare_agent_run`] so the batch and streaming paths cannot drift.
673pub async fn run_agent<D>(
674    module: alloc::sync::Arc<IrModule>,
675    agent: &Agent,
676    dispatcher: Arc<D>,
677    initial_messages: Vec<Message>,
678) -> Result<RunOutcome, RuntimeError>
679where
680    D: ToolDispatcher + Send + Sync + 'static,
681{
682    let (rt, agent_def, manifest, history, initial_message, run_options) =
683        prepare_agent_run(module, agent, dispatcher, initial_messages)?;
684    rt.run_with_history(agent_def, manifest, history, initial_message, run_options)
685        .await
686}

688/// Execute one `Agent` node through the kernel agent loop, returning the
689/// uncollapsed [`crate::stream::RunEvent`] stream instead of a collapsed
690/// [`RunOutcome`].
691///
692/// Construction is shared with [`run_agent`] via [`prepare_agent_run`] so the
693/// batch and streaming paths cannot drift. The conformance dev profile drives
694/// this to observe the per-event run (TextDelta / ToolCall* / TurnCompleted /
695/// RunCompleted) rather than the collapsed terminal outcome.
696///
697/// The returned stream is `'static`: `Runtime::run_streaming_with_history`
698/// snapshots its tool registry and backend into Arc-clones that the stream
699/// owns, so the `Runtime` built here may be dropped after this call returns.
700pub async fn run_agent_streaming<D>(
701    module: alloc::sync::Arc<IrModule>,
702    agent: &Agent,
703    dispatcher: Arc<D>,
704    initial_messages: Vec<Message>,
705) -> Result<impl futures_core::Stream<Item = crate::stream::RunEvent> + 'static, RuntimeError>
706where
707    D: ToolDispatcher + Send + Sync + 'static,
708{
709    let (rt, agent_def, manifest, history, initial_message, run_options) =
710        prepare_agent_run(module, agent, dispatcher, initial_messages)?;
711    rt.run_streaming_with_history(agent_def, manifest, history, initial_message, run_options)
712        .await
713}

715/// Split a `Vec<Message>` into `(history, last)`, returning `None` when empty.
716fn split_history(mut messages: Vec<Message>) -> Option<(Vec<Message>, Message)> {
717    if messages.is_empty() {
718        return None;
719    }
720    let last = messages.pop().expect("non-empty checked above");
721    Some((messages, last))
722}
