24// Region admission counters
25// ---------------------------------------------------------------------------

27/// Per-`Dynamic` step bounds accounting: total spawns ever admitted and
28/// currently in-flight spawns. Shared (`Arc`) across every `SpawnTool`
29/// offered by one dynamic region so all offered kinds draw from the same
30/// pool.
31pub(crate) struct RegionCounters {
32    max_spawns: u64,
33    max_concurrency: u64,
34    spawned: AtomicU64,
35    in_flight: AtomicU64,
36}
37
38/// Typed admission refusal — carries the snapshot needed to render both
39/// the soft-deny tool-result text and the `runtime.dynamic.spawn_denied`
40/// tracing event.
41#[derive(Debug, Clone, Copy)]
42pub(crate) enum AdmitError {
43    /// The region's `max_spawns` (lifetime spawn count) is exhausted.
44    Bounds { spawned: u64, max: u64 },
45    /// The region's `max_concurrency` (simultaneously in-flight spawns) is
46    /// exhausted. Defensive only — unreachable under today's sequential
47    /// per-turn tool dispatch (no two `invoke()` calls run concurrently
48    /// against the same region), but guards the invariant if that changes.
49    Concurrency { in_flight: u64, max: u64 },
50}
51
52impl RegionCounters {
53    pub(crate) fn new(max_spawns: u64, max_concurrency: u64) -> Self {
54        Self {
55            max_spawns,
56            max_concurrency,
57            spawned: AtomicU64::new(0),
58            in_flight: AtomicU64::new(0),
59        }
60    }
61
62    /// Admit one spawn: returns its 0-based index, or the typed refusal.
63    ///
64    /// Reserves a `spawned` slot via a compare-exchange loop (bounds
65    /// check), then bumps `in_flight` and verifies it didn't overshoot
66    /// `max_concurrency` — rolling the `in_flight` bump back on overshoot
67    /// (the `spawned` reservation is NOT rolled back: a concurrency
68    /// refusal still counts against the region's lifetime spawn budget,
69    /// matching `max_spawns` semantics of "attempts", not "successes").
70    pub(crate) fn try_admit(&self) -> Result<u64, AdmitError> {
71        let index = loop {
72            let cur = self.spawned.load(Ordering::SeqCst);
73            if cur >= self.max_spawns {
74                return Err(AdmitError::Bounds {
75                    spawned: cur,
76                    max: self.max_spawns,
77                });
78            }
79            if self
80                .spawned
81                .compare_exchange(cur, cur + 1, Ordering::SeqCst, Ordering::SeqCst)
82                .is_ok()
83            {
84                break cur;
85            }
86        };
87
88        let in_flight = self.in_flight.fetch_add(1, Ordering::SeqCst) + 1;
89        if in_flight > self.max_concurrency {
90            self.in_flight.fetch_sub(1, Ordering::SeqCst);
91            return Err(AdmitError::Concurrency {
92                in_flight: in_flight - 1,
93                max: self.max_concurrency,
94            });
95        }
96        Ok(index)
97    }
98
99    /// Paired with every successful `try_admit`; call when the child finishes.
100    pub(crate) fn release(&self) {
101        self.in_flight.fetch_sub(1, Ordering::SeqCst);
102    }
103}
104
105/// RAII guard pairing a successful `try_admit` with `release()`.
106///
107/// A bare `counters.release()` statement placed after an `.await` is only
108/// reached if that `.await` returns normally — a panic unwinding through it
109/// (the workspace default is `panic = "unwind"`) or the `invoke()` future
110/// being dropped without being polled to completion (cancellation, a
111/// `select!`, an external timeout) both skip it, permanently leaking an
112/// `in_flight` slot and wedging the region at `max_concurrency` for the
113/// rest of its lifetime. Binding this guard immediately after a successful
114/// `try_admit` and holding it across the child run makes `release()` run
115/// on every exit path via `Drop`, unwind or not. No `unsafe`; `AtomicU64`
116/// stays `Send + Sync` under `no_std`.
117struct ReleaseOnDrop<'a>(&'a RegionCounters);
118
119impl Drop for ReleaseOnDrop<'_> {
120    fn drop(&mut self) {
121        self.0.release();
122    }
123}

125// ---------------------------------------------------------------------------
126// Attenuation: child grant = meet(region envelope, offered kind's caps)
127// ---------------------------------------------------------------------------
128
129/// The child agent's capability grant: the meet (greatest lower bound) of
130/// the dynamic region's envelope and the spawned kind's declared
131/// capabilities. Composes with `AttenuatedDispatcher`'s own frame-nesting
132/// meet exactly like a subflow frame — see
133/// `docs/superpowers/specs/2026-07-19-subflow-runtime-attenuation-design.md`.
134pub(crate) fn child_grant(
135    envelope: &CapabilityRequirements,
136    kind_caps: &CapabilityRequirements,
137) -> CapabilityRequirements {
138    CapabilityRequirements {
139        declared: tau_domain::package::capability::lattice::meet(
140            &envelope.declared,
141            &kind_caps.declared,
142        ),
143    }
144}

152fn json_to_domain_value(v: serde_json::Value) -> tau_domain::Value {

154}

156fn make_tool_spec(name: &str, description: &str, input_schema: &serde_json::Value) -> ToolSpec {
157    let input_schema_domain = json_to_domain_value(input_schema.clone());
158    let json = serde_json::json!({
159        "name": name,
160        "description": description,
161        "input_schema": input_schema_domain,
162    });
163    serde_json::from_value(json).unwrap_or_else(|_| {
164        let fallback = serde_json::json!({
165            "name": name,
166            "description": description,
167            "input_schema": tau_domain::Value::Object(Default::default()),
168        });
169        serde_json::from_value(fallback).expect("fallback ToolSpec must deserialize")
170    })
171}

177/// A `tau_ports::Tool` implementing `agent.<kind>.spawn`: the LLM-facing
178/// surface for spawning a dynamic-region child agent of one offered
179/// `kind`. `invoke()` is the runtime gate: bounds/concurrency admission →

181pub(crate) struct SpawnTool<D> {
182    spawn: tau_ir::pipeline::DynamicSpawn,
183    envelope: CapabilityRequirements,
184    counters: Arc<RegionCounters>,
185    region_step: String,
186    module: Arc<IrModule>,
187    dispatcher: Arc<D>,
188    tool_name: String,
189}
190

200        module: Arc<IrModule>,
201        dispatcher: Arc<D>,

207            counters,
208            region_step,
209            module,

216impl<D> tau_ports::tool::Tool for SpawnTool<D>
217where
218    D: ToolDispatcher + Send + Sync + 'static,
219{
220    type Session = ();
221
222    fn name(&self) -> &str {
223        &self.tool_name
224    }
225
226    fn schema(&self) -> ToolSpec {
227        make_tool_spec(
228            &self.tool_name,
229            &self.spawn.description,
230            &serde_json::json!({
231                "type": "object",
232                "properties": { "message": { "type": "string" } },
233                "required": ["message"]
234            }),
235        )
236    }
237
238    async fn init(&self, _ctx: SessionContext) -> Result<Self::Session, ToolError> {
239        Ok(())
240    }
241
242    async fn invoke(
243        &self,
244        _session: &mut Self::Session,
245        args: tau_domain::Value,
246    ) -> Result<ToolResult, ToolError> {
247        // 1. Parse `message: String` from args.
248        let message = match args
249            .as_object()
250            .and_then(|o| o.get("message"))
251            .and_then(|v| v.as_string())
252        {
253            Some(m) => m.to_string(),
254            None => {
255                return Ok(ToolResult::new(
256                    alloc::vec![ToolContent::Text {
257                        text: alloc::format!("{}: missing required arg `message`", self.tool_name),
258                    }],
259                    true,
260                ));
261            }
262        };
263
264        // 2. Bounds/concurrency admission gate — denied BEFORE any child is
265        //    constructed.
266        let n = match self.counters.try_admit() {
267            Ok(n) => n,
268            Err(err) => {
269                // Neutral `count`/`limit` field names (not `spawned`/`max_spawns`)
270                // because this arm covers BOTH refusal kinds and each binds a
271                // different pair of counters: Bounds's are the lifetime
272                // spawn count vs. `max_spawns`; Concurrency's are the
273                // in-flight count vs. `max_concurrency`. A single field name
274                // reused across both with only one arm's semantics (as
275                // before) silently mislabels the other arm's tracing event —
276                // a real defect in a security-relevant admission-denial log.
277                let (reason, count, limit, text) = match err {
278                    AdmitError::Bounds { spawned, max } => (
279                        "bounds",
280                        spawned,
281                        max,
282                        alloc::format!(
283                            "spawn denied: region `{}` max_spawns exhausted ({spawned}/{max}) — kind `{}`; proceed with the results you have",
284                            self.region_step, self.spawn.kind,
285                        ),
286                    ),
287                    AdmitError::Concurrency { in_flight, max } => (
288                        "concurrency",
289                        in_flight,
290                        max,
291                        alloc::format!(
292                            "spawn denied: region `{}` max_concurrency exceeded ({in_flight}/{max}) — kind `{}`; proceed with the results you have",
293                            self.region_step, self.spawn.kind,
294                        ),
295                    ),
296                };
297                tracing::warn!(
298                    name = "runtime.dynamic.spawn_denied",
299                    region_step = %self.region_step,
300                    kind = %self.spawn.kind,
301                    reason = %reason,
302                    count = count,
303                    limit = limit,
304                );
305                return Ok(ToolResult::new(
306                    alloc::vec![ToolContent::Text { text }],
307                    true,
308                ));
309            }
310        };
311        // RAII: pairs this successful admission with `release()` on every
312        // exit path (normal return, early `?`, or panic unwind) — see
313        // `ReleaseOnDrop`'s doc comment.
314        let _release_guard = ReleaseOnDrop(&self.counters);
315
316        // 3-5. Child id, meet-attenuated grant, child Agent node.
317        let child_id = alloc::format!("{}:{}#{n}", self.region_step, self.spawn.kind);
318        let grant = child_grant(&self.envelope, &self.spawn.capabilities);
319        let child_agent = tau_ir::node::Agent {
320            id: tau_ir::ids::AgentId(child_id.clone()),
321            prompt: self.spawn.prompt.clone(),
322            model_ref: self.spawn.model_ref.clone(),
323            tool_refs: self.spawn.tool_refs.clone(),
324            context: None,
325            budget: tau_ir::budget::AgentBudget {
326                max_turns: None,
327                max_tokens: None,
328            },
329            produces: alloc::vec::Vec::new(),
330            output_schema: None,
331            durable: None,
332        };
333
334        // 6. Admission accepted. `index` (not `spawned`, which the denial
335        //    event above uses for a *count*) — `n` is the 0-based admission
336        //    index, a different quantity from a spawned-count snapshot.
337        tracing::info!(
338            name = "runtime.dynamic.spawned",
339            region_step = %self.region_step,
340            kind = %self.spawn.kind,
341            child_id = %child_id,
342            index = n,
343            max_spawns = self.counters.max_spawns,
344        );
345
346        // 7. Attenuated dispatcher: the child (and its descendants) run
347        //    under `grant`, with its own dynamic-region denial event name.
348        let att = Arc::new(AttenuatedDispatcher::new_with_event(
349            grant,
350            ToolId(self.tool_name.clone()),
351            child_id.clone(),
352            self.module.clone(),
353            self.dispatcher.clone(),
354            "runtime.dynamic.attenuation_denied",
355        ));
356
357        // 8. Run the child agent. `_release_guard` releases the in_flight
358        //    slot on drop — reached on every exit path, not just the
359        //    fall-through after this `.await`.
360        let user_msg = user_message(&message);
361        let outcome = alloc::boxed::Box::pin(run_agent(
362            self.module.clone(),
363            &child_agent,
364            att,
365            alloc::vec![user_msg],
366        ))
367        .await;
368
369        // 9. Map the outcome to a ToolResult.
370        let (is_error, text) = match &outcome {
371            Err(e) => (true, alloc::format!("spawn `{child_id}` failed: {e}")),
372            Ok(o @ RunOutcome::Failed { status, .. }) => {
373                let last_text = last_assistant_text(o);
374                let detail = if last_text.is_empty() {
375                    alloc::format!("{status:?}")
376                } else {
377                    last_text
378                };
379                (true, alloc::format!("spawn `{child_id}` failed: {detail}"))
380            }
381            Ok(o) => (false, last_assistant_text(o)),
382        };
383
384        Ok(ToolResult::new(
385            alloc::vec![ToolContent::Text { text }],
386            is_error,
387        ))
388    }
389
390    async fn teardown(&self, _session: Self::Session) -> Result<(), ToolError> {
391        Ok(())
392    }
393}

594    #[derive(Default)]
595    struct SpawnDeniedFields {
596        reason: Option<String>,
597        count: Option<u64>,
598        limit: Option<u64>,
599    }
600
601    impl Visit for SpawnDeniedFields {
602        fn record_u64(&mut self, field: &Field, value: u64) {
603            match field.name() {
604                "count" => self.count = Some(value),
605                "limit" => self.limit = Some(value),
606                _ => {}
607            }
608        }
609        fn record_i64(&mut self, field: &Field, value: i64) {
610            if value < 0 {
611                return;
612            }
613            match field.name() {
614                "count" => self.count = Some(value as u64),
615                "limit" => self.limit = Some(value as u64),
616                _ => {}
617            }
618        }
619        fn record_str(&mut self, field: &Field, value: &str) {
620            if field.name() == "reason" {
621                self.reason = Some(value.to_string());
622            }
623        }
624        fn record_debug(&mut self, field: &Field, value: &dyn core::fmt::Debug) {
625            let raw = alloc::format!("{value:?}");
626            match field.name() {
627                "reason" => self.reason = Some(raw.trim_matches('"').to_string()),
628                "count" => {
629                    if let Ok(n) = raw.trim().parse::<u64>() {
630                        self.count = Some(n);
631                    }
632                }
633                "limit" => {
634                    if let Ok(n) = raw.trim().parse::<u64>() {
635                        self.limit = Some(n);
636                    }
637                }
638                _ => {}
639            }
640        }
641    }

643    /// `(reason, count, limit)` captured from one `runtime.dynamic.spawn_denied`
644    /// event — named to satisfy clippy's `type_complexity` lint on
645    /// [`CapturedSpawnDenials`]'s field.
646    type SpawnDenialRecord = (Option<String>, Option<u64>, Option<u64>);
647
648    #[derive(Default, Clone)]
649    struct CapturedSpawnDenials(Arc<Mutex<alloc::vec::Vec<SpawnDenialRecord>>>);
650
651    impl<S: Subscriber> Layer<S> for CapturedSpawnDenials {
652        fn on_new_span(&self, _attrs: &Attributes<'_>, _id: &Id, _ctx: LayerContext<'_, S>) {}
653
654        fn on_event(&self, event: &Event<'_>, _ctx: LayerContext<'_, S>) {
655            let mut visitor = SpawnDeniedFields::default();
656            event.record(&mut visitor);
657            if visitor.reason.is_some() {
658                self.0
659                    .lock()
660                    .expect("captured-events mutex poisoned")
661                    .push((visitor.reason, visitor.count, visitor.limit));
662            }
663        }
664    }

666    #[tokio::test]
667    async fn spawn_denied_concurrency_event_reports_in_flight_not_max_spawns() {
668        // IMPORTANT-2 regression: the concurrency-denial arm previously
669        // reused the bounds arm's `spawned`/`max_spawns` field names while
670        // emitting `in_flight`/`max_concurrency` values — mislabeling a
671        // security-relevant admission-denial event. Assert the emitted
672        // event's structured fields directly (not just its message text).
673        let captured = CapturedSpawnDenials::default();
674        let _tracing_guard = tracing_subscriber::registry()
675            .with(captured.clone())
676            .set_default();
677
678        let counters = Arc::new(RegionCounters::new(4, 1));
679        // Saturate concurrency without releasing — spawned=1, in_flight=1.
680        counters.try_admit().expect("first admit");
681        let tool = SpawnTool::new(
682            test_spawn("researcher"),
683            CapabilityRequirements::default(),
684            counters,
685            "fanout".into(),
686            test_module(),
687            Arc::new(PanicDispatcher),
688        );
689        let mut session = ();
690        tool.invoke(
691            &mut session,
692            domain_args(serde_json::json!({"message": "go"})),
693        )
694        .await
695        .expect("soft-deny returns Ok(ToolResult)");
696
697        let events = captured.0.lock().expect("poisoned").clone();
698        let (reason, count, limit) = events
699            .iter()
700            .find(|(reason, ..)| reason.as_deref() == Some("concurrency"))
701            .cloned()
702            .expect("expected a concurrency spawn_denied event");
703        assert_eq!(reason.as_deref(), Some("concurrency"));
704        // `count` must be the in-flight snapshot (1), and `limit` must be
705        // `max_concurrency` (1) — NOT `max_spawns` (4), which the old
706        // mislabeled `max_spawns` field name would have implied.
707        assert_eq!(count, Some(1), "count must report in_flight, not spawned");
708        assert_eq!(
709            limit,
710            Some(1),
711            "limit must report max_concurrency (1), not max_spawns (4)"
712        );
713    }
