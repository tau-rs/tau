1//! v0 partial-interpret driver for `tau_ir::IrModule`.
2//!
3//! Per the design spec D-5, v0 (β.2) carries the IR as data and runs it
4//! through this interpreter. The interpreter is a thin layer over the
5//! existing `Runtime` agent loop — for each agent node, it builds a
6//! `Runtime` configured with the agent's tools (resolved via the
7//! caller's tool registry) and dispatches its budget.
8//!
9//! The same module is what `tau dev` calls (with callbacks-for-tools)
10//! and what the bundle's wasm component calls (with WASI- / tau-host-
11//! gated tool dispatch). The interpreter does not distinguish; the
12//! difference lives in the `ToolDispatcher` implementation the caller
13//! supplies.
14

34/// Drive an `IrModule` from its single entry agent to completion.
35///
36/// `entry` names which agent in the module to start with. Future v0.x
37/// will infer it from a `[workflow]` block; v0.0 requires the caller
38/// to supply it.
39///
40/// `module` is taken as `Arc<IrModule>` so the dispatcher's
41/// `DispatcherTool` can share it across recursive subflow invocations
42/// without copying the IR.
43pub async fn run_ir<D>(
44    module: alloc::sync::Arc<IrModule>,
45    entry: &AgentId,
46    dispatcher: Arc<D>,
47    initial_messages: Vec<Message>,
48) -> Result<RunOutcome, RuntimeError>
49where
50    D: tool_dispatch::ToolDispatcher + Send + Sync + 'static,
51{
52    let agent_node =
53        module
54            .workflow
55            .agents
56            .get(entry)
57            .ok_or_else(|| RuntimeError::AgentNotFound {
58                agent: entry.0.clone(),
59            })?;
60    // Clone the Agent node out of the Arc so the borrow doesn't escape;
61    // it's small (id + prompt + a few Vec<...>) and avoids a self-borrow
62    // through run_agent's signature.
63    let agent_node = agent_node.clone();
64    agent_loop::run_agent(module, &agent_node, dispatcher, initial_messages).await
65}

67/// Drive an `IrModule` from its single entry agent, returning the
68/// uncollapsed [`crate::stream::RunEvent`] stream instead of a collapsed
69/// [`RunOutcome`].
70///
71/// Streaming counterpart of [`run_ir`]: identical entry-agent lookup, but
72/// delegates to [`agent_loop::run_agent_streaming`] so the conformance dev
73/// profile can observe the per-event run (the stream terminates with exactly
74/// one `RunEvent::RunCompleted`). The returned stream is `'static` (see
75/// [`agent_loop::run_agent_streaming`]).
76pub async fn run_ir_streaming<D>(
77    module: alloc::sync::Arc<IrModule>,
78    entry: &AgentId,
79    dispatcher: Arc<D>,
80    initial_messages: Vec<Message>,
81) -> Result<impl futures_core::Stream<Item = crate::stream::RunEvent> + 'static, RuntimeError>
82where
83    D: tool_dispatch::ToolDispatcher + Send + Sync + 'static,
84{
85    let agent_node =
86        module
87            .workflow
88            .agents
89            .get(entry)
90            .ok_or_else(|| RuntimeError::AgentNotFound {
91                agent: entry.0.clone(),
92            })?;
93    // Clone the Agent node out of the Arc so the borrow doesn't escape
94    // (same rationale as `run_ir`).
95    let agent_node = agent_node.clone();
96    agent_loop::run_agent_streaming(module, &agent_node, dispatcher, initial_messages).await
97}
