1//! Agent-loop methods on [`Runtime`] and pure helper functions.
2//!
3//! This module contains the methods that can run in any async executor
4//! (no tokio specifics), plus the pure free-function helpers used by
5//! the streaming pump (`stream.rs` in the host shell).
6//!
7//! # What lives here vs. the host shell
8//!
9//! - [`Runtime::invoke_tool`] — single-tool direct dispatch; no LLM loop.
10//! - [`build_policy_denied_outcome`], [`agent_messages_to_provider_messages`],
11//!   [`flatten_content_to_string`], [`content_to_value`] — pure helpers used
12//!   by the streaming pump.
13//! - `narrowed_capability_for_session` — stays in `tau-runtime-tokio::run`
14//!   until `capability_override` migrates to core (it consumes a
15//!   `tau-pkg::EffectiveCapability`, which tau-pkg owns).
16//!
17//! - `run`, `run_with_history`, `run_default`, `spawn_root_agent` —
18//!   **stay in the host shell** until `stream.rs` and the orchestration
19//!   submodules migrate to core (Tasks 3.6/3.7).
20//!
21//! # Error routing
22//!
23//! Methods return `crate::error::RuntimeError` (the core error). Host-shell
24//! callers whose return type is the shell-level `RuntimeError` get automatic
25//! `?` conversion via `#[from] CoreRuntimeError`.

64    /// Invoke a single tool by name without engaging the LLM loop.
65    ///
66    /// Bypasses the multi-turn agent driver — useful for callers that
67    /// want to compose tools directly (e.g., `tau-workflow`'s
68    /// `tool.call` step kind). The tool's capability requirements are
69    /// still checked against the `agent_def`'s package grant set, so
70    /// the caller must pass the workflow's default-agent definition.
71    ///
72    /// Follows the same sequence as the run loop's tool-dispatch arm:
73    /// `resolve_tool → capability check → init → invoke → teardown`.
74    ///
75    /// The `clock` and `random` parameters supply entropy for minting the
76    /// tool session ID. Pass `None` to fall back to deterministic
77    /// test-fixture defaults (acceptable in tests; production callers
78    /// must supply real implementations via their shell's `drive` entry
79    /// point).
80    ///
81    /// # Errors
82    ///
83    /// - [`RuntimeError::ToolNotRegistered`] — the tool name is unknown.
84    /// - [`RuntimeError::Internal`] — the agent's package does not grant
85    ///   a capability required by the tool.
86    /// - [`RuntimeError::Tool`] — the tool's `init`, `invoke`, or
87    ///   `teardown` returned a [`tau_ports::ToolError`].
88    ///
89    /// # Example
90    ///
91    /// ```rust,ignore
92    /// // Runtime is #[non_exhaustive]; construct via builder.
93    /// let result = runtime
94    ///     .invoke_tool(&agent_def, &manifest, "echo", Value::Null, None, None)
95    ///     .await?;
96    /// ```
97    #[instrument(
98        name = "dispatch.tool",
99        skip_all,
100        fields(tool_name = %tool_name),
101    )]
102    pub async fn invoke_tool_with(
103        &self,
104        agent_def: &tau_domain::AgentDefinition,
105        package_manifest: &PackageManifest,
106        tool_name: &str,
107        args: tau_domain::Value,
108        clock: Option<Arc<dyn tau_ports::Clock>>,
109        random: Option<Arc<dyn tau_ports::RandomSource>>,
110    ) -> Result<tau_ports::ToolResult, RuntimeError> {
111        use tau_ports::SessionContext;
112
113        let tool = self.resolve_tool(tool_name)?.clone();
114        debug!(
115            name = "dispatch.tool_resolved",
116            tool_name = %tool_name,
117            plugin_id = %tool.name(),
118        );
119
120        // Capability check: mirror the run loop's structural check.
121        let granted: Vec<Capability> = package_manifest.capabilities().to_vec();
122        let required: &[Capability] = tool.capabilities();
123        if let Some(missing) = crate::capability::check_capabilities(&granted, required) {
124            let denial = CapabilityDenial::new(
125                agent_def.id.to_string(),
126                agent_def.package.name.to_string(),
127                tool_name.to_owned(),
128                capability_kind_str(missing),
129                alloc::format!("{missing:?}"),
130            );
131            return Err(RuntimeError::Internal {
132                message: alloc::format!("capability denied: {denial}"),
133            });
134        }
135
136        // Mint a session UUID from the random source. When the caller
137        // passes `None` (backwards-compat / test path), fall back to
138        // the test-fixture DeterministicRandom if available, otherwise
139        // produce a nil UUID. Production callers must supply a real
140        // RandomSource via their shell's `drive` entry point.
141        // Resolve the entropy source once, then mint both the session UUID
142        // and the agent-instance id from the `Clock`/`RandomSource` ports so
143        // ids are reproducible under deterministic ports (the no_std kernel
144        // never reaches `AgentInstanceId::new`, which is std-only). When the
145        // caller passes `None`, fall back to the test-fixture
146        // DeterministicRandom if available, otherwise a nil/zero id.
147        let instance_millis = clock.as_ref().map(|c| c.now().max(0) as u64).unwrap_or(0);
148        let (session_uuid, instance_id) = match random {
149            Some(ref r) => {
150                let mut rb = [0u8; 10];
151                r.fill(&mut rb);
152                (
153                    crate::ids::uuid_v4(r),
154                    AgentInstanceId::from_parts(instance_millis, rb),
155                )
156            }
157            None => {
158                #[cfg(any(test, feature = "test-fixtures"))]
159                {
160                    let r: Arc<dyn tau_ports::RandomSource> =
161                        Arc::new(tau_ports::DeterministicRandom::seeded(0));
162                    let mut rb = [0u8; 10];
163                    r.fill(&mut rb);
164                    (
165                        crate::ids::uuid_v4(&r),
166                        AgentInstanceId::from_parts(instance_millis, rb),
167                    )
168                }
169                #[cfg(not(any(test, feature = "test-fixtures")))]
170                {
171                    (
172                        uuid::Uuid::nil(),
173                        AgentInstanceId::from_parts(instance_millis, [0u8; 10]),
174                    )
175                }
176            }
177        };
178
179        // Build a minimal SessionContext (no deny entries). `deadline` is a
180        // `process`-feature-gated field (no_std hosts have no `SystemTime`),
181        // so the constructor arity differs by feature.
182        #[cfg(feature = "process")]
183        let ctx =
184            SessionContext::new(instance_id, session_uuid, None).with_granted_capabilities(granted);
185        #[cfg(not(feature = "process"))]
186        let ctx = SessionContext::new(instance_id, session_uuid).with_granted_capabilities(granted);
187
188        tool.init(ctx.clone()).await.map_err(RuntimeError::from)?;
189        let result = tool.invoke(&ctx, &mut (), args).await;
190        // teardown best-effort: don't mask invoke's error if both fail.
191        let _ = tool.teardown(()).await;
192        result.map_err(RuntimeError::from)
193    }

195    /// Invoke a single tool by name, using deterministic test-fixture
196    /// defaults for clock/random. Thin 4-arg wrapper over
197    /// [`Self::invoke_tool_with`].
198    pub async fn invoke_tool(
199        &self,
200        agent_def: &tau_domain::AgentDefinition,
201        package_manifest: &PackageManifest,
202        tool_name: &str,
203        args: tau_domain::Value,
204    ) -> Result<tau_ports::ToolResult, RuntimeError> {

207    }

247    /// Stream an agent run, prepending conversation history before the
248    /// initial message.
249    pub async fn run_streaming_with_history(
250        &self,
251        agent_def: tau_domain::AgentDefinition,
252        package_manifest: PackageManifest,
253        history: Vec<Message>,
254        initial_message: Message,
255        options: crate::options::RunOptions,
256    ) -> Result<impl futures_core::Stream<Item = crate::stream::RunEvent> + 'static, RuntimeError>
257    {
258        use hashbrown::HashMap as HbHashMap;
259        use tau_ports::DenyEntry;
260
261        tracing::info!(name = "runtime.run_streaming_started");
262
263        // Step 1: Determine the effective grant set.
264        let (granted_for_kernel, granted_for_session, deny_entries) =
265            if let Some(override_grant) = options.granted_capabilities_override.as_ref() {
266                tracing::debug!(
267                    name = "runtime.streaming_capability_set_loaded_from_override",
268                    count = override_grant.len(),
269                    reason = "agent.<kind>.spawn child run",
270                );
271                let kernel: Vec<Capability> = override_grant.clone();
272                let session: Vec<Capability> = override_grant.clone();
273                (kernel, session, Vec::<DenyEntry>::new())
274            } else if let Some(resolver) = options.capability_resolver.as_ref() {
275                let resolved = resolver
276                    .resolve(package_manifest.capabilities())
277                    .map_err(|e| {
278                        tracing::warn!(
279                            name = "runtime.streaming_capability_override_rejected",
280                            agent_id = %agent_def.id,
281                            package_id = %agent_def.package.name,
282                            kind = %e.kind,
283                            reason = %e.reason,
284                        );
285                        RuntimeError::CapabilityOverrideExpands {
286                            kind: e.kind,
287                            reason: e.reason,
288                        }
289                    })?;
290                tracing::debug!(
291                    name = "runtime.streaming_capability_set_loaded",
292                    count = resolved.for_kernel.len(),
293                );
294                (
295                    resolved.for_kernel,
296                    resolved.for_session,
297                    resolved.deny_entries,
298                )
299            } else {
300                // No resolver and no override: use the manifest capabilities
301                // unchanged (no narrowing, no deny entries). Right default for
302                // shells with no override system.
303                let kernel: Vec<Capability> = package_manifest.capabilities().to_vec();
304                let session = kernel.clone();
305                tracing::debug!(
306                    name = "runtime.streaming_capability_set_loaded_passthrough",
307                    count = kernel.len(),
308                );
309                (kernel, session, Vec::<DenyEntry>::new())
310            };
311        let granted: &[Capability] = &granted_for_kernel;
312
313        // Step 5: Resolve LLM backend.
314        let backend = self
315            .resolve_llm_backend(agent_def.id.as_str(), agent_def.llm_backend.as_str())?
316            .clone();
317
318        // Step 6: Build capability-filtered tool_specs.
319        let mut tool_specs: Vec<tau_ports::ToolSpec> = Vec::with_capacity(self.tools().len());
320        for (name, tool) in self.tools().iter() {
321            let required = tool.capabilities();
322            if crate::capability::check_capabilities(granted, required).is_none() {
323                tool_specs.push(tool.schema());
324            } else {
325                tracing::debug!(
326                    name = "runtime.streaming_tool_filtered",
327                    tool_name = name.as_str(),
328                    "tool filtered out: missing capability",
329                );
330            }
331        }
332
333        // Step 7: Snapshot tools registry (Arc-clones).
334        let tools: HbHashMap<alloc::string::String, Arc<dyn crate::builder::DynTool>> = self
335            .tools()
336            .iter()
337            .map(|(k, v)| (k.clone(), v.clone()))
338            .collect();
339
340        // Step 8: Snapshot tool_validators registry.
341        // When `tool-validation` is enabled the validators are pre-compiled
342        // at Runtime::build time and stored in the registry; without
343        // `tool-validation` (e.g. the wasm guest) no schema validation runs.
344        #[cfg(feature = "tool-validation")]
345        let tool_validators: HbHashMap<
346            alloc::string::String,
347            crate::tool_args::ToolArgsValidator,
348        > = self.tool_validators().clone().into_iter().collect();
349        #[cfg(not(feature = "tool-validation"))]
350        let tool_validators: HbHashMap<alloc::string::String, ()> = HbHashMap::default();
351
352        // Step 9: Construct and return the stream.
353        let stream = crate::stream::run_streaming_inner(
354            backend,
355            agent_def,
356            package_manifest,
357            history,
358            initial_message,
359            options,
360            tools,
361            tool_validators,
362            granted_for_kernel,
363            tool_specs,
364            deny_entries,
365            granted_for_session,
366        );
367        Ok(stream)
368    }

390    /// Run an agent with a pre-existing conversation history. See `run.rs`
391    /// module docs for the full loop semantics, error/failure dichotomy,
392    /// and tracing vocabulary.
393    #[instrument(
394        name = "runtime.agent_run",
395        skip_all,
396        fields(
397            agent_id = %agent_def.id,
398            display_name = %agent_def.display_name,
399            package_id = %agent_def.package.name,
400            llm_backend_name = %agent_def.llm_backend,
401            max_turns = options.max_turns,
402            history_len = history.len(),
403        ),
404    )]
405    pub async fn run_with_history(
406        &self,
407        agent_def: tau_domain::AgentDefinition,
408        package_manifest: PackageManifest,
409        history: Vec<Message>,
410        initial_message: Message,
411        options: crate::options::RunOptions,
412    ) -> Result<RunOutcome, RuntimeError> {
413        use crate::stream::RunEvent;
414        use futures_core::Stream as _;
415        use tau_ports::{LlmError, ToolError};
416
417        // Capture the orchestration trace context BEFORE `agent_def` and
418        // `options` are moved into the streaming delegate. When this run is
419        // part of a multi-agent orchestration (an `orchestration_state` with
420        // its `TraceStream` is present, and the host injected clock+random),
421        // the collector below translates the pump's per-turn
422        // `RunEvent::TurnCompleted` and terminal `RunCompleted` into the
423        // multi-agent `Turn`/`Completion` trace events that
424        // `output_orchestration::run_printer` aggregates into the per-agent
425        // summary (issue #469). Every agent — root, spawned children, and
426        // interpreter-driven agents — funnels through this one method, so a
427        // single emit site covers them all. Non-orchestrated single-agent
428        // runs leave `orchestration_state` unset and skip emission entirely.
429        let trace_ctx = match (
430            options.orchestration_state.clone(),
431            options.clock.clone(),
432            options.random.clone(),
433        ) {
434            (Some(state), Some(clock), Some(random)) => {
435                Some((state, clock, random, agent_def.id.to_string()))
436            }
437            _ => None,
438        };
439        // Wall-clock mark for per-turn duration: initialised to run start,
440        // advanced on each `TurnCompleted`.
441        let mut last_turn_mark = trace_ctx
442            .as_ref()
443            .map(|(_, clock, _, _)| crate::ids::now_utc(clock));
444
445        // Delegate all agent-loop logic to run_streaming_with_history.
446        let stream = self
447            .run_streaming_with_history(
448                agent_def,
449                package_manifest,
450                history,
451                initial_message,
452                options,
453            )
454            .await?;
455        let mut stream = alloc::boxed::Box::pin(stream);
456        loop {
457            let next = core::future::poll_fn(|cx| stream.as_mut().poll_next(cx)).await;
458            match next {
459                Some(RunEvent::TurnCompleted { turn, usage, .. }) => {
460                    // Translate the pump's per-turn signal into a multi-agent
461                    // `Turn` trace event, timed off the injected clock.
462                    if let Some((state, clock, random, agent_id)) = trace_ctx.as_ref() {
463                        let now = crate::ids::now_utc(clock);
464                        let duration_ms = last_turn_mark
465                            .map(|prev| (now - prev).num_milliseconds().max(0) as u64)
466                            .unwrap_or(0);
467                        last_turn_mark = Some(now);
468                        // input + output, matching how `aggregated_tokens`
469                        // (and thus the Summary header `tokens_used`) sums.
470                        let tokens = usage
471                            .map(|u| u64::from(u.input_tokens) + u64::from(u.output_tokens))
472                            .unwrap_or(0);
473                        let s = state.borrow();
474                        s.trace.emit(tau_ports::TraceEvent {
475                            id: crate::ids::ulid(clock, random),
476                            ts: now,
477                            run_id: s.run_id.clone(),
478                            agent_id: Some(agent_id.clone()),
479                            kind: tau_ports::TraceEventKind::Turn {
480                                agent_id: agent_id.clone(),
481                                // `RunEvent::TurnCompleted.turn` is 1-indexed;
482                                // `TraceEventKind::Turn.turn_index` is 0-based.
483                                turn_index: turn.saturating_sub(1),
484                                duration_ms,
485                                tokens,
486                            },
487                        });
488                    }
489                    continue;
490                }
491                Some(RunEvent::RunCompleted { outcome }) => {
492                    // Emit the agent's `Completion` before returning so the
493                    // per-agent summary has an entry even for a zero-turn or
494                    // failed agent.
495                    if let Some((state, clock, random, agent_id)) = trace_ctx.as_ref() {
496                        let status = if matches!(outcome, RunOutcome::Completed { .. }) {
497                            "completed"
498                        } else {
499                            "failed"
500                        };
501                        let s = state.borrow();
502                        s.trace.emit(tau_ports::TraceEvent {
503                            id: crate::ids::ulid(clock, random),
504                            ts: crate::ids::now_utc(clock),
505                            run_id: s.run_id.clone(),
506                            agent_id: Some(agent_id.clone()),
507                            kind: tau_ports::TraceEventKind::Completion {
508                                agent_id: agent_id.clone(),
509                                status: status.to_string(),
510                            },
511                        });
512                    }
513                    return Ok(outcome);
514                }
515                Some(RunEvent::FatalError {
516                    kind,
517                    detail,
518                    context_json,
519                    tool_error_variant,
520                }) => {
521                    return Err(match kind.as_str() {
522                        "ToolNotRegistered" => {
523                            let (tool_name, registered) = context_json
524                                .as_deref()
525                                .and_then(|s| serde_json::from_str::<serde_json::Value>(s).ok())
526                                .and_then(|v| {
527                                    let tn = v["tool_name"].as_str()?.to_owned();
528                                    let reg: Vec<alloc::string::String> = v["registered"]
529                                        .as_array()?
530                                        .iter()
531                                        .filter_map(|x| x.as_str().map(alloc::string::String::from))
532                                        .collect();
533                                    Some((tn, reg))
534                                })
535                                .unwrap_or_else(|| (detail.clone(), alloc::vec![]));
536                            RuntimeError::ToolNotRegistered {
537                                tool_name,
538                                registered,
539                            }
540                        }
541                        "Llm" => RuntimeError::Llm(LlmError::Internal { message: detail }),
542                        "Tool" => {
543                            let tool_err = match tool_error_variant.as_deref() {
544                                Some("BadArgs") => ToolError::BadArgs { reason: detail },
545                                Some("SessionDead") => ToolError::SessionDead { reason: detail },
546                                Some("DeadlineExceeded") => ToolError::DeadlineExceeded,
547                                Some("CapabilityDenied") => {
548                                    ToolError::CapabilityDenied { capability: detail }
549                                }
550                                _ => ToolError::Internal { message: detail },
551                            };
552                            RuntimeError::Tool(tool_err)
553                        }
554                        "ContextPipeline" => RuntimeError::ContextPipeline { detail },
555                        _ => RuntimeError::Internal { message: detail },
556                    });
557                }
558                Some(_) => continue,
559                None => unreachable!(
560                    "run_streaming_inner must yield exactly one RunCompleted before stream end"
561                ),
562            }
563        }
564    }

590    ///
591    /// The kernel-resident half of the spawn_root_agent flow. Host shells
592    /// pass pre-attached trace subscribers (e.g. a tokio JSONL writer)

594    /// `Vec::new()`. The full `spawn_root_agent_with_scope` wrapper in

597    #[allow(clippy::too_many_arguments)]
598    pub async fn spawn_root_agent_inner(
599        self: Arc<Self>,
600        root_agent_def: tau_domain::AgentDefinition,
601        root_manifest: PackageManifest,
602        initial_message: Message,
603        budget: tau_ports::RunBudget,
604        subscribers: Vec<Arc<dyn crate::orchestration::TraceSubscriber>>,
605        clock: Option<Arc<dyn tau_ports::Clock>>,
606        random: Option<Arc<dyn tau_ports::RandomSource>>,
607        // Optional pre-minted run-id. When `Some`, the inner uses it
608        // verbatim — host shells supply this so the JSONL log path and
609        // the run-id inside trace events agree.
610        run_id_override: Option<alloc::string::String>,
611        // Project-scope root as a String. Host shells convert their PathBuf
612        // via `path.to_string_lossy().into_owned()`. Used by the streaming
613        // pump's skill-spawn intercept to resolve installed skills.
614        scope_root: Option<alloc::string::String>,
615        // Skill resolver for `skill.<name>.spawn` dispatch. Host shells
616        // build a `TauPkgSkillResolver` from their scope; guest shells
617        // pass `None` (or a `NoSkillResolver`). Carried into `RunOptions`.
618        skill_resolver: Option<Arc<dyn tau_ports::SkillResolver>>,
619    ) -> Result<tau_ports::RunSnapshot, RuntimeError> {
620        use core::cell::RefCell;
621
622        // Default clock/random for test paths if caller passes None.
623        let clock = clock.unwrap_or_else(|| {
624            #[cfg(feature = "test-fixtures")]
625            {
626                Arc::new(tau_ports::MockClock::new()) as Arc<dyn tau_ports::Clock>
627            }
628            #[cfg(not(feature = "test-fixtures"))]
629            {
630                panic!(
631                    "spawn_root_agent_inner: clock must be supplied unless test-fixtures \
632                     is enabled"
633                );
634            }
635        });
636        let random = random.unwrap_or_else(|| {
637            #[cfg(feature = "test-fixtures")]
638            {
639                Arc::new(tau_ports::DeterministicRandom::seeded(0))
640                    as Arc<dyn tau_ports::RandomSource>
641            }
642            #[cfg(not(feature = "test-fixtures"))]
643            {
644                panic!(
645                    "spawn_root_agent_inner: random must be supplied unless test-fixtures \
646                     is enabled"
647                );
648            }
649        });
650
651        let run_id = run_id_override.unwrap_or_else(|| crate::ids::ulid(&clock, &random));
652        let root_agent_id = root_agent_def.id.to_string();
653        let now = crate::ids::now_utc(&clock);
654
655        let mut state = crate::orchestration::run_state::RunState::new(
656            run_id.clone(),
657            root_agent_id,
658            budget,
659            now,
660        );
661
662        // Attach pre-built subscribers (JSONL writer, etc).
663        for sub in subscribers {
664            state.trace.add_subscriber(sub);
665        }
666
667        // Arc<RefCell<RunState>>: kernel futures are non-Send by design
668        // (see builder::BoxFuture alias), so RunState is single-task and
669        // RefCell is the honest representation. Arc is kept (not Rc) so the
670        // refcount machinery is uniform with the rest of the kernel surface.
671        #[allow(clippy::arc_with_non_send_sync)]
672        let state_arc = Arc::new(RefCell::new(state));
673
674        let opts = crate::options::RunOptions {
675            orchestration_state: Some(state_arc.clone()),
676            orchestration_runtime: Some(self.clone()),
677            clock: Some(clock.clone()),
678            random: Some(random.clone()),
679            scope_root,
680            skill_resolver,
681            ..Default::default()
682        };
683
684        let outcome = self
685            .run_with_history(
686                root_agent_def,
687                root_manifest,
688                Vec::new(),
689                initial_message,
690                opts,
691            )
692            .await?;
693
694        let now_end = crate::ids::now_utc(&clock);
695        {
696            let mut s = state_arc.borrow_mut();
697            s.ended_at = Some(now_end);
698            let success = matches!(outcome, RunOutcome::Completed { .. });
699            let orphans_present = !s.task_list.all_terminal();
700            s.status = if success && !orphans_present {
701                tau_ports::RunStatus::Completed
702            } else {
703                tau_ports::RunStatus::Failed
704            };
705            if orphans_present {
706                let orphan_ids: Vec<_> = s
707                    .task_list
708                    .all()
709                    .into_iter()
710                    .filter(|t| {
711                        !matches!(
712                            t.status,
713                            tau_ports::TaskStatus::Done
714                                | tau_ports::TaskStatus::Failed
715                                | tau_ports::TaskStatus::Discarded
716                        )
717                    })
718                    .map(|t| t.id)
719                    .collect();
720                s.trace.emit(tau_ports::TraceEvent {
721                    id: crate::ids::ulid(&clock, &random),
722                    ts: now_end,
723                    run_id: run_id.clone(),
724                    agent_id: None,
725                    kind: tau_ports::TraceEventKind::OrphanedTasksAtTermination {
726                        task_ids: orphan_ids,
727                    },
728                });
729            }
730        }
731
732        let snapshot = {
733            let s = state_arc.borrow();
734            s.snapshot(now_end)
735        };
736        Ok(snapshot)
737    }
738}

761/// Project the agent's [`Message`] history onto the LLM-call shape.
762///
763/// Per `tau_ports::llm` module-level docs, `tau_domain::Message`
764/// (universal envelope) and [`LlmProviderMessage`] (provider call
765/// shape) are intentionally distinct. This function is the single
766/// projection point in the kernel.
767pub fn agent_messages_to_provider_messages(history: &[Message]) -> Vec<LlmProviderMessage> {
768    let mut out = Vec::with_capacity(history.len());
769    // Provider id of the most recent tool-call, so each tool-result
770    // references the `tool_use` it answers rather than its own message id.
771    // Anthropic rejects a `tool_result` whose `tool_use_id` matches no
772    // preceding `tool_use.id` (orphaned pair) on multi-turn tool runs.
773    // A result message's `parent_id` (the message it replies to) is the
774    // authoritative link; when absent we fall back to the nearest preceding
775    // tool-call — the run loop pushes each call immediately before its
776    // result, so calls and results strictly alternate in the history.
777    let mut last_tool_use_id: Option<String> = None;
778    for m in history {
779        match (&m.sender, &m.payload) {
780            (tau_domain::Address::User, MessagePayload::Text { content }) => {
781                out.push(LlmProviderMessage::user(vec![ContentBlock::Text(
782                    content.clone(),
783                )]));
784            }
785            (tau_domain::Address::Agent(_), MessagePayload::Text { content }) => {
786                out.push(LlmProviderMessage::assistant(vec![ContentBlock::Text(
787                    content.clone(),
788                )]));
789            }
790            (tau_domain::Address::Agent(_), MessagePayload::ToolCall { args }) => {
791                let tool_name = match &m.recipient {
792                    tau_domain::Address::Tool(name) => name.clone(),
793                    _ => String::new(),
794                };
795                let tool_use_id = alloc::format!("toolu_{}", m.id);
796                last_tool_use_id = Some(tool_use_id.clone());
797                out.push(LlmProviderMessage::assistant(vec![ContentBlock::ToolUse(
798                    ToolUse::new(tool_use_id, tool_name, args.clone()),
799                )]));
800            }
801            (tau_domain::Address::Tool(_), MessagePayload::ToolResult { body }) => {
802                out.push(LlmProviderMessage::tool_result(
803                    answered_tool_use_id(m, last_tool_use_id.as_deref()),
804                    vec![ContentBlock::Text(value_to_preview_string(body))],
805                    false,
806                ));
807            }
808            (
809                tau_domain::Address::Tool(_),
810                MessagePayload::ToolError {
811                    kind: _,
812                    message,
813                    details: _,
814                },
815            ) => {
816                out.push(LlmProviderMessage::tool_result(
817                    answered_tool_use_id(m, last_tool_use_id.as_deref()),
818                    vec![ContentBlock::Text(message.clone())],
819                    true,
820                ));
821            }
822            _ => {}
823        }
824    }
825    out
826}
827
828/// Provider `tool_use_id` for a tool-result/error message: prefer the
829/// message's `parent_id` (authoritative link to the tool-call it answers),
830/// then the nearest preceding tool-call id, then — for a malformed history
831/// with neither — the result's own id (degenerate, preserves old behavior).
832fn answered_tool_use_id(m: &Message, last_tool_use_id: Option<&str>) -> String {
833    match &m.parent_id {
834        Some(parent) => alloc::format!("toolu_{parent}"),
835        None => last_tool_use_id
836            .map(String::from)
837            .unwrap_or_else(|| alloc::format!("toolu_{}", m.id)),
838    }
839}

841/// Flatten a tool's content blocks into a single human-readable string.
842pub fn flatten_content_to_string(blocks: &[ToolContent]) -> String {
843    let mut out = String::new();
844    for block in blocks {
845        match block {
846            ToolContent::Text { text } => {
847                if !out.is_empty() {
848                    out.push('\n');
849                }
850                out.push_str(text);
851            }
852            ToolContent::Json { data } => {
853                if !out.is_empty() {
854                    out.push('\n');
855                }
856                out.push_str(&value_to_preview_string(data));
857            }
858            _ => {}
859        }
860    }
861    out
862}

864/// Build a [`Value`] from a tool's content blocks.
865pub fn content_to_value(blocks: &[ToolContent]) -> Value {
866    if blocks.len() == 1 {
867        if let ToolContent::Json { data } = &blocks[0] {
868            return data.clone();
869        }
870    }
871    let arr: Vec<Value> = blocks
872        .iter()
873        .map(|b| match b {
874            ToolContent::Text { text } => Value::String(text.clone()),
875            ToolContent::Json { data } => data.clone(),
876            _ => Value::Null,
877        })
878        .collect();
879    let mut obj = BTreeMap::new();
880    obj.insert("content".to_string(), Value::Array(arr));
881    Value::Object(obj)
882}
883
884/// Compact preview string for a [`Value`].
885pub fn value_to_preview_string(v: &Value) -> String {
886    match v {
887        Value::String(s) => s.clone(),
888        Value::Null => "null".to_string(),
889        Value::Bool(b) => b.to_string(),
890        Value::Integer(i) => i.to_string(),
891        Value::Float(f) => f.to_string(),
892        Value::Bytes(b) => alloc::format!("<{} bytes>", b.len()),
893        Value::Array(_) | Value::Object(_) => alloc::format!("{v:?}"),
894        _ => alloc::format!("{v:?}"),
895    }
896}
