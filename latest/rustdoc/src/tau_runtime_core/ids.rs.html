1//! UUID / ULID / now-timestamp helpers routed through `Clock` and `RandomSource` ports.
2//!
3//! These are the only places in the kernel that materialize a session
4//! ID, run ID, trace ID, or wall-clock timestamp. The kernel never
5//! reaches `chrono::Utc::now()` / `ulid::Ulid::generate()` /
6//! `uuid::Uuid::new_v4()` directly — every call site routes through
7//! these helpers, fed by ports.
8
9use alloc::string::String;
10use alloc::sync::Arc;
11
12use tau_ports::{Clock, RandomSource};

14/// Mint a UUID v4 from the supplied `RandomSource`. Returns the
15/// canonical 36-character hyphenated form.
16///
17/// # Example
18///
19/// ```
20/// use tau_runtime_core::ids::uuid_v4;
21/// use tau_ports::DeterministicRandom;
22/// use std::sync::Arc;
23///
24/// let random: Arc<dyn tau_ports::RandomSource> = Arc::new(DeterministicRandom::seeded(42));
25/// let id = uuid_v4(&random);
26/// let s = id.to_string();
27/// assert_eq!(s.len(), 36);
28/// // version nibble must be 4 (UUID v4)
29/// assert_eq!(&s[14..15], "4");
30/// ```
31pub fn uuid_v4(random: &Arc<dyn RandomSource>) -> uuid::Uuid {
32    let mut bytes = [0u8; 16];
33    random.fill(&mut bytes);
34    // Variant + version bits per RFC 4122 §4.4.
35    bytes[6] = (bytes[6] & 0x0F) | 0x40;
36    bytes[8] = (bytes[8] & 0x3F) | 0x80;
37    uuid::Uuid::from_bytes(bytes)
38}

40/// Mint a ULID from the supplied `Clock` + `RandomSource`. Returns the
41/// base32-encoded canonical form (26 chars).
42///
43/// # Example
44///
45/// ```
46/// use tau_runtime_core::ids::ulid;
47/// use tau_ports::{MockClock, DeterministicRandom};
48/// use std::sync::Arc;
49///
50/// let clock: Arc<dyn tau_ports::Clock> = Arc::new(MockClock::at(1_000));
51/// let random: Arc<dyn tau_ports::RandomSource> = Arc::new(DeterministicRandom::seeded(7));
52/// let id = ulid(&clock, &random);
53/// assert_eq!(id.len(), 26);
54/// ```
55pub fn ulid(clock: &Arc<dyn Clock>, random: &Arc<dyn RandomSource>) -> String {
56    let ts = clock.now().max(0) as u64;
57    let mut rand_bytes = [0u8; 10];
58    random.fill(&mut rand_bytes);
59    let mut b = [0u8; 16];
60    b[..10].copy_from_slice(&rand_bytes);
61    let rand_part = u128::from_le_bytes(b);
62    // `to_str` (no_std-safe) rather than `Display`/`to_string` (ulid gates
63    // its `Display` impl behind its `std` feature, which this no_std kernel
64    // does not enable). A canonical ULID is always 26 base32 chars.
65    let mut buf = [0u8; 26];
66    let s = ::ulid::Ulid::from_parts(ts, rand_part).array_to_str(&mut buf);
67    String::from(&*s)
68}

70/// Wall-clock now as `chrono::DateTime<Utc>` from the supplied `Clock`.
71///
72/// # Example
73///
74/// ```
75/// use tau_runtime_core::ids::now_utc;
76/// use tau_ports::MockClock;
77/// use std::sync::Arc;
78///
79/// let clock: Arc<dyn tau_ports::Clock> = Arc::new(MockClock::new());
80/// let dt = now_utc(&clock);
81/// // MockClock increments per call; verify post-epoch + non-negative.
82/// assert!(dt.timestamp_millis() > 0);
83/// ```
84pub fn now_utc(clock: &Arc<dyn Clock>) -> chrono::DateTime<chrono::Utc> {
85    chrono::DateTime::<chrono::Utc>::from_timestamp_millis(clock.now())
86        .unwrap_or_else(|| chrono::DateTime::<chrono::Utc>::from_timestamp_millis(0).unwrap())
87}

89/// Mint a [`tau_domain::MessageId`] (UUID v7) from the `Clock` + `RandomSource`
90/// ports. The no_std kernel uses this instead of `MessageId::new` (which is
91/// std-only and reads the ambient clock/RNG) so message ids are reproducible
92/// under deterministic ports — the property wasm conformance relies on.
93pub fn message_id(clock: &Arc<dyn Clock>, random: &Arc<dyn RandomSource>) -> tau_domain::MessageId {
94    let mut rand_bytes = [0u8; 10];
95    random.fill(&mut rand_bytes);
96    tau_domain::MessageId::from_parts(clock.now().max(0) as u64, rand_bytes)
97}
