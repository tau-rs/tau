1//! Skill resolution for `skill.<name>.spawn` virtual tool dispatch.
2//!
3//! Skills-4 (ROADMAP §16). Three pure-ish functions:
4//!
5//! - [`substitute_skill_dir`] — replace `${SKILL_DIR}` literal in
6//!   path-based capabilities.
7//! - [`apply_scope_paths`] — narrow fs.* capability paths per
8//!   caller's `scope_paths` arg.
9//! - [`resolve_skill_for_spawn`] — end-to-end: lookup + substitute +
10//!   scope + subset check. Returns a `SkillSpawnRequest` ready for
11//!   the existing v1.1 spawn machinery.
12//!   **Requires the `host-fs` feature** (returns a `PathBuf`). Skill lookup + SKILL.md reading are delegated to the injected `tau_ports::SkillResolver` port.

31/// A validated, ready-to-spawn skill invocation.
32///
33/// Produced by [`resolve_skill_for_spawn`]; consumed by the kernel's
34/// `skill.<name>.spawn` virtual-tool handler. Fields carry the
35/// fully-resolved capability grant and initial message for the child run.
36///
37/// Gated behind `host-fs` because it contains `std::path::PathBuf`.
38///
39/// # Example
40///
41/// ```
42/// use std::path::PathBuf;
43/// use tau_runtime_core::orchestration::skill_resolve::SkillSpawnRequest;
44///
45/// // Demonstrate field access (in practice, SkillSpawnRequest is produced
46/// // by resolve_skill_for_spawn, not constructed directly).
47/// let req = SkillSpawnRequest {
48///     skill_name: "critic".into(),
49///     install_path: PathBuf::from("/skills/critic"),
50///     system_prompt: "You are a critical reviewer.".into(),
51///     grant: vec![],
52///     message: "Review this draft.".into(),
53/// };
54/// assert_eq!(req.skill_name, "critic");
55/// assert_eq!(req.message, "Review this draft.");
56/// assert!(req.grant.is_empty());
57/// ```
58#[cfg(feature = "host-fs")]
59#[derive(Debug, Clone)]
60pub struct SkillSpawnRequest {
61    /// Skill name (as referenced in `skill.<name>.spawn`).
62    pub skill_name: String,
63    /// Absolute path to the installed skill directory.
64    pub install_path: std::path::PathBuf,
65    /// SKILL.md body (or caller's override if supplied).
66    pub system_prompt: String,
67    /// Resolved + narrowed capability grant for the child agent.
68    pub grant: Vec<Capability>,
69    /// Initial user message for the spawned child.
70    pub message: String,
71}

73/// Caller's spawn args.
74///
75/// # Example
76///
77/// ```
78/// use tau_runtime_core::orchestration::skill_resolve::SkillSpawnArgs;
79///
80/// let args = SkillSpawnArgs {
81///     message: "Summarise this document.".into(),
82///     system_prompt: Some("Be concise.".into()),
83///     scope_paths: Some(vec!["/workspace/docs/**".into()]),
84/// };
85/// assert_eq!(args.message, "Summarise this document.");
86/// assert_eq!(args.system_prompt.as_deref(), Some("Be concise."));
87/// ```
88#[derive(Debug, Clone, Default)]
89pub struct SkillSpawnArgs {
90    /// Required initial user message.
91    pub message: String,
92    /// Optional override for the skill's default SKILL.md body.
93    pub system_prompt: Option<String>,
94    /// Optional path narrowing. Each entry must be covered by at
95    /// least one declared fs.* path.
96    pub scope_paths: Option<Vec<String>>,
97}

99/// Substitute `${SKILL_DIR}` literal in capability `paths` entries
100/// with `install_path`. Non-path capabilities (net.http,
101/// task_list, plan, agent.spawn, skill.spawn, custom) pass through.
102///
103/// # Example
104///
105/// ```
106/// use tau_domain::{Capability, FsCapability, SKILL_DIR_VAR};
107/// use tau_runtime_core::orchestration::skill_resolve::substitute_skill_dir;
108///
109/// let cap: Capability = serde_json::from_value(serde_json::json!({
110///     "kind": "fs.read",
111///     "paths": [format!("{SKILL_DIR_VAR}/refs/**")]
112/// })).expect("valid capability");
113///
114/// let out = substitute_skill_dir(&[cap], "/skills/critic/1.0.0");
115/// if let Capability::Filesystem(FsCapability::Read { paths, .. }) = &out[0] {
116///     assert_eq!(paths[0], "/skills/critic/1.0.0/refs/**");
117/// }
118/// ```
119pub fn substitute_skill_dir(caps: &[Capability], install_path: &str) -> Vec<Capability> {
120    let subst = |paths: &[String]| -> Vec<String> {
121        paths
122            .iter()
123            .map(|p| p.replace(SKILL_DIR_VAR, install_path))
124            .collect()
125    };
126    caps.iter()
127        .map(|c| match c {
128            Capability::Filesystem(FsCapability::Read { paths, .. }) => {
129                // Use serde round-trip to avoid #[non_exhaustive] struct-construction ban.
130                serde_json::from_value(serde_json::json!({
131                    "kind": "fs.read",
132                    "paths": subst(paths)
133                }))
134                .unwrap_or_else(|_| c.clone())
135            }
136            Capability::Filesystem(FsCapability::Write {
137                paths, max_bytes, ..
138            }) => {
139                let mut v = serde_json::json!({
140                    "kind": "fs.write",
141                    "paths": subst(paths)
142                });
143                if let Some(mb) = max_bytes {
144                    v["max_bytes"] = serde_json::json!(mb);
145                }
146                serde_json::from_value(v).unwrap_or_else(|_| c.clone())
147            }
148            Capability::Filesystem(FsCapability::Exec { paths, .. }) => {
149                serde_json::from_value(serde_json::json!({
150                    "kind": "fs.exec",
151                    "paths": subst(paths)
152                }))
153                .unwrap_or_else(|_| c.clone())
154            }
155            other => other.clone(),
156        })
157        .collect()
158}

160/// Test if `glob` covers `candidate` (i.e. candidate is a subset).
161#[cfg(feature = "host-fs")]
162fn glob_covers(glob: &str, candidate: &str) -> bool {
163    // Strip ${SKILL_DIR} prefix if both sides have it — globset's
164    // `{` char would otherwise be parsed as alternation.
165    match GlobBuilder::new(glob).literal_separator(false).build() {
166        Ok(g) => g.compile_matcher().is_match(candidate),
167        Err(_) => false,
168    }
169}

171/// Narrow fs.* paths per caller's `scope_paths`. For each fs.*
172/// capability the skill declares, the child's effective paths =
173/// intersect_with_scope(declared, scope). If empty → drop the
174/// capability entirely. Non-fs caps pass through.
175///
176/// Hard-fail if any `scope_path` is not covered by ANY declared
177/// fs.* path (typo detection).
178///
179/// # Example
180///
181/// ```
182/// use tau_domain::Capability;
183/// use tau_runtime_core::orchestration::skill_resolve::apply_scope_paths;
184///
185/// let cap: Capability = serde_json::from_value(serde_json::json!({
186///     "kind": "fs.read", "paths": ["/workspace/**"]
187/// })).expect("valid capability");
188///
189/// // Narrowing: only keep the sub-path.
190/// let narrowed = apply_scope_paths(vec![cap], &["/workspace/project/**".to_string()])
191///     .expect("path is covered");
192/// assert_eq!(narrowed.len(), 1);
193///
194/// // Out-of-scope path → error.
195/// let cap2: Capability = serde_json::from_value(serde_json::json!({
196///     "kind": "fs.read", "paths": ["/workspace/**"]
197/// })).expect("valid capability");
198/// let err = apply_scope_paths(vec![cap2], &["/home/alice/**".to_string()]);
199/// assert!(err.is_err());
200/// ```
201#[cfg(feature = "host-fs")]
202pub fn apply_scope_paths(
203    caps: Vec<Capability>,
204    scope_paths: &[String],
205) -> Result<Vec<Capability>, OrchestrationError> {
206    if scope_paths.is_empty() {
207        return Ok(caps);
208    }
209
210    // Collect every declared fs.* path across all declared capabilities.
211    let all_fs_paths: Vec<&str> = caps
212        .iter()
213        .flat_map(|c| match c {
214            Capability::Filesystem(FsCapability::Read { paths, .. }) => paths.iter(),
215            Capability::Filesystem(FsCapability::Write { paths, .. }) => paths.iter(),
216            Capability::Filesystem(FsCapability::Exec { paths, .. }) => paths.iter(),
217            _ => [].iter(),
218        })
219        .map(|s| s.as_str())
220        .collect();
221
222    // Typo check: each scope_path must be covered by at least one declared.
223    for sp in scope_paths {
224        let covered = all_fs_paths.iter().any(|d| glob_covers(d, sp));
225        if !covered {
226            return Err(OrchestrationError::SkillScopePathNotCovered { path: sp.clone() });
227        }
228    }
229
230    let intersect = |declared: &[String]| -> Vec<String> {
231        scope_paths
232            .iter()
233            .filter(|sp| declared.iter().any(|d| glob_covers(d, sp)))
234            .cloned()
235            .collect()
236    };
237
238    let mut out = Vec::new();
239    for c in caps {
240        match c {
241            Capability::Filesystem(FsCapability::Read { paths, .. }) => {
242                let narrowed = intersect(&paths);
243                if !narrowed.is_empty() {
244                    // Use serde round-trip to avoid #[non_exhaustive] ban.
245                    if let Ok(cap) = serde_json::from_value(serde_json::json!({
246                        "kind": "fs.read",
247                        "paths": narrowed
248                    })) {
249                        out.push(cap);
250                    }
251                }
252            }
253            Capability::Filesystem(FsCapability::Write {
254                paths, max_bytes, ..
255            }) => {
256                let narrowed = intersect(&paths);
257                if !narrowed.is_empty() {
258                    let mut v = serde_json::json!({
259                        "kind": "fs.write",
260                        "paths": narrowed
261                    });
262                    if let Some(mb) = max_bytes {
263                        v["max_bytes"] = serde_json::json!(mb);
264                    }
265                    if let Ok(cap) = serde_json::from_value(v) {
266                        out.push(cap);
267                    }
268                }
269            }
270            Capability::Filesystem(FsCapability::Exec { paths, .. }) => {
271                let narrowed = intersect(&paths);
272                if !narrowed.is_empty() {
273                    if let Ok(cap) = serde_json::from_value(serde_json::json!({
274                        "kind": "fs.exec",
275                        "paths": narrowed
276                    })) {
277                        out.push(cap);
278                    }
279                }
280            }
281            other => out.push(other),
282        }
283    }
284    Ok(out)
285}

287/// End-to-end resolution. Looks up the skill via the injected
288/// [`SkillResolver`] port, substitutes `${SKILL_DIR}`, narrows by
289/// `scope_paths`, verifies the subset law, returns the request.
290///
291/// `parent_grant` is the parent agent's effective capability grant —
292/// used for the v1.1 capability subset law.
293///
294/// **Requires the `host-fs` feature**: returns a [`SkillSpawnRequest`]
295/// whose `install_path` is a `std::path::PathBuf`. Guest shells without
296/// `host-fs` cannot construct the request and never call this; they ship
297/// `tau_ports::NoSkillResolver` so a `skill.<name>.spawn` fails gracefully.
298#[cfg(feature = "host-fs")]
299pub fn resolve_skill_for_spawn(
300    skill_name: &str,
301    args: &SkillSpawnArgs,
302    parent_grant: &[Capability],
303    resolver: &dyn SkillResolver,
304) -> Result<SkillSpawnRequest, OrchestrationError> {
305    use tau_ports::SkillResolveError;
306
307    let resolved = resolver.resolve(skill_name).map_err(|e| match e {
308        SkillResolveError::NotFound => OrchestrationError::SkillNotInstalled {
309            name: skill_name.to_string(),
310        },
311        SkillResolveError::InstallPathMissing { expected_path } => {
312            OrchestrationError::SkillInstallPathMissing {
313                name: skill_name.to_string(),
314                expected_path: std::path::PathBuf::from(expected_path),
315            }
316        }
317        SkillResolveError::Invalid { detail } => OrchestrationError::SkillContentInvalid {
318            name: skill_name.to_string(),
319            detail,
320        },
321    })?;
322
323    // 1. system_prompt: caller override OR the resolved SKILL.md body.
324    let system_prompt = args.system_prompt.clone().unwrap_or(resolved.system_prompt);
325
326    // 2. ${SKILL_DIR} substitution in capabilities.
327    let install_path = resolved.install_path;
328    let substituted = substitute_skill_dir(&resolved.capabilities, &install_path);
329
330    // 3. Apply caller's scope_paths if provided.
331    let scoped = if let Some(sp) = &args.scope_paths {
332        apply_scope_paths(substituted, sp)?
333    } else {
334        substituted
335    };
336
337    // 4. Subset law: child grant ⊆ parent grant.
338    check_capability_subset(parent_grant, &scoped)?;
339
340    Ok(SkillSpawnRequest {
341        skill_name: skill_name.to_string(),
342        install_path: std::path::PathBuf::from(install_path),
343        system_prompt,
344        grant: scoped,
345        message: args.message.clone(),
346    })
347}

357    fn fs_read(paths: Vec<&str>) -> Capability {
358        let paths_json: Vec<serde_json::Value> = paths
359            .iter()
360            .map(|p| serde_json::Value::String(p.to_string()))
361            .collect();
362        serde_json::from_value(serde_json::json!({
363            "kind": "fs.read",
364            "paths": paths_json
365        }))
366        .unwrap()
367    }

370        let paths_json: Vec<serde_json::Value> = paths

381    fn net_http(hosts: Vec<&str>) -> Capability {
382        let hosts_json: Vec<serde_json::Value> = hosts
383            .iter()
