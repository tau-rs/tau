1//! Typed errors raised by multi-agent orchestration operations.
2
3use alloc::string::String;
4use alloc::vec::Vec;
5
6use tau_ports::{AgentId, TaskId};
7
8/// Errors surfaced by virtual-tool dispatch + TaskList state transitions.
9///
10/// # Example
11///
12/// ```
13/// use tau_runtime_core::orchestration::error::OrchestrationError;
14///
15/// let not_found = OrchestrationError::TaskNotFound { task: "99".into() };
16/// assert!(not_found.to_string().contains("99"));
17/// assert!(not_found.to_string().contains("not found"));
18///
19/// let budget_err = OrchestrationError::BudgetExceeded {
20///     budget: "max_total_tokens".into(),
21///     value: 150,
22///     limit: 100,
23/// };
24/// assert!(budget_err.to_string().contains("max_total_tokens"));
25/// ```
26#[derive(Debug, thiserror::Error)]
27#[non_exhaustive]
28pub enum OrchestrationError {
29    /// `agent.<kind>.spawn`: the parent agent's grant does not include
30    /// the requested child kind in its `Agent::Spawn { allowed_kinds }`.
31    #[error("agent {parent:?} not authorized to spawn kind {kind:?}")]
32    SpawnNotAuthorized {
33        /// Parent agent id.
34        parent: AgentId,
35        /// Requested child kind.
36        kind: String,
37    },
38
39    /// Generic capability check failure for a virtual tool call.
40    #[error("agent {agent:?} lacks capability {needed}")]
41    CapabilityMissing {
42        /// The agent.
43        agent: AgentId,
44        /// Description of the missing capability.
45        needed: String,
46    },
47
48    /// `task.claim`: task is currently owned by another agent and its
49    /// lease has not expired.
50    #[error("task {task:?} already locked by {by:?} until {until}")]
51    TaskLocked {
52        /// Task id.
53        task: TaskId,
54        /// Current lock holder.
55        by: AgentId,
56        /// Lease expiry (RFC 3339).
57        until: String,
58    },
59
60    /// `task.update` / `task.complete` / `task.heartbeat`: the caller is
61    /// not the current owner.
62    #[error("task {task:?}: agent {agent:?} is not the owner")]
63    NotTaskOwner {
64        /// Task id.
65        task: TaskId,
66        /// Calling agent.
67        agent: AgentId,
68    },
69
70    /// `task.get` / `task.update`: id doesn't exist.
71    #[error("task {task:?} not found")]
72    TaskNotFound {
73        /// Task id.
74        task: TaskId,
75    },
76
77    /// Invalid state transition (e.g. claim a task that's already Done).
78    #[error("task {task:?}: cannot transition to {target}")]
79    InvalidTaskTransition {
80        /// Task id.
81        task: TaskId,
82        /// Target status.
83        target: String,
84    },
85
86    /// Budget exceeded; aborting.
87    #[error("budget {budget} exceeded: {value} / {limit}")]
88    BudgetExceeded {
89        /// Budget name.
90        budget: String,
91        /// Final value.
92        value: u64,
93        /// Limit.
94        limit: u64,
95    },
96
97    /// Capability subset law violated at agent.spawn time: child grant
98    /// contains capabilities not present in the parent's grant.
99    #[error("child grant exceeds parent grant: extra = {extras:?}")]
100    GrantNotSubset {
101        /// Capabilities in child but not in parent (JSON-serialized form).
102        extras: Vec<String>,
103    },
104
105    /// Persistence I/O failed.
106    ///
107    /// Gated behind `host-fs` because `std::io::Error` is a std type.
108    #[cfg(feature = "host-fs")]
109    #[error("orchestration persistence error: {0}")]
110    PersistenceError(#[from] std::io::Error),
111
112    /// Argument parsing failure (malformed virtual-tool args).
113    #[error("virtual tool {tool}: {detail}")]
114    ArgError {
115        /// Tool name.
116        tool: String,
117        /// Detail.
118        detail: String,
119    },
120
121    /// Skills-4: `skill.<name>.spawn` — no installed skill matches `name`.
122    #[error("skill not installed: {name:?}")]
123    SkillNotInstalled {
124        /// The unresolved skill name.
125        name: String,
126    },
127
128    /// Skills-4: skill's lockfile entry exists but install path is
129    /// missing on disk.
130    ///
131    /// Gated behind `host-fs` because `std::path::PathBuf` is a std type.
132    #[cfg(feature = "host-fs")]
133    #[error("skill {name:?} install path missing at {expected_path:?}")]
134    SkillInstallPathMissing {
135        /// Skill name.
136        name: String,
137        /// The expected install path (the manifest location).
138        expected_path: std::path::PathBuf,
139    },
140
141    /// Skills-4: SKILL.md couldn't be read or parsed.
142    #[error("skill {name:?} content invalid: {detail}")]
143    SkillContentInvalid {
144        /// Skill name.
145        name: String,
146        /// Reason (read error, YAML parse failure, missing required field).
147        detail: String,
148    },
149
150    /// Skills-4: caller's `scope_paths` includes a path not covered
151    /// by any declared fs.* path. Typo detection.
152    #[error("skill scope_path {path:?} is not covered by any declared fs.* path")]
153    SkillScopePathNotCovered {
154        /// The offending scope_path entry.
155        path: String,
156    },
157
158    /// Skills-4: parent's `Capability::Skill(SkillCapability::Spawn)`
159    /// doesn't include the requested skill in `allowed_skills`.
160    #[error("agent {parent:?} not authorized to spawn skill {name:?}")]
161    SkillSpawnNotAuthorized {
162        /// Parent agent id.
163        parent: AgentId,
164        /// The requested skill name.
165        name: String,
166    },
167}
