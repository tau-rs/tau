1//! Run-time options for the kernel and the token-usage report.
2//!
3//! `TokenUsage` is defined here for both `RunOptions` and `RunOutcome`.
4//! The full host-shell `RunOptions` (with tokio/orchestration fields) is
5//! defined in `tau-runtime`; this module carries the executor-agnostic
6//! subset that no_std kernels and host shells both consume.
7
8use alloc::string::String;
9use alloc::sync::Arc;
10use alloc::vec::Vec;
11use core::cell::RefCell;

13/// Token usage reported by the LLM backend, summed across the run.
14///
15/// Some backends report `total_tokens`; some report only input/output.
16/// `Default` returns all zeros (useful when no backend was called yet).
17///
18/// # Example
19///
20/// ```
21/// use tau_runtime_core::options::TokenUsage;
22///
23/// let usage = TokenUsage::default();
24/// assert_eq!(usage.input_tokens, 0);
25/// assert_eq!(usage.output_tokens, 0);
26/// assert_eq!(usage.total_tokens, None);
27/// ```
28#[non_exhaustive]
29#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
30#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
31#[cfg_attr(feature = "schema", schemars(rename = "RunTokenUsage"))]
32pub struct TokenUsage {
33    /// Total input (prompt) tokens used across the run.
34    pub input_tokens: u64,
35    /// Total output (completion) tokens emitted across the run.
36    pub output_tokens: u64,
37    /// `Some(n)` when the backend reports a unified count; `None` otherwise.
38    pub total_tokens: Option<u64>,
39}

41/// Executor-agnostic core options for a kernel run.
42///
43/// The full host-shell `RunOptions` in `tau-runtime` re-exports this
44/// type and extends it with tokio/orchestration fields (not visible
45/// here). Constructed via `RunOptions::default()` then mutated.
46///
47/// # Example
48///
49/// ```
50/// use tau_runtime_core::options::RunOptions;
51///
52/// let mut opts = RunOptions::default();
53/// opts.max_turns = 8;
54/// opts.trace_label = Some("session-abc".into());
55/// assert_eq!(opts.max_turns, 8);
56/// assert_eq!(opts.trace_label.as_deref(), Some("session-abc"));
57/// ```
58#[non_exhaustive]
59#[derive(Clone)]
60pub struct RunOptions {
61    /// Hard cap on agent loop iterations. Hitting this returns
62    /// `Ok(RunOutcome::Failed { kind: OutOfResources, .. })`.
63    /// Default: 16.
64    pub max_turns: u32,
65
66    /// Optional caller-supplied label included in tracing spans for
67    /// log correlation (e.g. session UUID from a TUI).
68    pub trace_label: Option<String>,
69
70    /// Set by the orchestration recursion path (v1.1+). When `Some`,
71    /// short-circuits the `compute_effective(manifest + project_override)`
72    /// calculation and uses this list directly as the agent's effective
73    /// grant. The capability subset law (`check_capability_subset`) is the
74    /// authoritative gate before this is set, so the kernel trusts it as a
75    /// pre-validated narrowing of the parent's grant.
76    pub granted_capabilities_override: Option<Vec<tau_domain::Capability>>,
77
78    /// Clock used by the runtime to stamp wall-clock times on trace events,
79    /// run snapshots, and ULID/UUID minting. Host shells inject their impl
80    /// (TokioClock on tokio, EmbassyClock on embassy). If `None`, the
81    /// kernel uses a zero-valued internal default — meaningful only for
82    /// tests; production callers must supply one through their shell's
83    /// `drive` entry point.
84    pub clock: Option<Arc<dyn tau_ports::Clock>>,
85
86    /// Random source used by the runtime to mint session IDs (UUID v4),
87    /// run IDs (ULID), trace event IDs (ULID), and other entropy consumers
88    /// in the kernel. Host shells inject their impl (OsRandom on std hosts,
89    /// HwRandom on MCU). If `None`, the kernel uses a deterministic
90    /// fixture — meaningful only for tests.
91    pub random: Option<Arc<dyn tau_ports::RandomSource>>,
92
93    /// Capability resolver used to apply project-level overrides to a
94    /// package manifest's declared capabilities. Host shells supply
95    /// their impl — tau-runtime ships `TauPkgCapabilityResolver` over
96    /// `tau_pkg::capability_override::compute_effective`; Embassy/wasm
97    /// shells either ship a passthrough impl or leave this `None`.
98    ///
99    /// When `None` AND [`Self::granted_capabilities_override`] is also
100    /// `None`, the kernel uses the manifest capabilities unchanged
101    /// (no narrowing, no deny entries). This is the right default for
102    /// shells with no override system.
103    pub capability_resolver: Option<Arc<dyn tau_ports::CapabilityResolver>>,
104
105    /// Skill resolver used to look up installed skills for
106    /// `skill.<name>.spawn` virtual-tool dispatch. Host shells supply
107    /// their impl — tau-runtime-tokio ships `TauPkgSkillResolver` over
108    /// `tau_pkg::find_installed_skill`; guest (wasm/embassy) shells ship
109    /// `tau_ports::NoSkillResolver` or leave this `None`.
110    ///
111    /// When `None`, a `skill.<name>.spawn` call fails gracefully with a
112    /// "no skill resolver available" tool error (the kernel never reads
113    /// the filesystem itself). Single-agent runs leave this `None`.
114    pub skill_resolver: Option<Arc<dyn tau_ports::SkillResolver>>,
115
116    /// Set by `Runtime::spawn_root_agent` when running inside a
117    /// multi-agent orchestrated run. When present, virtual tool calls
118    /// (`task.*`, `run.*`, `agent.<kind>.spawn`) are intercepted before
119    /// plugin dispatch and routed through `crate::orchestration`.
120    /// Callers using single-agent `Runtime::run` should leave this `None`.
121    ///
122    /// `Arc<RefCell<RunState>>` (not async Mutex) is the honest
123    /// representation: the kernel futures are non-Send by design
124    /// (`builder::BoxFuture` is `Pin<Box<dyn Future>>` without a `Send`
125    /// bound), so the agent loop is single-task already.
126    pub orchestration_state: Option<Arc<RefCell<crate::orchestration::run_state::RunState>>>,
127
128    /// Set by `Runtime::spawn_root_agent` (v1.1+). Carries the
129    /// `Arc<Runtime>` recursion handle so the in-stream
130    /// `agent.<kind>.spawn` / `skill.<name>.spawn` intercept can
131    /// recursively invoke a child run on the same kernel via
132    /// `run_with_history`. `None` for single-agent runs.
133    pub orchestration_runtime: Option<Arc<crate::builder::Runtime>>,
134
135    /// Project-scope root on disk — used for skill resolution and JSONL
136    /// run-log paths. Set by the host shell's wrapper around
137    /// `spawn_root_agent` from its scope discovery.
138    ///
139    /// `String` (not `PathBuf`) so this field stays usable in `no_std +
140    /// alloc` shells with no `std::path`. Tokio host code wraps/unwraps
141    /// via `PathBuf::from` / `Path::to_str`.
142    pub scope_root: Option<String>,
143
144    /// β.4 per-turn context pipeline. Empty = no context management
145    /// (full history every turn — pre-β.4 behavior).
146    pub context_pipeline: Vec<Arc<dyn crate::context::ContextTransformer>>,
147
148    /// Token estimator used by `fit_budget`. Defaults to the heuristic.
149    pub token_estimator: Arc<dyn crate::context::TokenEstimator>,
150
151    /// Durable checkpoint store (ADR-0053). Set by the interpreter only for
152    /// agents that declare `durable` AND when the host dispatcher supplies a
153    /// store. When `Some` (with [`Self::run_id`] also set), the agent loop
154    /// persists a `TurnCheckpoint` after each `TurnCompleted`. `None` =
155    /// durability off (default, and for all non-durable agents).
156    pub checkpoint_store: Option<Arc<dyn tau_ports::orchestration::CheckpointStore>>,
157
158    /// Run id used to key checkpoints + as the `--resume` handle. Set by the
159    /// host shell alongside [`Self::checkpoint_store`].
160    pub run_id: Option<String>,
161
162    /// Rehydrated checkpoint for a resumed run (`tau run --resume`). When
163    /// `Some`, the loop seeds its history + turn counter from this checkpoint
164    /// and re-enters at the next turn, instead of starting fresh — so already
165    /// committed turns are not re-billed.
166    pub resume_from: Option<tau_ports::orchestration::TurnCheckpoint>,
167
168    /// The declaring agent's checkpoint granularity, when durable. Gates the
169    /// mid-turn (per-tool) checkpoint + resume path (ADR-0053 follow-up).
170    pub durable_granularity: Option<tau_ir::durable::CheckpointGranularity>,
171}

173impl core::fmt::Debug for RunOptions {
174    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
175        f.debug_struct("RunOptions")
176            .field("max_turns", &self.max_turns)
177            .field("trace_label", &self.trace_label)
178            .field(
179                "granted_capabilities_override",
180                &self.granted_capabilities_override,
181            )
182            .field("clock", &self.clock.as_ref().map(|_| "<Clock>"))
183            .field("random", &self.random.as_ref().map(|_| "<RandomSource>"))
184            .field(
185                "capability_resolver",
186                &self
187                    .capability_resolver
188                    .as_ref()
189                    .map(|_| "<CapabilityResolver>"),
190            )
191            .field(
192                "skill_resolver",
193                &self.skill_resolver.as_ref().map(|_| "<SkillResolver>"),
194            )
195            .field(
196                "orchestration_state",
197                &self.orchestration_state.as_ref().map(|_| "<RunState>"),
198            )
199            .field(
200                "orchestration_runtime",
201                &self.orchestration_runtime.as_ref().map(|_| "<Runtime>"),
202            )
203            .field("scope_root", &self.scope_root)
204            .field(
205                "context_pipeline",
206                &alloc::format!("<{} transformers>", self.context_pipeline.len()),
207            )
208            .field("token_estimator", &"<TokenEstimator>")
209            .field(
210                "checkpoint_store",
211                &self.checkpoint_store.as_ref().map(|_| "<CheckpointStore>"),
212            )
213            .field("run_id", &self.run_id)
214            .field("resume_from", &self.resume_from.as_ref().map(|c| c.turn))
215            .field("durable_granularity", &self.durable_granularity)
216            .finish()
217    }
218}

220impl Default for RunOptions {
221    fn default() -> Self {
222        Self {
223            max_turns: 16,
224            trace_label: None,
225            granted_capabilities_override: None,
226            clock: None,
227            random: None,
228            capability_resolver: None,
229            skill_resolver: None,
230            orchestration_state: None,
231            orchestration_runtime: None,
232            scope_root: None,
233            context_pipeline: Vec::new(),
234            token_estimator: Arc::new(crate::context::HeuristicEstimator),
235            checkpoint_store: None,
236            run_id: None,
237            resume_from: None,
238            durable_granularity: None,
239        }
240    }
241}

243#[cfg(test)]
244mod tests {

247    #[test]
248    fn run_options_default_max_turns_is_16() {
249        let opts = RunOptions::default();
250        assert_eq!(opts.max_turns, 16);
251        assert_eq!(opts.trace_label, None);
252    }
253
254    #[test]
255    #[allow(clippy::field_reassign_with_default)]
256    fn run_options_can_override_max_turns_and_trace_label() {
257        let mut opts = RunOptions::default();
258        opts.max_turns = 100;
259        opts.trace_label = Some("session-abc".into());
260        assert_eq!(opts.max_turns, 100);
261        assert_eq!(opts.trace_label.as_deref(), Some("session-abc"));
262    }
263
264    #[test]
265    fn token_usage_default_is_all_zero() {
266        let usage = TokenUsage::default();
267        assert_eq!(usage.input_tokens, 0);
268        assert_eq!(usage.output_tokens, 0);
269        assert_eq!(usage.total_tokens, None);
270    }
271
272    #[test]
273    fn token_usage_is_copy() {
274        let a = TokenUsage {
275            input_tokens: 1,
276            output_tokens: 2,
277            total_tokens: Some(3),
278        };
279        let b = a; // requires Copy
280        assert_eq!(a, b);
281    }
