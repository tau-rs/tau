5//! Fixed `&'static str` names for every span and event in ADR-0006 §3.9.
6//!
7//! Mirror of `tau_observe::vocabulary` — kept in `tau-runtime-core` so the
8//! executor-agnostic kernel does not depend on `tau-observe` (which uses
9//! tokio + tracing-subscriber and is std-only). The host shell's
10//! `tau_observe::vocabulary` remains the canonical source of truth.
11//!
12//! These constants are `#[doc(hidden)] pub` (not the usual `pub(crate)`) so
13//! the cross-crate drift test in `tau-runtime-tokio` can cross-reference
14//! every name + value. Treat them as crate-private for normal use.
15

203    #[test]
204    fn check_span_and_event_constants_equal_their_literals() {
205        assert_eq!(SPAN_PIPELINE_CHECK, "pipeline.check");
206        assert_eq!(EV_CHECK_EVALUATED, "check.evaluated");
207        assert_eq!(EV_CHECK_RETRY, "check.retry");

210    #[test]

212        assert_eq!(FN_BUILTIN_EXISTS, "__tau::goal::exists");
213        assert_eq!(FN_BUILTIN_NON_EMPTY, "__tau::goal::non_empty");
214        assert_eq!(FN_BUILTIN_EQUALS, "__tau::goal::equals");
215        assert_eq!(FN_BUILTIN_MATCHES, "__tau::goal::matches");
216        assert_eq!(FN_BUILTIN_MIN_COUNT, "__tau::goal::min_count");
217        assert_eq!(FN_BUILTIN_SCHEMA_VALID, "__tau::goal::schema_valid");
