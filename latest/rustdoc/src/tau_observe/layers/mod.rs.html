11/// Build the per-layer filter that a layer passed to
12/// [`InstallOptions::extra_layers`](crate::install::InstallOptions::extra_layers)
13/// is expected to carry: "only events on `target`, never anything more
14/// verbose than `max_level`".
15///
16/// [`crate::install::install`] attaches the caller's `EnvFilter` as a
17/// *per-layer* filter on the console sinks rather than as a global
18/// filter (tau-rs/tau#694), so extra layers sit outside it and see every
19/// event the process emits. That is what makes on-disk artifacts survive
20/// `--quiet` / `RUST_LOG=error` — and it means each extra layer must
21/// narrow the firehose itself.
22///
23/// `max_level` is not cosmetic. `Filter::max_level_hint` feeds
24/// `tracing`'s process-wide max-level cache: with a hint of `INFO`,
25/// every DEBUG/TRACE callsite in the process is disabled outright
26/// instead of being evaluated and rejected once per event. Pass the
27/// level the target's producers actually emit at — a hint that is too
28/// coarse costs performance, one that is too fine silently drops
29/// records.
30///
31/// ```
32/// use tau_observe::layers::{only_target, workflow_run_log};
33/// use tracing_subscriber::filter::LevelFilter;
34///
35/// // Wire a layer so it only ever sees its own target's events.
36/// use tracing_subscriber::Layer as _;
37/// let layer = workflow_run_log::WorkflowRunLogLayer::new("/tmp/run.jsonl".into())
38///     .with_filter(only_target(workflow_run_log::TARGET, LevelFilter::INFO));
39/// let _boxed: Box<dyn tracing_subscriber::Layer<tracing_subscriber::Registry> + Send + Sync> =
40///     Box::new(layer);
41/// ```
42pub fn only_target(
43    target: &'static str,
44    max_level: LevelFilter,
45) -> FilterFn<impl Fn(&Metadata<'_>) -> bool + Clone> {
46    filter_fn(move |meta: &Metadata<'_>| meta.target() == target).with_max_level_hint(max_level)
47}
