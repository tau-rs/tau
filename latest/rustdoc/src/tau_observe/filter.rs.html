1//! Shared `EnvFilter` builders.
2//!
3//! Every tau binary or library that initializes a tracing subscriber
4//! goes through these helpers so the resolution order (RUST_LOG > caller
5//! directive > default) is identical everywhere.
6
7use tracing_subscriber::filter::EnvFilter;
8
9/// Build an `EnvFilter` from the `RUST_LOG` env var if set, otherwise
10/// from the `fallback` directive (e.g. `"tau=info"`).
11///
12/// `RUST_LOG` is parsed verbatim. The fallback is *not* a default for a
13/// missing var key — it is the entire filter, used only when `RUST_LOG`
14/// is unset.
15///
16/// ```
17/// use tau_observe::filter::env_or_directive;
18///
19/// // Resolves to `RUST_LOG` when set, otherwise to the fallback
20/// // directive. Either way the result is a usable, non-empty filter.
21/// // (The exact RUST_LOG-vs-fallback branch is covered hermetically by
22/// // the unit tests, which guard the shared env var with a mutex; a
23/// // doctest must not mutate process-global `RUST_LOG`.)
24/// let filter = env_or_directive("tau=info");
25/// assert!(!filter.to_string().is_empty());
26/// ```
27pub fn env_or_directive(fallback: &str) -> EnvFilter {
28    if let Ok(env) = std::env::var("RUST_LOG") {
29        return EnvFilter::new(env);
30    }
31    EnvFilter::new(fallback)
32}

34#[cfg(test)]
35mod tests {
36    use super::*;
37    use std::sync::{Mutex, OnceLock};
38
39    fn rust_log_lock() -> std::sync::MutexGuard<'static, ()> {
40        static LOCK: OnceLock<Mutex<()>> = OnceLock::new();
41        LOCK.get_or_init(|| Mutex::new(()))
42            .lock()
43            .unwrap_or_else(|p| p.into_inner())
44    }
45
46    #[test]
47    fn fallback_used_when_rust_log_unset() {
48        let _g = rust_log_lock();
49        std::env::remove_var("RUST_LOG");
50        let f = env_or_directive("tau=info");
51        assert!(f.to_string().contains("tau=info"), "got: {f}");
52    }
53
54    #[test]
55    fn rust_log_overrides_fallback() {
56        let _g = rust_log_lock();
57        std::env::set_var("RUST_LOG", "my_plugin=trace");
58        let f = env_or_directive("tau=info");
59        assert!(f.to_string().contains("my_plugin=trace"), "got: {f}");
60        std::env::remove_var("RUST_LOG");
61    }
62
63    #[test]
64    fn empty_rust_log_still_overrides() {
65        let _g = rust_log_lock();
66        std::env::set_var("RUST_LOG", "");
67        let f = env_or_directive("tau=info");
68        // EnvFilter::new("") yields an empty filter — that's the intent
69        // when the user explicitly clears RUST_LOG.
70        assert!(!f.to_string().contains("tau=info"), "got: {f}");
71        std::env::remove_var("RUST_LOG");
72    }
73}
