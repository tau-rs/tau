1//! Canonical tracing-subscriber installer.
2//!
3//! Two install paths supported at v1: human-readable to stderr (CLI),
4//! and JSON to stderr (plugin SDK). Both go through [`install`] so the
5//! filter-resolution and idempotency behavior are identical.

11/// Output format for the fmt layer.
12///
13/// ```
14/// use tau_observe::install::Format;
15///
16/// assert_ne!(Format::Human, Format::Json);
17/// // Format is Copy, so it can be stored and passed cheaply.
18/// let fmt: Format = Format::Json;
19/// let fmt2 = fmt; // copy
20/// assert_eq!(fmt, fmt2);
21/// ```
22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23pub enum Format {
24    /// Human-readable (timestamp + level + target + fields + message).
25    Human,
26    /// JSON Lines, one event per line.
27    Json,
28}
29
30/// Where the subscriber writes serialized events.
31///
32/// ```
33/// use tau_observe::install::Writer;
34///
35/// assert_ne!(Writer::Stderr, Writer::Stdout);
36/// // Writer is Copy, so it can be stored and passed cheaply.
37/// let w: Writer = Writer::Stderr;
38/// let w2 = w; // copy
39/// assert_eq!(w, w2);
40/// ```
41#[derive(Debug, Clone, Copy, PartialEq, Eq)]
42pub enum Writer {
43    /// Standard error.
44    Stderr,
45    /// Standard output.
46    Stdout,
47}

49/// File rotation policy when writing to a file sink.
50///
51/// ```
52/// use tau_observe::install::Rotation;
53///
54/// // `Never` is the default.
55/// assert_eq!(Rotation::default(), Rotation::Never);
56/// // All three variants are distinct.
57/// assert_ne!(Rotation::Never, Rotation::Daily);
58/// assert_ne!(Rotation::Daily, Rotation::Hourly);
59/// ```
60#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
61pub enum Rotation {
62    /// No rotation — append forever to a single file.
63    #[default]
64    Never,
65    /// Roll over each day at UTC midnight (filename gains a date suffix).
66    Daily,
67    /// Roll over each hour (filename gains a date+hour suffix).
68    Hourly,
69}

71/// All knobs the canonical installer accepts.
72///
73/// Construct via [`InstallOptions::cli_default`] or
74/// [`InstallOptions::plugin_sdk`], then override individual fields as
75/// needed.
76///
77/// ```
78/// use tau_observe::install::{InstallOptions, Format, Writer, Rotation};
79///
80/// let opts = InstallOptions::cli_default();
81/// assert_eq!(opts.format, Format::Human);
82/// assert_eq!(opts.writer, Writer::Stderr);
83/// assert!(!opts.non_blocking);
84/// assert!(opts.file_path.is_none());
85/// assert_eq!(opts.rotation, Rotation::Never);
86/// assert!(opts.extra_layers.is_empty());
87/// ```
88pub struct InstallOptions {
89    /// Filter to apply. Build via `tau_observe::filter::env_or_directive`.
90    pub filter: EnvFilter,
91    /// Serialization format.
92    pub format: Format,
93    /// Sink.
94    pub writer: Writer,
95    /// When `true`, the fmt layer writes through a non-blocking MPSC
96    /// channel. Requires feature `non_blocking` and `file_path` to be
97    /// set; otherwise the flag is ignored (blocking install runs).
98    pub non_blocking: bool,
99    /// Optional file sink. When set together with `non_blocking`,
100    /// overrides [`Writer`]. Requires feature `non_blocking`.
101    pub file_path: Option<std::path::PathBuf>,
102    /// File rotation policy. Ignored unless `file_path` is set.
103    pub rotation: Rotation,
104    /// Extra layers to compose into the registry alongside the fmt layer.
105    /// Used by callers that need to wire custom on-disk JSONL sinks (e.g.
106    /// `WorkflowRunLogLayer`, `PluginRecordingLayer`).
107    ///
108    /// Each layer must implement `Layer<Registry>` so it composes against
109    /// the raw registry. [`Self::filter`] is attached as a *per-layer*
110    /// filter on the console sinks (fmt + OTLP), NOT as a global filter,
111    /// so these extras genuinely sit outside it: they are called for
112    /// every event the process emits, whatever `-v` / `--quiet` /
113    /// `RUST_LOG` say.
114    ///
115    /// That is the point. These layers write files the user asked for
116    /// explicitly (`--record-protocol <path>`, the `tau workflow resume`
117    /// run log); console verbosity must not decide whether those files
118    /// get written. tau-rs/tau#694 is what happens otherwise — the
119    /// `EnvFilter` used to be layered on top of the registry as a global
120    /// filter, `--quiet` (`tau=warn`) short-circuited the INFO-level
121    /// artifact events before these layers ever saw them, and `tau
122    /// workflow run --quiet` exited 0 having written nothing.
123    ///
124    /// The flip side is that each extra layer owns its own filtering.
125    /// Attach a per-layer filter with `Layer::with_filter` — see
126    /// [`crate::layers::only_target`], which also pins the
127    /// `max_level_hint` so the process-wide max level stays bounded.
128    /// A layer without one is called for every event in every crate.
129    pub extra_layers: Vec<
130        Box<dyn tracing_subscriber::Layer<tracing_subscriber::Registry> + Send + Sync + 'static>,
131    >,
132    /// When set, an OpenTelemetry layer exports spans over OTLP/gRPC.
133    /// Requires feature `otlp`.
134    #[cfg(feature = "otlp")]
135    pub otlp: Option<crate::otlp::OtlpEndpoint>,
136}

138impl std::fmt::Debug for InstallOptions {
139    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
140        let mut dbg = f.debug_struct("InstallOptions");
141        dbg.field("filter", &self.filter)
142            .field("format", &self.format)
143            .field("writer", &self.writer)
144            .field("non_blocking", &self.non_blocking)
145            .field("file_path", &self.file_path)
146            .field("rotation", &self.rotation)
147            .field(
148                "extra_layers",
149                &format_args!("[{} layers]", self.extra_layers.len()),
150            );
151        #[cfg(feature = "otlp")]
152        dbg.field("otlp", &self.otlp);
153        dbg.finish()
154    }
155}

207/// Errors from [`install`].
208///
209/// ```
210/// use tau_observe::install::InstallError;
211///
212/// // The Display message matches the `thiserror` annotation.
213/// let err = InstallError::AlreadyInstalled;
214/// assert!(err.to_string().contains("already installed"));
215/// ```
216#[derive(Debug, Error)]
217pub enum InstallError {
218    /// A subscriber is already installed in this process and the global
219    /// init was attempted a second time. Calls that want idempotent
220    /// install go through [`install`] (which short-circuits) — this
221    /// error is reserved for explicit `install_unique`-style entry
222    /// points that may be added later.
223    #[error("a tracing subscriber is already installed for this process")]
224    AlreadyInstalled,
225}

227/// Guard returned by [`install`]. Drop runs after-effects (currently
228/// holds the non-blocking appender's `WorkerGuard` when the
229/// `non_blocking` feature is enabled and the non-blocking install
230/// path was taken).
231#[derive(Debug)]
232pub struct InstallGuard {
233    #[cfg(feature = "non_blocking")]
234    _appender_guard: Option<tracing_appender::non_blocking::WorkerGuard>,
235    _private: (),
236}

238impl InstallGuard {
239    fn empty() -> Self {
240        Self {
241            #[cfg(feature = "non_blocking")]
242            _appender_guard: None,
243            _private: (),
244        }
245    }

250/// Build a `tracing-opentelemetry` layer from an [`OtlpEndpoint`].
251///
252/// Panics if the OTLP pipeline cannot be constructed (e.g. malformed
253/// endpoint). v1 deliberately treats a misconfigured OTLP endpoint as a
254/// fatal user error rather than degrading silently.
255#[cfg(feature = "otlp")]
256fn build_otel_layer<S>(
257    otlp_ep: &crate::otlp::OtlpEndpoint,
258) -> tracing_opentelemetry::OpenTelemetryLayer<S, opentelemetry_sdk::trace::Tracer>
259where
260    S: tracing::Subscriber + for<'span> tracing_subscriber::registry::LookupSpan<'span>,
261{
262    use opentelemetry::trace::TracerProvider as _;
263    use opentelemetry_otlp::{WithExportConfig as _, WithTonicConfig as _};
264
265    let mut exporter_builder = opentelemetry_otlp::SpanExporter::builder()
266        .with_tonic()
267        .with_endpoint(&otlp_ep.endpoint);
268    if !otlp_ep.headers.is_empty() {
269        let mut metadata = tonic::metadata::MetadataMap::new();
270        for (k, v) in &otlp_ep.headers {
271            if let (Ok(name), Ok(val)) = (
272                k.parse::<tonic::metadata::MetadataKey<tonic::metadata::Ascii>>(),
273                v.parse::<tonic::metadata::MetadataValue<tonic::metadata::Ascii>>(),
274            ) {
275                metadata.insert(name, val);
276            }
277        }
278        exporter_builder = exporter_builder.with_metadata(metadata);
279    }
280    let exporter = exporter_builder.build().expect("build OTLP span exporter");
281
282    let resource = opentelemetry_sdk::Resource::builder()
283        .with_attributes([
284            opentelemetry::KeyValue::new("service.name", "tau"),
285            opentelemetry::KeyValue::new("service.version", env!("CARGO_PKG_VERSION")),
286        ])
287        .build();
288
289    let provider = opentelemetry_sdk::trace::SdkTracerProvider::builder()
290        .with_batch_exporter(exporter)
291        .with_resource(resource)
292        .build();
293
294    let tracer = provider.tracer("tau");
295    tracing_opentelemetry::layer().with_tracer(tracer)
296}

298/// Install the global tracing subscriber. Idempotent: subsequent calls
299/// after a successful install are no-ops that return a fresh guard
300/// without re-installing.
301///
302/// `no_run`: installs a process-global tracing subscriber. Running this
303/// in a parallel doctest context would race with other tests that also
304/// install a subscriber; the call shape is demonstrated without execution.
305/// ```no_run
306/// use tau_observe::install::{install, InstallOptions};
307///
308/// // Install once at process start; the guard must stay alive.
309/// let _guard = install(InstallOptions::cli_default())
310///     .expect("tracing subscriber install failed");
311/// ```
312pub fn install(opts: InstallOptions) -> Result<InstallGuard, InstallError> {
313    let cell = INSTALL_ONCE.get_or_init(|| Mutex::new(false));
314    let mut installed = cell.lock().unwrap_or_else(|p| p.into_inner());
315    if *installed {
316        return Ok(InstallGuard::empty());
317    }
318
319    #[cfg(feature = "non_blocking")]
320    {
321        if opts.non_blocking && opts.file_path.is_some() {
322            let guard = install_non_blocking_inner(opts)?;
323            *installed = true;
324            return Ok(guard);
325        }
326    }
327
328    use tracing_subscriber::fmt;
329    use tracing_subscriber::layer::Layer as _;
330    use tracing_subscriber::layer::SubscriberExt;
331    use tracing_subscriber::util::SubscriberInitExt;
332
333    // Compose caller-provided extra layers onto the raw registry first.
334    // `Vec<Box<dyn Layer<Registry> + Send + Sync>>` implements `Layer<S>`
335    // via tracing-subscriber's blanket impl for `Vec<L>`. We wrap the
336    // vec in `Option` so it disappears entirely when empty — the Vec
337    // impl's `enabled` is `iter().any(…)`, which is `false` for an empty
338    // vec and would suppress every event process-wide.
339    let extras = if opts.extra_layers.is_empty() {
340        None
341    } else {
342        Some(opts.extra_layers)
343    };
344
345    // The `EnvFilter` is a PER-LAYER filter on the console sinks, not a
346    // global one (tau-rs/tau#694). Layering it onto the registry made it
347    // a global filter, and a global filter gates every layer in the
348    // stack — including layers carrying their own per-layer filter. A
349    // per-layer filter can only narrow what its layer sees; it can never
350    // widen past a global filter. That made `--quiet` silently drop the
351    // workflow run log and the `--record-protocol` recording, both of
352    // which are on-disk artifacts the user asked for by name.
353    //
354    // The optional OpenTelemetry layer is an operator-facing sink too,
355    // so it rides under the same filter. `Layer::and_then` folds it and
356    // the fmt layer into a single `Layer<Registry>`, which lets one
357    // `EnvFilter` cover both — `EnvFilter` is not `Clone`, so filtering
358    // them separately would mean re-parsing the directives. `None` is a
359    // zero-cost no-op because `Option<Layer>` is itself a `Layer`.
360    #[cfg(feature = "otlp")]
361    let console_extra = opts.otlp.as_ref().map(build_otel_layer);
362    #[cfg(not(feature = "otlp"))]
363    let console_extra: Option<tracing_subscriber::layer::Identity> = None;
364
365    let filter = opts.filter;
366    let base = tracing_subscriber::registry().with(extras);
367
368    let result = match (opts.format, opts.writer) {
369        (Format::Human, Writer::Stderr) => base
370            .with(
371                fmt::layer()
372                    .with_writer(std::io::stderr)
373                    .and_then(console_extra)
374                    .with_filter(filter),
375            )
376            .try_init(),
377        (Format::Human, Writer::Stdout) => base
378            .with(
379                fmt::layer()
380                    .with_writer(std::io::stdout)
381                    .and_then(console_extra)
382                    .with_filter(filter),
383            )
384            .try_init(),
385        (Format::Json, Writer::Stderr) => base
386            .with(
387                fmt::layer()
388                    .json()
389                    .with_writer(std::io::stderr)
390                    .with_current_span(true)
391                    .with_span_list(false)
392                    .and_then(console_extra)
393                    .with_filter(filter),
394            )
395            .try_init(),
396        (Format::Json, Writer::Stdout) => base
397            .with(
398                fmt::layer()
399                    .json()
400                    .with_writer(std::io::stdout)
401                    .with_current_span(true)
402                    .with_span_list(false)
403                    .and_then(console_extra)
404                    .with_filter(filter),
405            )
406            .try_init(),
407    };
408
409    match result {
410        Ok(()) => {
411            *installed = true;
412            Ok(InstallGuard::empty())
413        }
414        // `try_init` returns Err when a subscriber is already installed.
415        // We treat that as success because another part of the process
416        // (e.g. a foreign test harness) has already initialized one.
417        // The guard the caller receives is a no-op.
418        Err(_) => {
419            *installed = true;
420            Ok(InstallGuard::empty())
421        }
422    }
423}

425/// Split a file path into the (directory, file-name) pair that
426/// `tracing-appender` needs.
427///
428/// `Path::parent()` returns `Some("")` for a bare filename like
429/// `PathBuf::from("app.log")` (it is `None` only for paths like `/`
430/// and `""`). `tracing-appender` treats an empty directory as the
431/// process CWD silently — we make that explicit by normalizing to
432/// `"."` so logs always land somewhere predictable.
433///
434/// The `file_name()` `.expect()` stays: paths ending in `/` or `..`
435/// have no filename, and the resulting panic surfaces a programmer
436/// error instead of passing nonsense to `tracing-appender`.
437#[cfg(feature = "non_blocking")]
438fn resolve_appender_paths(path: &std::path::Path) -> (std::path::PathBuf, String) {
439    let dir = match path.parent() {
440        Some(p) if !p.as_os_str().is_empty() => p.to_path_buf(),
441        _ => std::path::PathBuf::from("."),
442    };
443    let prefix = path
444        .file_name()
445        .expect("file_path has no filename component")
446        .to_string_lossy()
447        .into_owned();
448    (dir, prefix)
449}

451/// Build the non-blocking install path. Caller must have validated
452/// that `opts.file_path.is_some()` and holds the install lock.
453#[cfg(feature = "non_blocking")]
454fn install_non_blocking_inner(opts: InstallOptions) -> Result<InstallGuard, InstallError> {
455    use tracing_appender::rolling;
456    use tracing_subscriber::fmt;
457    use tracing_subscriber::layer::Layer as _;
458    use tracing_subscriber::layer::SubscriberExt;
459    use tracing_subscriber::util::SubscriberInitExt;
460
461    let path = opts
462        .file_path
463        .clone()
464        .expect("install_non_blocking_inner called without file_path");
465    let (dir, prefix) = resolve_appender_paths(&path);
466
467    let file_appender = match opts.rotation {
468        Rotation::Never => rolling::never(&dir, &prefix),
469        Rotation::Daily => rolling::daily(&dir, &prefix),
470        Rotation::Hourly => rolling::hourly(&dir, &prefix),
471    };
472    let (writer, worker_guard) = tracing_appender::non_blocking(file_appender);
473
474    // Per-layer filter on the fmt layer, matching the blocking path —
475    // see the `install` body for why the `EnvFilter` must not be a
476    // global filter (tau-rs/tau#694). This path does not compose
477    // `extra_layers` or the OTLP layer at all yet; that gap is #699.
478    let registry = tracing_subscriber::registry();
479    let result = match opts.format {
480        Format::Human => registry
481            .with(fmt::layer().with_writer(writer).with_filter(opts.filter))
482            .try_init(),
483        Format::Json => registry
484            .with(
485                fmt::layer()
486                    .json()
487                    .with_writer(writer)
488                    .with_current_span(true)
489                    .with_span_list(false)
490                    .with_filter(opts.filter),
491            )
492            .try_init(),
493    };
494    // `try_init` returns Err when a subscriber is already installed —
495    // treat that as success (parity with the blocking path).
496    let _ = result;
497
498    Ok(InstallGuard {
499        _appender_guard: Some(worker_guard),
500        _private: (),
501    })
502}
