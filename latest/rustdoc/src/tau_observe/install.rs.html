1//! Canonical tracing-subscriber installer.
2//!
3//! Two install paths supported at v1: human-readable to stderr (CLI),
4//! and JSON to stderr (plugin SDK). Both go through [`install`] so the
5//! filter-resolution and idempotency behavior are identical.

11/// Output format for the fmt layer.
12///
13/// ```
14/// use tau_observe::install::Format;
15///
16/// assert_ne!(Format::Human, Format::Json);
17/// // Format is Copy, so it can be stored and passed cheaply.
18/// let fmt: Format = Format::Json;
19/// let fmt2 = fmt; // copy
20/// assert_eq!(fmt, fmt2);
21/// ```
22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23pub enum Format {
24    /// Human-readable (timestamp + level + target + fields + message).
25    Human,
26    /// JSON Lines, one event per line.
27    Json,
28}
29
30/// Where the subscriber writes serialized events.
31///
32/// ```
33/// use tau_observe::install::Writer;
34///
35/// assert_ne!(Writer::Stderr, Writer::Stdout);
36/// // Writer is Copy, so it can be stored and passed cheaply.
37/// let w: Writer = Writer::Stderr;
38/// let w2 = w; // copy
39/// assert_eq!(w, w2);
40/// ```
41#[derive(Debug, Clone, Copy, PartialEq, Eq)]
42pub enum Writer {
43    /// Standard error.
44    Stderr,
45    /// Standard output.
46    Stdout,
47}
48
49/// File rotation policy when writing to a file sink.
50///
51/// ```
52/// use tau_observe::install::Rotation;
53///
54/// // `Never` is the default.
55/// assert_eq!(Rotation::default(), Rotation::Never);
56/// // All three variants are distinct.
57/// assert_ne!(Rotation::Never, Rotation::Daily);
58/// assert_ne!(Rotation::Daily, Rotation::Hourly);
59/// ```
60#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
61pub enum Rotation {
62    /// No rotation — append forever to a single file.
63    #[default]
64    Never,
65    /// Roll over each day at UTC midnight (filename gains a date suffix).
66    Daily,
67    /// Roll over each hour (filename gains a date+hour suffix).
68    Hourly,
69}

71/// All knobs the canonical installer accepts.
72///
73/// Construct via [`InstallOptions::cli_default`] or
74/// [`InstallOptions::plugin_sdk`], then override individual fields as
75/// needed.
76///
77/// ```
78/// use tau_observe::install::{InstallOptions, Format, Writer, Rotation};
79///
80/// let opts = InstallOptions::cli_default();
81/// assert_eq!(opts.format, Format::Human);
82/// assert_eq!(opts.writer, Writer::Stderr);
83/// assert!(!opts.non_blocking);
84/// assert!(opts.file_path.is_none());
85/// assert_eq!(opts.rotation, Rotation::Never);
86/// assert!(opts.extra_layers.is_empty());
87/// ```
88pub struct InstallOptions {
89    /// Filter to apply. Build via `tau_observe::filter::env_or_directive`.
90    pub filter: EnvFilter,
91    /// Serialization format.
92    pub format: Format,
93    /// Sink.
94    pub writer: Writer,
95    /// When `true`, the fmt layer writes through a non-blocking MPSC
96    /// channel. Requires feature `non_blocking` and `file_path` to be
97    /// set; otherwise the flag is ignored (blocking install runs).
98    pub non_blocking: bool,
99    /// Optional file sink. When set together with `non_blocking`,
100    /// overrides [`Writer`]. Requires feature `non_blocking`.
101    pub file_path: Option<std::path::PathBuf>,
102    /// File rotation policy. Ignored unless `file_path` is set.
103    pub rotation: Rotation,
104    /// Extra layers to compose into the registry alongside the fmt layer.
105    /// Used by callers that need to wire custom on-disk JSONL sinks (e.g.
106    /// `WorkflowRunLogLayer`, `PluginRecordingLayer`).
107    ///
108    /// Each layer must implement `Layer<Registry>` so it composes against
109    /// the raw registry before the `EnvFilter` is layered on top. The
110    /// installer applies these via a fold BEFORE the `EnvFilter`, so the
111    /// `EnvFilter`'s global-filter effect still gates them — layer impls
112    /// that need to bypass the env filter should attach their own
113    /// per-layer filter via `Layer::with_filter`.
114    pub extra_layers: Vec<
115        Box<dyn tracing_subscriber::Layer<tracing_subscriber::Registry> + Send + Sync + 'static>,
116    >,
117    /// When set, an OpenTelemetry layer exports spans over OTLP/gRPC.
118    /// Requires feature `otlp`.
119    #[cfg(feature = "otlp")]
120    pub otlp: Option<crate::otlp::OtlpEndpoint>,
121}

123impl std::fmt::Debug for InstallOptions {
124    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
125        let mut dbg = f.debug_struct("InstallOptions");
126        dbg.field("filter", &self.filter)
127            .field("format", &self.format)
128            .field("writer", &self.writer)
129            .field("non_blocking", &self.non_blocking)
130            .field("file_path", &self.file_path)
131            .field("rotation", &self.rotation)
132            .field(
133                "extra_layers",
134                &format_args!("[{} layers]", self.extra_layers.len()),
135            );
136        #[cfg(feature = "otlp")]
137        dbg.field("otlp", &self.otlp);
138        dbg.finish()
139    }
140}

192/// Errors from [`install`].
193///
194/// ```
195/// use tau_observe::install::InstallError;
196///
197/// // The Display message matches the `thiserror` annotation.
198/// let err = InstallError::AlreadyInstalled;
199/// assert!(err.to_string().contains("already installed"));
200/// ```
201#[derive(Debug, Error)]
202pub enum InstallError {
203    /// A subscriber is already installed in this process and the global
204    /// init was attempted a second time. Calls that want idempotent
205    /// install go through [`install`] (which short-circuits) — this
206    /// error is reserved for explicit `install_unique`-style entry
207    /// points that may be added later.
208    #[error("a tracing subscriber is already installed for this process")]
209    AlreadyInstalled,
210}

212/// Guard returned by [`install`]. Drop runs after-effects (currently
213/// holds the non-blocking appender's `WorkerGuard` when the
214/// `non_blocking` feature is enabled and the non-blocking install
215/// path was taken).
216#[derive(Debug)]
217pub struct InstallGuard {
218    #[cfg(feature = "non_blocking")]
219    _appender_guard: Option<tracing_appender::non_blocking::WorkerGuard>,
220    _private: (),
221}
222
223impl InstallGuard {
224    fn empty() -> Self {
225        Self {
226            #[cfg(feature = "non_blocking")]
227            _appender_guard: None,
228            _private: (),
229        }
230    }
231}

235/// Build a `tracing-opentelemetry` layer from an [`OtlpEndpoint`].
236///
237/// Panics if the OTLP pipeline cannot be constructed (e.g. malformed
238/// endpoint). v1 deliberately treats a misconfigured OTLP endpoint as a
239/// fatal user error rather than degrading silently.
240#[cfg(feature = "otlp")]
241fn build_otel_layer<S>(
242    otlp_ep: &crate::otlp::OtlpEndpoint,
243) -> tracing_opentelemetry::OpenTelemetryLayer<S, opentelemetry_sdk::trace::Tracer>
244where
245    S: tracing::Subscriber + for<'span> tracing_subscriber::registry::LookupSpan<'span>,
246{
247    use opentelemetry::trace::TracerProvider as _;
248    use opentelemetry_otlp::{WithExportConfig as _, WithTonicConfig as _};
249
250    let mut exporter_builder = opentelemetry_otlp::SpanExporter::builder()
251        .with_tonic()
252        .with_endpoint(&otlp_ep.endpoint);
253    if !otlp_ep.headers.is_empty() {
254        let mut metadata = tonic::metadata::MetadataMap::new();
255        for (k, v) in &otlp_ep.headers {
256            if let (Ok(name), Ok(val)) = (
257                k.parse::<tonic::metadata::MetadataKey<tonic::metadata::Ascii>>(),
258                v.parse::<tonic::metadata::MetadataValue<tonic::metadata::Ascii>>(),
259            ) {
260                metadata.insert(name, val);
261            }
262        }
263        exporter_builder = exporter_builder.with_metadata(metadata);
264    }
265    let exporter = exporter_builder.build().expect("build OTLP span exporter");
266
267    let resource = opentelemetry_sdk::Resource::builder()
268        .with_attributes([
269            opentelemetry::KeyValue::new("service.name", "tau"),
270            opentelemetry::KeyValue::new("service.version", env!("CARGO_PKG_VERSION")),
271        ])
272        .build();
273
274    let provider = opentelemetry_sdk::trace::SdkTracerProvider::builder()
275        .with_batch_exporter(exporter)
276        .with_resource(resource)
277        .build();
278
279    let tracer = provider.tracer("tau");
280    tracing_opentelemetry::layer().with_tracer(tracer)
281}

283/// Install the global tracing subscriber. Idempotent: subsequent calls
284/// after a successful install are no-ops that return a fresh guard
285/// without re-installing.
286///
287/// `no_run`: installs a process-global tracing subscriber. Running this
288/// in a parallel doctest context would race with other tests that also
289/// install a subscriber; the call shape is demonstrated without execution.
290/// ```no_run
291/// use tau_observe::install::{install, InstallOptions};
292///
293/// // Install once at process start; the guard must stay alive.
294/// let _guard = install(InstallOptions::cli_default())
295///     .expect("tracing subscriber install failed");
296/// ```
297pub fn install(opts: InstallOptions) -> Result<InstallGuard, InstallError> {
298    let cell = INSTALL_ONCE.get_or_init(|| Mutex::new(false));
299    let mut installed = cell.lock().unwrap_or_else(|p| p.into_inner());
300    if *installed {
301        return Ok(InstallGuard::empty());
302    }
303
304    #[cfg(feature = "non_blocking")]
305    {
306        if opts.non_blocking && opts.file_path.is_some() {
307            let guard = install_non_blocking_inner(opts)?;
308            *installed = true;
309            return Ok(guard);
310        }
311    }
312
313    use tracing_subscriber::fmt;
314    use tracing_subscriber::layer::SubscriberExt;
315    use tracing_subscriber::util::SubscriberInitExt;
316
317    // Compose caller-provided extra layers onto the raw registry first.
318    // `Vec<Box<dyn Layer<Registry> + Send + Sync>>` implements `Layer<S>`
319    // via tracing-subscriber's blanket impl for `Vec<L>`. We wrap the
320    // vec in `Option` so it disappears entirely when empty — the Vec
321    // impl's `max_level_hint` returns `LevelFilter::OFF` for an empty
322    // vec, which would otherwise suppress all events at the registry
323    // level (see tracing-subscriber v0.3 `Vec<L>::max_level_hint`).
324    let extras = if opts.extra_layers.is_empty() {
325        None
326    } else {
327        Some(opts.extra_layers)
328    };
329    let base = tracing_subscriber::registry()
330        .with(extras)
331        .with(opts.filter);
332
333    // Optional OpenTelemetry layer goes BELOW the fmt layer so that its
334    // `Layer<S>` impl resolves against the stable `Layered<EnvFilter,
335    // Registry>` subscriber type (independent of which fmt branch the
336    // match selects below). `Option<Layer>` is itself a `Layer`, so
337    // `None` is a zero-cost no-op.
338    #[cfg(feature = "otlp")]
339    let base = base.with(opts.otlp.as_ref().map(build_otel_layer));
340
341    let result = match (opts.format, opts.writer) {
342        (Format::Human, Writer::Stderr) => base
343            .with(fmt::layer().with_writer(std::io::stderr))
344            .try_init(),
345        (Format::Human, Writer::Stdout) => base
346            .with(fmt::layer().with_writer(std::io::stdout))
347            .try_init(),
348        (Format::Json, Writer::Stderr) => base
349            .with(
350                fmt::layer()
351                    .json()
352                    .with_writer(std::io::stderr)
353                    .with_current_span(true)
354                    .with_span_list(false),
355            )
356            .try_init(),
357        (Format::Json, Writer::Stdout) => base
358            .with(
359                fmt::layer()
360                    .json()
361                    .with_writer(std::io::stdout)
362                    .with_current_span(true)
363                    .with_span_list(false),
364            )
365            .try_init(),
366    };
367
368    match result {
369        Ok(()) => {
370            *installed = true;
371            Ok(InstallGuard::empty())
372        }
373        // `try_init` returns Err when a subscriber is already installed.
374        // We treat that as success because another part of the process
375        // (e.g. a foreign test harness) has already initialized one.
376        // The guard the caller receives is a no-op.
377        Err(_) => {
378            *installed = true;
379            Ok(InstallGuard::empty())
380        }
381    }
382}

384/// Split a file path into the (directory, file-name) pair that
385/// `tracing-appender` needs.
386///
387/// `Path::parent()` returns `Some("")` for a bare filename like
388/// `PathBuf::from("app.log")` (it is `None` only for paths like `/`
389/// and `""`). `tracing-appender` treats an empty directory as the
390/// process CWD silently — we make that explicit by normalizing to
391/// `"."` so logs always land somewhere predictable.
392///
393/// The `file_name()` `.expect()` stays: paths ending in `/` or `..`
394/// have no filename, and the resulting panic surfaces a programmer
395/// error instead of passing nonsense to `tracing-appender`.
396#[cfg(feature = "non_blocking")]
397fn resolve_appender_paths(path: &std::path::Path) -> (std::path::PathBuf, String) {
398    let dir = match path.parent() {
399        Some(p) if !p.as_os_str().is_empty() => p.to_path_buf(),
400        _ => std::path::PathBuf::from("."),
401    };
402    let prefix = path
403        .file_name()
404        .expect("file_path has no filename component")
405        .to_string_lossy()
406        .into_owned();
407    (dir, prefix)
408}

410/// Build the non-blocking install path. Caller must have validated
411/// that `opts.file_path.is_some()` and holds the install lock.
412#[cfg(feature = "non_blocking")]
413fn install_non_blocking_inner(opts: InstallOptions) -> Result<InstallGuard, InstallError> {
414    use tracing_appender::rolling;
415    use tracing_subscriber::fmt;
416    use tracing_subscriber::layer::SubscriberExt;
417    use tracing_subscriber::util::SubscriberInitExt;
418
419    let path = opts
420        .file_path
421        .clone()
422        .expect("install_non_blocking_inner called without file_path");
423    let (dir, prefix) = resolve_appender_paths(&path);
424
425    let file_appender = match opts.rotation {
426        Rotation::Never => rolling::never(&dir, &prefix),
427        Rotation::Daily => rolling::daily(&dir, &prefix),
428        Rotation::Hourly => rolling::hourly(&dir, &prefix),
429    };
430    let (writer, worker_guard) = tracing_appender::non_blocking(file_appender);
431
432    let registry = tracing_subscriber::registry().with(opts.filter);
433    let result = match opts.format {
434        Format::Human => registry.with(fmt::layer().with_writer(writer)).try_init(),
435        Format::Json => registry
436            .with(
437                fmt::layer()
438                    .json()
439                    .with_writer(writer)
440                    .with_current_span(true)
441                    .with_span_list(false),
442            )
443            .try_init(),
444    };
445    // `try_init` returns Err when a subscriber is already installed —
446    // treat that as success (parity with the blocking path).
447    let _ = result;
448
449    Ok(InstallGuard {
450        _appender_guard: Some(worker_guard),
451        _private: (),
452    })
453}
