1//! Sensitive-data preview helpers (ADR-0006 §3.9 discipline).
2//!
3//! The kernel emits structured logs containing arguments, message
4//! payloads, and LLM responses. Per §3.9 these bodies are previewed
5//! (256 bytes, UTF-8-boundary-clipped) at `DEBUG` and below; full
6//! content is emitted only at `TRACE`. These helpers make that policy
7//! mechanical at every call site.
8//!
9//! Per ADR-0006 NG9 ("tau does not redact for the caller") this module
10//! is **kernel-internal**. Plugin authors may use it but are not
11//! required to.
12
13use std::fmt::{self, Display, Formatter};
14
15const PREVIEW_LIMIT_BYTES: usize = 256;
16
17/// Render a `&str` truncated to at most 256 bytes ending on a UTF-8
18/// boundary, with a `"…"` ellipsis if truncation occurred.
19///
20/// Use at `DEBUG` (and below) call sites for argument / payload /
21/// message content.
22///
23/// ```
24/// use tau_observe::preview::preview;
25///
26/// // Short strings pass through unchanged.
27/// assert_eq!(format!("{}", preview("hello")), "hello");
28///
29/// // Strings longer than 256 bytes are truncated with an ellipsis.
30/// let long = "x".repeat(300);
31/// let out = format!("{}", preview(&long));
32/// assert!(out.ends_with('…'), "expected ellipsis suffix");
33/// assert!(out.len() <= 256 + '…'.len_utf8());
34/// ```
35pub fn preview(value: &str) -> impl Display + '_ {
36    Preview(value)
37}
38
39struct Preview<'a>(&'a str);
40
41impl Display for Preview<'_> {
42    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
43        let s = self.0;
44        if s.len() <= PREVIEW_LIMIT_BYTES {
45            f.write_str(s)
46        } else {
47            let mut cut = PREVIEW_LIMIT_BYTES;
48            while cut > 0 && !s.is_char_boundary(cut) {
49                cut -= 1;
50            }
51            f.write_str(&s[..cut])?;
52            f.write_str("…")
53        }
54    }
55}

57/// Render a `serde_json::Value` as compact JSON, truncated to at most
58/// 256 bytes ending on a UTF-8 boundary, with a `"…"` ellipsis if
59/// truncation occurred.
60///
61/// Use at `DEBUG` (and below) call sites for JSON-shaped payloads.
62///
63/// ```
64/// use tau_observe::preview::preview_json;
65///
66/// // Small values render as compact JSON without truncation.
67/// let v = serde_json::json!({"name": "ada"});
68/// assert_eq!(format!("{}", preview_json(&v)), r#"{"name":"ada"}"#);
69///
70/// // Large values are truncated with an ellipsis.
71/// let big = serde_json::json!({"data": "y".repeat(500)});
72/// let out = format!("{}", preview_json(&big));
73/// assert!(out.ends_with('…'), "expected ellipsis for large payload");
74/// ```
75pub fn preview_json(value: &serde_json::Value) -> impl Display + '_ {
76    PreviewJson(value)
77}
78
79struct PreviewJson<'a>(&'a serde_json::Value);
80
81impl Display for PreviewJson<'_> {
82    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
83        let s = serde_json::to_string(&self.0).unwrap_or_else(|_| "<unserializable>".to_string());
84        write!(f, "{}", Preview(s.as_str()))
85    }
86}

88/// Render a `&str` in full.
89///
90/// **Only call this at `tracing::trace!` (or below) sites.** At any
91/// higher level, the macros emit the event unconditionally (subject to
92/// the filter) and the full content gets persisted. Use [`preview`]
93/// instead for DEBUG and above.
94///
95/// ```
96/// use tau_observe::preview::full;
97///
98/// // Returns the entire string regardless of length.
99/// let s = "x".repeat(1000);
100/// assert_eq!(format!("{}", full(&s)), s);
101/// ```
102pub fn full(value: &str) -> impl Display + '_ {
103    Full(value)
104}
105
106struct Full<'a>(&'a str);
107
108impl Display for Full<'_> {
109    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
110        f.write_str(self.0)
111    }
112}

114/// Render a `serde_json::Value` as compact JSON in full.
115///
116/// Same rule as [`full`]: TRACE-only call sites.
117///
118/// ```
119/// use tau_observe::preview::full_json;
120///
121/// // Returns the compact JSON representation in full, without truncation.
122/// let v = serde_json::json!({"data": "x".repeat(1000)});
123/// let out = format!("{}", full_json(&v));
124/// assert!(out.contains(&"x".repeat(1000)));
125/// assert!(!out.ends_with('…'), "full_json must not truncate");
126/// ```
127pub fn full_json(value: &serde_json::Value) -> impl Display + '_ {
128    FullJson(value)
129}
130
131struct FullJson<'a>(&'a serde_json::Value);
132
133impl Display for FullJson<'_> {
134    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
135        let s = serde_json::to_string(&self.0).unwrap_or_else(|_| "<unserializable>".to_string());
136        f.write_str(&s)
137    }
138}

140#[cfg(test)]
141mod tests {
142    use super::*;
143
144    #[test]
145    fn short_string_passes_through_verbatim() {
146        let s = "hello";
147        assert_eq!(format!("{}", preview(s)), "hello");
148    }
149
150    #[test]
151    fn empty_string_passes_through() {
152        assert_eq!(format!("{}", preview("")), "");
153    }
154
155    #[test]
156    fn exactly_256_bytes_no_truncation() {
157        let s = "a".repeat(256);
158        assert_eq!(format!("{}", preview(&s)), s);
159    }
160
161    #[test]
162    fn over_256_bytes_truncates_with_ellipsis() {
163        let s = "a".repeat(300);
164        let out = format!("{}", preview(&s));
165        assert!(out.ends_with('…'));
166        let body = out.trim_end_matches('…');
167        assert!(body.len() <= 256, "body was {} bytes", body.len());
168    }
169
170    #[test]
171    fn truncation_respects_utf8_boundary_at_3_byte_codepoint() {
172        let mut s = "a".repeat(254);
173        s.push('€');
174        s.push_str(&"b".repeat(50));
175        let out = format!("{}", preview(&s));
176        let body = out.trim_end_matches('…');
177        assert!(
178            std::str::from_utf8(body.as_bytes()).is_ok(),
179            "invalid UTF-8 in preview"
180        );
181    }
182
183    #[test]
184    fn truncation_respects_utf8_boundary_at_4_byte_codepoint() {
185        let mut s = "a".repeat(253);
186        s.push('𝄞');
187        s.push_str(&"b".repeat(50));
188        let out = format!("{}", preview(&s));
189        let body = out.trim_end_matches('…');
190        assert!(
191            std::str::from_utf8(body.as_bytes()).is_ok(),
192            "invalid UTF-8 in preview"
193        );
194    }
195
196    #[test]
197    fn preview_json_short_value_passes_through() {
198        let v = serde_json::json!({"name": "ada"});
199        let out = format!("{}", preview_json(&v));
200        assert_eq!(out, r#"{"name":"ada"}"#);
201    }
202
203    #[test]
204    fn preview_json_long_value_truncates() {
205        let v = serde_json::json!({"data": "x".repeat(500)});
206        let out = format!("{}", preview_json(&v));
207        assert!(out.ends_with('…'));
208        let body = out.trim_end_matches('…');
209        assert!(body.len() <= 256);
210    }
211
212    #[test]
213    fn full_returns_string_verbatim_regardless_of_length() {
214        let s = "x".repeat(1000);
215        assert_eq!(format!("{}", full(&s)), s);
216    }
217
218    #[test]
219    fn full_json_returns_value_verbatim_regardless_of_size() {
220        let v = serde_json::json!({"data": "x".repeat(1000)});
221        let out = format!("{}", full_json(&v));
222        assert!(out.contains(&"x".repeat(1000)));
223    }
