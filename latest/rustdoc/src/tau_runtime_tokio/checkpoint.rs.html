1//! Filesystem [`CheckpointStore`] for durable execution (ADR-0053).
2//!
3//! Per-turn snapshot files under `<scope>/.tau/runs/<run_id>/turn-<n>.json`,
4//! one file per completed turn, keep-all. Writes go to `turn-<n>.json.tmp`
5//! then atomic-rename to `turn-<n>.json`, so a crash mid-write leaves the
6//! previous turn's file intact — the resume point, which is exactly the
7//! at-least-once boundary A-minimal promises. `load_latest` returns the
8//! highest-`n` file.
9//!
10//! Under `PerToolCall`, multiple checkpoints are written within one turn —
11//! they share `turn-<n>.json` (last-write-wins; the atomic `.tmp`-rename
12//! makes overwrite safe), and the turn-boundary write finalizes it with an
13//! empty `pending_tool_uses`. `load_latest` still returns the highest-`n`
14//! file, whose `pending_tool_uses` directs mid-turn resume.
15
16use std::path::PathBuf;
17
18use tau_ports::{CheckpointError, CheckpointStore, RunId, TurnCheckpoint};
19
20/// A [`CheckpointStore`] backed by per-turn JSON snapshot files.
21pub struct FileCheckpointStore {
22    /// Project scope root; the runs directory is `<root>/.tau/runs`.
23    scope_root: PathBuf,
24}
25
26impl FileCheckpointStore {
27    /// Construct a store rooted at `scope_root` (the project scope dir).
28    pub fn new(scope_root: impl Into<PathBuf>) -> Self {
29        Self {
30            scope_root: scope_root.into(),
31        }
32    }
33
34    /// `<scope>/.tau/runs/<run_id>` — the directory holding one run's
35    /// per-turn snapshot files.
36    fn run_dir(&self, run_id: &str) -> PathBuf {
37        self.scope_root.join(".tau").join("runs").join(run_id)
38    }
39}
40
41fn io_err(e: std::io::Error) -> CheckpointError {
42    CheckpointError::Io(e.to_string())
43}
44
45impl CheckpointStore for FileCheckpointStore {
46    fn persist(&self, ckpt: &TurnCheckpoint) -> Result<(), CheckpointError> {
47        let dir = self.run_dir(&ckpt.run_id);
48        std::fs::create_dir_all(&dir).map_err(io_err)?;
49
50        let json = serde_json::to_vec_pretty(ckpt)
51            .map_err(|e| CheckpointError::Serialization(e.to_string()))?;
52
53        // Atomic write: stage to `.tmp`, then rename onto the final name.
54        let final_path = dir.join(format!("turn-{}.json", ckpt.turn));
55        let tmp_path = dir.join(format!("turn-{}.json.tmp", ckpt.turn));
56        std::fs::write(&tmp_path, &json).map_err(io_err)?;
57        std::fs::rename(&tmp_path, &final_path).map_err(io_err)?;
58        Ok(())
59    }
60
61    fn load_latest(&self, run_id: &RunId) -> Result<Option<TurnCheckpoint>, CheckpointError> {
62        let dir = self.run_dir(run_id);
63        if !dir.exists() {
64            return Ok(None);
65        }
66
67        // Find the highest committed `turn-<n>.json`. A half-written
68        // `turn-<n>.json.tmp` ends in `.tmp`, not `.json`, so it is ignored.
69        let mut best: Option<(u32, PathBuf)> = None;
70        for entry in std::fs::read_dir(&dir).map_err(io_err)? {
71            let entry = entry.map_err(io_err)?;
72            let file_name = entry.file_name();
73            let name = file_name.to_string_lossy();
74            let Some(stem) = name
75                .strip_prefix("turn-")
76                .and_then(|s| s.strip_suffix(".json"))
77            else {
78                continue;
79            };
80            let Ok(turn) = stem.parse::<u32>() else {
81                continue;
82            };
83            if best.as_ref().is_none_or(|(b, _)| turn > *b) {
84                best = Some((turn, entry.path()));
85            }
86        }
87
88        let Some((_, path)) = best else {
89            return Ok(None);
90        };
91        let bytes = std::fs::read(&path).map_err(io_err)?;
92        let ckpt: TurnCheckpoint = serde_json::from_slice(&bytes)
93            .map_err(|e| CheckpointError::Serialization(e.to_string()))?;
94        Ok(Some(ckpt))
95    }
96}

98#[cfg(test)]
99mod tests {
100    use super::*;
101    use tau_domain::{Address, Message, MessagePayload};
102
103    fn ckpt(run_id: &str, turn: u32) -> TurnCheckpoint {
104        TurnCheckpoint {
105            run_id: run_id.to_string(),
106            turn,
107            history: vec![Message::new(
108                Address::User,
109                Address::System,
110                MessagePayload::Text {
111                    content: format!("turn {turn}"),
112                },
113            )],
114            input_tokens: turn as u64 * 10,
115            output_tokens: turn as u64 * 5,
116            pending_tool_uses: vec![],
117        }
118    }
119
120    #[test]
121    fn persist_then_load_latest_round_trips_highest_turn() {
122        let tmp = tempfile::tempdir().unwrap();
123        let store = FileCheckpointStore::new(tmp.path());
124
125        // No checkpoints yet.
126        assert!(store.load_latest(&"run-1".to_string()).unwrap().is_none());
127
128        store.persist(&ckpt("run-1", 1)).unwrap();
129        store.persist(&ckpt("run-1", 2)).unwrap();
130        store.persist(&ckpt("run-1", 3)).unwrap();
131
132        let latest = store.load_latest(&"run-1".to_string()).unwrap().unwrap();
133        assert_eq!(latest.turn, 3);
134        assert_eq!(latest.input_tokens, 30);
135        assert_eq!(latest.history.len(), 1);
136
137        // All three turn files were kept (keep-all).
138        let dir = tmp.path().join(".tau/runs/run-1");
139        let files: Vec<_> = std::fs::read_dir(&dir)
140            .unwrap()
141            .map(|e| e.unwrap().file_name().to_string_lossy().into_owned())
142            .collect();
143        assert_eq!(files.iter().filter(|f| f.ends_with(".json")).count(), 3);
144        assert!(
145            files.iter().all(|f| !f.ends_with(".tmp")),
146            "no temp files should remain: {files:?}"
147        );
148    }
149
150    #[test]
151    fn load_latest_isolates_by_run_id() {
152        let tmp = tempfile::tempdir().unwrap();
153        let store = FileCheckpointStore::new(tmp.path());
154        store.persist(&ckpt("run-a", 5)).unwrap();
155        store.persist(&ckpt("run-b", 1)).unwrap();
156        assert_eq!(
157            store
158                .load_latest(&"run-a".to_string())
159                .unwrap()
160                .unwrap()
161                .turn,
162            5
163        );
164        assert_eq!(
165            store
166                .load_latest(&"run-b".to_string())
167                .unwrap()
168                .unwrap()
169                .turn,
170            1
171        );
172    }
