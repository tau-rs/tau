1//! Protocol recording: capture every wire frame to the
2//! `tau::plugin::frame` tracing target.
3//!
4//! When [`crate::plugin_host::PluginHostOptions::recording`] is set to
5//! [`crate::plugin_host::RecordingSink::JsonlFile`], the read-loop and
6//! the writer-mutex tap every frame and emit a [`tracing::event!`]
7//! carrying the frame's metadata + base64-encoded body. A
8//! [`tau_observe::layers::plugin_recording::PluginRecordingLayer`]
9//! installed by the CLI is the actual writer to disk; the [`Recorder`]
10//! itself is now a stateless emitter holding only the plugin name.
11//!
12//! On-disk JSONL shape (unchanged from the pre-D direct-write era):
13//!
14//! ```text
15//! {"ts":1714316451.123,"plugin":"echo-llm","dir":"h2p",
16//!  "msgid":1,"method":"meta.handshake","frame":"<base64>"}
17//! ```
18//!
19//! Direction codes are `h2p` (host → plugin) and `p2h` (plugin → host).
20//! `method` and `msgid` are decoded from the frame body for indexing
21//! convenience but are also redundantly encoded inside `frame` (base64)
22//! for lossless replay.
23//!
24//! Recording is best-effort. The layer logs write errors via
25//! `tracing::warn!` and does not propagate — the runtime continues even
26//! if the file is full or permission-denied. Likewise, failure to open
27//! the file on first event lazy-loads to a no-op for subsequent events.
28//!
29//! See spec §7.8 (recording), §9.1 (debug tier — protocol recording),
30//! and Logging Sub-project D Task 5 (tracing layer migration).
31

40/// Direction of a recorded frame.
41///
42/// `pub` (rather than `pub(crate)`) so the test-only `__internals`
43/// re-exports can surface it; production code only sees the JSONL
44/// `dir` string field, never the typed enum.
45///
46/// # Example
47///
48/// ```
49/// use tau_runtime_tokio::plugin_host::Direction;
50///
51/// let d = Direction::HostToPlugin;
52/// assert!(matches!(d, Direction::HostToPlugin));
53///
54/// let d2 = Direction::PluginToHost;
55/// assert!(matches!(d2, Direction::PluginToHost));
56/// ```
57#[derive(Debug, Clone, Copy)]
58pub enum Direction {
59    /// Host → plugin (request, host-side notification).
60    HostToPlugin,
61    /// Plugin → host (response, plugin-side notification).
62    PluginToHost,
63}
64
65impl Direction {
66    fn as_str(&self) -> &'static str {
67        match self {
68            Direction::HostToPlugin => "h2p",
69            Direction::PluginToHost => "p2h",
70        }
71    }
72}

74/// A stateless emitter that wraps every recorded frame into a
75/// `tracing::event!(target = "tau::plugin::frame", …)`.
76///
77/// The actual JSONL file is owned by
78/// [`tau_observe::layers::plugin_recording::PluginRecordingLayer`],
79/// which the CLI installs alongside the global subscriber when
80/// `--record-protocol <path>` is passed. The [`Recorder`] keeps only
81/// the plugin name so per-plugin attribution can be carried as a field
82/// on the emitted event.
83///
84/// `pub` (rather than `pub(crate)`) because the test-only
85/// `__internals` re-exports under [`crate::plugin_host::__internals`]
86/// need to surface the type to integration tests; production code
87/// reaches the recorder only via
88/// [`crate::plugin_host::PluginHostOptions::recording`].
89#[derive(Debug)]
90pub struct Recorder {
91    plugin_name: String,
92}
93
94impl Recorder {
95    /// Construct a stateless recorder for `plugin_name`. The recorder
96    /// emits tracing events on each `record(...)`; the actual file
97    /// write happens in a `PluginRecordingLayer` installed elsewhere.
98    ///
99    /// # Example
100    ///
101    /// ```
102    /// use tau_runtime_tokio::plugin_host::Recorder;
103    ///
104    /// let recorder = Recorder::new("echo-llm");
105    /// assert!(format!("{recorder:?}").contains("echo-llm"));
106    /// ```
107    pub fn new(plugin_name: impl Into<String>) -> Self {
108        Self {
109            plugin_name: plugin_name.into(),
110        }
111    }
112
113    /// Record a frame. Emits a `tracing::event!` at INFO level on the
114    /// `tau::plugin::frame` target; if no
115    /// [`tau_observe::layers::plugin_recording::PluginRecordingLayer`] is
116    /// installed, this is a cheap no-op (tracing's event-emission cost
117    /// degrades to a target-string compare when no layer matches).
118    ///
119    /// INFO (rather than TRACE) is chosen so the default
120    /// `tau=info` filter doesn't short-circuit recording events via
121    /// `Subscriber::max_level_hint` before the per-layer filter on
122    /// the recording layer has a chance to evaluate them.
123    /// Recording is opt-in via `--record-protocol`; the level is an
124    /// implementation detail of routing through the tracing
125    /// infrastructure.
126    ///
127    /// `frame_bytes` is the raw MessagePack frame body (post-framing
128    /// decode); the function additionally decodes a [`Frame`] to extract
129    /// `msgid` + `method` for indexing convenience. Decoding failures
130    /// are tolerated: the event is still emitted with the optional
131    /// fields omitted (the layer then materializes them as `null`)
132    /// and the (lossless) base64-encoded frame body intact.
133    pub async fn record(&self, dir: Direction, frame_bytes: &[u8]) {
134        let ts = SystemTime::now()
135            .duration_since(UNIX_EPOCH)
136            .map(|d| d.as_secs_f64())
137            .unwrap_or(0.0);
138        let (msgid, method) = decode_frame_metadata(frame_bytes);
139        let encoded = B64.encode(frame_bytes);
140
141        // tracing macros don't support runtime-omitted fields: the four
142        // (msgid, method) cases each enumerate which optionals are
143        // present. Absent fields surface as `null` in the layer's
144        // serialized JSONL (see `PluginRecordingLayer`).
145        //
146        // - `ts` is emitted as an `f64` so the layer's `record_f64`
147        //   path stores a JSON number (not a Display-stringified form).
148        // - `msgid` is cast to `u64` (always non-negative) to match the
149        //   layer's `record_u64` visitor.
150        // - String fields use the `%` display sigil so tracing
151        //   dispatches via the `Display` Debug impl (which the layer's
152        //   `record_debug` handler maps to a JSON string without
153        //   double-quoting).
154        match (msgid, method) {
155            (Some(id), Some(m)) => tracing::event!(
156                target: REC_TARGET,
157                tracing::Level::INFO,
158                ts = ts,
159                plugin = %self.plugin_name,
160                dir = %dir.as_str(),
161                msgid = id as u64,
162                method = %m,
163                frame = %encoded,
164            ),
165            (Some(id), None) => tracing::event!(
166                target: REC_TARGET,
167                tracing::Level::INFO,
168                ts = ts,
169                plugin = %self.plugin_name,
170                dir = %dir.as_str(),
171                msgid = id as u64,
172                frame = %encoded,
173            ),
174            (None, Some(m)) => tracing::event!(
175                target: REC_TARGET,
176                tracing::Level::INFO,
177                ts = ts,
178                plugin = %self.plugin_name,
179                dir = %dir.as_str(),
180                method = %m,
181                frame = %encoded,
182            ),
183            (None, None) => tracing::event!(
184                target: REC_TARGET,
185                tracing::Level::INFO,
186                ts = ts,
187                plugin = %self.plugin_name,
188                dir = %dir.as_str(),
189                frame = %encoded,
190            ),
191        }
192    }
193}

195/// Best-effort decode of the frame body to extract `msgid` and `method`
196/// for the JSONL `msgid` / `method` indexing fields. Decode failures
197/// (or future [`Frame`] variants — the type is `#[non_exhaustive]`)
198/// map to `(None, None)` — the raw base64-encoded frame is still
199/// recorded losslessly under the `frame` field.
200fn decode_frame_metadata(frame_bytes: &[u8]) -> (Option<u32>, Option<String>) {
201    match Frame::decode(frame_bytes) {
202        Ok(Frame::Request { id, method, .. }) => (Some(id), Some(method)),
203        Ok(Frame::Response { id, .. }) => (Some(id), None),
204        Ok(Frame::Notification { method, .. }) => (None, Some(method)),
205        // `Frame` is `#[non_exhaustive]`; a future variant decodes to
206        // an indexer-less line rather than panicking.
207        Ok(_) => (None, None),
208        Err(_) => (None, None),
209    }
210}

228    /// Wait until `path` contains at least `expected` JSONL lines, or
229    /// panic after `timeout`. The recording layer's writes are
230    /// fire-and-forget `tokio::spawn` tasks, so deterministic
231    /// observation requires polling rather than a fixed sleep.
232    async fn await_layer_flush(path: &Path, expected: usize, timeout: Duration) {
233        let start = std::time::Instant::now();
234        loop {
235            if let Ok(s) = tokio::fs::read_to_string(path).await {
236                let count = s.lines().filter(|l| !l.is_empty()).count();
237                if count >= expected {
238                    return;
239                }
240            }
241            if start.elapsed() >= timeout {
242                let actual = std::fs::read_to_string(path)
243                    .map(|s| s.lines().filter(|l| !l.is_empty()).count())
244                    .unwrap_or(0);
245                panic!(
246                    "plugin recording layer did not flush {expected} record(s) to {path:?} within {timeout:?} (actual: {actual})"
247                );
248            }
249            tokio::time::sleep(Duration::from_millis(10)).await;
250        }
251    }

253    #[tokio::test]
254    async fn record_appends_line_with_metadata_for_known_request_frame() {
255        let tmp = tempfile::tempdir().unwrap();
256        let path = tmp.path().join("rec.jsonl");
257        let layer = PluginRecordingLayer::new(path.clone());
258        let subscriber = Registry::default().with(layer);
259
260        let frame = Frame::Request {
261            id: 7,
262            method: "llm.complete".to_string(),
263            params: rmp_serde::to_vec::<Vec<()>>(&Vec::new()).unwrap(),
264        };
265        let bytes = frame.encode().unwrap();
266
267        let _guard = tracing::subscriber::set_default(subscriber);
268        let recorder = Recorder::new("plug");
269        recorder.record(Direction::HostToPlugin, &bytes).await;
270
271        await_layer_flush(&path, 1, Duration::from_secs(5)).await;
272
273        let contents = std::fs::read_to_string(&path).unwrap();
274        let line = contents.lines().next().expect("at least one line");
275        let parsed: serde_json::Value = serde_json::from_str(line).unwrap();
276        assert_eq!(parsed["plugin"], "plug");
277        assert_eq!(parsed["dir"], "h2p");
278        assert_eq!(parsed["msgid"], 7);
279        assert_eq!(parsed["method"], "llm.complete");
280        assert!(parsed["frame"].is_string());
281        assert!(parsed["ts"].is_number());
282    }

284    #[tokio::test]
285    async fn record_tolerates_undecodable_frame_bytes() {
286        let tmp = tempfile::tempdir().unwrap();
287        let path = tmp.path().join("rec.jsonl");
288        let layer = PluginRecordingLayer::new(path.clone());
289        let subscriber = Registry::default().with(layer);
290
291        let _guard = tracing::subscriber::set_default(subscriber);
292        let recorder = Recorder::new("plug");
293        // Garbage that won't parse as a Frame.
294        recorder
295            .record(Direction::PluginToHost, b"\xff\xff\xff\xff")
296            .await;
297
298        await_layer_flush(&path, 1, Duration::from_secs(5)).await;
299
300        let contents = std::fs::read_to_string(&path).unwrap();
301        let line = contents.lines().next().expect("at least one line");
302        let parsed: serde_json::Value = serde_json::from_str(line).unwrap();
303        assert_eq!(parsed["dir"], "p2h");
304        assert!(parsed["msgid"].is_null());
305        assert!(parsed["method"].is_null());
306        assert!(parsed["frame"].is_string());
307    }
