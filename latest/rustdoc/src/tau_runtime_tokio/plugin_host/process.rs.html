1//! Owns a running plugin subprocess: child handle, framer, dispatch
2//! task, stderr re-emit task, and in-flight call tracking. Shared
3//! across the per-port IPC adapters
4//! (`IpcLlmBackend` / `IpcTool` / etc., landing in Task 15) via
5//! [`std::sync::Arc`].
6//!
7//! The split between this module and
8//! [`crate::plugin_host::handshake`] mirrors the SDK side: `process`
9//! owns lifecycle (spawn / read-loop / stderr / shutdown);
10//! `handshake` is the host-side handshake driver that runs once
11//! before the read loop is allowed to consume inbound frames.
12//!
13//! See `docs/superpowers/specs/2026-04-28-plugin-loading-design.md`
14//! §7.3 (PluginProcess + read loop + stderr task + shutdown sequence)
15//! and §9.2 (wire-decode tracing).

35/// Type-erased async writer used by [`PluginProcess`] for outbound
36/// frames. Boxing here lets the same struct wrap a real
37/// [`tokio::process::ChildStdin`] (production spawn path) or an
38/// in-memory [`tokio::io::DuplexStream`] (`#[cfg(test)]` /
39/// `feature = "test-support"` path used by the per-port IPC adapter
40/// integration tests). The runtime cost is one virtual call per
41/// `write_all` — negligible against the rmp-serde encode and the
42/// kernel's tokio scheduler hop.
43pub type DynAsyncWriter = Box<dyn AsyncWrite + Send + Unpin>;

49/// Result delivered to a waiting `oneshot::Sender` for a host-issued
50/// request. `Ok(bytes)` carries the rmp-serde-encoded result payload;
51/// `Err(envelope)` carries the protocol-level error envelope from a
52/// `Frame::Response`. Cancellation (sender dropped because the read
53/// loop terminated) maps to a synthetic `Internal` envelope at the
54/// caller boundary in Task 15.
55pub(crate) type RpcResult = Result<Vec<u8>, RpcErrorEnvelope>;
56
57/// Shared map of in-flight request msgids → response delivery channel.
58pub(crate) type InFlightResponses = Arc<Mutex<HashMap<u32, oneshot::Sender<RpcResult>>>>;
59
60/// Shared map of in-flight stream msgids → chunk delivery channel.
61pub(crate) type InFlightStreams =
62    Arc<Mutex<HashMap<u32, mpsc::Sender<tau_ports::CompletionChunk>>>>;

64/// Owns a spawned plugin subprocess and the per-process dispatch
65/// machinery. Not constructed directly outside this module in the
66/// production path: the per-port `load_*` entry points wrap this in
67/// `Arc<dyn Dyn*>` shims after [`PluginProcess::spawn_and_handshake`]
68/// returns.
69///
70/// Public for the same `__internals` test-export reasons as
71/// [`super::ipc_llm::IpcLlmBackend`]; the `new_for_test` constructor
72/// gated by `feature = "test-support"` lets integration tests build
73/// instances without spawning a real subprocess.
74pub struct PluginProcess {
75    /// Plugin name (typically `LockedPlugin::manifest.name`). Used
76    /// for tracing fields and error messages.
77    pub(crate) name: String,
78    /// Monotonic msgid generator. Starts at 2; msgid `1` is reserved
79    /// for the handshake exchange driven before this struct is built.
80    pub(crate) next_msgid: AtomicU32,
81    /// Writer for outbound frames. `Mutex`-wrapped so multiple
82    /// concurrent IPC method calls can serialize their writes. The
83    /// concrete writer is type-erased ([`DynAsyncWriter`]) so the same
84    /// struct is shared between the production spawn path
85    /// (wrapping a [`tokio::process::ChildStdin`]) and the test path
86    /// (wrapping a [`tokio::io::DuplexStream`]).
87    pub(crate) writer: Mutex<FramedWriter<DynAsyncWriter>>,
88    /// Map of msgid → response delivery channel for outstanding
89    /// requests. The read loop completes entries here; the per-port
90    /// adapter inserts them before sending the request.
91    pub(crate) in_flight_responses: InFlightResponses,
92    /// Map of msgid → stream chunk delivery channel for outstanding
93    /// streaming requests. The read loop forwards `stream.chunk`
94    /// notifications here; [`super::ipc_llm::IpcLlmBackend::stream`]
95    /// inserts an entry per `llm.stream` call and pairs it with an
96    /// `in_flight_responses` entry under the same msgid. The
97    /// [`super::stream_router`] consumes both halves to assemble a
98    /// [`tau_ports::CompletionStream`].
99    pub(crate) in_flight_streams: InFlightStreams,
100    /// The spawned child process. Wrapped in `Mutex<Option<_>>` so
101    /// shutdown can `take()` it before awaiting `Child::wait`.
102    child: Mutex<Option<Child>>,
103    /// Notified when shutdown starts; the read/stderr tasks observe
104    /// this only via the underlying transport closing, not directly.
105    /// Reserved for Task 17's recording-flush coordination.
106    #[allow(dead_code)]
107    shutdown_signal: Notify,
108    /// Optional protocol recorder. When set, every outbound frame
109    /// (via [`PluginProcess::send_frame`]) and every inbound frame
110    /// (in the read loop) is mirrored to the recorder before
111    /// dispatch. See `crate::plugin_host::recording`.
112    pub(crate) recorder: Option<RecorderHandle>,
113    /// How long [`PluginProcess::shutdown`] waits between escalation
114    /// steps (post-`meta.shutdown` graceful → SIGTERM, and after
115    /// SIGTERM → SIGKILL). Default 2s from `PluginHostOptions`.
116    shutdown_timeout: Duration,
117    /// Read-loop join handle; kept so the runtime can observe it for
118    /// diagnostics. Dropped on `PluginProcess` drop, which aborts the
119    /// task if it is still running.
120    _read_task: JoinHandle<()>,
121    /// Stderr re-emit task join handle. Same drop semantics as
122    /// `_read_task`.
123    _stderr_task: JoinHandle<()>,
124    /// Holds adapter-side sandbox resources (e.g., container ID).
125    /// Dropped after the child exits, running adapter cleanup via
126    /// `CapabilityHandle::drop`. `Option` because test/no-sandbox paths
127    /// construct without one (via `new_for_test`).
128    ///
129    /// Wrapped in `Mutex` because `CapabilityHandle` contains a
130    /// `Box<dyn FnOnce()>` which is `!Sync`; the `Mutex` adds the
131    /// `Sync` bound required by `Arc<PluginProcess>` (the `Dyn*`
132    /// port traits are `Send + Sync`).
133    #[allow(dead_code)] // keeps the handle alive; cleanup runs on Mutex drop
134    _sandbox_handle: std::sync::Mutex<Option<CapabilityHandle>>,
135}

137/// Secret env-var names re-injected into every plugin child after
138/// [`tokio::process::Command::env_clear`]. This is an **explicit
139/// allowlist**, not the host's full environment: `env_clear()` exists
140/// precisely so plugins run in a minimal, reproducible env, and only
141/// these well-known secret names are passed back through.
142///
143/// One entry per shipped LLM plugin's default env-var name
144/// (`default_api_key_env` / `default_bearer_token_env`):
145/// - `ANTHROPIC_API_KEY` — `tau-plugins/anthropic`
146/// - `OPENAI_API_KEY` — `tau-plugins/openai`
147/// - `OLLAMA_BEARER_TOKEN` — `tau-plugins/ollama`
148///
149/// A plugin configured with a *custom* `api_key_env` name still needs
150/// the cleartext-config path or a future per-plugin env declaration; the
151/// host cannot enumerate arbitrary names without re-introducing the
152/// pass-everything behavior `env_clear()` removed.
153const SECRET_ENV_ALLOWLIST: &[&str] =
154    &["ANTHROPIC_API_KEY", "OPENAI_API_KEY", "OLLAMA_BEARER_TOKEN"];
155
156/// Apply the plugin child's minimal, reproducible environment to
157/// `command`: clear the inherited env, then re-add only the two
158/// `TAU_PLUGIN_*` identity vars, `PATH` (for shared-library
159/// resolution), and any [`SECRET_ENV_ALLOWLIST`] names present in the
160/// parent env (resolved via `env_lookup`).
161///
162/// Without the allowlist re-injection, `env_clear()` wiped env-provided
163/// secrets like `ANTHROPIC_API_KEY` before the plugin's `from_config`
164/// `std::env::var` lookup could read them — forcing real keys into
165/// plaintext `tau.toml`. Only allowlisted names cross over, so the
166/// security intent of `env_clear()` (no ambient host env) is preserved.
167///
168/// `env_lookup` is injected (rather than calling `std::env::var`
169/// directly) so the policy is unit-testable without spawning a
170/// subprocess or mutating the test process's real environment. The
171/// production caller passes `|n| std::env::var(n).ok()`.
172fn configure_plugin_command_env(
173    command: &mut Command,
174    run_id: &str,
175    agent_id: &str,
176    env_lookup: impl Fn(&str) -> Option<String>,
177) {
178    command
179        .env_clear()
180        .env("TAU_PLUGIN_RUN_ID", run_id)
181        .env("TAU_PLUGIN_AGENT_ID", agent_id)
182        // Inherit PATH so shared-library lookups (libc, libssl, …) work
183        // the same as for the host. Anything more ambient than PATH and
184        // the allowlisted secrets below should be added via the
185        // per-plugin config payload, not the env, so plugin behavior
186        // stays reproducible across hosts.
187        .env("PATH", env_lookup("PATH").unwrap_or_default());
188
189    // Re-inject env-provided secrets so plaintext `tau.toml` is no
190    // longer the only working path. Only allowlisted names cross over,
191    // and only when actually set in the parent env.
192    for name in SECRET_ENV_ALLOWLIST {
193        if let Some(value) = env_lookup(name) {
194            command.env(name, value);
195        }
196    }
197}

199/// Resolve each declared credential through `chain` and set it on the
200/// command's environment under the declaration's env name. Returns the
201/// env names actually injected (for assertability + tracing).
202///
203/// Resolution outcomes:
204/// * `Ok(Some(_))` → inject under `decl.env` (UTF-8 secrets only;
205///   non-UTF-8 surfaces [`RuntimeError::CredentialResolution`]).
206/// * `Ok(None)` → not found anywhere; leave whatever
207///   `configure_plugin_command_env` already set (today's env passthrough).
208///   Backward-compatible.
209/// * `Err(_)` → a provider owned the request and failed; surfaces
210///   [`RuntimeError::CredentialResolution`].
211///
212/// `chain == None` is a no-op (legacy embedders without a chain).
213async fn inject_credentials(
214    command: &mut tokio::process::Command,
215    chain: Option<&tau_ports::credential::CredentialChain>,
216    credentials: &[tau_pkg::project::project::AgentCredential],
217    plugin_name: &str,
218) -> Result<Vec<String>, RuntimeError> {
219    use tau_ports::credential::{CredentialProvider, CredentialRequest};
220    let Some(chain) = chain else {
221        return Ok(Vec::new());
222    };
223    let mut injected = Vec::new();
224    for decl in credentials {
225        let req = CredentialRequest::new(decl.id.clone()).with_env_name(decl.env.clone());
226        // Disambiguate: both `CredentialChain::resolve` (inherent via the
227        // `CredentialProvider` impl) and the `DynCredentialProvider::resolve`
228        // blanket method are in scope; qualify through `CredentialProvider`.
229        match CredentialProvider::resolve(chain, &req).await {
230            Ok(Some(resolved)) => match resolved.secret.expose_str() {
231                Ok(s) => {
232                    command.env(&decl.env, s);
233                    injected.push(decl.env.clone());
234                }
235                Err(_) => {
236                    return Err(RuntimeError::CredentialResolution {
237                        plugin: plugin_name.to_string(),
238                        id: decl.id.to_string(),
239                        reason: "resolved secret is not valid UTF-8".to_string(),
240                    })
241                }
242            },
243            Ok(None) => {} // not found anywhere: leave today's env passthrough.
244            Err(e) => {
245                return Err(RuntimeError::CredentialResolution {
246                    plugin: plugin_name.to_string(),
247                    id: decl.id.to_string(),
248                    reason: e.to_string(),
249                })
250            }
251        }
252    }
253    Ok(injected)
254}

256impl PluginProcess {
257    /// Spawn a plugin subprocess and install the framer, dispatch
258    /// read loop, and stderr re-emit task. The handshake is **not**
259    /// driven here — callers run [`crate::plugin_host::handshake`]
260    /// over a borrowed reader+writer first, then feed the post-
261    /// handshake reader into [`PluginProcess::install_read_loop`] via
262    /// this constructor.
263    ///
264    /// Steps:
265    ///
266    /// 1. `tokio::process::Command::spawn` with `stdin`/`stdout`/`stderr`
267    ///    piped, `env_clear()`, and the two `TAU_PLUGIN_*` env vars
268    ///    plus `PATH` for shared-library resolution.
269    /// 2. `kill_on_drop(true)` so a panicking host doesn't leak the
270    ///    child.
271    /// 3. The handshake driver runs against the borrowed
272    ///    `FramedReader`/`FramedWriter` returned through the
273    ///    `pre_handshake` closure.
274    /// 4. After the handshake completes, the read loop and stderr
275    ///    task spawn and the `PluginProcess` is returned.
276    ///
277    /// Errors:
278    ///
279    /// * [`RuntimeError::PluginSpawnFailed`] if `Command::spawn`
280    ///   fails (binary missing, not executable, sandbox denied, …).
281    /// * Whatever the `pre_handshake` closure returns.
282    // Nine params (two beyond clippy's default ceiling) is the natural
283    // shape of "spawn + sandbox + drive handshake" — splitting into a
284    // builder wouldn't simplify the call sites in
285    // `crate::plugin_host::load_*`.
286    #[allow(clippy::too_many_arguments)]
287    pub(crate) async fn spawn_and_handshake<F, T>(
288        binary_path: &Path,
289        plugin_name: String,
290        run_id: &str,
291        agent_id: &str,
292        framer_options: FramerOptions,
293        shutdown_timeout: Duration,
294        recorder: Option<RecorderHandle>,
295        // Optional sandbox plan + adapter pair. When `Some((plan, adapter))`:
296        // 1. `validate_plan_against_adapter` is called (Layer 3 cross-check).
297        // 2. `adapter.wrap_spawn(plan, &mut command)` is called to apply
298        //    enforcement. The resulting `CapabilityHandle` is stored on the
299        //    `PluginProcess` so its `Drop` runs adapter cleanup on plugin
300        //    exit. When `None`, the spawn proceeds without sandboxing (test
301        //    paths; use `MockSandbox` for behavioral tests instead).
302        sandbox: Option<(&CapabilityPlan, &SandboxAdapter)>,
303        // Resolved credential chain + the agent's credential declarations.
304        // For each declaration, the resolved secret is injected into the
305        // child's env under the declared name, AFTER the env scrub and
306        // BEFORE sandbox wrap_spawn. `None` chain (or empty decls) is a
307        // no-op preserving today's env passthrough.
308        credential_chain: Option<&tau_ports::credential::CredentialChain>,
309        agent_credentials: &[tau_pkg::project::project::AgentCredential],
310        pre_handshake: F,
311    ) -> Result<(Arc<PluginProcess>, T), RuntimeError>
312    where
313        // The HRTB on the future return type lets the closure body
314        // capture the `&mut` borrows without us having to spell out a
315        // separate `for<'a>` lifetime in the call site. The returned
316        // future is type-erased to `Pin<Box<dyn Future + 'a>>` so
317        // calling code can use `async move {...}` without tripping
318        // the higher-ranked-closure lifetime inference bug
319        // (rust-lang/rust#90696).
320        F: for<'a> FnOnce(
321            &'a mut FramedReader<ChildStdout>,
322            &'a mut FramedWriter<DynAsyncWriter>,
323        ) -> std::pin::Pin<
324            Box<dyn std::future::Future<Output = Result<T, RuntimeError>> + Send + 'a>,
325        >,
326    {
327        tracing::debug!(
328            target: "tau_runtime_tokio::plugin_host",
329            plugin = plugin_name.as_str(),
330            binary_path = ?binary_path,
331            "plugin.spawning"
332        );
333
334        let mut command = Command::new(binary_path);
335        command
336            .stdin(std::process::Stdio::piped())
337            .stdout(std::process::Stdio::piped())
338            .stderr(std::process::Stdio::piped())
339            .kill_on_drop(true);
340        // Clear the inherited env and re-add only the identity vars,
341        // PATH, and the secret-env allowlist (so env-provided keys like
342        // ANTHROPIC_API_KEY reach the plugin without plaintext config).
343        configure_plugin_command_env(&mut command, run_id, agent_id, |n| std::env::var(n).ok());
344
345        // Resolve declared credentials through the chain and inject them
346        // into the child's env, AFTER the env scrub and BEFORE sandbox
347        // wrap_spawn. Backward-compatible: no chain / no decls = no-op.
348        let injected = inject_credentials(
349            &mut command,
350            credential_chain,
351            agent_credentials,
352            &plugin_name,
353        )
354        .await?;
355        // Audit trail: record WHICH env names a secret was injected
356        // under (names only, never values) so a security-relevant
357        // injection leaves a trace.
358        if !injected.is_empty() {
359            tracing::info!(
360                target: "tau_runtime_tokio::plugin_host",
361                plugin = plugin_name.as_str(),
362                injected = ?injected,
363                "plugin_host.credentials_injected"
364            );
365        }
366
367        // Layer 3 + 4: sandbox validation and enforcement.
368        //
369        // Order is deliberate: validate_plan_against_adapter (Layer 3)
370        // runs BEFORE wrap_spawn (Layer 4) so a bad plan never reaches
371        // the adapter.
372        let sandbox_handle: Option<CapabilityHandle> = if let Some((plan, adapter)) = sandbox {
373            // Layer 3: cross-check plan capabilities against adapter shapes.
374            validate_plan_against_adapter(&plugin_name, plan, adapter).map_err(
375                |errors: Vec<SandboxValidationError>| RuntimeError::SandboxValidationFailed {
376                    plugin: plugin_name.clone(),
377                    errors,
378                },
379            )?;
380
381            // Layer 4: apply sandbox enforcement to the Command.
382            // `as_std_mut()` is the bridge between `tokio::process::Command`
383            // and `std::process::Command` required by `Sandbox::wrap_spawn`.
384            let handle = adapter
385                .wrap_spawn(plan, command.as_std_mut())
386                .await
387                .map_err(|source| RuntimeError::SandboxWrapFailed {
388                    plugin: plugin_name.clone(),
389                    source,
390                })?;
391            Some(handle)
392        } else {
393            None
394        };
395
396        let mut child = command
397            .spawn()
398            .map_err(|source| RuntimeError::PluginSpawnFailed {
399                plugin: plugin_name.clone(),
400                source,
401            })?;
402
403        let stdin = child
404            .stdin
405            .take()
406            .expect("stdin piped via stdin(Stdio::piped())");
407        let stdout = child
408            .stdout
409            .take()
410            .expect("stdout piped via stdout(Stdio::piped())");
411        let stderr = child
412            .stderr
413            .take()
414            .expect("stderr piped via stderr(Stdio::piped())");
415
416        let pid = child.id();
417        tracing::info!(
418            target: "tau_runtime_tokio::plugin_host",
419            plugin = plugin_name.as_str(),
420            pid,
421            "plugin.spawned"
422        );
423
424        // Box the writer up-front so `PluginProcess::writer` can hold a
425        // type-erased writer (this lets the test-only constructor share
426        // the same field type with a `DuplexStream`-backed writer).
427        // The handshake driver is generic over `W: AsyncWrite + Unpin`,
428        // so passing a `FramedWriter<DynAsyncWriter>` is sound.
429        let mut writer: FramedWriter<DynAsyncWriter> =
430            FramedWriter::new(Box::new(stdin) as DynAsyncWriter);
431        let mut reader = FramedReader::new(stdout, framer_options);
432
433        // Drive the handshake (or whatever the caller needs to run
434        // *before* the dispatch read loop starts consuming frames).
435        let handshake_outcome = match pre_handshake(&mut reader, &mut writer).await {
436            Ok(value) => value,
437            Err(err) => {
438                // Best-effort cleanup: kill the child so we don't leak
439                // it. `kill_on_drop` handles drop-time cleanup, but
440                // explicit kill yields more deterministic teardown
441                // for the failure path.
442                let _ = child.kill().await;
443                return Err(err);
444            }
445        };
446
447        let in_flight_responses: InFlightResponses = Arc::new(Mutex::new(HashMap::new()));
448        let in_flight_streams: InFlightStreams = Arc::new(Mutex::new(HashMap::new()));
449
450        let read_task = tokio::spawn(read_loop(
451            reader,
452            plugin_name.clone(),
453            in_flight_responses.clone(),
454            in_flight_streams.clone(),
455            recorder.clone(),
456        ));
457
458        let stderr_task = tokio::spawn(stderr_loop(stderr, plugin_name.clone()));
459
460        let process = Arc::new(PluginProcess {
461            name: plugin_name,
462            // msgid 1 was used for the handshake; subsequent calls
463            // start at 2.
464            next_msgid: AtomicU32::new(2),
465            writer: Mutex::new(writer),
466            in_flight_responses,
467            in_flight_streams,
468            child: Mutex::new(Some(child)),
469            shutdown_signal: Notify::new(),
470            recorder,
471            shutdown_timeout,
472            _read_task: read_task,
473            _stderr_task: stderr_task,
474            _sandbox_handle: std::sync::Mutex::new(sandbox_handle),
475        });
476
477        Ok((process, handshake_outcome))
478    }
479
480    /// Send a single outbound frame, tapping the protocol recorder
481    /// (if any) before acquiring the writer mutex. Centralizes the
482    /// host-side write path so per-port adapters
483    /// (`IpcLlmBackend` / `IpcTool` / `IpcStorage`) get
484    /// recording for free without each having to reach into the
485    /// recorder themselves.
486    ///
487    /// Returns the framer's [`ProtocolError`] verbatim on write failure
488    /// so call sites can wrap it in their port-specific error variants.
489    pub(crate) async fn send_frame(&self, frame_bytes: &[u8]) -> Result<(), ProtocolError> {
490        if let Some(recorder) = &self.recorder {
491            recorder.record(Direction::HostToPlugin, frame_bytes).await;
492        }
493        let mut writer = self.writer.lock().await;
494        writer.write_frame(frame_bytes).await
495    }

497    /// Drive the spec §7.3 shutdown sequence:
498    ///
499    /// 1. Send `meta.shutdown` notification (`[2, "meta.shutdown",
500    ///    []]`).
501    /// 2. Wait up to `shutdown_timeout` for the child to exit.
502    /// 3. If still alive, send SIGTERM (`Child::start_kill` =
503    ///    `SIGKILL` on Unix per `tokio` — see note below) and wait
504    ///    500 ms.
505    /// 4. If still alive, force-kill via `Child::kill`.
506    /// 5. Emit `plugin.exited` with `clean: bool`.
507    ///
508    /// **Tokio detail**: `tokio::process::Child` does not expose a
509    /// portable SIGTERM API. We approximate the spec sequence with
510    /// the available primitives: graceful → `start_kill` (SIGKILL on
511    /// Unix) after `shutdown_timeout` → `kill().await` (idempotent)
512    /// after a 500 ms grace. Replacing `start_kill` with explicit
513    /// SIGTERM via `nix` is a Phase-1 follow-up and additive to this
514    /// API.
515    #[allow(dead_code)] // wired by Task 24 via explicit close on Runtime drop.
516    pub(crate) async fn shutdown(self: Arc<Self>) {
517        // Step 1: send meta.shutdown notification (best-effort).
518        let shutdown_frame = Frame::Notification {
519            method: tau_plugin_protocol::handshake::meta::SHUTDOWN_METHOD.to_string(),
520            // params is an empty rmp-serde-encoded array. The smallest
521            // legitimate payload (per `Frame::encode`'s
522            // `EmptyFrameSlot` rejection) is the one-byte `[0x90]`
523            // (empty MessagePack array). Build it via rmp-serde so
524            // the wire encoding is canonical.
525            params: rmp_serde::to_vec::<Vec<()>>(&Vec::new())
526                .expect("encoding empty Vec<()> never fails"),
527        };
528        if let Ok(body) = shutdown_frame.encode() {
529            let mut writer = self.writer.lock().await;
530            let _ = writer.write_frame(&body).await;
531            tracing::debug!(
532                target: "tau_runtime_tokio::plugin_host",
533                plugin = self.name.as_str(),
534                "plugin.shutdown_sent"
535            );
536        }
537
538        // Step 2-5: take the child out of its mutex so we can wait/kill.
539        let mut guard = self.child.lock().await;
540        let Some(mut child) = guard.take() else {
541            return;
542        };
543        drop(guard);
544
545        // Wait up to `shutdown_timeout` for clean exit.
546        let exit_status = match tokio::time::timeout(self.shutdown_timeout, child.wait()).await {
547            Ok(Ok(status)) => Some((status, true)),
548            Ok(Err(_)) | Err(_) => None,
549        };
550
551        let (exit_status, clean) = if let Some(es) = exit_status {
552            es
553        } else {
554            // Step 3: SIGKILL fallback (tokio's only portable knob).
555            // Spec calls for SIGTERM first; see method docs.
556            let _ = child.start_kill();
557            match tokio::time::timeout(Duration::from_millis(500), child.wait()).await {
558                Ok(Ok(status)) => (status, false),
559                Ok(Err(_)) | Err(_) => {
560                    // Step 4: hard kill.
561                    let _ = child.kill().await;
562                    match child.wait().await {
563                        Ok(status) => (status, false),
564                        Err(_) => return, // process gone, give up
565                    }
566                }
567            }
568        };
569
570        let exit_code = exit_status.code();
571        tracing::info!(
572            target: "tau_runtime_tokio::plugin_host",
573            plugin = self.name.as_str(),
574            exit_code = ?exit_code,
575            clean,
576            "plugin.exited"
577        );
578    }

581    /// without spawning a subprocess.
582    ///
583    /// Used by the per-port IPC adapter integration tests in
584    /// `tau-runtime/tests/plugin_host_ipc_*.rs`: the test harness pairs
585    /// a [`tau_plugin_protocol::test_support::FakeStdioPeer`] with the
586    /// host-side adapter via two duplex streams, drives the handshake
587    /// out of band (or skips it for unit-style tests), and constructs
588    /// the [`PluginProcess`] via this constructor so the adapter has a
589    /// working `Arc<PluginProcess>` to dispatch through.
590    ///
591    /// # Caller responsibilities
592    ///
593    /// - The handshake must already have run on `reader`+`writer`
594    ///   (msgid 1 consumed).
595    /// - `next_msgid` is initialized to 2 to match
596    ///   [`PluginProcess::spawn_and_handshake`]'s post-handshake state.
597    /// - There is no underlying child process: the [`PluginProcess::shutdown`]
598    ///   path is a no-op for test instances.
599    #[cfg(any(test, feature = "test-support"))]
600    pub fn new_for_test<R>(
601        plugin_name: String,
602        reader: FramedReader<R>,
603        writer: FramedWriter<DynAsyncWriter>,
604        shutdown_timeout: Duration,
605    ) -> Arc<Self>
606    where
607        R: AsyncRead + Unpin + Send + 'static,
608    {

610    }

612    /// Variant of [`PluginProcess::new_for_test`] that accepts an
613    /// optional recorder. Used by `tau-runtime`'s recording integration
614    /// tests to verify that the read- and write-side tap points fire
615    /// without spawning a real subprocess. Production code must not
616    /// reach for this constructor; the public `load_*` entry points
617    /// build the recorder from [`crate::plugin_host::PluginHostOptions`]
618    /// and feed it into [`PluginProcess::spawn_and_handshake`].
619    #[cfg(any(test, feature = "test-support"))]
620    pub fn new_for_test_with_recorder<R>(
621        plugin_name: String,
622        reader: FramedReader<R>,
623        writer: FramedWriter<DynAsyncWriter>,
624        shutdown_timeout: Duration,
625        recorder: Option<RecorderHandle>,
626    ) -> Arc<Self>
627    where
628        R: AsyncRead + Unpin + Send + 'static,
629    {
630        let in_flight_responses: InFlightResponses = Arc::new(Mutex::new(HashMap::new()));
631        let in_flight_streams: InFlightStreams = Arc::new(Mutex::new(HashMap::new()));
632
633        let read_task = tokio::spawn(read_loop(
634            reader,
635            plugin_name.clone(),
636            in_flight_responses.clone(),
637            in_flight_streams.clone(),
638            recorder.clone(),
639        ));
640
641        // Test instances have no real stderr to re-emit; spawn a stub
642        // task that immediately returns so the field has a valid
643        // `JoinHandle<()>` to drop with.
644        let stderr_task = tokio::spawn(async {});
645
646        Arc::new(PluginProcess {
647            name: plugin_name,
648            next_msgid: AtomicU32::new(2),
649            writer: Mutex::new(writer),
650            in_flight_responses,
651            in_flight_streams,
652            child: Mutex::new(None),
653            shutdown_signal: Notify::new(),
654            recorder,
655            shutdown_timeout,
656            _read_task: read_task,
657            _stderr_task: stderr_task,
658            // Test constructors have no sandbox; cleanup is a no-op.
659            _sandbox_handle: std::sync::Mutex::new(None),
660        })
661    }
662}

664/// Per-process dispatch read loop. Runs on a dedicated tokio task
665/// spawned by [`PluginProcess::spawn_and_handshake`]. Terminates on
666/// EOF or framer error and drains both in-flight maps so callers
667/// don't await forever.
668///
669/// Generic over the reader IO type so the production spawn path can
670/// pass a [`tokio::process::ChildStdout`] and the test path can pass
671/// a [`tokio::io::DuplexStream`].
672async fn read_loop<R>(
673    mut reader: FramedReader<R>,
674    plugin_name: String,
675    in_flight_responses: InFlightResponses,
676    in_flight_streams: InFlightStreams,
677    recorder: Option<RecorderHandle>,
678) where
679    R: AsyncRead + Unpin + Send + 'static,
680{
681    loop {
682        let body = match reader.next_frame().await {
683            Ok(Some(b)) => b,
684            Ok(None) => {
685                tracing::info!(
686                    target: "tau_runtime_tokio::plugin_host",
687                    plugin = plugin_name.as_str(),
688                    "plugin stdout EOF"
689                );
690                break;
691            }
692            Err(err) => {
693                tracing::error!(
694                    target: "tau_runtime_tokio::plugin_host",
695                    plugin = plugin_name.as_str(),
696                    err = %err,
697                    "plugin stdout frame read error"
698                );
699                break;
700            }
701        };
702        // Tap the inbound frame *before* decode-and-dispatch so the
703        // recorded byte sequence matches exactly what came off the wire
704        // — including frames whose payload `Frame::decode` later
705        // rejects (those still get logged with `null` msgid/method via
706        // `decode_frame_metadata` and the raw base64 frame).
707        if let Some(recorder) = &recorder {
708            recorder.record(Direction::PluginToHost, &body).await;
709        }
710        let frame = match Frame::decode(&body) {
711            Ok(f) => f,
712            Err(err) => {
713                tracing::warn!(
714                    target: "tau_runtime_tokio::plugin_host",
715                    plugin = plugin_name.as_str(),
716                    err = %err,
717                    "plugin frame decode failed"
718                );
719                continue;
720            }
721        };
722        match frame {
723            Frame::Response { id, error, result } => {
724                tracing::trace!(
725                    target: "tau_runtime_tokio::plugin_host",
726                    plugin = plugin_name.as_str(),
727                    msgid = id,
728                    error_code = error.as_ref().map(|e| e.code),
729                    "plugin.response_received"
730                );
731                let mut map = in_flight_responses.lock().await;
732                if let Some(sender) = map.remove(&id) {
733                    let value = match (error, result) {
734                        (Some(env), _) => Err(env),
735                        // A `null` result is legitimate (per
736                        // `Frame::decode`'s `Value::Nil → None`
737                        // mapping); represent it as a zero-byte
738                        // payload so downstream decoders can detect
739                        // it. Per-port adapters in Task 15 are
740                        // responsible for treating empty-bytes as
741                        // "null" rather than as
742                        // rmp-serde-deserialization fodder.
743                        (None, Some(bytes)) => Ok(bytes),
744                        (None, None) => Ok(Vec::new()),
745                    };
746                    let _ = sender.send(value);
747                } else {
748                    tracing::warn!(
749                        target: "tau_runtime_tokio::plugin_host",
750                        plugin = plugin_name.as_str(),
751                        msgid = id,
752                        "response with no matching in-flight request"
753                    );
754                }
755            }
756            Frame::Notification { method, params } if method == STREAM_CHUNK_METHOD => {
757                // params shape: [originating_msgid, CompletionChunk]
758                // (see test_support::send_stream_chunk + spec §7.3).
759                let parsed: (u32, tau_ports::CompletionChunk) = match rmp_serde::from_slice(&params)
760                {
761                    Ok(p) => p,
762                    Err(err) => {
763                        tracing::warn!(
764                            target: "tau_runtime_tokio::plugin_host",
765                            plugin = plugin_name.as_str(),
766                            err = %err,
767                            "stream.chunk params decode failed"
768                        );
769                        continue;
770                    }
771                };
772                let (originating_id, chunk) = parsed;
773                tracing::trace!(
774                    target: "tau_runtime_tokio::plugin_host",
775                    plugin = plugin_name.as_str(),
776                    msgid = originating_id,
777                    "plugin.stream_chunk"
778                );
779                let map = in_flight_streams.lock().await;
780                if let Some(tx) = map.get(&originating_id) {
781                    // Best-effort send; if the receiver dropped, the
782                    // chunk is silently discarded (the streaming call
783                    // was cancelled or the host stopped reading).
784                    let _ = tx.send(chunk).await;
785                }
786            }
787            // Other notifications and any plugin-initiated requests
788            // are not part of the v0.1 host contract; ignore them.
789            _ => {}
790        }
791    }
792
793    // EOF/error path: notify all waiting callers that the plugin
794    // exited mid-call so they don't await forever. Streams have their
795    // senders dropped which terminates the receiver naturally.
796    let mut responses = in_flight_responses.lock().await;
797    for (_, sender) in responses.drain() {
798        let _ = sender.send(Err(RpcErrorEnvelope::new(
799            INTERNAL_ERROR,
800            "plugin process exited mid-call".to_string(),
801            None,
802        )));
803    }
804    let mut streams = in_flight_streams.lock().await;
805    streams.clear();
806}

808/// Re-emit plugin stderr lines as host-side tracing events. The SDK
809/// formats events as JSON via [`tau-plugin-sdk`]'s tracing layer; we
810/// best-effort parse each line and pull `level`/`message` to fan them
811/// out into the right `tracing::*!` macro under the
812/// `target = "plugin"` namespace, with the `plugin = <name>` field
813/// preserved so subscribers can filter.
814async fn stderr_loop(stderr: ChildStderr, plugin_name: String) {
815    let reader = tokio::io::BufReader::new(stderr);
816    let mut lines = reader.lines();
817    loop {
818        match lines.next_line().await {
819            Ok(Some(line)) => emit_plugin_line(&plugin_name, &line),
820            Ok(None) => break,
821            Err(err) => {
822                tracing::warn!(
823                    target: "tau_runtime_tokio::plugin_host",
824                    plugin = plugin_name.as_str(),
825                    err = %err,
826                    "plugin stderr read error"
827                );
828                break;
829            }
830        }
831    }
832}
833
834/// Notification method for streaming chunks (see spec §4.6 and §7.3).
835const STREAM_CHUNK_METHOD: &str = "stream.chunk";
836
837fn emit_plugin_line(plugin_name: &str, line: &str) {
838    // Try parsing as a JSON object first (the SDK's tracing layer
839    // shape). If that fails, emit the raw line under
840    // `target = "plugin"` at WARN.
841    match serde_json::from_str::<serde_json::Value>(line) {
842        Ok(json) => {
843            let level = json
844                .get("level")
845                .and_then(serde_json::Value::as_str)
846                .unwrap_or("INFO");
847            let message = json
848                .get("message")
849                .and_then(serde_json::Value::as_str)
850                .or_else(|| {
851                    json.get("fields")
852                        .and_then(|f| f.get("message"))
853                        .and_then(serde_json::Value::as_str)
854                })
855                .unwrap_or("");
856            // `tracing` macros require static `target` strings, so
857            // we use the constant `"plugin"` target and carry the
858            // plugin name as a structured field.
859            match level {
860                "ERROR" => tracing::error!(target: "plugin", plugin = plugin_name, "{message}"),
861                "WARN" => tracing::warn!(target: "plugin", plugin = plugin_name, "{message}"),
862                "DEBUG" => tracing::debug!(target: "plugin", plugin = plugin_name, "{message}"),
863                "TRACE" => tracing::trace!(target: "plugin", plugin = plugin_name, "{message}"),
864                _ => tracing::info!(target: "plugin", plugin = plugin_name, "{message}"),
865            }
866        }
867        Err(_) => {
868            tracing::warn!(
869                target: "plugin",
870                plugin = plugin_name,
871                raw = line,
872                "non-json plugin stderr"
873            );
874        }
875    }
