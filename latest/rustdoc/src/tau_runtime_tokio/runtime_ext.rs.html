1//! Host-shell free-function wrapper for `Runtime::spawn_root_agent_inner`.
2//!
3//! Pre-β.1.3.5c this module hosted a `RuntimeShellExt` extension trait
4//! providing `run`, `run_with_history`, `run_streaming`, `invoke_tool`,
5//! `spawn_root_agent`, and `run_default` on
6//! [`tau_runtime_core::Runtime`]. Those methods now live as inherent
7//! methods on the kernel `Runtime` directly; only the JSONL-persistence
8//! wiring of `spawn_root_agent` remained tokio-shell-specific, so this
9//! module shrinks to a single free function that builds the subscriber
10//! from the scope_root and delegates to the kernel inner.
11
12use std::path::PathBuf;
13use std::sync::Arc;
14
15use tau_domain::{AgentDefinition, Message, PackageManifest};
16use tau_ports::{RunBudget, RunSnapshot};
17
18use crate::builder::Runtime;
19use crate::error::RuntimeError;
20
21/// Multi-agent orchestrated run entry point (ROADMAP §9, v1).
22///
23/// Builds the JSONL run-log subscriber from `scope_root` and delegates
24/// to [`Runtime::spawn_root_agent_inner`].
25pub async fn spawn_root_agent_with_scope(
26    runtime: Arc<Runtime>,
27    root_agent_def: AgentDefinition,
28    root_manifest: PackageManifest,
29    initial_message: Message,
30    budget: RunBudget,
31    scope_root: PathBuf,
32) -> Result<RunSnapshot, RuntimeError> {
33    // Mint the run-id up front so the same id flows into the JSONL path and the
34    // RunState. Use the tokio shell's real clock/random so every orchestrated run
35    // gets a unique, wall-clock-ordered ULID. (The previous MockClock +
36    // DeterministicRandom::seeded(0) minted a FIXED run-id on every call — a
37    // latent collision bug — and forced `tau-ports/test-fixtures` into the
38    // production dependency graph.)
39    let clock: Arc<dyn tau_ports::Clock> = Arc::new(crate::TokioClock);
40    let random: Arc<dyn tau_ports::RandomSource> = Arc::new(crate::OsRandom);
41    let run_id = tau_runtime_core::ids::ulid(&clock, &random);
42
43    // Build the JSONL subscriber via the tokio mpsc channel.
44    let log_path = crate::orchestration::persistence::run_log_path(&scope_root, &run_id);
45    let subscriber = crate::orchestration::trace_mpsc::channel_with_writer(log_path);
46
47    // Delegate to the core's inner. Passing `run_id` keeps the inner's
48    // RunState in sync with the JSONL log path we just constructed.
49    let scope_root_str = scope_root.to_string_lossy().into_owned();
50    let skill_resolver: std::sync::Arc<dyn tau_ports::SkillResolver> = std::sync::Arc::new(
51        crate::skill_resolver_impl::TauPkgSkillResolver::new(scope_root.clone()),
52    );
53    runtime
54        .spawn_root_agent_inner(
55            root_agent_def,
56            root_manifest,
57            initial_message,
58            budget,
59            vec![subscriber],
60            Some(clock),
61            Some(random),
62            Some(run_id),
63            Some(scope_root_str),
64            Some(skill_resolver),
65        )
66        .await
67        .map_err(RuntimeError::Core)
68}
