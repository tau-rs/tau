1//! Tokio-shell zero-config entry: drive a [`Runtime`] with the production
2//! [`TokioClock`] + [`OsRandom`] defaults wired into
3//! [`tau_runtime_core::options::RunOptions`].
4//!
5//! This is the canonical production entry point for the tokio host shell.
6//! [`crate::runtime_ext::spawn_root_agent_with_scope`] retains
7//! `MockClock` + `DeterministicRandom` for tests that depend on
8//! deterministic run IDs.
9
10use core::future::Future;
11use std::path::PathBuf;
12use std::sync::Arc;
13
14use tau_domain::{AgentDefinition, Message, PackageManifest};
15use tau_ports::{Clock, RandomSource, RunBudget, RunSnapshot, TraceEvent};
16use tokio::sync::mpsc::UnboundedReceiver;
17
18use crate::builder::Runtime;
19use crate::error::RuntimeError;
20use crate::{OsRandom, TokioClock};

22/// Drive a multi-agent orchestrated run with tokio-shell defaults.
23///
24/// Mints the run-id from [`OsRandom`], stamps wall-clock from
25/// [`TokioClock`], builds the JSONL run-log subscriber from `scope_root`,
26/// and delegates to [`Runtime::spawn_root_agent_inner`].
27///
28/// CLI / workflow callers reach this through
29/// [`crate::runtime_ext::spawn_root_agent_with_scope`] only when they want
30/// deterministic test fixtures; production paths should call `drive` so
31/// run IDs and timestamps reflect real wall-clock.
32///
33/// This is the non-observing entry point: it attaches only the JSONL
34/// run-log writer and awaits the run to completion. Callers that want to
35/// render the trace live during the run use [`drive_with_live_trace`].
36pub async fn drive(
37    runtime: Arc<Runtime>,
38    root_agent_def: AgentDefinition,
39    root_manifest: PackageManifest,
40    initial_message: Message,
41    budget: RunBudget,
42    scope_root: PathBuf,
43) -> Result<RunSnapshot, RuntimeError> {
44    let (rx, run) = drive_with_live_trace(
45        runtime,
46        root_agent_def,
47        root_manifest,
48        initial_message,
49        budget,
50        scope_root,
51    );
52    // No live consumer: drop the receiver so the live subscriber's sends
53    // become no-ops (its channel is closed). The JSONL writer subscriber is
54    // unaffected, so persistence is identical to the un-observed path.
55    drop(rx);
56    run.await
57}
58
59/// Like [`drive`], but attaches a second, live trace subscriber and hands
60/// its receiver back so the caller can render the trace *as the run
61/// progresses* (issue #469) rather than reading the JSONL log post-hoc.
62///
63/// Returns `(rx, run)`:
64///
65/// - `rx` yields every [`TraceEvent`] as it is emitted; it closes once the
66///   run ends and every subscriber (including the one held by the run
67///   future) is dropped.
68/// - `run` is the run future; awaiting it yields the final
69///   [`RunSnapshot`].
70///
71/// Drive both concurrently — e.g. `tokio::join!(run, printer(rx))`. The
72/// JSONL run-log writer is attached exactly as in [`drive`], so
73/// persistence is unchanged; the live subscriber is purely additive.
74///
75/// The returned future is `'static` (it owns every input), so callers may
76/// also `tokio::spawn` it if they prefer.
77pub fn drive_with_live_trace(
78    runtime: Arc<Runtime>,
79    root_agent_def: AgentDefinition,
80    root_manifest: PackageManifest,
81    initial_message: Message,
82    budget: RunBudget,
83    scope_root: PathBuf,
84) -> (
85    UnboundedReceiver<TraceEvent>,
86    impl Future<Output = Result<RunSnapshot, RuntimeError>>,
87) {
88    let clock: Arc<dyn Clock> = Arc::new(TokioClock);
89    let random: Arc<dyn RandomSource> = Arc::new(OsRandom);
90
91    let run_id = tau_runtime_core::ids::ulid(&clock, &random);
92
93    let log_path = crate::orchestration::persistence::run_log_path(&scope_root, &run_id);
94    let writer_subscriber = crate::orchestration::trace_mpsc::channel_with_writer(log_path);
95
96    // Second subscriber: a live receiver handed back to the caller. The
97    // orchestration core's `TraceStream` already fans out to N subscribers,
98    // so this rides alongside the JSONL writer with no changes to the core.
99    let (live_subscriber, rx) = crate::orchestration::trace_mpsc::MpscTraceSubscriber::channel();
100
101    let scope_root_str = scope_root.to_string_lossy().into_owned();
102    let skill_resolver: Arc<dyn tau_ports::SkillResolver> = Arc::new(
103        crate::skill_resolver_impl::TauPkgSkillResolver::new(scope_root.clone()),
104    );
105
106    let run = async move {
107        runtime
108            .spawn_root_agent_inner(
109                root_agent_def,
110                root_manifest,
111                initial_message,
112                budget,
113                vec![writer_subscriber, Arc::new(live_subscriber)],
114                Some(clock),
115                Some(random),
116                Some(run_id),
117                Some(scope_root_str),
118                Some(skill_resolver),
119            )
120            .await
121            .map_err(RuntimeError::Core)
122    };
123
124    (rx, run)
125}
