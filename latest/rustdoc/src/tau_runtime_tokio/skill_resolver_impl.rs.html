1//! `tau_pkg`-backed implementation of the [`SkillResolver`] port.
2//!
3//! Wraps `tau_pkg::find_installed_skill` + the SKILL.md read so the
4//! kernel can resolve `skill.<name>.spawn` targets through a stable trait
5//! without linking `tau-pkg` (which pulls tokio/rustix and does not
6//! cross-compile to `wasm32-wasip2`).
7//!
8//! Construct with [`TauPkgSkillResolver::new`] from the project scope
9//! root; the kernel calls `resolve()` each time a `skill.<name>.spawn`
10//! virtual tool fires.
11
12use std::path::PathBuf;
13
14use tau_pkg::{find_installed_skill, FindSkillError, Scope};
15use tau_ports::{ResolvedSkill, SkillResolveError, SkillResolver};
16
17/// Production skill resolver: resolves a `tau_pkg::Scope` from the
18/// project scope root, looks up the installed skill, and reads its
19/// SKILL.md body.
20///
21/// Build once per orchestrated run from the scope root and stuff
22/// `Arc<TauPkgSkillResolver>` into `RunOptions.skill_resolver`.
23pub struct TauPkgSkillResolver {
24    scope_root: PathBuf,
25}
26
27impl TauPkgSkillResolver {
28    /// Construct from the project scope root (the directory containing
29    /// `.tau/`). Scope resolution is deferred to `resolve()` so a bad
30    /// root surfaces as a typed `SkillResolveError::Invalid` rather than
31    /// a constructor panic.
32    pub fn new(scope_root: PathBuf) -> Self {
33        Self { scope_root }
34    }
35}
36
37impl SkillResolver for TauPkgSkillResolver {
38    fn resolve(&self, name: &str) -> Result<ResolvedSkill, SkillResolveError> {
39        let scope = Scope::resolve(&self.scope_root).map_err(|e| SkillResolveError::Invalid {
40            detail: format!("resolving scope at {:?}: {e}", self.scope_root),
41        })?;
42
43        let installed = match find_installed_skill(&scope, name) {
44            Ok(Some(s)) => s,
45            Ok(None) => return Err(SkillResolveError::NotFound),
46            Err(FindSkillError::InstallPathMissing { path, .. }) => {
47                return Err(SkillResolveError::InstallPathMissing {
48                    expected_path: path.display().to_string(),
49                });
50            }
51            Err(e) => {
52                return Err(SkillResolveError::Invalid {
53                    detail: e.to_string(),
54                });
55            }
56        };
57
58        // Read + parse SKILL.md for the default system prompt.
59        let skill_md_path = installed.install_path.join(&installed.skill.content);
60        let text =
61            std::fs::read_to_string(&skill_md_path).map_err(|e| SkillResolveError::Invalid {
62                detail: format!("reading SKILL.md at {skill_md_path:?}: {e}"),
63            })?;
64        let parsed = tau_domain::parse_skill_md(&text).map_err(|e| SkillResolveError::Invalid {
65            detail: format!("parsing SKILL.md: {e}"),
66        })?;
67
68        Ok(ResolvedSkill {
69            install_path: installed.install_path.display().to_string(),
70            capabilities: installed.capabilities,
71            system_prompt: parsed.body,
72        })
73    }
74}

76#[cfg(test)]
77mod tests {
78    use super::*;
79
80    /// Point `$TAU_HOME` at a process-local tempdir so the global-scope
81    /// fallback resolves.
82    ///
83    /// `Scope::resolve` walks up for `.tau/`, finds none, and falls back to
84    /// `Scope::global()`, which reads `$TAU_HOME`/`$XDG_DATA_HOME`/`$HOME`. On
85    /// Windows CI none are set, so resolution fails with `HomeNotFound`
86    /// (surfacing as `SkillResolveError::Invalid`) instead of the intended
87    /// `NotFound`. A dedicated tempdir with `config.toml` pre-created, set once,
88    /// gives it a real home without racing on the write. Mirrors
89    /// `e2e_common::ensure_home_env` (added in #545). See #530.
90    fn ensure_home_env() {
91        use std::sync::OnceLock;
92        static TAU_HOME_INIT: OnceLock<()> = OnceLock::new();
93        TAU_HOME_INIT.get_or_init(|| {
94            let dir = std::env::temp_dir()
95                .join(format!("tau-skill-resolver-test-{}", std::process::id()));
96            std::fs::create_dir_all(&dir).expect("create tau_home tempdir");
97            let cfg = dir.join("config.toml");
98            if !cfg.exists() {
99                let _ = std::fs::write(&cfg, "");
100            }
101            std::env::set_var("TAU_HOME", dir);
102        });
103    }
104
105    #[test]
106    fn no_lockfile_scope_returns_not_found() {
107        // A tempdir with no `.tau/` anywhere up the tree resolves to a
108        // global scope whose lockfile does not exist, so
109        // `find_installed_skill` returns `Ok(None)` → `NotFound`.
110        // This exercises the real `Scope::resolve` + lookup path
111        // without requiring an installed skill.
112        ensure_home_env();
113        let dir = tempfile::tempdir().expect("tempdir");
114        let resolver = TauPkgSkillResolver::new(dir.path().to_path_buf());
115        let err = resolver.resolve("definitely-not-installed").unwrap_err();
116        assert!(matches!(err, SkillResolveError::NotFound));
117    }
118}
