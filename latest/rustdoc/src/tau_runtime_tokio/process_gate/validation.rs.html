1//! Layer 3 pre-flight plan validation.
2//!
3//! [`validate_plan_against_adapter`] cross-checks every capability in a
4//! [`tau_ports::CapabilityPlan`] against the adapter's declared
5//! `supported_shapes`. All validation errors are collected and returned at
6//! once so callers see the complete picture in a single pass.
7
8use tau_domain::Capability;
9use tau_ports::{CapabilityGate, CapabilityPlan};
10
11/// A single capability-shape rejection produced by
12/// [`validate_plan_against_adapter`].
13///
14/// Carries enough context for callers to format a user-facing message
15/// without further synthesis.
16#[non_exhaustive]
17#[derive(Debug, Clone)]
18pub struct SandboxValidationError {
19    /// Identifier the caller passed in (typically a plugin id).
20    pub plan_id: String,
21    /// The capability that produced the error.
22    pub capability: Capability,
23    /// Human-readable reason — typically the shape rejected and what
24    /// the adapter supports. Concrete enough to render directly to the
25    /// user without further synthesis.
26    pub reason: String,
27}
28
29impl SandboxValidationError {
30    /// Construct a [`SandboxValidationError`].
31    ///
32    /// `#[non_exhaustive]` blocks struct-literal construction outside this
33    /// crate; use this constructor instead.
34    ///
35    /// # Example
36    ///
37    /// ```
38    /// use tau_domain::Capability;
39    /// use tau_runtime_tokio::process_gate::SandboxValidationError;
40    ///
41    /// let cap: Capability = serde_json::from_value(serde_json::json!({"kind": "fs.read", "paths": ["/tmp/**"]})).unwrap();
42    /// let err = SandboxValidationError::new("my-plugin", cap, "shape not supported");
43    /// assert_eq!(err.plan_id, "my-plugin");
44    /// assert!(err.to_string().contains("my-plugin"));
45    /// assert!(err.to_string().contains("shape not supported"));
46    /// ```
47    pub fn new(
48        plan_id: impl Into<String>,
49        capability: Capability,
50        reason: impl Into<String>,
51    ) -> Self {
52        Self {
53            plan_id: plan_id.into(),
54            capability,
55            reason: reason.into(),
56        }
57    }
58}
59
60impl std::fmt::Display for SandboxValidationError {
61    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
62        write!(f, "plan {}: {}", self.plan_id, self.reason)
63    }
64}
65
66impl std::error::Error for SandboxValidationError {}

68/// Cross-check every capability in `plan` against `adapter.supported_shapes()`.
69///
70/// **All** unsupported capabilities are collected before returning — the error
71/// `Vec` may contain more than one entry. This is the key value of Layer 3:
72/// a single `tau check` run surfaces every problem at once.
73///
74/// # Arguments
75///
76/// * `plan_id` — free-form identifier carried into each [`SandboxValidationError`]
77///   so callers can format messages like `"plugin foo: capability fs.read shape
78///   unsupported by adapter X"`. Typically a plugin id.
79/// * `plan` — the [`CapabilityPlan`] to validate.
80/// * `adapter` — any [`CapabilityGate`] implementor (e.g. `SandboxAdapter` from the
81///   chain, `MockCapabilityGate` from fixtures, or another test double).
82///
83/// # Returns
84///
85/// `Ok(())` if every capability shape in `plan` is in `adapter.supported_shapes()`.
86/// `Err(errors)` with the complete list of failures otherwise.
87///
88/// # Example
89///
90/// ```
91/// use tau_domain::Capability;
92/// use tau_ports::{CapabilityPlan, fixtures::MockCapabilityGate};
93/// use tau_runtime_tokio::process_gate::validate_plan_against_adapter;
94///
95/// let adapter = MockCapabilityGate::new("mock");
96///
97/// // A standard fs.read capability is supported by MockCapabilityGate.
98/// let cap: Capability = serde_json::from_value(
99///     serde_json::json!({"kind": "fs.read", "paths": ["/tmp/**"]})
100/// ).unwrap();
101/// let plan = CapabilityPlan::new(vec![cap], None, None);
102/// assert!(validate_plan_against_adapter("my-plugin", &plan, &adapter).is_ok());
103///
104/// // An empty plan always passes.
105/// let empty = CapabilityPlan::new(vec![], None, None);
106/// assert!(validate_plan_against_adapter("my-plugin", &empty, &adapter).is_ok());
107/// ```
108pub fn validate_plan_against_adapter<S: CapabilityGate>(
109    plan_id: &str,
110    plan: &CapabilityPlan,
111    adapter: &S,
112) -> Result<(), Vec<SandboxValidationError>> {
113    let supported = adapter.supported_shapes();
114
115    let sup_list: String = supported
116        .iter()
117        .map(|s| format!("{s:?}"))
118        .collect::<Vec<_>>()
119        .join(", ");
120
121    let mut errors: Vec<SandboxValidationError> = Vec::new();
122
123    for cap in &plan.capabilities {
124        let required = cap.required_shape();
125        if !supported.contains(&required) {
126            let reason =
127                format!("adapter does not support shape {required:?} (supported: {sup_list})");
128            errors.push(SandboxValidationError::new(plan_id, cap.clone(), reason));
129        }
130    }
131
132    if errors.is_empty() {
133        Ok(())
134    } else {
135        Err(errors)
136    }
137}

139#[cfg(test)]
140mod tests {
141    use super::*;
142    use tau_domain::CapabilityShape;
143    use tau_ports::fixtures::MockCapabilityGate;
144
145    fn cap(json: &str) -> Capability {
146        serde_json::from_str(json).expect("test capability JSON must be valid")
147    }
148
149    fn plan_with(caps: Vec<Capability>) -> CapabilityPlan {
150        CapabilityPlan::new(caps, None, None)
151    }
152
153    #[test]
154    fn validation_passes_when_all_shapes_supported() {
155        // MockCapabilityGate supports the 5 standard shapes (not Custom).
156        let adapter = MockCapabilityGate::new("mock");
157        let plan = plan_with(vec![cap(r#"{"kind":"fs.read","paths":["/tmp/**"]}"#)]);
158        assert!(
159            validate_plan_against_adapter("test-plugin", &plan, &adapter).is_ok(),
160            "expected Ok for fs.read on MockCapabilityGate"
161        );
162    }
163
164    #[test]
165    fn validation_returns_all_errors_not_just_first() {
166        // Two Custom capabilities — MockCapabilityGate does not support CapabilityShape::Custom.
167        let adapter = MockCapabilityGate::new("mock");
168        let plan = plan_with(vec![
169            cap(r#"{"kind":"custom.mcp.tool.use","tool":"x"}"#),
170            cap(r#"{"kind":"custom.mcp.tool.other","tool":"y"}"#),
171        ]);
172        let errors = validate_plan_against_adapter("test-plugin", &plan, &adapter)
173            .expect_err("expected validation errors for Custom capabilities");
174        assert_eq!(
175            errors.len(),
176            2,
177            "both custom capabilities should produce errors, got {:?}",
178            errors.len()
179        );
180        for e in &errors {
181            assert_eq!(e.plan_id, "test-plugin");
182        }
183    }
184
185    #[test]
186    fn validation_includes_plan_id_in_each_error() {
187        let adapter = MockCapabilityGate::new("mock");
188        let plan = plan_with(vec![cap(r#"{"kind":"custom.mcp.tool.use","tool":"x"}"#)]);
189        let errors = validate_plan_against_adapter("test-plugin", &plan, &adapter)
190            .expect_err("expected error for Custom capability");
191        assert_eq!(errors.len(), 1);
192        assert_eq!(errors[0].plan_id, "test-plugin");
193    }
194
195    #[test]
196    fn validation_empty_plan_is_ok() {
197        let adapter = MockCapabilityGate::new("mock");
198        let plan = plan_with(vec![]);
199        assert!(
200            validate_plan_against_adapter("test-plugin", &plan, &adapter).is_ok(),
201            "empty plan should always pass"
202        );
203    }
204
205    #[test]
206    fn sandbox_validation_error_display_includes_id_and_reason() {
207        let cap_val = cap(r#"{"kind":"custom.mcp.tool.use","tool":"x"}"#);
208        let e = SandboxValidationError::new(
209            "my-plugin",
210            cap_val,
211            "adapter does not support shape Custom { name: \"custom.mcp.tool.use\" } (supported: FilesystemRead)",
212        );
213        let display = format!("{e}");
214        assert!(
215            display.contains("my-plugin"),
216            "display must contain plan_id; got: {display}"
217        );
218        assert!(
219            display.contains("adapter does not support shape"),
220            "display must contain the reason text; got: {display}"
221        );
222        // Verify shape info is in the output too.
223        assert!(
224            display.contains("Custom"),
225            "display must include shape info; got: {display}"
226        );
227        // Also verify the capability's required_shape is Custom (sanity check on the fixture).
228        let shape = e.capability.required_shape();
229        assert!(
230            matches!(shape, CapabilityShape::Custom { .. }),
231            "shape must be Custom"
232        );
233    }
