1//! Declarative-requirements adapter resolver.
2//!
3//! [`resolve_adapter`] implements the Bazel-style filter pipeline from spec §3:
4//!
5//! 1. **Platform filter** — only consider adapters that apply to the current OS.
6//! 2. **Probe filter** — instantiate the adapter and call `probe()`. Adapters
7//!    returning [`tau_ports::CapabilityProbe::Unavailable`] are rejected.
8//! 3. **Tier filter** — the adapter's delivered tier must be ≥ the effective
9//!    required tier (max of project tier and plugin-floor tier).
10//! 4. **Shape filter** — every shape in `requirements.required_shapes` must be
11//!    in the adapter's `supported_shapes`.
12//! 5. **Per-plugin tier filter** — every plugin's `required_tier` must be ≤
13//!    the adapter's delivered tier.
14//! 6. **Priority sort** — among survivors, pick the one with the highest
15//!    [`AdapterRegistration::priority`].
16//!
17//! Mock injection via `TAU_TESTING_ALLOW_MOCK_SANDBOX=1` bypasses the registry
18//! entirely; it is an opt-in, never a silent fallback.
19

43/// A concrete sandbox adapter selected by the resolver.
44///
45/// Enum dispatch — the `Sandbox` trait uses `async fn` in trait (AFIT) which
46/// is not dyn-compatible, so we use an explicit enum rather than `Arc<dyn Sandbox>`.
47#[non_exhaustive]
48#[allow(clippy::large_enum_variant)]
49pub enum SandboxAdapter {
50    /// `tau-sandbox-native` Linux adapter.
51    Native(NativeSandbox),
52    /// `tau-sandbox-darwin` macOS adapter (sandbox-exec + tau-sandbox-proxy).
53    /// Selected for `RegistryKind::Native` on macOS hosts.
54    #[cfg(target_os = "macos")]
55    Darwin(DarwinSandbox),
56    /// `tau-sandbox-windows` Windows adapter (AppContainer + tau-sandbox-proxy).
57    /// Selected for `RegistryKind::Native` on Windows hosts.
58    #[cfg(target_os = "windows")]
59    Windows(WindowsSandbox),
60    /// `tau-sandbox-container` docker/podman adapter.
61    Container(ContainerSandbox),
62    /// `tau_ports::fixtures::MockCapabilityGate` — present ONLY under `cargo test`
63    /// or the opt-in `mock-sandbox` feature; absent from a default production
64    /// build. Instantiated solely via the `TAU_TESTING_ALLOW_MOCK_SANDBOX=1`
65    /// env-var injection path below, never as a silent fallback.
66    #[cfg(any(feature = "mock-sandbox", test))]
67    Mock(tau_ports::fixtures::MockCapabilityGate),
68    /// No isolation; explicit opt-out path.
69    Passthrough(PassthroughSandbox),
70}

302/// Public wrapper for `instantiate` — used by `tau sandbox status` to
303/// probe each registered adapter regardless of platform compatibility.
304pub fn instantiate_for_probe(kind: RegistryKind) -> Result<SandboxAdapter, String> {

308fn instantiate(kind: RegistryKind) -> Result<SandboxAdapter, String> {
309    // The catch-all arm exists for forward-compat: when a new RegistryKind
310    // variant is added before the match arms are updated, the compiler will
311    // continue to compile (since RegistryKind is #[non_exhaustive] externally).
312    // Within the same crate the pattern is currently unreachable, hence the allow.
313    #[allow(unreachable_patterns)]
314    match kind {
315        #[cfg(target_os = "macos")]
316        RegistryKind::Native => Ok(SandboxAdapter::Darwin(DarwinSandbox::new("native"))),
317        #[cfg(target_os = "windows")]
318        RegistryKind::Native => Ok(SandboxAdapter::Windows(WindowsSandbox::new("native"))),
319        #[cfg(not(any(target_os = "macos", target_os = "windows")))]
320        RegistryKind::Native => Ok(SandboxAdapter::Native(NativeSandbox::new(
321            "native",
322            CapabilityTier::Strict,
323        ))),
324        RegistryKind::Container => Ok(SandboxAdapter::Container(ContainerSandbox::new(
325            "container",
326            ContainerRuntime::Auto,
327        ))),
328        RegistryKind::Remote => {
329            Err("remote backend not implemented at v0.2 (Phase 2 sub-project F)".into())
330        }
331        RegistryKind::Passthrough => Ok(SandboxAdapter::Passthrough(PassthroughSandbox::new())),
332        // catch-all for #[non_exhaustive]
333        other => Err(format!("unknown adapter kind: {other:?}")),
334    }
335}

337// ---------------------------------------------------------------------------
338// resolve_adapter — the filter pipeline
339// ---------------------------------------------------------------------------
340
341/// Select the best sandbox adapter for the given project requirements and
342/// plugin list.
343///
344/// Implements the Bazel-style filter pipeline from spec §3:
345/// platform → probe → tier → shape → plugin-tier-floor → priority sort.
346///
347/// # Mock injection
348///
349/// When `TAU_TESTING_ALLOW_MOCK_SANDBOX=1` is set, the registry is bypassed
350/// entirely and `SandboxAdapter::Mock` is returned. This is the only way to
351/// get a Mock adapter; it is never a silent fallback.
352pub async fn resolve_adapter(
353    requirements: &tau_pkg::scope::SandboxRequirements,
354    plugins: &[tau_domain::PluginSandboxRequirements],
355) -> Result<SandboxAdapter, ResolutionError> {
356    use tau_pkg::scope::SandboxRequiredTier;
357
358    // Mock injection via env var — bypasses the registry entirely.
359    // Gated out of production builds so the env-var string never ships.
360    #[cfg(any(feature = "mock-sandbox", test))]
361    if std::env::var("TAU_TESTING_ALLOW_MOCK_SANDBOX")
362        .map(|v| v == "1")
363        .unwrap_or(false)
364    {
365        return Ok(SandboxAdapter::Mock(
366            tau_ports::fixtures::MockCapabilityGate::new("mock"),
367        ));
368    }
369
370    let platform = detect_platform();
371
372    // Map config-side tier types to tau_ports::CapabilityTier
373    let project_tier: CapabilityTier = match requirements.required_tier {
374        SandboxRequiredTier::None => CapabilityTier::None,
375        SandboxRequiredTier::Light => CapabilityTier::Light,
376        SandboxRequiredTier::Strict => CapabilityTier::Strict,
377        // catch-all for #[non_exhaustive] forward-compat
378        other => {
379            return Err(ResolutionError::ConfigError {
380                message: format!("unknown required_tier variant: {other:?}"),
381            })
382        }
383    };
384
385    // Plugin floor: highest required_tier across all plugins.
386    let plugin_floor: CapabilityTier = plugins
387        .iter()
388        .filter_map(|p| p.required_tier.as_ref())
389        .map(|t| match t {
390            tau_domain::PluginRequiredTier::None => CapabilityTier::None,
391            tau_domain::PluginRequiredTier::Light => CapabilityTier::Light,
392            tau_domain::PluginRequiredTier::Strict => CapabilityTier::Strict,
393            _ => CapabilityTier::None,
394        })
395        .max()
396        .unwrap_or(CapabilityTier::None);
397
398    let effective_required_tier = std::cmp::max(project_tier, plugin_floor);
399
400    let required_shapes_set = if requirements.required_shapes.is_empty() {
401        // Auto-derive: caller has not given explicit shapes. The resolver
402        // permits any shape combination; downstream Layer 3 still validates
403        // per-plugin against adapter.supported_shapes via `validate_plan`.
404        // For the resolver's filter, treat empty-required as "no shape filter"
405        // (every adapter passes the shape filter).
406        tau_domain::CapabilityShapeSet::new()
407    } else {
408        let mut s = tau_domain::CapabilityShapeSet::new();
409        for shape in &requirements.required_shapes {
410            s.insert(shape.clone());
411        }
412        s
413    };
414
415    let mut tried: Vec<(String, ResolutionRejection)> = Vec::new();
416    let mut candidates: Vec<&AdapterRegistration> = Vec::new();
417
418    for registration in REGISTRY.iter() {
419        let name = registration.kind.name().to_owned();
420
421        // Filter 1: platform match
422        if !registration.platforms.includes(platform) {
423            tried.push((name.clone(), ResolutionRejection::PlatformMismatch));
424            continue;
425        }
426
427        // Filter 2: probe (instantiate adapter, await probe)
428        let adapter = match instantiate(registration.kind) {
429            Ok(a) => a,
430            Err(msg) => {
431                tried.push((name.clone(), ResolutionRejection::ProbeUnavailable(msg)));
432                continue;
433            }
434        };
435        let delivered = match adapter.probe().await {
436            tau_ports::CapabilityProbe::Available { tier, .. } => tier,
437            tau_ports::CapabilityProbe::Unavailable { reason } => {
438                tried.push((name.clone(), ResolutionRejection::ProbeUnavailable(reason)));
439                continue;
440            }
441            other => {
442                tried.push((
443                    name.clone(),
444                    ResolutionRejection::ProbeUnavailable(format!("{other:?}")),
445                ));
446                continue;
447            }
448        };
449
450        // Filter 3: delivered tier >= effective required tier
451        if delivered < effective_required_tier {
452            tried.push((
453                name.clone(),
454                ResolutionRejection::TierTooLow {
455                    delivered,
456                    required: effective_required_tier,
457                },
458            ));
459            continue;
460        }
461
462        // Filter 4: required shapes ⊆ adapter's supported shapes
463        let supported_shapes = (registration.shapes_supported_fn)();
464        let mut missing = tau_domain::CapabilityShapeSet::new();
465        for shape in required_shapes_set.iter() {
466            if !supported_shapes.contains(shape) {
467                missing.insert(shape.clone());
468            }
469        }
470        if !missing.is_empty() {
471            tried.push((
472                name.clone(),
473                ResolutionRejection::ShapesUnsupported { missing },
474            ));
475            continue;
476        }
477
478        // Filter 5: every plugin's required_tier <= delivered_tier
479        let mut plugin_failure: Option<(String, tau_domain::PluginRequiredTier)> = None;
480        for (idx, plugin) in plugins.iter().enumerate() {
481            if let Some(p_tier) = &plugin.required_tier {
482                let p_tier_mapped = match p_tier {
483                    tau_domain::PluginRequiredTier::None => CapabilityTier::None,
484                    tau_domain::PluginRequiredTier::Light => CapabilityTier::Light,
485                    tau_domain::PluginRequiredTier::Strict => CapabilityTier::Strict,
486                    _ => CapabilityTier::None,
487                };
488                if p_tier_mapped > delivered {
489                    // We don't know plugin name in this context — use index
490                    // for diagnostic; callers wrap a richer diagnostic
491                    // (Task 6 wires plugin names into PluginSandboxRequirements
492                    // via context).
493                    plugin_failure = Some((format!("plugin[{idx}]"), *p_tier));
494                    break;
495                }
496            }
497        }
498        if let Some((plugin_id, required)) = plugin_failure {
499            tried.push((
500                name.clone(),
501                ResolutionRejection::PluginTierTooLow {
502                    plugin: plugin_id,
503                    required,
504                },
505            ));
506            continue;
507        }
508
509        candidates.push(registration);
510    }
511
512    // §3.5 — pick highest-priority match
513    candidates.sort_by_key(|r| Reverse(r.priority));
514    let chosen_reg =
515        candidates
516            .into_iter()
517            .next()
518            .ok_or_else(|| ResolutionError::NoAdapterMatches {
519                tried,
520                platform: platform.to_owned(),
521                required_tier: effective_required_tier,
522            })?;
523
524    instantiate(chosen_reg.kind).map_err(|m| ResolutionError::ConfigError { message: m })
525}

527// ---------------------------------------------------------------------------
528// resolve_strict_for_validation — picks highest-priority non-passthrough adapter
529// ---------------------------------------------------------------------------
530
531/// Walk the registry and pick the highest-priority non-passthrough adapter
532/// that probes `Available` on the current platform.
533///
534/// Used by `tau resolve --check-sandbox` when the project's `required_tier`
535/// is `None` (so the project's actual resolution would yield passthrough) —
536/// `--check-sandbox` should still validate against the strictest available
537/// "real" adapter so users see what would happen if they strengthened the
538/// requirement.
539///
540/// When `TAU_TESTING_ALLOW_MOCK_SANDBOX=1` is set, returns the mock adapter
541/// (same as `resolve_adapter`) so tests remain reliable across platforms.
542pub async fn resolve_strict_for_validation() -> Result<SandboxAdapter, ResolutionError> {
543    // Mock injection — mirrors resolve_adapter behaviour.
544    // Gated out of production builds so the env-var string never ships.
545    #[cfg(any(feature = "mock-sandbox", test))]
546    if std::env::var("TAU_TESTING_ALLOW_MOCK_SANDBOX")
547        .map(|v| v == "1")
548        .unwrap_or(false)
549    {
550        return Ok(SandboxAdapter::Mock(
551            tau_ports::fixtures::MockCapabilityGate::new("mock"),
552        ));
553    }
554
555    let platform = detect_platform();
556    let mut tried: Vec<(String, ResolutionRejection)> = Vec::new();
557
558    // Collect non-passthrough candidates in priority order (highest first).
559    let mut candidates: Vec<&AdapterRegistration> = REGISTRY
560        .iter()
561        .filter(|r| r.kind != RegistryKind::Passthrough)
562        .collect();
563    candidates.sort_by_key(|r| Reverse(r.priority));
564
565    for entry in candidates {
566        let name = entry.kind.name().to_owned();
567
568        if !entry.platforms.includes(platform) {
569            tried.push((name, ResolutionRejection::PlatformMismatch));
570            continue;
571        }
572
573        let adapter = match instantiate(entry.kind) {
574            Ok(a) => a,
575            Err(msg) => {
576                tried.push((name, ResolutionRejection::ProbeUnavailable(msg)));
577                continue;
578            }
579        };
580
581        match adapter.probe().await {
582            tau_ports::CapabilityProbe::Available { .. } => return Ok(adapter),
583            tau_ports::CapabilityProbe::Unavailable { reason } => {
584                tried.push((name, ResolutionRejection::ProbeUnavailable(reason)));
585            }
586            other => {
587                tried.push((
588                    name,
589                    ResolutionRejection::ProbeUnavailable(format!("{other:?}")),
590                ));
591            }
592        }
593    }
594
595    Err(ResolutionError::NoAdapterMatches {
596        tried,
597        platform: platform.to_owned(),
598        required_tier: CapabilityTier::None,
599    })
600}

602// ---------------------------------------------------------------------------
603// resolve_adapter_forced — forced single-adapter path for --sandbox <kind>
604// ---------------------------------------------------------------------------
605
606/// Force-instantiate a specific adapter kind, bypassing the registry filter.
607///
608/// Probes the named adapter; returns `Ok(adapter)` iff
609/// [`tau_ports::CapabilityProbe::Available`], otherwise
610/// [`ResolutionError::ConfigError`] with a guided message (e.g.
611/// `"--sandbox native is not applicable on macOS"`).
612pub async fn resolve_adapter_forced(kind: RegistryKind) -> Result<SandboxAdapter, ResolutionError> {
613    let adapter = instantiate(kind).map_err(|m| ResolutionError::ConfigError {
614        message: format!("--sandbox {kind:?}: {m}"),
615    })?;
616    match adapter.probe().await {
617        tau_ports::CapabilityProbe::Available { .. } => Ok(adapter),
618        tau_ports::CapabilityProbe::Unavailable { reason } => Err(ResolutionError::ConfigError {
619            message: format!("--sandbox {kind:?} not applicable: {reason}"),
620        }),
621        other => Err(ResolutionError::ConfigError {
622            message: format!("--sandbox {kind:?}: unexpected probe result: {other:?}"),
623        }),
624    }
625}
