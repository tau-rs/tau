1//! Span tree + time-axis math over `tau_ports::TraceEvent`.
2//!
3//! [`TraceModel`] is a pure, incremental fold: [`TraceModel::apply`] consumes
4//! one [`tau_ports::TraceEvent`] at a time and appends/updates [`Span`]s. It
5//! holds no clock, no I/O, no runtime dependency — callers (e.g. a `tau run`
6//! event subscriber, or a replay of a recorded trace file) drive it.
7
8use std::collections::HashMap;
9
10use chrono::{DateTime, Duration, Utc};
11use tau_ports::{AgentId, CapabilityVerdict, TraceEvent, TraceEventKind};
12
13/// What a [`Span`] represents in the execution tree.
14///
15/// `Reasoning`, `Branch`, `Parallel`, `Loop`, and `Suspend` are not produced
16/// by any [`TraceEventKind`] as of M1 — they exist so a later milestone can
17/// add new `apply` arms (e.g. control-flow spans) without breaking this
18/// enum's shape.
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20pub enum SpanKind {
21    /// One agent turn.
22    Agent,
23    /// One tool call.
24    Tool,
25    /// A model reasoning step (not yet produced; reserved for M2).
26    Reasoning,
27    /// A `Branch` control-flow node (not yet produced; reserved).
28    Branch,
29    /// A `Parallel` control-flow node (not yet produced; reserved).
30    Parallel,
31    /// A `Loop` control-flow node (not yet produced; reserved).
32    Loop,
33    /// A `Suspend` control-flow node (not yet produced; reserved).
34    Suspend,
35}
36
37/// Terminal (or in-flight) status of a [`Span`].
38#[derive(Debug, Clone, Copy, PartialEq, Eq)]
39pub enum SpanStatus {
40    /// Started, not yet observed to complete.
41    Running,
42    /// Completed successfully.
43    Ok,
44    /// Completed with an error/failure.
45    Failed,
46}

48/// One renderable node in the execution trace.
49#[derive(Debug, Clone, PartialEq)]
50pub struct Span {
51    /// Index into [`TraceModel::spans`]; stable for the life of the model.
52    pub id: usize,
53    /// What kind of unit of work this span represents.
54    pub kind: SpanKind,
55    /// Human-readable label (agent id, tool name, ...).
56    pub label: String,
57    /// Derived start time (`ts - duration`).
58    pub start: DateTime<Utc>,
59    /// Derived end time; `None` while still running.
60    pub end: Option<DateTime<Utc>>,
61    /// Token count, if this span carries one (turns only).
62    pub tokens: Option<u64>,
63    /// Governance verdict, if this span was capability-gated (tool calls only).
64    pub capability: Option<CapabilityVerdict>,
65    /// Parent span id, for tree rendering.
66    pub parent: Option<usize>,
67    /// Current status.
68    pub status: SpanStatus,
69}

71/// Incremental fold of `TraceEvent`s into a renderable span tree + time axis.
72#[derive(Debug, Default)]
73pub struct TraceModel {
74    spans: Vec<Span>,
75    /// Owning agent for each span, index-aligned with `spans`. `None` for
76    /// spans with no known owner (e.g. a `ToolCall` whose top-level
77    /// `TraceEvent.agent_id` was `None`). Used, together with
78    /// `agent_parent`, to (re)compute `Span::parent` in a full pass after
79    /// every `apply` — see `resolve_parents`. Real emission order (traced
80    /// against `tau-runtime-core`) has a `Spawn` arrive before the child's
81    /// own `Turn`s, and mid-turn `ToolCall`s arrive before the `Turn` event
82    /// that summarizes them, so parent resolution cannot be a one-shot
83    /// lookup at push time — it must be re-derivable from whatever has
84    /// arrived so far, regardless of order.
85    owners: Vec<Option<AgentId>>,
86    /// Turn index each span belongs to, index-aligned with `spans`. For an
87    /// `Agent` span this is the `Turn` event's `turn_index`; for a `Tool`
88    /// span it is the `ToolCall` event's `turn_index`. `None` for spans with
89    /// no turn association. Used by `resolve_parents` to nest a tool under
90    /// its *own* turn's agent span rather than the agent's first turn.
91    turns: Vec<Option<u32>>,
92    /// child agent id -> parent agent id, populated by `Spawn` events.
93    agent_parent: HashMap<AgentId, AgentId>,
94    /// Max `ts` seen across every applied event (not just span bounds) —
95    /// the upper bound of the time axis while spans are still running.
96    max_ts: Option<DateTime<Utc>>,
97    /// Run id, captured from the first event carrying a non-empty one.
98    /// Surfaced to the TUI toolbar; `None` until the first event arrives.
99    run_id: Option<String>,
100}
101
102impl TraceModel {
103    /// Empty model.
104    pub fn new() -> Self {
105        Self::default()
106    }
107
108    /// Fold one event into the model, appending or updating spans.
109    pub fn apply(&mut self, event: &TraceEvent) {
110        self.max_ts = Some(match self.max_ts {
111            Some(t) if t >= event.ts => t,
112            _ => event.ts,
113        });
114        if self.run_id.is_none() && !event.run_id.is_empty() {
115            self.run_id = Some(event.run_id.clone());
116        }
117
118        match &event.kind {
119            TraceEventKind::Turn {
120                agent_id,
121                turn_index,
122                duration_ms,
123                tokens,
124            } => {
125                let start = event.ts - Duration::milliseconds(*duration_ms as i64);
126                self.push_span(
127                    Span {
128                        id: 0,
129                        kind: SpanKind::Agent,
130                        label: agent_id.clone(),
131                        start,
132                        end: Some(event.ts),
133                        tokens: Some(*tokens),
134                        capability: None,
135                        parent: None, // filled in by resolve_parents below
136                        status: SpanStatus::Ok,
137                    },
138                    Some(agent_id.clone()),
139                    Some(*turn_index),
140                );
141            }
142            TraceEventKind::ToolCall {
143                tool_name,
144                duration_ms,
145                status,
146                capability,
147                turn_index,
148            } => {
149                let start = event.ts - Duration::milliseconds(*duration_ms as i64);
150                self.push_span(
151                    Span {
152                        id: 0,
153                        kind: SpanKind::Tool,
154                        label: tool_name.clone(),
155                        start,
156                        end: Some(event.ts),
157                        tokens: None,
158                        capability: capability.clone(),
159                        parent: None, // filled in by resolve_parents below
160                        status: status_from_str(status),
161                    },
162                    event.agent_id.clone(),
163                    Some(*turn_index),
164                );
165            }
166            TraceEventKind::Spawn { child_id, .. } => {
167                if let Some(parent_id) = &event.agent_id {
168                    self.agent_parent
169                        .insert(child_id.clone(), parent_id.clone());
170                }
171            }
172            TraceEventKind::Completion { agent_id, status } => {
173                if let Some(&idx) = self.agent_anchors().get(agent_id.as_str()) {
174                    self.spans[idx].status = status_from_str(status);
175                }
176            }
177            // Abort/budget/orphan/task-mutation/plan-note: no bar in M1.
178            // Forward-compat catch-all for any future TraceEventKind variant.
179            TraceEventKind::Abort { .. }
180            | TraceEventKind::TaskMutation { .. }
181            | TraceEventKind::PlanNote { .. }
182            | TraceEventKind::BudgetWarn { .. }
183            | TraceEventKind::BudgetExceeded { .. }
184            | TraceEventKind::OrphanedTasksAtTermination { .. } => {}
185        }
186
187        // Re-derive every span's parent from the full current state rather
188        // than pinning it once at push time. Real trace order can deliver a
189        // `Spawn` before the parent's own `Turn`, or a `ToolCall` before the
190        // `Turn` that summarizes it — a one-shot lookup at push time leaves
191        // those permanently un-parented. O(n) per event is fine at
192        // TUI-sized trace volumes.
193        self.resolve_parents();
194    }
195
196    /// All spans in emission order; index == [`Span::id`].
197    pub fn spans(&self) -> &[Span] {
198        &self.spans
199    }
200
201    /// Run id captured from the first applied event, if any.
202    pub fn run_id(&self) -> Option<&str> {
203        self.run_id.as_deref()
204    }
205
206    /// The time axis' bounds: `(min start, max end)`.
207    ///
208    /// A span still running (`end.is_none()`) contributes the max `ts` seen
209    /// across every applied event as its provisional end, so the axis keeps
210    /// extending while work is in flight. Once every span has completed, the
211    /// axis pins to the latest span *end* — a trailing no-op event (a
212    /// `Completion`/`Spawn` after the last bar) no longer stretches the
213    /// window with dead space to its right.
214    pub fn window(&self) -> Option<(DateTime<Utc>, DateTime<Utc>)> {
215        if self.spans.is_empty() {
216            return None;
217        }
218        let lo = self.spans.iter().map(|s| s.start).min()?;
219        let any_running = self.spans.iter().any(|s| s.end.is_none());
220        let max_end = self.spans.iter().filter_map(|s| s.end).max();
221        let hi = if any_running {
222            // A running span extends the axis to the latest ts observed.
223            let fallback_hi = self.max_ts.unwrap_or(lo);
224            max_end.map_or(fallback_hi, |e| e.max(fallback_hi))
225        } else {
226            // All spans complete: ignore trailing no-op events' ts.
227            max_end.unwrap_or(lo)
228        };
229        Some((lo, hi))
230    }
231
232    /// Map `span`'s `[start, end]` onto `[0, width_cols]` given the current
233    /// [`TraceModel::window`]. Returns `(offset_cols, len_cols)`, both
234    /// clamped to `[0, width_cols]`. A zero-length span renders `len == 1`.
235    pub fn bar(&self, span: &Span, width_cols: u16) -> (u16, u16) {
236        let Some((lo, hi)) = self.window() else {
237            return (0, 0);
238        };
239        let width = i64::from(width_cols);
240        let total_ms = (hi - lo).num_milliseconds().max(1);
241
242        let to_col = |t: DateTime<Utc>| -> i64 {
243            let ms = (t - lo).num_milliseconds();
244            (ms * width / total_ms).clamp(0, width)
245        };
246
247        let offset = to_col(span.start);
248        let end_off = to_col(span.end.unwrap_or(hi)).max(offset);
249        let mut len = (end_off - offset).max(1);
250        if offset + len > width {
251            len = (width - offset).max(1);
252        }
253        (offset as u16, len as u16)
254    }
255
256    /// Each agent's representative span — the earliest `Agent`-kind span
257    /// owned by that agent id — used as the nesting anchor both for that
258    /// agent's own `Spawn`ed children and for its `Tool` spans. Recomputed
259    /// on demand (not cached) so it always reflects whatever spans exist
260    /// so far, independent of arrival order.
261    fn agent_anchors(&self) -> HashMap<&str, usize> {
262        let mut anchors: HashMap<&str, usize> = HashMap::new();
263        for (i, owner) in self.owners.iter().enumerate() {
264            if self.spans[i].kind != SpanKind::Agent {
265                continue;
266            }
267            if let Some(owner) = owner {
268                anchors.entry(owner.as_str()).or_insert(i);
269            }
270        }
271        anchors
272    }
273
274    /// Each `(agent id, turn index)` pair's `Agent` span — the earliest
275    /// `Agent`-kind span owned by that agent that carries that turn index.
276    /// A tool call nests under the anchor for its *own* turn so a run with
277    /// several turns per agent renders each turn's tools under the right
278    /// turn bar, not all under turn 0.
279    fn turn_anchors(&self) -> HashMap<(&str, u32), usize> {
280        let mut anchors: HashMap<(&str, u32), usize> = HashMap::new();
281        for (i, owner) in self.owners.iter().enumerate() {
282            if self.spans[i].kind != SpanKind::Agent {
283                continue;
284            }
285            if let (Some(owner), Some(turn)) = (owner, self.turns[i]) {
286                anchors.entry((owner.as_str(), turn)).or_insert(i);
287            }
288        }
289        anchors
290    }
291
292    /// Recompute `Span::parent` for every span from the current
293    /// `owners`/`turns`/`agent_parent`/anchor state. A `Tool` span's parent
294    /// is the `Agent` span for its own `(owner, turn)` — falling back to the
295    /// owner's first turn if that turn's span has not arrived yet (mid-turn
296    /// tool calls precede the `Turn` event that summarizes them). An `Agent`
297    /// span's parent is the anchor span of whichever agent spawned it (per
298    /// `agent_parent`), if known.
299    fn resolve_parents(&mut self) {
300        let anchors = self.agent_anchors();
301        let turn_anchors = self.turn_anchors();
302        let mut parents = vec![None; self.spans.len()];
303        for (i, owner) in self.owners.iter().enumerate() {
304            let Some(owner) = owner else { continue };
305            let parent = match self.spans[i].kind {
306                SpanKind::Tool => self.turns[i]
307                    .and_then(|turn| turn_anchors.get(&(owner.as_str(), turn)).copied())
308                    .or_else(|| anchors.get(owner.as_str()).copied()),
309                SpanKind::Agent => self
310                    .agent_parent
311                    .get(owner)
312                    .and_then(|p| anchors.get(p.as_str()).copied()),
313                _ => None,
314            };
315            // Guard against a self-loop (e.g. malformed/self-spawned data).
316            parents[i] = parent.filter(|&p| p != i);
317        }
318        for (span, parent) in self.spans.iter_mut().zip(parents) {
319            span.parent = parent;
320        }
321    }
322
323    fn push_span(&mut self, mut span: Span, owner: Option<AgentId>, turn: Option<u32>) -> usize {
324        let idx = self.spans.len();
325        span.id = idx;
326        self.spans.push(span);
327        self.owners.push(owner);
328        self.turns.push(turn);
329        idx
330    }
331}

333fn status_from_str(status: &str) -> SpanStatus {
334    match status {
335        "ok" | "completed" => SpanStatus::Ok,
336        _ => SpanStatus::Failed,
337    }
