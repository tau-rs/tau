1//! Malformed-tolerant jsonl line ingestion.
2//!
3//! [`parse_line`] turns one line of a `.tau/runs/<id>.jsonl` file into a
4//! [`tau_ports::TraceEvent`], accepting both the writer's
5//! `{"line_kind":"trace_event","event":{…}}` envelope and a bare event
6//! (spec §13.5). Only a blank/whitespace-only line returns
7//! `Ok(None)` — that is the one case a live tail (e.g. the TUI reading a
8//! file the engine is still writing to) can expect to see routinely (a
9//! trailing newline, or a read that lands exactly on EOF). Any non-empty
10//! line that fails to parse — including a half-written/truncated line read
11//! mid-write — returns `Err`, it is not silently swallowed. A live-tail
12//! caller is expected to treat an `Err` on the last/still-growing line as
13//! retryable (re-read on the next fsync boundary) and only skip/surface an
14//! `Err` once the line is no longer growing (i.e. a later read confirms it
15//! wasn't a transient partial write).
16
17use tau_ports::TraceEvent;
18
19/// Error parsing one jsonl line into a [`TraceEvent`].
20///
21/// Reserved for genuinely malformed *complete* lines — never returned for
22/// blank/whitespace-only lines (see [`parse_line`]).
23#[derive(thiserror::Error, Debug)]
24pub enum IngestError {
25    /// The line was non-empty but not a valid JSON-encoded `TraceEvent`.
26    #[error("bad trace line: {0}")]
27    Json(String),
28}
29
30/// Mirrors the tag shape of `tau_runtime_tokio::orchestration::persistence::RunLogLine`
31/// (`{"line_kind": "...", ...}`), read manually via `serde_json::Value`.
32///
33/// `tau-trace` is pure and headless — it must not depend on the tokio host
34/// crate — so the wire shape is mirrored here instead of imported. It is
35/// read via `serde_json::Value` rather than a `#[derive(Deserialize)]`
36/// mirror enum because `tau-trace` has no direct dependency on the `serde`
37/// crate (only `serde_json`, which does not re-export `serde`'s derive
38/// macros), and adding one is a dependency change outside this fix's scope.
39/// The `envelope_shape_matches_the_writer` test guards against drift: if
40/// the writer's tag name (`line_kind`), tag value (`trace_event`), or
41/// payload field (`event`) ever changes, that test fails rather than the
42/// reader silently rendering an empty waterfall (spec §13.5).
43const ENVELOPE_TAG_FIELD: &str = "line_kind";
44const ENVELOPE_TRACE_EVENT_TAG: &str = "trace_event";
45const ENVELOPE_PAYLOAD_FIELD: &str = "event";
46
47/// Parse one `.tau/runs/<id>.jsonl` line into a [`TraceEvent`].
48///
49/// Accepts both line shapes:
50/// - the **envelope** the run-log writer produces,
51///   `{"line_kind":"trace_event","event":{…}}` — non-`trace_event` kinds
52///   (e.g. `task_mutation`) yield `Ok(None)`;
53/// - a **bare** `TraceEvent`, for older logs and test fixtures.
54///
55/// Returns `Ok(None)` for a blank/whitespace-only line (a trailing newline,
56/// or a live tail reading past the last complete line) and for envelope
57/// lines that carry no trace event. Returns `Err` for any other non-empty
58/// line that fails to deserialize — including a half-written/truncated line
59/// read mid-write, which is *not* treated as `Ok(None)`. A live-tail caller
60/// should treat such an `Err` on the still-growing tail line as retryable.
61pub fn parse_line(line: &str) -> Result<Option<TraceEvent>, IngestError> {
62    let trimmed = line.trim();
63    if trimmed.is_empty() {
64        return Ok(None);
65    }
66    // Try the envelope first: it is what the writer emits today. A
67    // truncated/malformed line fails JSON parsing outright here (it never
68    // becomes a `Value::Object`), so it falls through to the bare-event
69    // attempt below and surfaces that `Err` — it never spuriously matches
70    // the envelope shape.
71    if let Ok(serde_json::Value::Object(mut obj)) =
72        serde_json::from_str::<serde_json::Value>(trimmed)
73    {
74        if let Some(serde_json::Value::String(tag)) = obj.get(ENVELOPE_TAG_FIELD) {
75            return if tag == ENVELOPE_TRACE_EVENT_TAG {
76                let payload = obj.remove(ENVELOPE_PAYLOAD_FIELD).ok_or_else(|| {
77                    IngestError::Json(format!(
78                        "envelope line missing '{ENVELOPE_PAYLOAD_FIELD}' field"
79                    ))
80                })?;
81                serde_json::from_value::<TraceEvent>(payload)
82                    .map(Some)
83                    .map_err(|e| IngestError::Json(e.to_string()))
84            } else {
85                Ok(None)
86            };
87        }
88    }
89    serde_json::from_str::<TraceEvent>(trimmed)
90        .map(Some)
91        .map_err(|e| IngestError::Json(e.to_string()))
92}

100    #[test]
101    fn parses_a_turn_line() {
102        // Round-trip via the real Serialize impl, per Step 0 — this is
103        // immune to field-name drift because we never hand-author the JSON.
104        let evt = TraceEvent {
105            id: "evt-1".into(),
106            ts: Utc.timestamp_opt(1_700_000_000, 0).unwrap(),
107            run_id: "run-1".into(),
108            agent_id: Some("agent-1".into()),
109            kind: TraceEventKind::Turn {
110                agent_id: "agent-1".into(),
111                turn_index: 3,
112                duration_ms: 250,
113                tokens: 42,
114            },
115        };
116        let line = serde_json::to_string(&evt).unwrap();
117
118        let parsed = parse_line(&line).unwrap().unwrap();
119
120        assert_eq!(parsed, evt);
121        assert!(matches!(parsed.kind, TraceEventKind::Turn { .. }));
122    }

161    #[test]
162    fn truncated_mid_write_line_is_err_not_panic() {
163        // Simulates a live tail reading a partially-flushed line: valid
164        // JSON prefix, but cut off before the closing brace.
165        let evt = TraceEvent {
166            id: "evt-3".into(),
167            ts: Utc.timestamp_opt(1_700_000_200, 0).unwrap(),
168            run_id: "run-1".into(),
169            agent_id: None,
170            kind: TraceEventKind::Abort {
171                reason: "watchdog".into(),
172            },
173        };
174        let full_line = serde_json::to_string(&evt).unwrap();
175        // Byte-index slicing is only safe here because every field in
176        // `evt` is ASCII; a non-ASCII value would need a char-boundary-safe
177        // split to avoid panicking on a multi-byte UTF-8 codepoint.
178        let truncated = &full_line[..full_line.len() / 2];
179
180        assert!(parse_line(truncated).is_err());
181    }

245    #[test]
246    fn envelope_shape_matches_the_writer() {
247        // Drift guard (spec §13.5): this literal is the exact shape
248        // `tau_runtime_tokio::orchestration::persistence::spawn_writer`
249        // serializes via `RunLogLine::TraceEvent`. If the writer's tag name
250        // (`line_kind`), tag value (`trace_event`), or payload field
251        // (`event`) ever changes, this test fails loudly instead of
252        // `tau trace` silently rendering an empty waterfall.
253        let evt = TraceEvent {
254            id: "evt-11".into(),
255            ts: Utc.timestamp_opt(1_700_000_500, 0).unwrap(),
256            run_id: "run-11".into(),
257            agent_id: None,
258            kind: TraceEventKind::Turn {
259                agent_id: "a".into(),
260                turn_index: 0,
261                duration_ms: 1,
262                tokens: 0,
263            },
264        };
265        let mut line = serde_json::Map::new();
266        line.insert(
267            "line_kind".into(),
268            serde_json::Value::String("trace_event".into()),
269        );
270        line.insert("event".into(), serde_json::to_value(&evt).unwrap());
271        let encoded = serde_json::Value::Object(line).to_string();
272
273        assert_eq!(parse_line(&encoded).unwrap().unwrap(), evt);
274    }
