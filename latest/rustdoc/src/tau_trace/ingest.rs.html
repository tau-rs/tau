1//! Malformed-tolerant jsonl line ingestion.
2//!
3//! [`parse_line`] turns one line of a `.tau/runs/<id>.jsonl` file into a
4//! [`tau_ports::TraceEvent`]. Only a blank/whitespace-only line returns
5//! `Ok(None)` — that is the one case a live tail (e.g. the TUI reading a
6//! file the engine is still writing to) can expect to see routinely (a
7//! trailing newline, or a read that lands exactly on EOF). Any non-empty
8//! line that fails to parse — including a half-written/truncated line read
9//! mid-write — returns `Err`, it is not silently swallowed. A live-tail
10//! caller is expected to treat an `Err` on the last/still-growing line as
11//! retryable (re-read on the next fsync boundary) and only skip/surface an
12//! `Err` once the line is no longer growing (i.e. a later read confirms it
13//! wasn't a transient partial write).
14
15use tau_ports::TraceEvent;
16
17/// Error parsing one jsonl line into a [`TraceEvent`].
18///
19/// Reserved for genuinely malformed *complete* lines — never returned for
20/// blank/whitespace-only lines (see [`parse_line`]).
21#[derive(thiserror::Error, Debug)]
22pub enum IngestError {
23    /// The line was non-empty but not a valid JSON-encoded `TraceEvent`.
24    #[error("bad trace line: {0}")]
25    Json(String),
26}
27
28/// Parse one `.tau/runs/<id>.jsonl` line into a [`TraceEvent`].
29///
30/// Returns `Ok(None)` only for a blank/whitespace-only line (e.g. a
31/// trailing newline, or a live tail reading past the last complete line).
32/// Returns `Err` for any non-empty line that fails to deserialize as a
33/// `TraceEvent` — this includes a half-written/truncated line read
34/// mid-write, which is *not* treated as `Ok(None)`. A live-tail caller
35/// should treat such an `Err` on the still-growing tail line as retryable.
36pub fn parse_line(line: &str) -> Result<Option<TraceEvent>, IngestError> {
37    let trimmed = line.trim();
38    if trimmed.is_empty() {
39        return Ok(None);
40    }
41    serde_json::from_str::<TraceEvent>(trimmed)
42        .map(Some)
43        .map_err(|e| IngestError::Json(e.to_string()))
44}

46#[cfg(test)]
47mod tests {
48    use super::*;
49    use chrono::{TimeZone, Utc};
50    use tau_ports::TraceEventKind;
51
52    #[test]
53    fn parses_a_turn_line() {
54        // Round-trip via the real Serialize impl, per Step 0 — this is
55        // immune to field-name drift because we never hand-author the JSON.
56        let evt = TraceEvent {
57            id: "evt-1".into(),
58            ts: Utc.timestamp_opt(1_700_000_000, 0).unwrap(),
59            run_id: "run-1".into(),
60            agent_id: Some("agent-1".into()),
61            kind: TraceEventKind::Turn {
62                agent_id: "agent-1".into(),
63                turn_index: 3,
64                duration_ms: 250,
65                tokens: 42,
66            },
67        };
68        let line = serde_json::to_string(&evt).unwrap();
69
70        let parsed = parse_line(&line).unwrap().unwrap();
71
72        assert_eq!(parsed, evt);
73        assert!(matches!(parsed.kind, TraceEventKind::Turn { .. }));
74    }
75
76    #[test]
77    fn parses_a_tool_call_line() {
78        let evt = TraceEvent {
79            id: "evt-2".into(),
80            ts: Utc.timestamp_opt(1_700_000_100, 0).unwrap(),
81            run_id: "run-1".into(),
82            agent_id: Some("agent-1".into()),
83            kind: TraceEventKind::ToolCall {
84                tool_name: "net.http".into(),
85                duration_ms: 12,
86                status: "ok".into(),
87                capability: None,
88                turn_index: 0,
89            },
90        };
91        let line = serde_json::to_string(&evt).unwrap();
92
93        let parsed = parse_line(&line).unwrap().unwrap();
94
95        assert_eq!(parsed, evt);
96        assert!(matches!(parsed.kind, TraceEventKind::ToolCall { .. }));
97    }
98
99    #[test]
100    fn blank_line_is_none() {
101        assert!(parse_line("").unwrap().is_none());
102        assert!(parse_line("   ").unwrap().is_none());
103        assert!(parse_line("\n").unwrap().is_none());
104        assert!(parse_line("  \t  ").unwrap().is_none());
105    }
106
107    #[test]
108    fn garbage_is_err_not_panic() {
109        let err = parse_line("{not json").unwrap_err();
110        assert!(matches!(err, IngestError::Json(_)));
111    }
112
113    #[test]
114    fn truncated_mid_write_line_is_err_not_panic() {
115        // Simulates a live tail reading a partially-flushed line: valid
116        // JSON prefix, but cut off before the closing brace.
117        let evt = TraceEvent {
118            id: "evt-3".into(),
119            ts: Utc.timestamp_opt(1_700_000_200, 0).unwrap(),
120            run_id: "run-1".into(),
121            agent_id: None,
122            kind: TraceEventKind::Abort {
123                reason: "watchdog".into(),
124            },
125        };
126        let full_line = serde_json::to_string(&evt).unwrap();
127        // Byte-index slicing is only safe here because every field in
128        // `evt` is ASCII; a non-ASCII value would need a char-boundary-safe
129        // split to avoid panicking on a multi-byte UTF-8 codepoint.
130        let truncated = &full_line[..full_line.len() / 2];
131
132        assert!(parse_line(truncated).is_err());
133    }
