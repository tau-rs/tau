1//! AppContainer profile generation from a [`CapabilityPlan`].
2//!
3//! Pure functions — no I/O, no Win32 calls. Tested on any platform.
4
5use tau_domain::{Capability, FsCapability, NetCapability};
6use tau_ports::CapabilityPlan;
7
8/// Result of [`build_appcontainer_caps`]: the AppContainer-shape inputs that
9/// the spawn layer turns into Win32 calls.
10#[derive(Debug, Clone, PartialEq, Eq)]
11#[non_exhaustive]
12pub struct AppContainerCaps {
13    /// Filesystem paths the plugin needs to read. Spawn layer adds an ACL
14    /// grant on each (the AppContainer SID + GENERIC_READ).
15    pub fs_read_paths: Vec<String>,
16    /// Filesystem paths the plugin needs to write. Spawn layer adds an ACL
17    /// grant on each (the AppContainer SID + GENERIC_READ + GENERIC_WRITE).
18    pub fs_write_paths: Vec<String>,
19    /// Whether the plan requests outbound HTTP. When true, the spawn layer
20    /// spawns a per-container named-pipe proxy and routes the command
21    /// through `tau-net-bridge-win` (`launcher -- bridge --pipe <name> --
22    /// <orig> <args...>`); the bridge dials the pipe and relays the
23    /// plugin's traffic to the host-side `tau_sandbox_proxy` allowlist
24    /// enforcement. **No capability SIDs are added** (spike #626:
25    /// same-package loopback and SID-ACL'd pipes need none, and
26    /// `internetClient` would allow bypassing the allowlist).
27    pub has_http: bool,
28    /// Whether the plan grants process-spawn. AppContainer children inherit
29    /// the same security context, so this doesn't widen the sandbox; we
30    /// just don't restrict child spawn explicitly.
31    pub has_process_spawn: bool,
32}
33
34/// Translate a `CapabilityPlan` into AppContainer-shape inputs for the spawn
35/// layer. Pure; no Win32, no I/O.
36pub fn build_appcontainer_caps(plan: &CapabilityPlan) -> AppContainerCaps {
37    let mut fs_read_paths = Vec::new();
38    let mut fs_write_paths = Vec::new();
39    let mut has_http = false;
40    let mut has_process_spawn = false;
41
42    for cap in &plan.capabilities {
43        match cap {
44            Capability::Filesystem(FsCapability::Read { paths, .. }) => {
45                for p in paths {
46                    fs_read_paths.push(clean_glob_suffix(p));
47                }
48            }
49            Capability::Filesystem(FsCapability::Write { paths, .. }) => {
50                for p in paths {
51                    fs_write_paths.push(clean_glob_suffix(p));
52                }
53            }
54            Capability::Network(NetCapability::Http { .. }) => {
55                has_http = true;
56            }
57            Capability::Process(_) => {
58                has_process_spawn = true;
59            }
60            _ => {}
61        }
62    }
63
64    AppContainerCaps {
65        fs_read_paths,
66        fs_write_paths,
67        has_http,
68        has_process_spawn,
69    }
70}

72/// The write paths of `plan` whose *raw* capability spelling was a
73/// directory subtree (`<dir>/**`, `<dir>/*`, or a trailing `/`),
74/// normalised exactly like [`build_appcontainer_caps`] normalises them.
75///
76/// The Windows spawn layer creates these directories when they are
77/// missing, before granting the AppContainer SID write access on them.
78/// Two facts make that necessary rather than optional:
79///
80/// - `SetNamedSecurityInfoW` fails with `ERROR_FILE_NOT_FOUND`
81///   (`WIN32_ERROR(2)`) on a path that does not exist, so the grant —
82///   and with it the whole `wrap_spawn` — fails closed. `tau-pkg`'s
83///   build envelope grants write on `<package>/target/**`, which cargo
84///   has not created yet at wrap time; #622's CI round 1 failed exactly
85///   there.
86/// - The container cannot create the directory itself: the *parent* is
87///   not write-granted, by design.
88///
89/// Only glob-shaped entries qualify. A bare path (`C:\out\report.txt`)
90/// may well name a file the plan wants created, and silently turning it
91/// into a directory would be worse than the current loud failure — so
92/// those still fail closed.
93pub fn dir_shaped_write_paths(plan: &CapabilityPlan) -> Vec<String> {
94    let mut out = Vec::new();
95    for cap in &plan.capabilities {
96        if let Capability::Filesystem(FsCapability::Write { paths, .. }) = cap {
97            for p in paths {
98                if p.ends_with("/**") || p.ends_with("/*") || p.ends_with('/') {
99                    out.push(clean_glob_suffix(p));
100                }
101            }
102        }
103    }
104    out
105}
106
107/// Strip trailing glob suffixes from a path. AppContainer ACLs are
108/// per-directory + inherited; `/srv/data/**` and `/srv/data` both grant
109/// the same scope so we normalise to the parent path.
110fn clean_glob_suffix(p: &str) -> String {
111    p.trim_end_matches("/**")
112        .trim_end_matches("/*")
113        .trim_end_matches('/')
114        .to_string()
115}

117#[cfg(test)]
118mod tests {
119    use super::*;
120    use serde_json::json;
121
122    fn plan_from(capabilities: serde_json::Value) -> CapabilityPlan {
123        let plan_json = json!({
124            "capabilities": capabilities,
125            "context": null,
126            "limits": null,
127        });
128        serde_json::from_value(plan_json).expect("decode plan")
129    }
130
131    #[test]
132    fn empty_plan_emits_empty_caps() {
133        let plan = plan_from(json!([]));
134        let caps = build_appcontainer_caps(&plan);
135        assert!(caps.fs_read_paths.is_empty());
136        assert!(caps.fs_write_paths.is_empty());
137        assert!(!caps.has_http);
138        assert!(!caps.has_process_spawn);
139    }
140
141    #[test]
142    fn fs_read_paths_collected() {
143        let plan = plan_from(json!([
144            { "kind": "fs.read", "paths": ["/etc/foo", "/data/cache"] }
145        ]));
146        let caps = build_appcontainer_caps(&plan);
147        assert_eq!(
148            caps.fs_read_paths,
149            vec!["/etc/foo".to_string(), "/data/cache".to_string()]
150        );
151        assert!(caps.fs_write_paths.is_empty());
152    }
153
154    #[test]
155    fn fs_write_separate_from_read() {
156        let plan = plan_from(json!([
157            { "kind": "fs.read",  "paths": ["/etc/cfg"] },
158            { "kind": "fs.write", "paths": ["/data/scratch"] }
159        ]));
160        let caps = build_appcontainer_caps(&plan);
161        assert_eq!(caps.fs_read_paths, vec!["/etc/cfg".to_string()]);
162        assert_eq!(caps.fs_write_paths, vec!["/data/scratch".to_string()]);
163    }
164
165    #[test]
166    fn glob_suffix_stripped() {
167        let plan = plan_from(json!([
168            { "kind": "fs.read", "paths": ["/srv/data/**", "/etc/*", "/tmp/"] }
169        ]));
170        let caps = build_appcontainer_caps(&plan);
171        assert_eq!(
172            caps.fs_read_paths,
173            vec![
174                "/srv/data".to_string(),
175                "/etc".to_string(),
176                "/tmp".to_string()
177            ]
178        );
179    }
180
181    #[test]
182    fn dir_shaped_write_paths_only_matches_globs() {
183        let plan = plan_from(json!([
184            { "kind": "fs.write", "paths": [
185                "C:\\pkg\\target/**", "/var/cache/*", "/srv/out/", "C:\\out\\report.txt"
186            ] },
187            // Read globs must NOT be included: only write grants are
188            // allowed to materialise a directory.
189            { "kind": "fs.read", "paths": ["/etc/cfg/**"] }
190        ]));
191        assert_eq!(
192            dir_shaped_write_paths(&plan),
193            vec![
194                "C:\\pkg\\target".to_string(),
195                "/var/cache".to_string(),
196                "/srv/out".to_string(),
197            ]
198        );
199    }
200
201    #[test]
202    fn dir_shaped_write_paths_normalises_like_build_caps() {
203        // Whatever this returns must be comparable against
204        // `AppContainerCaps::fs_write_paths` entry-for-entry — the spawn
205        // layer looks paths up in a set built from it.
206        let plan = plan_from(json!([
207            { "kind": "fs.write", "paths": ["/a/b/**", "/c/d"] }
208        ]));
209        let caps = build_appcontainer_caps(&plan);
210        let dirs = dir_shaped_write_paths(&plan);
211        assert_eq!(
212            caps.fs_write_paths,
213            vec!["/a/b".to_string(), "/c/d".to_string()]
214        );
215        assert_eq!(dirs, vec!["/a/b".to_string()]);
216        assert!(caps.fs_write_paths.contains(&dirs[0]));
217    }
