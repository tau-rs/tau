1//! Windows AppContainer sandbox adapter for tau.
2//!
3//! Wraps plugin commands with Windows AppContainer (Windows 8+) so the
4//! plugin runs inside a kernel-level isolated container. Strict tier:
5//! filesystem isolation per-capability via per-AppContainer-SID ACL
6//! grants + outbound network restricted to a per-container named pipe
7//! (SID-ACL'd, no capability SIDs granted) fronting the host-side
8//! `tau-sandbox-proxy` allowlist logic; the in-container
9//! `tau-net-bridge-win` relays the plugin's traffic over the pipe.
10//!
11//! Compared to [`tau_sandbox_native`] (Linux landlock + seccomp + namespaces)
12//! and [`tau_sandbox_darwin`] (macOS sandbox-exec):
13//! - **Pros:** native Windows kernel sandboxing; same security envelope
14//!   as Linux/macOS strict from the plugin's perspective; reuses
15//!   `tau-sandbox-proxy` for HTTPS allowlist enforcement.
16//! - **Cons:** AppContainer programming is verbose (Win32 API);
17//!   development requires a Windows host (not testable on macOS dev);
18//!   no per-syscall filtering (no Windows equivalent of seccomp).
19//!
20//! ## Development constraint
21//!
22//! This crate cannot be exercised on macOS or Linux. The pure-logic
23//! parts (`profile`) compile and unit-test on any platform; the Win32
24//! parts (`acl`, the launcher command rebuild in `lib`) are
25//! `cfg(target_os = "windows")`-gated. Windows CI runners are the only
26//! place runtime behavior is verified.
27

83    /// Spawn a pipe proxy DACL'd to `profile` and return the bare pipe
84    /// name plus an opaque keep-alive guard (drop aborts the accept
85    /// loop). For the foreign-container pipe-access control test, which
86    /// targets the pipe DACL directly rather than going through
87    /// `wrap_spawn`.
88    pub fn spawn_pipe_proxy(
89        profile: &str,
90        hosts: tau_sandbox_proxy::HostAllow,
91    ) -> std::io::Result<(String, Box<dyn std::any::Any + Send>)> {
92        let sid = crate::acl::AppContainerSid {
93            profile_name: profile.to_string(),
94        };
95        let h = crate::pipe_proxy::spawn_pipe_proxy(hosts, &sid)?;
96        Ok((h.pipe_name().to_string(), Box::new(h)))
97    }

111/// Windows AppContainer adapter.
112pub struct WindowsSandbox {
113    name: String,
114    /// Probe is cached lazily on the first call.
115    probe_cache: Arc<OnceCell<CapabilityProbe>>,
116}
117
118impl WindowsSandbox {
119    /// Construct a Windows adapter.
120    pub fn new(name: impl Into<String>) -> Self {
121        Self {
122            name: name.into(),
123            probe_cache: Arc::new(OnceCell::new()),
124        }
125    }
126}

128impl CapabilityGate for WindowsSandbox {
129    fn name(&self) -> &str {
130        &self.name
131    }
132
133    async fn probe(&self) -> CapabilityProbe {
134        self.probe_cache
135            .get_or_init(|| async { run_probe().await })
136            .await
137            .clone()
138    }
139
140    fn supported_shapes(&self) -> CapabilityShapeSet {
141        let mut set = CapabilityShapeSet::new();
142        set.insert(tau_domain::CapabilityShape::FilesystemRead);
143        set.insert(tau_domain::CapabilityShape::FilesystemWrite);
144        set.insert(tau_domain::CapabilityShape::ProcessExec);
145        set.insert(tau_domain::CapabilityShape::NetworkHttp);
146        set
147    }
148
149    fn validate_plan(&self, plan: &CapabilityPlan) -> Result<(), CapabilityError> {
150        let supported = self.supported_shapes();
151        for cap in &plan.capabilities {
152            let shape = cap.required_shape();
153            if !supported.contains(&shape) {
154                return Err(CapabilityError::ShapeUnsupported { shape });
155            }
156        }
157        let mut exact: Vec<String> = Vec::new();
158        for cap in &plan.capabilities {
159            if let Capability::Network(NetCapability::Http { hosts, .. }) = cap {
160                if !hosts.is_any() {
161                    exact.extend(hosts.exact_hosts());
162                }
163            }
164        }
165        if !exact.is_empty() {
166            tau_sandbox_proxy::validate_hosts(&exact).map_err(|e| CapabilityError::Proxy {
167                message: format!("host validation: {e}"),
168            })?;
169        }
170        Ok(())
171    }
172}

174impl ProcessCapabilityGate for WindowsSandbox {

176    async fn wrap_spawn(
177        &self,
178        plan: &CapabilityPlan,
179        cmd: &mut Command,

189        _cmd: &mut Command,

191        Err(CapabilityError::Unavailable {
192            reason: "tau-sandbox-windows is Windows-only".to_string(),
193        })
194    }

197/// Probe for AppContainer availability. Cached per `WindowsSandbox`
198/// instance via `OnceCell`.
199///
200/// The real Win32 AppContainer implementation has landed (`acl.rs`
201/// grant/revoke, `wrap_spawn` via the launcher exec-wrapper), so this
202/// returns `Available { tier: Strict }` on Windows — filesystem and
203/// process isolation are enforced. Network egress (#622) is enforced via
204/// a per-container named-pipe proxy: the AppContainer gets no network
205/// capability SIDs, and the pipe's DACL admits only the current user and
206/// the container's own SID.
207async fn run_probe() -> CapabilityProbe {
208    if !cfg!(target_os = "windows") {
209        return CapabilityProbe::Unavailable {
210            reason: "not running on Windows".to_string(),
211        };
212    }
213    CapabilityProbe::Available {
214        tier: tau_ports::CapabilityTier::Strict,
215        details: "AppContainer (FS + process isolation + proxied egress via named pipe)"
216            .to_string(),
217    }
218}

220#[cfg(target_os = "windows")]
221async fn wrap_spawn_windows(
222    plan: &CapabilityPlan,
223    cmd: &mut Command,
224) -> Result<CapabilityHandle, CapabilityError> {
225    use std::sync::atomic::{AtomicU64, Ordering};
226    static COUNTER: AtomicU64 = AtomicU64::new(0);
227
228    let caps = build_appcontainer_caps(plan);
229
230    // Generate a unique AppContainer profile name + SID per spawn.
231    let counter = COUNTER.fetch_add(1, Ordering::Relaxed);
232    let profile_name = format!("tau-sandbox-{}-{}", std::process::id(), counter);
233    let app_sid = acl::create_appcontainer_profile(&profile_name).map_err(|e| {
234        CapabilityError::WrapFailed {
235            message: format!("create_appcontainer_profile: {e}"),
236        }
237    })?;
238
239    // Grant ACLs on plan-specified paths to the AppContainer SID.
240    let mut granted_paths: Vec<(String, acl::AccessKind)> = Vec::new();
241    for path in &caps.fs_read_paths {
242        acl::grant_access(&app_sid, path, acl::AccessKind::Read).map_err(|e| {
243            CapabilityError::WrapFailed {
244                message: format!("grant read on {path}: {e}"),
245            }
246        })?;
247        granted_paths.push((path.clone(), acl::AccessKind::Read));
248    }
249    // A write grant needs its target to EXIST: `SetNamedSecurityInfoW`
250    // answers `WIN32_ERROR(2)` (ERROR_FILE_NOT_FOUND) otherwise and the
251    // whole plan fails closed — which is what #622's CI round 1 hit on
252    // cargo's not-yet-created `<package>/target`. The container cannot
253    // create it either (its parent is deliberately not write-granted),
254    // so the spawn layer materialises the directory here. Only
255    // glob-shaped (`<dir>/**`) write entries qualify; see
256    // [`dir_shaped_write_paths`].
257    let dir_shaped: std::collections::HashSet<String> =
258        dir_shaped_write_paths(plan).into_iter().collect();
259    for path in &caps.fs_write_paths {
260        if dir_shaped.contains(path) && !std::path::Path::new(path).exists() {
261            std::fs::create_dir_all(path).map_err(|e| CapabilityError::WrapFailed {
262                message: format!("create write-grant dir {path}: {e}"),
263            })?;
264        }
265        acl::grant_access(&app_sid, path, acl::AccessKind::Write).map_err(|e| {
266            CapabilityError::WrapFailed {
267                message: format!("grant write on {path}: {e}"),
268            }
269        })?;
270        granted_paths.push((path.clone(), acl::AccessKind::Write));
271    }
272
273    // Egress (#622): spawn the per-container pipe proxy. The container
274    // gets NO network capability SIDs — the SID-ACL'd pipe into this
275    // HostAllow proxy is its only route out. Fail closed on any setup
276    // error (ADR-0014).
277    let proxy_handle = if caps.has_http {
278        let mut any = false;
279        let mut exact: Vec<String> = Vec::new();
280        for cap in &plan.capabilities {
281            if let Capability::Network(NetCapability::Http { hosts, .. }) = cap {
282                if hosts.is_any() {
283                    any = true;
284                } else {
285                    exact.extend(hosts.exact_hosts());
286                }
287            }
288        }
289        let policy = if any {
290            tau_sandbox_proxy::HostAllow::Any
291        } else {
292            tau_sandbox_proxy::HostAllow::Exact(exact)
293        };
294        let handle =
295            pipe_proxy::spawn_pipe_proxy(policy, &app_sid).map_err(|e| CapabilityError::Proxy {
296                message: format!("spawn_pipe_proxy: {e}"),
297            })?;
298        Some(handle)
299    } else {
300        None
301    };
302
303    // Resolve the bridge exe (env override -> sibling of tau.exe ->
304    // PATH) and grant the container read+execute on it so the image
305    // load inside the AppContainer succeeds.
306    let bridge_exe: Option<std::path::PathBuf> = if proxy_handle.is_some() {
307        let p = std::env::var_os("TAU_NET_BRIDGE_WIN_PATH")
308            .map(std::path::PathBuf::from)
309            .or_else(|| {
310                std::env::current_exe()
311                    .ok()
312                    .and_then(|e| e.parent().map(|d| d.join("tau-net-bridge-win.exe")))
313                    .filter(|p| p.exists())
314            })
315            .unwrap_or_else(|| std::path::PathBuf::from("tau-net-bridge-win"));
316        if p.exists() {
317            let ps = p.to_string_lossy().into_owned();
318            acl::grant_access(&app_sid, &ps, acl::AccessKind::Read).map_err(|e| {
319                CapabilityError::WrapFailed {
320                    message: format!("grant read on bridge exe {ps}: {e}"),
321                }
322            })?;
323            granted_paths.push((ps, acl::AccessKind::Read));
324        }
325        Some(p)
326    } else {
327        None
328    };
329
330    // Rebuild the command to run the target THROUGH the launcher, which
331    // does CreateProcessAsUserW inside the AppContainer. Mirrors the
332    // darwin sandbox-exec rebuild (`*cmd = Command::new(...)`).
333    let launcher = std::env::var_os("TAU_APPCONTAINER_LAUNCHER_PATH")
334        .unwrap_or_else(|| std::ffi::OsString::from("tau-appcontainer-launcher"));
335    let orig_program = cmd.get_program().to_os_string();
336    let orig_args: Vec<std::ffi::OsString> = cmd.get_args().map(|a| a.to_os_string()).collect();
337    let orig_envs: Vec<(std::ffi::OsString, Option<std::ffi::OsString>)> = cmd
338        .get_envs()
339        .map(|(k, v)| (k.to_os_string(), v.map(|x| x.to_os_string())))
340        .collect();
341    let orig_cwd = cmd.get_current_dir().map(|p| p.to_path_buf());
342
343    *cmd = Command::new(launcher);
344    cmd.arg("--profile").arg(&profile_name);
345    // No capability SIDs are added for network (#622 design): the pipe
346    // proxy's DACL is the enforcement point, not a well-known SID grant.
347    cmd.arg("--");
348    if let (Some(proxy), Some(bridge)) = (&proxy_handle, &bridge_exe) {
349        // launcher -- <bridge> --pipe <name> -- <orig> <args...>
350        cmd.arg(bridge)
351            .arg("--pipe")
352            .arg(proxy.pipe_name())
353            .arg("--");
354    }
355    cmd.arg(orig_program).args(orig_args);
356    for (k, v) in orig_envs {
357        match v {
358            Some(val) => {
359                cmd.env(k, val);
360            }
361            None => {
362                cmd.env_remove(k);
363            }
364        }
365    }
366    if let Some(dir) = orig_cwd {
367        cmd.current_dir(dir);
368    }
369
370    // Build the CapabilityHandle. On drop:
371    // 1. revoke ACL grants in reverse order
372    // 2. delete the AppContainer profile
373    let cleanup_sid = app_sid.clone();
374    let cleanup_profile = profile_name.clone();
375    let cleanup_paths = granted_paths;
376    let mut handle = CapabilityHandle::new(move || {
377        for (path, kind) in cleanup_paths.iter().rev() {
378            let _ = acl::revoke_access(&cleanup_sid, path, *kind);
379        }
380        let _ = acl::delete_appcontainer_profile(&cleanup_profile);
381    });
382    if let Some(p) = proxy_handle {
383        handle.nest_handle(Box::new(p));
384    }
385
386    Ok(handle)
387}

447    #[test]

449        // `HostSet`'s deserializer now rejects "*" as a non-hostname at

456            "context": null,
457            "limits": null,

459        let err = serde_json::from_value::<CapabilityPlan>(plan_json)

461        assert!(

463            "expected wildcard-rejection error, got {err}"
