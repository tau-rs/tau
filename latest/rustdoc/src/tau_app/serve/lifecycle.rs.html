16/// Main serve entry point. Builds runtime, spawns tasks, blocks until shutdown.
17pub async fn run(opts: ServeOptions) -> Result<()> {
18    super::tracing_init::install();
19
20    info!(project = %opts.project_path.display(), "serve starting");
21    // Echo the resolved configuration so an operator can confirm from the
22    // logs that their flags (`--idle-timeout`, `--max-concurrent`, …) took
23    // effect (audit O1).
24    info!(?opts, "serve config");
25
26    let project = Arc::new(
27        Project::load(&opts.project_path)
28            .await
29            .context("load project")?,
30    );
31
32    let runtime = build_runtime(&project).await.context("build runtime")?;
33
34    let (in_tx, in_rx) = mpsc::channel(64);
35    let (out_tx, out_rx) = mpsc::channel(256);
36
37    // Linux: set PDEATHSIG so we die when parent dies.
38    #[cfg(target_os = "linux")]
39    set_pdeathsig();
40
41    let cancel_reg = CancelRegistry::default();
42    // Tripped when an outbound send fails (writer task gone). The dispatch
43    // loop selects on it to begin graceful shutdown (audit O4).
44    let writer_gone = tokio_util::sync::CancellationToken::new();
45    let dispatcher = Dispatcher {
46        project: project.clone(),
47        runtime: Arc::new(runtime),
48        handshake: HandshakeState::default(),
49        cancel_reg: cancel_reg.clone(),
50        max_concurrent: opts.max_concurrent,
51        out_tx: out_tx.clone(),
52        writer_gone: writer_gone.clone(),
53    };
54
55    let local_set = LocalSet::new();
56
57    // Shared idle clock — touched by the reader on every inbound line and by
58    // the writer on every outbound message, so an in-flight run that keeps
59    // streaming holds the timer open.
60    let activity = Activity::new();
61
62    // Reader and writer tasks are Send-friendly — spawn on multi-thread side.
63    let reader_handle = tokio::spawn(framing::reader_task(in_tx, activity.clone()));
64    let writer_handle = tokio::spawn(framing::writer_task(out_rx, activity.clone()));
65
66    if opts.ready_on_stderr {
67        eprintln!("tau-serve ready");
68    }
69
70    // Run dispatcher loop on the LocalSet so per-request tasks have a
71    // current_thread executor available for non-Send streams.
72    // spawn_local (used inside Dispatcher::spawn_run) works within any
73    // active LocalSet on the current thread — no &LocalSet borrow needed.
74    let idle_timeout = opts.idle_timeout;
75    let writer_gone_signal = writer_gone.clone();
76    let dispatch_result = local_set
77        .run_until(async move {
78            let shutdown_signal = wait_for_shutdown_signal();
79            // When no `--idle-timeout` is set, this future never resolves and
80            // the select arm is effectively disabled.
81            let idle = async {
82                match idle_timeout {
83                    Some(d) => idle::wait_for_idle(activity, d).await,
84                    None => std::future::pending::<()>().await,
85                }
86            };
87            tokio::select! {
88                r = dispatcher.run(in_rx) => r,
89                _ = shutdown_signal => Ok(()),
90                _ = idle => {
91                    info!(timeout = ?idle_timeout, "idle timeout reached, shutting down");
92                    Ok(())
93                }
94                _ = writer_gone_signal.cancelled() => {
95                    warn!("writer task gone; shutting down dispatcher");
96                    Ok(())
97                }
98            }
99        })
100        .await;
101
102    // Graceful drain.
103    cancel_reg.cancel_all();
104    let grace_result = tokio::time::timeout(opts.shutdown_grace, async {
105        let _ = reader_handle.await;
106    })
107    .await;
108    if grace_result.is_err() {
109        warn!(grace = ?opts.shutdown_grace, "shutdown grace expired");
110    }
111    drop(out_tx);
112    let _ = writer_handle.await;
113
114    info!("serve shutdown complete");
115    dispatch_result?;
116    Ok(())
117}

119/// Resolve an agent entry's LLM backend package name via the project
120/// `[models]` table (per-agent model resolution). `validate_models`
121/// (D7 stage 1) refuses a project whose agent names an unknown alias, so this
122/// is infallible for a validated config; the error arm is defense-in-depth.
123fn agent_llm_backend(
124    entry: &tau_pkg::project::project::AgentEntry,
125    models: &std::collections::BTreeMap<String, tau_pkg::project::project::ModelEntry>,
126) -> Result<String> {
127    models
128        .get(&entry.model)
129        .map(|m| m.backend.clone())
130        .ok_or_else(|| {
131            anyhow::anyhow!(
132                "agent {:?} references model alias {:?} absent from [models]",
133                entry.id,
134                entry.model
135            )
136        })
137}

139/// Build the `Runtime` from a loaded `Project`.
140///
141/// Iterates every agent in the project's `tau.toml`, reads the lockfile to
142/// locate the LLM-backend and tool plugin binaries, and spawns them via
143/// `tau_runtime_tokio::plugin_host`. Deduplicates by plugin name so that two
144/// agents sharing the same backend/tool spawn only one subprocess.
145///
146/// Sandbox enforcement (audit S4): serve mirrors the CLI `run`/`chat`
147/// path. It resolves one [`SandboxAdapter`] for the whole serve process
148/// and builds a per-plugin [`tau_ports::CapabilityPlan`] from each
149/// plugin's declared capabilities, then passes both into `load_*`.
150/// `plugin_host` only enforces when both are present (`plan.zip(adapter)`),
151/// so resolving the adapter is as load-bearing as building the plan —
152/// passing one without the other silently disables the Layer-4 backstop.
153///
154/// Serve mode v1 simplifications (deferred to future iterations):
155/// - Recorder / trace context: a synthetic trace context is generated
156///   per plugin (no recorder ledger, no run-level trace propagation).
157/// - `[agents.<id>.config]` is forwarded to the LLM backend as JSON;
158///   tools always receive `{}` (per-tool config selectors not yet landed).
159/// - `working_context` / resource `limits` are not yet threaded into the
160///   plan (both `None`), matching the CLI tool path.
161async fn build_runtime(project: &Project) -> Result<tau_runtime_tokio::Runtime> {
162    use tau_pkg::LockFile;
163    use tau_plugin_protocol::handshake::TraceContext;
164    use tau_runtime_tokio::plugin_host;
165
166    let lockfile_path = project.scope.lockfile_path();
167    let lockfile = LockFile::load(&lockfile_path)
168        .with_context(|| format!("load lockfile {}", lockfile_path.display()))?;
169
170    // Resolve a single sandbox adapter for the serve process. Serve has no
171    // `--no-sandbox` / `--sandbox` flags, so we always ask the resolver for
172    // the best adapter satisfying the scope's `[sandbox]` requirements and
173    // every referenced plugin's declared tier floor (audit S4).
174    let scope_reqs = read_scope_sandbox_requirements(&project.scope)?;
175    let plugin_reqs = gather_plugin_sandbox_reqs(&project.scope, &lockfile, &project.config);
176    let adapter = resolve_serve_adapter(&scope_reqs, &plugin_reqs).await?;
177    let mut host_options = plugin_host::PluginHostOptions::default();
178    host_options.sandbox_adapter = Some(adapter);
179
180    let mut builder = tau_runtime_tokio::Runtime::builder();
181
182    // Dedup by plugin name — multiple agents may reference the same package.
183    let mut seen_llm_backends: std::collections::HashSet<String> = Default::default();
184    let mut seen_tools: std::collections::HashSet<String> = Default::default();
185
186    for entry in project.config.agents.values() {
187        // ---- LLM backend ----
188        // The agent's backend package is derived from the project `[models]`
189        // table via its `model` alias (per-agent model resolution); the
190        // removed `AgentEntry.llm_backend` field no longer carries it.
191        let backend_name = agent_llm_backend(entry, &project.config.models)?;
192        if seen_llm_backends.insert(backend_name.clone()) {
193            let pkg_name: tau_domain::PackageName = backend_name
194                .parse()
195                .with_context(|| format!("invalid LLM backend package name {backend_name:?}"))?;
196            let pkg = lockfile.find(&pkg_name).ok_or_else(|| {
197                anyhow::anyhow!(
198                    "LLM backend {:?} not installed in scope (run `tau install <url>`)",
199                    backend_name
200                )
201            })?;
202            let plugin = pkg.plugin.as_ref().ok_or_else(|| {
203                anyhow::anyhow!(
204                    "package {:?} has no [plugin] table in its tau.toml (data-only package \
205                     cannot be used as an llm_backend)",
206                    backend_name
207                )
208            })?;
209            let trace_context = TraceContext::new(
210                "serve".to_string(),
211                entry.id.clone(),
212                "serve-root".to_string(),
213            );
214            let config = agent_config_to_json(entry);
215            // Build this plugin's capability plan from its manifest, narrowed
216            // by the agent's project-level capability overrides (audit S4).
217            let caps = read_plugin_caps(&project.scope, &lockfile, &backend_name);
218            let plan = plan_for_plugin(&caps, &entry.capability_overrides).with_context(|| {
219                format!("building sandbox plan for LLM backend {backend_name:?}")
220            })?;
221            let backend = plugin_host::load_llm_backend(
222                plugin,
223                config,
224                trace_context,
225                host_options.clone(),
226                Some(&plan),
227            )
228            .await
229            .with_context(|| format!("load LLM backend {backend_name:?}"))?;
230            builder = builder.with_dyn_llm_backend(backend);
231        }
232
233        // ---- Tools ----
234        for tool in &entry.requires.tools {
235            let tool_name = tool.name.to_string();
236            if seen_tools.insert(tool_name.clone()) {
237                let pkg = lockfile.find(&tool.name).ok_or_else(|| {
238                    anyhow::anyhow!(
239                        "tool {:?} not installed in scope (run `tau install <url>`)",
240                        tool_name
241                    )
242                })?;
243                let plugin = pkg.plugin.as_ref().ok_or_else(|| {
244                    anyhow::anyhow!(
245                        "package {:?} has no [plugin] table in its tau.toml (data-only package \
246                         cannot be used as a tool)",
247                        tool_name
248                    )
249                })?;
250                let trace_context = TraceContext::new(
251                    "serve".to_string(),
252                    entry.id.clone(),
253                    "serve-root".to_string(),
254                );
255                // Per-tool capability overrides are not addressable in the
256                // project tau.toml at v0.1 — pass an empty override slice, as
257                // the CLI tool path does (audit S4).
258                let caps = read_plugin_caps(&project.scope, &lockfile, &tool_name);
259                let plan = plan_for_plugin(&caps, &[])
260                    .with_context(|| format!("building sandbox plan for tool {tool_name:?}"))?;
261                let loaded = plugin_host::load_tool(
262                    plugin,
263                    serde_json::Value::Object(Default::default()),
264                    trace_context,
265                    host_options.clone(),
266                    Some(&plan),
267                )
268                .await
269                .with_context(|| format!("load tool {:?}", tool_name))?;
270                builder = builder.with_dyn_tool(loaded);
271            }
272        }
273    }
274
275    // Use `build_allow_empty` so a project with zero agents still starts
276    // successfully. Serve mode accepts `meta.handshake` and `meta.ping` on
277    // any project; `runtime.run` calls will return -32010 UNKNOWN_AGENT
278    // which is the correct behaviour when no agents are defined.
279    builder.build_allow_empty().context("build Runtime")
280}

282/// Resolve the sandbox adapter for serve mode (audit S4).
283///
284/// Mirrors the non-force branch of the CLI `run`/`chat` path
285/// (`plugin_loader::load_plugins`): serve exposes no `--no-sandbox` /
286/// `--sandbox` flags, so it always asks the resolver for the best adapter
287/// satisfying the scope's `[sandbox]` requirements and every referenced
288/// plugin's declared tier floor. The returned adapter is the half
289/// `plugin_host` zips against each plugin's [`tau_ports::CapabilityPlan`];
290/// without it the Layer-4 backstop is silently disabled.
291async fn resolve_serve_adapter(
292    requirements: &tau_pkg::scope::SandboxRequirements,
293    plugin_reqs: &[tau_domain::PluginSandboxRequirements],
294) -> Result<std::sync::Arc<tau_runtime_tokio::process_gate::SandboxAdapter>> {
295    tau_runtime_tokio::process_gate::resolve_adapter(requirements, plugin_reqs)
296        .await
297        .map(std::sync::Arc::new)
298        .map_err(|e| anyhow::anyhow!("resolve sandbox adapter for serve mode: {e}"))
299}

301/// Build a plugin's Layer-3 capability plan (audit S4).
302///
303/// Thin serve-side wrapper over [`tau_runtime_tokio::process_gate::build_plan`]:
304/// serve defers `working_context` and resource `limits` (both `None`),
305/// exactly like the CLI tool path. Returns a real `CapabilityPlan` so serve
306/// can pass `Some(&plan)` into `load_*` and have `plugin_host` enforce it.
307fn plan_for_plugin(
308    caps: &[tau_domain::Capability],
309    overrides: &[tau_pkg::capability_override::CapabilityOverride],
310) -> Result<tau_ports::CapabilityPlan> {
311    tau_runtime_tokio::process_gate::build_plan(caps, overrides, None, None)
312        .map_err(|e| anyhow::anyhow!("build sandbox plan: {e}"))
313}

315/// Read the scope's `[sandbox]` requirements from `config.toml`, falling
316/// back to defaults when the file doesn't exist (a freshly-created scope
317/// without an explicit config). Mirrors `plugin_loader::load_plugins`.
318fn read_scope_sandbox_requirements(
319    scope: &tau_pkg::Scope,
320) -> Result<tau_pkg::scope::SandboxRequirements> {
321    let config_path = scope.config_path();
322    if config_path.exists() {
323        let text = std::fs::read_to_string(&config_path)
324            .with_context(|| format!("reading scope config at {config_path:?}"))?;
325        let scope_config = tau_pkg::scope::ScopeConfig::read_from_str(&text)
326            .with_context(|| format!("parsing scope config at {config_path:?}"))?;
327        Ok(scope_config.sandbox)
328    } else {
329        Ok(tau_pkg::scope::SandboxRequirements::default())
330    }
331}

333/// Collect the sandbox-tier floors declared by every plugin the project's
334/// agents reference (LLM backends + tools), deduplicated by package name.
335/// Feeds the resolver so a plugin demanding e.g. `Strict` can't be admitted
336/// under a weaker adapter (audit S4).
337fn gather_plugin_sandbox_reqs(
338    scope: &tau_pkg::Scope,
339    lockfile: &tau_pkg::LockFile,
340    config: &tau_pkg::project::ProjectConfig,
341) -> Vec<tau_domain::PluginSandboxRequirements> {
342    let mut seen: std::collections::HashSet<String> = Default::default();
343    let mut reqs = Vec::new();
344    for entry in config.agents.values() {
345        // Backend package derived from the project `[models]` table; skip
346        // agents whose model alias does not resolve (validate_models refuses
347        // these at build time — defensive skip here).
348        if let Some(backend) = config.models.get(&entry.model).map(|m| m.backend.clone()) {
349            if seen.insert(backend.clone()) {
350                if let Some(req) = read_plugin_sandbox_req(scope, lockfile, &backend) {
351                    reqs.push(req);
352                }
353            }
354        }
355        for tool in &entry.requires.tools {
356            let name = tool.name.to_string();
357            if seen.insert(name.clone()) {
358                if let Some(req) = read_plugin_sandbox_req(scope, lockfile, &name) {
359                    reqs.push(req);
360                }
361            }
362        }
363    }
364    reqs
365}

367/// Read a plugin's `[sandbox]` requirements from its on-disk `tau.toml`.
368///
369/// Mirrors `plugin_loader::read_plugin_sandbox_req`. Returns `None` if the
370/// package is absent from the lockfile (treated as "no floor", not an error).
371fn read_plugin_sandbox_req(
372    scope: &tau_pkg::Scope,
373    lockfile: &tau_pkg::LockFile,
374    package_name: &str,
375) -> Option<tau_domain::PluginSandboxRequirements> {
376    let pkg_name: tau_domain::PackageName = package_name.parse().ok()?;
377    let locked = lockfile.find(&pkg_name)?;
378    let pkg_dir = scope.package_dir(&locked.name, &locked.active_version);
379    match tau_pkg::read_manifest(&pkg_dir.join("tau.toml")) {
380        Ok(manifest) => Some(manifest.sandbox().clone()),
381        Err(_) => Some(tau_domain::PluginSandboxRequirements::default()),
382    }
383}

385/// Read a plugin's declared capabilities from its on-disk `tau.toml`, used
386/// to construct its [`tau_ports::CapabilityPlan`].
387///
388/// Mirrors `plugin_loader::read_plugin_caps_by_name`. Returns an empty vec
389/// when the package is absent from the lockfile or its manifest can't be
390/// read — an empty plan denies everything, the safe default.
391fn read_plugin_caps(
392    scope: &tau_pkg::Scope,
393    lockfile: &tau_pkg::LockFile,
394    package_name: &str,
395) -> Vec<tau_domain::Capability> {
396    let Ok(pkg_name) = package_name.parse::<tau_domain::PackageName>() else {
397        return Vec::new();
398    };
399    let Some(locked) = lockfile.find(&pkg_name) else {
400        return Vec::new();
401    };
402    let pkg_dir = scope.package_dir(&locked.name, &locked.active_version);
403    match tau_pkg::read_manifest(&pkg_dir.join("tau.toml")) {
404        Ok(manifest) => manifest.capabilities().to_vec(),
405        Err(_) => Vec::new(),
406    }
407}

409/// Convert an agent's `[agents.<id>.config]` TOML sub-table to a
410/// `serde_json::Value` for the plugin handshake.
411///
412/// Returns `{}` (not `Null`) when no config is set: plugin config structs
413/// typically don't deserialize from JSON null, so an empty object lets every
414/// plugin with default-able config construct its defaults.
415fn agent_config_to_json(entry: &tau_pkg::project::AgentEntry) -> serde_json::Value {
416    if entry.config.is_empty() {
417        return serde_json::Value::Object(serde_json::Map::new());
418    }
419    let mut map = serde_json::Map::with_capacity(entry.config.len());
420    for (k, v) in &entry.config {
421        map.insert(k.clone(), toml_to_json(v.clone()));
422    }
423    serde_json::Value::Object(map)
424}
425
426fn toml_to_json(v: toml::Value) -> serde_json::Value {
427    match v {
428        toml::Value::String(s) => serde_json::Value::String(s),
429        toml::Value::Integer(i) => serde_json::Value::Number(i.into()),
430        toml::Value::Float(f) => serde_json::Number::from_f64(f)
431            .map(serde_json::Value::Number)
432            .unwrap_or(serde_json::Value::Null),
433        toml::Value::Boolean(b) => serde_json::Value::Bool(b),
434        toml::Value::Datetime(dt) => serde_json::Value::String(dt.to_string()),
435        toml::Value::Array(arr) => {
436            serde_json::Value::Array(arr.into_iter().map(toml_to_json).collect())
437        }
438        toml::Value::Table(t) => {
439            serde_json::Value::Object(t.into_iter().map(|(k, v)| (k, toml_to_json(v))).collect())
440        }
441    }
442}

444/// Wait for any of: SIGTERM, SIGINT (Unix), Ctrl-C (all platforms).
445///
446/// Unix uses dedicated `SignalKind::terminate` + `SignalKind::interrupt`
447/// streams (`tokio::signal::unix`). Windows has no `unix` submodule;
448/// we use `tokio::signal::ctrl_c` as the cross-platform interrupt
449/// signal. SIGTERM has no analog on Windows — parent-death detection
450/// there relies on stdin EOF (see lifecycle::run).
451#[cfg(unix)]
452async fn wait_for_shutdown_signal() {
453    use tokio::signal::unix::{signal, SignalKind};
454    let mut term = match signal(SignalKind::terminate()) {
455        Ok(s) => s,
456        Err(_) => return,
457    };
458    let mut int = match signal(SignalKind::interrupt()) {
459        Ok(s) => s,
460        Err(_) => return,
461    };
462    tokio::select! {
463        _ = term.recv() => info!("received SIGTERM"),
464        _ = int.recv() => info!("received SIGINT"),
465    }
466}
467
468#[cfg(not(unix))]
469async fn wait_for_shutdown_signal() {
470    let _ = tokio::signal::ctrl_c().await;
471    info!("received Ctrl-C");
472}

474/// On Linux, ask the kernel to deliver SIGTERM to us when our parent dies.
475#[cfg(target_os = "linux")]
476#[allow(unsafe_code)]
477fn set_pdeathsig() {
478    // SAFETY: prctl is async-signal-safe; the SIGTERM target is the
479    // current process which always exists. This is the only unsafe
480    // block in tau-app — see the crate-level lib.rs note for context.
481    unsafe {
482        libc::prctl(
483            libc::PR_SET_PDEATHSIG,
484            libc::SIGTERM as libc::c_ulong,
485            0,
486            0,
487            0,
488        );
489    }
490}

503    /// S4 regression — the adapter half.
504    ///
505    /// `plugin_host::load_*` enforces the sandbox only when **both** the
506    /// capability plan and the adapter are present (`plan.zip(adapter)`).
507    /// Serve previously passed `PluginHostOptions::default()`, whose
508    /// `sandbox_adapter` is `None`, so the zip collapsed to `None` and every
509    /// plugin ran unsandboxed. Serve must now resolve a real adapter.
510    #[tokio::test]

512        // Mock injection makes `resolve_adapter` deterministic across host
513        // platforms (no real landlock/SBPL tooling needed in CI).
514        std::env::set_var("TAU_TESTING_ALLOW_MOCK_SANDBOX", "1");
515        let reqs = tau_pkg::scope::SandboxRequirements::default();
516        let result = resolve_serve_adapter(&reqs, &[]).await;
517        std::env::remove_var("TAU_TESTING_ALLOW_MOCK_SANDBOX");
518
519        let adapter = result.expect("serve must resolve a sandbox adapter");
520        assert!(
521            matches!(
522                *adapter,
523                tau_runtime_tokio::process_gate::SandboxAdapter::Mock(_)
524            ),
525            "expected a resolved adapter (the half plugin_host zips against the plan), \
526             got a non-mock variant under TAU_TESTING_ALLOW_MOCK_SANDBOX",
527        );
528    }

530    /// S4 regression — the plan half.
531    ///
532    /// Serve previously passed `sandbox_plan = None` to every `load_*` call.
533    /// `plan_for_plugin` is the serve-side construction point that produces a
