1//! Per-request executor for runtime.run and runtime.run_streaming.
2//!
3//! Wires JSON-RPC requests to tau_runtime_tokio::Runtime::run and
4//! Runtime::run_streaming. Streaming emits one runtime.event
5//! notification per RunEvent, correlated by the request id, then
6//! a final result response.
7

20/// Execute a runtime.run or runtime.run_streaming request.
21pub async fn execute(disp: Dispatcher, req: Request, streaming: bool) {
22    // 1. Parse params.
23    let params = match req.params.as_ref() {
24        Some(v) => v,
25        None => {
26            disp.send_err(
27                req.id,
28                error_codes::INVALID_PARAMS,
29                "params missing".into(),
30                None,
31            )
32            .await;
33            return;
34        }
35    };
36    let agent_id = match params.get("agent").and_then(|v| v.as_str()) {
37        Some(s) => s.to_string(),
38        None => {
39            disp.send_err(
40                req.id,
41                error_codes::INVALID_PARAMS,
42                "params.agent missing or not a string".into(),
43                None,
44            )
45            .await;
46            return;
47        }
48    };
49    let prompt = match params.get("prompt").and_then(|v| v.as_str()) {
50        Some(s) => s.to_string(),
51        None => {
52            disp.send_err(
53                req.id,
54                error_codes::INVALID_PARAMS,
55                "params.prompt missing or not a string".into(),
56                None,
57            )
58            .await;
59            return;
60        }
61    };
62
63    // 2. Resolve the agent. Unknown agents and package/manifest failures both
64    // surface here through one typed path (no string-prefix matching).
65    let (agent_def, manifest) = match disp.project.resolve(&agent_id) {
66        Ok(pair) => pair,
67        Err(ResolveError::AgentNotFound { agent_id, .. }) => {
68            disp.send_err(
69                req.id,
70                error_codes::UNKNOWN_AGENT,
71                format!("agent_id not found: {}", agent_id),
72                Some(json!({ "agent_id": agent_id })),
73            )
74            .await;
75            return;
76        }
77        Err(e) => {
78            // Manifest invalid, package not installed at the requested
79            // version, etc. — anything that isn't an unknown agent id.
80            disp.send_err(
81                req.id,
82                error_codes::RUNTIME_ERROR,
83                format!("agent resolution failed: {}", e),
84                Some(json!({ "agent_id": agent_id })),
85            )
86            .await;
87            return;
88        }
89    };
90
91    // 3. Build initial Message. tau_domain::Message has no Message::user()
92    // constructor outside tau-domain (non_exhaustive + struct-literal blocked).
93    // Use Message::new with Address::User sender/recipient and a Text payload.
94    // The runtime overwrites the recipient address internally.
95    let initial = Message::new(
96        Address::User,
97        Address::User,
98        MessagePayload::Text { content: prompt },
99    );
100
101    // Host-shell contract (`stream.rs::clock_ref`): serve drives the core
102    // streaming path directly, so it must inject the production clock +
103    // randomness or `run_streaming_inner` panics on the first port use.
104    let mut opts = RunOptions::default();
105    opts.clock = Some(std::sync::Arc::new(tau_runtime_tokio::TokioClock));
106    opts.random = Some(std::sync::Arc::new(tau_runtime_tokio::OsRandom));
107    let cancel = disp.cancel_reg.register(req.id.clone());
108
109    let result: Result<(), tau_runtime_tokio::RuntimeError> = if streaming {
110        execute_streaming(
111            &disp,
112            req.id.clone(),
113            agent_def,
114            manifest,
115            initial,
116            opts,
117            cancel,
118        )
119        .await
120    } else {
121        execute_batch(
122            &disp,
123            req.id.clone(),
124            agent_def,
125            manifest,
126            initial,
127            opts,
128            cancel,
129        )
130        .await
131    };
132
133    disp.cancel_reg.forget(&req.id);
134
135    if let Err(err) = result {
136        let obj = from_runtime_error(&err);
137        disp.send_err(req.id, obj.code, obj.message, obj.data).await;
138    }
139}

141async fn execute_batch(
142    disp: &Dispatcher,
143    id: RequestId,
144    agent_def: tau_domain::AgentDefinition,
145    manifest: tau_domain::PackageManifest,
146    initial: Message,
147    opts: RunOptions,
148    cancel: tokio_util::sync::CancellationToken,
149) -> Result<(), tau_runtime_tokio::RuntimeError> {
150    use tokio::select;
151    // Runtime::run signature (verified in Task 11 reconciliation):
152    //   pub async fn run(&self, agent_def, package_manifest, initial_message, options)
153    let fut = disp.runtime.run(agent_def, manifest, initial, opts);
154    select! {
155        outcome = fut => {
156            let outcome = outcome?;
157            let body = wire::outcome_to_json(&outcome);
158            disp.send_ok(id, body).await;
159            Ok(())
160        }
161        _ = cancel.cancelled() => {
162            disp.send_err(
163                id,
164                error_codes::CANCELLED,
165                "Cancelled by client".into(),
166                None,
167            ).await;
168            Ok(())
169        }
170    }
171}

173async fn execute_streaming(
174    disp: &Dispatcher,
175    id: RequestId,
176    agent_def: tau_domain::AgentDefinition,
177    manifest: tau_domain::PackageManifest,
178    initial: Message,
179    opts: RunOptions,
180    cancel: tokio_util::sync::CancellationToken,
181) -> Result<(), tau_runtime_tokio::RuntimeError> {
182    use tokio::select;
183    let stream = disp
184        .runtime
185        .run_streaming(agent_def, manifest, initial, opts)
186        .await?;
187    // run_streaming returns an `impl Stream + 'static` that is NOT Unpin.
188    // Pin it to the stack so StreamExt::next() is callable in the select! loop.
189    tokio::pin!(stream);
190
191    let mut last_token_usage: Option<Value> = None;
192    let mut stop_reason: Option<String> = None;
193
194    loop {
195        select! {
196            biased;
197            _ = cancel.cancelled() => {
198                disp.send_err(
199                    id,
200                    error_codes::CANCELLED,
201                    "Cancelled by client".into(),
202                    None,
203                ).await;
204                return Ok(());
205            }
206            event = stream.next() => {
207                match event {
208                    None => break,
209                    Some(ev) => emit_event(disp, &id, &ev, &mut last_token_usage, &mut stop_reason).await,
210                }
211            }
212        }
213    }
214
215    let body = json!({
216        "final": true,
217        "token_usage": last_token_usage,
218        "stop_reason": stop_reason,
219    });
220    disp.send_ok(id, body).await;
221    Ok(())
222}

224async fn emit_event(
225    disp: &Dispatcher,
226    id: &RequestId,
227    event: &RunEvent,
228    last_token_usage: &mut Option<Value>,
229    stop_reason: &mut Option<String>,
230) {
231    // The wire-shape projection (including the field renames vs the runtime
232    // types) lives in `wire::event_to_wire`, the single source of truth.
233    // `wire` carries the streaming side-effect updates the dispatcher must
234    // remember for the final summary.
235    let w = wire::event_to_wire(event);
236    if let Some(sr) = w.stop_reason {
237        *stop_reason = Some(sr);
238    }
239    if let Some(tu) = w.token_usage {
240        *last_token_usage = Some(tu);
241    }
242    disp.send_notification(
243        methods::RUNTIME_EVENT,
244        json!({ "id": id, "kind": w.kind, "data": w.data }),
245    )
246    .await;
247}
