1//! Project + agent resolution wrapper for serve mode.
2//!
3//! Thin orchestration over `tau_pkg::project::*`. Loads a project's
4//! `tau.toml`, holds the parsed `ProjectConfig`, and resolves
5//! per-call agent ids to `(AgentDefinition, PackageManifest)` pairs
6//! via `tau_pkg::project::build_agent_definition`.
7//!
8//! See ADR-0031 spec §4 and the post-refactor note in the plan.
9//!
10//! ## Reconciliation note (post-PR #133)
11//!
12//! The plan assumed `Scope::default_for_cwd` and
13//! `AgentResolutionError::AgentNotFound`; neither exists in the actual
14//! tau-pkg API. Adaptations made:
15//!
16//! - `Scope::resolve(root)` is used instead (walk-up from project root,
17//!   no side-effecting directory creation).
18//! - Unknown-agent detection is done in [`Project::resolve`] by returning
19//!   the typed [`ResolveError::AgentNotFound`] variant; the dispatcher
20//!   (Task 10) maps that single typed path to JSON-RPC `-32010 Unknown
21//!   agent`. There is no string-prefix error contract.
22
23use anyhow::{Context, Result};
24use std::path::{Path, PathBuf};
25use tau_domain::{AgentDefinition, PackageManifest};
26use tau_pkg::project::{build_agent_definition, AgentResolutionError, ProjectConfig};
27use tau_pkg::Scope;
28
29/// Loaded project state held by a serve-mode process for its lifetime.
30///
31/// Built once at startup from `--project <path>` (see Task 12).
32/// Read-only thereafter. `Arc<Project>` is shared across concurrent
33/// dispatcher tasks.
34#[derive(Debug, Clone)]
35pub struct Project {
36    /// Canonical project root (the directory containing tau.toml).
37    pub root: PathBuf,
38    /// Scope used for agent resolution. Captured at load time.
39    pub scope: Scope,
40    /// Parsed tau.toml.
41    pub config: ProjectConfig,
42}
43
44/// Error returned by [`Project::resolve`].
45///
46/// Typed so the dispatcher routes unknown agents to `-32010 Unknown agent`
47/// through one match arm — no string-prefix sniffing.
48#[derive(Debug, thiserror::Error)]
49pub enum ResolveError {
50    /// `agent_id` is not defined in the project's `tau.toml`.
51    #[error("agent {agent_id:?} is not defined in {root}")]
52    AgentNotFound {
53        /// The unknown agent id, echoed back in the JSON-RPC error data.
54        agent_id: String,
55        /// Project root, for the human-readable message.
56        root: String,
57    },
58    /// Package resolution failed (not installed, version unsatisfied, manifest
59    /// invalid). Flows to `-32006 RUNTIME_ERROR` at the dispatcher.
60    #[error(transparent)]
61    Resolution(#[from] AgentResolutionError),
62}
63
64impl Project {
65    /// Load a project from disk.
66    ///
67    /// Reads + parses `tau.toml`, captures the active `Scope` via
68    /// `Scope::resolve` (walk-up from `root`). Does not resolve agents
69    /// (that happens lazily per-call via [`Project::resolve`]).
70    pub async fn load(root: &Path) -> Result<Self> {
71        let root = std::fs::canonicalize(root)
72            .with_context(|| format!("canonicalize project root {}", root.display()))?;
73        let config = ProjectConfig::from_path(root.join("tau.toml"))
74            .with_context(|| format!("parse tau.toml at {}", root.display()))?;
75        let scope = Scope::resolve(&root).with_context(|| "compute scope for project root")?;
76        Ok(Self {
77            root,
78            scope,
79            config,
80        })
81    }
82
83    /// Resolve an agent id to `(AgentDefinition, PackageManifest)`.
84    ///
85    /// Returns [`ResolveError::AgentNotFound`] for an unknown agent id and
86    /// [`ResolveError::Resolution`] for package/manifest failures. The
87    /// dispatcher maps these to `-32010` and `-32006` respectively.
88    pub fn resolve(
89        &self,
90        agent_id: &str,
91    ) -> std::result::Result<(AgentDefinition, PackageManifest), ResolveError> {
92        let entry =
93            self.config
94                .agents
95                .get(agent_id)
96                .ok_or_else(|| ResolveError::AgentNotFound {
97                    agent_id: agent_id.to_string(),
98                    root: self.root.display().to_string(),
99                })?;
100        Ok(build_agent_definition(
101            entry,
102            &self.root,
103            &self.scope,
104            &self.config.models,
105        )?)
106    }
107
108    /// List all agent ids defined in the project's `tau.toml`.
109    pub fn agent_ids(&self) -> Vec<String> {
110        self.config.agents.keys().cloned().collect()
111    }
112}

114#[cfg(test)]
115mod tests {
116    use super::*;
117
118    fn fixture_dir() -> PathBuf {
119        PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("tests/fixtures/handshake-only")
120    }
121
122    #[tokio::test]
123    async fn resolve_unknown_agent_returns_typed_variant() {
124        // The handshake-only fixture defines zero agents, so any id is unknown.
125        let tau_home = std::env::temp_dir().join("tau-project-unit-test");
126        std::fs::create_dir_all(&tau_home).unwrap();
127        std::env::set_var("TAU_HOME", &tau_home);
128
129        let project = Project::load(&fixture_dir()).await.expect("load fixture");
130        let err = project.resolve("no-such-agent").unwrap_err();
131        assert!(
132            matches!(err, ResolveError::AgentNotFound { ref agent_id, .. } if agent_id == "no-such-agent"),
133            "expected typed AgentNotFound, got: {err:?}"
134        );
135    }
