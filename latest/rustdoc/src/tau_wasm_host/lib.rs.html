1//! β.7.5 wasm host — a std [`wasmtime`] embedder for the `tau-wasm-guest`
2//! WASI 0.2 component.
3//!
4//! The guest (built for `wasm32-wasip2`, see the `tau-wasm-guest` crate)
5//! exports `run(prompt) -> result<string, string>` from the `tau:host/runner`
6//! world and imports four host ports it cannot satisfy in-wasm:
7//!
8//! - `complete(request-json) -> result<string, string>` — delegated
9//!   inference (credentials live host-side, β.5).
10//! - `now-millis() -> u64` — wall clock.
11//! - `next-u64() -> u64` — randomness.
12//! - `emit-event(event-json: string)` — streams one `RunEvent` at a time
13//!   (fire-and-forget; see [`run_component`]'s return value).
14//!
15//! This crate offers those imports two implementations, through one seam
16//! ([`embed::EmbedPorts`]): a **deterministic** one for conformance —
17//! [`run_component`] / [`run_component_with_caps`], where time is a
18//! fixed-step counter, randomness is a seeded SplitMix64 PRNG, and inference
19//! replays a caller-supplied queue of canned `CompletionResponse` JSON
20//! strings (cassette-style), so the same `(component, prompt,
21//! llm_responses)` triple always yields the same bytes (the property β.6
22//! conformance / `WasmProfile` depends on) — and a caller-supplied one for
23//! embedding products ([`embed::run_component_with_ports`], EPIC 7.2), where
24//! the product owns the clock, entropy, and LLM client.
25//!
26//! [`run_component`] and [`run_component_with_caps`] instantiate the guest,
27//! drive `run`, and hand back the guest's payload (an empty sentinel —
28//! events flow via `emit-event`) plus the `RunEvent` JSON strings streamed
29//! via `emit-event`, in order. [`embed::run_component_with_ports`] follows
30//! the same shape but returns only the payload — events reach the caller
31//! live through `EmbedPorts::on_event` instead of being buffered. **`Ok` from
32//! any of these three means the guest's `run` returned its `ok` arm — it
33//! does NOT mean the workflow succeeded.** Once baked IR decodes, `run`
34//! always reaches its `ok` arm; a mid-run failure (e.g. an LLM error) is
35//! reported as a `RunEvent::FatalError` on the event stream, not as an
36//! `Err` here. `Err` is reserved for load/instantiate/trap failures and the
37//! guest's own load-time error arm (malformed IR before execution starts).
38//! Callers that care whether the run actually completed must watch for a
39//! terminal `RunEvent::RunCompleted` (see `crates/tau-wasm-embed-example`'s
40//! `completed` flag for the pattern).

74/// Deterministic step (ms) added to the clock on every `now-millis` call.
75const CLOCK_STEP_MILLIS: u64 = 1;
76/// Seed for the [`HostState`] PRNG. Fixed so randomness is reproducible.
77const PRNG_SEED: u64 = 0x9E37_79B9_7F4A_7C15;
78
79/// The four host functions a product supplies to run a tau component —
80/// the Rust face of the `tau:host/runner` WIT imports (EPIC 7.2).
81///
82/// Ports are owned by the run (`Box<dyn EmbedPorts>`; wasmtime's
83/// `Store<T: 'static>` forbids borrows). Impls that need state back after
84/// the run hold an `Arc` handle and keep a clone outside.
85pub trait EmbedPorts: Send {
86    /// Live inference. Typed: the host layer (de)serializes across the WIT
87    /// boundary; the `Err` string crosses into the guest's `result` error
88    /// arm. The WIT import is synchronous — an async product client blocks
89    /// inside its impl.
90    fn complete(&mut self, req: CompletionRequest) -> Result<CompletionResponse, String>;
91    /// Wall clock in milliseconds.
92    fn now_millis(&mut self) -> u64;
93    /// Next value from the product's entropy source.
94    fn next_u64(&mut self) -> u64;
95    /// One serialized `RunEvent` per call, in order, live as the guest emits
96    /// it. Raw JSON so this crate needs no tau-runtime-core dependency;
97    /// products deserialize with their own dep if they want typed events.
98    fn on_event(&mut self, event_json: &str);
99}

101/// The conformance ports: canned-cassette LLM, fixed-step clock, seeded
102/// SplitMix64 — exactly the behavior `run_component` has always had, now as
103/// one `EmbedPorts` impl.
104pub(crate) struct DeterministicPorts {
105    responses: VecDeque<String>,
106    clock_millis: u64,
107    prng_state: u64,
108    emitted: Arc<Mutex<Vec<String>>>,
109}
110
111impl DeterministicPorts {
112    /// Returns the ports plus the shared event-buffer handle the caller
113    /// drains after the run.
114    pub(crate) fn new(responses: Vec<String>) -> (Self, Arc<Mutex<Vec<String>>>) {
115        let emitted = Arc::new(Mutex::new(Vec::new()));
116        (
117            Self {
118                responses: responses.into(),
119                clock_millis: 0,
120                prng_state: PRNG_SEED,
121                emitted: Arc::clone(&emitted),
122            },
123            emitted,
124        )
125    }
126}
127
128impl EmbedPorts for DeterministicPorts {
129    fn complete(&mut self, _req: CompletionRequest) -> Result<CompletionResponse, String> {
130        let raw = self
131            .responses
132            .pop_front()
133            .ok_or_else(|| "tau-wasm-host: no canned completion response left".to_string())?;
134        serde_json::from_str(&raw)
135            .map_err(|e| format!("tau-wasm-host: invalid canned CompletionResponse: {e}"))
136    }
137
138    fn now_millis(&mut self) -> u64 {
139        let now = self.clock_millis;
140        self.clock_millis = self.clock_millis.wrapping_add(CLOCK_STEP_MILLIS);
141        now
142    }
143
144    fn next_u64(&mut self) -> u64 {
145        // SplitMix64 — identical to the legacy HostState sequence.
146        self.prng_state = self.prng_state.wrapping_add(0x9E37_79B9_7F4A_7C15);
147        let mut z = self.prng_state;
148        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
149        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
150        z ^ (z >> 31)
151    }
152
153    fn on_event(&mut self, event_json: &str) {
154        self.emitted
155            .lock()
156            .expect("event buffer lock")
157            .push(event_json.to_string());
158    }
159}

161/// Errors surfaced by [`run_component`], [`run_component_with_caps`], and
162/// [`embed::run_component_with_ports`]. `thiserror` at the crate boundary;
163/// `wasmtime`/`serde_json` failures are wrapped, never leaked as `anyhow`.
164/// None of these variants cover a mid-run workflow failure — that surfaces
165/// as a `RunEvent::FatalError` on the event stream while this crate still
166/// returns `Ok` (see the module docs above).
167#[derive(Debug, thiserror::Error)]
168pub enum WasmHostError {
169    /// The supplied bytes are not a loadable component.
170    #[error("failed to load wasm component: {0}")]
171    Load(#[source] anyhow::Error),
172    /// Linking the host imports or instantiating the component failed.
173    #[error("failed to instantiate component: {0}")]
174    Instantiate(#[source] anyhow::Error),
175    /// The guest trapped (panicked / unreachable) while running `run`.
176    #[error("guest `run` trapped: {0}")]
177    Trap(#[source] anyhow::Error),
178    /// The guest's `run` completed but returned its error arm (a load-time
179    /// failure, e.g. malformed IR — not a mid-run failure; those are
180    /// `FatalError` events, not this variant).
181    #[error("guest `run` returned an error: {0}")]
182    Guest(String),
183    /// A caller-supplied canned completion response is not valid
184    /// `CompletionResponse` JSON. Caught up-front so a malformed cassette
185    /// fails before instantiation rather than mid-run. Only reachable
186    /// through the deterministic/cassette entrypoints ([`run_component`],
187    /// [`run_component_with_caps`]) — [`embed::run_component_with_ports`]
188    /// takes live `EmbedPorts` instead of a cassette, so it never produces
189    /// this variant.
190    #[error("invalid canned completion response JSON: {0}")]
191    InvalidResponse(#[source] serde_json::Error),
192    /// Building the WASI context (e.g. a preopen dir failed to open, or a
193    /// resolved preopen would escape the sandbox root).
194    #[error("failed to configure WASI context: {0}")]
195    WasiConfig(#[source] anyhow::Error),
196}

198/// Store data backing the `tau:host/host` imports. One instance per
199/// [`run_component_with_ports`] call.
200///
201/// No `#[derive(Debug)]`: the WASI fields (`WasiCtx`/`WasiHttpCtx`, EPIC 3.3)
202/// are not `Debug`, and this struct is never formatted.
203struct HostState {
204    /// The four host functions, supplied by the caller (EPIC 7.2). The
205    /// conformance entrypoints pass `DeterministicPorts`.
206    ports: Box<dyn EmbedPorts>,
207    /// WASI 0.2 resource table (EPIC 3.3).
208    table: ResourceTable,
209    /// WASI 0.2 host context: exactly the preopens/network derived from the
210    /// component's allow-bounded caps (EPIC 3.3).
211    wasi: WasiCtx,
212    /// wasi:http resource bookkeeping (EPIC 3.3); the egress *decision* lives
213    /// in `egress`, not here.
214    http: WasiHttpCtx,
215    /// Network egress policy folded from the component's allow-bounded caps
216    /// (EPIC 3.3), sourced from the canonical `resolve_wasi_config`. Consulted
217    /// by the `WasiHttpHooks::send_request` override below before any outgoing
218    /// request is sent.
219    egress: EgressPolicy,
220}
221
222impl HostState {
223    fn new(ports: Box<dyn EmbedPorts>, wasi: WasiCtx, egress: EgressPolicy) -> Self {
224        Self {
225            ports,
226            table: ResourceTable::new(),
227            wasi,
228            http: WasiHttpCtx::new(),
229            egress,
230        }
231    }
232}

234impl host::Host for HostState {
235    fn complete(&mut self, request_json: String) -> Result<String, String> {
236        let req: CompletionRequest = serde_json::from_str(&request_json)
237            .map_err(|e| format!("tau-wasm-host: malformed CompletionRequest from guest: {e}"))?;
238        let resp = self.ports.complete(req)?;
239        serde_json::to_string(&resp)
240            .map_err(|e| format!("tau-wasm-host: failed to serialize CompletionResponse: {e}"))
241    }
242
243    fn now_millis(&mut self) -> u64 {
244        self.ports.now_millis()
245    }
246
247    fn next_u64(&mut self) -> u64 {
248        self.ports.next_u64()
249    }
250
251    fn emit_event(&mut self, event_json: String) {
252        self.ports.on_event(&event_json);
253    }
254}

278/// The egress gate: every `wasi:http` outgoing request is routed through
279/// `WasiHttpHooks::send_request` before wasmtime opens a socket. An authority
280/// or HTTP method that the allow-bounded caps didn't authorize is rejected
281/// here — the guest never gets a connection to it.
282impl WasiHttpHooks for EgressPolicy {
283    fn send_request(
284        &mut self,
285        request: hyper::Request<HyperOutgoingBody>,
286        config: OutgoingRequestConfig,
287    ) -> HttpResult<HostFutureIncomingResponse> {
288        let authority = request
289            .uri()
290            .authority()
291            .map(|a| a.as_str().to_string())
292            .unwrap_or_default();
293        let method = request.method().as_str();
294        if !self.permits(&authority, method) {
295            return Err(WasiHttpErrorCode::HttpRequestDenied.into());
296        }
297        Ok(default_send_request(request, config))
298    }
299}

301/// Build a `WasiCtx` that preopens exactly `cfg.preopens` (each resolved
302/// under `sandbox_root`) and denies all network egress by default (wasi:http
303/// egress is gated separately by the [`EgressPolicy`] filter; raw
304/// `wasi:sockets` stays default-deny).
305///
306/// `resolve_wasi_config` already dropped any non-G2 / `..`-bearing path, so
307/// every `host_dir` is absolute and escape-free; the `starts_with` check is
308/// defense-in-depth against a future resolver regression.
309fn wasi_ctx_from_config(
310    cfg: &WasiConfiguration,
311    sandbox_root: &Path,
312) -> Result<WasiCtx, WasmHostError> {
313    let mut builder = WasiCtxBuilder::new();
314    for p in &cfg.preopens {
315        // Defense-in-depth: `resolve_wasi_config` already drops non-G2 /
316        // `..`-bearing paths, but this is a public API over arbitrary caps, so
317        // re-check lexically. Scan segments explicitly — a component-wise
318        // `PathBuf::starts_with(sandbox_root)` AFTER `join` does NOT catch
319        // `..` (e.g. `<root>/../x` still `starts_with` `<root>`), so it would
320        // be a false guard.
321        if !p.host_dir.starts_with('/') || p.host_dir.split('/').any(|seg| seg == "..") {
322            return Err(WasmHostError::WasiConfig(anyhow::anyhow!(
323                "unsafe preopen path (not absolute or contains `..`): {}",
324                p.host_dir
325            )));
326        }
327        // Map the guest-visible absolute `host_dir` under the sandbox root.
328        let host_path = sandbox_root.join(p.host_dir.trim_start_matches('/'));
329        // Ensure the host dir exists so preopen succeeds.
330        std::fs::create_dir_all(&host_path).map_err(|e| WasmHostError::WasiConfig(e.into()))?;
331        let (dir_perms, file_perms) = match p.access {
332            PreopenAccess::ReadOnly => (DirPerms::READ, FilePerms::READ),
333            PreopenAccess::ReadWrite => (DirPerms::all(), FilePerms::all()),
334        };
335        builder
336            .preopened_dir(&host_path, &p.host_dir, dir_perms, file_perms)
337            .map_err(|e| WasmHostError::WasiConfig(e.into()))?;
338    }
339    Ok(builder.build())
340}

342/// Build the determinism `wasmtime::Config` (spec §7): canonicalise NaNs and
343/// turn off the nondeterministic relaxed-SIMD lowerings so float/SIMD codegen
344/// is bit-stable across hosts.
345fn determinism_config() -> wasmtime::Result<Config> {
346    let mut config = Config::new();
347    config.cranelift_nan_canonicalization(true);
348    config.wasm_relaxed_simd(false);
349    Ok(config)
350}

352/// Load `wasm_bytes` as a component, satisfy the four `tau:host/host` imports
353/// with deterministic stubs, instantiate it under a determinism `Config`,
354/// and drive the exported `run(prompt)`.
355///
356/// `llm_responses` is the cassette: each `complete` call pops the next entry
357/// (validated as `CompletionResponse` JSON up-front). Returns a
358/// `(payload, emitted)` pair: `payload` is the JSON string the guest's `run`
359/// produced on its `ok` arm — an empty sentinel now that events flow through
360/// `emit-event` instead (design D2) — and `emitted` is every `RunEvent` JSON
361/// string streamed via `emit-event`, in order.
362///
363/// Determinism contract: for fixed inputs the returned bytes are identical
364/// across calls and hosts — the property `WasmProfile` conformance relies on.
365/// EPIC 3.3: run a component whose WASI authority is bounded by `caps`.
366/// Filesystem globs resolve to preopens under `sandbox_root`; network egress
367/// is denied unless a `net.http` cap authorizes the target host.
368pub fn run_component_with_caps(
369    wasm_bytes: &[u8],
370    prompt: &str,
371    llm_responses: Vec<String>,
372    caps: &[tau_domain::Capability],
373    sandbox_root: &Path,
374) -> Result<(String, Vec<String>), WasmHostError> {
375    // Fail fast on a malformed cassette before touching wasmtime.
376    for resp in &llm_responses {
377        serde_json::from_str::<CompletionResponse>(resp).map_err(WasmHostError::InvalidResponse)?;
378    }
379    let (ports, emitted) = DeterministicPorts::new(llm_responses);
380    let payload =
381        run_component_with_ports(wasm_bytes, prompt, Box::new(ports), caps, sandbox_root)?;
382    let emitted = std::mem::take(&mut *emitted.lock().expect("event buffer lock"));
383    Ok((payload, emitted))
384}

386/// Variant A embedding entry (EPIC 7.2): run a built tau component with
387/// caller-supplied [`EmbedPorts`]. Events reach `ports.on_event` live, one
388/// serialized `RunEvent` per call; the `Ok` value is the guest's payload
389/// sentinel (empty today, reserved for forward-compat).
390///
391/// **`Ok` does not mean the workflow succeeded** — once baked IR decodes,
392/// the guest's `run` always returns its `ok` arm. A mid-run failure (e.g.
393/// the LLM port erroring) surfaces only as a `RunEvent::FatalError` on the
394/// `on_event` stream; `Err` from this function covers load/instantiate/trap
395/// and the guest's own load-time error arm. To know whether the run actually
396/// completed, watch for a terminal `RunEvent::RunCompleted` on `on_event`
397/// (see `crates/tau-wasm-embed-example`'s `completed` flag for the pattern).
398///
399/// `caps` bounds the WASI authority granted to the component (which
400/// preopens exist, which network egress is allowed); today the only
401/// supported way to obtain a non-empty, meaningfully-scoped `caps` slice is
402/// deserializing it from a package manifest via `tau-pkg` (the curated
403/// `tau_domain::Capability` constructors are behind tau-domain's
404/// test-only `test-fixtures` feature, not part of this crate's public
405/// surface). The shipped example ([`crate`]-level docs;
406/// `crates/tau-wasm-embed-example`) deliberately grants none (`&[]`) — no
407/// fs/net reaches the workflow.
408///
409/// `sandbox_root` is security-relevant: every preopen `caps` authorizes is
410/// resolved *under* this path, so it is the ceiling on what the
411/// component's filesystem grants can reach even before
412/// `caps` narrows further. `Path::new(".")` is only a safe value when
413/// `caps` grants no filesystem access at all — with any fs cap granted, pick
414/// a real sandbox directory.
415///
416/// The component and this host must come from compatible tau versions: the
417/// `complete` port round-trips `tau_ports::llm::CompletionRequest` /
418/// `CompletionResponse` as JSON across the wasm boundary, and neither type
419/// has serde defaults. An older component run against a newer host (or vice
420/// versa) after a field is added fails every `complete` call — reported as a
421/// `FatalError` event carrying a "malformed CompletionRequest" message, not
422/// as an `Err` return.
423pub fn run_component_with_ports(
424    wasm_bytes: &[u8],
425    prompt: &str,
426    ports: Box<dyn EmbedPorts>,
427    caps: &[tau_domain::Capability],
428    sandbox_root: &Path,
429) -> Result<String, WasmHostError> {
430    let cfg = resolve_wasi_config(caps);
431    let wasi = wasi_ctx_from_config(&cfg, sandbox_root)?;
432    let egress = EgressPolicy::from_config(&cfg);
433
434    let config = determinism_config().map_err(|e| WasmHostError::Instantiate(e.into()))?;
435    let engine = Engine::new(&config).map_err(|e| WasmHostError::Instantiate(e.into()))?;
436    let component =
437        Component::new(&engine, wasm_bytes).map_err(|e| WasmHostError::Load(e.into()))?;
438
439    let mut linker: Linker<HostState> = Linker::new(&engine);
440    wasmtime_wasi::p2::add_to_linker_sync(&mut linker)
441        .map_err(|e| WasmHostError::Instantiate(e.into()))?;
442    add_only_http_to_linker_sync(&mut linker).map_err(|e| WasmHostError::Instantiate(e.into()))?;
443    Runner::add_to_linker::<_, HasSelf<HostState>>(&mut linker, |state| state)
444        .map_err(|e| WasmHostError::Instantiate(e.into()))?;
445
446    let mut store = Store::new(&engine, HostState::new(ports, wasi, egress));
447    let runner = Runner::instantiate(&mut store, &component, &linker)
448        .map_err(|e| WasmHostError::Instantiate(e.into()))?;
449
450    match runner.call_run(&mut store, prompt) {
451        Ok(Ok(payload)) => Ok(payload),
452        Ok(Err(guest_err)) => Err(WasmHostError::Guest(guest_err)),
453        Err(trap) => Err(WasmHostError::Trap(trap.into())),
454    }
455}

457/// Determinism-conformance entry: no capabilities, so no WASI grants.
458/// Returns the same `(payload, emitted)` pair as [`run_component_with_caps`].
459pub fn run_component(
460    wasm_bytes: &[u8],
461    prompt: &str,
462    llm_responses: Vec<String>,
463) -> Result<(String, Vec<String>), WasmHostError> {
464    run_component_with_caps(wasm_bytes, prompt, llm_responses, &[], Path::new("."))
465}

569    fn preopen_with_dotdot_is_rejected_before_preopen() {
570        // Defense-in-depth: `resolve_wasi_config` never emits a `..` host_dir,
571        // but a hand-built config with one must be refused (a lexical
572        // component scan, since `join(..)` + `starts_with` would NOT catch it).
573        use tau_ports::target::{
574            PreopenAccess, PreopenGranularity, ResolvedPreopen, WasiConfiguration,
575        };

577            allowed_hosts: tau_domain::package::host::HostSet::Exact(Default::default()),
578            methods: None,
579            preopens: vec![ResolvedPreopen {
580                host_dir: "/../etc".to_string(),
581                access: PreopenAccess::ReadOnly,
582                granularity: PreopenGranularity::Exact,
583                from: vec![],
584            }],
585        };
586        let dir = tempfile::tempdir().expect("tempdir");
587        // Guard fires BEFORE any create_dir_all/preopen, so nothing is created
588        // outside the sandbox.

590        let existed_before = escaped.exists();
591        // `WasiCtx` isn't `Debug`, so match rather than `unwrap_err`.

593            Ok(_) => panic!("escaping preopen was accepted"),
594            Err(e) => e,
595        };
596        assert!(matches!(err, WasmHostError::WasiConfig(_)), "got: {err:?}");
597        assert_eq!(
598            escaped.exists(),
599            existed_before,
600            "guard must not create a dir outside the sandbox"
