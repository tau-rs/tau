1//! β.7.5 wasm host — a std [`wasmtime`] embedder for the `tau-wasm-guest`
2//! WASI 0.2 component.
3//!
4//! The guest (built for `wasm32-wasip2`, see the `tau-wasm-guest` crate)
5//! exports `run(prompt) -> result<string, string>` from the `tau:host/runner`
6//! world and imports three host ports it cannot satisfy in-wasm:
7//!
8//! - `complete(request-json) -> result<string, string>` — delegated
9//!   inference (credentials live host-side, β.5).
10//! - `now-millis() -> u64` — wall clock.
11//! - `next-u64() -> u64` — randomness.
12//!
13//! This crate satisfies those imports with **deterministic** stubs so the
14//! same `(component, prompt, llm_responses)` triple always yields the same
15//! bytes — the property β.6 conformance (`WasmProfile`) depends on. Time is
16//! a fixed-step counter, randomness is a seeded SplitMix64 PRNG, and
17//! inference replays a caller-supplied queue of canned `CompletionResponse`
18//! JSON strings (cassette-style).
19//!
20//! [`run_component`] is the whole public surface: feed it the guest bytes
21//! and it instantiates, drives `run`, and hands back the JSON string the
22//! guest returned.

54/// Deterministic step (ms) added to the clock on every `now-millis` call.
55const CLOCK_STEP_MILLIS: u64 = 1;
56/// Seed for the [`HostState`] PRNG. Fixed so randomness is reproducible.
57const PRNG_SEED: u64 = 0x9E37_79B9_7F4A_7C15;
58
59/// Errors surfaced by [`run_component`]. `thiserror` at the crate boundary;
60/// `wasmtime`/`serde_json` failures are wrapped, never leaked as `anyhow`.
61#[derive(Debug, thiserror::Error)]
62pub enum WasmHostError {
63    /// The supplied bytes are not a loadable component.
64    #[error("failed to load wasm component: {0}")]
65    Load(#[source] anyhow::Error),
66    /// Linking the host imports or instantiating the component failed.
67    #[error("failed to instantiate component: {0}")]
68    Instantiate(#[source] anyhow::Error),
69    /// The guest trapped (panicked / unreachable) while running `run`.
70    #[error("guest `run` trapped: {0}")]
71    Trap(#[source] anyhow::Error),
72    /// The guest's `run` completed but returned its error arm.
73    #[error("guest `run` returned an error: {0}")]
74    Guest(String),
75    /// A caller-supplied canned completion response is not valid
76    /// `CompletionResponse` JSON. Caught up-front so a malformed cassette
77    /// fails before instantiation rather than mid-run.
78    #[error("invalid canned completion response JSON: {0}")]
79    InvalidResponse(#[source] serde_json::Error),
80    /// Building the WASI context (e.g. a preopen dir failed to open, or a
81    /// resolved preopen would escape the sandbox root).
82    #[error("failed to configure WASI context: {0}")]
83    WasiConfig(#[source] anyhow::Error),
84}

86/// Store data backing the three `tau:host/host` imports with deterministic
87/// behaviour. One instance per [`run_component`] call.
88struct HostState {
89    /// Queue of canned `CompletionResponse` JSON strings, popped front-first
90    /// on each `complete` call. Empty queue → `complete` returns its error
91    /// arm so a guest that over-calls fails loudly rather than hangs.
92    responses: VecDeque<String>,
93    /// Monotonic millisecond counter; advances by [`CLOCK_STEP_MILLIS`].
94    clock_millis: u64,
95    /// SplitMix64 state, seeded from [`PRNG_SEED`].
96    prng_state: u64,
97    /// WASI 0.2 resource table (EPIC 3.3).
98    table: ResourceTable,
99    /// WASI 0.2 host context: exactly the preopens/network derived from the
100    /// component's allow-bounded caps (EPIC 3.3).
101    wasi: WasiCtx,
102    /// wasi:http resource bookkeeping (EPIC 3.3); the egress *decision* lives
103    /// in `egress`, not here.
104    http: WasiHttpCtx,
105    /// Network egress policy folded from the component's allow-bounded caps
106    /// (EPIC 3.3), sourced from the canonical `resolve_wasi_config`. Consulted
107    /// by the `WasiHttpHooks::send_request` override below before any outgoing
108    /// request is sent.
109    egress: EgressPolicy,
110}
111
112impl HostState {
113    fn new(responses: Vec<String>, wasi: WasiCtx, egress: EgressPolicy) -> Self {
114        Self {
115            responses: responses.into(),
116            clock_millis: 0,
117            prng_state: PRNG_SEED,
118            table: ResourceTable::new(),
119            wasi,
120            http: WasiHttpCtx::new(),
121            egress,
122        }
123    }
124}
125
126impl host::Host for HostState {
127    fn complete(&mut self, _request_json: String) -> Result<String, String> {
128        match self.responses.pop_front() {
129            Some(resp) => Ok(resp),
130            None => Err("tau-wasm-host: no canned completion response left".to_string()),
131        }
132    }
133
134    fn now_millis(&mut self) -> u64 {
135        let now = self.clock_millis;
136        self.clock_millis = self.clock_millis.wrapping_add(CLOCK_STEP_MILLIS);
137        now
138    }
139
140    fn next_u64(&mut self) -> u64 {
141        // SplitMix64 — small, fast, fully deterministic from a seed.
142        self.prng_state = self.prng_state.wrapping_add(0x9E37_79B9_7F4A_7C15);
143        let mut z = self.prng_state;
144        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
145        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
146        z ^ (z >> 31)
147    }
148}

172/// The egress gate: every `wasi:http` outgoing request is routed through
173/// `WasiHttpHooks::send_request` before wasmtime opens a socket. An authority
174/// or HTTP method that the allow-bounded caps didn't authorize is rejected
175/// here — the guest never gets a connection to it.
176impl WasiHttpHooks for EgressPolicy {
177    fn send_request(
178        &mut self,
179        request: hyper::Request<HyperOutgoingBody>,
180        config: OutgoingRequestConfig,
181    ) -> HttpResult<HostFutureIncomingResponse> {
182        let authority = request
183            .uri()
184            .authority()
185            .map(|a| a.as_str().to_string())
186            .unwrap_or_default();
187        let method = request.method().as_str();
188        if !self.permits(&authority, method) {
189            return Err(WasiHttpErrorCode::HttpRequestDenied.into());
190        }
191        Ok(default_send_request(request, config))
192    }
193}

195/// Build a `WasiCtx` that preopens exactly `cfg.preopens` (each resolved
196/// under `sandbox_root`) and denies all network egress by default (wasi:http
197/// egress is gated separately by the [`EgressPolicy`] filter; raw
198/// `wasi:sockets` stays default-deny).
199///
200/// `resolve_wasi_config` already dropped any non-G2 / `..`-bearing path, so
201/// every `host_dir` is absolute and escape-free; the `starts_with` check is
202/// defense-in-depth against a future resolver regression.
203fn wasi_ctx_from_config(
204    cfg: &WasiConfiguration,
205    sandbox_root: &Path,
206) -> Result<WasiCtx, WasmHostError> {
207    let mut builder = WasiCtxBuilder::new();
208    for p in &cfg.preopens {
209        // Defense-in-depth: `resolve_wasi_config` already drops non-G2 /
210        // `..`-bearing paths, but this is a public API over arbitrary caps, so
211        // re-check lexically. Scan segments explicitly — a component-wise
212        // `PathBuf::starts_with(sandbox_root)` AFTER `join` does NOT catch
213        // `..` (e.g. `<root>/../x` still `starts_with` `<root>`), so it would
214        // be a false guard.
215        if !p.host_dir.starts_with('/') || p.host_dir.split('/').any(|seg| seg == "..") {
216            return Err(WasmHostError::WasiConfig(anyhow::anyhow!(
217                "unsafe preopen path (not absolute or contains `..`): {}",
218                p.host_dir
219            )));
220        }
221        // Map the guest-visible absolute `host_dir` under the sandbox root.
222        let host_path = sandbox_root.join(p.host_dir.trim_start_matches('/'));
223        // Ensure the host dir exists so preopen succeeds.
224        std::fs::create_dir_all(&host_path).map_err(|e| WasmHostError::WasiConfig(e.into()))?;
225        let (dir_perms, file_perms) = match p.access {
226            PreopenAccess::ReadOnly => (DirPerms::READ, FilePerms::READ),
227            PreopenAccess::ReadWrite => (DirPerms::all(), FilePerms::all()),
228        };
229        builder
230            .preopened_dir(&host_path, &p.host_dir, dir_perms, file_perms)
231            .map_err(|e| WasmHostError::WasiConfig(e.into()))?;
232    }
233    Ok(builder.build())
234}

236/// Build the determinism `wasmtime::Config` (spec §7): canonicalise NaNs and
237/// turn off the nondeterministic relaxed-SIMD lowerings so float/SIMD codegen
238/// is bit-stable across hosts.
239fn determinism_config() -> wasmtime::Result<Config> {
240    let mut config = Config::new();
241    config.cranelift_nan_canonicalization(true);
242    config.wasm_relaxed_simd(false);
243    Ok(config)
244}

246/// Load `wasm_bytes` as a component, satisfy the three `tau:host/host` imports
247/// with deterministic stubs, instantiate it under a determinism `Config`,
248/// and drive the exported `run(prompt)`.
249///
250/// `llm_responses` is the cassette: each `complete` call pops the next entry
251/// (validated as `CompletionResponse` JSON up-front). Returns the JSON string
252/// the guest's `run` produced on its `ok` arm.
253///
254/// Determinism contract: for fixed inputs the returned bytes are identical
255/// across calls and hosts — the property `WasmProfile` conformance relies on.
256/// EPIC 3.3: run a component whose WASI authority is bounded by `caps`.
257/// Filesystem globs resolve to preopens under `sandbox_root`; network egress
258/// is denied unless a `net.http` cap authorizes the target host.
259pub fn run_component_with_caps(
260    wasm_bytes: &[u8],
261    prompt: &str,
262    llm_responses: Vec<String>,
263    caps: &[tau_domain::Capability],
264    sandbox_root: &Path,
265) -> Result<String, WasmHostError> {
266    // Fail fast on a malformed cassette before touching wasmtime.
267    for resp in &llm_responses {
268        serde_json::from_str::<CompletionResponse>(resp).map_err(WasmHostError::InvalidResponse)?;
269    }
270
271    // Consume the canonical no_std cap→config fold (tau-ports, PR #533); the
272    // host only translates its output into a wasmtime `WasiCtx` + egress gate.
273    let cfg = resolve_wasi_config(caps);
274    let wasi = wasi_ctx_from_config(&cfg, sandbox_root)?;
275    let egress = EgressPolicy::from_config(&cfg);
276
277    let config = determinism_config().map_err(|e| WasmHostError::Instantiate(e.into()))?;
278    let engine = Engine::new(&config).map_err(|e| WasmHostError::Instantiate(e.into()))?;
279    let component =
280        Component::new(&engine, wasm_bytes).map_err(|e| WasmHostError::Load(e.into()))?;
281
282    let mut linker: Linker<HostState> = Linker::new(&engine);
283    wasmtime_wasi::p2::add_to_linker_sync(&mut linker)
284        .map_err(|e| WasmHostError::Instantiate(e.into()))?;
285    add_only_http_to_linker_sync(&mut linker).map_err(|e| WasmHostError::Instantiate(e.into()))?;
286    Runner::add_to_linker::<_, HasSelf<HostState>>(&mut linker, |state| state)
287        .map_err(|e| WasmHostError::Instantiate(e.into()))?;
288
289    let mut store = Store::new(&engine, HostState::new(llm_responses, wasi, egress));
290    let runner = Runner::instantiate(&mut store, &component, &linker)
291        .map_err(|e| WasmHostError::Instantiate(e.into()))?;
292
293    match runner.call_run(&mut store, prompt) {
294        Ok(Ok(payload)) => Ok(payload),
295        Ok(Err(guest_err)) => Err(WasmHostError::Guest(guest_err)),
296        Err(trap) => Err(WasmHostError::Trap(trap.into())),
297    }
298}

300/// Determinism-conformance entry: no capabilities, so no WASI grants.
301pub fn run_component(
302    wasm_bytes: &[u8],
303    prompt: &str,
304    llm_responses: Vec<String>,
305) -> Result<String, WasmHostError> {
306    run_component_with_caps(wasm_bytes, prompt, llm_responses, &[], Path::new("."))
307}

314    fn canned_response() -> String {
315        // Minimal valid CompletionResponse JSON for cassette validation.

319    /// A `WasiCtx` with no preopens and no network — the no-caps baseline.

324    /// The deny-all egress policy the no-caps path produces (empty cap set →

327        let no_caps: [tau_domain::Capability; 0] = [];
328        EgressPolicy::from_config(&resolve_wasi_config(&no_caps))

331    #[test]
332    fn clock_advances_by_fixed_step() {
333        let mut state = HostState::new(vec![], empty_wasi_ctx(), empty_egress());
334        assert_eq!(state.now_millis(), 0);
335        assert_eq!(state.now_millis(), CLOCK_STEP_MILLIS);
336        assert_eq!(state.now_millis(), CLOCK_STEP_MILLIS * 2);
337    }

339    #[test]
340    fn prng_is_deterministic_and_seeded() {
341        let mut a = HostState::new(vec![], empty_wasi_ctx(), empty_egress());
342        let mut b = HostState::new(vec![], empty_wasi_ctx(), empty_egress());
343        let seq_a: Vec<u64> = (0..4).map(|_| a.next_u64()).collect();
344        let seq_b: Vec<u64> = (0..4).map(|_| b.next_u64()).collect();
345        assert_eq!(seq_a, seq_b, "same seed must yield same sequence");
346        assert_ne!(seq_a[0], seq_a[1], "PRNG must actually advance");
347    }

391    #[test]
392    fn preopen_with_dotdot_is_rejected_before_preopen() {
393        // Defense-in-depth: `resolve_wasi_config` never emits a `..` host_dir,
394        // but a hand-built config with one must be refused (a lexical
395        // component scan, since `join(..)` + `starts_with` would NOT catch it).
396        use tau_ports::target::{
397            PreopenAccess, PreopenGranularity, ResolvedPreopen, WasiConfiguration,
398        };
399        let cfg = WasiConfiguration {
400            allowed_hosts: tau_domain::package::host::HostSet::Exact(Default::default()),
401            methods: None,
402            preopens: vec![ResolvedPreopen {
403                host_dir: "/../etc".to_string(),
404                access: PreopenAccess::ReadOnly,
405                granularity: PreopenGranularity::Exact,
406                from: vec![],
407            }],
408        };
409        let dir = tempfile::tempdir().expect("tempdir");
410        // Guard fires BEFORE any create_dir_all/preopen, so nothing is created
411        // outside the sandbox.
412        let escaped = dir.path().parent().unwrap().join("etc");
413        let existed_before = escaped.exists();
414        // `WasiCtx` isn't `Debug`, so match rather than `unwrap_err`.
415        let err = match wasi_ctx_from_config(&cfg, dir.path()) {
416            Ok(_) => panic!("escaping preopen was accepted"),
417            Err(e) => e,
418        };
419        assert!(matches!(err, WasmHostError::WasiConfig(_)), "got: {err:?}");
420        assert_eq!(
421            escaped.exists(),
422            existed_before,
423            "guard must not create a dir outside the sandbox"
424        );
425    }
