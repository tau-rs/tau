1//! β.7.5 wasm host — a std [`wasmtime`] embedder for the `tau-wasm-guest`
2//! WASI 0.2 component.
3//!
4//! The guest (built for `wasm32-wasip2`, see the `tau-wasm-guest` crate)
5//! exports `run(prompt) -> result<string, string>` from the `tau:host/runner`
6//! world and imports three host ports it cannot satisfy in-wasm:
7//!
8//! - `complete(request-json) -> result<string, string>` — delegated
9//!   inference (credentials live host-side, β.5).
10//! - `now-millis() -> u64` — wall clock.
11//! - `next-u64() -> u64` — randomness.
12//!
13//! This crate satisfies those imports with **deterministic** stubs so the
14//! same `(component, prompt, llm_responses)` triple always yields the same
15//! bytes — the property β.6 conformance (`WasmProfile`) depends on. Time is
16//! a fixed-step counter, randomness is a seeded SplitMix64 PRNG, and
17//! inference replays a caller-supplied queue of canned `CompletionResponse`
18//! JSON strings (cassette-style).
19//!
20//! [`run_component`] is the whole public surface: feed it the guest bytes
21//! and it instantiates, drives `run`, and hands back the JSON string the
22//! guest returned.
23

37/// Deterministic step (ms) added to the clock on every `now-millis` call.
38const CLOCK_STEP_MILLIS: u64 = 1;
39/// Seed for the [`HostState`] PRNG. Fixed so randomness is reproducible.
40const PRNG_SEED: u64 = 0x9E37_79B9_7F4A_7C15;
41
42/// Errors surfaced by [`run_component`]. `thiserror` at the crate boundary;
43/// `wasmtime`/`serde_json` failures are wrapped, never leaked as `anyhow`.
44#[derive(Debug, thiserror::Error)]
45pub enum WasmHostError {
46    /// The supplied bytes are not a loadable component.
47    #[error("failed to load wasm component: {0}")]
48    Load(#[source] anyhow::Error),
49    /// Linking the host imports or instantiating the component failed.
50    #[error("failed to instantiate component: {0}")]
51    Instantiate(#[source] anyhow::Error),
52    /// The guest trapped (panicked / unreachable) while running `run`.
53    #[error("guest `run` trapped: {0}")]
54    Trap(#[source] anyhow::Error),
55    /// The guest's `run` completed but returned its error arm.
56    #[error("guest `run` returned an error: {0}")]
57    Guest(String),
58    /// A caller-supplied canned completion response is not valid
59    /// `CompletionResponse` JSON. Caught up-front so a malformed cassette
60    /// fails before instantiation rather than mid-run.
61    #[error("invalid canned completion response JSON: {0}")]
62    InvalidResponse(#[source] serde_json::Error),
63}
64
65/// Store data backing the three `tau:host/host` imports with deterministic
66/// behaviour. One instance per [`run_component`] call.
67struct HostState {
68    /// Queue of canned `CompletionResponse` JSON strings, popped front-first
69    /// on each `complete` call. Empty queue → `complete` returns its error
70    /// arm so a guest that over-calls fails loudly rather than hangs.
71    responses: VecDeque<String>,
72    /// Monotonic millisecond counter; advances by [`CLOCK_STEP_MILLIS`].
73    clock_millis: u64,
74    /// SplitMix64 state, seeded from [`PRNG_SEED`].
75    prng_state: u64,
76}
77
78impl HostState {
79    fn new(responses: Vec<String>) -> Self {
80        Self {
81            responses: responses.into(),
82            clock_millis: 0,
83            prng_state: PRNG_SEED,
84        }
85    }
86}
87
88impl host::Host for HostState {
89    fn complete(&mut self, _request_json: String) -> Result<String, String> {
90        match self.responses.pop_front() {
91            Some(resp) => Ok(resp),
92            None => Err("tau-wasm-host: no canned completion response left".to_string()),
93        }
94    }
95
96    fn now_millis(&mut self) -> u64 {
97        let now = self.clock_millis;
98        self.clock_millis = self.clock_millis.wrapping_add(CLOCK_STEP_MILLIS);
99        now
100    }
101
102    fn next_u64(&mut self) -> u64 {
103        // SplitMix64 — small, fast, fully deterministic from a seed.
104        self.prng_state = self.prng_state.wrapping_add(0x9E37_79B9_7F4A_7C15);
105        let mut z = self.prng_state;
106        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
107        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
108        z ^ (z >> 31)
109    }
110}

112/// Build the determinism `wasmtime::Config` (spec §7): canonicalise NaNs and
113/// turn off the nondeterministic relaxed-SIMD lowerings so float/SIMD codegen
114/// is bit-stable across hosts.
115fn determinism_config() -> wasmtime::Result<Config> {
116    let mut config = Config::new();
117    config.cranelift_nan_canonicalization(true);
118    config.wasm_relaxed_simd(false);
119    Ok(config)
120}

122/// Load `wasm_bytes` as a component, satisfy the three `tau:host/host` imports
123/// with deterministic stubs, instantiate it under a determinism `Config`,
124/// and drive the exported `run(prompt)`.
125///
126/// `llm_responses` is the cassette: each `complete` call pops the next entry
127/// (validated as `CompletionResponse` JSON up-front). Returns the JSON string
128/// the guest's `run` produced on its `ok` arm.
129///
130/// Determinism contract: for fixed inputs the returned bytes are identical
131/// across calls and hosts — the property `WasmProfile` conformance relies on.
132pub fn run_component(
133    wasm_bytes: &[u8],
134    prompt: &str,
135    llm_responses: Vec<String>,
136) -> Result<String, WasmHostError> {
137    // Fail fast on a malformed cassette before touching wasmtime.
138    for resp in &llm_responses {
139        serde_json::from_str::<CompletionResponse>(resp).map_err(WasmHostError::InvalidResponse)?;
140    }
141
142    let config = determinism_config().map_err(|e| WasmHostError::Instantiate(e.into()))?;
143    let engine = Engine::new(&config).map_err(|e| WasmHostError::Instantiate(e.into()))?;
144    let component =
145        Component::new(&engine, wasm_bytes).map_err(|e| WasmHostError::Load(e.into()))?;
146
147    let mut linker: Linker<HostState> = Linker::new(&engine);
148    Runner::add_to_linker::<_, HasSelf<HostState>>(&mut linker, |state| state)
149        .map_err(|e| WasmHostError::Instantiate(e.into()))?;
150
151    let mut store = Store::new(&engine, HostState::new(llm_responses));
152    let runner = Runner::instantiate(&mut store, &component, &linker)
153        .map_err(|e| WasmHostError::Instantiate(e.into()))?;
154
155    match runner.call_run(&mut store, prompt) {
156        Ok(Ok(payload)) => Ok(payload),
157        Ok(Err(guest_err)) => Err(WasmHostError::Guest(guest_err)),
158        Err(trap) => Err(WasmHostError::Trap(trap.into())),
159    }
160}

162#[cfg(test)]
163mod tests {
164    use super::*;
165    use host::Host as _;
166
167    fn canned_response() -> String {
168        // Minimal valid CompletionResponse JSON for cassette validation.
169        r#"{"text":"","tool_uses":[],"stop_reason":"EndTurn","usage":null}"#.to_string()
170    }
171
172    #[test]
173    fn clock_advances_by_fixed_step() {
174        let mut state = HostState::new(vec![]);
175        assert_eq!(state.now_millis(), 0);
176        assert_eq!(state.now_millis(), CLOCK_STEP_MILLIS);
177        assert_eq!(state.now_millis(), CLOCK_STEP_MILLIS * 2);
178    }
179
180    #[test]
181    fn prng_is_deterministic_and_seeded() {
182        let mut a = HostState::new(vec![]);
183        let mut b = HostState::new(vec![]);
184        let seq_a: Vec<u64> = (0..4).map(|_| a.next_u64()).collect();
185        let seq_b: Vec<u64> = (0..4).map(|_| b.next_u64()).collect();
186        assert_eq!(seq_a, seq_b, "same seed must yield same sequence");
187        assert_ne!(seq_a[0], seq_a[1], "PRNG must actually advance");
188    }
189
190    #[test]
191    fn complete_pops_responses_then_errors() {
192        let mut state = HostState::new(vec!["first".to_string(), "second".to_string()]);
193        assert_eq!(state.complete(String::new()), Ok("first".to_string()));
194        assert_eq!(state.complete(String::new()), Ok("second".to_string()));
195        assert!(
196            state.complete(String::new()).is_err(),
197            "exhausted queue errors"
198        );
199    }
200
201    #[test]
202    fn malformed_cassette_rejected_before_wasm() {
203        let err = run_component(&[], "p", vec!["not json".to_string()]).unwrap_err();
204        assert!(matches!(err, WasmHostError::InvalidResponse(_)));
205    }
206
207    #[test]
208    fn well_formed_cassette_passes_validation_then_fails_at_load() {
209        // Empty bytes are a valid cassette but not a loadable component:
210        // proves validation precedes the wasmtime load path.
211        let err = run_component(&[], "p", vec![canned_response()]).unwrap_err();
212        assert!(matches!(err, WasmHostError::Load(_)), "got: {err:?}");
213    }
214}
