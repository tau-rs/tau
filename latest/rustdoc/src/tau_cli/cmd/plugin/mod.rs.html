1//! `tau plugin {describe, run, protocol decode}` subcommands.
2//!
3//! Spec §9 (debug tier): these handlers expose the plugin protocol to
4//! human operators for inspection, REPL-style probing, and post-hoc
5//! analysis of recorded sessions.
6//!
7//! - [`describe`] resolves a plugin from the lockfile, drives one
8//!   handshake, prints the advertised metadata + per-method schemas,
9//!   then shuts the plugin down cleanly.
10//! - [`run`] spawns an arbitrary plugin binary and drives either an
11//!   interactive REPL (`<method> <json-args>` per line) or a scripted
12//!   JSONL input file. Standalone — no agent, no kernel.
13//! - [`protocol_decode`] reads a JSONL recording produced by
14//!   `--record-protocol` and emits a human-readable transcript with
15//!   filter/time-range/JSON projection options.
16
17pub mod describe;
18pub mod protocol_decode;
19pub mod run;

26/// Dispatch a `tau plugin <action>` subcommand.
27///
28/// `record_protocol` is the global `--record-protocol` flag; it only
29/// affects [`describe`] (which spawns a plugin) — neither
30/// [`protocol_decode`] (offline) nor [`run`] (which manages its own
31/// process IO directly) consult it.
32pub async fn dispatch(
33    action: PluginAction,
34    record_protocol: Option<PathBuf>,
35    output: &mut Output,
36) -> anyhow::Result<()> {
37    match action {
38        PluginAction::Describe(args) => describe::run(&args, record_protocol, output).await,
39        PluginAction::Run(args) => run::run(&args, output).await,
40        PluginAction::Protocol { action } => match action {
41            PluginProtocolAction::Decode(args) => protocol_decode::run(&args, output).await,
42        },
43    }
44}
