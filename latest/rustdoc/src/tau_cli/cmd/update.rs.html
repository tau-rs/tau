1//! `tau update` — update an installed package to a newer or specific version.
2//!
3//! Per spec §2:
4//!
5//! - Resolves the active [`Scope`] (project or global).
6//! - Parses the package name and optional version pin.
7//! - Delegates to [`tau_pkg::update::update_package`].
8//! - Emits a result-summary after the synchronous call returns:
9//!   the library function runs end-to-end and returns an `UpdateResult`;
10//!   intermediate "installing…" progress events are not available without
11//!   library hooks, so output is post-call summary only.
12//!
13//! Exit codes per ADR-0007 §7:
14//! - 0: success.
15//! - 2: any `UpdateError` (maps through `anyhow` to `ExitCode::Error`).
16
17use std::str::FromStr;
18
19use semver::Version;
20use tau_domain::PackageName;
21use tau_pkg::{update::update_package, LockFile, Scope};
22
23use crate::cli::UpdateArgs;
24use crate::output::Output;
25
26/// Run `tau update`.
27pub async fn run(args: &UpdateArgs, output: &mut Output) -> anyhow::Result<()> {
28    // 1. Resolve scope.
29    let scope = if args.global {
30        Scope::global()?
31    } else {
32        let cwd = std::env::current_dir()?;
33        Scope::resolve(&cwd)?
34    };
35
36    // 2. Parse package name.
37    let name = PackageName::from_str(&args.package)
38        .map_err(|e| anyhow::anyhow!("invalid package name {:?}: {}", args.package, e))?;
39
40    // 3. Parse optional version pin.
41    let version_pin: Option<Version> = args
42        .version
43        .as_deref()
44        .map(|v| Version::parse(v).map_err(|e| anyhow::anyhow!("invalid version {:?}: {}", v, e)))
45        .transpose()?;
46
47    // 4. Look up the current active version from the lockfile (needed for
48    //    JSON update_started event emitted before calling the library).
49    let current_version: Option<String> = if scope.lockfile_path().exists() {
50        LockFile::load(&scope.lockfile_path())
51            .ok()
52            .and_then(|lf| lf.find(&name).map(|p| p.active_version.to_string()))
53    } else {
54        None
55    };
56
57    // 5. JSON: emit update_started BEFORE calling the library.
58    if output.is_json() {
59        output.json(&serde_json::json!({
60            "event": "update_started",
61            "name": name.as_str(),
62            "current": current_version.as_deref().unwrap_or("unknown"),
63        }))?;
64    }
65
66    // 6. Delegate to tau_pkg::update_package.
67    let result = update_package(&name, version_pin, &scope, args.prune)
68        .map_err(|e| anyhow::anyhow!("{}", e))?;
69
70    // 7. Emit output.
71    let transitive: Vec<serde_json::Value> = result
72        .transitive_deps_changed
73        .iter()
74        .map(|(dep_name, dep_ver)| {
75            serde_json::json!({ "name": dep_name.as_str(), "version": dep_ver.to_string() })
76        })
77        .collect();
78
79    if output.is_json() {
80        output.json(&serde_json::json!({
81            "event": "update_completed",
82            "name": name.as_str(),
83            "from": result.from_version.to_string(),
84            "to": result.to_version.to_string(),
85            "pruned": args.prune,
86            "transitive_deps_changed": transitive,
87        }))?;
88    } else {
89        output.human(&format!(
90            "\u{2713} Updated {}: {} \u{2192} {}",
91            name.as_str(),
92            result.from_version,
93            result.to_version,
94        ))?;
95        output.human(&format!("  Active version: {}", result.to_version))?;
96        if args.prune {
97            output.human(&format!("  Pruned: {}", result.from_version))?;
98        }
99        if !result.transitive_deps_changed.is_empty() {
100            let deps_str: Vec<String> = result
101                .transitive_deps_changed
102                .iter()
103                .map(|(n, v)| format!("{}@{}", n.as_str(), v))
104                .collect();
105            output.human(&format!("  New transitive deps: {}", deps_str.join(", ")))?;
106        }
107    }
108
109    Ok(())
110}
