1//! `tau build wasm <project>` — IR-to-wasm AOT compiler (β.7.5).
2//!
3//! Pipeline: load + validate the project, lower its IR for `any-wasi-strict`
4//! (capability-fit refuses `process-exec`/`agent-spawn` here), serialize the
5//! canonical IR bytes, bake them into the `tau-wasm-guest` crate via the
6//! `TAU_IR_BYTES` env handshake, shell `cargo build … --target wasm32-wasip2`,
7//! and emit the produced `.wasm` plus its IR hash.
8//!
9//! The guest source lives in *this* workspace; the command shells cargo
10//! against it (located via `CARGO_MANIFEST_DIR`). Shipping the guest for a
11//! user-facing `tau run --wasm` is a γ concern.
12

24/// The triple every wasm build lowers for.
25const WASM_TARGET: &str = "any-wasi-strict";
26
27/// Load + validate the project and lower its IR for `any-wasi-strict`.
28///
29/// Returns the lowered module and its canonical bytes. Capability-fit is
30/// enforced inside `lower_project`; a `process-exec`/`agent-spawn` project
31/// surfaces `LowerError::CapabilityFitFailed` here (no bypass flag).
32///
33/// This function is also the entry point for the Task 4 e2e integration test
34/// so it must remain `pub`.
35pub fn lower_to_wasm_ir(project: &Path) -> Result<(tau_ir::IrModule, Vec<u8>)> {
36    let loaded = load_project(project)
37        .with_context(|| format!("loading project from {}", project.display()))?;
38
39    let target: tau_ports::target::TargetTriple = WASM_TARGET
40        .parse()
41        .expect("any-wasi-strict is a registered triple");
42    if tau_ports::target::lookup(&target).is_none() {
43        bail!("internal: target `{WASM_TARGET}` not found in the registry");
44    }
45
46    let caches = tau_ir_lower::Caches {
47        native_tool: &|name: &str| native_tool_hash(name),
48        mcp_contract: &|_url| None,
49        skill: &|_name| None,
50        prompt_file: &|p: &std::path::Path| {
51            tau_pkg::bundle::read_prompt_file(p, project)
52                .map_err(|e| tau_ir_lower::PromptFileError(e.to_string()))
53        },
54    };
55
56    let out =
57        tau_ir_lower::lower_project(&loaded.project, &target, &caches).map_err(|e| match e {
58            LowerError::CapabilityFitFailed {
59                ref missing,
60                ref tools,
61            } => anyhow::anyhow!(
62                "capability-fit refused for {WASM_TARGET}: \
63                 the following capability shape(s) are not supported by wasm guests: {missing:?}. \
64                 Offending tools: {tools:?}. \
65                 Remove or replace tools that require ProcessExec or AgentSpawn \
66                 before building for wasm.",
67            ),
68            other => anyhow::anyhow!("lowering for {WASM_TARGET} failed: {other}"),
69        })?;
70    // NOTE(D6-B PR3): embedding the content-addressed asset store into the
71    // wasm component (so the guest can resolve `PromptSource::Asset` prompts)
72    // lands with the wasm/WIT lane. Until then, warn rather than silently ship
73    // a wasm module whose prompt assets the guest cannot resolve.
74    if !out.assets.is_empty() {
75        tracing::warn!(
76            asset_count = out.assets.len(),
77            "wasm build: {WASM_TARGET} does not yet embed the prompt asset store (D6-B PR3); \
78             agents using `system_file` prompts will not resolve them in the wasm guest yet"
79        );
80    }
81    let module = out.module;
82    let bytes = tau_ir::to_canonical_bytes(&module);
83    Ok((module, bytes))
84}

86/// Locate the tau source workspace root at compile time.
87///
88/// The guest source lives in the same Cargo workspace. We locate the workspace
89/// root at compile time so the command works regardless of the working
90/// directory at runtime (e.g. inside a tmp project directory).
91fn workspace_root() -> PathBuf {
92    PathBuf::from(env!("CARGO_MANIFEST_DIR"))
93        .parent()
94        .and_then(Path::parent)
95        .expect("tau-cli is two levels below the workspace root")
96        .to_path_buf()
97}

99/// Shell `cargo build -p tau-wasm-guest` with the baked IR and return the
100/// produced `.wasm` bytes.
101///
102/// The IR path is injected via `TAU_IR_BYTES` env var. A dedicated
103/// `CARGO_TARGET_DIR` (`<workspace>/target/tau-build-wasm`) ensures this
104/// build never contends with the main agent's target dir (CLAUDE.md Rule 1).
105fn build_guest_with_ir(ir_path: &Path) -> Result<Vec<u8>> {
106    let root = workspace_root();
107    let target_dir = root.join("target/tau-build-wasm");
108
109    let output = Command::new(env!("CARGO"))
110        .current_dir(&root)
111        .args([
112            "build",
113            "-p",
114            "tau-wasm-guest",
115            "--target",
116            "wasm32-wasip2",
117            "--release",
118            "--message-format=json",
119        ])
120        .env("CARGO_INCREMENTAL", "0")
121        .env("CARGO_TARGET_DIR", &target_dir)
122        .env("TAU_IR_BYTES", ir_path)
123        .output()
124        .context("failed to spawn cargo for the guest build")?;
125
126    if !output.status.success() {
127        bail!(
128            "guest build failed (is wasm32-wasip2 installed?):\n{}",
129            String::from_utf8_lossy(&output.stderr)
130        );
131    }
132
133    let stdout = String::from_utf8(output.stdout).context("cargo json output is utf-8")?;
134    let wasm_path = stdout
135        .lines()
136        .filter_map(|l| serde_json::from_str::<serde_json::Value>(l).ok())
137        .filter(|m| m["reason"] == "compiler-artifact")
138        .filter(|m| {
139            m["target"]["name"]
140                .as_str()
141                .is_some_and(|n| n == "tau-wasm-guest" || n == "tau_wasm_guest")
142        })
143        .flat_map(|m| {
144            m["filenames"]
145                .as_array()
146                .into_iter()
147                .flatten()
148                .filter_map(|f| f.as_str().map(str::to_string))
149                .collect::<Vec<_>>()
150        })
151        .find(|f| f.ends_with(".wasm"))
152        .context("no .wasm artifact in cargo output for tau-wasm-guest")?;
153
154    std::fs::read(&wasm_path).with_context(|| format!("reading built component {wasm_path}"))
155}

157/// CLI entry point for `tau build wasm`.
158pub async fn run(args: &BuildWasmArgs, output: &mut Output) -> Result<()> {
159    let project = args
160        .project
161        .clone()
162        .unwrap_or_else(|| std::env::current_dir().expect("cwd is readable"));
163
164    let (module, bytes) = lower_to_wasm_ir(&project)?;
165    let ir_hash = hex_lower(&tau_ir::compute_hash(&module));
166
167    // Bake the IR bytes into a tempfile the guest build reads via TAU_IR_BYTES.
168    // The file must stay alive until after the cargo call returns.
169    let ir_file = tempfile::NamedTempFile::new().context("creating IR scratch file")?;
170    std::fs::write(ir_file.path(), &bytes).context("writing IR scratch bytes")?;
171
172    let wasm = build_guest_with_ir(ir_file.path())?;
173    drop(ir_file); // bytes are consumed; safe to remove the scratch file now.
174
175    let out_path = args
176        .output
177        .clone()
178        .unwrap_or_else(|| project.join(format!("{}.wasm", project_stem(&project))));
179    std::fs::write(&out_path, &wasm).with_context(|| format!("writing {}", out_path.display()))?;
180
181    let _ = output.human(&format!(
182        "built {} ({} bytes, ir {})",
183        out_path.display(),
184        wasm.len(),
185        ir_hash
186    ));
187    Ok(())
188}

190fn project_stem(project: &Path) -> String {
191    project
192        .file_name()
193        .and_then(|n| n.to_str())
194        .unwrap_or("project")
195        .to_string()
