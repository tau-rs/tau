1//! `tau install` — install a package from a git URL.
2//!
3//! Per spec §3.13 install row:
4//!
5//! - Parses the URL into a [`PackageSource`].
6//! - Resolves the active [`Scope`] (project or global).
7//! - For `--dry-run`: clones the source into a [`tempfile::TempDir`],
8//!   parses the manifest, prints a preview of what would be installed,
9//!   and writes nothing to the scope.
10//! - Otherwise: delegates to [`tau_pkg::install_with_options`] and
11//!   prints either a one-line summary or, when `--json` is set, a
12//!   structured payload.
13//!
14//! Errors propagate via `anyhow::Error` and exit 2 through the
15//! top-level `dispatch` in `crate::run_main`.
16

33/// Run `tau install`.
34pub async fn run(args: &InstallArgs, output: &mut Output) -> anyhow::Result<()> {
35    // 1. Parse URL into a PackageSource.
36    let source = PackageSource::from_str(&args.url)
37        .map_err(|e| anyhow::anyhow!("invalid package URL {:?}: {}", args.url, e))?;
38
39    // 2. Resolve scope.
40    let scope = if args.global {
41        Scope::global()?
42    } else {
43        let cwd = std::env::current_dir()?;
44        Scope::resolve(&cwd)?
45    };
46
47    // 3. Dry-run path: clone to tmp, parse manifest, print preview.
48    if args.dry_run {
49        let preview = preview_install(&source, &scope)?;
50        emit_dry_run(&preview, output)?;
51        return Ok(());
52    }
53
54    // 4. Real install. Resolve the host's sandbox adapter and inject it so the
55    //    install-time `cargo build` + cross-check run sandboxed (audit S2).
56    //    The build path fails closed unless --allow-unsandboxed-build is set.
57    output.status(format!("installing from {}...", args.url))?;
58
59    // The strict-tier egress proxy task spawned by `wrap_spawn` runs on this
60    // (the ambient) runtime; the gate's guard aborts it on drop. We pass a
61    // `Handle` rather than owning a `Runtime` so nothing is dropped in async
62    // context (which would panic in tokio's blocking shutdown).
63    let handle = tokio::runtime::Handle::current();
64    let requirements = tau_pkg::scope::SandboxRequirements::default();
65    let adapter = resolve_adapter(&requirements, &[])
66        .await
67        .map_err(|e| anyhow::anyhow!("resolving install sandbox adapter: {e}"))?;
68    let enforced = matches!(
69        adapter.probe().await,
70        CapabilityProbe::Available { tier, .. } if tier > CapabilityTier::None
71    );
72    let gate = RuntimeInstallSandbox::new(adapter, handle, enforced);
73
74    let mut options = InstallOptions::default();
75    options.sandbox = Some(std::sync::Arc::new(gate));
76    options.allow_unsandboxed_build = args.allow_unsandboxed_build;
77
78    let installed = install_with_options(&source, &scope, options)
79        .map_err(|e| anyhow::anyhow!("install failed: {e}"))?;
80
81    if output.is_json() {
82        let payload = serde_json::json!({
83            "name": installed.name.as_str(),
84            "version": installed.version.to_string(),
85            "scope": scope_kind_str(&scope),
86            "path": installed.installed_path,
87        });
88        output.json(&payload)?;
89    } else {
90        output.human(&format!(
91            "installed: {} {} ({})",
92            installed.name,
93            installed.version,
94            scope_kind_str(&scope),
95        ))?;
96    }
97
98    Ok(())
99}

101/// Return `"global"` or `"project"` for the current scope's kind.
102fn scope_kind_str(scope: &Scope) -> &'static str {
103    match scope.kind() {
104        ScopeKind::Global => "global",
105        ScopeKind::Project => "project",
106        // ScopeKind is `#[non_exhaustive]`; cover unknown future variants.
107        _ => "unknown",
108    }
109}

111/// Preview shape used for both human and JSON dry-run output.
112struct DryRunPreview {
113    name: String,
114    version: String,
115    url: String,
116    target_path: PathBuf,
117    kind: String,
118    capabilities: Vec<String>,
119    dependencies: Vec<String>,
120}
121
122/// Clone the source into a temp dir, parse the manifest, and produce the
123/// preview that the dry-run path emits. The temp dir auto-removes on drop,
124/// so the dry-run leaves no on-disk state behind.
125fn preview_install(source: &PackageSource, scope: &Scope) -> anyhow::Result<DryRunPreview> {
126    let tmp_dir = tempfile::TempDir::new()?;
127
128    // Render the source's URL/scp string for the clone subprocess. v0.1
129    // PackageSource has only the Git variant, but it is `#[non_exhaustive]`,
130    // so we surface a clear error for hypothetical future variants.
131    let (location_str, rev_opt) = match source {
132        PackageSource::Git { location, rev } => (location.to_string(), rev.clone()),
133        other => {
134            anyhow::bail!("unsupported package source for dry-run: {other:?}");
135        }
136    };
137
138    // Shell out to `git` directly. We mirror tau-pkg's clone flags
139    // (protocol.file.allow=always for file:// fixtures + `--branch <rev>
140    // --single-branch` for revision pins) so dry-run accepts the same
141    // sources as the real install path.
142    let dest = tmp_dir.path();
143    let mut cmd = Command::new("git");
144    cmd.env("GIT_CONFIG_COUNT", "1")
145        .env("GIT_CONFIG_KEY_0", "protocol.file.allow")
146        .env("GIT_CONFIG_VALUE_0", "always");
147    cmd.arg("clone");
148    if let Some(rev) = &rev_opt {
149        cmd.arg("--branch").arg(rev).arg("--single-branch");
150    }
151    cmd.arg(&location_str).arg(dest);
152
153    let output = cmd
154        .output()
155        .map_err(|e| anyhow::anyhow!("spawning `git clone`: {e}"))?;
156    if !output.status.success() {
157        anyhow::bail!(
158            "git clone failed (exit {}): {}",
159            output.status.code().unwrap_or(-1),
160            String::from_utf8_lossy(&output.stderr).trim(),
161        );
162    }
163
164    let manifest = read_manifest(&dest.join("tau.toml"))
165        .map_err(|e| anyhow::anyhow!("reading cloned manifest: {e}"))?;
166
167    let target_path = scope.package_dir(manifest.name(), manifest.version());
168
169    Ok(DryRunPreview {
170        name: manifest.name().as_str().to_owned(),
171        version: manifest.version().to_string(),
172        url: source.to_string(),
173        target_path,
174        kind: kind_str(manifest.kind()).to_owned(),
175        capabilities: manifest
176            .capabilities()
177            .iter()
178            .map(capability_kind_str)
179            .collect(),
180        dependencies: dep_strings(&manifest),
181    })
182}

184/// Render a [`PackageKind`] as its canonical kind string (e.g. `"tool"`,
185/// `"llm-backend"`). v0.1 ships only `Custom { kind }`, but the type is
186/// `#[non_exhaustive]`; future typed variants fall back to `Debug`.
187fn kind_str(kind: &PackageKind) -> &str {
188    match kind {
189        PackageKind::Custom { kind } => kind.as_str(),
190        // Forward-compat for typed variants added later.
191        _ => "unknown",
192    }
193}

195/// Render a [`Capability`] as its dot-namespaced kind string (matches the
196/// manifest `kind = "..."` form in [`Capability`]'s serde).
197fn capability_kind_str(cap: &Capability) -> String {
198    match cap {
199        Capability::Filesystem(FsCapability::Read { .. }) => "fs.read".into(),
200        Capability::Filesystem(FsCapability::Write { .. }) => "fs.write".into(),
201        Capability::Filesystem(FsCapability::Exec { .. }) => "fs.exec".into(),
202        Capability::Network(NetCapability::Http { .. }) => "net.http".into(),
203        Capability::Process(ProcessCapability::Spawn { .. }) => "process.spawn".into(),
204        Capability::Agent(AgentCapability::Spawn { .. }) => "agent.spawn".into(),
205        Capability::TaskList { .. } => "task_list".into(),
206        Capability::Plan { .. } => "plan".into(),
207        Capability::Custom { name, .. } => name.clone(),
208        // Forward-compat: future typed variants fall back to debug.
209        other => format!("{other:?}"),
210    }
211}

213/// Render dependencies as `"<name>@<version_req>"` strings.
214fn dep_strings(manifest: &PackageManifest) -> Vec<String> {
215    manifest
216        .dependencies()
217        .iter()
218        .map(|dep| format!("{}@{}", dep.name.as_str(), dep.version_req))
219        .collect()
220}

222/// Emit the dry-run preview to the user. Format per task spec:
223///
224/// ```text
225/// [dry-run] would install: <name> <version>
226/// [dry-run] from:          <url>
227/// [dry-run] to:            <target-path>
228/// [dry-run] kind:          <kind>
229/// [dry-run] capabilities:  <list> (or "none")
230/// [dry-run] dependencies:  <list> (or "none")
231/// [dry-run] no changes written.
232/// ```
233///
234/// With `--json`, emits a single JSON object on stdout instead.
235fn emit_dry_run(preview: &DryRunPreview, output: &mut Output) -> anyhow::Result<()> {
236    if output.is_json() {
237        let payload = serde_json::json!({
238            "dry_run": true,
239            "name": preview.name,
240            "version": preview.version,
241            "url": preview.url,
242            "target_path": preview.target_path,
243            "kind": preview.kind,
244            "capabilities": preview.capabilities,
245            "dependencies": preview.dependencies,
246        });
247        output.json(&payload)?;
248        return Ok(());
249    }
250
251    output.dry_run(format!(
252        "would install: {} {}",
253        preview.name, preview.version
254    ))?;
255    output.dry_run(format!("from:          {}", preview.url))?;
256    output.dry_run(format!("to:            {}", preview.target_path.display()))?;
257    output.dry_run(format!("kind:          {}", preview.kind))?;
258    let caps = if preview.capabilities.is_empty() {
259        "none".to_string()
260    } else {
261        preview.capabilities.join(", ")
262    };
263    output.dry_run(format!("capabilities:  {caps}"))?;
264    let deps = if preview.dependencies.is_empty() {
265        "none".to_string()
266    } else {
267        preview.dependencies.join(", ")
268    };
269    output.dry_run(format!("dependencies:  {deps}"))?;
270    output.dry_run("no changes written.")?;
271    Ok(())
272}
