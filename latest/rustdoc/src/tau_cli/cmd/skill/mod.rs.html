1//! `tau skill <subcommand>` — inspect installed skill packages.
2//!
3//! Skills-3 sub-project from ROADMAP §16. Two subcommands:
4//! - [`list`] — enumerate installed skills (lockfile-only).
5//! - [`show`] — detailed view of one skill, optionally including
6//!   the SKILL.md body (rendered or raw).
7//!
8//! Skills-5 adds two more:
9//! - [`import`] — convert an Anthropic-format source into a tau-skill dir.
10//! - [`export`] — strip tau.toml from an installed skill (Task 6 stub).
11//!
12//! See `docs/superpowers/specs/2026-05-13-skills-3-discovery-design.md`
13//! for the design and ADR-0027 for the decision rationale.
14

22use crate::cli::SkillSubcommand;
23use crate::output::Output;

25/// Route `tau skill <subcommand>` to its impl module.
26pub async fn dispatch(sub: SkillSubcommand, output: &mut Output) -> anyhow::Result<()> {
27    match sub {
28        SkillSubcommand::List(args) => list::run(&args, output).await,
29        SkillSubcommand::Show(args) => show::run(&args, output).await,
30        SkillSubcommand::Import(args) => {
31            import::run(args).map_err(|e| anyhow::anyhow!("{e}"))?;
32            Ok(())
33        }
34        SkillSubcommand::Export(args) => {
35            export::run(args).map_err(|e| anyhow::anyhow!("{e}"))?;
36            Ok(())
37        }
38    }
39}
