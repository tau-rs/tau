16/// Entry point for `tau dev`.
17///
18/// When `--prompt` (`-p`) is supplied, runs exactly one turn and exits
19/// (single-shot mode). When omitted, enters the interactive REPL loop.
20/// When `--watch` is set, the REPL auto-applies pending manifest reloads at
21/// the top of each iteration without requiring the user to type `:reload`.
22/// If both `-p` and `--watch` are set, `--watch` is a no-op (one-shot doesn't
23/// loop) and a warning is emitted.
24pub async fn run(args: DevArgs, output: &mut Output) -> Result<()> {
25    let mut session = DevSession::load(args.project, args.agent).await?;
26    match args.prompt {
27        Some(p) => {
28            // --watch is meaningless in one-shot mode; warn but proceed.
29            if args.watch {
30                eprintln!("warning: --watch is ignored when -p is set");
31            }
32            session.run_turn(&p, output).await
33        }
34        None => repl::run_loop(&mut session, output, args.watch).await,
35    }
36}
