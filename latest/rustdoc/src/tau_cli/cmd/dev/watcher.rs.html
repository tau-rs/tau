1//! File watcher for `tau dev` — wraps `notify::RecommendedWatcher`.
2//!
3//! Watches:
4//!   - `tau.toml`
5//!   - `workflows/*.toml`
6//!   - every external prompt file referenced by `[agents.X.prompt] system_file`
7//!   - every `[dirs]` root (`agents`/`tools`), recursively
8//!
9//! On any `Modify`, `Create`, or `Remove` event for a watched path,
10//! `pending_reload` is set to `true`. The caller must hold the returned
11//! [`notify::RecommendedWatcher`] alive — dropping it stops the watcher.
12
13use std::collections::HashSet;
14use std::path::{Path, PathBuf};
15use std::sync::atomic::{AtomicBool, AtomicU64, Ordering};
16use std::sync::Arc;
17
18use anyhow::{Context, Result};
19use notify::{EventKind, RecommendedWatcher, RecursiveMode, Watcher};
20use tau_pkg::project::project::{AgentEntry, ProjectConfig, PromptEntry};

22/// Spawn a watcher over the project's relevant paths.
23///
24/// Returns the watcher handle — the caller **MUST** hold it alive.
25/// Dropping it stops all file watching.
26///
27/// On any modify/create/remove event for a watched path,
28/// `pending_reload` is set to `true` (Acquire/Release ordering).
29///
30/// `project_file_path` is the original path supplied by the caller (may be
31/// a `.ts` file or a directory). When it ends in `.ts`, the watcher tracks
32/// the `.ts` file instead of `<root>/tau.toml`.
33pub fn spawn(
34    project_root: &Path,
35    project_file_path: &Path,
36    project: &ProjectConfig,
37    pending_reload: Arc<AtomicBool>,
38) -> Result<RecommendedWatcher> {
39    let paths = resolve_watch_paths(project_root, project_file_path, project);
40    let dir_roots = resolve_watch_dirs(project_root, project);
41
42    // The set of files we actually care about, canonicalized so the callback's
43    // path matching survives symlinked temp dirs (e.g. macOS `/var` ->
44    // `/private/var`). This is the crux of correct filtering: some notify
45    // backends (notably FSEvents) watch the *parent directory* of a watched
46    // file and deliver events for sibling files too. Without this filter a
47    // write to e.g. `tau-lock.toml` — a sibling of the watched `tau.toml` —
48    // would spuriously set `pending_reload`. Match each event's paths against
49    // this set so only genuine changes to watched files trigger a reload.
50    let watched: HashSet<PathBuf> = paths
51        .iter()
52        .map(|p| std::fs::canonicalize(p).unwrap_or_else(|_| p.clone()))
53        .collect();
54
55    // `[dirs]` roots, canonicalized the same way as `watched` above. These
56    // are matched by prefix rather than exact equality — any path under a
57    // watched root (including nested subdirectories and moved/renamed
58    // files, which notify reports as a remove+add pair) is in scope.
59    let watched_dirs: Vec<PathBuf> = dir_roots
60        .iter()
61        .map(|p| std::fs::canonicalize(p).unwrap_or_else(|_| p.clone()))
62        .collect();
63
64    // Content fingerprint of the watched files at boot. The watcher flips
65    // `pending_reload` only when an event *actually changes* watched content
66    // — not on metadata-only events (`Modify(Metadata)`), and not on the
67    // historical `Create`/`Modify` events that FSEvents replays for the
68    // watched file when the watch first registers. Those replayed events
69    // otherwise race the boot sequence and cause spurious reloads.
70    let last_hash = Arc::new(AtomicU64::new(hash_watched(&watched)));
71
72    // `[dirs]`-subtree analogue of `last_hash` above, for the same reason:
73    // FSEvents replays historical Create/Modify events for files that
74    // already existed under a root when the recursive watch registers, and
75    // under load (nextest parallelism) those replays race the boot
76    // sequence — see `hash_watched`'s doc comment / commit 6d1a8552 (#363),
77    // which fixed the identical race for the flat file watchlist. Real
78    // projects have pre-existing files under `agents/`/`tools/` at
79    // registration time, so this is the common case, not an edge case.
80    // Fingerprint is `(path, len, mtime)` per file, not file bytes: cheap
81    // enough to recompute on every dir event (a `stat`, not a `read`, per
82    // file) while still detecting adds/removes/edits (mtime and/or length
83    // change on write).
84    let last_dir_hash = Arc::new(AtomicU64::new(hash_watched_dirs(&watched_dirs)));
85
86    let pending_for_callback = pending_reload.clone();
87    let watched_for_callback = watched.clone();
88    let watched_dirs_for_callback = watched_dirs.clone();
89    let last_hash_for_callback = last_hash.clone();
90    let last_dir_hash_for_callback = last_dir_hash.clone();
91    let mut watcher = RecommendedWatcher::new(
92        move |res: std::result::Result<notify::Event, notify::Error>| {
93            let Ok(event) = res else { return };
94            if !matches!(
95                event.kind,
96                EventKind::Modify(_) | EventKind::Create(_) | EventKind::Remove(_)
97            ) {
98                return;
99            }
100            // Canonicalize first (matches FSEvents' `/private/…` paths),
101            // falling back to the raw path for removes (the file is gone,
102            // so canonicalize fails).
103            let canon_paths: Vec<PathBuf> = event
104                .paths
105                .iter()
106                .map(|p| std::fs::canonicalize(p).unwrap_or_else(|_| p.clone()))
107                .collect();
108
109            // Case 1: an exact watched file (tau.toml, workflows/*.toml,
110            // an external prompt file). Some backends (FSEvents) watch the
111            // parent directory and deliver sibling-file events too, so this
112            // exact-match filter is load-bearing — without it a write to
113            // e.g. `tau-lock.toml` (a sibling of watched `tau.toml`) would
114            // spuriously set `pending_reload`.
115            let touches_watched_file = event.paths.iter().zip(&canon_paths).any(|(p, canon)| {
116                watched_for_callback.contains(canon) || watched_for_callback.contains(p)
117            });
118
119            // Case 2: any path under a watched `[dirs]` root. A move/rename
120            // inside the root surfaces as a remove+add pair; both events'
121            // paths land under the root, so both branches reach here and
122            // both flip `pending_reload`.
123            let touches_watched_dir = canon_paths.iter().any(|canon| {
124                watched_dirs_for_callback
125                    .iter()
126                    .any(|d| canon.starts_with(d))
127            });
128
129            if !touches_watched_file && !touches_watched_dir {
130                return;
131            }
132
133            if touches_watched_dir {
134                // Metadata-based change detection, same purpose as the
135                // content-hash check below but scoped to `[dirs]`
136                // subtrees: ignore the FSEvents at-registration replay of
137                // Create events for files that already existed at boot
138                // (their (path, len, mtime) triple is unchanged from the
139                // initial snapshot), while still catching real adds,
140                // removes, edits, and renames (renames change the set of
141                // present paths, so the fingerprint changes regardless of
142                // which half of the remove+add pair is processed first).
143                let new_dir_hash = hash_watched_dirs(&watched_dirs_for_callback);
144                if new_dir_hash != last_dir_hash_for_callback.swap(new_dir_hash, Ordering::AcqRel) {
145                    pending_for_callback.store(true, Ordering::Release);
146                }
147                return;
148            }
149
150            // Content-based change detection for exact-file matches: ignore
151            // metadata-only and replayed events whose bytes match what we
152            // already have.
153            let new_hash = hash_watched(&watched_for_callback);
154            if new_hash != last_hash_for_callback.swap(new_hash, Ordering::AcqRel) {
155                pending_for_callback.store(true, Ordering::Release);
156            }
157        },
158        notify::Config::default(),
159    )
160    .context("create notify watcher")?;
161
162    for path in &paths {
163        if path.exists() {
164            watcher
165                .watch(path, RecursiveMode::NonRecursive)
166                .with_context(|| format!("watch {}", path.display()))?;
167        }
168    }
169
170    for dir in &dir_roots {
171        if dir.exists() {
172            watcher
173                .watch(dir, RecursiveMode::Recursive)
174                .with_context(|| format!("watch {}", dir.display()))?;
175        }
176    }
177
178    Ok(watcher)
179}

181/// Combined content fingerprint of all watched files, order-independent
182/// (paths sorted for determinism). A missing file contributes a sentinel so
183/// create/remove still register as a change. Used to suppress reloads from
184/// metadata-only and replayed filesystem events whose content is unchanged.
185fn hash_watched(watched: &HashSet<PathBuf>) -> u64 {
186    use std::hash::{Hash, Hasher};
187    let mut paths: Vec<&PathBuf> = watched.iter().collect();
188    paths.sort();
189    let mut hasher = std::collections::hash_map::DefaultHasher::new();
190    for p in paths {
191        p.hash(&mut hasher);
192        match std::fs::read(p) {
193            Ok(bytes) => bytes.hash(&mut hasher),
194            Err(_) => (-1i8).hash(&mut hasher),
195        }
196    }
197    hasher.finish()
198}

200/// Combined `(path, len, mtime)` fingerprint of every file under the given
201/// `[dirs]` roots, order-independent (paths sorted for determinism).
202///
203/// This is the `[dirs]`-subtree analogue of `hash_watched`: same boot-race
204/// purpose (see the call site's comment), but hashing file *metadata*
205/// rather than *bytes* — reading every file under `agents/`/`tools/` on
206/// every fs event would be needless I/O for a category whose purpose is
207/// just "did anything under this subtree change", and metadata already
208/// answers that (a write changes length and/or mtime; an add/remove/rename
209/// changes which paths are present at all).
210fn hash_watched_dirs(dirs: &[PathBuf]) -> u64 {
211    use std::hash::{Hash, Hasher};
212    let mut entries: Vec<(PathBuf, u64, Option<std::time::Duration>)> = Vec::new();
213    for root in dirs {
214        collect_dir_snapshot(root, &mut entries);
215    }
216    entries.sort_by(|a, b| a.0.cmp(&b.0));
217    let mut hasher = std::collections::hash_map::DefaultHasher::new();
218    for (path, len, mtime) in entries {
219        path.hash(&mut hasher);
220        len.hash(&mut hasher);
221        mtime.hash(&mut hasher);
222    }
223    hasher.finish()
224}

226/// Recursively collect `(path, len, mtime-since-epoch)` for every file
227/// under `dir`, appending to `out`. Unreadable directories/entries are
228/// silently skipped (mirrors `resolve_watch_paths`' `.flatten()` pattern
229/// elsewhere in this file) — a transient stat failure just means that
230/// entry doesn't contribute to the fingerprint this round, which is no
231/// worse than the file not existing yet.
232fn collect_dir_snapshot(dir: &Path, out: &mut Vec<(PathBuf, u64, Option<std::time::Duration>)>) {
233    let Ok(entries) = std::fs::read_dir(dir) else {
234        return;
235    };
236    for entry in entries.flatten() {
237        let path = entry.path();
238        let Ok(file_type) = entry.file_type() else {
239            continue;
240        };
241        if file_type.is_dir() {
242            collect_dir_snapshot(&path, out);
243        } else if file_type.is_file() {
244            if let Ok(meta) = entry.metadata() {
245                let mtime = meta
246                    .modified()
247                    .ok()
248                    .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok());
249                out.push((path, meta.len(), mtime));
250            }
251        }
252    }
253}

255/// Resolve the full set of paths to watch for this project:
256///   - for `.ts` projects: `project_file_path` (the `.ts` source)
257///   - for TOML projects: `<root>/tau.toml`
258///   - `<root>/workflows/*.toml` (if directory exists)
259///   - external prompt files declared as `prompt.system_file = "..."` in any agent
260fn resolve_watch_paths(
261    project_root: &Path,
262    project_file_path: &Path,
263    project: &ProjectConfig,
264) -> Vec<PathBuf> {
265    let ext = project_file_path.extension().and_then(|s| s.to_str());
266    let manifest_path = if ext == Some("ts") {
267        project_file_path.to_path_buf()
268    } else {
269        project_root.join("tau.toml")
270    };
271    let mut paths = vec![manifest_path];
272
273    // workflows/*.toml
274    let workflows_dir = project_root.join("workflows");
275    if workflows_dir.is_dir() {
276        if let Ok(entries) = std::fs::read_dir(&workflows_dir) {
277            for entry in entries.flatten() {
278                let p = entry.path();
279                if p.extension().and_then(|s| s.to_str()) == Some("toml") {
280                    paths.push(p);
281                }
282            }
283        }
284    }
285
286    // External prompt files (PromptEntry::File variant).
287    // If an agent uses `prompt.system = "..."` (inline), no extra path is added.
288    for agent in project.agents.values() {
289        if let Some(file_path) = agent_prompt_file(agent) {
290            let resolved = if file_path.is_absolute() {
291                file_path
292            } else {
293                project_root.join(file_path)
294            };
295            paths.push(resolved);
296        }
297    }
298
299    paths
300}

302/// `[dirs]` roots to watch recursively (empty when the project has none).
303///
304/// Unlike [`resolve_watch_paths`] these are directories, watched with
305/// [`RecursiveMode::Recursive`] so any file added, removed, or edited
306/// anywhere under an `agents`/`tools` root triggers a reload — including
307/// newly created subdirectories, which a flat file-watch list can't express.
308fn resolve_watch_dirs(project_root: &Path, project: &ProjectConfig) -> Vec<PathBuf> {
309    let Some(dirs) = &project.dirs else {
310        return Vec::new();
311    };
312    [dirs.agents.as_ref(), dirs.tools.as_ref()]
313        .into_iter()
314        .flatten()
315        .map(|rel| project_root.join(rel))
316        .collect()
317}
318
319/// Extract the external prompt file path from an agent's `PromptEntry`, if any.
320///
321/// Returns `None` for `PromptEntry::None` and `PromptEntry::Inline` (the
322/// watcher doesn't need to track inline strings). The trailing wildcard arm
323/// handles any future variants added to the `#[non_exhaustive]` enum.
324fn agent_prompt_file(agent: &AgentEntry) -> Option<PathBuf> {
325    match &agent.prompt {
326        PromptEntry::File(p) => Some(p.clone()),
327        PromptEntry::Inline(_) | PromptEntry::None => None,
328        // Forward-compat: future PromptEntry variants default to no file.
329        _ => None,
330    }
331}

337    #[test]
338    fn watch_paths_include_dirs_roots() {
339        let tmp = tempfile::tempdir().expect("tempdir");
340        let root = tmp.path();
341        std::fs::write(
342            root.join("tau.toml"),
343            "[project]\nname = \"p\"\n[dirs]\nagents = \"agents\"\n",
344        )
345        .expect("write tau.toml");
346        std::fs::create_dir_all(root.join("agents")).expect("mkdir agents");
347
348        let project =
349            ProjectConfig::from_path(root.join("tau.toml")).expect("project must validate");
350        assert!(project.dirs.is_some(), "expected [dirs] to be populated");
351
352        let dirs = resolve_watch_dirs(root, &project);
353        assert_eq!(dirs, vec![root.join("agents")]);
354    }
