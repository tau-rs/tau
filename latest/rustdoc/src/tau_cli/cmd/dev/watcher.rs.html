1//! File watcher for `tau dev` — wraps `notify::RecommendedWatcher`.
2//!
3//! Watches:
4//!   - `tau.toml`
5//!   - `workflows/*.toml`
6//!   - every external prompt file referenced by `[agents.X.prompt] system_file`
7//!
8//! On any `Modify`, `Create`, or `Remove` event for a watched path,
9//! `pending_reload` is set to `true`. The caller must hold the returned
10//! [`notify::RecommendedWatcher`] alive — dropping it stops the watcher.
11
12use std::collections::HashSet;
13use std::path::{Path, PathBuf};
14use std::sync::atomic::{AtomicBool, AtomicU64, Ordering};
15use std::sync::Arc;
16
17use anyhow::{Context, Result};
18use notify::{EventKind, RecommendedWatcher, RecursiveMode, Watcher};
19use tau_pkg::project::project::{AgentEntry, ProjectConfig, PromptEntry};

21/// Spawn a watcher over the project's relevant paths.
22///
23/// Returns the watcher handle — the caller **MUST** hold it alive.
24/// Dropping it stops all file watching.
25///
26/// On any modify/create/remove event for a watched path,
27/// `pending_reload` is set to `true` (Acquire/Release ordering).
28///
29/// `project_file_path` is the original path supplied by the caller (may be
30/// a `.ts` file or a directory). When it ends in `.ts`, the watcher tracks
31/// the `.ts` file instead of `<root>/tau.toml`.
32pub fn spawn(
33    project_root: &Path,
34    project_file_path: &Path,
35    project: &ProjectConfig,
36    pending_reload: Arc<AtomicBool>,
37) -> Result<RecommendedWatcher> {
38    let paths = resolve_watch_paths(project_root, project_file_path, project);
39
40    // The set of files we actually care about, canonicalized so the callback's
41    // path matching survives symlinked temp dirs (e.g. macOS `/var` ->
42    // `/private/var`). This is the crux of correct filtering: some notify
43    // backends (notably FSEvents) watch the *parent directory* of a watched
44    // file and deliver events for sibling files too. Without this filter a
45    // write to e.g. `tau-lock.toml` — a sibling of the watched `tau.toml` —
46    // would spuriously set `pending_reload`. Match each event's paths against
47    // this set so only genuine changes to watched files trigger a reload.
48    let watched: HashSet<PathBuf> = paths
49        .iter()
50        .map(|p| std::fs::canonicalize(p).unwrap_or_else(|_| p.clone()))
51        .collect();
52
53    // Content fingerprint of the watched files at boot. The watcher flips
54    // `pending_reload` only when an event *actually changes* watched content
55    // — not on metadata-only events (`Modify(Metadata)`), and not on the
56    // historical `Create`/`Modify` events that FSEvents replays for the
57    // watched file when the watch first registers. Those replayed events
58    // otherwise race the boot sequence and cause spurious reloads.
59    let last_hash = Arc::new(AtomicU64::new(hash_watched(&watched)));
60
61    let pending_for_callback = pending_reload.clone();
62    let watched_for_callback = watched.clone();
63    let last_hash_for_callback = last_hash.clone();
64    let mut watcher = RecommendedWatcher::new(
65        move |res: std::result::Result<notify::Event, notify::Error>| {
66            let Ok(event) = res else { return };
67            if !matches!(
68                event.kind,
69                EventKind::Modify(_) | EventKind::Create(_) | EventKind::Remove(_)
70            ) {
71                return;
72            }
73            // Only consider events touching an actually-watched file. Some
74            // backends (FSEvents) watch the parent directory and deliver
75            // sibling-file events; canonicalize first (matches FSEvents'
76            // `/private/…` paths), falling back to the raw path for removes.
77            let touches_watched = event.paths.iter().any(|p| {
78                let canon = std::fs::canonicalize(p).unwrap_or_else(|_| p.clone());
79                watched_for_callback.contains(&canon) || watched_for_callback.contains(p)
80            });
81            if !touches_watched {
82                return;
83            }
84            // Content-based change detection: ignore metadata-only and
85            // replayed events whose bytes match what we already have.
86            let new_hash = hash_watched(&watched_for_callback);
87            if new_hash != last_hash_for_callback.swap(new_hash, Ordering::AcqRel) {
88                pending_for_callback.store(true, Ordering::Release);
89            }
90        },
91        notify::Config::default(),
92    )
93    .context("create notify watcher")?;
94
95    for path in &paths {
96        if path.exists() {
97            watcher
98                .watch(path, RecursiveMode::NonRecursive)
99                .with_context(|| format!("watch {}", path.display()))?;
100        }
101    }
102
103    Ok(watcher)
104}

106/// Combined content fingerprint of all watched files, order-independent
107/// (paths sorted for determinism). A missing file contributes a sentinel so
108/// create/remove still register as a change. Used to suppress reloads from
109/// metadata-only and replayed filesystem events whose content is unchanged.
110fn hash_watched(watched: &HashSet<PathBuf>) -> u64 {
111    use std::hash::{Hash, Hasher};
112    let mut paths: Vec<&PathBuf> = watched.iter().collect();
113    paths.sort();
114    let mut hasher = std::collections::hash_map::DefaultHasher::new();
115    for p in paths {
116        p.hash(&mut hasher);
117        match std::fs::read(p) {
118            Ok(bytes) => bytes.hash(&mut hasher),
119            Err(_) => (-1i8).hash(&mut hasher),
120        }
121    }
122    hasher.finish()
123}

125/// Resolve the full set of paths to watch for this project:
126///   - for `.ts` projects: `project_file_path` (the `.ts` source)
127///   - for TOML projects: `<root>/tau.toml`
128///   - `<root>/workflows/*.toml` (if directory exists)
129///   - external prompt files declared as `prompt.system_file = "..."` in any agent
130fn resolve_watch_paths(
131    project_root: &Path,
132    project_file_path: &Path,
133    project: &ProjectConfig,
134) -> Vec<PathBuf> {
135    let ext = project_file_path.extension().and_then(|s| s.to_str());
136    let manifest_path = if ext == Some("ts") {
137        project_file_path.to_path_buf()
138    } else {
139        project_root.join("tau.toml")
140    };
141    let mut paths = vec![manifest_path];
142
143    // workflows/*.toml
144    let workflows_dir = project_root.join("workflows");
145    if workflows_dir.is_dir() {
146        if let Ok(entries) = std::fs::read_dir(&workflows_dir) {
147            for entry in entries.flatten() {
148                let p = entry.path();
149                if p.extension().and_then(|s| s.to_str()) == Some("toml") {
150                    paths.push(p);
151                }
152            }
153        }
154    }
155
156    // External prompt files (PromptEntry::File variant).
157    // If an agent uses `prompt.system = "..."` (inline), no extra path is added.
158    for agent in project.agents.values() {
159        if let Some(file_path) = agent_prompt_file(agent) {
160            let resolved = if file_path.is_absolute() {
161                file_path
162            } else {
163                project_root.join(file_path)
164            };
165            paths.push(resolved);
166        }
167    }
168
169    paths
170}

172/// Extract the external prompt file path from an agent's `PromptEntry`, if any.
173///
174/// Returns `None` for `PromptEntry::None` and `PromptEntry::Inline` (the
175/// watcher doesn't need to track inline strings). The trailing wildcard arm
176/// handles any future variants added to the `#[non_exhaustive]` enum.
177fn agent_prompt_file(agent: &AgentEntry) -> Option<PathBuf> {
178    match &agent.prompt {
179        PromptEntry::File(p) => Some(p.clone()),
180        PromptEntry::Inline(_) | PromptEntry::None => None,
181        // Forward-compat: future PromptEntry variants default to no file.
182        _ => None,
183    }
