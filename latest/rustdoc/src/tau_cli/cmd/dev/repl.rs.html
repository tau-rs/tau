10/// Parsed user input from the REPL prompt.
11#[derive(Debug, Clone, PartialEq, Eq)]
12pub enum Command {
13    /// Plain text — send to the current agent as a turn.
14    Prompt(String),
15    /// `:reload` — apply pending manifest changes, keep history.
16    Reload,
17    /// `:state` — print session stats.
18    State,
19    /// `:history` — print message log (last 20).
20    History,
21    /// `:agents` — list agents in the project.
22    Agents,
23    /// `:agent <name>` — switch the active agent.
24    SwitchAgent(String),
25    /// `:clear` — reset history, keep manifest.
26    Clear,
27    /// `:help` — print command list.
28    Help,
29    /// `:quit` — exit (also fired by Ctrl-D / EOF).
30    Quit,
31    /// Empty line — no-op.
32    Empty,
33    /// Unrecognised `:foo` — print error, stay at prompt.
34    UnknownColon(String),
35}
36
37/// Parse one line of REPL input.
38pub fn parse_command(line: &str) -> Command {
39    let trimmed = line.trim();
40    if trimmed.is_empty() {
41        return Command::Empty;
42    }
43    if !trimmed.starts_with(':') {
44        return Command::Prompt(trimmed.to_string());
45    }
46    let mut parts = trimmed.splitn(2, char::is_whitespace);
47    let verb = parts.next().unwrap_or("");
48    let arg = parts.next().unwrap_or("").trim();
49    match verb {
50        ":reload" => Command::Reload,
51        ":state" => Command::State,
52        ":history" => Command::History,
53        ":agents" => Command::Agents,
54        ":agent" => {
55            if arg.is_empty() {
56                Command::UnknownColon(":agent (missing name)".into())
57            } else {
58                Command::SwitchAgent(arg.to_string())
59            }
60        }
61        ":clear" => Command::Clear,
62        ":help" => Command::Help,
63        ":quit" => Command::Quit,
64        other => Command::UnknownColon(other.to_string()),
65    }
66}

68/// Run the REPL loop until the user quits.
69///
70/// When `watch_mode` is `true`, a pending reload is applied automatically at
71/// the top of each iteration (before the next prompt). When `false`, the REPL
72/// prints a hint asking the user to type `:reload` instead.
73pub async fn run_loop(
74    session: &mut DevSession,
75    output: &mut Output,
76    watch_mode: bool,
77) -> Result<()> {
78    let mut editor = DefaultEditor::new()?;
79    print_banner(output, session);
80    loop {
81        use std::sync::atomic::Ordering;
82        if session.pending_reload.load(Ordering::Acquire) {
83            if watch_mode {
84                match session.auto_reload_if_pending().await {
85                    Ok(true) => output.human(&format!(
86                        "(auto-reloaded; {} messages preserved)",
87                        session.history.len()
88                    ))?,
89                    Ok(false) => {} // race: cleared between observing and acting
90                    Err(e) => output.human(&format!("auto-reload failed: {e}"))?,
91                }
92            } else {
93                output.human("(manifest changed; type :reload to apply)")?;
94            }
95        }
96        let prompt = format!("({}) > ", session.current_agent_name());
97        let line = match editor.readline(&prompt) {
98            Ok(l) => l,
99            Err(ReadlineError::Interrupted) => {
100                output.human("(Ctrl-C: use :quit or Ctrl-D to exit)")?;
101                continue;
102            }
103            Err(ReadlineError::Eof) => break,
104            Err(e) => return Err(e.into()),
105        };
106        editor.add_history_entry(&line).ok();
107
108        match parse_command(&line) {
109            Command::Prompt(p) => {
110                if let Err(e) = session.run_turn(&p, output).await {
111                    let _ = output.human(&format!("turn failed: {e:#}"));
112                }
113            }
114            Command::Reload => match session.reload().await {
115                Ok(true) => output.human(&format!(
116                    "reloaded; {} messages preserved",
117                    session.history.len()
118                ))?,
119                Ok(false) => output.human("nothing to reload")?,
120                Err(e) => output.human(&format!(
121                    "reload failed: {e}\n(keeping previous config; fix and try :reload again)"
122                ))?,
123            },
124            Command::State => output.human("(:state stub — Phase 5)")?,
125            Command::History => output.human("(:history stub — Phase 5)")?,
126            Command::Agents => print_agents(session, output)?,
127            Command::SwitchAgent(name) => switch_agent(session, &name, output)?,
128            Command::Clear => {
129                session.history.clear();
130                output.human("history cleared")?;
131            }
132            Command::Help => print_help(output)?,
133            Command::Quit => break,
134            Command::Empty => continue,
135            Command::UnknownColon(s) => {
136                output.human(&format!("unknown command `{s}` — try :help"))?;
137            }
138        }
139    }
140    Ok(())
141}

143fn print_banner(output: &mut Output, session: &DevSession) {
144    let _ = output.human(&format!(
145        "tau dev — {} ({} agents, {} tools)",
146        session.project_root.display(),
147        session.project.agents.len(),
148        session.project.tools.len(),
149    ));
150    let _ = output.human("type :help, :reload, :state, :quit");
151}
152
153fn print_help(output: &mut Output) -> Result<()> {
154    output.human(
155        "commands:\n  \
156         > <text>           run a turn with the current agent\n  \
157         :reload            apply pending manifest changes (history preserved)\n  \
158         :state             session stats\n  \
159         :history           recent messages\n  \
160         :agents            list agents\n  \
161         :agent <name>      switch active agent\n  \
162         :clear             reset history (manifest unchanged)\n  \
163         :help              this list\n  \
164         :quit | Ctrl-D     exit\n\n\
165         note: Ctrl-C during a turn cancels best-effort; the underlying turn\n\
166         may complete in background (β.3 PR-5.1 deferral).",
167    )?;
168    Ok(())
169}
170
171fn print_agents(session: &DevSession, output: &mut Output) -> Result<()> {
172    for name in session.project.agents.keys() {
173        let marker = if name == session.current_agent_name() {
174            "*"
175        } else {
176            " "
177        };
178        output.human(&format!(" {marker} {name}"))?;
179    }
180    Ok(())
181}
182
183fn switch_agent(session: &mut DevSession, name: &str, output: &mut Output) -> Result<()> {
184    if !session.project.agents.contains_key(name) {
185        output.human(&format!("agent `{name}` not in tau.toml"))?;
186        return Ok(());
187    }
188    session.current_agent = name.to_string();
189    output.human(&format!("switched to `{name}`"))?;
190    Ok(())
191}

194mod tests {

197    #[test]
198    fn parses_prompt() {
199        assert_eq!(

208        assert_eq!(parse_command("  :reload  "), Command::Reload);
209    }
210
211    #[test]
212    fn parses_switch_agent() {
213        assert_eq!(
214            parse_command(":agent fan-monitor"),

217    }
218
219    #[test]
220    fn empty_line_is_empty() {
221        assert_eq!(parse_command(""), Command::Empty);
222        assert_eq!(parse_command("   "), Command::Empty);
223    }
224
225    #[test]
226    fn unknown_colon_command() {
227        match parse_command(":foobar") {
228            Command::UnknownColon(s) => assert_eq!(s, ":foobar"),

230        }
