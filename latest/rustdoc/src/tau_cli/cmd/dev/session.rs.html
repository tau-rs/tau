25/// All the long-lived state for one `tau dev` invocation.
26pub struct DevSession {
27    /// Original project path supplied by the caller.
28    ///
29    /// May be a directory (TOML-based project) or a `.ts` file (β.8
30    /// TypeScript-based project). Used by the file watcher and `:reload`
31    /// to re-extract the project config from the correct source.
32    pub project_file_path: PathBuf,
33    /// Project root (the directory containing the manifest file).
34    pub project_root: PathBuf,
35    /// Parsed + validated project config.
36    pub project: ProjectConfig,
37    /// Lowered IR module for the current project.
38    pub ir: IrModule,
39    /// Content-addressed asset store (D6-B) for the current IR — the
40    /// `system_file` prompt bytes `self.ir` references. Re-derived on
41    /// `:reload` alongside `self.ir`; handed to the interpreter so
42    /// `PromptSource::Asset` prompts resolve.
43    pub assets: BTreeMap<String, tau_ir::asset::AssetBlob>,
44    /// Name of the agent the REPL is currently driving.
45    pub current_agent: String,
46    /// Multi-turn conversation history (in-memory only in v1).
47    pub history: Vec<Message>,
48    /// Set true by the file watcher (Phase 4) when a watched file changes.
49    /// Cleared by `:reload`.
50    pub pending_reload: Arc<AtomicBool>,
51    /// Watcher handle — kept alive to keep file-watching active.
52    /// `None` if the watcher failed to register at boot (degraded mode).
53    pub notify_handle: Option<notify::RecommendedWatcher>,
54}
55
56impl fmt::Debug for DevSession {
57    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
58        f.debug_struct("DevSession")
59            .field("project_file_path", &self.project_file_path)
60            .field("project_root", &self.project_root)
61            .field("current_agent", &self.current_agent)
62            .field("history_len", &self.history.len())
63            .field(
64                "pending_reload",
65                &self
66                    .pending_reload
67                    .load(std::sync::atomic::Ordering::Relaxed),
68            )
69            .field(
70                "notify_handle",
71                &self.notify_handle.as_ref().map(|_| "<watcher>"),
72            )
73            .finish_non_exhaustive()
74    }
75}

77impl DevSession {
78    /// Load + validate + lower a project into a fresh session.
79    ///
80    /// `project_path` may be a directory (TOML-based project) or a `.ts` file
81    /// (β.8 TypeScript-based project). File-extension dispatch happens inside
82    /// [`crate::cmd::project_load::load_project`].
83    pub async fn load(project_path: PathBuf, agent_override: Option<String>) -> Result<Self> {
84        let crate::cmd::project_load::LoadedProject {
85            project_root,
86            project,
87        } = crate::cmd::project_load::load_project(&project_path)?;
88
89        // `project.agents` is a `BTreeMap<String, AgentEntry>`, so `.keys()` iterates
90        // in alphabetical order — the first key is the alphabetical default.
91        let current_agent = match agent_override {
92            Some(name) => {
93                if !project.agents.contains_key(&name) {
94                    return Err(anyhow!("agent `{name}` not found in project"));
95                }
96                name
97            }
98            None => project
99                .agents
100                .keys()
101                .next()
102                .ok_or_else(|| anyhow!("project declares no agents"))?
103                .clone(),
104        };
105
106        let (ir, assets) =
107            lower_project_to_ir(&project, &project_root).context("lower project to IR")?;
108
109        let pending_reload = Arc::new(AtomicBool::new(false));
110
111        let notify_handle = match crate::cmd::dev::watcher::spawn(
112            &project_root,
113            &project_path,
114            &project,
115            pending_reload.clone(),
116        ) {
117            Ok(w) => Some(w),
118            Err(e) => {
119                eprintln!("warning: file watcher unavailable ({e}); use :reload manually");
120                None
121            }
122        };
123
124        Ok(Self {
125            project_file_path: project_path,
126            project_root,
127            project,
128            ir,
129            assets,
130            current_agent,
131            history: Vec::new(),
132            pending_reload,
133            notify_handle,
134        })
135    }

137    /// Name of the agent the REPL is currently driving.
138    pub fn current_agent_name(&self) -> &str {
139        &self.current_agent
140    }

142    /// Read-only access to the conversation history.
143    pub fn history(&self) -> &[Message] {
144        &self.history
145    }

147    /// Apply pending manifest changes.
148    ///
149    /// Re-parses `tau.toml`, re-lowers IR, and updates `self.project` and
150    /// `self.ir`. Conversation history is intentionally preserved so that
151    /// the REPL retains multi-turn context across hot-reloads.
152    ///
153    /// MCP clients are created per-turn in [`run_turn`][Self::run_turn] (the `_mcp_lifetime`
154    /// binding), so there are no long-lived MCP handles to drop here —
155    /// the next call to `run_turn` automatically picks up the new config.
156    ///
157    /// Returns:
158    /// - `Ok(true)`  — manifest reloaded successfully.
159    /// - `Ok(false)` — `pending_reload` was not set; nothing to do.
160    /// - `Err(e)`    — parse or lowering error; old config is kept intact
161    ///   and `pending_reload` is restored so the user can fix and retry.
162    pub async fn reload(&mut self) -> anyhow::Result<bool> {
163        use std::sync::atomic::Ordering;
164
165        if !self.pending_reload.swap(false, Ordering::AcqRel) {
166            return Ok(false);
167        }
168
169        let new_project = match crate::cmd::project_load::load_project(&self.project_file_path) {
170            Ok(loaded) => loaded.project,
171            Err(e) => {
172                self.pending_reload.store(true, Ordering::Release);
173                return Err(anyhow!("reload project: {e}"));
174            }
175        };
176
177        let (new_ir, new_assets) = match lower_project_to_ir(&new_project, &self.project_root) {
178            Ok(pair) => pair,
179            Err(e) => {
180                self.pending_reload.store(true, Ordering::Release);
181                return Err(anyhow!("lower IR: {e}"));
182            }
183        };
184
185        // If the current agent was removed in the new manifest, fall back
186        // to the first agent (alphabetical order) so the session stays valid.
187        if !new_project.agents.contains_key(&self.current_agent) {
188            if let Some(first) = new_project.agents.keys().next() {
189                self.current_agent = first.clone();
190            }
191        }
192
193        self.project = new_project;
194        self.ir = new_ir;
195        self.assets = new_assets;
196        // history intentionally NOT touched
197        Ok(true)
198    }

200    /// Watch-mode counterpart to [`reload`][Self::reload].
201    ///
202    /// Has the same effect (re-parse + re-lower + keep history) but signals
203    /// "auto-applied by `--watch`" rather than "user typed `:reload`". The thin
204    /// wrapper makes the watch-mode path explicit at call sites.
205    pub async fn auto_reload_if_pending(&mut self) -> anyhow::Result<bool> {
206        self.reload().await
207    }

209    /// Run one turn against the current agent.
210    ///
211    /// Mirrors the construction in `crate::cmd::ir_dispatcher::run_via_ir`:
212    /// 1. Resolve the agent entry from `self.project`.
213    /// 2. Load the per-scope lockfile + plugins via `plugin_loader`.
214    /// 3. Boot MCP servers from the project config + lockfile.
215    /// 4. Build a `ForwardingDispatcher`.
216    /// 5. Append the user prompt to `self.history` and call [`run_ir`].
217    /// 6. Append the agent's reply to `self.history` and render the outcome.
218    ///
219    /// History is preserved across turns so the REPL has multi-turn context.
220    /// On error the history is left unchanged (the failed turn is not appended).
221    pub async fn run_turn(
222        &mut self,
223        prompt: &str,
224        output: &mut crate::output::Output,
225    ) -> Result<()> {
226        // 1. Resolve the agent entry.
227        let agent_entry = self
228            .project
229            .agents
230            .get(&self.current_agent)
231            .ok_or_else(|| {
232                anyhow!(
233                    "agent `{}` not in project (was it removed between loads?)",
234                    self.current_agent
235                )
236            })?
237            .clone();
238
239        // 2. Resolve the package scope from the project root.
240        let scope =
241            tau_pkg::Scope::resolve(&self.project_root).context("resolving package scope")?;
242
243        // 3. Build plugin host options (no recording, no forced sandbox in dev mode).
244        let run_id = format!(
245            "tau-dev-{}",
246            std::time::SystemTime::now()
247                .duration_since(std::time::UNIX_EPOCH)
248                .map(|d| d.as_nanos())
249                .unwrap_or(0)
250        );
251        let trace_context =
252            TraceContext::new(run_id, self.current_agent.clone(), "dev-root".to_string());
253        let host_options = plugin_loader::build_host_options(
254            None,  // no --record-protocol in dev mode
255            false, // no --no-sandbox override
256            None,  // no forced adapter kind
257        );
258        let loaded = plugin_loader::load_plugins(
259            &agent_entry,
260            &scope,
261            &self.project.models,
262            trace_context,
263            host_options,
264        )
265        .await?;
266        let runtime = loaded
267            .builder
268            .build()
269            .context("failed to build runtime from spawned plugins")?;
270
271        // 4. Pull DynLlmBackend + DynTool handles.
272        let llm_backend: Arc<dyn DynLlmBackend> =
273            runtime
274                .llm_backends()
275                .values()
276                .next()
277                .cloned()
278                .ok_or_else(|| anyhow!("runtime has no LLM backend after plugin load"))?;
279
280        let mut tools_by_id: BTreeMap<tau_ir::ToolId, Arc<dyn DynTool>> = BTreeMap::new();
281        for ir_tool_id in self.ir.workflow.tools.keys() {
282            if let Some(handle) = runtime.tools().get(&ir_tool_id.0) {
283                tools_by_id.insert(ir_tool_id.clone(), handle.clone());
284            }
285        }
286
287        // 5. Boot MCP servers (from lockfile + project config) when a lockfile exists.
288        //    Non-MCP projects (or projects not yet locked) skip this step safely.
289        let lockfile_path = scope.lockfile_path();
290        let _mcp_lifetime; // keep inbound-dispatch handles alive for the full turn
291        if lockfile_path.exists() {
292            let lockfile = tau_pkg::lockfile::LockFile::load(&lockfile_path)
293                .with_context(|| format!("loading lockfile at {}", lockfile_path.display()))?;
294            let mcp_setup = setup_mcp_runtime(
295                &self.project,
296                &lockfile,
297                llm_backend.clone(),
298                loaded.sandbox_adapter.clone(),
299            )
300            .await
301            .map_err(|e| anyhow!("MCP setup failed: {e}"))?;
302            for (id, tool) in mcp_setup.tools {
303                tools_by_id.insert(id, tool);
304            }
305            _mcp_lifetime = mcp_setup.inbound_handles;
306        } else {
307            _mcp_lifetime = Vec::new();
308        }
309
310        // 6. Build dispatcher and append the user turn to history. The
311        //    dispatcher supplies `TokioClock`/`OsRandom`, so `run_agent`'s
312        //    "clock must be supplied" invariant holds on both the pipeline
313        //    and single-agent paths. It is rebuilt every turn (and reads
314        //    `self.ir` fresh), so `:reload` — which swaps `self.ir` — is
315        //    automatically reflected on the next turn.
316        let llm_backends: BTreeMap<String, Arc<dyn DynLlmBackend>> = runtime
317            .llm_backends()
318            .iter()
319            .map(|(name, handle)| (name.clone(), handle.clone()))
320            .collect();
321        let dispatcher = Arc::new(
322            ForwardingDispatcher::new(llm_backends, tools_by_id).with_assets(self.assets.clone()),
323        );
324        let initial = Message::new(
325            Address::User,
326            Address::Agent(AgentInstanceId::new()),
327            MessagePayload::Text {
328                content: prompt.to_string(),
329            },
330        );
331        self.history.push(initial);
332
333        // 6a. Pipeline dispatch — mirrors `cmd::run::try_run_pipeline`.
334        //     When the project lowered to an `IrModule` whose
335        //     `workflow.pipeline` is `Some` (a `[[pipeline.steps]]` block was
336        //     declared), sequence the whole pipeline via `run_pipeline`,
337        //     threading each step's output to later steps, and render the
338        //     LAST step's stored output. When no pipeline is declared (the
339        //     overwhelming majority of projects) this branch is skipped and
340        //     the single-agent path below runs unchanged.
341        if self.ir.workflow.pipeline.is_some() {
342            let module = Arc::new(self.ir.clone());
343            // The id of the LAST step — its stored output is the run result.
344            // An empty `steps` vec cannot lower to `Some(pipeline)` (the
345            // parser rejects it), but guard anyway rather than index blindly.
346            let last_step_id = module
347                .workflow
348                .pipeline
349                .as_ref()
350                .and_then(|p| p.steps.last())
351                .map(|s| s.id.0.clone());
352
353            let store_result = run_pipeline(module, prompt.to_string(), dispatcher).await;
354
355            drop(runtime);
356            plugin_loader::flush_recorders().await;
357
358            let store = store_result.context("running pipeline via IR interpreter")?;
359            let last_step_id = last_step_id.ok_or_else(|| anyhow!("pipeline declared no steps"))?;
360            return render_pipeline_result(&store, &last_step_id, output);
361        }
362
363        // 6b. Single-agent dispatch (default; unchanged).
364        let entry_id = AgentId(self.current_agent.clone());
365        let history_snapshot = self.history.clone();
366
367        let run_outcome = run_ir(
368            Arc::new(self.ir.clone()),
369            &entry_id,
370            dispatcher,
371            history_snapshot,
372        )
373        .await;
374
375        drop(runtime);
376        plugin_loader::flush_recorders().await;
377
378        let outcome = run_outcome.context("running agent via IR interpreter")?;
379        render_outcome(outcome, output)
380    }
381}

383/// Shared helper: lower a `ProjectConfig` to an `IrModule`.
384///
385/// Uses `TargetTriple::PASSTHROUGH` (no capability-fit gate) so that `tau
386/// dev` works on any host regardless of sandbox tier. All three caches are
387/// stubs (`|_| None`) — MCP contracts and skill hashes are not pinned in
388/// dev mode.
389///
390/// Used by both `load` (Phase 2) and the `:reload` command (Phase 5).
391fn lower_project_to_ir(
392    project: &ProjectConfig,
393    project_root: &std::path::Path,
394) -> Result<(IrModule, BTreeMap<String, tau_ir::asset::AssetBlob>)> {
395    // `system_file` prompts resolve relative to the project root, matching
396    // `tau run` and the bundle builder (D6-B).
397    let caches = Caches {
398        native_tool: &|_| None,
399        mcp_contract: &|_| None,
400        skill: &|_| None,
401        prompt_file: &|p: &std::path::Path| {
402            tau_pkg::bundle::read_prompt_file(p, project_root)
403                .map_err(|e| tau_ir_lower::PromptFileError(e.to_string()))
404        },
405    };
406    lower_project(project, &TargetTriple::PASSTHROUGH, &caches)
407        .map(|out| (out.module, out.assets))
408        .map_err(|e| anyhow!("IR lowering failed: {e}"))
409}
