1//! `tau run --bundle` IR-interpreter dispatch path.
2//!
3//! When a verified bundle carries an [`tau_pkg::bundle::manifest::IrPayload`]
4//! ("v2" bundle), this module decodes the canonical IR bytes, builds a
5//! forwarding [`ToolDispatcher`] over the host's plugin registry, and
6//! drives the IR to completion: a bundled `[[pipeline.steps]]` block runs
7//! through [`tau_runtime_core::interpreter::pipeline::run_pipeline`], while
8//! a pipeline-less bundle runs its single entry agent via
9//! [`tau_runtime_core::interpreter::run_ir`]. Either result is mapped to
10//! stdout / [`crate::cmd::run::AgentFailed`] using the same shape the
11//! cwd-based path uses, so callers see identical exit-code semantics
12//! regardless of which path executed the run.
13//!
14//! Legacy v1 bundles (no `ir_payload`) continue to use the cwd-based
15//! agent path — see [`crate::cmd::run::run`].

42/// Run a verified v2 bundle through the IR interpreter.
43///
44/// When the bundle's IR carries a `[[pipeline.steps]]` block
45/// (`module.workflow.pipeline.is_some()`), the engine sequences the whole
46/// pipeline via [`tau_runtime_core::interpreter::pipeline::run_pipeline`],
47/// threading each step's output to later steps and rendering the LAST
48/// step's output — symmetric with the cwd path's `try_run_pipeline` (see
49/// [`crate::cmd::run`]). Otherwise it runs the entry agent's single loop,
50/// where the entry is the positional `<agent>` argument resolved against
51/// the module's agents (#623 — matching the dev path's `tau run <agent>`
52/// semantics; an id absent from the module is a hard error).
53///
54/// Returns `Ok(())` on a `RunOutcome::Completed`, `Err(AgentFailed)` on
55/// `RunOutcome::Failed`, and any other kernel/CLI error as a wrapped
56/// [`anyhow::Error`] — matching the cwd-based path's error shape so
57/// `lib::run_main`'s downcast continues to map exit codes correctly.
58#[allow(clippy::too_many_arguments)]
59/// Build the runtime asset map from a bundle's `[[assets]]` store (D6-B),
60/// decoding each blob and verifying its bytes hash matches its key. The bundle
61/// self-hash proves the manifest is untampered as a unit, but not that each
62/// asset *key* matches its bytes — this re-hash closes that gap so the IR's
63/// content-addressed reference can be trusted.
64pub(crate) fn asset_map_from_bundle(
65    assets: &[tau_pkg::bundle::manifest::BundleAsset],
66) -> anyhow::Result<BTreeMap<String, tau_ir::asset::AssetBlob>> {
67    let mut map = BTreeMap::new();
68    for a in assets {
69        let bytes = a
70            .bytes()
71            .map_err(|e| anyhow::anyhow!("bundle asset {:?}: {e}", a.hash))?;
72        let computed = tau_ir::asset::asset_hash(&bytes);
73        if computed != a.hash {
74            return Err(anyhow::anyhow!(
75                "bundle asset key {:?} does not match its content hash {:?} — tampered bundle.",
76                a.hash,
77                computed
78            ));
79        }
80        let kind = match a.kind.as_str() {
81            "prompt" => tau_ir::asset::AssetKind::Prompt,
82            other => {
83                return Err(anyhow::anyhow!(
84                    "bundle asset {:?} has unknown kind {:?}",
85                    a.hash,
86                    other
87                ))
88            }
89        };
90        map.insert(a.hash.clone(), tau_ir::asset::AssetBlob { kind, bytes });
91    }
92    Ok(map)
93}

95/// Every [`ToolId`] the module provides as [`tau_ir::ToolImpl::Native`].
96///
97/// Native bodies are statically linked (`tau-native-tools`), never
98/// installed as plugins, so they never appear in the runtime's tool
99/// registry. Both `ForwardingDispatcher` construction sites (this module's
100/// bundle path and [`crate::cmd::run`]'s cwd pipeline path) pass this set
101/// so a native tool_ref the agent calls is dispatchable rather than a
102/// hard "no tool registered" failure (#639).
103pub(crate) fn native_tool_ids(module: &IrModule) -> std::collections::BTreeSet<ToolId> {
104    module
105        .workflow
106        .tools
107        .iter()
108        .filter(|(_, t)| matches!(t.impl_, tau_ir::ToolImpl::Native { .. }))
109        .map(|(id, _)| id.clone())
110        .collect()
111}

113pub(crate) async fn run_via_ir(
114    module: IrModule,
115    assets: BTreeMap<String, tau_ir::asset::AssetBlob>,
116    args: &RunArgs,
117    record_protocol: Option<PathBuf>,
118    force_passthrough: bool,
119    force_adapter_kind: Option<tau_runtime_tokio::process_gate::registry::RegistryKind>,
120    output: &mut Output,
121) -> anyhow::Result<()> {
122    // 1. Resolve the entry agent from the positional `<agent>` argument,
123    //    validated against the module's agents (#623 — the argument used to
124    //    be silently ignored in favour of the alphabetically-first module
125    //    agent, so the plugin backend was configured from the wrong agent's
126    //    `[agents.<id>.config]`).
127    let entry_agent_id = tau_ir::ids::AgentId(args.agent_id.clone());
128    if !module.workflow.agents.contains_key(&entry_agent_id) {
129        let available: Vec<&str> = module
130            .workflow
131            .agents
132            .keys()
133            .map(|a| a.0.as_str())
134            .collect();
135        return Err(anyhow::anyhow!(
136            "agent id {:?} not found in the bundle's IR module (available \
137             agents: {available:?})",
138            args.agent_id
139        ));
140    }
141
142    // 2. Load the project config from the cwd (proven byte-clean by the
143    //    bundle verify gate). The IR module names agents using the IR's
144    //    own `AgentId`; the project tau.toml uses `AgentEntry.id`. v0
145    //    requires them to agree (lowering preserves the id verbatim).
146    let cwd = std::env::current_dir()?;
147    let project_path = cwd.join("tau.toml");
148    let project = ProjectConfig::from_path(&project_path)
149        .with_context(|| format!("project tau.toml required at {project_path:?}"))?;
150
151    let agent_entry = project.agents.get(&entry_agent_id.0).ok_or_else(|| {
152        anyhow::anyhow!(
153            "IR entry agent {:?} not found in project tau.toml (the IR \
154             lowering should keep agent ids verbatim; this is a build/run skew)",
155            entry_agent_id.0
156        )
157    })?;
158
159    // 3. Resolve + install missing packages for this agent.
160    let scope = tau_pkg::Scope::resolve(&cwd).context("resolving package scope")?;
161    crate::cmd::resolve_helpers::resolve_and_install_for_agent(
162        agent_entry,
163        &scope,
164        args.no_install,
165        output,
166    )?;
167
168    // 4. Build plugin host options + spawn plugins (same flow the cwd path uses).
169    // One ULID per run, shared by the plugin TraceContext (log grouping) and
170    // the orchestration trace sink (spec §13.5), so `.tau/runs/<id>.jsonl`,
171    // `tau trace <id>` and the plugin logs all agree on the run's identity.
172    let run_id = crate::cmd::run::mint_run_id();
173    let trace_context =
174        TraceContext::new(run_id.clone(), entry_agent_id.0.clone(), "root".to_string());
175    let host_options = plugin_loader::build_host_options(
176        record_protocol.as_deref(),
177        force_passthrough,
178        force_adapter_kind,
179    );
180    let loaded = plugin_loader::load_plugins(
181        agent_entry,
182        &scope,
183        project.effective_models(),
184        trace_context,
185        host_options,
186    )
187    .await?;
188    let runtime = loaded
189        .builder
190        .build()
191        .context("failed to build runtime from spawned plugins")?;
192
193    // 5. Pull Arc<dyn DynLlmBackend> + Arc<dyn DynTool> handles for the
194    //    forwarding dispatcher. ToolId.0 == tool.spec.name == DynTool::name()
195    //    by lowering construction; see crates/tau-ir/src/lower/parse.rs.
196    // Hold the WHOLE name-keyed registry so the forwarding dispatcher can
197    // select a backend per agent/judge by the name baked into its resolved
198    // `model_ref` (per-agent model resolution) — not just one ambient backend.
199    let llm_backends: BTreeMap<String, Arc<dyn DynLlmBackend>> = runtime
200        .llm_backends()
201        .iter()
202        .map(|(name, handle)| (name.clone(), handle.clone()))
203        .collect();
204    if llm_backends.is_empty() {
205        return Err(anyhow::anyhow!(
206            "runtime has no LLM backend after plugin load"
207        ));
208    }
209    // A representative backend for MCP sampling setup. Per-server backend
210    // selection is out of scope for the per-agent-model track.
211    let mcp_backend = llm_backends
212        .values()
213        .next()
214        .cloned()
215        .expect("llm_backends is non-empty (checked above)");
216
217    let mut tools_by_id: BTreeMap<ToolId, Arc<dyn DynTool>> = BTreeMap::new();
218    for ir_tool_id in module.workflow.tools.keys() {
219        if let Some(handle) = runtime.tools().get(&ir_tool_id.0) {
220            tools_by_id.insert(ir_tool_id.clone(), handle.clone());
221        }
222        // ForwardingDispatcher::invoke still returns a typed RuntimeError
223        // for any unknown ToolId as defense-in-depth, but the
224        // build-time-style pre-check below is the canonical gate so the
225        // run aborts before any LLM tokens are spent.
226    }
227
228    // 5c. MCP extension: open each [tools.<entry>] mcp = "..." server,
229    //     verify drift vs lockfile, spawn inbound-dispatch tasks, and
230    //     insert one McpBackedTool per server-tool into tools_by_id.
231    //     Non-MCP projects (empty mcp_entries) produce an empty setup — no-op.
232    let lockfile =
233        tau_pkg::lockfile::LockFile::load(&scope.lockfile_path()).with_context(|| {
234            format!(
235                "loading lockfile for MCP setup: {}",
236                scope.lockfile_path().display()
237            )
238        })?;
239    let mcp_setup = setup_mcp_runtime(
240        &project,
241        &lockfile,
242        mcp_backend.clone(),
243        loaded.sandbox_adapter.clone(),
244    )
245    .await
246    .map_err(|e| anyhow::anyhow!("MCP setup: {e}"))?;
247    for (id, tool) in mcp_setup.tools {
248        tools_by_id.insert(id, tool);
249    }
250    // Hold inbound-dispatch task handles for the lifetime of the run.
251    // Dropping them here would abort the pumps immediately.
252    let _mcp_lifetime = mcp_setup.inbound_handles;
253
254    // Ids the IR provides natively — exempt from the plugin-install
255    // pre-check below, and served by the dispatcher from the linked
256    // `tau-native-tools` crate at invoke time (#639).
257    let native_ids = native_tool_ids(&module);
258
259    // 5b. Pre-check: every ToolId referenced by the entry agent must be
260    //     resolvable through the dispatcher. tau's general stance (per
261    //     CLAUDE.md / "feedback_tau_rust_like_build_enforcement"): any
262    //     check that *can* run at startup MUST run at startup — never
263    //     trickle through at invoke time. Collect every missing id so
264    //     the operator sees the full set in one shot.
265    let entry_agent = module.workflow.agents.get(&entry_agent_id).ok_or_else(|| {
266        anyhow::anyhow!(
267            "IR module has no entry agent {:?} (lowering invariant violated)",
268            entry_agent_id.0
269        )
270    })?;
271    let missing: Vec<&ToolId> = entry_agent
272        .tool_refs
273        .iter()
274        .filter(|tid| !tools_by_id.contains_key(*tid))
275        // Native tools (`ToolImpl::Native`) are statically-linked bodies
276        // (see `tau-native-tools`), not plugin-installed binaries — their
277        // absence from the plugin runtime is not an install skew. The
278        // dispatcher serves them from the linked crate instead (#639).
279        .filter(|tid| !native_ids.contains(*tid))
280        .collect();
281    if !missing.is_empty() {
282        let names: Vec<&str> = missing.iter().map(|t| t.0.as_str()).collect();
283        return Err(anyhow::anyhow!(
284            "IR entry agent {:?} references tools not present in the runtime: {:?}. \
285             This is a build/install skew — the bundle's IR was compiled against a \
286             different plugin set than the one installed in this scope.",
287            entry_agent_id.0,
288            names
289        ));
290    }
291
292    // 5c. Closed-world asset check (D6-B): every `PromptSource::Asset` the
293    //     module references must resolve to a bundle asset, and every bundle
294    //     asset must be referenced (no orphans) — keeps bundles canonical and
295    //     mirrors the tool_refs pre-check above. Load-time, before any tokens.
296    {
297        let mut referenced: std::collections::BTreeSet<&str> = std::collections::BTreeSet::new();
298        for agent in module.workflow.agents.values() {
299            if let Some(h) = agent.prompt.asset_hash() {
300                if !assets.contains_key(h) {
301                    return Err(anyhow::anyhow!(
302                        "IR agent {:?} prompt references asset {:?} not present in the bundle's \
303                         asset store — build/install skew or a tampered bundle.",
304                        agent.id.0,
305                        h
306                    ));
307                }
308                referenced.insert(h);
309            }
310        }
311        if let Some(orphan) = assets.keys().find(|k| !referenced.contains(k.as_str())) {
312            return Err(anyhow::anyhow!(
313                "bundle asset {:?} is not referenced by any agent prompt (orphan) — the bundle \
314                 is not canonical.",
315                orphan
316            ));
317        }
318    }
319
320    // 5d. ADR-0053: wire durable checkpoint/resume when the entry agent
321    //     declares `[durable]`. The store is rooted at the project scope so
322    //     checkpoints land under `<scope>/.tau/runs/<run_id>/`. On
323    //     `--resume <id>` we load the latest committed checkpoint; on a fresh
324    //     durable run we mint a run id and tell the operator how to resume.
325    let mut dispatcher =
326        ForwardingDispatcher::new(llm_backends, tools_by_id).with_native(native_ids);
327    if let Some(durability) = entry_agent.durable.as_ref() {
328        let store: Arc<dyn tau_ports::CheckpointStore> = Arc::new(
329            tau_runtime_tokio::FileCheckpointStore::new(scope.path().to_path_buf()),
330        );
331        let (durable_run_id, resume) = match &args.resume {
332            Some(rid) => {
333                let resume = store
334                    .load_latest(rid)
335                    .map_err(|e| anyhow::anyhow!("loading checkpoint for --resume {rid:?}: {e}"))?;
336                match &resume {
337                    Some(c) => {
338                        output.status(format!(
339                            "resuming durable run {rid:?} from turn {} (turns 1..={} not re-billed)",
340                            c.turn, c.turn
341                        ))?;
342                    }
343                    None => {
344                        output.warn(format!(
345                            "no checkpoint found for --resume {rid:?}; starting a fresh run under that id"
346                        ))?;
347                    }
348                }
349                (rid.clone(), resume)
350            }
351            None => {
352                // Collision-resistant ULID (matches `cmd::run`'s durable and
353                // workflow run-id minting). The previous `run-<nanos>` scheme
354                // could collide under fast successive runs or a backward clock
355                // step, corrupting the checkpoint directory shared by both runs.
356                let rid = crate::cmd::run::mint_run_id();
357                output.status(format!(
358                    "durable run id: {rid} (resume after a crash with `tau run --resume {rid}`)"
359                ))?;
360                (rid, None)
361            }
362        };
363        // EPIC 6.1: resolve the agent's durability for THIS host's target and
364        // refuse to start a run we cannot make durable — symmetric with
365        // `tau check --target` failing at build time.
366        let host_target = tau_ports::target::TargetTriple::host();
367        let resolved = tau_runtime_core::resolve_durability(durability, &host_target)
368            .require_supported(&host_target)
369            .map_err(|e| anyhow::anyhow!("{e}"))?;
370        dispatcher = dispatcher.with_durable(store, durable_run_id, resume, resolved.checkpoint);
371    }
372    // D6-B: hand the interpreter the content-addressed asset store so it can
373    // resolve `PromptSource::Asset` prompt references (empty map => no-op).
374    let dispatcher = dispatcher.with_assets(assets);
375
376    // Spec §13.5: attach the run-log writer (and, under --tui, a live
377    // channel) so the interpreter's agent loop emits TraceEvents. Reuses the
378    // same writer + file namespace as the multi-agent path, so
379    // `tau trace --last` picks up bundle runs with no reader changes.
380    //
381    // Built inline (not via `trace_mpsc::channel_with_writer`) so this call
382    // site keeps the writer's `JoinHandle`: that handle is what
383    // `flush_jsonl_writer` below awaits, closing a #650-shaped race where a
384    // detached `tokio::spawn` writer task is never scheduled again before
385    // the CLI process exits, leaving `.tau/runs/<id>.jsonl` truncated to
386    // zero bytes even though the run completed successfully.
387    let mut trace_subscribers: Vec<
388        Arc<dyn tau_runtime_core::orchestration::trace::TraceSubscriber>,
389    > = Vec::new();
390    let (jsonl_subscriber, jsonl_rx) =
391        tau_runtime_tokio::orchestration::trace_mpsc::MpscTraceSubscriber::channel();
392    let jsonl_writer_handle = tau_runtime_tokio::orchestration::persistence::spawn_writer(
393        tau_runtime_tokio::orchestration::persistence::run_log_path(scope.path(), &run_id),
394        jsonl_rx,
395    );
396    trace_subscribers.push(Arc::new(jsonl_subscriber));
397    let tui_rx = if args.tui {
398        let (subscriber, rx) =
399            tau_runtime_tokio::orchestration::trace_mpsc::MpscTraceSubscriber::channel();
400        trace_subscribers.push(Arc::new(subscriber));
401        Some(rx)
402    } else {
403        None
404    };
405    let dispatcher = Arc::new(dispatcher.with_trace(run_id.clone(), trace_subscribers));
406
407    // 6. Build the initial prompt text from --prompt / stdin.
408    let prompt_text = match &args.prompt {
409        Some(s) => s.clone(),
410        None => {
411            let mut buf = String::new();
412            std::io::stdin()
413                .read_to_string(&mut buf)
414                .context("reading initial prompt from stdin")?;
415            buf
416        }
417    };
418
419    let module = std::sync::Arc::new(module);
420
421    // 7. Drive the IR. A bundle whose IR carries a `[[pipeline.steps]]`
422    //    block is engine-sequenced via `run_pipeline` (rendering the LAST
423    //    step's output); a bundle with NO pipeline keeps the
424    //    single-entry-agent `run_ir` path below BYTE-FOR-BYTE unchanged.
425    if module.workflow.pipeline.is_some() {
426        // The id of the LAST NON-CHECK, NON-SUSPEND pipeline step — its
427        // stored output is the run's final result. Trailing `StepRun::Check`
428        // steps (lowered from `[goals.*]` / `[deliverables.*]`) evaluate
429        // postconditions but store NO output, so rendering one would hit the
430        // "no output" guard even when all checks pass. `Suspend` likewise
431        // records no output. `Pipeline::final_leaf_step_id` skips both and
432        // renders the last output-producing step. The parser rejects an
433        // empty pipeline (see `tau_pkg`), but guard anyway rather than index
434        // blindly.
435        let last_step_id = module
436            .workflow
437            .pipeline
438            .as_ref()
439            .and_then(|p| p.final_leaf_step_id())
440            .map(|id| id.0.clone())
441            .ok_or_else(|| anyhow::anyhow!("IR pipeline has no steps"))?;
442
443        let store = drive_with_optional_tui(
444            tau_runtime_core::interpreter::pipeline::run_pipeline(module, prompt_text, dispatcher),
445            tui_rx,
446        )
447        .await;
448
449        // `dispatcher` (and every trace_subscribers clone it carried,
450        // including `jsonl_subscriber`'s sender) is fully dropped by now —
451        // `run_pipeline` owned it and has returned. Awaiting the writer
452        // handle here guarantees every already-queued TraceEvent is
453        // fsync'd to `.tau/runs/<run_id>.jsonl` before this function
454        // reports the outcome.
455        flush_jsonl_writer(jsonl_writer_handle).await;
456
457        // Drop runtime + flush recorders before rendering, identical to
458        // the single-agent path's discipline below so plugin processes are
459        // reaped and recording files are flushed before the process exits.
460        drop(runtime);
461        plugin_loader::flush_recorders().await;
462
463        let store = store.context("running pipeline via IR interpreter")?;
464        return crate::cmd::run::render_pipeline_result(&store, &last_step_id, output);
465    }
466
467    let initial = Message::new(
468        Address::User,
469        Address::Agent(AgentInstanceId::new()),
470        MessagePayload::Text {
471            content: prompt_text,
472        },
473    );
474
475    // 8. Drive the single entry agent.
476    let run_outcome = drive_with_optional_tui(
477        run_ir(module, &entry_agent_id, dispatcher, vec![initial]),
478        tui_rx,
479    )
480    .await;
481
482    // `dispatcher` is fully dropped by now (see the pipeline branch's
483    // identical comment above) — flush the run-log writer before anything
484    // else observes the run as finished.
485    flush_jsonl_writer(jsonl_writer_handle).await;
486
487    // 9. Drop runtime + flush recorders before rendering, identical to
488    //    the cwd path's discipline so plugin processes are reaped and
489    //    recording files are flushed before the process exits.
490    drop(runtime);
491    plugin_loader::flush_recorders().await;
492
493    let outcome = run_outcome.context("running agent via IR interpreter")?;
494    render_outcome(outcome, output)
495}

497/// Await the run-log JSONL writer's task to completion.
498///
499/// [`tau_runtime_tokio::orchestration::persistence::spawn_writer`] is a
500/// detached `tokio::spawn`: nothing else in the process joins it, so
501/// without this call the writer can still be sitting on its first
502/// `rx.recv().await` when `main` returns and the tokio runtime is torn
503/// down — the process exits with `.tau/runs/<run_id>.jsonl` created
504/// (via `OpenOptions::create`) but zero bytes long, even though the run
505/// itself completed successfully. This is the same shape of bug as #650
506/// (a detached `tokio::spawn` writer dropped at runtime shutdown), just in
507/// this newer §13.5 writer rather than `WorkflowRunLogLayer`. Call sites
508/// await this only after every `Arc<dyn TraceSubscriber>` clone (the
509/// `ForwardingDispatcher` and its `RunState`) has gone out of scope, so
510/// the writer's channel is already closed and this simply waits for the
511/// backlog to drain and fsync rather than blocking forever.
512async fn flush_jsonl_writer(handle: tokio::task::JoinHandle<()>) {
513    if let Err(e) = handle.await {
514        tracing::warn!("run-log writer task panicked: {e}");
515    }
516}

518/// Drive `fut` to completion, joining the blocking execution-trace TUI
519/// alongside it when `tui_rx` is `Some` (spec §13.5). Shared by both of
520/// `run_via_ir`'s drive sites (`run_pipeline` and `run_ir`) so `--tui`
521/// covers pipeline bundles as well as single-agent ones — wiring only one
522/// site would leave `--tui` silently inert for pipeline bundles.
523///
524/// Mirrors `cmd/run.rs`'s multi-agent `--tui` join (`run.rs:366-411`):
525/// `run_tui` is a blocking raw-mode loop, so it moves to the blocking pool
526/// while `fut` keeps driving concurrently on the current task via
527/// `tokio::join!`. `fut` is deliberately never `tokio::spawn`ed — the
528/// interpreter's synthetic `RunState` is `!Send` (spec §13.3), and moving
529/// it to another task would violate that.
530///
531/// Error precedence matches `run.rs:389-411` LITERALLY (not just in spirit):
532/// when `fut` itself fails, that `RuntimeError` is what the operator needs
533/// to act on, so it becomes the returned error's root cause — but a
534/// concurrent TUI failure/panic is chained onto it via `anyhow::Context`
535/// rather than only logged, so it survives in the rendered error chain
536/// (`--debug` / the default one-line render both see it) instead of only a
537/// `tracing::warn!` line that may not be visible at the active log level.
538/// A TUI failure with no run failure is still surfaced as the sole error.
539async fn drive_with_optional_tui<F, T>(
540    fut: F,
541    tui_rx: Option<tokio::sync::mpsc::UnboundedReceiver<tau_ports::TraceEvent>>,
542) -> anyhow::Result<T>
543where
544    F: Future<Output = Result<T, RuntimeError>>,
545{
546    let Some(rx) = tui_rx else {
547        return fut.await.map_err(anyhow::Error::new);
548    };
549    let tui_task =
550        tokio::task::spawn_blocking(move || crate::tui::run_tui(crate::tui::TraceSource::Live(rx)));
551    let (run_res, tui_res) = tokio::join!(fut, tui_task);
552    match run_res {
553        Err(run_err) => {
554            let err = anyhow::Error::new(run_err);
555            let err = match tui_res {
556                Ok(Ok(())) => err,
557                Ok(Err(e)) => err.context(format!("execution-trace TUI also errored: {e}")),
558                Err(join_err) => err.context(format!(
559                    "execution-trace TUI task also panicked: {join_err}"
560                )),
561            };
562            Err(err)
563        }
564        Ok(value) => match tui_res {
565            Ok(Ok(())) => Ok(value),
566            Ok(Err(e)) => Err(e.context("execution-trace TUI")),
567            Err(join_err) => Err(anyhow::anyhow!(
568                "execution-trace TUI task panicked: {join_err}"
569            )),
570        },
571    }
572}

581/// The dispatcher owns:
582/// * `backend`: the LLM-backend handle the interpreter hands to each
583///   agent-loop construction (see `agent_loop::run_agent`).
584/// * `tools`: a `BTreeMap<ToolId, Arc<dyn DynTool>>` from IR ids to
585///   real plugin tool handles. Lookup is by ToolId (which matches
586///   `DynTool::name()` by lowering construction).
587///
588/// Unknown ToolIds surface as `RuntimeError::Internal` — there is no

726impl ToolDispatcher for ForwardingDispatcher {
727    /// Invoke a tool by [`ToolId`] and return a [`ToolInvocationResult`].
728    ///
729    /// # Runtime requirement
730    ///
731    /// This dispatcher hops onto `tokio::task::spawn_blocking` and then
732    /// drives the non-`Send` `DynTool::{init,invoke,teardown}` futures
733    /// via `tokio::runtime::Handle::current().block_on(...)` inside the
734    /// blocking closure. `block_on` requires a worker thread that is
735    /// not the only thread driving the runtime, so this **must be
736    /// called from a multi-thread tokio runtime** (the default for
737    /// `#[tokio::main]` / `Runtime::new()`). Calling from a
738    /// `current_thread` runtime will deadlock the blocking task on the
739    /// only available worker.
740    ///
741    /// # Body shape
742    ///
743    /// The joined-text result is round-tripped through
744    /// `serde_json::from_str` so structured tool output (e.g. a tool
745    /// that emits `{"temp":22}`) lands in `body` as a JSON object, not
746    /// as a `Value::String("{\"temp\":22}")`. Plain text (`hello`) is
747    /// not valid JSON, so the fallback `Value::String(joined_text)`
748    /// preserves it verbatim.
749    ///
750    /// That symmetry matters because
751    /// [`tau_runtime_core::interpreter::agent_loop`]'s
752    /// `DispatcherTool::invoke` (the inverse direction) special-cases
753    /// `Value::String` to extract the raw `String` and uses
754    /// `format!("{v}")` for structured shapes. Together this gives a
755    /// lossless round-trip in both directions: plain text → raw text,
756    /// structured shapes → compact JSON.
757    fn invoke<'a>(
758        &'a self,
759        tool_id: &'a ToolId,
760        args: &'a serde_json::Value,
761    ) -> Pin<Box<dyn Future<Output = Result<ToolInvocationResult, RuntimeError>> + Send + 'a>> {
762        // Clone `args` into the async block so the conversion error is
763        // surfaced as a real `RuntimeError` via `?` — silently mapping
764        // a failed conversion to `Value::Null` would corrupt every
765        // call site that passed a non-trivial argument.
766        let args_owned = args.clone();
767        let tool = self.tools.get(tool_id).cloned();
768        let tool_id_str = tool_id.0.clone();
769        // Native fallback (#639): only when no plugin handle serves this id.
770        let is_native = tool.is_none() && self.native.contains(tool_id);
771
772        Box::pin(async move {
773            // Native tools resolve from the statically-linked shared crate,
774            // keyed by ToolId — the SAME contract the wasm guest
775            // (`tau-wasm-guest/src/dispatcher.rs`) and the conformance
776            // profile use, so every profile returns byte-identical bodies.
777            // Handled before the domain-args conversion because native
778            // bodies take `serde_json::Value` directly.
779            if is_native {
780                return match tau_native_tools::invoke(&tool_id_str, &args_owned) {
781                    Some(body) => Ok(ToolInvocationResult {
782                        body: Some(body),
783                        error: None,
784                    }),
785                    // Declared Native but absent from the linked crate:
786                    // a build/install skew, not an empty result.
787                    None => Err(RuntimeError::Internal {
788                        message: format!(
789                            "IR declares tool {tool_id_str:?} as native but tau-native-tools \
790                             has no such body — the bundle's IR was built against a different \
791                             tau version than this host binary."
792                        ),
793                    }),
794                };
795            }
796
797            let domain_args: tau_domain::Value =
798                serde_json::from_value(args_owned).map_err(|e| RuntimeError::Internal {
799                    message: format!("argument conversion failed: {e}"),
800                })?;
801
802            let tool = tool.ok_or_else(|| RuntimeError::Internal {
803                message: format!(
804                    "ForwardingDispatcher: no tool registered for IR ToolId {tool_id_str:?}"
805                ),
806            })?;
807
808            // Mint a fresh SessionContext per invocation. The IR-driven
809            // path does not currently thread agent grants / deny entries
810            // through to plugin tools — this is symmetric with the
811            // cwd-based path at v0 (per-tool capability overrides are
812            // a deferred feature in plugin_loader.rs).
813            let ctx = SessionContext::new(AgentInstanceId::new(), uuid::Uuid::new_v4(), None);
814
815            // `DynTool::{init,invoke,teardown}` return non-`Send` futures
816            // (their concrete impls — e.g. `IpcTool` in
817            // `tau-runtime-tokio::plugin_host` — never explicitly annotate
818            // `+ Send` in the trait-object return type). The interpreter's
819            // `ToolDispatcher::invoke` contract, however, requires a
820            // `+ Send` future so `run_ir` can be Sync across worker
821            // boundaries.
822            //
823            // To bridge: hop onto `tokio::task::spawn_blocking` (which
824            // takes a `Send + 'static` closure that returns a `Send`
825            // value) and drive the non-`Send` futures inside via
826            // `Handle::current().block_on(...)`. The closure's captures
827            // (`Arc<dyn DynTool>`, `SessionContext`, `tau_domain::Value`)
828            // are all `Send + 'static`, and the resulting `ToolResult` is
829            // also `Send`, so the spawn_blocking signature is satisfied.
830            let tool_for_blocking = tool.clone();
831            let ctx_for_blocking = ctx.clone();
832            let tool_id_str_for_blocking = tool_id_str.clone();
833            let blocking = tokio::task::spawn_blocking(move || {
834                tokio::runtime::Handle::current().block_on(async move {
835                    tool_for_blocking
836                        .init(ctx_for_blocking.clone())
837                        .await
838                        .map_err(|e| RuntimeError::Internal {
839                            message: format!(
840                                "tool {tool_id_str_for_blocking:?} init failed: {e:?}"
841                            ),
842                        })?;
843                    let mut session: () = ();
844                    let result = tool_for_blocking
845                        .invoke(&ctx_for_blocking, &mut session, domain_args)
846                        .await
847                        .map_err(|e| RuntimeError::Internal {
848                            message: format!(
849                                "tool {tool_id_str_for_blocking:?} invoke failed: {e:?}"
850                            ),
851                        });
852                    // Best-effort teardown — never let cleanup turn a
853                    // successful invoke into a failure.
854                    let _ = tool_for_blocking.teardown(session).await;
855                    result
856                })
857            });
858
859            let result = blocking.await.map_err(|join_err| RuntimeError::Internal {
860                message: format!(
861                    "tool {tool_id_str:?} blocking task joined with error: {join_err}"
862                ),
863            })??;
864
865            // Joined text from every content block (Json blocks degrade
866            // to their Debug form to keep the contract simple — v0
867            // tool plugins overwhelmingly return Text).
868            let joined_text = result
869                .content
870                .iter()
871                .map(|c| match c {
872                    ToolContent::Text { text } => text.clone(),
873                    other => format!("{other:?}"),
874                })
875                .collect::<Vec<_>>()
876                .join("");
877
878            if result.is_error {
879                Ok(ToolInvocationResult {
880                    body: None,
881                    error: Some(joined_text),
882                })
883            } else {
884                // Try-parse the joined text as JSON first so structured
885                // tool output (a tool returning `{"temp":22}`) lands as
886                // a `Value::Object` — the inverse direction in
887                // `agent_loop::DispatcherTool::invoke` renders `body`
888                // via `format!("{v}")`, and `Display` on a JSON value
889                // produces compact JSON for objects/arrays but the
890                // literal *quoted* form for strings. Falling back to
891                // `Value::String` only when parsing fails preserves
892                // plain-text round-tripping (`"hello"` → `hello`).
893                let body = match serde_json::from_str::<serde_json::Value>(joined_text.trim()) {
894                    Ok(v) => v,
895                    Err(_) => serde_json::Value::String(joined_text),
896                };
897                Ok(ToolInvocationResult {
898                    body: Some(body),
899                    error: None,
900                })
901            }
902        })
903    }

1101/// directly so unit tests can exercise it without a live `McpClient`.
1102///
1103/// Returns `Err(RuntimeError::McpContractDriftAtBoot)` when the
1104/// canonical hash of `contract` differs from `entry.contract_hash`,
1105/// or `Err(RuntimeError::McpSetupFailed)` when hashing itself fails.
1106pub(crate) fn verify_hash_against_lockfile(
1107    entry: &LockedMcpEntry,
1108    contract: &tau_mcp::contract::ServerContract,
1109) -> Result<(), RuntimeError> {
1110    let actual_hash = canonical_hash(contract).map_err(|e| RuntimeError::McpSetupFailed {
1111        entry: entry.entry.clone(),
1112        reason: format!("canonical_hash failed: {e}"),
1113    })?;
1114    let actual_hex = hash_to_hex(&actual_hash);
1115    if actual_hex != entry.contract_hash {
1116        return Err(RuntimeError::McpContractDriftAtBoot {
1117            entry: entry.entry.clone(),
1118            expected_hash: entry.contract_hash.clone(),
1119            actual_hash: actual_hex,
1120        });
1121    }
1122    Ok(())
1123}

1154/// Build the sandbox [`CapabilityPlan`] for one MCP server entry.
1155///
1156/// The base grant is the author's `[tools.<entry>]` capability envelope:
1157/// `tau build` already refuses an envelope that does not cover every
1158/// server-tool's declared caps (`McpBuildError::EnvelopeCoversContract`),
1159/// so the envelope is the widest capability set the pinned contract can
1160/// exercise. The runtime drift check right after the handshake rejects a
1161/// server whose live contract no longer matches that pin.
1162///
1163/// Governed projects clamp the envelope against the `[allow.mcp.<entry>]`
1164/// host ceiling: `net.http` caps are met (lattice intersection) with the
1165/// ceiling hosts, so a net cap with no allowed host is dropped rather than
1166/// granted (fail-closed). Non-network caps pass through — the `[allow.mcp]`
1167/// registry carries only a host ceiling; fs/process grants are bounded by
1168/// the root `[allow]` ceiling at build time.
1169///
1170/// A governed project whose entry is missing from `[allow.mcp]` is a
1171/// build/run skew (`tau build` + `tau check` both refuse it) and fails
1172/// here rather than running the server unlisted.
1173fn mcp_capability_plan(
1174    entry: &str,
1175    envelope: &[tau_domain::Capability],
1176    allow: Option<&AllowConfig>,
1177) -> Result<CapabilityPlan, String> {
1178    let Some(allow) = allow else {
1179        // Ungoverned project (`--allow-ungoverned` / no `[allow]`): the
1180        // envelope alone is the grant.
1181        return Ok(CapabilityPlan::new(envelope.to_vec(), None, None));
1182    };
1183    let Some(mcp_allow) = allow.mcp.get(entry) else {
1184        return Err(format!(
1185            "governed project has no [allow.mcp.{entry}] registration — \
1186             build/check refuse this, so the lockfile and tau.toml have skewed"
1187        ));
1188    };
1189    // Ceiling shape: net.http over the registered hosts, any method.
1190    // Capability variants are only constructable via the unified parser.
1191    let ceiling: tau_domain::Capability = serde_json::from_value(serde_json::json!({
1192        "kind": "net.http",
1193        "hosts": mcp_allow.hosts,
1194    }))
1195    .map_err(|e| format!("[allow.mcp.{entry}].hosts is not a valid net.http ceiling: {e}"))?;
1196    let (net, mut caps): (Vec<tau_domain::Capability>, Vec<tau_domain::Capability>) = envelope
1197        .iter()
1198        .cloned()
1199        .partition(|c| matches!(c, tau_domain::Capability::Network(_)));
1200    caps.extend(tau_domain::meet(&net, &[ceiling]));
1201    Ok(CapabilityPlan::new(caps, None, None))
1202}

1204/// Per-server-tool narrowed authority (execution-trace TUI spec §12.3).
1205///
1206/// `declared` is the entry's **author envelope**
1207/// (`[tools.<entry>].capabilities`), not the server contract's `caps` —
1208/// see #712: a real handshake always reports empty contract caps, so
1209/// clamping against those could never fire outside a test.
1210///
1211/// `Some(effective)` iff the entry's meet-clamped [`CapabilityPlan`]
1212/// narrows the declared `net.http` caps — i.e. the declared net caps are
1213/// not a subset of the plan's. The effective set is the non-net declared
1214/// caps plus `meet(declared_net, plan_net)` (which may contain no net cap
1215/// at all after a fail-closed empty meet — still a clamp; the kernel
1216/// renders it `none`). `None` = not narrowed: no net caps declared,
1217/// ungoverned plan, or the ceiling already covers the declared hosts.
1218/// Only hosts narrow today — the `[allow.mcp]` registry is an any-method
1219/// host ceiling — but the comparison spans full net caps so a
1220/// method-carrying ceiling needs no rework here.
1221fn tool_effective_capabilities(
1222    declared: &[tau_domain::Capability],
1223    plan: &CapabilityPlan,
1224) -> Option<Vec<tau_domain::Capability>> {
1225    let (declared_net, declared_rest): (Vec<tau_domain::Capability>, Vec<tau_domain::Capability>) =
1226        declared
1227            .iter()
1228            .cloned()
1229            .partition(|c| matches!(c, tau_domain::Capability::Network(_)));
1230    if declared_net.is_empty() {
1231        return None;
1232    }
1233    let plan_net: Vec<tau_domain::Capability> = plan
1234        .capabilities
1235        .iter()
1236        .filter(|c| matches!(c, tau_domain::Capability::Network(_)))
1237        .cloned()
1238        .collect();
1239    if tau_domain::capability_subset(&declared_net, &plan_net).is_ok() {
1240        return None;
1241    }
1242    let mut effective = declared_rest;
1243    effective.extend(tau_domain::meet(&declared_net, &plan_net));
1244    Some(effective)
1245}

1247/// Boot the MCP runtime: per-entry handshake + drift check +
1248/// `WiredHostHandlers` + inbound dispatch + `McpBackedTool` registration.
1249///
1250/// Each stdio server is spawned under `gate` (the resolved sandbox
1251/// adapter) with the per-entry [`CapabilityPlan`] from
1252/// [`mcp_capability_plan`] enforced at the OS boundary.
1253///
1254/// Errors out before `ForwardingDispatcher` is constructed if any entry
1255/// fails (drift, network, parse). Returns an empty setup struct when
1256/// `lockfile.mcp_entries` is empty (non-MCP projects are unaffected).
1257pub(crate) async fn setup_mcp_runtime(
1258    config: &crate::config::ProjectConfig,
1259    lockfile: &LockFile,
1260    backend: Arc<dyn DynLlmBackend>,
1261    gate: Arc<dyn DynProcessCapabilityGate>,
1262) -> Result<McpRuntimeSetup, RuntimeError> {
1263    let mut tools: Vec<(tau_ir::ids::ToolId, Arc<dyn DynTool>)> = Vec::new();
1264    let mut inbound_handles: Vec<InboundDispatchHandle> = Vec::new();
1265
1266    for locked in &lockfile.mcp_entries {
1267        // Locate the corresponding tau.toml entry (sampling.models + roots).
1268        let tool_entry =
1269            config
1270                .tools
1271                .get(&locked.entry)
1272                .ok_or_else(|| RuntimeError::McpSetupFailed {
1273                    entry: locked.entry.clone(),
1274                    reason: format!(
1275                        "lockfile names entry {:?} but [tools.{}] missing in tau.toml",
1276                        locked.entry, locked.entry
1277                    ),
1278                })?;
1279
1280        let url = match &tool_entry.body {
1281            tau_pkg::project::project::ToolBody::Mcp(u) => u.clone(),
1282            other => {
1283                return Err(RuntimeError::McpSetupFailed {
1284                    entry: locked.entry.clone(),
1285                    reason: format!("[tools.{}] body is not Mcp: {other:?}", locked.entry),
1286                });
1287            }
1288        };
1289
1290        let sampling_models = tool_entry
1291            .sampling
1292            .as_ref()
1293            .map(|s| s.models.clone())
1294            .unwrap_or_default();
1295        let roots = tool_entry.roots.clone();
1296
1297        // Open the MCP server (handshake) under the resolved sandbox
1298        // adapter, with the per-entry plan enforced at the OS boundary
1299        // (stdio servers; HTTP/cassette transports spawn no process).
1300        let plan = mcp_capability_plan(
1301            &locked.entry,
1302            &tool_entry.capabilities,
1303            config.allow.as_ref(),
1304        )
1305        .map_err(|reason| RuntimeError::McpSetupFailed {
1306            entry: locked.entry.clone(),
1307            reason,
1308        })?;
1309        let client = mcp_open(&url, &plan, gate.clone(), McpClientOptions::default())
1310            .await
1311            .map_err(|e| RuntimeError::McpSetupFailed {
1312                entry: locked.entry.clone(),
1313                reason: format!("open failed: {e}"),
1314            })?;
1315
1316        // Drift check: live contract hash must match lockfile-recorded hash.
1317        verify_lockfile_against_live(locked, &client)?;
1318
1319        // Wrap in Arc; spawn inbound-dispatch task.
1320        let arc_client = Arc::new(client);
1321        let handlers = WiredHostHandlers::new(backend.clone(), sampling_models, roots);
1322        let handle = arc_client.start_inbound_dispatch(handlers);
1323        inbound_handles.push(handle);
1324
1325        // Per server-tool in the contract, register one McpBackedTool.
1326        for st in &arc_client.contract().tools {
1327            let ir_tool_id = tau_ir::ids::ToolId(format!("{}.{}", locked.entry, st.name));
1328            // #712: clamp against the AUTHOR ENVELOPE, not `st.caps`.
1329            // A real handshake always yields `st.caps == []` — MCP does not
1330            // standardize per-tool capability declaration and `McpTool` has
1331            // no extension field to carry a tau-specific one, so
1332            // `ServerContract::from_handshake`'s extractor has nothing to
1333            // read. Clamping against it made this `None` for every tool of
1334            // every real server. The envelope is also the more meaningful
1335            // basis: the server is the untrusted party, so "you declared
1336            // hosts X, governance clamped you to Y" is the statement a user
1337            // can act on.
1338            let effective = tool_effective_capabilities(&tool_entry.capabilities, &plan);
1339            let mcp_tool: Arc<dyn DynTool> = McpBackedTool::new(
1340                ir_tool_id.0.clone(),
1341                arc_client.clone(),
1342                st.name.clone(),
1343                st.caps.clone(),
1344                effective,
1345                st.input_schema.0.clone(),
1346                st.description.clone().unwrap_or_default(),
1347            );
1348            tools.push((ir_tool_id, mcp_tool));
1349        }
1350    }
1351
1352    Ok(McpRuntimeSetup {
1353        tools,
1354        inbound_handles,
1355    })
1356}
