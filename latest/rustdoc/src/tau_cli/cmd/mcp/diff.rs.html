1//! `tau mcp diff <name>` — read-only diff between pin file and live probe.
2//! Exit 0 if no drift, exit 64 if drift detected.
3
4use anyhow::{Context, Result};
5use tau_mcp::contract::pinned::PinnedContract;
6
7use crate::cli::McpDiffArgs;
8use crate::cmd::mcp::probe_only;
9use crate::output::Output;
10
11/// Run `tau mcp diff`.
12pub async fn run(args: McpDiffArgs, _output: &mut Output) -> Result<()> {
13    let project_root = std::env::current_dir().context("get cwd")?;
14    let pin_path = project_root
15        .join(".tau/mcp")
16        .join(format!("{}.contract.json", args.name));
17    let bytes = std::fs::read(&pin_path)
18        .with_context(|| format!("no pin file at {}", pin_path.display()))?;
19    let pinned: PinnedContract =
20        serde_json::from_slice(&bytes).with_context(|| format!("parse {}", pin_path.display()))?;
21
22    let live = probe_only(&args.name, None, &project_root)
23        .await
24        .context("probe live MCP server")?;
25
26    let drift = pinned.contract_hash_hex != live.contract_hash_hex;
27
28    if args.json {
29        let payload = serde_json::json!({
30            "name": &args.name,
31            "drift": drift,
32            "pin_hash": &pinned.contract_hash_hex,
33            "live_hash": &live.contract_hash_hex,
34            "pin_tools": pinned.contract.tools.len(),
35            "live_tools": live.contract.tools.len(),
36            "pin_server_version": &pinned.contract.server_info.version,
37            "live_server_version": &live.contract.server_info.version,
38        });
39        println!("{}", serde_json::to_string_pretty(&payload)?);
40    } else if !drift {
41        println!("no drift: `{}` matches live server", args.name);
42    } else {
43        println!("DRIFT detected for `{}`:", args.name);
44        println!("  pin hash:  {}", pinned.contract_hash_hex);
45        println!("  live hash: {}", live.contract_hash_hex);
46        println!(
47            "  pin tools / live tools: {} / {}",
48            pinned.contract.tools.len(),
49            live.contract.tools.len()
50        );
51        println!(
52            "  pin server / live server: {} v{} / {} v{}",
53            pinned.contract.server_info.name,
54            pinned.contract.server_info.version,
55            live.contract.server_info.name,
56            live.contract.server_info.version,
57        );
58    }
59
60    if drift {
61        std::process::exit(64);
62    }
63    Ok(())
64}
