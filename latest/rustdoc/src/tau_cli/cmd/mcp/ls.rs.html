1//! `tau mcp ls` — list pinned MCP contracts in the current project.
2//!
3//! See spec at `docs/superpowers/specs/2026-06-01-beta-3-mcp-facilitator-design.md`
4//! §10 (CLI surface) and ADR-0038.
5
6use std::path::PathBuf;
7
8use anyhow::{Context, Result};
9use serde::Serialize;
10use tau_mcp::contract::pinned::PinnedContract;
11
12use crate::cli::McpLsArgs;
13use crate::output::Output;
14
15#[derive(Serialize)]
16struct PinSummary {
17    name: String,
18    url: String,
19    server_name: String,
20    tools_count: usize,
21    contract_hash_hex: String,
22    path: PathBuf,
23}
24
25/// Run `tau mcp ls`.
26pub async fn run(args: McpLsArgs, _output: &mut Output) -> Result<()> {
27    let project_root = std::env::current_dir().context("get cwd")?;
28    let pin_dir = project_root.join(".tau/mcp");
29    let mut pins = Vec::new();
30    if pin_dir.is_dir() {
31        for entry in
32            std::fs::read_dir(&pin_dir).with_context(|| format!("read {}", pin_dir.display()))?
33        {
34            let entry = entry?;
35            let path = entry.path();
36            let Some(name) = path
37                .file_name()
38                .and_then(|s| s.to_str())
39                .and_then(|s| s.strip_suffix(".contract.json"))
40                .map(String::from)
41            else {
42                continue;
43            };
44            let bytes = std::fs::read(&path).with_context(|| format!("read {}", path.display()))?;
45            let pinned: PinnedContract = serde_json::from_slice(&bytes)
46                .with_context(|| format!("parse {}", path.display()))?;
47            pins.push(PinSummary {
48                name,
49                url: pinned.url.clone(),
50                server_name: pinned.contract.server_info.name.clone(),
51                tools_count: pinned.contract.tools.len(),
52                contract_hash_hex: pinned.contract_hash_hex.clone(),
53                path,
54            });
55        }
56        pins.sort_by(|a, b| a.name.cmp(&b.name));
57    }
58
59    if args.json {
60        let payload = serde_json::json!({ "pins": pins });
61        println!("{}", serde_json::to_string_pretty(&payload)?);
62    } else if pins.is_empty() {
63        println!("no pinned MCP contracts (run `tau mcp pin <name>`)");
64    } else {
65        for p in &pins {
66            println!(
67                "{:24} {:8} {} → {} ({} tools, hash {})",
68                p.name,
69                "MCP",
70                p.url,
71                p.server_name,
72                p.tools_count,
73                &p.contract_hash_hex[..16],
74            );
75        }
76    }
77    Ok(())
78}
