1//! `tau mcp show <name>` — show a pinned MCP contract.
2
3use anyhow::{Context, Result};
4use tau_mcp::contract::pinned::PinnedContract;
5
6use crate::cli::McpShowArgs;
7use crate::cmd::mcp::{render_sarif, OutputFormat};
8use crate::output::Output;
9
10/// Run `tau mcp show`.
11pub async fn run(args: McpShowArgs, _output: &mut Output) -> Result<()> {
12    let project_root = std::env::current_dir().context("get cwd")?;
13    let pin_path = project_root
14        .join(".tau/mcp")
15        .join(format!("{}.contract.json", args.name));
16    let bytes = std::fs::read(&pin_path)
17        .with_context(|| format!("no pin file at {}", pin_path.display()))?;
18    let pinned: PinnedContract =
19        serde_json::from_slice(&bytes).with_context(|| format!("parse {}", pin_path.display()))?;
20
21    match OutputFormat::from_flags(args.json, args.sarif) {
22        OutputFormat::Json => {
23            println!("{}", serde_json::to_string_pretty(&pinned)?);
24        }
25        OutputFormat::Sarif => {
26            let payload = serde_json::to_value(&pinned)?;
27            let sarif = render_sarif("tau-mcp/show", payload);
28            println!("{}", serde_json::to_string_pretty(&sarif)?);
29        }
30        OutputFormat::Human => {
31            println!("name:       {}", args.name);
32            println!("url:        {}", pinned.url);
33            println!(
34                "server:     {} v{}",
35                pinned.contract.server_info.name, pinned.contract.server_info.version
36            );
37            println!("hash:       {}", pinned.contract_hash_hex);
38            println!("tools:      {}", pinned.contract.tools.len());
39            for t in &pinned.contract.tools {
40                println!("  - {}", t.name);
41            }
42        }
43    }
44    Ok(())
45}
