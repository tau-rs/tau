1//! `tau build` — see spec `2026-05-27-tau-build-design.md`.
2//!
3//! Thin CLI shim over [`tau_pkg::bundle::build`](tau_pkg::bundle::build()). Resolves the project
4//! root from the current directory, calls the bundle builder with the
5//! host target + default output path, prints progress to stderr and
6//! the bundle path to stdout, then exits with the appropriate code
7//! per spec §6 (0 success, 2 config/parse, 3 install-state, 70 internal).

22/// CLI entry point for `tau build`. The function is async to match the
23/// dispatcher's signature; MCP contract resolution requires async I/O.
24pub async fn run(args: &BuildArgs, output: &mut Output) -> Result<()> {
25    // Resolve the project path: prefer the CLI positional arg, fall back to cwd.
26    let project_path = match &args.project {
27        Some(p) => p.clone(),
28        None => match std::env::current_dir() {
29            Ok(p) => p,
30            Err(e) => {
31                let _ = output.error(format!("cannot determine current directory: {e}"));
32                std::process::exit(70);
33            }
34        },
35    };
36
37    // Resolve the project root and optionally extract a ProjectConfig from a
38    // `.ts` source (β.8). For TOML-based projects the root IS the project
39    // directory; for `.ts` projects it is the parent directory.
40    let (project_root, ts_project) = {
41        let ext = project_path.extension().and_then(|s| s.to_str());
42        if project_path.is_file() && ext == Some("ts") {
43            match crate::cmd::project_load::load_project(&project_path) {
44                Ok(loaded) => (loaded.project_root, Some(loaded.project)),
45                Err(e) => {
46                    let _ = output.error(format!("{e}"));
47                    std::process::exit(2);
48                }
49            }
50        } else {
51            let root = if project_path.is_dir() {
52                project_path.clone()
53            } else {
54                project_path
55                    .parent()
56                    .map(|p| p.to_path_buf())
57                    .unwrap_or_else(|| project_path.clone())
58            };
59            (root, None)
60        }
61    };
62
63    let build_target = match resolve_target(args) {
64        Ok(t) => t,
65        Err(msg) => {
66            let _ = output.error(msg);
67            std::process::exit(2);
68        }
69    };
70    // Artifact-kind targets (EPIC 5.1) branch to their own pipelines; the
71    // bundle path below runs only for a hardware triple / default host.
72    let target = match build_target {
73        BuildTarget::Bundle(t) => t,
74        BuildTarget::WasmGuest => return dispatch_wasm_guest(args, output).await,
75        BuildTarget::RustLib => return dispatch_rust_lib(args, output).await,
76    };
77
78    // Governed-by-default gate (ADR-0057 / D2). Evaluate the `[allow]`
79    // constitution BEFORE any network I/O (MCP resolution): an absent ceiling
80    // is a hard error (GOV000) unless `--allow-ungoverned`, and a declared-
81    // but-violated ceiling refuses the build. The verdict is stamped into the
82    // bundle so `tau run --bundle` can enforce governed-by-default on its end.
83    let governance = evaluate_build_governance(&project_path, &target, args, output).await;
84
85    // Map the repeatable `--agent` flag to the builder's filter. Empty
86    // → None (build all). Parse each id to AgentId; a malformed id is a
87    // config-level input error (exit 2).
88    let agent_filter = if args.agents.is_empty() {
89        None
90    } else {
91        let mut parsed = Vec::with_capacity(args.agents.len());
92        for raw in &args.agents {
93            match raw.parse::<tau_domain::AgentId>() {
94                Ok(id) => parsed.push(id),
95                Err(e) => {
96                    let _ = output.error(format!("invalid agent id '{raw}': {e}"));
97                    std::process::exit(2);
98                }
99            }
100        }
101        Some(parsed)
102    };
103
104    // Resolve MCP contracts (pinned or live) before lowering the IR.
105    // This is async; the result is passed into lower_ir as a sync cache.
106    let (mcp_entries_meta, mcp_cache_ir) =
107        match resolve_mcp_cache(&project_root, args.offline).await {
108            Ok(v) => v,
109            Err(e) => {
110                let _ = output.error(format!("{e}"));
111                std::process::exit(2);
112            }
113        };
114
115    // Lower the project IR. Load the project config (same pipeline the
116    // bundle builder uses), then call lower_project with a deterministic
117    // SHA-256-of-name cache for native tools (see `lower_ir`'s doc-comment
118    // for the forward-stability semantic). On IrError, render a human-
119    // readable diagnostic and exit 2.
120    //
121    // For `.ts` projects `ts_project` is already parsed; pass it through so
122    // `lower_ir` does not try to read a non-existent `tau.toml`.
123    let LowerIrResult {
124        payload: ir_payload,
125        triggers: trigger_bindings,
126        lower_error,
127        assets: ir_assets,
128    } = lower_ir(&project_root, &target, &mcp_cache_ir, ts_project.as_ref());
129
130    // A typecheck/lowering error (e.g. an invalid context pipeline where
131    // `fit_budget` is not the last step) is a hard build failure: surface
132    // the diagnostic and exit 2 rather than silently dropping the IR payload.
133    if let Some(e) = lower_error {
134        let _ = output.error(format!("{e}"));
135        std::process::exit(2);
136    }
137
138    let opts = BuildOptions {
139        project_root: project_root.clone(),
140        target,
141        output_path: args.output.clone(),
142        agent_filter,
143        ir_payload,
144        governance,
145        assets: ir_assets_to_bundle(ir_assets),
146    };
147
148    let _ = output.status("Building bundle…");
149
150    match build(opts) {
151        Ok(artifact) => {
152            // After a successful build, persist MCP entries to tau.lock.
153            if !mcp_entries_meta.is_empty() {
154                let lockfile_path = project_root.join("tau.lock");
155                match tau_pkg::lockfile::LockFile::load(&lockfile_path) {
156                    Ok(mut lf) => {
157                        lf.mcp_entries = mcp_entries_meta;
158                        if let Err(e) = lf.save(&lockfile_path) {
159                            tracing::warn!("failed to write mcp_entries to tau.lock: {e}");
160                        }
161                    }
162                    Err(e) => {
163                        tracing::warn!("failed to reload tau.lock for mcp_entries: {e}");
164                    }
165                }
166            }
167            emit_artifact(&artifact, output);
168            if let Some(adapter) = &args.emit_trigger {
169                emit_trigger_descriptors(adapter, &trigger_bindings, &artifact, output);
170            }
171            Ok(())
172        }
173        Err(e) => {
174            let _ = output.error(format!("{e}"));
175            std::process::exit(exit_code_for(&e) as i32);
176        }
177    }
178}

180/// Run the governed-by-default gate (ADR-0057 / D2) for `tau build`.
181///
182/// Loads the project through the same `CheckCtx` the `tau check governance`
183/// category uses, evaluates the `[allow]` constitution against the build
184/// flags, and either returns the [`GovernanceRecord`] to stamp into the
185/// bundle or terminates the process:
186///
187/// - `NoConstitution` → prints `GOV000` and exits 2.
188/// - `Violations`     → prints the refused-build diagnostic and exits 2.
189/// - malformed project → returns `None`; the bundle builder surfaces the
190///   precise parse/validation error (also exit 2) rather than a vague GOV000.
191async fn evaluate_build_governance(
192    project_path: &std::path::Path,
193    target: &TargetTriple,
194    args: &BuildArgs,
195    output: &mut Output,
196) -> Option<GovernanceRecord> {
197    use crate::cmd::check::{
198        evaluate_governance, render_no_constitution, render_violations, CheckCtx, GovernanceFlags,
199        GovernanceOutcome,
200    };
201
202    let flags = GovernanceFlags {
203        allow_ungoverned: args.allow_ungoverned,
204        no_governance: args.no_governance,
205    };
206    let ctx = match CheckCtx::load(project_path.to_path_buf(), false, Some(*target)).await {
207        Ok(c) => c,
208        Err(e) => {
209            let _ = output.error(format!("cannot evaluate governance: {e}"));
210            std::process::exit(2);
211        }
212    };
213    let Some(project) = &ctx.project else {
214        // Unparseable project — defer to the bundle builder's precise error.
215        return None;
216    };
217    match evaluate_governance(project, &ctx, flags) {
218        GovernanceOutcome::Proceed(verdict) => Some(GovernanceRecord { verdict }),
219        GovernanceOutcome::NoConstitution => {
220            let _ = output.diagnostic(render_no_constitution());
221            std::process::exit(2);
222        }
223        GovernanceOutcome::Violations(findings) => {
224            let _ = output.diagnostic(render_violations(&findings));
225            std::process::exit(2);
226        }
227    }
228}

230/// Discover all MCP entries from tau.toml and resolve their contracts.
231///
232/// Returns:
233/// - `Vec<McpEntryMeta>` — per-entry metadata for writing to `tau.lock`.
234/// - `BTreeMap<String, tau_ir_lower::ResolvedMcpContract>` — URL-keyed
235///   cache for `Caches::mcp_contract`.
236///
237/// On `--offline`, reads `.tau/mcp/<entry>.contract.json` (error if missing).
238/// On the live path, performs MCP handshakes and writes pinned files.
239async fn resolve_mcp_cache(
240    project_root: &std::path::Path,
241    offline: bool,
242) -> anyhow::Result<(
243    Vec<LockedMcpEntry>,
244    BTreeMap<String, tau_ir_lower::ResolvedMcpContract>,
245)> {
246    use tau_pkg::project::project::{ToolBody, UncheckedProjectConfig};
247
248    // Parse tau.toml to find MCP entries.
249    let tau_toml_path = project_root.join("tau.toml");
250    let tau_toml_str = match std::fs::read_to_string(&tau_toml_path) {
251        Ok(s) => s,
252        Err(_) => {
253            // No tau.toml → no MCP entries. lower_ir will warn separately.
254            return Ok((Vec::new(), BTreeMap::new()));
255        }
256    };
257    let unchecked: UncheckedProjectConfig = match toml::from_str(&tau_toml_str) {
258        Ok(u) => u,
259        Err(_) => {
260            // Parse failure → no MCP entries. lower_ir will warn separately.
261            return Ok((Vec::new(), BTreeMap::new()));
262        }
263    };
264    let config = match unchecked.validate() {
265        Ok(c) => c,
266        Err(_) => {
267            return Ok((Vec::new(), BTreeMap::new()));
268        }
269    };
270
271    // Collect all MCP entries.
272    let mcp_entries: Vec<(String, String)> = config
273        .tools
274        .iter()
275        .filter_map(|(name, t)| match &t.body {
276            ToolBody::Mcp(url) => Some((name.clone(), url.clone())),
277            _ => None,
278        })
279        .collect();
280
281    if mcp_entries.is_empty() {
282        return Ok((Vec::new(), BTreeMap::new()));
283    }
284
285    let pin_base = project_root.join(".tau").join("mcp");
286
287    if offline {
288        // Pinned path: read `.tau/mcp/<entry>.contract.json`.
289        use tau_mcp::contract::McpContractResolver as _;
290        let resolver = tau_mcp::contract::resolver::PinnedResolver::new(&pin_base);
291        let mut ir_cache: BTreeMap<String, tau_ir_lower::ResolvedMcpContract> = BTreeMap::new();
292        let mut locked_entries: Vec<LockedMcpEntry> = Vec::new();
293        for (entry, url) in &mcp_entries {
294            let resolved = resolver
295                .resolve(entry, url)
296                .map_err(|e| anyhow::anyhow!("MCP pin resolve failed for {entry:?}: {e}"))?;
297            locked_entries.push(mcp_entry_to_locked(
298                entry,
299                url,
300                &resolved,
301                Some(format!(".tau/mcp/{entry}.contract.json")),
302            ));
303            ir_cache.insert(url.clone(), to_ir_shape(resolved));
304        }
305        Ok((locked_entries, ir_cache))
306    } else {
307        // Live path: perform MCP handshakes.
308        let inputs: Vec<tau_mcp_tokio::resolver::McpEntryInput> = mcp_entries
309            .iter()
310            .map(|(entry, url)| tau_mcp_tokio::resolver::McpEntryInput {
311                entry: entry.clone(),
312                url: url.clone(),
313                plan: tau_ports::CapabilityPlan::new(Vec::new(), None, None),
314            })
315            .collect();
316        let live = tau_mcp_tokio::resolver::resolve_all(inputs)
317            .await
318            .map_err(|e| anyhow::anyhow!("MCP live resolve failed: {e}"))?;
319
320        // Write pinned files for next-time --offline.
321        std::fs::create_dir_all(&pin_base)
322            .map_err(|e| anyhow::anyhow!("failed to create .tau/mcp/: {e}"))?;
323        for (entry, url) in &mcp_entries {
324            if let Some(lr) = live.get(url) {
325                let path = pin_base.join(format!("{entry}.contract.json"));
326                let bytes = serde_json::to_vec_pretty(&lr.pinned)
327                    .map_err(|e| anyhow::anyhow!("serialize pinned contract for {entry:?}: {e}"))?;
328                std::fs::write(&path, bytes)
329                    .map_err(|e| anyhow::anyhow!("write {}: {e}", path.display()))?;
330            }
331        }
332
333        let mut ir_cache: BTreeMap<String, tau_ir_lower::ResolvedMcpContract> = BTreeMap::new();
334        let mut locked_entries: Vec<LockedMcpEntry> = Vec::new();
335        for (entry, url) in &mcp_entries {
336            if let Some(lr) = live.get(url) {
337                locked_entries.push(mcp_entry_to_locked(
338                    entry,
339                    url,
340                    &lr.resolved,
341                    Some(format!(".tau/mcp/{entry}.contract.json")),
342                ));
343                ir_cache.insert(url.clone(), to_ir_shape(lr.resolved.clone()));
344            }
345        }
346        Ok((locked_entries, ir_cache))
347    }
348}

350/// Convert a `tau_mcp` resolver output to tau-ir's structurally-identical type.
351fn to_ir_shape(
352    r: tau_mcp::contract::resolver::ResolvedMcpContract,
353) -> tau_ir_lower::ResolvedMcpContract {
354    use tau_ir_lower::{ResolvedMcpContract as IrR, ResolvedServerTool as IrS};
355    IrR {
356        hash: r.hash,
357        expanded_tools: r
358            .expanded_tools
359            .into_iter()
360            .map(|t| IrS {
361                name: t.name,
362                // v0 lossy: caps is `CapabilityRequirements` in tau-ir (structured),
363                // `Vec<String>` in tau-mcp (wire kind names). Use empty default;
364                // the author's envelope is the source of truth at build time,
365                // and runtime drift-check uses the lockfile.
366                caps: tau_ir::capability::CapabilityRequirements::default(),
367                input_schema: t.input_schema,
368            })
369            .collect(),
370        requires_sampling: r.requires_sampling,
371    }
372}

374/// Build a `LockedMcpEntry` from a resolved contract + metadata.
375fn mcp_entry_to_locked(
376    entry: &str,
377    url: &str,
378    resolved: &tau_mcp::contract::resolver::ResolvedMcpContract,
379    pinned_contract: Option<String>,
380) -> LockedMcpEntry {
381    let expanded_tools = resolved
382        .expanded_tools
383        .iter()
384        .map(|st| {
385            LockedMcpExpandedTool::new(
386                st.name.clone(),
387                st.caps.clone(),
388                schema_hash_json(&st.input_schema),
389            )
390        })
391        .collect();
392    LockedMcpEntry::new(
393        entry.to_owned(),
394        url.to_owned(),
395        hex_lower(&resolved.hash),
396        pinned_contract,
397        expanded_tools,
398    )
399}

401/// SHA-256 of a JSON value's canonical bytes (compact serialization).
402fn schema_hash_json(v: &serde_json::Value) -> String {
403    use sha2::{Digest, Sha256};
404    let bytes = serde_json::to_vec(v).unwrap_or_default();
405    let mut h = Sha256::new();
406    h.update(&bytes);
407    hex_lower(&h.finalize())
408}

410/// Result of [`lower_ir`]: the optional IR payload to embed in the bundle,
411/// plus the lowered trigger bindings (consumed by `--emit-trigger`).
412pub(crate) struct LowerIrResult {
413    /// `Some` when lowering succeeded; `None` when it failed (the bundle is
414    /// still built, just without an IR payload).
415    pub payload: Option<IrPayload>,
416    /// Trigger bindings lowered from the project config (empty when lowering
417    /// failed or the project declares no triggers).
418    pub triggers: Vec<tau_ir::trigger::TriggerBinding>,
419    /// `Some` when `lower_project` returned a typecheck/lowering error. The
420    /// payload is `None` in that case. Callers that treat lowering as a
421    /// best-effort enrichment (verify, run, dev) ignore this; `tau build`
422    /// inspects it and rejects the build (exit 2) so a typecheck failure —
423    /// e.g. an invalid context pipeline — is surfaced at build time rather
424    /// than silently dropped (see ADR: build-time enforcement discipline).
425    pub lower_error: Option<tau_ir_lower::LowerError>,
426    /// Content-addressed assets (currently `system_file` prompts) the module
427    /// references, keyed by hash (`"sha256:" + 64 hex`). Empty when lowering
428    /// failed or the project uses only inline prompts. `tau build` persists
429    /// these into the bundle's asset store; verify/dev use them to resolve
430    /// prompt references at run time (D6-B).
431    pub assets: BTreeMap<String, tau_ir::asset::AssetBlob>,
432}

434/// Attempt to lower the project IR, returning `Some(IrPayload)` on
435/// success or `None` if lowering fails (non-fatal — the bundle is still
436/// built, but without an IR payload; a warning is logged).
437///
438/// Native-tool content hashes are derived from `sha2::Sha256(symbolic_name)`.
439/// This is a deterministic, non-zero stand-in until an actual native-tool
440/// registry lands in `tau-pkg`: when that registry exists, replace the
441/// `native_tool` closure with the registry's source-content hash. Bundles
442/// produced before the switch will rebuild on the next `tau build` because
443/// their `canonical_ir_hash` will change — that's the honest forward-stability
444/// semantic (D-6): a change in tool identity is a change in workflow identity.
445///
446/// `mcp_cache` is keyed by MCP URL and populated by `resolve_mcp_cache`.
447///
448/// `preloaded_config` — when `Some`, the config is used directly instead of
449/// reading `tau.toml` from `project_root`. Used by the `.ts` source path
450/// (β.8) where there is no `tau.toml`.
451pub(crate) fn lower_ir(
452    project_root: &std::path::Path,
453    target: &TargetTriple,
454    mcp_cache: &BTreeMap<String, tau_ir_lower::ResolvedMcpContract>,
455    preloaded_config: Option<&tau_pkg::project::ProjectConfig>,
456) -> LowerIrResult {
457    use tau_pkg::project::project::UncheckedProjectConfig;
458
459    let config_owned;
460    let config = if let Some(c) = preloaded_config {
461        c
462    } else {
463        let tau_toml_path = project_root.join("tau.toml");
464        let tau_toml_str = match std::fs::read_to_string(&tau_toml_path) {
465            Ok(s) => s,
466            Err(e) => {
467                tracing::warn!("IR lowering: failed to read tau.toml: {e}");
468                return LowerIrResult {
469                    payload: None,
470                    triggers: Vec::new(),
471                    lower_error: None,
472                    assets: BTreeMap::new(),
473                };
474            }
475        };
476        let unchecked: UncheckedProjectConfig = match toml::from_str(&tau_toml_str) {
477            Ok(u) => u,
478            Err(e) => {
479                tracing::warn!("IR lowering: failed to parse tau.toml: {e}");
480                return LowerIrResult {
481                    payload: None,
482                    triggers: Vec::new(),
483                    lower_error: None,
484                    assets: BTreeMap::new(),
485                };
486            }
487        };
488        config_owned = match unchecked.validate() {
489            Ok(c) => c,
490            Err(e) => {
491                tracing::warn!("IR lowering: project config validation failed: {e}");
492                return LowerIrResult {
493                    payload: None,
494                    triggers: Vec::new(),
495                    lower_error: None,
496                    assets: BTreeMap::new(),
497                };
498            }
499        };
500        &config_owned
501    };
502
503    // Deterministic stand-in cache (β.2.6.1): hash the symbolic name with
504    // SHA-256. Two distinct names produce two distinct hashes (so future
505    // drift detection actually works), the value is non-zero (typecheck's
506    // sentinel check passes), and it's stable across builds (so the IR
507    // module hash stays reproducible). When a real native-tool registry
508    // lands in tau-pkg, replace this closure with the registry's
509    // source-content hash — see this fn's doc-comment.
510    let caches = tau_ir_lower::Caches {
511        native_tool: &|name: &str| Some(sha256_name(name)),
512        mcp_contract: &|url| mcp_cache.get(url).cloned(),
513        skill: &|_name| None,
514        // D6-B: read `system_file` prompts at build time (missing/unreadable
515        // => LowerError::PromptFileUnreadable => build fails). Routed through
516        // tau-pkg's `read_prompt_file` so the IR asset hash is computed over
517        // the same bytes as the bundle's `system_prompt_sha256`.
518        prompt_file: &|p: &std::path::Path| {
519            tau_pkg::bundle::read_prompt_file(p, project_root)
520                .map_err(|e| tau_ir_lower::PromptFileError(e.to_string()))
521        },
522    };
523
524    match tau_ir_lower::lower_project(config, target, &caches) {
525        Ok(out) => {
526            let bytes = tau_ir::to_canonical_bytes(&out.module);
527            let hash_bytes = tau_ir::compute_hash(&out.module);
528            let payload = Some(IrPayload {
529                ir_format: out.module.ir_format.0.clone(),
530                canonical_ir_hash: hex_lower(&hash_bytes),
531                canonical_ir_bytes_hex: hex_lower(&bytes),
532            });
533            LowerIrResult {
534                payload,
535                triggers: out.module.triggers,
536                lower_error: None,
537                assets: out.assets,
538            }
539        }
540        Err(e) => {
541            tracing::warn!("IR lowering failed: {e}");
542            LowerIrResult {
543                payload: None,
544                triggers: Vec::new(),
545                lower_error: Some(e),
546                assets: BTreeMap::new(),
547            }
548        }
549    }
550}

552/// Convert the asset blobs `tau_ir_lower` collected into the bundle's
553/// `[[assets]]` shape (bytes hex-encoded), sorted by hash for determinism.
554/// Shared by `tau build` and `tau verify --bundle` so both derive an
555/// identical asset store from the same source (D6-B).
556pub(crate) fn ir_assets_to_bundle(
557    ir_assets: BTreeMap<String, tau_ir::asset::AssetBlob>,
558) -> Vec<tau_pkg::bundle::manifest::BundleAsset> {
559    ir_assets
560        .into_iter()
561        .map(|(hash, blob)| tau_pkg::bundle::manifest::BundleAsset {
562            hash,
563            kind: blob.kind.as_str().to_string(),
564            bytes_hex: hex_lower(&blob.bytes),
565        })
566        .collect()
567}

569/// `Caches::native_tool`-shaped stand-in: `Some(SHA-256(name))`.
570///
571/// Shared by [`lower_ir`] (bundle build) and `cmd::run`'s pipeline
572/// lowering so both compute the SAME native-tool content hash — a drift
573/// between the two would make a pipeline's runtime IR diverge from its
574/// built bundle IR. Always `Some` (never the unknown-tool sentinel).
575pub(crate) fn native_tool_hash(name: &str) -> Option<[u8; 32]> {
576    Some(sha256_name(name))
577}

579/// Deterministic content-hash stand-in for a native tool's symbolic name.
580///
581/// Returns `SHA-256(name.as_bytes())`. Used by [`lower_ir`]'s `Caches::native_tool`
582/// closure until a real native-tool registry lands in `tau-pkg`. Distinct
583/// names always produce distinct hashes, and the value is non-zero so
584/// `tau_ir_lower::typecheck` won't reject it as the unknown-tool sentinel.
585fn sha256_name(name: &str) -> [u8; 32] {
586    use sha2::{Digest, Sha256};
587    let mut h = Sha256::new();
588    h.update(name.as_bytes());
589    h.finalize().into()
590}

592/// Encode a byte slice as lowercase hex.
593pub(crate) fn hex_lower(bytes: &[u8]) -> String {
594    const HEX: &[u8] = b"0123456789abcdef";
595    let mut out = String::with_capacity(bytes.len() * 2);
596    for b in bytes {
597        out.push(HEX[(b >> 4) as usize] as char);
598        out.push(HEX[(b & 0x0f) as usize] as char);
599    }
600    out
601}

603/// The selected build artifact kind (EPIC 5.1). `--target` accepts two
604/// artifact-kind keywords resolved ahead of hardware-triple parsing; any
605/// other value is a hardware triple producing a `.tau` bundle.
606#[derive(Debug)]
607pub(crate) enum BuildTarget {
608    /// Default / hardware triple → `.tau` bundle.
609    Bundle(TargetTriple),
610    /// `--target wasm-guest` → fully-linked wasm component.
611    WasmGuest,
612    /// `--target rust-lib` → generated no_std Rust library crate.
613    RustLib,
614}
615
616/// Resolve the build target. Keywords `wasm-guest`/`rust-lib` select an
617/// embedding artifact; `None` → host bundle; anything else is parsed as an
618/// Available triple (ADR-0034). Returns a human-readable error on invalid input.
619fn resolve_target(args: &BuildArgs) -> Result<BuildTarget, String> {
620    match args.target.as_deref() {
621        None => Ok(BuildTarget::Bundle(TargetTriple::host())),
622        Some("wasm-guest") => Ok(BuildTarget::WasmGuest),
623        Some("rust-lib") => Ok(BuildTarget::RustLib),
624        Some(s) => {
625            let triple: TargetTriple = s.parse().map_err(|e| {
626                format!(
627                    "invalid --target '{s}': {e}. Expected an artifact kind \
628                     (wasm-guest, rust-lib) or an Available triple: {}",
629                    available_triples_joined(),
630                )
631            })?;
632            let available = tau_ports::target::lookup(&triple)
633                .is_some_and(|e| matches!(e.status, tau_ports::target::TripleStatus::Available));
634            if !available {
635                return Err(format!(
636                    "target '{triple}' is not an Available build target. Expected an \
637                     artifact kind (wasm-guest, rust-lib) or an Available triple: {}",
638                    available_triples_joined(),
639                ));
640            }
641            Ok(BuildTarget::Bundle(triple))
642        }
643    }
644}

646/// Test seam: classify a `--target` value into its artifact-kind label without
647/// running a build. Returns `"bundle" | "wasm-guest" | "rust-lib" | "invalid"`.
648pub fn classify_target_for_test(target: Option<&str>) -> &'static str {

667/// Emit the rust-lib embedding crate for `project` into `out_dir`. Test seam:
668/// lowers for `any-wasi-strict` (cap-fit refuses ProcessExec/AgentSpawn), derives
669/// the WIT world, renders the scaffold, and writes it. Does not run governance
670/// or touch `Output` — the CLI dispatch wraps this.
671pub fn emit_rust_lib_to(
672    project: &std::path::Path,
673    out_dir: &std::path::Path,
674) -> Result<RustLibArtifact> {
675    use crate::cmd::build_wasm::{lower_to_wasm_ir, world_from_module};
676    let (module, bytes) = lower_to_wasm_ir(project)?;
677    let ir_hash = hex_lower(&tau_ir::compute_hash(&module));
678    let wit = world_from_module(&module)?;
679    let stem = project
680        .file_name()
681        .and_then(|n| n.to_str())
682        .unwrap_or("workflow");
683    let crate_name = sanitize_crate_name(stem);
684
685    let files = tau_sdk_codegen::render_rust_lib(tau_sdk_codegen::RustLibInput {
686        crate_name: &crate_name,
687        ir_bytes: &bytes,
688        ir_hash: &ir_hash,
689        wit: &wit,
690        tau_version: env!("CARGO_PKG_VERSION"),
691    });
692
693    let mut written = 0usize;
694    for (rel, contents) in &files {
695        let path = out_dir.join(rel);
696        if let Some(parent) = path.parent() {
697            std::fs::create_dir_all(parent)?;
698        }
699        std::fs::write(&path, contents)?;
700        written += 1;
701    }
702    Ok(RustLibArtifact {
703        out_dir: out_dir.to_path_buf(),
704        ir_hash,
705        files: written,
706    })
707}

719/// Lowercase, replace non-alphanumeric with `_`, so the stem is a valid crate name.
720fn sanitize_crate_name(stem: &str) -> String {
721    let name: String = stem
722        .chars()
723        .map(|c| {
724            if c.is_ascii_alphanumeric() {
725                c.to_ascii_lowercase()
726            } else {
727                '_'
728            }
729        })
730        .collect();
731    if name.is_empty() {
732        "workflow".to_string()
733    } else {
734        name
735    }
736}

738/// `--target rust-lib` dispatch: governed-by-default gate, then emit the crate.
739async fn dispatch_rust_lib(args: &BuildArgs, output: &mut Output) -> Result<()> {
740    let project = args
741        .project
742        .clone()
743        .unwrap_or_else(|| std::env::current_dir().expect("cwd is readable"));
744    let flags = crate::cmd::check::GovernanceFlags {
745        allow_ungoverned: args.allow_ungoverned,
746        no_governance: args.no_governance,
747    };
748    if let Err(diag) = crate::cmd::build_wasm::wasm_governance_gate(&project, flags).await {
749        let _ = output.diagnostic(diag);
750        std::process::exit(2);
751    }
752    let stem = project
753        .file_name()
754        .and_then(|n| n.to_str())
755        .unwrap_or("workflow");
756    let out_dir = args
757        .output
758        .clone()
759        .unwrap_or_else(|| project.join(format!("{stem}-rust-lib")));
760    let artifact = match emit_rust_lib_to(&project, &out_dir) {
761        Ok(a) => a,
762        Err(e) => {
763            let _ = output.error(format!("{e}"));
764            std::process::exit(2);
765        }
766    };
767    if output.is_json() {
768        let _ = output.json(&serde_json::json!({
769            "kind": "rust-lib",
770            "path": artifact.out_dir.display().to_string(),
771            "ir_hash": artifact.ir_hash,
772            "files": artifact.files,
773        }));
774    } else {
775        let _ = output.human(&format!(
776            "built rust-lib crate: {} ({} files, ir {})",
777            artifact.out_dir.display(),
778            artifact.files,
779            artifact.ir_hash,
780        ));
781    }
782    Ok(())
783}

807/// Artifact rendering with JSON support. Emits JSON under --json,
808/// human-readable text otherwise.
809fn emit_artifact(artifact: &BundleArtifact, output: &mut Output) {
810    if output.is_json() {
811        let obj = serde_json::json!({
812            "path": artifact.path.display().to_string(),
813            "sha256": artifact.sha256,
814            "size_bytes": artifact.size_bytes,
815        });
816        let _ = output.json(&obj);
817    } else {
818        let sha = &artifact.sha256;
819        let head = &sha[..sha.len().min(6)];
820        let tail = &sha[sha.len().saturating_sub(6)..];
821        let _ = output.status(format!(
822            "Wrote bundle: {} (sha256: {head}…{tail}, {} bytes)",
823            artifact.path.display(),
824            artifact.size_bytes,
825        ));
826        let _ = output.human(&artifact.path.display().to_string());
827    }
828}

830/// Write host-adapter descriptors for the lowered cron triggers next to the
831/// bundle. Manual triggers and systemd-unconvertible cron schedules are noted
832/// and skipped. Errors writing a descriptor are surfaced but non-fatal — the
833/// bundle is already built.
834fn emit_trigger_descriptors(
835    adapter: &str,
836    bindings: &[tau_ir::trigger::TriggerBinding],
837    artifact: &BundleArtifact,
838    output: &mut Output,
839) {
840    use tau_ir::trigger::{emit_k8s, emit_systemd, TriggerKind};
841
842    let artifact_ref = artifact.path.display().to_string();
843    let descriptors = match adapter {
844        "systemd" => emit_systemd(bindings, &artifact_ref),
845        "k8s" => emit_k8s(bindings, &artifact_ref),
846        other => {
847            let _ = output.error(format!("unknown --emit-trigger adapter: {other}"));
848            return;
849        }
850    };
851
852    if descriptors.is_empty() {
853        let cron_count = bindings
854            .iter()
855            .filter(|b| b.kind == TriggerKind::Cron)
856            .count();
857        if cron_count == 0 {
858            let _ = output.status("No cron triggers to emit (manual triggers need no scheduler).");
859        } else {
860            let _ = output.status(format!(
861                "{cron_count} cron trigger(s) present but none were emittable for {adapter} \
862                 (systemd's OnCalendar supports only `*` and plain-integer cron fields; \
863                 ranges/lists/steps are skipped — use --emit-trigger=k8s for those)."
864            ));
865        }
866        return;
867    }
868
869    let dir = artifact
870        .path
871        .parent()
872        .map(|p| p.to_path_buf())
873        .unwrap_or_else(|| std::path::PathBuf::from("."));
874    for (filename, content) in descriptors {
875        let path = dir.join(&filename);
876        match std::fs::write(&path, content.as_bytes()) {
877            Ok(()) => {
878                let _ = output.status(format!("Wrote trigger descriptor: {}", path.display()));
879            }
880            Err(e) => {
881                let _ = output.error(format!("failed to write {}: {e}", path.display()));
882            }
883        }
884    }
885}

887/// Maps a [`BuildError`] to its CLI exit code per spec §6.
888fn exit_code_for(err: &BuildError) -> u8 {
889    match err {
890        BuildError::MissingLockfile
891        | BuildError::PackageNotInstalled { .. }
892        | BuildError::AgentHomePackageMissing { .. } => 3,
893        BuildError::ProjectConfig(_)
894        | BuildError::LockfileLoad(_)
895        | BuildError::ManifestInvalid(_)
896        | BuildError::UnknownAgent { .. }
897        | BuildError::AgentHomePackageManifest { .. } => 2,
898        BuildError::TreeHashFailed { .. }
899        | BuildError::PromptResolveFailed { .. }
900        | BuildError::CapabilityOverrideFailed { .. }
901        | BuildError::WriteFailed { .. } => 70,
902    }
