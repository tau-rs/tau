1//! `tau list` — list installed packages or project-declared agents.
2//!
3//! Per spec §3.13 list row.
4//!
5//! - `tau list` / `tau list packages` reads the per-scope lockfile via
6//!   [`tau_pkg::list`] and prints one row per installed package.
7//!   `--global` forces global scope, `--all` lists project + global
8//!   side-by-side, otherwise scope is auto-resolved per cwd.
9//! - `tau list agents` reads the project `tau.toml` via
10//!   [`crate::config::ProjectConfig::from_path`] and prints one row per
11//!   declared agent. Errors with exit 2 if no project tau.toml is found.
12//! - `--dry-run` is rejected as nonsensical for a read-only command.
13//!
14//! Both modes support `--json` for scriptable output.
15

22/// Run `tau list`.
23pub async fn run(args: &ListArgs, output: &mut Output) -> anyhow::Result<()> {
24    if args.dry_run {
25        anyhow::bail!("--dry-run is not supported on tau list (read-only command)");
26    }
27
28    match args.resource {
29        ListResource::Packages => list_packages(args, output),
30        ListResource::Agents => list_agents(args, output),
31    }
32}

34/// Tabular row produced for one installed package.
35#[derive(Debug, Serialize)]
36struct PackageRow {
37    name: String,
38    version: String,
39    source: String,
40    scope: String,
41    version_count: usize,
42}
43
44/// Tabular row produced for one declared agent.
45#[derive(Debug, Serialize)]
46struct AgentRow {
47    id: String,
48    display_name: String,
49    package: String,
50    /// The agent's `model` alias (the `[models]` key it references).
51    model: String,
52    /// Backend package the agent's `model` alias resolves to via `[models]`.
53    llm_backend: String,
54    /// Effective capability set after applying the project override.
55    /// `None` when --capabilities flag is not set, OR the package is
56    /// not installed.
57    #[serde(skip_serializing_if = "Option::is_none")]
58    effective_capabilities: Option<Vec<EffectiveCapabilityRow>>,
59}

61/// Per-capability render row in the JSON output. Field naming mirrors
62/// the project tau.toml schema (allow_paths/deny_paths for fs.*,
63/// allow_hosts/deny_hosts for net.http, allow_commands/deny_commands
64/// for process.spawn, max_bytes for fs.write).
65#[derive(Debug, Serialize)]
66struct EffectiveCapabilityRow {
67    kind: String,
68    #[serde(skip_serializing_if = "Option::is_none")]
69    allow_paths: Option<Vec<String>>,
70    #[serde(skip_serializing_if = "Vec::is_empty", default)]
71    deny_paths: Vec<String>,
72    #[serde(skip_serializing_if = "Option::is_none")]
73    allow_hosts: Option<Vec<String>>,
74    #[serde(skip_serializing_if = "Vec::is_empty", default)]
75    deny_hosts: Vec<String>,
76    #[serde(skip_serializing_if = "Option::is_none")]
77    allow_commands: Option<Vec<String>>,
78    #[serde(skip_serializing_if = "Vec::is_empty", default)]
79    deny_commands: Vec<String>,
80    #[serde(skip_serializing_if = "Option::is_none")]
81    max_bytes: Option<u64>,
82}

84fn list_packages(args: &ListArgs, output: &mut Output) -> anyhow::Result<()> {
85    use tau_pkg::{Scope, ScopeKind};
86
87    let cwd = std::env::current_dir()?;
88
89    let mut rows: Vec<PackageRow> = Vec::new();
90
91    if args.all {
92        // Project (if any) + global, side-by-side.
93        // We can only detect a project scope by walking up from cwd; if the
94        // resolved scope is global, there's no project to list.
95        let resolved = Scope::resolve(&cwd)?;
96        if matches!(resolved.kind(), ScopeKind::Project) {
97            collect_packages(&resolved, "project", &mut rows)?;
98        }
99        let global_scope = Scope::global()?;
100        collect_packages(&global_scope, "global", &mut rows)?;
101    } else if args.global {
102        let global_scope = Scope::global()?;
103        collect_packages(&global_scope, "global", &mut rows)?;
104    } else {
105        let scope = Scope::resolve(&cwd)?;
106        let label = scope_label(scope.kind());
107        collect_packages(&scope, label, &mut rows)?;
108    }
109
110    if output.is_json() {
111        output.json(&rows)?;
112    } else if rows.is_empty() {
113        output.human("(no packages installed)")?;
114    } else {
115        output.human(
116            "name                version    source                                   scope",
117        )?;
118        output
119            .human("--                  --         --                                       --")?;
120        for r in &rows {
121            output.human(&format!(
122                "{:<20}{:<11}{:<41}{}",
123                truncate(&r.name, 19),
124                truncate(&r.version, 10),
125                truncate(&r.source, 40),
126                r.scope,
127            ))?;
128        }
129    }
130
131    Ok(())
132}

134fn collect_packages(
135    scope: &tau_pkg::Scope,
136    label: &str,
137    rows: &mut Vec<PackageRow>,
138) -> anyhow::Result<()> {
139    let listing = tau_pkg::list(scope).map_err(|e| anyhow::anyhow!("listing packages: {e}"))?;
140    for pkg in listing {
141        rows.push(PackageRow {
142            name: pkg.name.as_str().to_owned(),
143            version: pkg.active_version.to_string(),
144            source: pkg.source.to_string(),
145            scope: label.to_owned(),
146            version_count: pkg.installed_versions.len(),
147        });
148    }
149    Ok(())
150}

152fn scope_label(kind: tau_pkg::ScopeKind) -> &'static str {
153    match kind {
154        tau_pkg::ScopeKind::Global => "global",
155        tau_pkg::ScopeKind::Project => "project",
156        // ScopeKind is `#[non_exhaustive]`; future-proof the match.
157        _ => "unknown",
158    }
159}

161/// Render `s` truncated to at most `n` characters, no wrapping. Tabular
162/// rows use this so a long URL or version doesn't blow out the column.
163fn truncate(s: &str, n: usize) -> String {
164    if s.chars().count() <= n {
165        s.to_owned()
166    } else {
167        let cut: String = s.chars().take(n.saturating_sub(1)).collect();
168        format!("{cut}\u{2026}")
169    }
170}

172fn list_agents(args: &ListArgs, output: &mut Output) -> anyhow::Result<()> {
173    let cwd = std::env::current_dir()?;
174    let path = cwd.join("tau.toml");
175
176    let config = match ProjectConfig::from_path(&path) {
177        Ok(cfg) => cfg,
178        Err(ProjectConfigError::NotFound) => {
179            anyhow::bail!("no project tau.toml found at {path:?}; run `tau init` to create one");
180        }
181        Err(e) => return Err(e.into()),
182    };
183
184    // For --capabilities, resolve the package scope once. Failure here
185    // (e.g. cwd has no scope) is not fatal for the basic listing —
186    // each per-agent lookup will independently fall back to "no manifest".
187    let scope = if args.capabilities {
188        Some(tau_pkg::Scope::resolve(&cwd)?)
189    } else {
190        None
191    };
192
193    let mut rows: Vec<AgentRow> = Vec::with_capacity(config.agents.len());
194    for agent in config.agents.values() {
195        let effective_capabilities = if args.capabilities {
196            // Best-effort: if the manifest can't be loaded, render row without
197            // the effective set. compute_effective failure (override expands)
198            // is a hard error.
199            match crate::config::build_agent_definition(
200                agent,
201                &cwd,
202                scope
203                    .as_ref()
204                    .expect("scope was resolved when --capabilities was set"),
205                &config.models,
206            ) {
207                Ok((_def, manifest)) => {
208                    let effective = tau_runtime_tokio::capability_override::compute_effective(
209                        manifest.capabilities(),
210                        &agent.capability_overrides,
211                    )
212                    .map_err(|e| anyhow::anyhow!("agent {:?}: {}", agent.id, e))?;
213                    Some(
214                        effective
215                            .iter()
216                            .map(effective_capability_to_row)
217                            .collect::<Vec<_>>(),
218                    )
219                }
220                Err(_) => None, // package not installed; non-fatal
221            }
222        } else {
223            None
224        };
225
226        rows.push(AgentRow {
227            id: agent.id.clone(),
228            display_name: agent.display_name.clone(),
229            package: agent.package.clone(),
230            model: agent.model.clone(),
231            llm_backend: config
232                .models
233                .get(&agent.model)
234                .map(|m| m.backend.clone())
235                .unwrap_or_default(),
236            effective_capabilities,
237        });
238    }
239
240    if output.is_json() {
241        output.json(&rows)?;
242    } else if rows.is_empty() {
243        output.human("(no agents declared in project tau.toml)")?;
244    } else if args.capabilities {
245        output.human(
246            "id              display_name              package                  llm_backend       effective_capabilities",
247        )?;
248        output.human("--              --                        --                       --                --")?;
249        for r in &rows {
250            let caps_str = match &r.effective_capabilities {
251                Some(cap_rows) => format_effective_caps_human(cap_rows),
252                None => "(package not installed)".to_string(),
253            };
254            output.human(&format!(
255                "{:<16}{:<26}{:<25}{:<18}{}",
256                truncate(&r.id, 15),
257                truncate(&r.display_name, 25),
258                truncate(&r.package, 24),
259                truncate(&r.llm_backend, 17),
260                caps_str,
261            ))?;
262        }
263    } else {
264        output.human(
265            "id              display_name              package                  llm_backend",
266        )?;
267        output.human("--              --                        --                       --")?;
268        for r in &rows {
269            output.human(&format!(
270                "{:<16}{:<26}{:<25}{}",
271                truncate(&r.id, 15),
272                truncate(&r.display_name, 25),
273                truncate(&r.package, 24),
274                r.llm_backend,
275            ))?;
276        }
277    }
278
279    Ok(())
280}

282/// Build an `EffectiveCapabilityRow` for the JSON/human output from a
283/// runtime `EffectiveCapability`. Field selection is per-kind: paths
284/// for fs.*, hosts for net.http, commands for process.spawn.
285fn effective_capability_to_row(
286    eff: &tau_runtime_tokio::capability_override::EffectiveCapability,
287) -> EffectiveCapabilityRow {
288    use tau_domain::{Capability, FsCapability, NetCapability, ProcessCapability};
289    let kind = match &eff.source {
290        Capability::Filesystem(FsCapability::Read { .. }) => "fs.read",
291        Capability::Filesystem(FsCapability::Write { .. }) => "fs.write",
292        Capability::Filesystem(FsCapability::Exec { .. }) => "fs.exec",
293        Capability::Network(NetCapability::Http { .. }) => "net.http",
294        Capability::Process(ProcessCapability::Spawn { .. }) => "process.spawn",
295        Capability::Agent(_) => "agent.spawn",
296        Capability::Custom { name, .. } => {
297            return EffectiveCapabilityRow {
298                kind: name.clone(),
299                allow_paths: None,
300                deny_paths: Vec::new(),
301                allow_hosts: None,
302                deny_hosts: Vec::new(),
303                allow_commands: None,
304                deny_commands: Vec::new(),
305                max_bytes: None,
306            }
307        }
308        _ => "unknown",
309    }
310    .to_string();
311    let mut row = EffectiveCapabilityRow {
312        kind: kind.clone(),
313        allow_paths: None,
314        deny_paths: Vec::new(),
315        allow_hosts: None,
316        deny_hosts: Vec::new(),
317        allow_commands: None,
318        deny_commands: Vec::new(),
319        max_bytes: eff.max_bytes_override,
320    };
321    match kind.as_str() {
322        "fs.read" | "fs.write" | "fs.exec" => {
323            row.allow_paths = eff.allow_override.clone();
324            row.deny_paths = eff.deny.clone();
325        }
326        "net.http" => {
327            row.allow_hosts = eff.allow_override.clone();
328            row.deny_hosts = eff.deny.clone();
329        }
330        "process.spawn" => {
331            row.allow_commands = eff.allow_override.clone();
332            row.deny_commands = eff.deny.clone();
333        }
334        _ => {}
335    }
336    row
337}

339/// Render an `EffectiveCapabilityRow` set as a single human-readable line.
340fn format_effective_caps_human(rows: &[EffectiveCapabilityRow]) -> String {
341    if rows.is_empty() {
342        return "(none)".into();
343    }
344    rows.iter()
345        .map(format_one_cap)
346        .collect::<Vec<_>>()
347        .join(", ")
348}
349
350fn format_one_cap(row: &EffectiveCapabilityRow) -> String {
351    let mut parts: Vec<String> = Vec::new();
352    if let Some(allow) = &row.allow_paths {
353        parts.push(format!("allow={}", allow.join(",")));
354    } else if let Some(allow) = &row.allow_hosts {
355        parts.push(format!("allow={}", allow.join(",")));
356    } else if let Some(allow) = &row.allow_commands {
357        parts.push(format!("allow={}", allow.join(",")));
358    }
359    let denies = if !row.deny_paths.is_empty() {
360        Some(&row.deny_paths)
361    } else if !row.deny_hosts.is_empty() {
362        Some(&row.deny_hosts)
363    } else if !row.deny_commands.is_empty() {
364        Some(&row.deny_commands)
365    } else {
366        None
367    };
368    if let Some(d) = denies {
369        parts.push(format!("deny={}", d.join(",")));
370    }
371    if let Some(mb) = row.max_bytes {
372        parts.push(format!("max_bytes={mb}"));
373    }
374    if parts.is_empty() {
375        // Capability declared but no narrowing — package's grant unchanged.
376        format!("{}[unchanged]", row.kind)
377    } else {
378        format!("{}[{}]", row.kind, parts.join(";"))
379    }
380}

382#[cfg(test)]
383mod tests {
384    use super::*;
385
386    #[test]
387    fn scope_label_handles_global_and_project() {
388        assert_eq!(scope_label(tau_pkg::ScopeKind::Global), "global");
389        assert_eq!(scope_label(tau_pkg::ScopeKind::Project), "project");
390    }
391
392    #[test]
393    fn truncate_keeps_short_strings_unchanged() {
394        assert_eq!(truncate("abc", 10), "abc");
395    }
396
397    #[test]
398    fn truncate_appends_ellipsis_when_too_long() {
399        let out = truncate("abcdefghij", 5);
400        assert_eq!(out.chars().count(), 5);
401        assert!(out.ends_with('\u{2026}'));
402    }
403
404    #[test]
405    fn format_one_cap_renders_allow_and_deny() {
406        let row = EffectiveCapabilityRow {
407            kind: "fs.read".into(),
408            allow_paths: Some(vec!["/proj/src/**".into()]),
409            deny_paths: vec!["/proj/.env".into()],
410            allow_hosts: None,
411            deny_hosts: Vec::new(),
412            allow_commands: None,
413            deny_commands: Vec::new(),
414            max_bytes: None,
415        };
416        let out = format_one_cap(&row);
417        assert!(out.contains("fs.read"));
418        assert!(out.contains("allow=/proj/src/**"));
419        assert!(out.contains("deny=/proj/.env"));
420    }
421
422    #[test]
423    fn format_one_cap_unchanged_when_no_narrowing() {
424        let row = EffectiveCapabilityRow {
425            kind: "fs.read".into(),
426            allow_paths: None,
427            deny_paths: Vec::new(),
428            allow_hosts: None,
429            deny_hosts: Vec::new(),
430            allow_commands: None,
431            deny_commands: Vec::new(),
432            max_bytes: None,
433        };
434        let out = format_one_cap(&row);
435        assert!(out.contains("fs.read"));
436        assert!(out.contains("unchanged"));
437    }
