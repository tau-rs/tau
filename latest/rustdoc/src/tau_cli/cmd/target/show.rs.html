13/// Run `tau target show`.
14pub fn run(args: &TargetShowArgs, output: &mut Output) -> anyhow::Result<()> {
15    let triple = match TargetTriple::from_str(&args.triple) {
16        Ok(t) => t,
17        Err(e) => {
18            emit_parse_error(&args.triple, &e, output)?;
19            std::process::exit(64);
20        }
21    };
22
23    let entry = match tau_ports::target::lookup(&triple) {
24        Some(e) => e,
25        None => {
26            emit_unknown(&triple, output)?;
27            std::process::exit(64);
28        }
29    };
30
31    if output.is_json() {
32        render::render_json_event(entry, output)?;
33        return Ok(());
34    }
35
36    let shapes = (entry.shapes_fn)();
37    let mut shape_names: Vec<&'static str> = Vec::new();
38    for s in [
39        tau_domain::CapabilityShape::FilesystemRead,
40        tau_domain::CapabilityShape::FilesystemWrite,
41        tau_domain::CapabilityShape::ProcessExec,
42        tau_domain::CapabilityShape::NetworkHttp,
43        tau_domain::CapabilityShape::AgentSpawn,
44    ] {
45        if shapes.contains(&s) {
46            shape_names.push(render::shape_display(&s));
47        }
48    }
49    let shapes_csv = shape_names.join(", ");
50    let status_line = match entry.status {
51        TripleStatus::Available => "Available".to_string(),
52        TripleStatus::Reserved { reason } => format!("Reserved ({reason})"),
53        _ => "Unknown".to_string(),
54    };
55
56    output.human(&format!("{}", entry.triple))?;
57    output.human(&format!("  status:   {status_line}"))?;
58    output.human(&format!("  platform: {}", entry.triple.platform))?;
59    output.human(&format!("  adapter:  {}", entry.triple.adapter_family))?;
60    output.human(&format!(
61        "  tier:     {}",
62        render::tier_str(entry.triple.tier)
63    ))?;
64    output.human(&format!("  shapes:   {shapes_csv}"))?;
65    Ok(())
66}
67
68fn emit_parse_error(
69    input: &str,
70    err: &tau_ports::target::ParseError,
71    output: &mut Output,
72) -> anyhow::Result<()> {
73    if output.is_json() {
74        output.json(&json!({
75            "event": "error",
76            "kind": "parse",
77            "input": input,
78            "reason": err.to_string(),
79        }))?;
80    } else {
81        output.error(format!("could not parse triple `{input}`: {err}"))?;
82        if let Some(hint) = suggest(input) {
83            output.human(&format!("  did you mean: {hint}?"))?;
84        }
85    }
86    Ok(())
87}
88
89fn emit_unknown(triple: &TargetTriple, output: &mut Output) -> anyhow::Result<()> {
90    if output.is_json() {
91        output.json(&json!({
92            "event": "error",
93            "kind": "unknown",
94            "input": triple.to_string(),
95        }))?;
96    } else {
97        output.error(format!(
98            "unknown triple `{triple}` (parses but not registered)"
99        ))?;
100        if let Some(hint) = suggest(&triple.to_string()) {
101            output.human(&format!("  did you mean: {hint}?"))?;
102        }
103    }
104    Ok(())
105}
106
107fn suggest(input: &str) -> Option<String> {
108    let candidates: Vec<String> = tau_ports::target::list_all()
109        .map(|e| e.triple.to_string())
110        .collect();
111    closest_match(input, &candidates, 4).map(|s| s.to_string())
112}
