1//! Resolve an agent's package capabilities for the governance lattice
2//! (Story 1.5 L1/L2): lockfile → installed manifest → compute_effective →
3//! materialize EffectiveCapability back into concrete Capability.
4
5use serde_json::json;
6
7use crate::cmd::check::runner::CheckCtx;
8use tau_domain::Capability;
9use tau_pkg::capability_override::{compute_effective, EffectiveCapability, OverrideExpandError};
10use tau_pkg::project::AgentEntry;
11use tau_pkg::{read_manifest, LockFile};

13/// Outcome of resolving an agent's package capabilities.
14pub(crate) enum AgentCaps {
15    /// Package installed; `manifest` = declared caps, `effective` = manifest ∩ override.
16    Resolved {
17        manifest: Vec<Capability>,
18        effective: Vec<Capability>,
19    },
20    /// No lockfile / package not in lockfile / install dir or manifest missing.
21    NotInstalled,
22    /// The agent's override expanded its package grant (compute_effective failed).
23    OverrideExpands(OverrideExpandError),
24}
25
26/// Materialize an EffectiveCapability into a concrete Capability by applying
27/// the override deltas to `source` via the serde bridge.
28///
29/// `deny` is only applied to list-bearing kinds (fs/net/process); for other
30/// kinds it is a no-op (bounded; no defined semantics today).
31fn materialize(e: &EffectiveCapability) -> Capability {
32    if e.allow_override.is_none() && e.deny.is_empty() && e.max_bytes_override.is_none() {
33        return e.source.clone();
34    }
35    let mut v = match serde_json::to_value(&e.source) {
36        Ok(serde_json::Value::Object(o)) => o,
37        _ => return e.source.clone(),
38    };
39    let field = match v.get("kind").and_then(|k| k.as_str()) {
40        Some("fs.read") | Some("fs.write") | Some("fs.exec") => Some("paths"),
41        Some("net.http") => Some("hosts"),
42        Some("process.spawn") => Some("commands"),
43        _ => None,
44    };
45    if let Some(field) = field {
46        let mut list: Vec<String> = match &e.allow_override {
47            Some(a) => a.clone(),
48            None => v
49                .get(field)
50                .and_then(|x| x.as_array())
51                .map(|arr| {
52                    arr.iter()
53                        .filter_map(|s| s.as_str().map(String::from))
54                        .collect()
55                })
56                .unwrap_or_default(),
57        };
58        list.retain(|s| !e.deny.contains(s));
59        v.insert(field.to_string(), json!(list));
60    }
61    if let Some(mb) = e.max_bytes_override {
62        v.insert("max_bytes".to_string(), json!(mb));
63    }
64    serde_json::from_value::<Capability>(serde_json::Value::Object(v))
65        .unwrap_or_else(|_| e.source.clone())
66}

68/// Resolve the agent's package capabilities (manifest + effective).
69pub(crate) fn resolve_agent_caps(agent: &AgentEntry, ctx: &CheckCtx) -> AgentCaps {
70    let pkg_name = agent.package.split('@').next().unwrap_or(&agent.package);
71
72    let lockfile_path = ctx.scope.lockfile_path();
73    let Ok(lockfile) = LockFile::load(&lockfile_path) else {
74        return AgentCaps::NotInstalled;
75    };
76    let Some(pkg) = lockfile
77        .packages
78        .iter()
79        .find(|p| p.name.as_str() == pkg_name)
80    else {
81        return AgentCaps::NotInstalled;
82    };
83
84    let toml_path = ctx
85        .scope
86        .package_dir(&pkg.name, &pkg.active_version)
87        .join("tau.toml");
88    let Ok(manifest) = read_manifest(&toml_path) else {
89        return AgentCaps::NotInstalled;
90    };
91    let manifest_caps: Vec<Capability> = manifest.capabilities().to_vec();
92
93    match compute_effective(&manifest_caps, &agent.capability_overrides) {
94        Ok(eff) => AgentCaps::Resolved {
95            manifest: manifest_caps,
96            effective: eff.iter().map(materialize).collect(),
97        },
98        Err(e) => AgentCaps::OverrideExpands(e),
99    }
100}

102#[cfg(test)]
103mod tests {
104    use super::*;
105
106    fn cap(j: &str) -> Capability {
107        serde_json::from_str(j).unwrap()
108    }
109
110    #[test]
111    fn materialize_passthrough_when_no_override() {
112        // EffectiveCapability is #[non_exhaustive] + constructed by compute_effective;
113        // build one via compute_effective to keep within the public surface.
114        let pkg = vec![cap(r#"{"kind":"fs.read","paths":["/proj/**"]}"#)];
115        let eff = compute_effective(&pkg, &[]).unwrap();
116        let m = materialize(&eff[0]);
117        assert_eq!(m, cap(r#"{"kind":"fs.read","paths":["/proj/**"]}"#));
118    }
119
120    #[test]
121    fn materialize_applies_allow_override_and_deny() {
122        use tau_pkg::capability_override::CapabilityOverride;
123        let pkg = vec![cap(r#"{"kind":"fs.read","paths":["/proj/**"]}"#)];
124        let ov = vec![CapabilityOverride::new(
125            "fs.read".into(),
126            Some(vec!["/proj/src/**".into(), "/proj/docs/**".into()]),
127            vec!["/proj/docs/**".into()],
128            None,
129        )];
130        let eff = compute_effective(&pkg, &ov).unwrap();
131        let m = materialize(&eff[0]);
132        // allow_override minus deny = ["/proj/src/**"]
133        assert_eq!(m, cap(r#"{"kind":"fs.read","paths":["/proj/src/**"]}"#));
134    }
135}
