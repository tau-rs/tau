29/// Entry point for `tau check`. Parses CheckArgs, builds CheckCtx,
30/// selects category list, runs the orchestrator, dispatches output
31/// (Tasks 10-12 wire renderers), and exits with the appropriate code.
32pub async fn run(args: crate::cli::CheckArgs) -> Result<()> {
33    // Resolve project path. May be a directory (TOML-based project) or a
34    // `.ts` file (β.8 TypeScript-authoring surface).
35    let project_path = match args.project.as_ref() {
36        Some(p) => p.clone(),
37        None => std::env::current_dir()?,
38    };
39
40    let target: Option<TargetTriple> = if let Some(s) = args.target.as_deref() {
41        match TargetTriple::from_str(s) {
42            Ok(t) => {
43                if tau_ports::target::lookup(&t).is_none() {
44                    eprintln!("error: unknown triple `{t}` (parses but not registered)");
45                    std::process::exit(64);
46                }
47                Some(t)
48            }
49            Err(e) => {
50                eprintln!("error: could not parse triple `{s}`: {e}");
51                std::process::exit(64);
52            }
53        }
54    } else {
55        None
56    };
57
58    let ctx = runner::CheckCtx::load(project_path, args.fast, target).await?;
59
60    // --auto-resolve: attempt to install missing required tools for all agents
61    // before running checks. Errors are non-fatal — the packages category will
62    // report any remaining missing tools as findings.
63    if args.auto_resolve {
64        if let Some(project) = &ctx.project {
65            let mut resolve_output = crate::output::Output::with_writers(
66                Box::new(std::io::sink()),
67                Box::new(std::io::stderr()),
68                false,
69                false,
70                crate::output::ColorChoice::Never,
71            );
72            if let Err(e) = crate::cmd::resolve_helpers::resolve_and_install_for_project(
73                project.agents.values().cloned(),
74                &ctx.scope,
75                false, // no_install = false → actually install
76                false, // dry_run = false → apply changes
77                &mut resolve_output,
78            ) {
79                // Non-fatal: warn on stderr and continue; packages check
80                // will surface any remaining gaps as findings.
81                eprintln!("tau check: --auto-resolve encountered errors (continuing): {e:#}");
82            }
83        }
84    }
85
86    // Determine category list.
87    let categories: Vec<CheckCategory> = match args.category.as_deref() {
88        None => CheckCategory::ALL.to_vec(),
89        Some("config") => vec![CheckCategory::Config],
90        Some("lockfile") => vec![CheckCategory::Lockfile],
91        Some("packages") => vec![CheckCategory::Packages],
92        Some("sandbox") => vec![CheckCategory::Sandbox],
93        Some("plugins") => vec![CheckCategory::Plugins],
94        Some("skills") => vec![CheckCategory::Skills],
95        Some("mcp-contracts") => vec![CheckCategory::McpContracts],
96        Some("governance") => vec![CheckCategory::Governance],
97        Some(other) => anyhow::bail!("unknown check category: {other}"),
98    };
99
100    let results = runner::run_categories(&ctx, &categories).await;
101    let exit_code = compute_exit(&results);
102
103    if let Some(path) = args.sarif.as_ref() {
104        let rendered = output::sarif::render(&results);
105        if path == "-" {
106            print!("{rendered}");
107        } else {
108            std::fs::write(path, rendered)?;
109        }
110    } else if args.json {
111        let rendered = output::json::render(
112            &ctx.project_root,
113            &categories,
114            args.fast,
115            &results,
116            exit_code,
117        );
118        print!("{rendered}");
119    } else {
120        let use_color = std::env::var_os("NO_COLOR").is_none();
121        let rendered = output::human::render(&results, use_color, exit_code);
122        print!("{rendered}");
123    }
124
125    if exit_code != 0 {
126        std::process::exit(exit_code);
127    }
128    Ok(())
129}
