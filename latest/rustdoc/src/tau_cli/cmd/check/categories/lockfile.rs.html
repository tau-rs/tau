1//! `tau check lockfile` — recompute install-tree hashes vs lockfile.
2//!
3//! Wraps `tau_pkg::verify_all_with_options`. Each non-Ok `VerifyStatus`
4//! becomes one finding with severity `Error` (drift is a real bug).
5
6use crate::cmd::check::result::{
7    CheckCategory, CheckFinding, CheckResult, CheckStatus, FindingLocation, Severity,
8};
9use crate::cmd::check::runner::CheckCtx;
10use serde_json::json;
11use tau_pkg::verify_all_with_options;
12
13pub fn run_lockfile(ctx: &CheckCtx) -> CheckResult {
14    let lockfile_path = ctx.scope.lockfile_path();
15    let location = Some(FindingLocation {
16        path: lockfile_path.to_path_buf(),
17        line: None,
18        column: None,
19    });
20
21    let reports = match verify_all_with_options(&ctx.scope, /*anthropic_strict=*/ false) {
22        Ok(reports) => reports,
23        Err(e) => {
24            // Lockfile load/parse error becomes a single finding.
25            let findings = vec![CheckFinding {
26                category: CheckCategory::Lockfile,
27                severity: Severity::Error,
28                rule_id: "tau.lockfile.invalid",
29                summary: format!("lockfile error: {e}"),
30                detail: None,
31                location,
32                remediation: Some("run `tau install` to regenerate the lockfile".into()),
33                structured: json!({"kind": "LoadError"}),
34            }];
35            return CheckResult {
36                category: CheckCategory::Lockfile,
37                status: CheckStatus::Failed,
38                findings,
39                duration: std::time::Duration::ZERO,
40            };
41        }
42    };
43
44    let mut findings = Vec::new();
45    for report in &reports {
46        if !report.status.is_drift() {
47            continue;
48        }
49        findings.push(CheckFinding {
50            category: CheckCategory::Lockfile,
51            severity: Severity::Error,
52            rule_id: "tau.lockfile.drift",
53            summary: format!(
54                "{}@{} drift: {:?}",
55                report.name, report.version, report.status
56            ),
57            detail: None,
58            location: location.clone(),
59            remediation: Some("run `tau verify` for details, then `tau install` to refresh".into()),
60            structured: json!({
61                "package": report.name.to_string(),
62                "version": report.version.to_string(),
63                "status": format!("{:?}", report.status),
64            }),
65        });
66    }
67
68    let status = if findings.is_empty() {
69        CheckStatus::Ok
70    } else {
71        CheckStatus::Failed
72    };
73    CheckResult {
74        category: CheckCategory::Lockfile,
75        status,
76        findings,
77        duration: std::time::Duration::ZERO,
78    }
79}
