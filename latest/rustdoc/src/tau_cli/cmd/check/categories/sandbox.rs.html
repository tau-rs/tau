1//! `tau check sandbox` — validate sandbox plans for each installed plugin.
2//!
3//! Default (full): build plan AND validate against the resolved adapter.
4//! `--fast`: build plan only; skip adapter probe + validation.
5//!
6//! The per-plugin build/validate loop lives in
7//! `crate::cmd::resolve_helpers::check_plugin_sandbox`; this module
8//! handles the check-aggregator-specific output mapping (severity policy,
9//! `CheckFinding` synthesis, fast-mode adapter elision).
10
11use crate::cmd::check::result::{
12    CheckCategory, CheckFinding, CheckResult, CheckStatus, FindingLocation, Severity,
13};
14use crate::cmd::check::runner::CheckCtx;
15use crate::cmd::resolve_helpers::{
16    check_plugin_sandbox, check_plugin_sandbox_against_profile,
17    read_sandbox_requirements_for_check, resolve_sandbox_check_adapter, SandboxPluginOutcome,
18};
19use serde_json::json;

21pub async fn run_sandbox(ctx: &CheckCtx) -> CheckResult {
22    // project.is_none() means tau.toml is malformed — the config check
23    // reports this; we just skip to avoid duplicate noise.
24    if ctx.project.is_none() {
25        return skipped("tau.toml malformed (see config check)");
26    }
27
28    let tau_toml_path = ctx.project_root.join("tau.toml");
29    let sandbox_requirements = read_sandbox_requirements_for_check(&ctx.scope);
30
31    // Load the lockfile. If missing or unreadable, skip — the lockfile
32    // check will already report this.
33    let lockfile_path = ctx.scope.lockfile_path();
34    if !lockfile_path.exists() {
35        return skipped("lockfile missing or unreadable (see lockfile check)");
36    }
37    let lockfile = match tau_pkg::LockFile::load(&lockfile_path) {
38        Ok(lf) => lf,
39        Err(_) => return skipped("lockfile missing or unreadable (see lockfile check)"),
40    };
41
42    // Collect only packages that have a plugin entry — data-only packages
43    // don't need sandbox plans.
44    let plugin_pkgs: Vec<_> = lockfile
45        .packages
46        .iter()
47        .filter(|p| p.plugin.is_some())
48        .collect();
49
50    let mut findings: Vec<CheckFinding> = Vec::new();
51
52    // --target branch: validate against the target's documented profile
53    // instead of the locally resolved adapter.
54    //
55    // NOTE: the `plugin_pkgs.is_empty()` skip is intentionally NOT placed
56    // before this block.  Durability is an agent concern, independent of
57    // plugins.  A project with durable agents and no plugin packages must
58    // still receive durability findings (and the hard-fail path) when
59    // `--target` is set.  The per-plugin loop below no-ops naturally when
60    // `plugin_pkgs` is empty.
61    if let Some(target) = &ctx.target {
62        let Some(entry) = tau_ports::target::lookup(target) else {
63            // Should not happen — dispatch already validated the triple.
64            findings.push(CheckFinding {
65                category: CheckCategory::Sandbox,
66                severity: Severity::Error,
67                rule_id: "tau.sandbox.target_unknown",
68                summary: format!("target `{target}` is not registered"),
69                detail: None,
70                location: None,
71                remediation: Some("tau target list".into()),
72                structured: json!({ "kind": "TargetUnknown", "target": target.to_string() }),
73            });
74            return CheckResult {
75                category: CheckCategory::Sandbox,
76                status: CheckStatus::Failed,
77                findings,
78                duration: std::time::Duration::ZERO,
79            };
80        };
81        let profile = entry.profile();
82
83        // EPIC 6.1: print the host-resolved durability per durable agent.
84        if let Some(project) = &ctx.project {
85            findings.extend(durability_findings(&project.agents, target));
86        }
87
88        // Reserved → advisory Warning, but still validate against documented matrix.
89        if let tau_ports::target::TripleStatus::Reserved { reason } = entry.status {
90            findings.push(CheckFinding {
91                category: CheckCategory::Sandbox,
92                severity: Severity::Warning,
93                rule_id: "tau.sandbox.target_reserved",
94                summary: format!("target `{target}` is reserved: {reason}"),
95                detail: Some(
96                    "Reserved triples have a documented capability matrix but no shipping adapter; bundles compiled for them will not yet execute anywhere.".into(),
97                ),
98                location: None,
99                remediation: None,
100                structured: json!({ "kind": "TargetReserved", "target": target.to_string(), "reason": reason }),
101            });
102        }
103
104        // Adapter-availability check (Warning if no local adapter satisfies the triple).
105        if tau_runtime_tokio::process_gate::registration_for_triple(target).is_none() {
106            findings.push(CheckFinding {
107                category: CheckCategory::Sandbox,
108                severity: Severity::Warning,
109                rule_id: "tau.sandbox.target_no_local_adapter",
110                summary: format!(
111                    "no local adapter satisfies target `{target}`; cross-check is static only"
112                ),
113                detail: None,
114                location: None,
115                remediation: None,
116                structured: json!({ "kind": "TargetNoLocalAdapter", "target": target.to_string() }),
117            });
118        }
119
120        // Project required_tier must be ≤ target tier.
121        let project_tier = sandbox_requirements.required_tier;
122        let target_tier = target.tier;
123        if !tier_le(project_tier, target_tier) {
124            findings.push(CheckFinding {
125                category: CheckCategory::Sandbox,
126                severity: Severity::Error,
127                rule_id: "tau.sandbox.target_tier_mismatch",
128                summary: format!(
129                    "project requires tier {project_tier:?} but target `{target}` provides tier {target_tier:?}"
130                ),
131                detail: None,
132                location: Some(FindingLocation { path: tau_toml_path.clone(), line: None, column: None }),
133                remediation: None,
134                structured: json!({
135                    "kind": "TargetTierMismatch",
136                    "target": target.to_string(),
137                    "project_tier": format!("{project_tier:?}"),
138                    "target_tier": format!("{target_tier:?}"),
139                }),
140            });
141        }
142
143        // Per-plugin shape check.
144        for pkg in &plugin_pkgs {
145            let plugin_id = pkg.name.as_str().to_owned();
146            let pkg_dir = ctx.scope.package_dir(&pkg.name, &pkg.active_version);
147            let manifest_path = pkg_dir.join("tau.toml");
148
149            match check_plugin_sandbox_against_profile(&plugin_id, &manifest_path, &profile) {
150                SandboxPluginOutcome::Ok => {}
151                SandboxPluginOutcome::BuildPlanFailed(msg) => {
152                    findings.push(build_plan_finding(&plugin_id, msg, &tau_toml_path));
153                }
154                SandboxPluginOutcome::ValidateFailed(errors) => {
155                    for err in errors {
156                        findings.push(CheckFinding {
157                            category: CheckCategory::Sandbox,
158                            severity: Severity::Error,
159                            rule_id: "tau.sandbox.target_shape_unsupported",
160                            summary: format!("plugin `{plugin_id}`: {}", err.reason),
161                            detail: None,
162                            location: Some(FindingLocation {
163                                path: tau_toml_path.clone(),
164                                line: None,
165                                column: None,
166                            }),
167                            remediation: None,
168                            structured: json!({
169                                "kind": "TargetShapeUnsupported",
170                                "plugin_id": plugin_id,
171                                "reason": err.reason,
172                            }),
173                        });
174                    }
175                }
176                SandboxPluginOutcome::ManifestUnreadable(msg) => {
177                    if !ctx.fast {
178                        findings.push(CheckFinding {
179                            category: CheckCategory::Sandbox,
180                            severity: Severity::Warning,
181                            rule_id: "tau.sandbox.manifest_unreadable",
182                            summary: format!(
183                                "could not read manifest for `{plugin_id}`: {msg} — skipping capability check"
184                            ),
185                            detail: None,
186                            location: Some(FindingLocation {
187                                path: manifest_path,
188                                line: None,
189                                column: None,
190                            }),
191                            remediation: Some("tau resolve".into()),
192                            structured: json!({
193                                "plugin_id": plugin_id,
194                                "kind": "ManifestUnreadable",
195                                "error": msg,
196                            }),
197                        });
198                    }
199                }
200            }
201        }
202
203        let status = if findings.iter().any(|f| f.severity == Severity::Error) {
204            CheckStatus::Failed
205        } else {
206            CheckStatus::Ok
207        };
208        return CheckResult {
209            category: CheckCategory::Sandbox,
210            status,
211            findings,
212            duration: std::time::Duration::ZERO,
213        };
214    }
215
216    // No-target path: skip entirely when there are no plugin packages to
217    // validate. (The --target path above already returned, so we only reach
218    // here when ctx.target is None.)
219    if plugin_pkgs.is_empty() {
220        return skipped("no plugin packages in lockfile");
221    }
222
223    // Resolve adapter unless we're in --fast mode.
224    //
225    // When required_tier is None the runtime would pick Passthrough, which
226    // trivially accepts every plan. We use resolve_strict_for_validation
227    // via the helper, which picks the highest-priority non-passthrough
228    // adapter instead to surface what would happen if the user strengthens
229    // the requirement.
230    let adapter_opt = if ctx.fast {
231        None
232    } else {
233        match resolve_sandbox_check_adapter(&sandbox_requirements).await {
234            Ok(a) => Some(a),
235            Err(e) => {
236                // No adapter available — emit an advisory warning and skip
237                // validation rather than hard-failing.
238                findings.push(CheckFinding {
239                    category: CheckCategory::Sandbox,
240                    severity: Severity::Warning,
241                    rule_id: "tau.sandbox.no_adapter",
242                    summary: format!("no sandbox adapter available for validation: {e}"),
243                    detail: Some(
244                        "Sandbox plan shapes could not be validated. \
245                         Install a sandbox adapter (e.g. tau-sandbox-darwin) to enable full checks."
246                            .into(),
247                    ),
248                    location: None,
249                    remediation: None,
250                    structured: json!({ "kind": "NoAdapterAvailable", "error": e.to_string() }),
251                });
252                return CheckResult {
253                    category: CheckCategory::Sandbox,
254                    status: CheckStatus::Ok, // advisory only, not a hard failure
255                    findings,
256                    duration: std::time::Duration::ZERO,
257                };
258            }
259        }
260    };
261
262    for pkg in &plugin_pkgs {
263        let plugin_id = pkg.name.as_str().to_owned();
264        let pkg_dir = ctx.scope.package_dir(&pkg.name, &pkg.active_version);
265        let manifest_path = pkg_dir.join("tau.toml");
266
267        match check_plugin_sandbox(&plugin_id, &manifest_path, adapter_opt.as_ref()) {
268            SandboxPluginOutcome::Ok => {}
269            SandboxPluginOutcome::BuildPlanFailed(msg) => {
270                findings.push(build_plan_finding(&plugin_id, msg, &tau_toml_path));
271            }
272            SandboxPluginOutcome::ValidateFailed(errors) => {
273                for err in errors {
274                    findings.push(CheckFinding {
275                        category: CheckCategory::Sandbox,
276                        severity: Severity::Error,
277                        rule_id: "tau.sandbox.plan_invalid",
278                        summary: format!("plugin `{plugin_id}`: {}", err.reason),
279                        detail: None,
280                        location: Some(FindingLocation {
281                            path: tau_toml_path.clone(),
282                            line: None,
283                            column: None,
284                        }),
285                        remediation: None,
286                        structured: json!({
287                            "plugin_id": plugin_id,
288                            "kind": "SandboxValidationFailed",
289                            "reason": err.reason,
290                        }),
291                    });
292                }
293            }
294            SandboxPluginOutcome::ManifestUnreadable(msg) => {
295                // Fast mode preserves the prior silent-skip behavior; full
296                // mode surfaces a Warning so users see why a plugin was
297                // skipped without changing the result status.
298                if !ctx.fast {
299                    findings.push(CheckFinding {
300                        category: CheckCategory::Sandbox,
301                        severity: Severity::Warning,
302                        rule_id: "tau.sandbox.manifest_unreadable",
303                        summary: format!(
304                            "could not read manifest for `{plugin_id}`: {msg} — skipping capability check"
305                        ),
306                        detail: None,
307                        location: Some(FindingLocation {
308                            path: manifest_path,
309                            line: None,
310                            column: None,
311                        }),
312                        remediation: Some("tau resolve".into()),
313                        structured: json!({
314                            "plugin_id": plugin_id,
315                            "kind": "ManifestUnreadable",
316                            "error": msg,
317                        }),
318                    });
319                }
320            }
321        }
322    }
323
324    let status = if findings.iter().any(|f| f.severity == Severity::Error) {
325        CheckStatus::Failed
326    } else {
327        CheckStatus::Ok
328    };
329    CheckResult {
330        category: CheckCategory::Sandbox,
331        status,
332        findings,
333        duration: std::time::Duration::ZERO,
334    }
335}

337fn skipped(reason: &str) -> CheckResult {
338    CheckResult {
339        category: CheckCategory::Sandbox,
340        status: CheckStatus::Skipped {
341            reason: reason.into(),
342        },
343        findings: Vec::new(),
344        duration: std::time::Duration::ZERO,
345    }

348fn tier_le(a: tau_pkg::scope::SandboxRequiredTier, b: tau_ports::CapabilityTier) -> bool {
349    use tau_pkg::scope::SandboxRequiredTier as Req;
350    use tau_ports::CapabilityTier as Tier;
351    // SandboxTier is #[non_exhaustive]; catch-all is required for external match.
352    #[allow(unreachable_patterns)]
353    let to_rank = |t: Tier| match t {
354        Tier::None => 0,
355        Tier::Light => 1,
356        Tier::Strict => 2,
357        _ => 0,
358    };
359    // SandboxRequiredTier is also #[non_exhaustive]; catch-all required.
360    #[allow(unreachable_patterns)]
361    let req_rank = match a {
362        Req::None => 0,
363        Req::Light => 1,
364        Req::Strict => 2,
365        _ => 0,
366    };
367    req_rank <= to_rank(b)
368}

370/// Build the per-agent durability resolution findings for `tau check --target`.
371/// Honored → an informational `Note`; Unsupported → an `Error`.
372fn durability_findings(
373    agents: &std::collections::BTreeMap<String, tau_pkg::project::AgentEntry>,
374    target: &tau_ports::target::TargetTriple,
375) -> Vec<CheckFinding> {
376    use tau_runtime_core::Support;
377    let mut out = Vec::new();
378    for (id, agent) in agents {
379        let Some(entry) = agent.durable.as_ref() else {
380            continue;
381        };
382        let durability = tau_ir_lower::durable_entry_to_ir(entry);
383        let resolved = tau_runtime_core::resolve_durability(&durability, target);
384        let form = if resolved.from_intent.is_some() {
385            "intent"
386        } else {
387            "explicit"
388        };
389        let ckpt = match resolved.checkpoint {
390            tau_ir::durable::CheckpointGranularity::PerTurn => "per_turn",
391            tau_ir::durable::CheckpointGranularity::PerToolCall => "per_tool_call",
392            _ => "per_turn",
393        };
394        let store = match resolved.store {
395            tau_ir::durable::DurableStore::File => "file",
396            _ => "file",
397        };
398        let detail = match resolved.from_intent {
399            Some(_) => format!("survive-restarts → {ckpt} checkpoints, {store} store"),
400            None => format!("explicit {ckpt} + {store}"),
401        };
402        #[allow(unreachable_patterns)]
403        match resolved.support {
404            Support::Honored => out.push(CheckFinding {
405                category: CheckCategory::Sandbox,
406                severity: Severity::Note,
407                rule_id: "tau.durability.resolved",
408                summary: format!("{id}: {detail}  [resolved for {target}]"),
409                detail: None,
410                location: None,
411                remediation: None,
412                structured: json!({
413                    "kind": "DurabilityResolved",
414                    "agent": id,
415                    "form": form,
416                    "checkpoint": ckpt,
417                    "store": store,
418                    "support": "honored",
419                    "target": target.to_string(),
420                }),
421            }),
422            Support::Unsupported { reason } => out.push(CheckFinding {
423                category: CheckCategory::Sandbox,
424                severity: Severity::Error,
425                rule_id: "tau.durability.unsupported",
426                summary: format!("{id}: target `{target}` cannot honor durability: {reason}"),
427                detail: None,
428                location: None,
429                remediation: None,
430                structured: json!({
431                    "kind": "DurabilityUnsupported",
432                    "agent": id,
433                    "support": "unsupported",
434                    "reason": reason,
435                    "target": target.to_string(),
436                }),
437            }),
438            _ => {}
439        }
440    }
441    out
442}

444fn build_plan_finding(
445    plugin_id: &str,
446    message: String,
447    tau_toml_path: &std::path::Path,
448) -> CheckFinding {
449    CheckFinding {
450        category: CheckCategory::Sandbox,
451        severity: Severity::Error,
452        rule_id: "tau.sandbox.plan_invalid",
453        summary: format!("build_plan failed for `{plugin_id}`: {message}"),
454        detail: None,
455        location: Some(FindingLocation {
456            path: tau_toml_path.to_path_buf(),
457            line: None,
458            column: None,
459        }),
460        remediation: None,
461        structured: json!({
462            "plugin_id": plugin_id,
463            "kind": "BuildPlanFailed",
464            "error": message,
465        }),
466    }

475    // ---------------------------------------------------------------------------
476    // Regression test for the `plugin_pkgs.is_empty()` early-return gap.
477    //
478    // Before the fix, `run_sandbox` would return `Skipped("no plugin packages in
479    // lockfile")` even when `--target` was supplied and the project had a durable
480    // agent, so durability findings were never emitted and the hard-fail path
481    // (`DurabilityUnsupported`) was silently bypassed.
482    // ---------------------------------------------------------------------------
483
484    /// Build a minimal CheckCtx pointing at a temp-dir project that has:
485    ///   - a tau.toml with one durable agent (survive-restarts intent)
486    ///   - a lockfile with NO plugin packages
487    ///   - a target triple that honors durability
488    fn make_ctx_durable_no_plugins(
489        project_root: &std::path::Path,
490        target: tau_ports::target::TargetTriple,
491    ) -> CheckCtx {
492        // Create .tau/ so Scope::resolve finds a project scope.
493        std::fs::create_dir_all(project_root.join(".tau")).unwrap();
494
495        let scope = tau_pkg::Scope::resolve(project_root).unwrap();
496
497        // Write an empty lockfile (no plugin packages).
498        let lf = tau_pkg::lockfile::LockFile::default();
499        lf.save(&scope.lockfile_path()).unwrap();
500
501        // Build a ProjectConfig with one durable agent inline.
502        // NOTE: `packages = ["mock-llm"]` declares the backend so the
503        // validator accepts `backend = "mock-llm"` in [models].
504        let toml_src = r#"
505packages = ["mock-llm"]
506
507[project]
508name = "test-durable"
509
510[models]
511default = { backend = "mock-llm", model = "mock" }
512
513[agents.bot]
514display_name = "Bot"
515package      = "p@^0.1"
516model        = "default"
517tool_refs    = []
518durable      = "survive-restarts"
519"#;
520        let project = tau_pkg::project::ProjectConfig::parse_str(toml_src).ok();
521
522        CheckCtx {
523            project_root: project_root.to_path_buf(),
524            scope,
525            project,
526            fast: false,
527            target: Some(target),
528        }
529    }

531    #[tokio::test]
532    async fn run_sandbox_with_target_emits_durability_finding_even_with_no_plugin_packages() {
533        let tmp = tempfile::tempdir().unwrap();
534        let target: tau_ports::target::TargetTriple = "any-wasi-strict".parse().unwrap();
535        let ctx = make_ctx_durable_no_plugins(tmp.path(), target);
536
537        let result = run_sandbox(&ctx).await;
538
539        // Must NOT be Skipped — durability is an agent concern, not a plugin concern.
540        assert!(
541            !matches!(result.status, CheckStatus::Skipped { .. }),
542            "expected non-Skipped result but got {:?}",
543            result.status
544        );
545
546        // Must contain at least one durability finding for the durable agent.
547        let durability_findings: Vec<_> = result
548            .findings
549            .iter()
550            .filter(|f| f.rule_id == "tau.durability.resolved")
551            .collect();
552        assert!(
553            !durability_findings.is_empty(),
554            "expected tau.durability.resolved finding; got findings: {:#?}",
555            result.findings
556        );
557        // The finding is a Note (Honored on a registered target).
558        assert_eq!(durability_findings[0].severity, Severity::Note);
559    }
