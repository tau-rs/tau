1//! `tau check packages` — every agent's `requires.tools` references
2//! are present in the lockfile at a satisfying version.
3
4use crate::cmd::check::result::{
5    CheckCategory, CheckFinding, CheckResult, CheckStatus, FindingLocation, Severity,
6};
7use crate::cmd::check::runner::CheckCtx;
8use serde_json::json;
9use tau_pkg::LockFile;
10
11pub fn run_packages(ctx: &CheckCtx) -> CheckResult {
12    let Some(project) = &ctx.project else {
13        return CheckResult {
14            category: CheckCategory::Packages,
15            status: CheckStatus::Skipped {
16                reason: "tau.toml malformed (see config check)".into(),
17            },
18            findings: Vec::new(),
19            duration: std::time::Duration::ZERO,
20        };
21    };
22
23    let lockfile_path = ctx.scope.lockfile_path();
24    let lockfile: Option<LockFile> = if lockfile_path.exists() {
25        LockFile::load(&lockfile_path).ok()
26    } else {
27        None
28    };
29
30    let tau_toml = ctx.project_root.join("tau.toml");
31    let location = Some(FindingLocation {
32        path: tau_toml,
33        line: None,
34        column: None,
35    });
36
37    let mut findings = Vec::new();
38    for (agent_id, agent) in &project.agents {
39        for required in &agent.requires.tools {
40            let satisfied = lockfile
41                .as_ref()
42                .map(|lf| is_satisfied(lf, &required.name.to_string(), &required.version_req))
43                .unwrap_or(false);
44            if !satisfied {
45                findings.push(CheckFinding {
46                    category: CheckCategory::Packages,
47                    severity: Severity::NeedsSetup,
48                    rule_id: "tau.packages.missing",
49                    summary: format!(
50                        "agent `{}` requires `{}` {} but it isn't installed at a satisfying version",
51                        agent_id, required.name, required.version_req
52                    ),
53                    detail: None,
54                    location: location.clone(),
55                    remediation: Some("tau resolve".into()),
56                    structured: json!({
57                        "agent_id": agent_id,
58                        "package": required.name.to_string(),
59                        "version_req": required.version_req.to_string(),
60                    }),
61                });
62            }
63        }
64    }
65
66    let status = if findings.is_empty() {
67        CheckStatus::Ok
68    } else {
69        CheckStatus::Failed
70    };
71    CheckResult {
72        category: CheckCategory::Packages,
73        status,
74        findings,
75        duration: std::time::Duration::ZERO,
76    }
77}
78
79/// True if `lockfile` has an installed version of `name` satisfying `req`.
80fn is_satisfied(lockfile: &LockFile, name: &str, req: &semver::VersionReq) -> bool {
81    for pkg in &lockfile.packages {
82        if pkg.name.to_string() != name {
83            continue;
84        }
85        for v in &pkg.installed_versions {
86            if req.matches(&v.version) {
87                return true;
88            }
89        }
90    }
91    false
92}
