1//! Interactive shell around the pure [`crate::tui::draw`] renderer.
2//!
3//! [`App`] holds the [`TraceModel`] + [`UiState`] and reduces key input via
4//! [`App::apply_key`], which is deliberately pure (no terminal I/O) so it is
5//! unit-testable without a real terminal. [`run_tui`] is the impure half: it
6//! owns raw-mode/alternate-screen setup and teardown, and drives the read
7//! loop — tailing a jsonl file or draining a live [`TraceEvent`] channel —
8//! redrawing after every batch of input.
9

25/// Result of feeding one key to [`App::apply_key`].
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum Loop {
28    /// Keep running the event loop.
29    Continue,
30    /// Exit the event loop (terminal teardown happens in [`run_tui`]).
31    Quit,
32}
33
34/// Where [`run_tui`] reads [`TraceEvent`]s from.
35pub enum TraceSource {
36    /// Tail a `.tau/runs/<id>.jsonl` file: read to EOF, then poll for
37    /// appended lines (the run may still be writing).
38    File(PathBuf),
39    /// Drain a live channel, e.g. the receiver handed back by
40    /// `tau_runtime_tokio::drive_with_live_trace` — this is the exact
41    /// channel type that function returns, so a caller in `tau run` can
42    /// hand its `rx` straight to [`run_tui`] with no adapter.
43    Live(UnboundedReceiver<TraceEvent>),
44}

46/// Owns the render model + interactive UI state for the execution-trace TUI.
47///
48/// `apply_key` and `ingest_line` are the only mutators; both are pure
49/// (no I/O), which is what makes `apply_key` unit-testable without a
50/// terminal. [`run_tui`] is the impure shell that drives them from real
51/// input and a real trace source.
52pub struct App {
53    model: TraceModel,
54    ui: UiState,
55    /// `true` while `/` search-input mode is active: subsequent
56    /// character keys append to `ui.search` instead of being reduced as
57    /// navigation/filter commands.
58    search_mode: bool,
59    /// `true` (the tail -f default) while the newest span should stay
60    /// selected as more events arrive. Cleared by `Up` (the user is
61    /// browsing history); restored once `Down` catches back up to the
62    /// newest visible row.
63    follow: bool,
64}
65
66impl Default for App {
67    fn default() -> Self {
68        Self::new()
69    }
70}
71
72impl App {
73    /// Fresh app over an empty [`TraceModel`].
74    pub fn new() -> Self {
75        Self::with_model(TraceModel::new())
76    }
77
78    /// Fresh app over a caller-supplied model (test construction, and any
79    /// future "load a finished run's jsonl, then attach" path).
80    pub fn with_model(model: TraceModel) -> Self {
81        Self {
82            model,
83            ui: UiState::default(),
84            search_mode: false,
85            follow: true,
86        }
87    }
88
89    /// Index into the filtered-visible span list currently highlighted.
90    pub fn selected(&self) -> usize {
91        self.ui.selected
92    }
93
94    /// The active row filter.
95    pub fn filter(&self) -> Filter {
96        self.ui.filter
97    }
98
99    /// Whether the detail/expand toggle (`Enter`) is currently on.
100    pub fn expanded(&self) -> bool {
101        self.ui.expanded
102    }
103
104    /// Parse `line` as one jsonl trace event and fold it into the model.
105    ///
106    /// Silently no-ops on a blank line (`Ok(None)`) or a line that fails to
107    /// parse (`Err`) — callers that need to distinguish a genuine parse
108    /// error from a still-growing tail line (see `tau_trace::parse_line`'s
109    /// docs) probe with `tau_trace::parse_line` themselves before calling
110    /// this, as the file-tail loop below does.
111    pub fn ingest_line(&mut self, line: &str) {
112        if let Ok(Some(event)) = tau_trace::parse_line(line) {
113            self.apply_event(&event);
114        }
115    }
116
117    /// Fold an already-parsed event into the model (shared by
118    /// `ingest_line` and the live-channel loop, which has no jsonl line to
119    /// parse). Re-clamps `selected` and, while `follow` is on, snaps it to
120    /// the newest visible row.
121    fn apply_event(&mut self, event: &TraceEvent) {
122        self.model.apply(event);
123        self.clamp_selected();
124        if self.follow {
125            self.ui.selected = self.visible_len().saturating_sub(1);
126        }
127    }
128
129    /// Reduce one key press. Pure: no I/O, so this is unit-testable without
130    /// a real terminal.
131    ///
132    /// `Down`/`Up` move `selected` (clamped to the filtered-visible
133    /// length); `f` cycles [`Filter`] All -> Errors -> Tools -> Reasoning
134    /// -> All; `/` enters search-input mode (subsequent chars append to
135    /// `ui.search`, `Backspace` deletes, `Enter`/`Esc` exits search mode);
136    /// `Enter` outside search mode toggles the detail/expand flag; `q`/
137    /// `Q`/`Esc` outside search mode quits.
138    pub fn apply_key(&mut self, key: KeyCode) -> Loop {
139        if self.search_mode {
140            match key {
141                KeyCode::Enter | KeyCode::Esc => self.search_mode = false,
142                KeyCode::Backspace => {
143                    self.ui.search.pop();
144                }
145                KeyCode::Char(c) => self.ui.search.push(c),
146                _ => {}
147            }
148            self.clamp_selected();
149            return Loop::Continue;
150        }
151
152        match key {
153            KeyCode::Down => {
154                let last = self.visible_len().saturating_sub(1);
155                if self.ui.selected < last {
156                    self.ui.selected += 1;
157                }
158                if self.ui.selected >= last {
159                    self.follow = true;
160                }
161            }
162            KeyCode::Up => {
163                if self.ui.selected > 0 {
164                    self.ui.selected -= 1;
165                }
166                self.follow = false;
167            }
168            KeyCode::Char('f') => {
169                self.ui.filter = next_filter(self.ui.filter);
170                self.clamp_selected();
171            }
172            KeyCode::Char('/') => self.search_mode = true,
173            KeyCode::Enter => self.ui.expanded = !self.ui.expanded,
174            KeyCode::Char('q') | KeyCode::Char('Q') | KeyCode::Esc => return Loop::Quit,
175            _ => {}
176        }
177        Loop::Continue
178    }
179
180    /// Number of spans currently visible under `ui.filter` + `ui.search`.
181    /// Uses the same [`span_matches`] predicate `draw` filters with, so the
182    /// two can never drift out of sync.
183    fn visible_len(&self) -> usize {
184        self.model
185            .spans()
186            .iter()
187            .filter(|s| span_matches(self.ui.filter, s, &self.ui.search))
188            .count()
189    }
190
191    /// Clamp `ui.selected` into `[0, visible_len())` (or `0` if nothing is
192    /// visible) after any change that can shrink the visible set: a new
193    /// event, a filter cycle, or a search-text edit.
194    fn clamp_selected(&mut self) {
195        let len = self.visible_len();
196        self.ui.selected = if len == 0 {
197            0
198        } else {
199            self.ui.selected.min(len - 1)
200        };
201    }
202}
203
204/// `All -> Errors -> Tools -> Reasoning -> All`.
205fn next_filter(filter: Filter) -> Filter {
206    match filter {
207        Filter::All => Filter::Errors,
208        Filter::Errors => Filter::Tools,
209        Filter::Tools => Filter::Reasoning,
210        Filter::Reasoning => Filter::All,
211    }
212}
213
214/// How long each iteration blocks waiting for a key press before polling
215/// the trace source again.
216const POLL_INTERVAL: Duration = Duration::from_millis(150);
217
218/// Restores the terminal on every exit path — normal return, `?`
219/// early-return, or panic (ratatui's `init()` installs a panic hook that
220/// calls `restore()` too; this `Drop` impl is the belt to that hook's
221/// braces, covering the non-panic early-return paths a hook can't see).
222struct TerminalGuard;
223
224impl Drop for TerminalGuard {
225    fn drop(&mut self) {
226        ratatui::restore();
227    }
228}

230/// Run the execution-trace TUI to completion.
231///
232/// Owns raw-mode + alternate-screen setup via `ratatui::init()` and
233/// guarantees teardown via [`TerminalGuard`]. Polls both keyboard input
234/// and `source` on a fixed interval, redrawing after each batch; returns
235/// once the user quits (`q`/`Esc`) or the underlying I/O fails.
236pub fn run_tui(source: TraceSource) -> anyhow::Result<()> {
237    let mut terminal = ratatui::init();
238    let _guard = TerminalGuard;
239    let mut app = App::new();
240
241    match source {
242        TraceSource::File(path) => run_with_file(&mut terminal, &mut app, &path),
243        TraceSource::Live(rx) => run_with_live(&mut terminal, &mut app, rx),
244    }
245}
246
247/// Handle one polled input event: reduce a key press through `apply_key`,
248/// or just let a resize fall through to the next `draw` (ratatui
249/// re-computes layout from `frame.area()` every call, so a resize needs no
250/// special handling here — see the M1 constraint against reworking
251/// `draw`'s fixed-width bar).
252///
253/// Returns `true` once the app should quit.
254fn handle_input(app: &mut App) -> anyhow::Result<bool> {
255    if !event::poll(POLL_INTERVAL)? {
256        return Ok(false);
257    }
258    match event::read()? {
259        Event::Key(key) if key.kind == KeyEventKind::Press => {
260            Ok(matches!(app.apply_key(key.code), Loop::Quit))
261        }
262        _ => Ok(false),
263    }
264}
265
266fn run_with_live(
267    terminal: &mut ratatui::DefaultTerminal,
268    app: &mut App,
269    mut rx: UnboundedReceiver<TraceEvent>,
270) -> anyhow::Result<()> {
271    loop {
272        while let Ok(event) = rx.try_recv() {
273            app.apply_event(&event);
274        }
275
276        terminal.draw(|f| draw(f, &app.model, &app.ui))?;
277
278        if handle_input(app)? {
279            return Ok(());
280        }
281    }
282}

284fn run_with_file(
285    terminal: &mut ratatui::DefaultTerminal,
286    app: &mut App,
287    path: &Path,
288) -> anyhow::Result<()> {
289    let mut file =
290        File::open(path).with_context(|| format!("opening trace file {}", path.display()))?;
291    let mut pos: u64 = 0;
292    let mut buf = String::new();
293
294    loop {
295        poll_file_tail(&mut file, &mut pos, &mut buf, app)?;
296
297        terminal.draw(|f| draw(f, &app.model, &app.ui))?;
298
299        if handle_input(app)? {
300            return Ok(());
301        }
302    }
303}
304
305/// Read whatever `file` has grown by since `pos` and ingest every
306/// complete (newline-terminated) line.
307///
308/// A still-growing last line (no trailing `\n` yet) is speculatively
309/// parsed too: if it already parses (the writer flushed the JSON before
310/// the trailing newline), it is ingested and consumed like any other
311/// line. If it fails to parse, that is treated as the expected
312/// mid-write/truncation artifact documented on `tau_trace::parse_line` —
313/// `pos` is *not* advanced past it, so the same bytes (plus whatever the
314/// writer appends next) are re-read on the next poll instead of being
315/// dropped or crashing the loop. A read that lands mid a multi-byte UTF-8
316/// character at the current EOF is treated the same way.
317fn poll_file_tail(
318    file: &mut File,
319    pos: &mut u64,
320    buf: &mut String,
321    app: &mut App,
322) -> anyhow::Result<()> {
323    file.seek(SeekFrom::Start(*pos))?;
324    buf.clear();
325    match file.read_to_string(buf) {
326        Ok(0) => return Ok(()),
327        Ok(_) => {}
328        Err(e) if e.kind() == ErrorKind::InvalidData => return Ok(()),
329        Err(e) => return Err(e).context("tailing trace file"),
330    }
331
332    let ends_with_newline = buf.ends_with('\n');
333    let mut segments: Vec<&str> = buf.split('\n').collect();
334    let trailing_partial = if ends_with_newline {
335        segments.pop(); // the empty segment after the final '\n'
336        None
337    } else {
338        segments.pop()
339    };
340
341    let mut consumed: u64 = 0;
342    for line in &segments {
343        app.ingest_line(line);
344        consumed += line.len() as u64 + 1; // + the '\n' that terminated it
345    }
346
347    if let Some(partial) = trailing_partial.filter(|p| !p.is_empty()) {
348        match tau_trace::parse_line(partial) {
349            Ok(_) => {
350                app.ingest_line(partial);
351                consumed += partial.len() as u64;
352            }
353            Err(_) => {
354                // Retryable: leave `pos` before this segment.
355            }
356        }
357    }
358
359    *pos += consumed;
360    Ok(())
361}

538    #[test]
539    fn poll_file_tail_retries_a_truncated_last_line() {
540        use std::io::Write;
541
542        let evt = TraceEvent {
543            id: "evt-1".into(),
544            ts: Utc.timestamp_opt(100, 0).unwrap(),
545            run_id: "run-1".into(),
546            agent_id: Some("agent-1".into()),
547            kind: TraceEventKind::ToolCall {
548                tool_name: "net.http".into(),
549                duration_ms: 10,
550                status: "ok".into(),
551                capability: None,
552                turn_index: 0,
553            },
554        };
555        let full_line = serde_json::to_string(&evt).unwrap();
556        let truncated = &full_line[..full_line.len() / 2];
557
558        let dir = tempfile::tempdir().unwrap();
559        let path = dir.path().join("run.jsonl");
560        std::fs::write(&path, truncated).unwrap();
561
562        let mut file = File::open(&path).unwrap();
563        let mut pos: u64 = 0;
564        let mut buf = String::new();
565        let mut app = App::new();
566
567        // First poll: only a truncated line is on disk. Must not crash,
568        // must not consume it (no event applied, `pos` stays at 0).
569        poll_file_tail(&mut file, &mut pos, &mut buf, &mut app).unwrap();
570        assert_eq!(app.model.spans().len(), 0);
571        assert_eq!(pos, 0);
572
573        // Writer finishes flushing the line (+ newline). Re-open to append
574        // since `file` above was opened read-only.
575        {
576            let mut f = std::fs::OpenOptions::new()
577                .append(true)
578                .open(&path)
579                .unwrap();
580            writeln!(f, "{}", &full_line[full_line.len() / 2..]).unwrap();
581        }
582
583        // Second poll re-reads from `pos` (still 0) and now sees the
584        // complete, newline-terminated line.
585        poll_file_tail(&mut file, &mut pos, &mut buf, &mut app).unwrap();
586        assert_eq!(app.model.spans().len(), 1);
587        assert!(pos > 0);
588    }
