1//! Pure frame rendering for the execution-trace waterfall TUI.
2//!
3//! [`draw`] renders a [`tau_trace::TraceModel`] snapshot plus [`UiState`]
4//! into a ratatui [`Frame`]. It performs no terminal I/O — no raw-mode
5//! setup, no alternate-screen switch, no stdout writes — so the whole
6//! layout is exercisable with ratatui's `TestBackend`, as the test below
7//! does. Terminal setup/teardown and the input event loop belong to the
8//! caller (a later task), not here.
9
10use ratatui::layout::{Constraint, Direction, Layout, Rect};
11use ratatui::style::{Color, Modifier, Style};
12use ratatui::widgets::{Block, Borders, Cell, Paragraph, Row, Table, Wrap};
13use ratatui::Frame;
14
15use tau_ports::CapabilityVerdict;
16use tau_trace::{Span as TraceSpan, SpanKind, SpanStatus, TraceModel};

18/// Fixed column width (in cells) of the waterfall bar, and the
19/// `width_cols` passed to [`TraceModel::bar`] for every row so the
20/// glyph string built in [`bar_cell`] lines up with the column.
21const BAR_WIDTH: u16 = 20;
22
23/// Which spans are visible in the waterfall table.
24#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
25pub enum Filter {
26    /// Show every span.
27    #[default]
28    All,
29    /// Only spans whose status is `Failed`.
30    Errors,
31    /// Only `Tool`-kind spans.
32    Tools,
33    /// Only `Reasoning`-kind spans.
34    Reasoning,
35}
36
37impl Filter {
38    /// Short label rendered as this filter's toolbar chip.
39    fn label(self) -> &'static str {
40        match self {
41            Filter::All => "All",
42            Filter::Errors => "Errors",
43            Filter::Tools => "Tools",
44            Filter::Reasoning => "Reasoning",
45        }
46    }
47}
48
49/// Whether `span` is visible under `filter` + a `search` substring match
50/// over `span.label`.
51///
52/// The single shared predicate for span visibility — `draw`'s table/detail
53/// filtering and `app::App`'s `visible_len`/selection-clamp both call this
54/// so the two can never drift out of sync (they used to be two hand-kept
55/// copies of the same match arms; see the M1 final-review fix wave).
56pub(crate) fn span_matches(filter: Filter, span: &TraceSpan, search: &str) -> bool {
57    let filter_matches = match filter {
58        Filter::All => true,
59        Filter::Errors => matches!(span.status, SpanStatus::Failed),
60        Filter::Tools => matches!(span.kind, SpanKind::Tool),
61        Filter::Reasoning => matches!(span.kind, SpanKind::Reasoning),
62    };
63    filter_matches && (search.is_empty() || span.label.contains(search))
64}

66/// Interactive state that [`draw`] renders. Owned and mutated by the
67/// event loop (a later task); this module only ever reads it.
68#[derive(Debug, Clone, Default)]
69pub struct UiState {
70    /// Index into the *filtered* (currently visible) span list that is
71    /// highlighted in the table and shown in the detail pane.
72    pub selected: usize,
73    /// Active row filter.
74    pub filter: Filter,
75    /// Free-text search over span labels (substring match); empty means
76    /// no search filtering.
77    pub search: String,
78    /// Vertical scroll offset (in rows) into the detail pane.
79    pub scroll: u16,
80}
81
82/// Render `model` + `ui` into `frame`.
83///
84/// Pure frame rendering: lays out a toolbar (run summary, filter chips,
85/// search box), a waterfall table (`Name | Tokens | Dur | Cap | Bar`)
86/// filtered by `ui.filter`/`ui.search`, and a detail pane for the span at
87/// `ui.selected`. Performs no terminal I/O, so it is safe to drive with
88/// `ratatui::backend::TestBackend`.
89pub fn draw(frame: &mut Frame, model: &TraceModel, ui: &UiState) {
90    let area = frame.area();
91    let chunks = Layout::default()
92        .direction(Direction::Vertical)
93        .constraints([
94            Constraint::Length(3),
95            Constraint::Min(4),
96            Constraint::Length(3),
97        ])
98        .split(area);
99
100    draw_toolbar(frame, chunks[0], ui);
101
102    let visible: Vec<&TraceSpan> = model
103        .spans()
104        .iter()
105        .filter(|s| span_matches(ui.filter, s, &ui.search))
106        .collect();
107
108    draw_table(frame, chunks[1], model, &visible, ui);
109    draw_detail(frame, chunks[2], &visible, ui);
110}

112/// Toolbar: static title, one chip per [`Filter`] (the active one
113/// bracketed), and the current search text.
114fn draw_toolbar(frame: &mut Frame, area: Rect, ui: &UiState) {
115    let chips = [
116        Filter::All,
117        Filter::Errors,
118        Filter::Tools,
119        Filter::Reasoning,
120    ]
121    .iter()
122    .map(|f| {
123        if *f == ui.filter {
124            format!("[{}]", f.label())
125        } else {
126            format!(" {} ", f.label())
127        }
128    })
129    .collect::<Vec<_>>()
130    .join(" ");
131    let text = format!("tau trace   {chips}   search: {}", ui.search);
132    let block = Block::default().borders(Borders::ALL).title("tau trace");
133    frame.render_widget(Paragraph::new(text).block(block), area);
134}

136/// Waterfall table: one row per visible span, columns
137/// `Name | Tokens | Dur | Cap | Bar`. The `Cap` cell is colored by
138/// [`CapabilityVerdict`] (green allow / amber clamp / red drop); the
139/// currently-selected row is reverse-video highlighted.
140fn draw_table(
141    frame: &mut Frame,
142    area: Rect,
143    model: &TraceModel,
144    visible: &[&TraceSpan],
145    ui: &UiState,
146) {
147    let header = Row::new(vec!["Name", "Tokens", "Dur", "Cap", "Bar"])
148        .style(Style::default().add_modifier(Modifier::BOLD));
149
150    let rows: Vec<Row> = visible
151        .iter()
152        .enumerate()
153        .map(|(i, span)| {
154            let tokens = span
155                .tokens
156                .map(|t| t.to_string())
157                .unwrap_or_else(|| "-".to_string());
158            let (badge, badge_style) = capability_badge(span.capability.as_ref());
159            let mut row = Row::new(vec![
160                Cell::from(span.label.clone()),
161                Cell::from(tokens),
162                Cell::from(duration_label(span)),
163                Cell::from(badge).style(badge_style),
164                Cell::from(bar_cell(model, span)),
165            ]);
166            if i == ui.selected {
167                row = row.style(Style::default().add_modifier(Modifier::REVERSED));
168            }
169            row
170        })
171        .collect();
172
173    let widths = [
174        Constraint::Length(30),
175        Constraint::Length(8),
176        Constraint::Length(10),
177        Constraint::Length(20),
178        Constraint::Length(BAR_WIDTH),
179    ];
180
181    let table = Table::new(rows, widths)
182        .header(header)
183        .block(Block::default().borders(Borders::ALL).title("Waterfall"));
184    frame.render_widget(table, area);
185}

187/// Detail pane for the span at `ui.selected` within `visible`. Rendered as
188/// a single wrapped line (rather than one line per field) so it stays
189/// legible even in the minimum-height layout a small terminal forces.
190fn draw_detail(frame: &mut Frame, area: Rect, visible: &[&TraceSpan], ui: &UiState) {
191    let block = Block::default().borders(Borders::ALL).title("Detail");
192    let text = match visible.get(ui.selected) {
193        Some(span) => format!(
194            "{}  |  kind: {:?}  |  status: {:?}  |  tokens: {}  |  capability: {}",
195            span.label,
196            span.kind,
197            span.status,
198            span.tokens
199                .map(|t| t.to_string())
200                .unwrap_or_else(|| "-".to_string()),
201            capability_detail(span.capability.as_ref()),
202        ),
203        None => "(no span selected)".to_string(),
204    };
205    frame.render_widget(
206        Paragraph::new(text)
207            .block(block)
208            .wrap(Wrap { trim: true })
209            .scroll((ui.scroll, 0)),
210        area,
211    );
212}

214/// Short badge word + verdict color for the `Cap` table cell.
215fn capability_badge(verdict: Option<&CapabilityVerdict>) -> (String, Style) {
216    match verdict {
217        None => ("-".to_string(), Style::default()),
218        Some(CapabilityVerdict::Allow) => ("allow".to_string(), Style::default().fg(Color::Green)),
219        Some(CapabilityVerdict::Clamp { to }) => {
220            (format!("clamp:{to}"), Style::default().fg(Color::Yellow))
221        }
222        Some(CapabilityVerdict::Drop { reason }) => {
223            (format!("drop:{reason}"), Style::default().fg(Color::Red))
224        }
225    }
226}
227
228/// Verbose verdict line for the detail pane.
229fn capability_detail(verdict: Option<&CapabilityVerdict>) -> String {
230    match verdict {
231        None => "-".to_string(),
232        Some(CapabilityVerdict::Allow) => "allow".to_string(),
233        Some(CapabilityVerdict::Clamp { to }) => format!("clamp -> {to}"),
234        Some(CapabilityVerdict::Drop { reason }) => format!("drop: {reason}"),
235    }
236}

238/// `"<ms>ms"` for a completed span, `"…"` while still running.
239fn duration_label(span: &TraceSpan) -> String {
240    match span.end {
241        Some(end) => {
242            let ms = (end - span.start).num_milliseconds().max(0);
243            format!("{ms}ms")
244        }
245        None => "…".to_string(),
246    }
247}

249/// Build a `BAR_WIDTH`-wide glyph string for `span`'s waterfall bar:
250/// `░` background, `▊` over the span's `[offset, offset+len)` columns as
251/// returned by [`TraceModel::bar`].
252fn bar_cell(model: &TraceModel, span: &TraceSpan) -> String {
253    let (offset, len) = model.bar(span, BAR_WIDTH);
254    let mut glyphs = vec!['░'; BAR_WIDTH as usize];
255    let start = usize::from(offset).min(glyphs.len());
256    let end = start.saturating_add(usize::from(len)).min(glyphs.len());
257    for glyph in &mut glyphs[start..end] {
258        *glyph = '▊';
259    }
260    glyphs.into_iter().collect()
261}

263#[cfg(test)]
264mod tests {
265    use super::*;
266    use chrono::{TimeZone, Utc};
267    use ratatui::backend::TestBackend;
268    use ratatui::Terminal;
269    use tau_ports::{TraceEvent, TraceEventKind};
270
271    #[test]
272    fn renders_tool_row_with_capability_badge() {
273        let mut model = TraceModel::new();
274        model.apply(&TraceEvent {
275            id: "evt-1".into(),
276            ts: Utc.timestamp_opt(100, 0).unwrap(),
277            run_id: "run-1".into(),
278            agent_id: Some("agent-1".into()),
279            kind: TraceEventKind::ToolCall {
280                tool_name: "net.http".into(),
281                duration_ms: 250,
282                status: "ok".into(),
283                capability: Some(CapabilityVerdict::Drop {
284                    reason: "egress denied".into(),
285                }),
286            },
287        });
288
289        let mut term = Terminal::new(TestBackend::new(120, 10)).unwrap();
290        let ui = UiState {
291            selected: 0,
292            filter: Filter::All,
293            search: String::new(),
294            scroll: 0,
295        };
296        term.draw(|f| draw(f, &model, &ui)).unwrap();
297        let buf = term.backend().buffer().clone();
298        let text: String = buf.content().iter().map(|c| c.symbol()).collect();
299
300        assert!(text.contains("drop"), "expected a drop badge in:\n{text}");
301        assert!(
302            text.contains("net.http"),
303            "expected the tool label in:\n{text}"
304        );
305    }
