26/// Top-level entry point used by `main` and integration tests.
27///
28/// Parses CLI arguments, dispatches to the appropriate `cmd::*::run`
29/// handler, and maps the result to a process exit code.
30///
31/// At v0.1 of Task 4, all subcommand handlers are stubs that return
32/// an "unimplemented" error. Tasks 10-14 land the real handlers.
33pub async fn run_main() -> std::process::ExitCode {
34    let cli = cli::Cli::parse();
35    // When the user is running `tau workflow run`, mint the run id and
36    // attach a `WorkflowRunLogLayer` to the global subscriber so step
37    // events written by the runner end up in the on-disk JSONL log.
38    // The minted run id is then handed to the workflow runner via
39    // `RunOpts::run_id` so both producer and layer point at the same
40    // file. If scope resolution fails here the install proceeds without
41    // the layer; the workflow handler will surface the same error
42    // shortly after with a richer context message.
43    let prepared_workflow_run = prepare_workflow_run_layer(&cli.command);
44    let mut extra_layers: Vec<
45        Box<
46            dyn ::tracing_subscriber::Layer<::tracing_subscriber::Registry> + Send + Sync + 'static,
47        >,
48    > = Vec::new();
49    let workflow_run_id = if let Some(prep) = prepared_workflow_run {
50        extra_layers.push(Box::new(prep.layer));
51        Some(prep.run_id)
52    } else {
53        None
54    };
55    // Sub-project D Task 5: when `--record-protocol <path>` is set,
56    // install a `PluginRecordingLayer` aimed at that path. The same
57    // layer handle is stashed in a process-global slot so the
58    // top-level command handlers (cmd::run / cmd::chat / cmd::plugin
59    // describe) can call `flush()` on exit to ensure pending writes
60    // reach disk before the process terminates.
61    //
62    // The layer is wrapped with a per-layer `FilterFn` that always
63    // enables the `tau::plugin::frame` target regardless of the
64    // global `EnvFilter`, and pins its `max_level_hint` to TRACE so
65    // the tracing macros don't short-circuit recording events when
66    // the user's filter is more restrictive than INFO (e.g.
67    // `--quiet` → `tau=warn`). Protocol recording is opt-in via the
68    // explicit `--record-protocol` flag — gating it behind any
69    // particular log-level directive would surprise users who
70    // reasonably expect "the flag is set, ergo frames are recorded".
71    if let Some(path) = cli.record_protocol.clone() {
72        use tracing_subscriber::filter::{filter_fn, LevelFilter};
73        use tracing_subscriber::layer::Layer as _;
74        let layer = tau_observe::layers::plugin_recording::PluginRecordingLayer::new(path);
75        crate::cmd::plugin_loader::set_plugin_recording_layer(layer.clone());
76        let per_layer_filter =
77            filter_fn(|meta| meta.target() == tau_observe::layers::plugin_recording::TARGET)
78                .with_max_level_hint(LevelFilter::TRACE);
79        let filtered = layer.with_filter(per_layer_filter);
80        extra_layers.push(Box::new(filtered));
81    }
82    tracing::install_with_extra_layers(&cli, extra_layers);
83    // Capture `cli.debug` before `dispatch` consumes the parsed `Cli`.
84    // When set, the error path renders the full `anyhow` chain via
85    // `{err:?}` instead of the single-line top-level message. This is
86    // the integration-level surface for `--debug` (the tracing module
87    // already promotes the filter to DEBUG independently).
88    let debug = cli.debug;
89    match dispatch(cli, workflow_run_id).await {
90        Ok(()) => ExitCode::Success.into(),
91        Err(err) => {
92            // The AgentFailed marker is emitted by `cmd::run` when the
93            // agent reaches `RunOutcome::Failed`. It must NOT print the
94            // generic "error:" prefix — the run handler has already
95            // emitted a structured failure to the user. All other
96            // errors are kernel/CLI failures; they get the prefix and
97            // map to `ExitCode::Error`.
98            if let Some(bvf) = err.downcast_ref::<crate::cmd::run::BundleVerifyFailed>() {
99                // D9: `tau run --bundle` verification failure. The message
100                // is already structured by `render_verify_error`; print it
101                // as-is (no generic "error:" prefix) and use the mapped
102                // §C.3 exit code instead of the catch-all ExitCode::Error.
103                eprint!("{}", bvf.rendered);
104                std::process::ExitCode::from(bvf.code as u8)
105            } else if err.downcast_ref::<crate::cmd::run::AgentFailed>().is_some() {
106                ExitCode::AgentFailed.into()
107            } else if err
108                .downcast_ref::<crate::cmd::run::SuspendedRun>()
109                .is_some()
110            {
111                // A pipeline paused at a `Suspend` step (HITL, EPIC 4.3). Not
112                // a failure: `render_pipeline_outcome` already rendered the
113                // structured `{"outcome":"suspended", ...}` payload (or the
114                // human "Resume with: tau run --resume ..." message), so —
115                // like AgentFailed — skip the generic "error:" prefix and map
116                // straight to the dedicated exit code.
117                ExitCode::Suspended.into()
118            } else {
119                if debug {
120                    eprintln!("error: {err:?}");
121                } else {
122                    eprintln!("error: {err}");
123                }
124                ExitCode::from(&err).into()
125            }
126        }
127    }
128}

130/// Outcome of inspecting the parsed CLI for a `tau workflow run`
131/// invocation: a pre-minted run id and the run-log layer to install.
132struct PreparedWorkflowRun {
133    run_id: String,
134    layer: tau_observe::layers::workflow_run_log::WorkflowRunLogLayer,
135}
136
137/// If `command` is `Workflow(Run(...))`, resolve the scope, mint a fresh
138/// run id, and build the `WorkflowRunLogLayer` pointing at the same
139/// `<scope>/.tau/workflow-runs/<name>-<run_id>.jsonl` path the runner
140/// will use. Returns `None` for any other command, or when scope
141/// resolution fails (the workflow handler will re-surface that error).
142fn prepare_workflow_run_layer(command: &cli::Command) -> Option<PreparedWorkflowRun> {
143    let cli::Command::Workflow(cli::WorkflowSubcommand::Run(run_args)) = command else {
144        return None;
145    };
146    let cwd = std::env::current_dir().ok()?;
147    let scope = tau_pkg::Scope::resolve(&cwd).ok()?;
148    let run_id = ulid::Ulid::new().to_string();
149    let log_path = tau_workflow::run_log_path(scope.path(), &run_args.name, &run_id);
150    let layer = tau_observe::layers::workflow_run_log::WorkflowRunLogLayer::new(log_path);
151    Some(PreparedWorkflowRun { run_id, layer })
152}

154async fn dispatch(cli: cli::Cli, workflow_run_id: Option<String>) -> anyhow::Result<()> {
155    use cli::SandboxKindArg;
156    use tau_runtime_tokio::process_gate::registry::RegistryKind;
157
158    let mut output = Output::from_cli(&cli);
159    let record_protocol = cli.record_protocol.clone();
160
161    // Resolve sandbox override flags (--no-sandbox / --sandbox <kind>) once,
162    // so every plugin-loading subcommand gets the same derived values.
163    let force_passthrough =
164        cli.no_sandbox || matches!(cli.sandbox, Some(SandboxKindArg::Passthrough));
165    let force_adapter_kind: Option<RegistryKind> = match cli.sandbox {
166        Some(SandboxKindArg::Native) => Some(RegistryKind::Native),
167        Some(SandboxKindArg::Container) => Some(RegistryKind::Container),
168        Some(SandboxKindArg::Passthrough) | None => None,
169    };
170
171    match cli.command {
172        cli::Command::Init(args) => cmd::init::run(&args, &mut output).await,
173        cli::Command::Install(args) => cmd::install::run(&args, &mut output).await,
174        cli::Command::List(args) => cmd::list::run(&args, &mut output).await,
175        cli::Command::Run(args) => {
176            cmd::run::run(
177                &args,
178                record_protocol,
179                force_passthrough,
180                force_adapter_kind,
181                &mut output,
182            )
183            .await
184        }
185        cli::Command::Chat(args) => {
186            cmd::chat::run(
187                &args,
188                record_protocol,
189                force_passthrough,
190                force_adapter_kind,
191                &mut output,
192            )
193            .await
194        }
195        cli::Command::Dev(args) => cmd::dev::run(args, &mut output).await,
196        cli::Command::Embed(ref args) => cmd::embed::run(args, &mut output).await,
197        cli::Command::Resolve(args) => cmd::resolve::run(&args, &mut output).await,
198        cli::Command::Uninstall(args) => cmd::uninstall::run(&args, &mut output).await,
199        cli::Command::Update(args) => cmd::update::run(&args, &mut output).await,
200        cli::Command::Verify(args) => cmd::verify::run(&args, &mut output).await,
201        cli::Command::Plugin { action } => {
202            cmd::plugin::dispatch(action, record_protocol, &mut output).await
203        }
204        cli::Command::Session(args) => cmd::session::run(&args, &mut output).await,
205        cli::Command::Sandbox(args) => cmd::sandbox::run(&args, &mut output).await,
206        cli::Command::Skill(sub) => cmd::skill::dispatch(sub, &mut output).await,
207        cli::Command::Target(sub) => cmd::target::run(&sub, &mut output).await,
208        cli::Command::Workflow(sub) => {
209            cmd::workflow::dispatch(sub, workflow_run_id, &mut output).await
210        }
211        cli::Command::Serve(args) => cmd::serve::run(&args).await,
212        cli::Command::Check(args) => cmd::check::run(args).await,
213        cli::Command::Build(build_cmd) => match build_cmd.subcommand {
214            Some(cli::BuildSubcommand::Wasm(ref wasm_args)) => {
215                cmd::build_wasm::run(wasm_args, &mut output).await
216            }
217            None => cmd::build::run(&build_cmd.args, &mut output).await,
218        },
219        cli::Command::Mcp(sub) => cmd::mcp::dispatch(sub, &mut output).await,
220    }
221}
