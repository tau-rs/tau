1//! Command-line argument parser. clap v4 derive.
2//!
3//! [`Cli`] is the top-level parser; [`Command`] enumerates the v0.1
4//! subcommands plus the debug-tier `plugin` subcommand group. Each
5//! subcommand has its own `*Args` struct.
6//!
7//! Global flags (verbose, quiet, debug, color, json, record-protocol)
8//! are accessible from any subcommand via the parsed [`Cli`].
9
10use std::path::PathBuf;
11
12use clap::{Args, Parser, Subcommand, ValueEnum};
13
14/// Top-level parser for the `tau` binary.
15#[derive(Parser, Debug)]
16#[command(name = "tau", version, about = "tau runtime CLI")]
17pub struct Cli {
18    /// The subcommand to dispatch.
19    #[command(subcommand)]
20    pub command: Command,
21
22    /// Verbosity level. Repeat for more (-v: DEBUG, -vv: TRACE).
23    #[arg(short, long, action = clap::ArgAction::Count, global = true)]
24    pub verbose: u8,
25
26    /// Suppress non-error output (sets tracing to WARN).
27    #[arg(short, long, global = true, conflicts_with = "verbose")]
28    pub quiet: bool,
29
30    /// Show full error chain + DEBUG-level tracing on failures.
31    #[arg(long, global = true)]
32    pub debug: bool,
33
34    /// Color mode for terminal output.
35    #[arg(long, global = true, value_enum, default_value = "auto")]
36    pub color: ColorMode,
37
38    /// Emit structured JSON output instead of human-readable.
39    /// Only supported on `install`, `list`, `run`, `init`.
40    #[arg(long, global = true)]
41    pub json: bool,
42
43    /// Mirror every plugin protocol frame to a JSONL file at this path.
44    ///
45    /// Useful for debugging plugin behavior and replaying via
46    /// `tau plugin protocol decode <path>`. The recording is best-effort
47    /// per spec §7.8: write/open failures are logged but never abort the
48    /// invocation.
49    #[arg(long, global = true, value_name = "PATH")]
50    pub record_protocol: Option<PathBuf>,
51
52    /// Disable sandboxing entirely for this invocation (shorthand for
53    /// `--sandbox passthrough`). For development and one-off debugging.
54    /// Auditable in shell history.
55    #[arg(long, global = true)]
56    pub no_sandbox: bool,
57
58    /// Force a specific sandbox adapter kind, overriding the resolver.
59    /// `--no-sandbox` is shorthand for `--sandbox passthrough`. Conflicts
60    /// with `--no-sandbox` if both set.
61    #[arg(long, global = true, value_enum, conflicts_with = "no_sandbox")]
62    pub sandbox: Option<SandboxKindArg>,
63
64    /// Write logs through a non-blocking MPSC channel (high-throughput).
65    /// Requires `--log-file`. Env: `TAU_LOG_NON_BLOCKING=1`.
66    #[arg(long, global = true, env = "TAU_LOG_NON_BLOCKING")]
67    pub log_non_blocking: bool,
68
69    /// Write logs to the given file instead of stderr. Required when
70    /// `--log-non-blocking` is set. Env: `TAU_LOG_FILE`.
71    #[arg(long, global = true, env = "TAU_LOG_FILE", value_name = "PATH")]
72    pub log_file: Option<PathBuf>,
73
74    /// Log file rotation policy. Ignored unless `--log-file` is set.
75    /// Env: `TAU_LOG_ROTATION`.
76    #[arg(
77        long,
78        global = true,
79        value_enum,
80        default_value_t,
81        env = "TAU_LOG_ROTATION"
82    )]
83    pub log_rotation: LogRotationCli,
84
85    /// Export traces over OTLP/gRPC to this endpoint, e.g.
86    /// `https://otel.example.com:4317`. Falls back to
87    /// `OTEL_EXPORTER_OTLP_ENDPOINT` env var. Auth + tenant headers
88    /// come from `OTEL_EXPORTER_OTLP_HEADERS` (comma-separated
89    /// `key=value` pairs). Requires the `otlp` cargo feature (on by
90    /// default).
91    #[arg(
92        long,
93        env = "OTEL_EXPORTER_OTLP_ENDPOINT",
94        global = true,
95        value_name = "URL"
96    )]
97    pub otlp_endpoint: Option<String>,
98}

100/// CLI value for `--log-rotation <policy>`. Translates to
101/// `tau_observe::install::Rotation` at the install boundary so
102/// `tau-observe` stays free of any clap dependency.
103#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, ValueEnum)]
104#[value(rename_all = "kebab-case")]
105pub enum LogRotationCli {
106    /// No rotation — append forever to a single file.
107    #[default]
108    Never,
109    /// Roll over each day at UTC midnight (filename gains a date suffix).
110    Daily,
111    /// Roll over each hour (filename gains a date+hour suffix).
112    Hourly,
113}
114
115impl From<LogRotationCli> for tau_observe::install::Rotation {
116    fn from(value: LogRotationCli) -> Self {
117        match value {
118            LogRotationCli::Never => Self::Never,
119            LogRotationCli::Daily => Self::Daily,
120            LogRotationCli::Hourly => Self::Hourly,
121        }
122    }
123}

161    Update(UpdateArgs),

163    Plugin {
164        /// Sub-action within the plugin group.
165        #[command(subcommand)]
166        action: PluginAction,
167    },
168    /// Session management (list, show, delete, export).
169    Session(SessionArgs),
170    /// Sandbox configuration and diagnostics.
171    Sandbox(SandboxArgs),
172    /// Workflow subcommand group (list, run, log, resume).
173    #[command(subcommand)]
174    Workflow(WorkflowSubcommand),
175    /// Inspect installed skill packages (list, show).
176    #[command(subcommand)]
177    Skill(SkillSubcommand),
178    /// Inspect the deployment-target registry (list, show).
179    #[command(subcommand)]
180    Target(TargetSubcommand),
181    /// Start serve mode: accept JSON-RPC requests over stdio.
182    Serve(ServeArgs),
183    /// Run pre-flight validation against the project (config, lockfile,
184    /// packages, sandbox, plugins, skills, mcp-contracts, governance).
185    /// Aggregates existing validators into one CI/IDE-friendly verb.
186    Check(CheckArgs),
187    /// Build a deployment bundle from this project (Phase 2 §C.2),
188    /// or compile a workflow to WebAssembly (`tau build wasm <project>`).
189    Build(BuildCommand),
190    /// Manage Model Context Protocol (MCP) server contracts.
191    #[command(subcommand)]
192    Mcp(McpSubcommand),
193    /// Hot-reload REPL driving the post-β.3 IR runtime.
194    ///
195    /// `tau dev <project>` boots a stdin REPL with file-watching;
196    /// editing `tau.toml` triggers a `:reload` hint that preserves
197    /// conversation history.
198    Dev(DevArgs),
199    /// Emit host-embedding glue for a target language (Phase 2 §5.2).
200    Embed(EmbedArgs),
201}

203/// Top-level arguments for `tau build [wasm] [...]`.
204///
205/// When `subcommand` is `Some(BuildSubcommand::Wasm(...))` the command
206/// compiles the workflow IR to a WebAssembly module (β.7.5). When
207/// `subcommand` is `None`, the flattened [`BuildArgs`] fields apply and
208/// the bundle pipeline runs unchanged.
209#[derive(Args, Debug)]
210pub struct BuildCommand {
211    /// Optional subcommand (`wasm`). When present the remaining args are
212    /// parsed by [`BuildWasmArgs`]; when absent, the flattened bundle args apply.
213    #[command(subcommand)]
214    pub subcommand: Option<BuildSubcommand>,
215
216    /// Bundle arguments (only meaningful when `subcommand` is `None`).
217    #[command(flatten)]
218    pub args: BuildArgs,
219}

228    /// lands in β.7.5 Phase 4.

230}

254/// Arguments for `tau build` (bundle — Phase 2 §C.2).
255#[derive(Args, Debug)]
256pub struct BuildArgs {
257    /// Path to the project to build. May be a directory (containing
258    /// `tau.toml`) or a `.ts` file (β.8 TypeScript-authoring surface).
259    /// Defaults to the current working directory.
260    #[arg(value_name = "PROJECT")]
261    pub project: Option<std::path::PathBuf>,
262    /// What to build: an artifact kind (`wasm-guest`, `rust-lib`) or an
263    /// Available hardware triple (default: host → `.tau` bundle). `wasm-guest`
264    /// emits the fully-linked wasm component; `rust-lib` emits a generated
265    /// no_std Rust embedding crate (EPIC 5.1).
266    #[arg(long, value_name = "KIND|TRIPLE")]
267    pub target: Option<String>,
268    /// Output path (default: `<project>/<name>-<version>.tau`).
269    #[arg(long, short = 'o', value_name = "PATH")]
270    pub output: Option<std::path::PathBuf>,
271    /// Restrict the bundle to one or more agents (repeatable:
272    /// `--agent a --agent b`). When omitted, every agent is built.
273    /// Selecting agents also prunes packages they don't reference.
274    /// To avoid overwriting a full bundle at the default path, pass
275    /// `-o`/`--output`.
276    #[arg(long = "agent", value_name = "ID")]
277    pub agents: Vec<String>,
278    /// Use pinned `.tau/mcp/<entry>.contract.json` files for MCP
279    /// contracts; never reach out to live MCP servers. Build errors
280    /// out if any pinned file is missing.
281    #[arg(long)]
282    pub offline: bool,
283    /// Also emit host-adapter descriptors for the project's cron triggers.
284    /// `systemd` writes `.timer` + `.service` units; `k8s` writes `CronJob`
285    /// manifests. Files are written next to the bundle. Manual triggers and
286    /// cron schedules systemd can't auto-translate are skipped with a note.
287    #[arg(long = "emit-trigger", value_name = "ADAPTER", value_parser = ["systemd", "k8s"])]
288    pub emit_trigger: Option<String>,
289    /// Authorize a build of a project that declares NO `[allow]` ceiling
290    /// (records bundle verdict `ungoverned`). Governed-by-default: without
291    /// this, a missing `[allow]` is a hard error (GOV000). Distinct from
292    /// `--no-governance`, which skips checking a ceiling that DOES exist.
293    #[arg(long, conflicts_with = "no_governance")]
294    pub allow_ungoverned: bool,
295    /// Build a project that HAS an `[allow]` ceiling without enforcing it
296    /// (records bundle verdict `skipped`). Distinct from `--allow-ungoverned`,
297    /// which authorizes having no ceiling at all.
298    #[arg(long)]
299    pub no_governance: bool,
300}
