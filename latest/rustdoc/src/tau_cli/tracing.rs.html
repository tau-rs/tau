1//! Tracing-subscriber configuration for tau-cli.
2//!
3//! The CLI's job is to map clap flags onto a final filter directive.
4//! Subscriber install itself lives in [`tau_observe::install`] so the
5//! CLI and plugin SDK share one code path (sub-project A consolidation).
6//!
7//! Per spec §3.7: stderr-targeted subscriber, default level INFO scoped
8//! to `tau=*`, verbosity flags promote (-v: DEBUG, -vv: TRACE), --quiet
9//! demotes to WARN, --debug behaves as -v plus expanded error chain at
10//! print time, RUST_LOG overrides everything.
11
12use tau_observe::filter::env_or_directive;
13use tau_observe::install::{install as observe_install, Format, InstallOptions, Writer};
14use tracing_subscriber::filter::EnvFilter;
15
16use crate::cli::Cli;
17
18/// Compute the `EnvFilter` from CLI flags + `RUST_LOG` env.
19///
20/// Resolution order:
21/// 1. `RUST_LOG` (if set) — used verbatim, overrides flags.
22/// 2. `--verbose` count >= 2 → `"tau=trace"`.
23/// 3. `--debug` OR `--verbose` count >= 1 → `"tau=debug"`.
24/// 4. `--quiet` → `"tau=warn"`.
25/// 5. Default → `"tau=info"`.
26pub fn build_filter(cli: &Cli) -> EnvFilter {
27    let directive = if cli.verbose >= 2 {
28        "tau=trace"
29    } else if cli.debug || cli.verbose >= 1 {
30        "tau=debug"
31    } else if cli.quiet {
32        "tau=warn"
33    } else {
34        "tau=info"
35    };
36    env_or_directive(directive)
37}

39/// Install the global tracing subscriber for the `tau` CLI.
40///
41/// Delegates to [`tau_observe::install::install`] with the CLI's
42/// human-format, stderr-writer configuration. Idempotent — the
43/// underlying installer short-circuits second calls.
44pub fn install(cli: &Cli) {
45    install_with_extra_layers(cli, Vec::new())
46}
47
48/// Same as [`install`], but accepts caller-provided `extra_layers` that
49/// will be composed into the registry alongside the fmt layer.
50///
51/// Used by `tau workflow run` to attach the
52/// [`WorkflowRunLogLayer`](tau_observe::layers::workflow_run_log::WorkflowRunLogLayer)
53/// so step events emitted via `tracing::event!(target = "tau::workflow::step",
54/// ...)` materialize into the on-disk JSONL run log.
55pub fn install_with_extra_layers(
56    cli: &Cli,
57    extra_layers: Vec<
58        Box<dyn tracing_subscriber::Layer<tracing_subscriber::Registry> + Send + Sync + 'static>,
59    >,
60) {
61    // --log-non-blocking without --log-file is a configuration error:
62    // the non-blocking install path requires a file sink (it has no
63    // meaning over stderr). Exit early so the misconfiguration is
64    // surfaced loudly instead of being silently downgraded.
65    if cli.log_non_blocking && cli.log_file.is_none() {
66        eprintln!("error: --log-non-blocking requires --log-file");
67        std::process::exit(2);
68    }
69
70    let opts = InstallOptions {
71        filter: build_filter(cli),
72        format: Format::Human,
73        writer: Writer::Stderr,
74        non_blocking: cli.log_non_blocking,
75        file_path: cli.log_file.clone(),
76        rotation: cli.log_rotation.into(),
77        extra_layers,
78        // Resolution order for OTLP endpoint:
79        //  1. `--otlp-endpoint` flag (clap also auto-reads
80        //     `OTEL_EXPORTER_OTLP_ENDPOINT` because of `env =` on the
81        //     flag, so the flag arm covers both cases).
82        //  2. `OtlpEndpoint::from_env()` — picks up
83        //     `OTEL_EXPORTER_OTLP_TRACES_ENDPOINT` too, with its own
84        //     header parsing. Reached when no flag was given AND the
85        //     base env var is unset (clap would otherwise have
86        //     populated `cli.otlp_endpoint`).
87        #[cfg(feature = "otlp")]
88        otlp: match cli.otlp_endpoint.clone() {
89            Some(endpoint) => {
90                let headers = std::env::var("OTEL_EXPORTER_OTLP_HEADERS")
91                    .ok()
92                    .map(parse_headers_str)
93                    .unwrap_or_default();
94                Some(tau_observe::otlp::OtlpEndpoint { endpoint, headers })
95            }
96            None => tau_observe::otlp::OtlpEndpoint::from_env(),
97        },
98    };
99    // The CLI does not propagate install errors; the only failure mode
100    // is "already installed", which the underlying installer maps to a
101    // no-op guard.
102    //
103    // The non-blocking guard MUST live for the process lifetime so the
104    // appender's worker thread keeps draining the MPSC channel. The
105    // CLI `main` returns immediately after the subcommand handler, so
106    // we forget the guard rather than `let _ =` it (which would drop
107    // at end-of-fn and flush+stop the worker mid-write).
108    let guard =
109        observe_install(opts).expect("tau_observe::install never returns Err in current impl");
110    std::mem::forget(guard);
111}
112
113/// Parse `OTEL_EXPORTER_OTLP_HEADERS` form (`key=value,key2=value2`) into a
114/// map. Duplicates `tau_observe::otlp::parse_headers` because that helper
115/// is module-private; the spec for the header format is fixed by the OTel
116/// SDK so divergence is unlikely.
117#[cfg(feature = "otlp")]
118fn parse_headers_str(raw: String) -> std::collections::HashMap<String, String> {
119    raw.split(',')
120        .filter_map(|pair| {
121            let mut parts = pair.splitn(2, '=');
122            let key = parts.next()?.trim().to_string();
123            let val = parts.next()?.trim().to_string();
124            if key.is_empty() {
125                None
126            } else {
127                Some((key, val))
128            }
129        })
130        .collect()
131}

162    /// Serialize tests that mutate `RUST_LOG` against each other in the
163    /// same process (cargo test runs unit tests in one binary, so env
164    /// state is shared).
165    fn rust_log_lock() -> std::sync::MutexGuard<'static, ()> {
166        use std::sync::{Mutex, OnceLock};
167        static LOCK: OnceLock<Mutex<()>> = OnceLock::new();
168        LOCK.get_or_init(|| Mutex::new(()))
169            .lock()
170            .unwrap_or_else(|p| p.into_inner())
171    }

174    fn build_filter_default_is_info() {
175        let _g = rust_log_lock();
176        std::env::remove_var("RUST_LOG");
177        let cli = make_cli(0, false, false);
178        let filter = build_filter(&cli);
179        assert!(filter.to_string().contains("tau=info"), "got: {filter}");
