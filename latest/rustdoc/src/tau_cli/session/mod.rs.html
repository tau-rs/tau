1//! Session storage for `tau chat` REPL persistence (ADR-0013 / Tier 3
2//! priority 11).
3//!
4//! Sessions are JSONL files at `<scope>/sessions/<uuid>.jsonl`:
5//! header line first, then `{"type":"message",...}` and
6//! `{"type":"turn_summary",...}` lines appended per turn.
7//!
8//! The `id` module owns UUID v7 generation + prefix resolution.
9//! The `store` module owns file I/O (`SessionWriter`, `SessionReader`,
10//! `list_sessions`). The `render` module owns markdown rendering for
11//! `tau session show`.
12

30/// Errors returned by the session storage layer.
31// Most variants are staged for Tasks 6-10 (resume, list, show, delete, export).
32#[allow(dead_code)]
33#[non_exhaustive]
34#[derive(Debug, thiserror::Error)]
35pub enum SessionError {
36    /// Session id (or prefix) not found in the scope.
37    #[error("session {id_or_prefix:?} not found in {scope_path}")]
38    NotFound {
39        /// What the user typed.
40        id_or_prefix: String,
41        /// The scope's sessions directory.
42        scope_path: PathBuf,
43    },
44
45    /// Multiple sessions match the prefix.
46    #[error("session prefix {prefix:?} is ambiguous: matches {candidates:?}")]
47    AmbiguousPrefix {
48        /// Prefix the user typed.
49        prefix: String,
50        /// All candidate ids (8-char prefixes).
51        candidates: Vec<String>,
52    },
53
54    /// Session header is missing or malformed.
55    #[error("session {id} has invalid header: {detail}")]
56    InvalidHeader {
57        /// Session id (or filename if header missing entirely).
58        id: String,
59        /// Free-form description of the problem.
60        detail: String,
61    },
62
63    /// Schema version is unsupported.
64    #[error("session {id} has unsupported schema {schema}; supported: {supported}")]
65    UnsupportedSchema {
66        /// Session id.
67        id: String,
68        /// What the file declared.
69        schema: u32,
70        /// What this binary supports.
71        supported: u32,
72    },
73
74    /// Resume drift detected (without --force).
75    #[error("session {id} drift: {field} was {expected:?}, now {actual:?}")]
76    AgentDrift {
77        /// Session id.
78        id: String,
79        /// Drifted field name (e.g. `"package.version"`).
80        field: String,
81        /// What the session header recorded.
82        expected: String,
83        /// What the current state resolves to.
84        actual: String,
85    },
86
87    /// Filesystem I/O error.
88    #[error("io error at {path}: {message}")]
89    Io {
90        /// Path involved.
91        path: PathBuf,
92        /// Human-readable error message.
93        message: String,
94    },
95
96    /// JSON parse error.
97    #[error("parse error at {path}:{line}: {message}")]
98    Parse {
99        /// Path involved.
100        path: PathBuf,
101        /// Line number (1-indexed).
102        line: usize,
103        /// Human-readable error message.
104        message: String,
105    },
106}
