1//! `McpClient` — live MCP server handle returned by `open`.
2//!
3//! Carries the `Arc<dyn Transport>` (a `McpStdioServer` in PR-2, a
4//! Streamable HTTP transport in PR-3, possibly cassette-replay in
5//! tests) and the captured `ServerContract` from the handshake.

22/// Options for an `McpClient` (handshake settings + tool-call defaults).
23#[derive(Debug, Clone)]
24pub struct McpClientOptions {
25    /// Handshake settings (timeout, client info).
26    pub handshake: HandshakeOptions,
27    /// Default per-tool-call timeout.
28    pub call_timeout: Duration,
29}
30
31impl Default for McpClientOptions {
32    fn default() -> Self {
33        Self {
34            handshake: HandshakeOptions::default(),
35            call_timeout: Duration::from_secs(60),
36        }
37    }
38}

40/// Live MCP server handle.
41pub struct McpClient {
42    transport: Arc<dyn Transport>,
43    contract: ServerContract,
44    options: McpClientOptions,
45    next_id: Mutex<i64>,
46}
47
48impl McpClient {
49    /// Construct from a live transport + already-completed handshake.
50    pub fn new(
51        transport: Arc<dyn Transport>,
52        contract: ServerContract,
53        options: McpClientOptions,
54    ) -> Self {
55        Self {
56            transport,
57            contract,
58            options,
59            next_id: Mutex::new(2),
60        }
61    }
62
63    /// The captured server contract (initialize + tools/list snapshot).
64    pub fn contract(&self) -> &ServerContract {
65        &self.contract
66    }
67
68    /// Call a server-side tool by name with the given JSON args.
69    pub async fn call_tool(
70        &self,
71        server_tool_name: &str,
72        args: serde_json::Value,
73    ) -> Result<ToolsCallResponse, McpError> {
74        let id = {
75            let mut next = self.next_id.lock().await;
76            let id = *next;
77            *next += 1;
78            id
79        };
80        let req = JsonRpcMessage::Request(JsonRpcRequest {
81            jsonrpc: JSONRPC_VERSION.to_string(),
82            id: RequestId::Number(id),
83            method: "tools/call".to_string(),
84            params: Some(serde_json::to_value(ToolsCallRequest {
85                name: server_tool_name.to_string(),
86                arguments: Some(args),
87            })?),
88        });
89        let call_timeout = self.options.call_timeout;
90        let inner = async {
91            self.transport.send_message(&req).await?;
92            let resp = recv_response_for(&*self.transport, &RequestId::Number(id)).await?;
93            let result: ToolsCallResponse = serde_json::from_value(resp)?;
94            Ok::<_, McpError>(result)
95        };
96        timeout(call_timeout, inner).await.map_err(|_| {
97            McpError::Transport(format!(
98                "tools/call {server_tool_name:?} timed out after {}ms",
99                call_timeout.as_millis()
100            ))
101        })?
102    }
103
104    /// Borrow the live transport (PR-5 wires the inbound-dispatch task here).
105    pub fn transport(&self) -> &Arc<dyn Transport> {
106        &self.transport
107    }
108
109    /// Spawn the inbound-dispatch task for this client.
110    ///
111    /// Routes server-initiated `sampling/createMessage` + `roots/list`
112    /// requests through `handlers`. Returns a handle whose `shutdown()`
113    /// or `Drop` aborts the pump.
114    ///
115    /// Call AT MOST ONCE per McpClient — the pump owns the inbound side
116    /// of the transport. Calling twice would race with `call_tool`'s
117    /// response loop. In v0 the pattern is: spawn the pump for clients
118    /// that need server-initiated routing; do nothing for outbound-only
119    /// clients (e.g. the live resolver from PR-4).
120    pub fn start_inbound_dispatch(
121        &self,
122        handlers: std::sync::Arc<dyn tau_mcp::host::handlers::HostHandlers>,
123    ) -> crate::host_lifecycle::InboundDispatchHandle {
124        crate::host_lifecycle::spawn_inbound_dispatch(self.transport.clone(), handlers)
125    }
126}
127
128async fn recv_response_for(
129    transport: &dyn Transport,
130    expected_id: &RequestId,
131) -> Result<serde_json::Value, McpError> {
132    loop {
133        let msg = transport
134            .next_message()
135            .await?
136            .ok_or_else(|| McpError::Transport("peer closed mid-call".into()))?;
137        match msg {
138            JsonRpcMessage::Response(JsonRpcResponse {
139                id, result, error, ..
140            }) if &id == expected_id => {
141                if let Some(e) = error {
142                    return Err(McpError::Protocol(format!(
143                        "server returned error code={} msg={}",
144                        e.code, e.message
145                    )));
146                }
147                return Ok(result.unwrap_or(serde_json::Value::Null));
148            }
149            _ => continue,
150        }
151    }
152}
