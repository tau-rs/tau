1//! MCP handshake driver.
2//!
3//! Sends `initialize` + `tools/list` over a Transport, captures the
4//! responses, and builds a `ServerContract` from them.
5
6use std::time::Duration;
7
8use tau_mcp::contract::ServerContract;
9use tau_mcp::protocol::{
10    initialize::{ClientInfo, InitializeRequest, InitializeResponse},
11    jsonrpc::{JsonRpcMessage, JsonRpcRequest, JsonRpcResponse, RequestId, JSONRPC_VERSION},
12    tools::{ToolsListRequest, ToolsListResponse},
13};
14use tau_mcp::transport::Transport;
15use tokio::time::timeout;
16use tracing::{debug, instrument};
17
18use crate::host_lifecycle::error::HandshakeError;
19
20/// MCP protocol version tau speaks.
21pub const MCP_PROTOCOL_VERSION: &str = "2025-03-26";
22
23/// Handshake options (timeout, client info).
24#[derive(Debug, Clone)]
25pub struct HandshakeOptions {
26    /// Timeout for the entire handshake (initialize + tools/list).
27    pub handshake_timeout: Duration,
28    /// Client name reported to the server.
29    pub client_name: String,
30    /// Client version reported to the server.
31    pub client_version: String,
32}
33
34impl Default for HandshakeOptions {
35    fn default() -> Self {
36        Self {
37            handshake_timeout: Duration::from_secs(30),
38            client_name: "tau".to_string(),
39            client_version: env!("CARGO_PKG_VERSION").to_string(),
40        }
41    }
42}
43
44/// Drive the MCP handshake. Returns a `ServerContract` capturing the
45/// server's reported info + tools/list snapshot.
46#[instrument(name = "mcp_handshake", skip(transport, options), fields(
47    client_name = %options.client_name,
48    handshake_timeout_ms = options.handshake_timeout.as_millis() as u64,
49))]
50pub async fn drive_handshake(
51    transport: &dyn Transport,
52    options: &HandshakeOptions,
53) -> Result<ServerContract, HandshakeError> {
54    let inner = async {
55        // 1. initialize
56        let init_req = JsonRpcMessage::Request(JsonRpcRequest {
57            jsonrpc: JSONRPC_VERSION.to_string(),
58            id: RequestId::Number(0),
59            method: "initialize".to_string(),
60            params: Some(
61                serde_json::to_value(InitializeRequest {
62                    protocol_version: MCP_PROTOCOL_VERSION.to_string(),
63                    client_info: ClientInfo {
64                        name: options.client_name.clone(),
65                        version: options.client_version.clone(),
66                        additional: Default::default(),
67                    },
68                    capabilities: None,
69                })
70                .map_err(|e| HandshakeError::Malformed(format!("encode initialize: {e}")))?,
71            ),
72        });
73        send(transport, &init_req).await?;
74        let init_resp = recv_response_for(transport, &RequestId::Number(0)).await?;
75        let init_result: InitializeResponse = serde_json::from_value(init_resp)
76            .map_err(|e| HandshakeError::Malformed(format!("decode initialize response: {e}")))?;
77        debug!(
78            server_name = %init_result.server_info.name,
79            server_version = %init_result.server_info.version,
80            "initialize response decoded"
81        );
82
83        // 2. tools/list
84        let list_req = JsonRpcMessage::Request(JsonRpcRequest {
85            jsonrpc: JSONRPC_VERSION.to_string(),
86            id: RequestId::Number(1),
87            method: "tools/list".to_string(),
88            params: Some(serde_json::to_value(ToolsListRequest::default()).unwrap_or_default()),
89        });
90        send(transport, &list_req).await?;
91        let list_resp = recv_response_for(transport, &RequestId::Number(1)).await?;
92        let list_result: ToolsListResponse = serde_json::from_value(list_resp)
93            .map_err(|e| HandshakeError::Malformed(format!("decode tools/list response: {e}")))?;
94        debug!(
95            tools_count = list_result.tools.len(),
96            "tools/list response decoded"
97        );
98
99        let contract = ServerContract::from_handshake(init_result, list_result, |_| Vec::new());
100        Ok::<_, HandshakeError>(contract)
101    };
102
103    timeout(options.handshake_timeout, inner)
104        .await
105        .map_err(|_| HandshakeError::Timeout {
106            millis: options.handshake_timeout.as_millis() as u64,
107        })?
108}
109
110async fn send(transport: &dyn Transport, msg: &JsonRpcMessage) -> Result<(), HandshakeError> {
111    transport
112        .send_message(msg)
113        .await
114        .map_err(|e| HandshakeError::Transport(format!("{e}")))
115}
116
117/// Receive messages until we see a response matching `expected_id`.
118/// Ignores notifications and out-of-order responses (logs but skips).
119async fn recv_response_for(
120    transport: &dyn Transport,
121    expected_id: &RequestId,
122) -> Result<serde_json::Value, HandshakeError> {
123    loop {
124        let msg = transport
125            .next_message()
126            .await
127            .map_err(|e| HandshakeError::Transport(format!("{e}")))?
128            .ok_or_else(|| HandshakeError::Transport("peer closed mid-handshake".into()))?;
129        match msg {
130            JsonRpcMessage::Response(JsonRpcResponse {
131                id, result, error, ..
132            }) if &id == expected_id => {
133                if let Some(e) = error {
134                    return Err(HandshakeError::ServerError {
135                        code: e.code,
136                        message: e.message,
137                    });
138                }
139                return Ok(result.unwrap_or(serde_json::Value::Null));
140            }
141            other => {
142                debug!(
143                    received_kind = ?std::mem::discriminant(&other),
144                    expected_id = ?expected_id,
145                    "skipping unexpected handshake message"
146                );
147            }
148        }
149    }
150}
