1//! `dial(url, options) → Arc<McpHttpServer>` — HTTP transport dial entrypoint.
2//!
3//! Composes URL host extraction → reqwest::Client build → HttpClientGuard
4//! → McpHttpServer construction. Called by host_lifecycle::open() for
5//! `http:` and `https:` URLs.
6
7use std::sync::Arc;
8use std::time::Duration;
9
10use tracing::{info, instrument};
11use url::Url;
12
13use crate::transport_http::error::HttpSpawnError;
14use crate::transport_http::guard::HttpClientGuard;
15use crate::transport_http::server::McpHttpServer;
16
17/// Options for HTTP dial.
18#[derive(Debug, Clone)]
19pub struct HttpDialOptions {
20    /// Per-request timeout for the reqwest client.
21    pub request_timeout: Duration,
22    /// User-Agent string sent with every request.
23    pub user_agent: String,
24}
25
26impl Default for HttpDialOptions {
27    fn default() -> Self {
28        Self {
29            request_timeout: Duration::from_secs(60),
30            user_agent: concat!("tau-mcp-tokio/", env!("CARGO_PKG_VERSION")).to_string(),
31        }
32    }
33}
34
35/// Dial an HTTP MCP server. Returns a ready `Arc<McpHttpServer>` that
36/// `host_lifecycle::open` then drives through the MCP handshake.
37#[instrument(name = "mcp_http_dial", skip(options), fields(url = %url))]
38pub fn dial(url: Url, options: HttpDialOptions) -> Result<Arc<McpHttpServer>, HttpSpawnError> {
39    let pinned_host = url.host().ok_or_else(|| HttpSpawnError::NoHost {
40        url: url.to_string(),
41    })?;
42    let pinned_host = match pinned_host {
43        url::Host::Domain(d) => url::Host::Domain(d.to_string()),
44        url::Host::Ipv4(a) => url::Host::Ipv4(a),
45        url::Host::Ipv6(a) => url::Host::Ipv6(a),
46    };
47    let client = reqwest::Client::builder()
48        .redirect(reqwest::redirect::Policy::none())
49        .timeout(options.request_timeout)
50        .user_agent(options.user_agent)
51        .build()
52        .map_err(|e| HttpSpawnError::ClientBuild(format!("{e}")))?;
53    let guard = HttpClientGuard::new(client, pinned_host);
54    info!(host = %guard.pinned_host(), "constructed pinned HTTP client");
55    Ok(McpHttpServer::new(guard, url))
56}
