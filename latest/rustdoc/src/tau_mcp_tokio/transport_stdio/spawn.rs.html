1//! Sandboxed subprocess spawn for stdio MCP servers.
2//!
3//! Wraps a `tokio::process::Command` via
4//! `tau_ports::DynProcessGate::wrap_spawn` exactly the same way
5//! `plugin_host::process::spawn` does — the
6//! `CapabilityPlan` is honored at the OS boundary
7//! (landlock/seccomp/sandbox-exec/podman per the four sandbox adapters).
8//!
9//! After `wrap_spawn` succeeds, the command is spawned under tokio.
10//! Stdin / stdout / stderr handles are piped so the caller can wire
11//! them into a `JsonLineFramer`.
12
13use std::process::Stdio;
14use std::sync::Arc;
15
16use tau_ports::CapabilityPlan;
17use tau_ports::DynProcessGate;
18use tokio::process::{Child, Command};
19
20use crate::transport_stdio::error::StdioSpawnError;
21
22/// Spawn an MCP server subprocess under the given capability gate.
23///
24/// The command's stdin/stdout/stderr are piped; the caller wires
25/// `child.stdin.take()` and `child.stdout.take()` into a
26/// `JsonLineFramer`.
27///
28/// # Errors
29///
30/// - [`StdioSpawnError::SandboxRefused`] — the capability gate refused
31///   the plan (e.g. the plan demands a sandbox shape the gate adapter
32///   doesn't support on this target).
33/// - [`StdioSpawnError::TokioSpawn`] — `tokio::process::Command::spawn`
34///   failed (binary missing, permission denied, etc.).
35pub async fn spawn(
36    mut cmd: Command,
37    gate: Arc<dyn DynProcessGate>,
38    plan: &CapabilityPlan,
39) -> Result<Child, StdioSpawnError> {
40    let _handle = gate
41        .wrap_spawn(plan, cmd.as_std_mut())
42        .await
43        .map_err(StdioSpawnError::SandboxRefused)?;
44
45    // Apply piped stdio + kill_on_drop AFTER `wrap_spawn`. The command-rebuilding
46    // sandbox adapters (darwin sandbox-exec, windows launcher, container docker
47    // run) do `*cmd = Command::new(...)` and re-emit only program/args/envs/cwd,
48    // so anything set before `wrap_spawn` is discarded — and `std::process::Command`
49    // exposes no stdio getters, so an adapter cannot re-apply it. Setting it here
50    // is safe for every adapter (native/mock/passthrough don't rebuild).
51    cmd.stdin(Stdio::piped())
52        .stdout(Stdio::piped())
53        .stderr(Stdio::piped())
54        .kill_on_drop(true);
55
56    let child = cmd.spawn().map_err(StdioSpawnError::from)?;
57    Ok(child)
58}

60#[cfg(test)]
61mod tests {
62    use super::*;
63
64    use std::process::Command as StdCommand;
65
66    use tau_ports::{
67        CapabilityError, CapabilityGate, CapabilityHandle, CapabilityProbe, CapabilityShapeSet,
68        CapabilityTier, ProcessCapabilityGate,
69    };
70
71    /// Test double that reproduces the `*cmd = Command::new(...)` command
72    /// rebuild performed by the real command-rebuilding sandbox adapters
73    /// (`tau-sandbox-darwin` / `-windows` / `-container`). Those adapters
74    /// re-emit only program + args + envs + cwd on the fresh `Command`, so
75    /// any stdio / `kill_on_drop` config the caller set *before* `wrap_spawn`
76    /// is discarded. `std::process::Command` exposes no stdio getters, so an
77    /// adapter physically cannot re-apply it — the caller must set stdio
78    /// *after* `wrap_spawn`. This gate is the platform-agnostic stand-in for
79    /// that behavior (a real adapter needs sandbox-exec / AppContainer /
80    /// docker and only rebuilds on its own OS).
81    struct RebuildingGate;
82
83    impl CapabilityGate for RebuildingGate {
84        fn name(&self) -> &str {
85            "rebuilding-test-gate"
86        }
87
88        async fn probe(&self) -> CapabilityProbe {
89            CapabilityProbe::Available {
90                tier: CapabilityTier::None,
91                details: "test: rebuilds command like sandbox-exec/launcher".into(),
92            }
93        }
94
95        fn supported_shapes(&self) -> CapabilityShapeSet {
96            CapabilityShapeSet::new()
97        }
98
99        fn validate_plan(&self, _plan: &CapabilityPlan) -> Result<(), CapabilityError> {
100            Ok(())
101        }
102    }
103
104    impl ProcessCapabilityGate for RebuildingGate {
105        async fn wrap_spawn(
106            &self,
107            _plan: &CapabilityPlan,
108            cmd: &mut StdCommand,
109        ) -> Result<CapabilityHandle, CapabilityError> {
110            // Re-emit program + args + envs on a fresh Command, exactly as the
111            // real rebuilding adapters do — deliberately NOT stdio/kill_on_drop.
112            let program = cmd.get_program().to_os_string();
113            let args: Vec<std::ffi::OsString> = cmd.get_args().map(|a| a.to_os_string()).collect();
114            let envs: Vec<(std::ffi::OsString, Option<std::ffi::OsString>)> = cmd
115                .get_envs()
116                .map(|(k, v)| (k.to_os_string(), v.map(|v| v.to_os_string())))
117                .collect();
118            *cmd = StdCommand::new(program);
119            cmd.args(args);
120            for (k, v) in envs {
121                match v {
122                    Some(val) => {
123                        cmd.env(k, val);
124                    }
125                    None => {
126                        cmd.env_remove(k);
127                    }
128                }
129            }
130            Ok(CapabilityHandle::noop())
131        }
132    }

134    /// A long-lived, cross-platform child that blocks on stdin so the piped
135    /// handles stay open long enough to inspect.
136    #[cfg(unix)]
137    const SPAWN_PROGRAM: &str = "/bin/cat";
138    #[cfg(windows)]
139    const SPAWN_PROGRAM: &str = "cmd";
140
141    /// Regression guard for the launcher/sandbox-exec stdio-loss bug: when the
142    /// gate rebuilds the command (as the darwin/windows/container adapters do),
143    /// `spawn()` must still hand back piped stdin/stdout/stderr because it
144    /// applies the stdio config *after* `wrap_spawn`. Before the fix, stdio was
145    /// set before `wrap_spawn` and the rebuild discarded it, leaving the child's
146    /// stdin/stdout/stderr as `None` (and panicking the plugin-host caller).
147    #[tokio::test]
148    async fn spawn_pipes_stdio_even_when_gate_rebuilds_command() {
149        let gate: Arc<dyn DynProcessGate> = Arc::new(RebuildingGate);
150        let plan = CapabilityPlan::new(Vec::new(), None, None);
151        let cmd = Command::new(SPAWN_PROGRAM);
152
153        let mut child = spawn(cmd, gate, &plan).await.expect("spawn should succeed");
154
155        assert!(
156            child.stdin.is_some(),
157            "stdin must be piped after a command-rebuilding wrap_spawn"
158        );
159        assert!(
160            child.stdout.is_some(),
161            "stdout must be piped after a command-rebuilding wrap_spawn"
162        );
163        assert!(
164            child.stderr.is_some(),
165            "stderr must be piped after a command-rebuilding wrap_spawn"
166        );
167
168        // Don't rely on kill_on_drop timing in the test harness; reap eagerly.
169        let _ = child.kill().await;
170    }
