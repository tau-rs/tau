1//! Sandboxed subprocess spawn for stdio MCP servers.
2//!
3//! Wraps a `tokio::process::Command` via
4//! `tau_ports::DynProcessGate::wrap_spawn` exactly the same way
5//! `plugin_host::process::spawn` does — the
6//! `CapabilityPlan` is honored at the OS boundary
7//! (landlock/seccomp/sandbox-exec/podman per the four sandbox adapters).
8//!
9//! After `wrap_spawn` succeeds, the command is spawned under tokio.
10//! Stdin / stdout / stderr handles are piped so the caller can wire
11//! them into a `JsonLineFramer`.
12
13use std::process::Stdio;
14use std::sync::Arc;
15
16use tau_ports::CapabilityPlan;
17use tau_ports::DynProcessGate;
18use tokio::process::{Child, Command};
19
20use crate::transport_stdio::error::StdioSpawnError;
21
22/// Spawn an MCP server subprocess under the given capability gate.
23///
24/// The command's stdin/stdout/stderr are piped; the caller wires
25/// `child.stdin.take()` and `child.stdout.take()` into a
26/// `JsonLineFramer`.
27///
28/// # Errors
29///
30/// - [`StdioSpawnError::SandboxRefused`] — the capability gate refused
31///   the plan (e.g. the plan demands a sandbox shape the gate adapter
32///   doesn't support on this target).
33/// - [`StdioSpawnError::TokioSpawn`] — `tokio::process::Command::spawn`
34///   failed (binary missing, permission denied, etc.).
35pub async fn spawn(
36    mut cmd: Command,
37    gate: Arc<dyn DynProcessGate>,
38    plan: &CapabilityPlan,
39) -> Result<Child, StdioSpawnError> {
40    cmd.stdin(Stdio::piped())
41        .stdout(Stdio::piped())
42        .stderr(Stdio::piped())
43        .kill_on_drop(true);
44
45    let _handle = gate
46        .wrap_spawn(plan, cmd.as_std_mut())
47        .await
48        .map_err(StdioSpawnError::SandboxRefused)?;
49
50    let child = cmd.spawn().map_err(StdioSpawnError::from)?;
51    Ok(child)
52}
