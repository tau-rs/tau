1//! Error types for the stdio transport.
2
3use tau_ports::CapabilityError;
4use thiserror::Error;
5
6/// Failure during stdio MCP server spawn.
7#[derive(Debug, Error)]
8#[non_exhaustive]
9pub enum StdioSpawnError {
10    /// `ProcessCapabilityGate::validate_plan` (which `wrap_spawn` calls
11    /// internally) refused the plan.
12    #[error("capability gate refused plan: {0}")]
13    SandboxRefused(#[from] CapabilityError),
14    /// `tokio::process::Command::spawn` failed (binary missing,
15    /// permission denied, etc.).
16    #[error("tokio spawn failed: {0}")]
17    TokioSpawn(String),
18}
19
20impl From<std::io::Error> for StdioSpawnError {
21    fn from(e: std::io::Error) -> Self {
22        StdioSpawnError::TokioSpawn(format!("{e}"))
23    }
24}
25
26/// Failure during a stdio transport read/write.
27#[derive(Debug, Error)]
28#[non_exhaustive]
29pub enum StdioTransportError {
30    /// I/O error on child stdin/stdout.
31    #[error("I/O on stdio transport: {0}")]
32    Io(String),
33    /// One of the framed JSON-RPC lines was not valid JSON.
34    #[error("malformed JSON on stdio transport: {0}")]
35    Json(String),
36    /// The child process exited mid-conversation.
37    #[error("child process exited (status: {status})")]
38    ChildExited {
39        /// Child exit status as a string (cross-platform).
40        status: String,
41    },
42}
43
44impl From<std::io::Error> for StdioTransportError {
45    fn from(e: std::io::Error) -> Self {
46        StdioTransportError::Io(format!("{e}"))
47    }
48}
49
50impl From<serde_json::Error> for StdioTransportError {
51    fn from(e: serde_json::Error) -> Self {
52        StdioTransportError::Json(format!("{e}"))
53    }
54}
