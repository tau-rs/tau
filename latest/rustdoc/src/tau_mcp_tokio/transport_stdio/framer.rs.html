1//! Line-delimited JSON-RPC framer over async I/O.
2//!
3//! Per MCP stdio transport spec: every JSON-RPC message is one JSON
4//! object terminated by `\n`. The framer reads lines from an
5//! `AsyncBufRead` and writes them to an `AsyncWrite`, deserializing
6//! / serializing through `JsonRpcMessage`.
7
8use tau_mcp::protocol::JsonRpcMessage;
9use tokio::io::{AsyncBufReadExt, AsyncWriteExt, BufReader};
10use tokio::io::{AsyncRead, AsyncWrite};
11
12use crate::transport_stdio::error::StdioTransportError;
13
14/// Line-delimited JSON-RPC framer.
15pub struct JsonLineFramer<R, W>
16where
17    R: AsyncRead + Unpin + Send,
18    W: AsyncWrite + Unpin + Send,
19{
20    reader: BufReader<R>,
21    writer: W,
22    line_buf: String,
23}
24
25impl<R, W> JsonLineFramer<R, W>
26where
27    R: AsyncRead + Unpin + Send,
28    W: AsyncWrite + Unpin + Send,
29{
30    /// Construct a framer over the given reader+writer.
31    pub fn new(reader: R, writer: W) -> Self {
32        Self {
33            reader: BufReader::new(reader),
34            writer,
35            line_buf: String::new(),
36        }
37    }
38
39    /// Read one MCP message. Returns `Ok(None)` on EOF (clean close).
40    pub async fn read_message(&mut self) -> Result<Option<JsonRpcMessage>, StdioTransportError> {
41        self.line_buf.clear();
42        let n = self.reader.read_line(&mut self.line_buf).await?;
43        if n == 0 {
44            return Ok(None);
45        }
46        let trimmed = self.line_buf.trim_end_matches(['\r', '\n']);
47        if trimmed.is_empty() {
48            // Skip blank lines as a robustness measure.
49            return self.read_message_boxed().await;
50        }
51        let msg: JsonRpcMessage = serde_json::from_str(trimmed)?;
52        Ok(Some(msg))
53    }
54
55    fn read_message_boxed<'a>(
56        &'a mut self,
57    ) -> std::pin::Pin<
58        Box<
59            dyn std::future::Future<Output = Result<Option<JsonRpcMessage>, StdioTransportError>>
60                + Send
61                + 'a,
62        >,
63    > {
64        Box::pin(self.read_message())
65    }
66
67    /// Write one MCP message followed by `\n`.
68    pub async fn write_message(&mut self, msg: &JsonRpcMessage) -> Result<(), StdioTransportError> {
69        let bytes = serde_json::to_vec(msg)?;
70        self.writer.write_all(&bytes).await?;
71        self.writer.write_all(b"\n").await?;
72        self.writer.flush().await?;
73        Ok(())
74    }
75}

77#[cfg(test)]
78mod tests {
79    use super::*;
80    use tau_mcp::protocol::jsonrpc::{
81        JsonRpcMessage, JsonRpcRequest, JsonRpcResponse, RequestId, JSONRPC_VERSION,
82    };
83    use tokio::io::duplex;
84
85    #[tokio::test(flavor = "current_thread")]
86    async fn write_then_read_round_trips_request() {
87        let (peer_r, mut peer_w) = duplex(4096);
88        let (mut my_r, _my_w) = duplex(4096);
89
90        // Use peer's write as MY read.
91        let msg = JsonRpcMessage::Request(JsonRpcRequest {
92            jsonrpc: JSONRPC_VERSION.to_string(),
93            id: RequestId::Number(7),
94            method: "tools/call".to_string(),
95            params: Some(serde_json::json!({"name":"echo"})),
96        });
97
98        let bytes = serde_json::to_vec(&msg).unwrap();
99        tokio::io::AsyncWriteExt::write_all(&mut peer_w, &bytes)
100            .await
101            .unwrap();
102        tokio::io::AsyncWriteExt::write_all(&mut peer_w, b"\n")
103            .await
104            .unwrap();
105        drop(peer_w);
106
107        let mut framer = JsonLineFramer::new(peer_r, &mut my_r);
108        let received = framer.read_message().await.unwrap().expect("got a message");
109        assert_eq!(received, msg);
110    }

112    #[tokio::test(flavor = "current_thread")]
113    async fn eof_returns_none() {
114        let (peer_r, peer_w) = duplex(4096);
115        let (mut my_r, _my_w) = duplex(4096);
116        drop(peer_w); // EOF immediately
117
118        let mut framer = JsonLineFramer::new(peer_r, &mut my_r);
119        let received = framer.read_message().await.unwrap();
120        assert!(
121            received.is_none(),
122            "EOF should yield Ok(None), got {received:?}"

140    #[tokio::test(flavor = "current_thread")]
141    async fn skip_blank_lines_then_read_message() {
142        let (peer_r, mut peer_w) = duplex(4096);
143        let (mut my_r, _my_w) = duplex(4096);
144
145        let msg = JsonRpcMessage::Response(JsonRpcResponse {
146            jsonrpc: JSONRPC_VERSION.to_string(),
147            id: RequestId::Number(1),
148            result: Some(serde_json::json!({"ok":true})),
149            error: None,
150        });
151        let bytes = serde_json::to_vec(&msg).unwrap();
152        tokio::io::AsyncWriteExt::write_all(&mut peer_w, b"\n\n")
153            .await
154            .unwrap();
155        tokio::io::AsyncWriteExt::write_all(&mut peer_w, &bytes)
156            .await
157            .unwrap();
158        tokio::io::AsyncWriteExt::write_all(&mut peer_w, b"\n")
159            .await
160            .unwrap();
161        drop(peer_w);
162
163        let mut framer = JsonLineFramer::new(peer_r, &mut my_r);
164        let received = framer.read_message().await.unwrap().expect("got a message");
165        assert_eq!(received, msg);

168    #[tokio::test(flavor = "current_thread")]
169    async fn write_message_writes_one_line_with_trailing_newline() {
170        let (mut peer_r, peer_w) = duplex(4096);
171        let (_my_r, my_w) = duplex(4096);
172        let mut framer = JsonLineFramer::new(_my_r, peer_w);
173
174        let msg = JsonRpcMessage::Request(JsonRpcRequest {
175            jsonrpc: JSONRPC_VERSION.to_string(),
176            id: RequestId::Number(0),
177            method: "initialize".to_string(),
178            params: None,
179        });
180        framer.write_message(&msg).await.unwrap();
181        // Drop the framer to close peer_w so read_to_end below sees EOF.
182        drop(framer);
183        drop(my_w);
184
185        let mut buf = Vec::new();
186        tokio::io::AsyncReadExt::read_to_end(&mut peer_r, &mut buf)
187            .await
188            .unwrap();
189        let s = std::str::from_utf8(&buf).unwrap();
190        assert!(s.ends_with('\n'));
191        assert_eq!(s.matches('\n').count(), 1);
192    }
