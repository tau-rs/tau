5//! `tau-sandbox-proxy` — userspace HTTP-CONNECT proxy for tau sandboxed plugins.
6//!
7//! Shared by both the native (Linux landlock/seccomp) and container
8//! (docker/podman) sandbox adapters. Extracted from `tau-sandbox-native`
9//! because the proxy logic is purely tokio-based and cross-platform, while
10//! `tau-sandbox-native` itself is Linux-specific.
11//!
12//! Architecture: a tokio task in tau's parent address space accepts
13//! Unix-socket connections from the per-plugin `tau-net-bridge` binary.
14//! Each connection arrives carrying either:
15//!   - An HTTP `CONNECT host:port` request (HTTPS tunnel), or
16//!   - A plain HTTP request (`GET http://host/path HTTP/1.1`).
17//!
18//! The proxy validates the host against the plan's allow-list, then
19//! handles accordingly:
20//!   - CONNECT: peeks the TLS ClientHello to verify SNI matches, splices.
21//!   - Plain HTTP: rewrites the request line to origin-form, splices.
22//!
23//! Pass-through mode only for HTTPS — proxy does NOT terminate TLS.
24//! Plugin's TLS handshake goes end-to-end with the real remote server.

34/// Opt-in stderr tracing for the proxy's data path.
35///
36/// `tracing` events are the primary instrument, but the Windows
37/// AppContainer egress chain (`tau-sandbox-windows`) runs with **no
38/// subscriber installed** — the events are invisible exactly when the
39/// data path breaks, which cost #622 a full 20-minute CI round. Setting
40/// `TAU_SANDBOX_PROXY_TRACE` to a non-empty, non-`0` value additionally
41/// prints each decision point to stderr with a `PROXY ` prefix, matching
42/// the `BRIDGE ` / `PIPEPROXY ` markers on the other two hops so a stall
43/// can be localised to one hop from one log.
44///
45/// Read once and cached: off by default, so nothing about the Unix /
46/// container proxy path changes unless a caller opts in.
47fn trace_enabled() -> bool {
48    static ON: std::sync::OnceLock<bool> = std::sync::OnceLock::new();
49    *ON.get_or_init(|| {
50        std::env::var("TAU_SANDBOX_PROXY_TRACE")
51            .map(|v| !v.is_empty() && v != "0")
52            .unwrap_or(false)
53    })
54}
55
56macro_rules! ptrace {
57    ($($arg:tt)*) => {
58        if $crate::trace_enabled() {
59            eprintln!("PROXY {}", format_args!($($arg)*));
60        }
61    };
62}

64/// Escape a request-head slice for a single-line stderr marker.
65fn head_preview(buf: &[u8]) -> String {
66    let cut = buf.len().min(120);
67    String::from_utf8_lossy(&buf[..cut])
68        .escape_debug()
69        .to_string()
70}

72/// Host egress policy for the proxy. `Any` = pass-all (reachable only from a
73/// `HostSet::Any` capability); `Exact` = only these (pre-validated, lowercase)
74/// hosts. Case-insensitive matching at runtime.
75#[derive(Debug, Clone, PartialEq, Eq)]
76pub enum HostAllow {
77    /// Allow every host (pass-all).
78    Any,
79    /// Allow exactly these hosts (case-insensitive).
80    Exact(Vec<String>),
81}
82
83impl HostAllow {
84    /// True iff `host` is permitted. Case-folds both sides.
85    pub fn permits(&self, host: &str) -> bool {
86        match self {
87            HostAllow::Any => true,
88            HostAllow::Exact(list) => list.iter().any(|h| h.eq_ignore_ascii_case(host)),
89        }
90    }
91}

135/// Handle to a running proxy task. Drop aborts the task and unlinks the
136/// temp Unix socket file.
137#[cfg(unix)]
138#[non_exhaustive]
139pub struct ProxyHandle {
140    sock_path: PathBuf,
141    sock_dir: PathBuf,
142    task: JoinHandle<()>,
143}
144
145#[cfg(unix)]
146impl ProxyHandle {
147    /// Returns the path to the Unix socket the proxy is listening on.
148    pub fn sock_path(&self) -> &Path {
149        &self.sock_path
150    }
151}
152
153#[cfg(unix)]
154impl Drop for ProxyHandle {
155    fn drop(&mut self) {
156        self.task.abort();
157        let _ = std::fs::remove_file(&self.sock_path);
158        // Remove the per-run directory too (best-effort; only succeeds once
159        // the socket file inside it is gone).
160        let _ = std::fs::remove_dir(&self.sock_dir);
161    }
162}

164/// Spawn a tokio task that listens for HTTP CONNECT requests on a
165/// temp Unix socket file. Returns a `ProxyHandle` whose Drop cleans up.
166///
167/// Caller is responsible for granting the child access to the returned
168/// socket path (e.g. via landlock rules for native, bind-mount for container)
169/// so the bridge inside the sandbox can dial it.
170#[cfg(unix)]
171pub fn spawn_proxy(hosts: HostAllow) -> std::io::Result<ProxyHandle> {
172    let (sock_dir, sock_path) = make_run_dir_and_sock_path()?;
173    let listener = UnixListener::bind(&sock_path)?;
174    // Make the socket inode other-writable (`0o666`). `connect(2)` to a Unix
175    // socket requires *write* permission on the inode, and the two legitimate
176    // callers reach it as neither owner nor a CAP_DAC_OVERRIDE-capable process:
177    // the container bridge bind-mounts the inode and, under rootful Docker,
178    // runs as uid 0 with `--cap-drop=ALL` (no DAC_OVERRIDE) while the socket is
179    // owned by the host user that spawned the proxy — so it is "other" and only
180    // the other-write bit lets it connect. (Rootless Podman maps container-root
181    // to the socket's owner, so owner-write sufficed there — which is why the
182    // OS-default `0o755` mode silently regressed only Docker CI.) The `0o700`
183    // per-run dir (see `make_run_dir_and_sock_path`) — not the socket mode —
184    // remains the access boundary against other *local* users, who cannot
185    // traverse into the dir to reach the socket regardless of its mode. This
186    // preserves the S6 hardening while restoring the container path.
187    {
188        use std::os::unix::fs::PermissionsExt;
189        std::fs::set_permissions(&sock_path, std::fs::Permissions::from_mode(0o666))?;
190    }
191    let task = tokio::spawn(accept_loop(listener, hosts));
192    Ok(ProxyHandle {
193        sock_path,
194        sock_dir,
195        task,
196    })
197}

199/// Create a private per-run directory (mode `0o700`) in the system temp dir
200/// and return `(dir, socket_path)` for the proxy's Unix socket.
201///
202/// The `0o700` directory — not the socket file mode — is the access boundary,
203/// mirroring the ssh-agent / gpg-agent socket-in-a-private-dir pattern. This
204/// replaces the former world-writable (`0o666`) socket sitting directly in
205/// shared `/tmp`, which let any local user dial the proxy and relay egress to
206/// allowlisted hosts for the lifetime of a run (audit S6).
207///
208/// The socket file itself is set other-writable (`0o666`) by [`spawn_proxy`]
209/// after bind — see the rationale there: a container-root bridge without
210/// CAP_DAC_OVERRIDE must be able to `connect(2)` to the bind-mounted inode. No
211/// other local user can traverse the `0o700` directory to reach the socket, so
212/// the socket mode is not the boundary; the native bridge runs as the same host
213/// user that owns the directory (so DAC traversal is permitted; landlock grants
214/// the socket path itself).
215#[cfg(unix)]
216fn make_run_dir_and_sock_path() -> std::io::Result<(PathBuf, PathBuf)> {
217    use std::os::unix::fs::DirBuilderExt;
218    use std::sync::atomic::{AtomicU64, Ordering};
219    static COUNTER: AtomicU64 = AtomicU64::new(0);
220    let n = COUNTER.fetch_add(1, Ordering::Relaxed);
221    let mut dir = std::env::temp_dir();
222    dir.push(format!("tau-proxy-{}-{}", std::process::id(), n));
223    // Clear any stale directory from a prior aborted run, then create fresh
224    // at 0o700 so only the owning user can traverse into it.
225    let _ = std::fs::remove_dir_all(&dir);
226    std::fs::DirBuilder::new().mode(0o700).create(&dir)?;
227    let sock_path = dir.join("proxy.sock");
228    Ok((dir, sock_path))
229}

231#[cfg(unix)]
232async fn accept_loop(listener: UnixListener, hosts: HostAllow) {
233    loop {
234        match listener.accept().await {
235            Ok((mut conn, _)) => {
236                let hosts = hosts.clone();
237                tokio::spawn(async move {
238                    if let Err(e) = handle_connection(&mut conn, &hosts).await {
239                        tracing::warn!(error = %e, "proxy connection failed");
240                    }
241                });
242            }
243            Err(e) => {
244                tracing::warn!(error = %e, "proxy accept failed");
245                return;
246            }
247        }
248    }
249}

251/// Serve one proxied connection over any duplex byte stream.
252///
253/// This is the platform-agnostic core shared by the Unix socket
254/// listener ([`spawn_proxy`]) and the Windows named-pipe front end in
255/// `tau-sandbox-windows`. Reads one request head, validates the host
256/// against `hosts`, then tunnels (CONNECT) or forwards (plain HTTP).
257pub async fn handle_connection<S>(conn: &mut S, hosts: &HostAllow) -> std::io::Result<()>
258where
259    S: tokio::io::AsyncRead + tokio::io::AsyncWrite + Unpin + Send,
260{
261    let mut buf = [0u8; 4096];
262    let n = conn.read(&mut buf).await?;
263    ptrace!("head read n={n} bytes head={:?}", head_preview(&buf[..n]));
264    let first_line: &[u8] = match buf[..n].iter().position(|&b| b == b'\n') {
265        Some(idx) => &buf[..idx],
266        None => &buf[..n],
267    };
268    if first_line.starts_with(b"CONNECT ") {
269        handle_connect(conn, &buf[..n], hosts).await
270    } else {
271        handle_http(conn, &buf[..n], hosts).await
272    }
273}

275async fn handle_connect<S>(
276    plugin_sock: &mut S,
277    initial: &[u8],
278    hosts: &HostAllow,
279) -> std::io::Result<()>
280where
281    S: tokio::io::AsyncRead + tokio::io::AsyncWrite + Unpin + Send,
282{
283    let req = match parse_connect_request(initial) {
284        Ok(r) => r,
285        Err(_) => {
286            ptrace!("connect parse FAILED -> 400");
287            plugin_sock
288                .write_all(b"HTTP/1.1 400 Bad Request\r\n\r\n")
289                .await?;
290            return Ok(());
291        }
292    };
293    ptrace!("connect parsed host={} port={}", req.host, req.port);
294    if !hosts.permits(&req.host) {
295        ptrace!("connect host DENIED host={} -> 403", req.host);
296        plugin_sock
297            .write_all(b"HTTP/1.1 403 Forbidden\r\n\r\n")
298            .await?;
299        return Ok(());
300    }
301    if req.port != 443 {
302        ptrace!("connect port-gate REJECT port={} -> 400", req.port);
303        plugin_sock
304            .write_all(b"HTTP/1.1 400 Bad Request\r\n\r\n")
305            .await?;
306        return Ok(());
307    }
308    let mut remote = TcpStream::connect((req.host.as_str(), req.port)).await?;
309    ptrace!("connect upstream connect OK {}:{}", req.host, req.port);
310    plugin_sock
311        .write_all(b"HTTP/1.1 200 Connection established\r\n\r\n")
312        .await?;
313    // Peek the first chunk — should be TLS ClientHello with SNI matching CONNECT host
314    let mut peek_buf = [0u8; 1024];
315    let n = plugin_sock.read(&mut peek_buf).await?;
316    if let Some(sni) = peek_sni(&peek_buf[..n]) {
317        if sni != req.host {
318            return Err(std::io::Error::other(format!(
319                "SNI mismatch: CONNECT={} SNI={}",
320                req.host, sni
321            )));
322        }
323    } else {
324        return Err(std::io::Error::other("missing SNI in TLS ClientHello"));
325    }
326    // Forward the peeked bytes onward, then splice
327    remote.write_all(&peek_buf[..n]).await?;
328    ptrace!("connect tunnel established host={} peek={n}B", req.host);
329    let (pr, pw) = tokio::io::split(&mut *plugin_sock);
330    let (rr, rw) = remote.split();
331    splice_bidirectional(pr, pw, rr, rw, &req.host).await;
332    ptrace!("connect splice returned host={}", req.host);
333    Ok(())
334}

336/// Splice both directions of a proxied connection and log each direction's
337/// outcome under target `tau::proxy`.
338///
339/// # Termination: response-direction-driven, NOT `join!` (#622 CI round 2)
340///
341/// This used to be a [`tokio::join!`] of the two copies, which only
342/// completes when **both** directions reach EOF. That is unreachable
343/// whenever the client stream cannot express a half-close, and one of
344/// this function's two callers is exactly that case: on Windows the
345/// client is a `NamedPipeServer`, and tokio's
346/// `NamedPipeServer::poll_shutdown` / `NamedPipeClient::poll_shutdown`
347/// are **no-ops** — they just `poll_flush` and always return `Ready`
348/// (verified in `tokio-1.53.1/src/net/windows/named_pipe.rs:922` and
349/// `:1711`). So `AsyncWriteExt::shutdown()` never signals half-close over
350/// a pipe: the in-container bridge keeps its pipe handle open while it
351/// waits for the response, the `client -> remote` copy therefore never
352/// sees EOF, `join!` never returns, the handler never returns, the pipe
353/// server instance is never closed — and the bridge, in turn, never sees
354/// EOF on its own read. Circular non-termination: one stuck task and one
355/// leaked pipe instance per request, and the plugin gets its response
356/// only if it happens to be flushed before its own read timeout.
357///
358/// The rule now is: **the exchange is over when the response direction
359/// (`remote -> client`) completes.** At that point the upstream has
360/// closed (HTTP/1.1 `Connection: close`, or the far end of a CONNECT
361/// tunnel went away), so nothing more can arrive for the client and the
362/// connection is torn down — which is what closes the pipe handle and
363/// releases the peer. The request direction completing is *not* a
364/// termination signal: a client that half-closes after sending its
365/// request is still waiting for the response, so an `Ok` there is
366/// recorded and we keep waiting. An *error* there does end the splice —
367/// the client is gone, there is nobody left to deliver a response to.
368///
369/// CONNECT tunnels keep working: they are long-lived precisely because
370/// neither side EOFs, and this only ends them when the remote closes (or
371/// the client connection breaks), which is exactly when a tunnel is over.
372///
373/// Both directions' outcomes are still recorded (the audit-O3 property):
374/// the abandoned direction is logged as `incomplete` with no byte count
375/// rather than being silently dropped.
376async fn splice_bidirectional<CR, CW, RR, RW>(
377    mut client_r: CR,
378    mut client_w: CW,
379    mut remote_r: RR,
380    mut remote_w: RW,
381    host: &str,
382) where
383    CR: tokio::io::AsyncRead + Unpin,
384    CW: tokio::io::AsyncWrite + Unpin,
385    RR: tokio::io::AsyncRead + Unpin,
386    RW: tokio::io::AsyncWrite + Unpin,
387{
388    let up = tokio::io::copy(&mut client_r, &mut remote_w);
389    let down = tokio::io::copy(&mut remote_r, &mut client_w);
390    tokio::pin!(up, down);
391
392    let mut up_res: Option<std::io::Result<u64>> = None;
393    let mut down_res: Option<std::io::Result<u64>> = None;
394    loop {
395        tokio::select! {
396            // Disabled once it has produced a value: a completed future
397            // must never be polled again.
398            r = &mut up, if up_res.is_none() => {
399                let client_gone = r.is_err();
400                up_res = Some(r);
401                if client_gone {
402                    break;
403                }
404            }
405            r = &mut down => {
406                down_res = Some(r);
407                break;
408            }
409        }
410    }
411
412    log_direction(host, "client->remote", up_res);
413    log_direction(host, "remote->client", down_res);
414}

416/// Emit the `tau::proxy` event for one splice direction.
417///
418/// `None` means the direction was still in flight when the connection was
419/// torn down (see [`splice_bidirectional`]) — recorded rather than dropped.
420fn log_direction(host: &str, direction: &'static str, res: Option<std::io::Result<u64>>) {
421    match res {
422        Some(Ok(bytes)) => {
423            ptrace!("splice {direction} host={host} bytes={bytes}");
424            tracing::debug!(
425                target: "tau::proxy",
426                host = %host,
427                bytes,
428                direction,
429                "proxy splice direction complete"
430            )
431        }
432        Some(Err(e)) => {
433            ptrace!("splice {direction} host={host} FAILED err={e}");
434            tracing::warn!(
435                target: "tau::proxy",
436                host = %host,
437                error = %e,
438                direction,
439                "proxy splice direction failed mid-stream"
440            )
441        }
442        None => {
443            ptrace!("splice {direction} host={host} incomplete (torn down)");
444            tracing::debug!(
445                target: "tau::proxy",
446                host = %host,
447                direction,
448                "proxy splice direction abandoned when the peer direction finished"
449            )
450        }
451    }
452}

454/// True if `host` is an IP literal that is a loopback address
455/// (`127.0.0.0/8` or `::1`). Non-IP hostnames are never loopback.
456/// Matches the loopback semantics of [`validate::validate_hosts`].
457fn is_loopback_host(host: &str) -> bool {
458    host.parse::<std::net::IpAddr>()
459        .map(|ip| ip.is_loopback())
460        .unwrap_or(false)
461}

463/// Plaintext HTTP egress port policy. Mirrors [`handle_connect`]'s 443-only
464/// rule: a remote (non-loopback) host may only be reached on the well-known
465/// HTTP port 80; loopback hosts may use any port so local servers (e.g. a
466/// local model server on `http://127.0.0.1:11434`) keep working.
467fn http_port_allowed(host: &str, port: u16) -> bool {
468    is_loopback_host(host) || port == 80
469}

471async fn handle_http<S>(
472    plugin_sock: &mut S,
473    initial: &[u8],
474    hosts: &HostAllow,
475) -> std::io::Result<()>
476where
477    S: tokio::io::AsyncRead + tokio::io::AsyncWrite + Unpin + Send,
478{
479    let req = match parse_http_request(initial) {
480        Ok(r) => r,
481        Err(e) => {
482            ptrace!("http parse FAILED err={e} -> 400");
483            plugin_sock
484                .write_all(b"HTTP/1.1 400 Bad Request\r\n\r\n")
485                .await?;
486            return Ok(());
487        }
488    };
489    ptrace!(
490        "http parsed method={} host={} port={} path={} line_end={}",
491        req.method,
492        req.host,
493        req.port,
494        req.path_and_query,
495        req.line_end
496    );
497    if !hosts.permits(&req.host) {
498        ptrace!("http host DENIED host={} -> 403", req.host);
499        plugin_sock
500            .write_all(b"HTTP/1.1 403 Forbidden\r\n\r\n")
501            .await?;
502        return Ok(());
503    }
504    // Mirror handle_connect's port restriction on the plaintext path: a
505    // remote allowlisted host is reachable over plaintext only on port 80;
506    // loopback hosts may use any port (local servers).
507    if !http_port_allowed(&req.host, req.port) {
508        ptrace!(
509            "http port-gate REJECT host={} port={} -> 400",
510            req.host,
511            req.port
512        );
513        tracing::warn!(
514            host = %req.host,
515            port = req.port,
516            "proxy rejected plaintext HTTP to non-80 port on non-loopback host"
517        );
518        plugin_sock
519            .write_all(b"HTTP/1.1 400 Bad Request\r\n\r\n")
520            .await?;
521        return Ok(());
522    }
523    // Open TCP to the destination host:port.
524    let mut remote = match TcpStream::connect((req.host.as_str(), req.port)).await {
525        Ok(s) => {
526            ptrace!("http upstream connect OK {}:{}", req.host, req.port);
527            s
528        }
529        Err(e) => {
530            ptrace!(
531                "http upstream connect FAILED {}:{} err={e} -> 502",
532                req.host,
533                req.port
534            );
535            plugin_sock
536                .write_all(
537                    format!("HTTP/1.1 502 Bad Gateway\r\n\r\nupstream connect: {e}\r\n").as_bytes(),
538                )
539                .await?;
540            return Ok(());
541        }
542    };
543    // Send the rewritten request line, then the rest of the original buffer
544    // (headers + maybe partial body) verbatim.
545    let rewritten = rewrite_request_line(&req);
546    remote.write_all(rewritten.as_bytes()).await?;
547    remote.write_all(&initial[req.line_end..]).await?;
548    ptrace!(
549        "http forwarded head line={}B rest={}B",
550        rewritten.len(),
551        initial.len().saturating_sub(req.line_end)
552    );
553    // Splice both directions for the rest of the conversation.
554    let (pr, pw) = tokio::io::split(&mut *plugin_sock);
555    let (rr, rw) = remote.split();
556    splice_bidirectional(pr, pw, rr, rw, &req.host).await;
557    ptrace!("http splice returned host={}", req.host);
558    Ok(())
559}

561/// One-shot loopback upstream shared by the tests below.
562#[cfg(test)]
563mod test_upstream {
564    use tokio::io::{AsyncReadExt, AsyncWriteExt};
565    use tokio::net::TcpStream;
566
567    /// Drain the WHOLE forwarded request head, then write `resp`, then
568    /// half-close so the peer observes a clean FIN.
569    ///
570    /// # Why draining matters (#622 CI round 3)
571    ///
572    /// [`handle_http`](super::handle_http) forwards a request as **two**
573    /// writes — the rewritten request line, then the rest of the head it
574    /// had already buffered. A fixture that issues a single `read()` and
575    /// then drops the socket leaves the second write unread in the
576    /// receive buffer; on Windows, closing a socket with unread received
577    /// data aborts the connection with an **RST** instead of a FIN,
578    /// which destroys the response still in flight the other way. That
579    /// is exactly how `tau-sandbox-windows`'s
580    /// `egress_allowlisted_host_succeeds_through_full_chain` failed
581    /// (`splice remote->client FAILED ... os error 10054`).
582    ///
583    /// These two fixtures had the identical shape and only survived
584    /// because loopback usually coalesces both writes into one read —
585    /// i.e. they were latently flaky on Windows CI, not correct.
586    ///
587    /// Bounded by a byte cap and a per-read timeout: a regression must
588    /// fail the assertion, never hang the job.
589    pub(super) async fn answer_after_draining_head(s: &mut TcpStream, resp: &[u8]) {
590        let mut head = Vec::new();
591        let mut chunk = [0u8; 1024];
592        while !head.windows(4).any(|w| w == b"\r\n\r\n") && head.len() < 64 * 1024 {
593            match tokio::time::timeout(std::time::Duration::from_secs(5), s.read(&mut chunk)).await
594            {
595                Ok(Ok(n)) if n > 0 => head.extend_from_slice(&chunk[..n]),
596                // EOF, read error, or timeout: answer with what we have
597                // and let the test's own assertion report the problem.
598                _ => break,
599            }
600        }
601        s.write_all(resp).await.expect("write resp");
602        s.shutdown().await.expect("shutdown upstream write side");
603    }
604}

826    #[derive(Clone, Default, Debug)]
827    struct CapturedEvent {
828        target: String,
829        host: Option<String>,
830        bytes: Option<u64>,
831    }
832
833    #[derive(Default)]
834    struct CaptureVisitor {
835        ev: CapturedEvent,
836    }
837    impl Visit for CaptureVisitor {
838        fn record_u64(&mut self, field: &Field, value: u64) {
839            if field.name() == "bytes" {
840                self.ev.bytes = Some(value);
841            }
842        }
843        fn record_debug(&mut self, field: &Field, value: &dyn std::fmt::Debug) {
844            if field.name() == "host" {
845                self.ev.host = Some(format!("{value:?}").trim_matches('"').to_string());
846            }
847        }
848    }
849
850    #[derive(Clone, Default)]
851    struct CaptureLayer {
852        events: Arc<Mutex<Vec<CapturedEvent>>>,
853    }
854    impl<S: tracing::Subscriber> Layer<S> for CaptureLayer {
855        fn on_event(&self, event: &tracing::Event<'_>, _ctx: Context<'_, S>) {
856            let mut v = CaptureVisitor::default();
857            v.ev.target = event.metadata().target().to_string();
858            event.record(&mut v);
859            self.events.lock().unwrap().push(v.ev);
860        }
861    }
862
863    // A clean plaintext-HTTP exchange through the proxy must emit a
864    // `tau::proxy` splice event carrying the destination `host` and a `bytes`
865    // count — proving the byte-splice result is no longer swallowed (audit O3).
866    #[tokio::test]
867    async fn splice_emits_event_with_host_and_bytes() {
868        let cap = CaptureLayer::default();
869        let events = cap.events.clone();
870        let subscriber = tracing_subscriber::registry()
871            .with(cap.with_filter(tracing_subscriber::filter::LevelFilter::DEBUG));
872        let _guard = tracing::subscriber::set_default(subscriber);
873
874        // Loopback upstream: read the forwarded request, reply, then close.
875        let upstream = tokio::net::TcpListener::bind("127.0.0.1:0")
876            .await
877            .expect("bind upstream");
878        let port = upstream.local_addr().expect("addr").port();
879        let up = tokio::spawn(async move {
880            let (mut s, _) = upstream.accept().await.expect("accept");
881            // Drain the whole head before answering, then FIN: see
882            // `test_upstream::answer_after_draining_head`.
883            super::test_upstream::answer_after_draining_head(
884                &mut s,
885                b"HTTP/1.1 200 OK\r\nContent-Length: 2\r\n\r\nhi",
886            )
887            .await;
888        });

890        let h = spawn_proxy(HostAllow::Exact(vec!["127.0.0.1".to_string()])).expect("spawn");
891        let mut conn = UnixStream::connect(h.sock_path()).await.expect("connect");
892        let req =

894        conn.write_all(req.as_bytes()).await.expect("write req");
895        // Half-close our write side so the client->remote copy reaches EOF and
896        // the splice can complete cleanly.
897        conn.shutdown().await.expect("shutdown write");
898        let mut resp = Vec::new();
899        let _ = conn.read_to_end(&mut resp).await;
900        up.await.expect("upstream task");

903        tokio::time::sleep(std::time::Duration::from_millis(100)).await;
