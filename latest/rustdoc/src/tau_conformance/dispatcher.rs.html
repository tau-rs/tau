1//! `ConformanceDispatcher` — the [`ToolDispatcher`] for the canonical
2//! fan-monitor scenario.
3//!
4//! Routes the scenario's three tools and supplies the scripted
5//! [`SequencedLlm`](crate::sequenced_llm::SequencedLlm) backend:
6//!
7//! - native `read_temp` → deterministic body `32`
8//! - native `set_fan`   → deterministic body `{"ok": true}`
9//! - MCP `weather`       → real cassette replay via an opened
10//!   [`McpClient`] (`client.call_tool("weather", args)`), with the
11//!   response's text content blocks mapped deterministically to JSON.
12//!
13//! Clock/random are intentionally NOT overridden (the trait defaults
14//! return `None`); under the crate's `test-fixtures` feature `run_agent`
15//! injects `MockClock` / `DeterministicRandom`, which is exactly what a
16//! cross-profile conformance run wants — identical, deterministic time
17//! and entropy in both dev and wasm profiles.

33/// [`ToolDispatcher`] for the fan-monitor conformance scenario.
34pub(crate) struct ConformanceDispatcher {
35    backend: Arc<dyn DynLlmBackend>,
36    weather: Option<Arc<McpClient>>,
37}
38
39impl ConformanceDispatcher {
40    /// Construct with both native tools and a live MCP `weather` client.
41    ///
42    /// Used by `DevProfile`, which opens the cassette-backed `weather`
43    /// server before driving the scenario.
44    pub(crate) fn new(backend: Arc<dyn DynLlmBackend>, weather: Arc<McpClient>) -> Self {
45        Self {
46            backend,
47            weather: Some(weather),
48        }
49    }
50
51    /// Construct with only the native tools wired (no MCP client).
52    ///
53    /// Invoking the `weather` tool against this dispatcher surfaces a
54    /// [`RuntimeError::Internal`] — a scenario that drives `weather`
55    /// without an MCP client wired is a wiring bug and should fail loudly.
56    pub(crate) fn new_native_only(backend: Arc<dyn DynLlmBackend>) -> Self {
57        Self {
58            backend,
59            weather: None,
60        }
61    }
62}

64impl ToolDispatcher for ConformanceDispatcher {
65    fn invoke<'a>(
66        &'a self,
67        tool_id: &'a ToolId,
68        args: &'a Value,
69    ) -> Pin<Box<dyn Future<Output = Result<ToolInvocationResult, RuntimeError>> + Send + 'a>> {
70        // Clone the borrowed inputs into the async move so the returned
71        // future is `Send + 'a` without holding the `&ToolId`/`&Value`
72        // borrows across the `.await` (mirrors ForwardingDispatcher).
73        let name = tool_id.0.clone();
74        let args_owned = args.clone();
75        let weather = self.weather.clone();
76
77        Box::pin(async move {
78            // Native tools come from the shared no_std crate so dev and the
79            // wasm guest return byte-identical bodies (PR-F).
80            if let Some(body) = tau_native_tools::invoke(&name, &args_owned) {
81                return Ok(ToolInvocationResult {
82                    body: Some(body),
83                    error: None,
84                });
85            }
86            match name.as_str() {
87                "weather" => {
88                    let client = weather.ok_or_else(|| RuntimeError::Internal {
89                        message: "weather invoked but no MCP client wired".to_string(),
90                    })?;
91                    let resp = client.call_tool("weather", args_owned).await.map_err(|e| {
92                        RuntimeError::Internal {
93                            message: format!("MCP weather call_tool failed: {e}"),
94                        }
95                    })?;
96                    Ok(ToolInvocationResult {
97                        body: Some(mcp_response_to_json(&resp)),
98                        error: None,
99                    })
100                }
101                other => Err(RuntimeError::Internal {
102                    message: format!("unknown conformance tool {other:?}"),
103                }),
104            }
105        })
106    }
107
108    fn llm_backend_for(&self, _backend: &str) -> Result<Arc<dyn DynLlmBackend>, RuntimeError> {
109        // Single-backend conformance dispatcher: every agent/judge resolves to
110        // the one mock backend regardless of the name in its model_ref.
111        Ok(self.backend.clone())
112    }
113}

115/// Map an MCP `tools/call` response to a deterministic JSON body.
116///
117/// Mirrors `McpBackedTool`'s content handling
118/// (`crates/tau-mcp-tokio/src/bridge.rs`): text blocks join in order;
119/// non-text blocks degrade to a stable placeholder so the body is
120/// byte-deterministic across profiles. The joined text is then parsed
121/// as JSON (so a server returning `{"temp":22}` lands as an object),
122/// falling back to a JSON string when the text is not valid JSON.
123fn mcp_response_to_json(resp: &ToolsCallResponse) -> Value {
124    let joined: String = resp
125        .content
126        .iter()
127        .map(|b| match b {
128            ContentBlock::Text { text } => text.clone(),
129            ContentBlock::Image { mime_type, .. } => {
130                format!("[image/{mime_type} — not rendered in v0]")
131            }
132            ContentBlock::Resource { .. } => "[resource — not rendered in v0]".to_string(),
133        })
134        .collect::<Vec<_>>()
135        .join("");
136
137    match serde_json::from_str::<Value>(joined.trim()) {
138        Ok(v) => v,
139        Err(_) => Value::String(joined),
140    }
141}

143#[cfg(test)]
144mod tests {
145    use super::*;
146    use std::sync::Arc;
147    use tau_runtime_core::interpreter::tool_dispatch::ToolDispatcher;
148
149    #[tokio::test(flavor = "current_thread")]
150    async fn native_tools_return_deterministic_bodies() {
151        let llm = Arc::new(crate::sequenced_llm::SequencedLlm::new("mock-llm", vec![]));
152        let d = ConformanceDispatcher::new_native_only(llm);
153        let r = d
154            .invoke(&tau_ir::ToolId("read_temp".into()), &serde_json::json!({}))
155            .await
156            .unwrap();
157        assert_eq!(r.body, Some(serde_json::json!(32)));
158        let r = d
159            .invoke(
160                &tau_ir::ToolId("set_fan".into()),
161                &serde_json::json!({"on":true}),
162            )
163            .await
164            .unwrap();
165        assert_eq!(r.body, Some(serde_json::json!({"ok": true})));
166        assert!(d
167            .invoke(&tau_ir::ToolId("nope".into()), &serde_json::json!({}))
168            .await
169            .is_err());
170    }
171
172    #[test]
173    fn mcp_response_text_blocks_join_and_parse_json() {
174        let resp = ToolsCallResponse {
175            content: vec![ContentBlock::Text {
176                text: "{\"forecast\":\"sunny\"}".to_string(),
177            }],
178            is_error: None,
179        };
180        assert_eq!(
181            mcp_response_to_json(&resp),
182            serde_json::json!({"forecast": "sunny"})
183        );
184    }
185
186    #[test]
187    fn mcp_response_plain_text_falls_back_to_string() {
188        let resp = ToolsCallResponse {
189            content: vec![ContentBlock::Text {
190                text: "sunny".to_string(),
191            }],
192            is_error: None,
193        };
194        assert_eq!(
195            mcp_response_to_json(&resp),
196            serde_json::Value::String("sunny".to_string())
197        );
198    }
