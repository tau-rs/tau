1//! The versioned, crate-owned conformance event model. The authoritative
2//! comparison contract (ADR-0048). Each variant is sourced from exactly
3//! one runtime channel during normalization (see `normalize.rs`).
4
5use serde::{Deserialize, Serialize};
6use serde_json::Value;
7
8/// Bump when the whitelist or any field projection changes; re-bless
9/// goldens in the same change. Recorded in `expected_events.json`.
10pub const CONFORMANCE_EVENT_VERSION: u32 = 1;
11
12#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
13#[serde(tag = "kind")]
14pub enum ConformanceEvent {
15    /// tracing `runtime.run_started`. run_id is modulo (not present here).
16    RunStarted,
17    /// tracing `runtime.context_step_ran`. Token counts compared.
18    ContextStepRan {
19        step: String,
20        tokens_in: u64,
21        tokens_out: u64,
22    },
23    /// tracing `llm.request_built`.
24    InferenceCallStarted,
25    /// tracing `llm.response_received` (+ folded `llm.token_usage`).
26    InferenceCallCompleted {
27        stop_reason: String,
28        tokens_in: u64,
29        tokens_out: u64,
30    },
31    /// RunEvent::ToolCallStarted. `call` is the canonical first-seen
32    /// ordinal (e.g. "tc#0"); the provider id is modulo.
33    ToolCallStarted {
34        name: String,
35        args: Value,
36        call: String,
37    },
38    /// RunEvent::ToolCallCompleted. `result` is the Ok body or a canonical
39    /// error marker; `call` matches the paired Started ordinal.
40    ToolCallCompleted {
41        name: String,
42        result: ToolOutcome,
43        call: String,
44    },
45    /// RunEvent::RunCompleted. Outcome discriminant only.
46    RunCompleted { outcome: String },
47}
48
49#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
50#[serde(tag = "status", rename_all = "snake_case")]
51pub enum ToolOutcome {
52    /// Tool ran. `is_error` is the runtime's semantic-error flag
53    /// (Ok(ToolResult{is_error:true})); `body` is the extracted content.
54    Ok { body: Value, is_error: bool },
55    /// Transport/dispatch error (Result::Err). Error text is modulo.
56    Err,
57}

59#[cfg(test)]
60mod tests {
61    use super::*;
62    #[test]
63    fn serde_round_trips_and_carries_version() {
64        let ev = ConformanceEvent::ToolCallStarted {
65            name: "read_temp".into(),
66            args: serde_json::json!({}),
67            call: "tc#0".into(),
68        };
69        let s = serde_json::to_string(&ev).unwrap();
70        let back: ConformanceEvent = serde_json::from_str(&s).unwrap();
71        assert_eq!(ev, back);
72        const { assert!(CONFORMANCE_EVENT_VERSION >= 1) };
73    }
74}
