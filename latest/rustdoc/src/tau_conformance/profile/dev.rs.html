1//! Interpreted dev profile: drive `run_ir_streaming` and normalize its
2//! single typed `RunEvent` stream into `ConformanceEvent`s (ADR-0049,
3//! supersedes ADR-0048's dual-channel interleave). No tracing subscriber
4//! is installed — the engine emits every gate event as a typed variant,
5//! which is exactly what a no_std wasm guest produces across the
6//! component boundary.
7
8use std::path::Path;
9use std::sync::Arc;

11use futures::StreamExt as _;

13use crate::dispatcher::ConformanceDispatcher;
14use crate::event::ConformanceEvent;
15use crate::normalize::{map_runevent, NormState};
16use crate::profile::{Profile, ProfileError};
17use crate::scenario::Scenario;
18use crate::sequenced_llm::SequencedLlm;

20/// The interpreted-runtime conformance profile. See module docs.
21pub struct DevProfile;
22
23#[async_trait::async_trait(?Send)]
24impl Profile for DevProfile {
25    fn name(&self) -> &str {
26        "dev"
27    }
28
29    async fn run(&self, scenario: &Scenario) -> Result<Vec<ConformanceEvent>, ProfileError> {
30        let backend = Arc::new(SequencedLlm::new("mock-llm", scenario.responses.clone()));
31
32        // Open the cassette-backed MCP weather client ONLY if the cassette
33        // file exists (native-only scenarios have none).
34        let dispatcher = if scenario.weather_cassette.exists() {
35            let client = open_weather_client(&scenario.weather_cassette)
36                .await
37                .map_err(|e| ProfileError(format!("open weather cassette: {e}")))?;
38            Arc::new(ConformanceDispatcher::new(backend, client))
39        } else {
40            Arc::new(ConformanceDispatcher::new_native_only(backend))
41        };
42
43        let stream = tau_runtime_core::interpreter::run_ir_streaming(
44            scenario.module.clone(),
45            &scenario.entry,
46            dispatcher,
47            Vec::new(),
48        )
49        .await
50        .map_err(|e| ProfileError(format!("run_ir_streaming: {e}")))?;
51        let mut stream = Box::pin(stream);
52
53        // Single-channel: every conformance event is sourced from the typed
54        // `RunEvent` stream. No tracing capture, no interleave.
55        let mut out: Vec<ConformanceEvent> = Vec::new();
56        let mut st = NormState::default();
57        while let Some(ev) = stream.next().await {
58            if let Some(ce) = map_runevent(ev, &mut st) {
59                out.push(ce);
60            }
61        }
62        Ok(out)
63    }
64}

66/// Open the cassette-backed `weather` MCP client for the dev profile.
67///
68/// Uses the `cassette:` URL scheme (replay, no real process spawn), a
69/// passthrough process gate, and an empty capability plan — the cassette
70/// transport needs no real capabilities.
71async fn open_weather_client(
72    path: &Path,
73) -> Result<Arc<tau_mcp_tokio::host_lifecycle::client::McpClient>, String> {
74    use tau_mcp_tokio::host_lifecycle::{open, McpClientOptions};
75    let url = format!("cassette:{}", path.display());
76    let gate: Arc<dyn tau_runtime_tokio::process_gate::DynProcessCapabilityGate> =
77        Arc::new(tau_runtime_tokio::process_gate::passthrough::PassthroughSandbox::new());
78    let plan = tau_ports::CapabilityPlan::new(Vec::new(), None, None);
79    let client = open(&url, &plan, gate, McpClientOptions::default())
80        .await
81        .map_err(|e| e.to_string())?;
82    Ok(Arc::new(client))
83}

88    use crate::event::ConformanceEvent;

91    fn write_native_fixture(dir: &std::path::Path) {
92        let toml = r#"
93packages = ["mock-llm"]
94
95[project]
96name = "mini"
97
98[models.mock-1]
99backend = "mock-llm"
100model = "mock-1"
101
102[agents.solo]
103display_name = "Solo"
104package      = "solo@^0.1"
105model        = "mock-1"
106tool_refs    = ["read_temp"]
107max_turns    = 3
108
109[tools.read_temp]
110native      = "ReadTemp"
111description = "Read temp."
112capabilities = []
113"#;
114        std::fs::write(dir.join("tau.toml"), toml).unwrap();
115        let mut f = std::fs::File::create(dir.join("mock_llm.jsonl")).unwrap();
116        writeln!(f, "{{\"response\": {{\"tool_uses\": [{{\"id\": \"t0\", \"name\": \"read_temp\", \"input\": {{}}}}], \"stop_reason\": \"tool_use\"}}}}").unwrap();
117        writeln!(
118            f,
119            "{{\"response\": {{\"text\": \"done\", \"stop_reason\": \"end_turn\"}}}}"
120        )
121        .unwrap();
122    }
123
124    #[tokio::test(flavor = "current_thread")]
125    async fn dev_profile_native_round_trip() {
126        let tmp = tempfile::tempdir().unwrap();
127        write_native_fixture(tmp.path());
128        let s = crate::scenario::Scenario::load(tmp.path().to_path_buf()).unwrap();
129        let events = DevProfile.run(&s).await.expect("dev runs");
130        assert!(
131            matches!(events.first(), Some(ConformanceEvent::RunStarted)),
132            "first={:?}",
133            events.first()
134        );
135        assert!(
136            matches!(events.last(), Some(ConformanceEvent::RunCompleted { .. })),
137            "last={:?}",
138            events.last()
139        );
140        assert!(
141            events.iter().any(
142                |e| matches!(e, ConformanceEvent::ToolCallStarted{name, ..} if name=="read_temp")
143            ),
144            "expected read_temp ToolCallStarted; got {events:#?}"
145        );
146        assert!(events.iter().any(
147            |e| matches!(e, ConformanceEvent::ToolCallCompleted{name, ..} if name=="read_temp")
148        ));
149    }
