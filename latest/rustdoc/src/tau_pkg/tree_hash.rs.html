1//! Source-agnostic whole-tree SHA-256 hashing for `tau verify`
2//! (ADR-0012 / Tier 2 priority 7).
3//!
4//! Walks the install dir, sorts files by relative path, and feeds
5//! `path\0content\0` for each into a single SHA-256 stream. Excludes
6//! `.git/`, `target/`, and any `*.tau-tmp/` directories.
7//!
8//! Symlinks are NOT followed (`walkdir::WalkDir::follow_links(false)`).
9//! A symlink contributes its target path bytes to the hash, not the
10//! resolved file's content. This prevents symlink-loop pitfalls.
11//!
12//! Used at install time (populate `LockedVersion.sha256`) and at
13//! verify time (recompute + compare).
14
15use sha2::{Digest, Sha256};
16use std::fs;
17use std::path::{Path, PathBuf};
18use walkdir::WalkDir;
19
20/// Lowercase-hex encode a byte slice. digest 0.11's `finalize()` returns
21/// a `hybrid_array::Array<u8, ..>` which no longer implements `LowerHex`
22/// directly (it did in 0.10 via `GenericArray`), so we format manually.
23pub(crate) fn to_hex_lower(bytes: &[u8]) -> String {
24    use std::fmt::Write as _;
25    let mut s = String::with_capacity(bytes.len() * 2);
26    for b in bytes {
27        let _ = write!(s, "{b:02x}");
28    }
29    s
30}

32/// Error from `tree_hash`.
33///
34/// # Example
35///
36/// ```
37/// use tau_pkg::tree_hash::TreeHashError;
38///
39/// let err = TreeHashError::Io {
40///     path: std::path::PathBuf::from("/no/such/file"),
41///     message: "reading file: No such file or directory".to_string(),
42/// };
43/// let display = format!("{err}");
44/// assert!(display.contains("io error at"));
45/// assert!(display.contains("/no/such/file"));
46/// ```
47#[non_exhaustive]
48#[derive(Debug, thiserror::Error)]
49pub enum TreeHashError {
50    /// I/O error reading a file or walking the tree.
51    #[error("io error at {path}: {message}")]
52    Io {
53        /// The path that errored.
54        path: PathBuf,
55        /// The error message.
56        message: String,
57    },
58}
59
60/// SHA-256 of a single file. Used for plugin binary hashing
61/// (`LockedPlugin.binary_sha256`).
62///
63/// Returns 64-char lowercase hex.
64///
65/// # Example
66///
67/// ```
68/// use tau_pkg::tree_hash::sha256_of_file;
69///
70/// # let tmp = tempfile::tempdir().expect("tempdir");
71/// # let path = tmp.path().join("file.bin");
72/// # std::fs::write(&path, b"hello world").expect("write");
73/// let hex = sha256_of_file(&path).expect("sha256_of_file");
74/// assert_eq!(hex.len(), 64);
75/// assert!(hex.chars().all(|c| c.is_ascii_hexdigit() && !c.is_ascii_uppercase()));
76/// ```
77pub fn sha256_of_file(path: &Path) -> Result<String, TreeHashError> {
78    let bytes = fs::read(path).map_err(|e| TreeHashError::Io {
79        path: path.to_path_buf(),
80        message: format!("reading file: {e}"),
81    })?;
82    let mut hasher = Sha256::new();
83    hasher.update(&bytes);
84    Ok(to_hex_lower(&hasher.finalize()))
85}

87/// Hash entry for a single file in the tree.
88///
89/// # Example
90///
91/// ```
92/// use tau_pkg::tree_hash::sha256_of_file;
93///
94/// // `FileHash` is produced internally by `tree_hash`; obtain a SHA-256
95/// // via `sha256_of_file` to verify the same hex encoding is used.
96/// # let tmp = tempfile::tempdir().expect("tempdir");
97/// # let path = tmp.path().join("a.txt");
98/// # std::fs::write(&path, b"content").expect("write");
99/// let hex = sha256_of_file(&path).expect("sha256");
100/// assert_eq!(hex.len(), 64, "SHA-256 is always 64 hex chars");
101/// ```
102#[non_exhaustive]
103#[derive(Debug, Clone, PartialEq, Eq)]
104pub struct FileHash {
105    /// Relative path from the tree root.
106    pub rel_path: String,
107    /// SHA-256 hex of the file contents (or symlink target bytes).
108    pub sha256: String,
109}

111/// Compute a whole-tree SHA-256 of `root`.
112///
113/// Excludes `.git/`, `target/`, and `*.tau-tmp/` directories. Files
114/// are sorted lexicographically by relative path. For each file:
115///
116/// ```text
117/// hasher.update(rel_path.as_bytes());
118/// hasher.update(&[0]);
119/// hasher.update(file_contents);
120/// hasher.update(&[0]);
121/// ```
122///
123/// Returns 64-char lowercase hex.
124///
125/// # Example
126///
127/// ```
128/// use tau_pkg::tree_hash;
129///
130/// # let tmp = tempfile::tempdir().unwrap();
131/// # std::fs::write(tmp.path().join("hello.txt"), b"hello world").unwrap();
132/// let hash = tree_hash(tmp.path()).unwrap();
133/// assert_eq!(hash.len(), 64);
134/// ```
135pub fn tree_hash(root: &Path) -> Result<String, TreeHashError> {
136    let mut entries: Vec<(String, PathBuf)> = Vec::new();
137
138    for entry in WalkDir::new(root).follow_links(false) {
139        let entry = entry.map_err(|e| TreeHashError::Io {
140            path: e.path().map(Path::to_path_buf).unwrap_or_default(),
141            message: format!("walking tree: {e}"),
142        })?;
143
144        if entry.file_type().is_dir() {
145            continue;
146        }
147
148        // Skip files inside excluded directories.
149        let path = entry.path();
150        if path.components().any(|c| {
151            let s = c.as_os_str().to_string_lossy();
152            s == ".git" || s == "target" || s.ends_with(".tau-tmp")
153        }) {
154            continue;
155        }
156
157        let rel = path.strip_prefix(root).map_err(|e| TreeHashError::Io {
158            path: path.to_path_buf(),
159            message: format!("computing relative path: {e}"),
160        })?;
161        let rel_str = rel.to_string_lossy().replace('\\', "/");
162        entries.push((rel_str, path.to_path_buf()));
163    }
164
165    entries.sort_by(|a, b| a.0.cmp(&b.0));
166
167    let mut hasher = Sha256::new();
168    for (rel_path, abs_path) in entries {
169        let bytes = if abs_path.is_symlink() {
170            let target = fs::read_link(&abs_path).map_err(|e| TreeHashError::Io {
171                path: abs_path.clone(),
172                message: format!("reading symlink: {e}"),
173            })?;
174            target.to_string_lossy().into_owned().into_bytes()
175        } else {
176            fs::read(&abs_path).map_err(|e| TreeHashError::Io {
177                path: abs_path.clone(),
178                message: format!("reading file: {e}"),
179            })?
180        };
181
182        hasher.update(rel_path.as_bytes());
183        hasher.update([0u8]);
184        hasher.update(&bytes);
185        hasher.update([0u8]);
186    }
187
188    Ok(to_hex_lower(&hasher.finalize()))
189}

191#[cfg(test)]
192mod tests {
193    use super::*;
194    use std::fs;
195    use tempfile::TempDir;
196
197    #[test]
198    fn empty_dir_hashes_to_empty_sha256() {
199        let dir = TempDir::new().unwrap();
200        let h = tree_hash(dir.path()).unwrap();
201        // SHA-256 of empty input is a known value:
202        assert_eq!(
203            h,
204            "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"
205        );
206    }
207
208    #[test]
209    fn single_file_is_deterministic() {
210        let dir = TempDir::new().unwrap();
211        fs::write(dir.path().join("a.txt"), b"hello").unwrap();
212        let h1 = tree_hash(dir.path()).unwrap();
213        let h2 = tree_hash(dir.path()).unwrap();
214        assert_eq!(h1, h2);
215        assert_eq!(h1.len(), 64);
216    }
217
218    #[test]
219    fn nested_dirs_hash_includes_all_files() {
220        let dir = TempDir::new().unwrap();
221        fs::create_dir_all(dir.path().join("src")).unwrap();
222        fs::write(dir.path().join("a.txt"), b"hello").unwrap();
223        fs::write(dir.path().join("src/b.rs"), b"fn main() {}").unwrap();
224        let h_two_files = tree_hash(dir.path()).unwrap();
225        // Drop one file → different hash.
226        fs::remove_file(dir.path().join("src/b.rs")).unwrap();
227        let h_one_file = tree_hash(dir.path()).unwrap();
228        assert_ne!(h_two_files, h_one_file);
229    }
230
231    #[test]
232    fn dot_git_dir_is_excluded() {
233        let dir = TempDir::new().unwrap();
234        fs::write(dir.path().join("a.txt"), b"hello").unwrap();
235        let h_no_git = tree_hash(dir.path()).unwrap();
236
237        fs::create_dir_all(dir.path().join(".git/objects")).unwrap();
238        fs::write(dir.path().join(".git/HEAD"), b"ref: refs/heads/main").unwrap();
239        fs::write(dir.path().join(".git/objects/abc"), b"blob").unwrap();
240        let h_with_git = tree_hash(dir.path()).unwrap();
241        assert_eq!(h_no_git, h_with_git, ".git/ should be excluded");
242    }
243
244    #[test]
245    fn target_dir_is_excluded() {
246        let dir = TempDir::new().unwrap();
247        fs::write(dir.path().join("Cargo.toml"), b"[package]").unwrap();
248        let h_no_target = tree_hash(dir.path()).unwrap();
249
250        fs::create_dir_all(dir.path().join("target/release")).unwrap();
251        fs::write(dir.path().join("target/release/foo"), b"binary").unwrap();
252        let h_with_target = tree_hash(dir.path()).unwrap();
253        assert_eq!(h_no_target, h_with_target, "target/ should be excluded");
254    }
