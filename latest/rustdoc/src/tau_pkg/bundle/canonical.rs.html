1//! Canonical TOML serialization for `BundleManifest`.
2//!
3//! `serde` + `toml`'s default emitter does not guarantee deterministic
4//! field ordering across versions; we hand-roll a small emitter that
5//! writes fields in a fixed order so the self-hash is reproducible.
6
7use std::fmt::Write;
8
9use crate::bundle::manifest::{
10    BackendRef, BundleAgent, BundleEffectiveCapabilities, BundleManifest, BundlePackage,
11    BundleTrigger, GovernanceRecord, GovernanceVerdict, IrPayload,
12};

14/// Emit the canonical TOML serialization of a `BundleManifest`.
15///
16/// Field order is fixed; arrays of tables (`[[packages]]`, `[[agents]]`)
17/// are emitted in the order they appear in the input struct. Empty
18/// optional values are omitted per the spec.
19pub fn to_canonical_toml(manifest: &BundleManifest) -> String {
20    let mut out = String::with_capacity(2048);
21    let _ = writeln!(out, "schema_version = {}", manifest.schema_version);
22
23    // [bundle]
24    out.push('\n');
25    out.push_str("[bundle]\n");
26    write_str_kv(&mut out, "sha256", &manifest.bundle.sha256);
27    write_str_kv(&mut out, "created_at", &manifest.bundle.created_at);
28    write_str_kv(&mut out, "tau_version", &manifest.bundle.tau_version);
29    write_str_kv(&mut out, "target", &manifest.bundle.target.to_string());
30    if let Some(sel) = &manifest.bundle.selected_agents {
31        write_string_array(&mut out, "selected_agents", sel.clone());
32    }
33
34    // [project]
35    out.push('\n');
36    out.push_str("[project]\n");
37    write_str_kv(&mut out, "name", &manifest.project.name);
38    write_str_kv(&mut out, "version", &manifest.project.version.to_string());
39    write_str_kv(
40        &mut out,
41        "tau_toml_sha256",
42        &manifest.project.tau_toml_sha256,
43    );
44
45    // [[packages]]
46    for pkg in &manifest.packages {
47        out.push('\n');
48        out.push_str("[[packages]]\n");
49        write_package(&mut out, pkg);
50    }
51
52    // [[agents]]
53    for agent in &manifest.agents {
54        out.push('\n');
55        out.push_str("[[agents]]\n");
56        write_agent(&mut out, agent);
57    }
58
59    // [[trigger]] — emitted when present so the bindings participate in the
60    // self-hash. Order is the struct's Vec order (lowering sorts by name).
61    for trigger in &manifest.triggers {
62        out.push('\n');
63        out.push_str("[[trigger]]\n");
64        write_trigger(&mut out, trigger);
65    }
66
67    // [governance] — emitted when present so the verdict participates in the
68    // self-hash (a verdict must not be rewritable post-build).
69    if let Some(gov) = &manifest.governance {
70        out.push('\n');
71        out.push_str("[governance]\n");
72        write_governance(&mut out, gov);
73    }
74
75    // [ir_payload] — emitted when present so the bytes participate in the self-hash.
76    if let Some(ir) = &manifest.ir_payload {
77        out.push('\n');
78        out.push_str("[ir_payload]\n");
79        write_ir_payload(&mut out, ir);
80    }
81
82    // [[assets]] — content-addressed asset store (D6-B). The `build` writer
83    // sorts by `hash`; emit in that order so the bytes participate in the
84    // self-hash deterministically.
85    for asset in &manifest.assets {
86        out.push('\n');
87        out.push_str("[[assets]]\n");
88        write_asset(&mut out, asset);
89    }
90
91    out
92}

94fn write_asset(out: &mut String, asset: &crate::bundle::manifest::BundleAsset) {
95    // Fixed field order for a deterministic hash; `bytes_hex` is already hex.
96    write_str_kv(out, "hash", &asset.hash);
97    write_str_kv(out, "kind", &asset.kind);
98    write_str_kv(out, "bytes_hex", &asset.bytes_hex);
99}

101fn write_governance(out: &mut String, gov: &GovernanceRecord) {
102    write_str_kv(out, "verdict", governance_verdict_to_str(gov.verdict));
103}

105/// Wire form of a governance verdict. MUST match the serde `kebab-case`
106/// rename on [`GovernanceVerdict`] so the canonical (hashed) emission and
107/// the serde round-trip agree byte-for-byte.
108fn governance_verdict_to_str(v: GovernanceVerdict) -> &'static str {
109    match v {
110        GovernanceVerdict::Governed => "governed",
111        GovernanceVerdict::Ungoverned => "ungoverned",
112        GovernanceVerdict::Skipped => "skipped",
113    }
114}

116fn write_package(out: &mut String, pkg: &BundlePackage) {
117    write_str_kv(out, "name", &pkg.name);
118    write_str_kv(out, "version", &pkg.version.to_string());
119    // PackageSource serializes to a string form via its own serde impl;
120    // we use that for consistency.
121    let source_str = match toml::Value::try_from(&pkg.source) {
122        Ok(toml::Value::String(s)) => s,
123        Ok(other) => other.to_string(),
124        Err(e) => panic!("PackageSource serialization must succeed for canonical TOML: {e}"),
125    };
126    write_str_kv(out, "source", &source_str);
127    write_str_kv(out, "tree_sha256", &pkg.tree_sha256);
128    if let Some(bin) = &pkg.binary_sha256 {
129        write_str_kv(out, "binary_sha256", bin);
130    }
131    if !pkg.required_shapes.is_empty() {
132        write_string_array(
133            out,
134            "required_shapes",
135            pkg.required_shapes
136                .iter()
137                .map(capability_shape_to_str)
138                .collect::<Vec<_>>(),
139        );
140    }
141}

143fn write_agent(out: &mut String, agent: &BundleAgent) {
144    write_str_kv(out, "id", agent.id.as_str());
145    write_backend_inline(out, &agent.backend);
146    write_str_kv(out, "system_prompt_sha256", &agent.system_prompt_sha256);
147    if !agent.required_tools.is_empty() {
148        write_string_array(out, "required_tools", agent.required_tools.to_vec());
149    }
150    if !agent.effective_capabilities.is_empty() {
151        out.push_str("\n[agents.effective_capabilities]\n");
152        write_effective_capabilities(out, &agent.effective_capabilities);
153    }
154}

156fn write_trigger(out: &mut String, t: &BundleTrigger) {
157    write_str_kv(out, "name", &t.name);
158    write_str_kv(out, "kind", &t.kind);
159    write_str_kv(out, "agent", &t.agent);
160    if let Some(s) = &t.schedule {
161        write_str_kv(out, "schedule", s);
162    }
163    if let Some(tz) = &t.timezone {
164        write_str_kv(out, "timezone", tz);
165    }
166    if let Some(r) = &t.retry {
167        // Inline sub-table for stable single-line emission.
168        out.push_str("retry = { ");
169        write!(out, "max_attempts = {}", r.max_attempts).unwrap();
170        write!(
171            out,
172            ", backoff_strategy = {}",
173            toml_string(&r.backoff_strategy)
174        )
175        .unwrap();
176        write!(out, ", backoff_base = {}", toml_string(&r.backoff_base)).unwrap();
177        write!(out, ", backoff_max = {}", toml_string(&r.backoff_max)).unwrap();
178        if let Some(dl) = &r.dead_letter {
179            write!(out, ", dead_letter = {}", toml_string(dl)).unwrap();
180        }
181        out.push_str(" }\n");
182    }
183}

185fn write_backend_inline(out: &mut String, backend: &BackendRef) {
186    out.push_str("backend = { ");
187    write!(out, "kind = {}", toml_string(&backend.kind)).unwrap();
188    if let Some(model) = &backend.model {
189        write!(out, ", model = {}", toml_string(model)).unwrap();
190    }
191    for (k, v) in &backend.extra {
192        let v_toml = v.to_string();
193        write!(out, ", {} = {}", toml_bare_key(k), v_toml).unwrap();
194    }
195    out.push_str(" }\n");
196}

198fn write_effective_capabilities(out: &mut String, caps: &BundleEffectiveCapabilities) {
199    if !caps.allow_fs_read.is_empty() {
200        write_string_array(out, "allow_fs_read", caps.allow_fs_read.clone());
201    }
202    if !caps.deny_fs_read.is_empty() {
203        write_string_array(out, "deny_fs_read", caps.deny_fs_read.clone());
204    }
205    if !caps.allow_fs_write.is_empty() {
206        write_string_array(out, "allow_fs_write", caps.allow_fs_write.clone());
207    }
208    if !caps.deny_fs_write.is_empty() {
209        write_string_array(out, "deny_fs_write", caps.deny_fs_write.clone());
210    }
211    if !caps.allow_exec.is_empty() {
212        write_string_array(out, "allow_exec", caps.allow_exec.clone());
213    }
214    if !caps.deny_exec.is_empty() {
215        write_string_array(out, "deny_exec", caps.deny_exec.clone());
216    }
217    if !caps.allow_net_http.is_empty() {
218        write_string_array(out, "allow_net_http", caps.allow_net_http.clone());
219    }
220    // Gated: only emitted when set, so pre-D7-B bundles hash identically.
221    if caps.any_net_http {
222        writeln!(out, "any_net_http = true").unwrap();
223    }
224    if !caps.deny_net_http.is_empty() {
225        write_string_array(out, "deny_net_http", caps.deny_net_http.clone());
226    }
227    if !caps.allow_agent_spawn.is_empty() {
228        write_string_array(out, "allow_agent_spawn", caps.allow_agent_spawn.clone());
229    }
230    if !caps.deny_agent_spawn.is_empty() {
231        write_string_array(out, "deny_agent_spawn", caps.deny_agent_spawn.clone());
232    }
233    if !caps.allow_skill_spawn.is_empty() {
234        write_string_array(out, "allow_skill_spawn", caps.allow_skill_spawn.clone());
235    }
236    if !caps.deny_skill_spawn.is_empty() {
237        write_string_array(out, "deny_skill_spawn", caps.deny_skill_spawn.clone());
238    }
239}

241fn write_ir_payload(out: &mut String, ir: &IrPayload) {
242    // Fields are already hex strings; emit in fixed order for deterministic hash.
243    write_str_kv(out, "ir_format", &ir.ir_format);
244    write_str_kv(out, "canonical_ir_hash", &ir.canonical_ir_hash);
245    write_str_kv(out, "canonical_ir_bytes_hex", &ir.canonical_ir_bytes_hex);
246}

248fn write_str_kv(out: &mut String, key: &str, value: &str) {
249    writeln!(out, "{} = {}", key, toml_string(value)).unwrap();
250}
251
252fn write_string_array(out: &mut String, key: &str, items: Vec<String>) {
253    out.push_str(key);
254    out.push_str(" = [");
255    for (i, s) in items.iter().enumerate() {
256        if i > 0 {
257            out.push_str(", ");
258        }
259        out.push_str(&toml_string(s));
260    }
261    out.push_str("]\n");
262}

264/// Emit a TOML basic-string literal (escapes per the TOML spec).
265fn toml_string(s: &str) -> String {
266    let mut out = String::with_capacity(s.len() + 2);
267    out.push('"');
268    for c in s.chars() {
269        match c {
270            '"' => out.push_str("\\\""),
271            '\\' => out.push_str("\\\\"),
272            '\n' => out.push_str("\\n"),
273            '\r' => out.push_str("\\r"),
274            '\t' => out.push_str("\\t"),
275            c if (c as u32) < 0x20 => {
276                write!(&mut out, "\\u{:04X}", c as u32).unwrap();
277            }
278            c => out.push(c),
279        }
280    }
281    out.push('"');
282    out
283}

285/// Emit a TOML bare key if `k` is ASCII alphanumeric/dash/underscore;
286/// otherwise quote it.
287fn toml_bare_key(k: &str) -> String {
288    if !k.is_empty()
289        && k.chars()
290            .all(|c| c.is_ascii_alphanumeric() || c == '-' || c == '_')
291    {
292        k.to_string()
293    } else {
294        toml_string(k)
295    }
296}

298fn capability_shape_to_str(s: &tau_domain::CapabilityShape) -> String {
299    match s {
300        tau_domain::CapabilityShape::FilesystemRead => "FilesystemRead".into(),
301        tau_domain::CapabilityShape::FilesystemWrite => "FilesystemWrite".into(),
302        tau_domain::CapabilityShape::ProcessExec => "ProcessExec".into(),
303        tau_domain::CapabilityShape::NetworkHttp => "NetworkHttp".into(),
304        tau_domain::CapabilityShape::AgentSpawn => "AgentSpawn".into(),
305        tau_domain::CapabilityShape::Custom { name } => format!("Custom({name})"),
306        // Forward-compat: new variants added in future tau-domain versions
307        // are serialized as their debug representation.
308        unknown => format!("{unknown:?}"),
309    }
310}

363        let h2 = compute_self_hash(&m2);
364        assert_ne!(
365            h1, h2,
366            "asset bytes must participate in the bundle self-hash"

406    #[test]
407    fn canonical_emits_all_bundle_meta_fields() {
408        // Guard against the serde/canonical drift landmine: every
409        // BundleMeta field that feeds the self-hash must appear in the
410        // canonical `[bundle]` table. selected_agents is the newest such
411        // field; assert it is emitted (after target) when Some.
412        let mut m = sample_manifest();
413        m.bundle.selected_agents = Some(vec!["alpha".to_string(), "beta".to_string()]);
414        let out = to_canonical_toml(&m);
415        // Appears in the [bundle] section, after target, before [project].
416        let bundle_start = out.find("[bundle]").expect("[bundle] header");
417        let project_start = out.find("[project]").expect("[project] header");
418        let sel = out
419            .find("selected_agents")
420            .expect("selected_agents emitted");
421        let target = out.find("target =").expect("target emitted");
422        assert!(
423            bundle_start < target && target < sel && sel < project_start,
424            "selected_agents must sit in [bundle] after target:\n{out}"
425        );
426        // And omitted entirely when None.
427        m.bundle.selected_agents = None;
428        let out_none = to_canonical_toml(&m);
429        assert!(
430            !out_none.contains("selected_agents"),
431            "selected_agents must be omitted when None:\n{out_none}"
432        );
433    }

448                backoff_strategy: "exponential".into(),
449                backoff_base: "30s".into(),
450                backoff_max: "10m".into(),
451                dead_letter: Some("dlq-sink".into()),
452            }),
453        }];
454        let out = to_canonical_toml(&m);
455        // Scope all searches to the [[trigger]] section to avoid false matches
456        // against `kind =` inside the `[[agents]]` backend inline-table.
457        let trigger_start = out.find("[[trigger]]").expect("[[trigger]] header");
458        let trigger_section = &out[trigger_start..];
459        let p_name = trigger_section.find("name =").expect("name");
460        let p_kind = trigger_section.find("kind =").expect("kind");
461        let p_agent = trigger_section.find("agent =").expect("agent");
462        let p_sched = trigger_section.find("schedule =").expect("schedule");
463        let p_tz = trigger_section.find("timezone =").expect("timezone");
464        let p_retry = trigger_section.find("retry =").expect("retry");
465        assert!(
466            p_name < p_kind
467                && p_kind < p_agent
468                && p_agent < p_sched
469                && p_sched < p_tz
470                && p_tz < p_retry,
471            "trigger fields out of canonical order:\n{out}"
472        );
473    }
