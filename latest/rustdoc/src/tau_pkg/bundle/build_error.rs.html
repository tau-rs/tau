1//! `BuildError` — typed failure type for the bundle build pipeline.
2
3use std::path::PathBuf;
4
5/// Errors returned by [`crate::bundle::build`](crate::bundle::build()).
6///
7/// Each variant maps to a CLI exit code via `tau-cli::cmd::build`:
8/// config/parse errors → 2, install-state errors → 3, internal/IO
9/// errors → 70.
10#[derive(Debug, thiserror::Error)]
11pub enum BuildError {
12    /// Failed to load or parse the project's `tau.toml`.
13    #[error("failed to load project tau.toml: {0}")]
14    ProjectConfig(String),
15
16    /// The project lockfile (`tau.lock`) is absent.
17    #[error("missing lockfile; run `tau install` first")]
18    MissingLockfile,
19
20    /// Lockfile present but could not be loaded or parsed.
21    #[error("failed to load lockfile: {0}")]
22    LockfileLoad(String),
23
24    /// A package recorded in the lockfile is missing from the local
25    /// install cache.
26    #[error("package `{name}` is locked but not installed at {path:?}; run `tau install`")]
27    PackageNotInstalled {
28        /// Logical name of the missing package.
29        name: String,
30        /// Expected install path on disk.
31        path: PathBuf,
32    },
33
34    /// Recomputing the tree hash of a package payload failed.
35    #[error("tree hash failed for `{name}`: {source}")]
36    TreeHashFailed {
37        /// Logical name of the package whose tree hash failed.
38        name: String,
39        /// Underlying tree-hash error.
40        #[source]
41        source: crate::tree_hash::TreeHashError,
42    },
43
44    /// Resolving an agent's system prompt (file or inline) failed.
45    #[error("system prompt resolution failed for agent `{id}`: {source}")]
46    PromptResolveFailed {
47        /// Agent id whose prompt failed to resolve.
48        id: String,
49        /// Underlying I/O error.
50        #[source]
51        source: std::io::Error,
52    },
53
54    /// Computing the effective capabilities for an agent failed.
55    #[error("capability override compute failed for agent `{id}`: {source}")]
56    CapabilityOverrideFailed {
57        /// Agent id whose capability override failed.
58        id: String,
59        /// Underlying override-expansion error.
60        #[source]
61        source: crate::capability_override::OverrideExpandError,
62    },
63
64    /// Assembled bundle manifest failed validation.
65    #[error("manifest assembly failed: {0}")]
66    ManifestInvalid(String),
67
68    /// An agent declares capability overrides but its home package (the
69    /// `[agents.<id>].package` ref) is absent from the resolved package
70    /// set — the effective grant set cannot be computed.
71    #[error("agent `{id}` has capability overrides but its home package `{package}` is not in the bundle's package set; run `tau install`")]
72    AgentHomePackageMissing {
73        /// Agent id with the dangling home-package reference.
74        id: String,
75        /// The unresolved home-package name.
76        package: String,
77    },
78
79    /// An agent's home-package manifest could not be read or parsed while
80    /// computing its effective capabilities.
81    #[error(
82        "failed to read home-package manifest for agent `{id}` (package `{package}`): {source}"
83    )]
84    AgentHomePackageManifest {
85        /// Agent id whose home-package manifest failed to load.
86        id: String,
87        /// The home-package name.
88        package: String,
89        /// Underlying manifest read/parse error.
90        #[source]
91        source: crate::error::ManifestReadError,
92    },
93
94    /// `--agent <id>` named an agent not present in the project config.
95    #[error("unknown agent `{id}`; available agents: {}", available.join(", "))]
96    UnknownAgent {
97        /// The requested agent id that does not exist.
98        id: String,
99        /// All agent ids declared in the project (sorted), for the hint.
100        available: Vec<String>,
101    },
102
103    /// Writing the bundle artifact to disk failed.
104    #[error("bundle write failed at {path:?}: {source}")]
105    WriteFailed {
106        /// Path the bundle was being written to.
107        path: PathBuf,
108        /// Underlying I/O error.
109        #[source]
110        source: std::io::Error,
111    },
112}
