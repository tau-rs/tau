3/// Errors raised when parsing bundle TOML.
4///
5/// # Example
6///
7/// ```
8/// use tau_pkg::bundle::error::BundleParseError;
9///
10/// let err = BundleParseError::UnsupportedSchemaVersion { found: 99 };
11/// let display = format!("{err}");
12/// assert!(display.contains("99"));
13/// assert!(display.contains("unsupported"));
14/// ```
15#[derive(Debug, thiserror::Error)]
16pub enum BundleParseError {
17    /// Underlying TOML syntax/schema error.
18    #[error("TOML parse error: {0}")]
19    Toml(#[from] toml::de::Error),
20    /// Bundle declares a `schema_version` this binary does not support.
21    #[error("unsupported schema_version {found}; this tau binary supports v1.x only")]
22    UnsupportedSchemaVersion {
23        /// The schema_version found in the manifest.
24        found: u32,
25    },
26    /// A bundle declares `[[trigger]]` entries but its `schema_version` is
27    /// below 3. Triggers require schema_version 3 (an old tau must reject a
28    /// trigger-bearing bundle rather than silently drop the bindings).
29    #[error("bundle declares triggers but schema_version is {schema_version} (triggers require schema_version >= 3)")]
30    TriggerSchemaVersionMismatch {
31        /// The bundle's declared schema_version.
32        schema_version: u32,
33    },
34    /// A bundle carries a `[governance]` record but its `schema_version` is
35    /// below 4. Governance requires schema_version 4 (an old tau must reject
36    /// a governance-bearing bundle rather than silently ignore the verdict).
37    #[error("bundle declares a [governance] record but schema_version is {schema_version} (governance requires schema_version >= 4)")]
38    GovernanceSchemaVersionMismatch {
39        /// The bundle's declared schema_version.
40        schema_version: u32,
41    },
42    /// A bundle carries an `[[assets]]` store but its `schema_version` is
43    /// below 5. The asset store (D6-B) requires schema_version 5 (an old tau
44    /// must reject the bundle rather than silently drop the prompt bytes the
45    /// IR references).
46    #[error("bundle declares an [[assets]] store but schema_version is {schema_version} (assets require schema_version >= 5)")]
47    AssetSchemaVersionMismatch {
48        /// The bundle's declared schema_version.
49        schema_version: u32,
50    },
51}

53/// Errors raised when reading + parsing a bundle from disk.
54#[derive(Debug, thiserror::Error)]
55pub enum BundleIoError {
56    /// Could not read the bundle file.
57    #[error("could not read bundle at {path}: {source}")]
58    Read {
59        /// Path attempted.
60        path: std::path::PathBuf,
61        /// Underlying IO error.
62        source: std::io::Error,
63    },
64    /// Parsing the bundle contents failed.
65    #[error(transparent)]
66    Parse(#[from] BundleParseError),
67}

69/// Errors raised when verifying a bundle's self-hash. Used by Task 3.
70///
71/// # Example
72///
73/// ```
74/// use tau_pkg::bundle::error::BundleIntegrityError;
75///
76/// let err = BundleIntegrityError::HashFieldEmpty;
77/// let display = format!("{err}");
78/// assert!(display.contains("empty"));
79///
80/// let err2 = BundleIntegrityError::HashMismatch {
81///     claimed: "aaaa".to_string(),
82///     computed: "bbbb".to_string(),
83/// };
84/// let display2 = format!("{err2}");
85/// assert!(display2.contains("mismatch"));
86/// ```
87#[derive(Debug, thiserror::Error)]
88pub enum BundleIntegrityError {
89    /// `bundle.sha256` does not match the recomputed canonical-TOML SHA-256.
90    #[error("bundle self-hash mismatch: claimed {claimed}, computed {computed}")]
91    HashMismatch {
92        /// Hash claimed by the bundle's `bundle.sha256` field.
93        claimed: String,
94        /// Hash computed from the bundle's canonical-TOML form.
95        computed: String,
96    },
97    /// `bundle.sha256` field is empty (zero-length string).
98    #[error("bundle.sha256 field is empty")]
99    HashFieldEmpty,
100}
