1//! Reusable capability subset relation (ADR-0057 decision 2) and the shared
2//! per-field comparison helpers used by both `compute_effective` (per-package
3//! narrowing) and `capability_set_subset` (lattice-link ceiling checks).
4//!
5//! D3 task 6: `capability_set_subset` and `CeilingViolation` now delegate to
6//! the sound `tau-domain` lattice primitive (`capability_subset`); the
7//! sampling-era per-kind comparison logic that used to live here has been
8//! deleted. `paths_subset` delegates to the sound G2 glob engine
9//! (`tau_domain::package::capability::lattice::glob::glob_subset_set`).
10//! `string_set_subset` and `max_bytes_le` remain — `compute_effective`
11//! (mod.rs) still calls them directly for per-package allow/max_bytes
12//! narrowing, which is a distinct relation from the lattice-link ceiling
13//! check above.

19/// Globbed path subset: every `child` path is a glob-subset of some `parent`
20/// path. `Err(offender)` names the first child path with no admitting parent.
21pub(crate) fn paths_subset(child: &[String], parent: &[String]) -> Result<(), String> {
22    glob_subset_set(child, parent)
23}
24
25/// Exact-set inclusion: every `child` entry equals some `parent` entry.
26/// `Err(offender)` names the first child entry not present in `parent`.
27pub(crate) fn string_set_subset(child: &[String], parent: &[String]) -> Result<(), String> {
28    for c in child {
29        if !parent.iter().any(|p| p == c) {
30            return Err(c.clone());
31        }
32    }
33    Ok(())
34}
35
36/// `max_bytes` tightening: `child <= parent`. `parent == None` means the
37/// ceiling is unlimited (any child is admitted). `Err` carries the child value.
38pub(crate) fn max_bytes_le(child: u64, parent: Option<u64>) -> Result<(), String> {
39    match parent {
40        None => Ok(()),
41        Some(max) if child <= max => Ok(()),
42        Some(_) => Err(format!("max_bytes={child}")),
43    }
44}

73    fn fs_write_max_bytes_higher_rejected() {
74        // D3: fs.write subset now requires JOINT per-grant coverage — a
75        // parent entry only "covers" the child if that SAME entry's
76        // max_bytes admits the child's max_bytes. Here the only parent
77        // entry (max_bytes=5000) does not cover the child's 9000, so it is
78        // excluded entirely and the child's path has zero eligible parent
79        // paths left to match against — the offender is now the path
80        // itself, not a "max_bytes=<n>" token.
81        let child = vec![cap(
82            r#"{"kind":"fs.write","paths":["/p/**"],"max_bytes":9000}"#,
83        )];
84        let parent = vec![cap(
85            r#"{"kind":"fs.write","paths":["/p/**"],"max_bytes":5000}"#,
86        )];
87        let v = capability_set_subset(&child, &parent).unwrap_err();
88        assert_eq!(v.kind, "fs.write");
89        assert_eq!(v.offender, "/p/**");
90        assert!(v.reason.contains("max_bytes"), "got reason: {}", v.reason);
91    }
