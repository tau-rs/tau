1//! Enumerate available versions at a [`PackageSource`].
2//!
3//! Used by the resolver (`crate::resolve`) to pick a concrete
4//! `Version` for a `(name, source, version_req)` triple. For
5//! `Git { rev: None }` we shell out to `git ls-remote --tags` and
6//! filter the tag list to those parsing as `semver::Version` (after
7//! stripping a leading `v` if present). For `Git { rev: Some(_) }`
8//! the source is a single point in version space — we shallow-clone
9//! the rev into a tempdir, read the manifest, and return that one
10//! version.
11//!
12//! See `docs/superpowers/specs/2026-04-30-transitive-deps-design.md` §5.4.
13

20use crate::manifest::read_manifest;

22/// List the versions available at `source`.
23pub fn list_versions_at_source(source: &PackageSource) -> Result<Vec<Version>, SourceListError> {
24    match source {
25        PackageSource::Git {
26            location,
27            rev: None,
28        } => list_git_tags(location),
29        PackageSource::Git {
30            location,
31            rev: Some(rev),
32        } => single_version_at_rev(location, rev),
33        // Unreachable today (Git is the only variant), but the enum is
34        // `#[non_exhaustive]`, so the catch-all is required for forward
35        // compatibility. Future variants land here as `Unsupported`
36        // until the resolver explicitly handles them.
37        _ => Err(SourceListError::Unsupported),
38    }
39}

41fn list_git_tags(location: &GitLocation) -> Result<Vec<Version>, SourceListError> {
42    let url = location.to_string();
43    let output = Command::new("git")
44        .arg("ls-remote")
45        .arg("--tags")
46        .arg(&url)
47        .output()
48        .map_err(|e| SourceListError::GitInvoke {
49            message: format!("spawning `git ls-remote`: {e}"),
50        })?;
51    if !output.status.success() {
52        return Err(SourceListError::GitLsRemote {
53            stderr: String::from_utf8_lossy(&output.stderr).into_owned(),
54        });
55    }
56    let stdout = String::from_utf8_lossy(&output.stdout);
57    Ok(parse_ls_remote_tags(&stdout))
58}
59
60/// Parse `git ls-remote --tags` stdout into a sorted `Vec<Version>`.
61///
62/// Format: each line is `<sha>\trefs/tags/<tag>` (or `refs/tags/<tag>^{}`
63/// for annotated-tag peels). We extract the tag, strip a leading `v`
64/// if present, parse as `semver::Version`, drop non-semver tags. The
65/// returned vec is sorted ascending; resolver picks the last entry
66/// satisfying constraints (= the highest).
67fn parse_ls_remote_tags(stdout: &str) -> Vec<Version> {
68    let mut versions: Vec<Version> = stdout
69        .lines()
70        .filter_map(|line| {
71            let tag = line.split('\t').nth(1)?;
72            // Strip refs/tags/ prefix.
73            let tag = tag.strip_prefix("refs/tags/")?;
74            // Drop the ^{} peel suffix (annotated tag commit pointer).
75            let tag = tag.strip_suffix("^{}").unwrap_or(tag);
76            // Strip leading `v`.
77            let tag = tag.strip_prefix('v').unwrap_or(tag);
78            Version::parse(tag).ok()
79        })
80        .collect();
81    versions.sort();
82    versions.dedup();
83    versions
84}

86fn single_version_at_rev(
87    location: &GitLocation,
88    rev: &str,
89) -> Result<Vec<Version>, SourceListError> {
90    let tempdir = TempDir::new().map_err(|e| SourceListError::TempDir {
91        message: format!("creating tempdir for shallow clone: {e}"),
92    })?;
93    let dest = tempdir.path().join("repo");
94    let url = location.to_string();
95    let output = Command::new("git")
96        .arg("clone")
97        .arg("--depth")
98        .arg("1")
99        .arg("--branch")
100        .arg(rev)
101        .arg("--single-branch")
102        .arg(&url)
103        .arg(&dest)
104        .output()
105        .map_err(|e| SourceListError::GitInvoke {
106            message: format!("spawning `git clone`: {e}"),
107        })?;
108    if !output.status.success() {
109        return Err(SourceListError::GitClone {
110            rev: rev.to_owned(),
111            stderr: String::from_utf8_lossy(&output.stderr).into_owned(),
112        });
113    }
114    let manifest_path = dest.join("tau.toml");
115    let manifest = read_manifest(&manifest_path).map_err(SourceListError::Manifest)?;
116    Ok(vec![manifest.version().clone()])
117}

119/// Errors produced by [`list_versions_at_source`].
120///
121/// # Example
122///
123/// ```
124/// use tau_pkg::source_list::SourceListError;
125///
126/// let err = SourceListError::GitInvoke {
127///     message: "spawning `git ls-remote`: No such file or directory".to_string(),
128/// };
129/// let display = format!("{err}");
130/// assert!(display.contains("git invocation failed"));
131///
132/// let err2 = SourceListError::Unsupported;
133/// let display2 = format!("{err2}");
134/// assert!(display2.contains("not supported"));
135/// ```
136#[non_exhaustive]
137#[derive(Debug, thiserror::Error)]
138pub enum SourceListError {
139    /// `git` binary not invocable.
140    #[error("git invocation failed: {message}")]
141    GitInvoke {
142        /// Human-readable failure context.
143        message: String,
144    },
145    /// `git ls-remote` exited non-zero.
146    #[error("git ls-remote failed: {stderr}")]
147    GitLsRemote {
148        /// Captured stderr.
149        stderr: String,
150    },
151    /// `git clone` for a `rev`-pinned source failed.
152    #[error("git clone of {rev:?} failed: {stderr}")]
153    GitClone {
154        /// The rev that was passed to `--branch`.
155        rev: String,
156        /// Captured stderr.
157        stderr: String,
158    },
159    /// Could not create a tempdir for the shallow clone.
160    #[error("tempdir creation failed: {message}")]
161    TempDir {
162        /// Human-readable failure context.
163        message: String,
164    },
165    /// Reading the manifest at the cloned rev failed.
166    #[error("manifest read failed: {0}")]
167    Manifest(#[from] crate::error::ManifestReadError),
168    /// `PackageSource` variant not supported by the resolver. Reserved
169    /// for future variants — `Git` is the only variant at v0.1.
170    #[error("source kind not supported by resolver")]
171    Unsupported,
172}

179    /// Build a stdout fixture mimicking `git ls-remote --tags` output.
180    /// Each line: `<40-char-sha>\trefs/tags/<tag>`.
181    fn fake_ls_remote(tags: &[&str]) -> String {
182        tags.iter()
183            .map(|t| format!("0123456789abcdef0123456789abcdef01234567\trefs/tags/{t}"))
184            .collect::<Vec<_>>()
185            .join("\n")
186    }

188    #[test]
189    fn parse_ls_remote_tags_returns_only_semver_parsable_tags() {
190        let stdout = fake_ls_remote(&["v0.1.0", "v0.2.0", "release-2024", "main"]);
191        let versions = parse_ls_remote_tags(&stdout);
192        assert_eq!(
193            versions,
194            vec![
195                Version::parse("0.1.0").unwrap(),
196                Version::parse("0.2.0").unwrap(),
197            ]
198        );

244    /// Set up a local git repository in a tempdir, with a tau.toml
245    /// declaring `name = "test-tool"` and `version = "<version>"`,
246    /// then commit + tag `v<version>`. Returns the tempdir guard +
247    /// the GitLocation of the repo.

249        let tempdir = TempDir::new().unwrap();
250        let repo = tempdir.path().join("test-tool");
251        std::fs::create_dir(&repo).unwrap();
252        let manifest_body = format!(
253            r#"
254name = "test-tool"
255version = "{version}"
256description = "fixture"
257authors = []
258source = "https://example.com/test.git"
259kind = "tool"
260dependencies = []
261capabilities = []
262"#
263        );
264        std::fs::write(repo.join("tau.toml"), manifest_body).unwrap();
265
266        let run = |args: &[&str]| {
267            let out = Command::new("git")
268                .current_dir(&repo)
269                .args(args)
270                .output()
271                .unwrap();
272            assert!(
273                out.status.success(),
274                "git {args:?} failed: {}",
275                String::from_utf8_lossy(&out.stderr)
276            );
277        };
278        run(&["init", "-q", "-b", "main"]);
279        run(&["config", "user.email", "test@example.com"]);
280        run(&["config", "user.name", "Test"]);
281        run(&["add", "."]);
282        run(&["commit", "-q", "-m", "fixture"]);
283        run(&["tag", &format!("v{version}")]);
284
285        let url = format!("file://{}", repo.display());
286        let location = GitLocation::from_str(&url).unwrap();
287        (tempdir, location)
288    }
