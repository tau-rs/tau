1//! Read and structurally validate `tau.toml` manifests from disk.
2//!
3//! [`read_manifest`] is a thin wrapper:
4//!
5//! 1. `fs::read_to_string(path)` — surfaces I/O errors as
6//!    [`ManifestReadError::Io`] or [`ManifestReadError::NotFound`].
7//! 2. `toml::from_str::<UncheckedManifest>(&text)` — surfaces parse
8//!    failures as [`ManifestReadError::Parse`].
9//! 3. [`tau_domain::UncheckedManifest::validate`] — surfaces structural
10//!    failures as [`ManifestReadError::Validation`] via `#[from]`.
11//!
12//! Used by the install lifecycle (Task 10) to read `tau.toml` from a
13//! freshly-cloned package source tree.
14
15use std::fs;
16use std::io::ErrorKind;
17use std::path::Path;
18
19use tau_domain::{PackageManifest, UncheckedManifest};
20
21use crate::error::ManifestReadError;

23/// Read and validate a manifest from disk.
24///
25/// `path` is the manifest file itself (typically
26/// `<package_root>/tau.toml`), not the package root directory.
27///
28/// # Errors
29///
30/// - [`ManifestReadError::NotFound`] — file does not exist.
31/// - [`ManifestReadError::Io`] — file exists but could not be read
32///   (permissions, broken symlink, etc.).
33/// - [`ManifestReadError::Parse`] — file is not valid TOML or doesn't
34///   match the [`UncheckedManifest`] schema.
35/// - [`ManifestReadError::Validation`] — TOML parsed but structural
36///   validation rejected the manifest (invalid name, capability,
37///   etc.).
38///
39/// # Example
40///
41/// ```
42/// use tau_pkg::read_manifest;
43///
44/// # let tmp = tempfile::tempdir().unwrap();
45/// # let path = tmp.path().join("tau.toml");
46/// # std::fs::write(&path, concat!(
47/// #     "name = \"acme-tool\"\n",
48/// #     "version = \"1.0.0\"\n",
49/// #     "description = \"A tool\"\n",
50/// #     "authors = [\"Acme <acme@example.com>\"]\n",
51/// #     "source = \"https://example.com/acme/tool.git\"\n",
52/// #     "kind = \"tool\"\n",
53/// #     "dependencies = []\n",
54/// #     "capabilities = []\n",
55/// # )).unwrap();
56/// let manifest = read_manifest(&path).unwrap();
57/// assert_eq!(manifest.name().as_str(), "acme-tool");
58/// ```
59pub fn read_manifest(path: &Path) -> Result<PackageManifest, ManifestReadError> {
60    let text = fs::read_to_string(path).map_err(|e| {
61        if e.kind() == ErrorKind::NotFound {
62            ManifestReadError::NotFound {
63                path: path.display().to_string(),
64            }
65        } else {
66            ManifestReadError::Io {
67                message: format!("reading manifest {}: {e}", path.display()),
68            }
69        }
70    })?;
71
72    let unchecked: UncheckedManifest =
73        toml::from_str(&text).map_err(|e| ManifestReadError::Parse {
74            reason: format!("{}: {e}", path.display()),
75        })?;
76
77    let manifest = unchecked.validate()?;
78    Ok(manifest)
79}

87    fn write_manifest(dir: &TempDir, contents: &str) -> std::path::PathBuf {

93    /// Minimal valid manifest using the natural TOML form (per ADR-0005).
94    /// `PackageSource` serializes via its `Display`/`FromStr` string form,
95    /// `PackageKind::Custom` as the inner `kind` string.
96    fn minimal_valid_manifest() -> &'static str {
97        r#"
98name = "acme-tool"
99version = "1.0.0"
100description = "A tool for testing"
101authors = ["Acme <support@acme.dev>"]
102source = "https://example.com/acme/tool.git"
103kind = "tool"
104dependencies = []
105capabilities = []
106"#
107    }

109    #[test]
110    fn read_manifest_returns_not_found_for_missing_file() {
111        let tmp = TempDir::new().unwrap();
112        let path = tmp.path().join("does-not-exist.toml");
113        let err = read_manifest(&path).unwrap_err();
114        assert!(matches!(err, ManifestReadError::NotFound { .. }));
115    }
116
117    #[test]
118    fn read_manifest_returns_parse_for_bad_toml() {
119        let tmp = TempDir::new().unwrap();
120        let path = write_manifest(&tmp, "this is not toml = = =");
121        let err = read_manifest(&path).unwrap_err();
122        assert!(matches!(err, ManifestReadError::Parse { .. }));
123    }

125    #[test]
126    fn read_manifest_returns_error_for_invalid_name() {
127        let tmp = TempDir::new().unwrap();
128        // PackageName rejects names with spaces/uppercase at deserialisation
129        // time, so this surfaces as Parse (serde rejects the field value).
130        // Either Parse or Validation is acceptable — we just want a typed
131        // error, not a panic.
132        let bad = r#"
133name = "INVALID NAME WITH SPACES"
134version = "1.0.0"
135description = "bad"
136authors = []
137source = "https://example.com/x.git"
138kind = "tool"
139dependencies = []
140capabilities = []
141"#;
142        let path = write_manifest(&tmp, bad);
143        let err = read_manifest(&path).unwrap_err();
144        match err {
145            ManifestReadError::Parse { .. } => {}
146            ManifestReadError::Validation(_) => {}
147            other => panic!("expected Parse or Validation, got {other:?}"),
148        }
149    }
150
151    #[test]
152    fn read_manifest_succeeds_for_minimal_valid_manifest() {
153        let tmp = TempDir::new().unwrap();
154        let path = write_manifest(&tmp, minimal_valid_manifest());
155        let manifest = read_manifest(&path).unwrap();
156        assert_eq!(manifest.name().as_str(), "acme-tool");
157        assert_eq!(manifest.version().to_string(), "1.0.0");
158        assert_eq!(manifest.description(), "A tool for testing");
159    }

161    /// Manifest with a `[plugin]` table parses, validates, and surfaces
162    /// the typed plugin manifest via `PackageManifest::plugin()`.
163    #[test]
164    fn read_manifest_extracts_plugin_table() {
165        let tmp = TempDir::new().unwrap();
166        let toml_text = r#"
167name = "echo-llm"
168version = "0.1.0"
169description = "Toy LlmBackend plugin"
170authors = ["Acme <support@acme.dev>"]
171source = "https://example.com/echo-llm.git"
172kind = "llm-backend"
173dependencies = []
174capabilities = []
175
176[plugin]
177provides = "llm_backend"
178kind     = "rust-cargo"
179bin      = "echo-llm"
180"#;
181        let path = write_manifest(&tmp, toml_text);
182        let manifest = read_manifest(&path).unwrap();
183        let plugin = manifest
184            .plugin()
185            .expect("manifest should expose [plugin] table");
186        assert_eq!(plugin.provides, tau_domain::PortKind::LlmBackend);
187        assert_eq!(plugin.kind, tau_domain::PluginKind::RustCargo);
188        assert_eq!(plugin.bin, "echo-llm");
189    }

191    /// Data-only packages (no `[plugin]` table) round-trip with
192    /// `plugin == None`. This is the existing behaviour, preserved
193    /// for backward compatibility.
194    #[test]
195    fn read_manifest_without_plugin_table_returns_none() {
196        let tmp = TempDir::new().unwrap();
197        let path = write_manifest(&tmp, minimal_valid_manifest());
198        let manifest = read_manifest(&path).unwrap();
199        assert!(
200            manifest.plugin().is_none(),
201            "data-only manifest should have plugin == None"
202        );
203    }

205    /// Unknown plugin `kind` strings (e.g., `python-pip` before that
206    /// variant lands) surface as a `Parse` error from the typed
207    /// `PluginKind` deserializer.
208    #[test]
209    fn read_manifest_invalid_plugin_kind_errors() {
210        let tmp = TempDir::new().unwrap();
211        let bad = r#"
212name = "bad-plugin"
213version = "0.1.0"
214description = "Plugin with unknown kind"
215authors = []
216source = "https://example.com/bad.git"
217kind = "llm-backend"
218dependencies = []
219capabilities = []
220
221[plugin]
222provides = "llm_backend"
223kind     = "python-pip"
224bin      = "bad-plugin"
225"#;
226        let path = write_manifest(&tmp, bad);
227        let result = read_manifest(&path);
228        assert!(
229            matches!(result, Err(ManifestReadError::Parse { .. })),
230            "expected Parse error for unknown plugin kind, got {result:?}",
