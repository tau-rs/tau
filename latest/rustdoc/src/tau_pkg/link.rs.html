1//! Static link phase: validate the installed world the IR binds to.
2//!
3//! `link()` is a *static linker* — it checks that every plugin, model
4//! backend, and skill the project references resolves against the
5//! installed set (lockfile + manifests + on-disk `SKILL.md`), with **no
6//! process spawning**. It produces a serializable [`LinkRecord`] embedded
7//! in the bundle (PR 2) and trusted by `run --bundle` after verification
8//! (PR 3). Tool-name binding and sandbox adapter probing are the runtime
9//! *loader*'s job and are out of scope here.
10//!
11//! This module defines the linked-world types ([`LinkedPlugin`],
12//! [`LinkedSkill`], [`LinkRecord`], [`LinkOutcome`]), their error type
13//! ([`LinkError`]), and the [`link()`](link) entry point that produces a
14//! [`LinkOutcome`] from a `ProjectConfig` + `IrModule` + `LockFile` +
15//! `Scope`.
16

31/// A plugin package resolved against the installed set during linking.
32#[non_exhaustive]
33#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
34pub struct LinkedPlugin {
35    /// Validated package name.
36    pub name: PackageName,
37    /// The installed version selected (highest satisfying the requirement).
38    pub version: Version,
39    /// SHA-256 (hex) of the built plugin binary, copied from the lockfile.
40    pub binary_sha256: String,
41    /// The port this plugin provides (`LlmBackend` or `Tool`).
42    pub provides: PortKind,
43}

45/// An installed skill package resolved + parsed during linking.
46#[non_exhaustive]
47#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
48pub struct LinkedSkill {
49    /// Validated skill package name.
50    pub name: PackageName,
51    /// SHA-256 (hex) of the `SKILL.md` bytes, for drift detection at verify.
52    pub content_sha256: String,
53    /// Whether `SKILL.md` parsed successfully at link time.
54    pub parsed_ok: bool,
55}

57/// The verified binding record produced by `link()`. Embedded in the
58/// bundle manifest; the parsed skill bodies travel separately in
59/// [`LinkOutcome::parsed_skills`], not in the serialized record.
60#[non_exhaustive]
61#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
62pub struct LinkRecord {
63    /// Every plugin (model backend + required tool) resolved, deterministically ordered by name.
64    pub resolved_plugins: Vec<LinkedPlugin>,
65    /// Every installed skill package resolved + parsed, ordered by name.
66    pub resolved_skills: Vec<LinkedSkill>,
67    /// Final model-alias → resolved [`ModelRef`] bindings; no later re-resolution.
68    pub model_bindings: BTreeMap<String, ModelRef>,
69    /// The IR target the bundle was linked for (run --bundle platform check).
70    pub platform: TargetTriple,
71    /// SHA-256 (hex) of the lockfile bytes the record was linked against.
72    pub lockfile_sha256: String,
73}

75/// The full result of `link()`: the serializable [`LinkRecord`] plus the
76/// parsed skill bodies (kept out of the record; used to seed the runtime
77/// `SkillResolver` — PR 2).
78#[non_exhaustive]
79#[derive(Debug, Clone)]
80pub struct LinkOutcome {
81    /// The verified, serializable record.
82    pub record: LinkRecord,
83    /// skill name → parsed `SKILL.md`, produced once at link time.
84    pub parsed_skills: BTreeMap<String, tau_domain::SkillContent>,
85}

87/// Errors from static linking. All variants are collected — `link()`
88/// never stops at the first — so the operator sees the whole set.
89#[non_exhaustive]
90#[derive(Debug, Clone, PartialEq, Eq, Error)]
91pub enum LinkError {
92    /// A referenced package is not installed in the scope.
93    #[error("plugin {package} not installed in scope")]
94    PluginNotInstalled {
95        /// The referenced package name.
96        package: String,
97    },
98    /// A referenced package is installed but provides the wrong port.
99    #[error("plugin {package} provides {found:?} but {expected:?} was required")]
100    PluginPortMismatch {
101        /// The referenced package name.
102        package: String,
103        /// The port the plugin actually declares.
104        found: PortKind,
105        /// The port the reference required.
106        expected: PortKind,
107    },
108    /// No installed version of a package satisfies the requirement.
109    #[error("no installed version of {package} satisfies {req}")]
110    VersionUnsatisfied {
111        /// The referenced package name.
112        package: String,
113        /// The semver requirement string.
114        req: String,
115    },
116    /// An installed skill package's `SKILL.md` is missing on disk.
117    #[error("skill {package}: SKILL.md missing at {path}")]
118    SkillMissing {
119        /// The skill package name.
120        package: String,
121        /// The expected `SKILL.md` path (lossy UTF-8).
122        path: String,
123    },
124    /// A skill package's `SKILL.md` failed to parse, or its `tau.toml`/
125    /// manifest (or the on-disk lockfile) couldn't be read/parsed;
126    /// `detail` names the specific cause.
127    #[error("skill {package}: SKILL.md parse failed: {detail}")]
128    SkillParse {
129        /// The skill package name.
130        package: String,
131        /// Parser detail.
132        detail: String,
133    },
134}

136/// Resolve one plugin reference against the lockfile: installed, provides
137/// the expected port, and has an installed version satisfying `version_req`.
138///
139/// Ports the installed/port checks from `tau-cli`'s `plugin_loader.rs`
140/// (`resolve_plugin`) and the version-satisfiability search from
141/// `tau-pkg`'s `project/agent.rs` (`build_agent_definition`) into a single
142/// pure helper returning a structured [`LinkError`] instead of `anyhow`.
143///
144/// Called from [`link()`] for every model backend and required tool.
145fn resolve_one_plugin(
146    lockfile: &crate::lockfile::LockFile,
147    package: &PackageName,
148    version_req: &semver::VersionReq,
149    expected: PortKind,
150) -> Result<LinkedPlugin, LinkError> {
151    let pkg = lockfile
152        .find(package)
153        .ok_or_else(|| LinkError::PluginNotInstalled {
154            package: package.to_string(),
155        })?;
156
157    // Must carry a [plugin] table and provide the required port
158    // (plugin_loader.rs shape). A data-only package (no [plugin] table)
159    // can't bind to a plugin reference, so it's PluginNotInstalled too.
160    // Checked before version satisfiability: structural validity (does
161    // this package even back a plugin?) takes precedence over "which
162    // version" — a data-only package is PluginNotInstalled regardless of
163    // what versions happen to be installed.
164    let plugin = pkg
165        .plugin
166        .as_ref()
167        .ok_or_else(|| LinkError::PluginNotInstalled {
168            package: package.to_string(),
169        })?;
170    if plugin.manifest.provides != expected {
171        return Err(LinkError::PluginPortMismatch {
172            package: package.to_string(),
173            found: plugin.manifest.provides,
174            expected,
175        });
176    }
177
178    // Highest installed version satisfying the requirement (agent.rs shape).
179    let version = pkg
180        .installed_versions
181        .iter()
182        .map(|v| &v.version)
183        .filter(|v| version_req.matches(v))
184        .max()
185        .cloned()
186        .ok_or_else(|| LinkError::VersionUnsatisfied {
187            package: package.to_string(),
188            req: version_req.to_string(),
189        })?;
190
191    Ok(LinkedPlugin {
192        name: package.clone(),
193        version,
194        binary_sha256: plugin.binary_sha256.clone(),
195        provides: plugin.manifest.provides,
196    })
197}

199/// Resolve every agent's `model` alias against `[models]`, producing the
200/// alias → [`ModelRef`] bindings [`LinkRecord::model_bindings`] carries.
201/// Agents with an empty `model` (no model declared) are skipped.
202///
203/// Infallible: `ProjectConfig::validate` already rejects an agent whose
204/// `model` is empty (`ProjectConfigError::MissingAgentModel`) or absent
205/// from `[models]` (`ProjectConfigError::UnknownModelAlias`) at parse
206/// time, so a `ProjectConfig` carrying either defect can't be
207/// constructed through the public validation path — every alias this
208/// function looks up is guaranteed present.
209///
210/// Thin wrapper over [`resolve_models_from`], which exists so this
211/// function's logic stays directly unit-testable regardless.
212///
213/// Called from [`link()`] to populate [`LinkRecord::model_bindings`].
214fn resolve_models(cfg: &ProjectConfig) -> BTreeMap<String, ModelRef> {
215    resolve_models_from(&cfg.agents, &cfg.models)
216}
217
218/// Core logic for [`resolve_models`], taking the agent and model maps
219/// directly. See [`resolve_models`] for why this split exists.
220fn resolve_models_from(
221    agents: &BTreeMap<String, AgentEntry>,
222    models: &BTreeMap<String, ModelEntry>,
223) -> BTreeMap<String, ModelRef> {
224    let mut bindings = BTreeMap::new();
225    for agent in agents.values() {
226        if agent.model.is_empty() {
227            continue; // agent declares no model
228        }
229        if let Some(ModelEntry { backend, model }) = models.get(&agent.model) {
230            bindings.insert(
231                agent.model.clone(),
232                ModelRef {
233                    backend: backend.clone(),
234                    model_id: model.clone(),
235                },
236            );
237        }
238        // An absent alias is unreachable on validated input
239        // (ProjectConfig::validate rejects it) — no error, no panic.
240    }
241    bindings
242}

244/// Resolve every installed skill package against the on-disk `SKILL.md`,
245/// parsing it exactly once and recording the outcome.
246///
247/// Enumerates lockfile entries in the *passed* `lockfile` where
248/// `pkg.skill.is_some()` (the same filter [`crate::find_installed_skill`]
249/// uses internally), locates each package's install path via that
250/// helper, reads + hashes `SKILL.md`, and parses it via
251/// [`tau_domain::parse_skill_md`]. Every skill produces exactly one
252/// [`LinkedSkill`] entry (with `parsed_ok` set accordingly), except for
253/// two classes of resolution failure that short-circuit before parsing:
254/// a missing install artifact — no `tau.toml` on disk
255/// (`FindSkillError::InstallPathMissing`), or — because
256/// `find_installed_skill` re-resolves against the on-disk lockfile via
257/// `scope` rather than trusting the passed `lockfile` — a lockfile/disk
258/// divergence where the on-disk lockfile no longer lists this package as
259/// a skill (`Ok(None)`) — surfaces as [`LinkError::SkillMissing`]; a
260/// present-but-unparseable `SKILL.md`, or a malformed `tau.toml`/
261/// manifest/lockfile (any other `find_installed_skill` error), surfaces
262/// as [`LinkError::SkillParse`] with the reason in `detail`. All errors
263/// are collected — this never stops at the first skill that fails, and a
264/// skill declared in `lockfile` is never silently dropped from the
265/// result.
266///
267/// Called from [`link()`] to populate [`LinkRecord::resolved_skills`] and
268/// [`LinkOutcome::parsed_skills`].
269fn resolve_skills(
270    scope: &crate::scope::Scope,
271    lockfile: &crate::lockfile::LockFile,
272) -> (
273    Vec<LinkedSkill>,
274    BTreeMap<String, tau_domain::SkillContent>,
275    Vec<LinkError>,
276) {
277    let mut linked = Vec::new();
278    let mut parsed_map = BTreeMap::new();
279    let mut errors = Vec::new();
280
281    for pkg in &lockfile.packages {
282        if pkg.skill.is_none() {
283            continue; // not a skill package
284        }
285        let name = pkg.name.clone();
286        // Locate the install path + SKILL.md filename via the existing helper.
287        let installed = match crate::find_installed_skill(scope, name.as_str()) {
288            Ok(Some(s)) => s,
289            Ok(None) => {
290                // The passed `lockfile` lists this package as a skill, but
291                // `find_installed_skill` re-resolves against the on-disk
292                // lockfile via `scope` and found no matching skill entry
293                // there (lockfile/disk divergence). That's a missing skill
294                // install, not "no skills" — record it instead of silently
295                // dropping it, or `resolve_skills` could return an empty
296                // result indistinguishable from "no skills declared".
297                let path = scope.package_dir(&name, &pkg.active_version);
298                errors.push(LinkError::SkillMissing {
299                    package: name.to_string(),
300                    path: path.display().to_string(),
301                });
302                continue;
303            }
304            Err(crate::FindSkillError::InstallPathMissing { path, .. }) => {
305                errors.push(LinkError::SkillMissing {
306                    package: name.to_string(),
307                    path: path.display().to_string(),
308                });
309                continue;
310            }
311            Err(e) => {
312                errors.push(LinkError::SkillParse {
313                    package: name.to_string(),
314                    detail: e.to_string(),
315                });
316                continue;
317            }
318        };
319        let skill_md = installed.install_path.join(&installed.skill.content);
320        let bytes = match std::fs::read(&skill_md) {
321            Ok(b) => b,
322            Err(_) => {
323                errors.push(LinkError::SkillMissing {
324                    package: name.to_string(),
325                    path: skill_md.display().to_string(),
326                });
327                continue;
328            }
329        };
330        let content_sha256 = sha256_hex(&bytes);
331        let text = String::from_utf8_lossy(&bytes);
332        match tau_domain::parse_skill_md(&text) {
333            Ok(content) => {
334                linked.push(LinkedSkill {
335                    name: name.clone(),
336                    content_sha256,
337                    parsed_ok: true,
338                });
339                parsed_map.insert(name.to_string(), content);
340            }
341            Err(e) => {
342                linked.push(LinkedSkill {
343                    name: name.clone(),
344                    content_sha256,
345                    parsed_ok: false,
346                });
347                errors.push(LinkError::SkillParse {
348                    package: name.to_string(),
349                    detail: e.to_string(),
350                });
351            }
352        }
353    }
354    (linked, parsed_map, errors)
355}

357/// Statically link the project against the installed set. Validates every
358/// referenced plugin, model backend, and skill; collects **all** errors
359/// (never stops at the first) so the operator sees the whole set at once.
360///
361/// On success returns a [`LinkOutcome`] carrying the serializable
362/// [`LinkRecord`] plus the parsed skill bodies (for seeding the runtime
363/// `SkillResolver`, PR 2). Does not spawn processes or probe sandbox
364/// adapters — that's the runtime *loader*'s job.
365///
366/// # Precondition
367///
368/// `lockfile` MUST be the same lockfile persisted at
369/// `scope.lockfile_path()`. Plugin resolution ([`resolve_one_plugin`])
370/// reads only the passed `lockfile`, but skill resolution
371/// ([`resolve_skills`]) and [`LinkRecord::lockfile_sha256`] both read
372/// straight from disk via `scope`. Passing a `lockfile` that has drifted
373/// from disk is undefined territory for skills specifically: a skill
374/// package listed in the passed `lockfile` but absent from the on-disk
375/// lockfile surfaces as [`LinkError::SkillMissing`] rather than being
376/// silently skipped (see [`resolve_skills`]'s doc comment). Callers (the
377/// `build`/`dev` entry points in PR 2) always pass the lockfile they just
378/// loaded from `scope.lockfile_path()`, so this invariant holds in
379/// practice.
380pub fn link(
381    cfg: &ProjectConfig,
382    module: &IrModule,
383    lockfile: &crate::lockfile::LockFile,
384    scope: &crate::scope::Scope,
385) -> Result<LinkOutcome, Vec<LinkError>> {
386    let mut errors = Vec::new();
387
388    // 1. Gather (package, version_req, expected_port) for every referenced
389    //    plugin: model backends (any version) and each agent's required
390    //    tools. Keyed by the package's string form — `PackageName` has no
391    //    `Ord`/`PartialOrd` impl, so it can't itself be a `BTreeMap` key.
392    let mut wanted: BTreeMap<String, (PackageName, semver::VersionReq, PortKind)> = BTreeMap::new();
393    for m in cfg.models.values() {
394        // validate() already guarantees every [models] backend parses as a
395        // package name; skip unparseable defensively rather than panic.
396        if let Ok(name) = PackageName::from_str(&m.backend) {
397            wanted.entry(name.to_string()).or_insert((
398                name,
399                semver::VersionReq::STAR,
400                PortKind::LlmBackend,
401            ));
402        }
403    }
404    for agent in cfg.agents.values() {
405        for tool in &agent.requires.tools {
406            wanted.entry(tool.name.to_string()).or_insert((
407                tool.name.clone(),
408                tool.version_req.clone(),
409                PortKind::Tool,
410            ));
411        }
412    }
413
414    // 2. Resolve every wanted plugin, collecting errors; keep successes
415    //    ordered by name (BTreeMap iteration is already name-ordered).
416    let mut resolved_plugins = Vec::new();
417    for (name, req, port) in wanted.values() {
418        match resolve_one_plugin(lockfile, name, req, *port) {
419            Ok(p) => resolved_plugins.push(p),
420            Err(e) => errors.push(e),
421        }
422    }
423    resolved_plugins.sort_by_key(|a| a.name.to_string());
424
425    // 3. Models — infallible; ProjectConfig::validate already guarantees
426    //    every agent's alias resolves.
427    let model_bindings = resolve_models(cfg);
428
429    // 4. Skills.
430    let (mut resolved_skills, parsed_skills, skill_errs) = resolve_skills(scope, lockfile);
431    errors.extend(skill_errs);
432    resolved_skills.sort_by_key(|a| a.name.to_string());
433
434    if !errors.is_empty() {
435        errors.sort_by_key(link_error_sort_key); // deterministic, no-drift bar
436        return Err(errors);
437    }
438
439    // lockfile_sha256 hashes the ON-DISK lockfile, not the passed one — see
440    // the precondition doc above. `run --bundle` re-derives this same hash
441    // from disk to detect drift.
442    let lockfile_sha256 = sha256_hex(&std::fs::read(scope.lockfile_path()).unwrap_or_default());
443
444    Ok(LinkOutcome {
445        record: LinkRecord {
446            resolved_plugins,
447            resolved_skills,
448            model_bindings,
449            platform: module.target,
450            lockfile_sha256,
451        },
452        parsed_skills,
453    })
454}

456/// Stable sort key for deterministic [`LinkError`] ordering across callers.
457fn link_error_sort_key(e: &LinkError) -> (u8, String) {
458    match e {
459        LinkError::PluginNotInstalled { package } => (0, package.clone()),
460        LinkError::PluginPortMismatch { package, .. } => (1, package.clone()),
461        LinkError::VersionUnsatisfied { package, .. } => (2, package.clone()),
462        LinkError::SkillMissing { package, .. } => (3, package.clone()),
463        LinkError::SkillParse { package, .. } => (4, package.clone()),
464    }
465}

467/// SHA-256 hex of `bytes`. `tree_hash::to_hex_lower` handles the
468/// digest-to-hex step; no crate-wide `sha256_hex(bytes) -> String`
469/// one-shot helper exists yet (`tree_hash::sha256_of_file` takes a
470/// path, not bytes), so this is the local equivalent.
471fn sha256_hex(bytes: &[u8]) -> String {
472    use sha2::{Digest, Sha256};
473    let mut hasher = Sha256::new();
474    hasher.update(bytes);
475    crate::tree_hash::to_hex_lower(&hasher.finalize())
476}
