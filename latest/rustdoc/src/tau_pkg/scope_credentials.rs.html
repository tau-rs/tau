1//! `[credentials]` chain configuration (β.5), stored in scope/home
2//! `config.toml`. Deployment-specific: the same bundle resolves
3//! credentials from env locally, files in k8s, or (later) Vault in prod.
4//!
5//! Unchecked→validate discipline mirrors `project::project`.
6
7use std::collections::BTreeMap;
8
9use serde::{Deserialize, Serialize};
10
11/// Unchecked `[credentials]` block.
12#[derive(Debug, Clone, Default, PartialEq, Eq, Deserialize, Serialize)]
13pub struct UncheckedCredentialsConfig {
14    /// Provider names in precedence order. Empty → implicit `["env"]`.
15    #[serde(default)]
16    pub chain: Vec<String>,
17    /// Provider definitions keyed by name.
18    #[serde(default)]
19    pub providers: BTreeMap<String, UncheckedProvider>,
20}
21
22/// Unchecked provider definition (`[credentials.providers.<name>]`).
23#[derive(Debug, Clone, PartialEq, Eq, Deserialize, Serialize)]
24pub struct UncheckedProvider {
25    /// Provider kind: `"env"` or `"file"`.
26    #[serde(rename = "type")]
27    pub kind: String,
28    /// `file`: secrets directory.
29    #[serde(default)]
30    pub dir: Option<String>,
31    /// `file`: credential-id → filename map.
32    #[serde(default)]
33    pub key_map: BTreeMap<String, String>,
34}
35
36/// Validated chain configuration.
37#[derive(Debug, Clone, PartialEq, Eq)]
38pub struct CredentialsChainConfig {
39    /// Ordered, validated providers.
40    pub chain: Vec<ProviderConfig>,
41}
42
43/// A validated provider configuration.
44#[derive(Debug, Clone, PartialEq, Eq)]
45pub enum ProviderConfig {
46    /// Environment-variable provider.
47    Env,
48    /// File provider with a secrets dir and id→filename map.
49    File {
50        /// Secrets directory.
51        dir: String,
52        /// Credential-id → filename.
53        key_map: BTreeMap<String, String>,
54    },
55}
56
57/// Errors validating a `[credentials]` block.
58#[non_exhaustive]
59#[derive(Debug, thiserror::Error, PartialEq, Eq)]
60pub enum CredentialsConfigError {
61    /// A name in `chain` has no matching provider definition.
62    #[error("chain references undefined provider {0:?}")]
63    UndefinedProvider(String),
64    /// A provider has an unknown `type`.
65    #[error("provider {name:?}: unknown type {kind:?}")]
66    UnknownKind {
67        /// Provider name.
68        name: String,
69        /// The unknown kind string.
70        kind: String,
71    },
72    /// A `file` provider is missing `dir`.
73    #[error("file provider {0:?}: missing `dir`")]
74    FileMissingDir(String),
75}
76
77impl Default for CredentialsChainConfig {
78    fn default() -> Self {
79        // Zero-config default: env-only.
80        Self {
81            chain: vec![ProviderConfig::Env],
82        }
83    }
84}
85
86impl UncheckedCredentialsConfig {
87    /// Validate into a [`CredentialsChainConfig`]. An empty `chain`
88    /// defaults to `["env"]`. `"env"` needs no provider definition.
89    pub fn validate(self) -> Result<CredentialsChainConfig, CredentialsConfigError> {
90        let names = if self.chain.is_empty() {
91            vec!["env".to_string()]
92        } else {
93            self.chain
94        };
95
96        let mut chain = Vec::with_capacity(names.len());
97        for name in names {
98            if name == "env" && !self.providers.contains_key("env") {
99                chain.push(ProviderConfig::Env);
100                continue;
101            }
102            let def = self
103                .providers
104                .get(&name)
105                .ok_or_else(|| CredentialsConfigError::UndefinedProvider(name.clone()))?;
106            match def.kind.as_str() {
107                "env" => chain.push(ProviderConfig::Env),
108                "file" => {
109                    let dir = def
110                        .dir
111                        .clone()
112                        .ok_or_else(|| CredentialsConfigError::FileMissingDir(name.clone()))?;
113                    chain.push(ProviderConfig::File {
114                        dir,
115                        key_map: def.key_map.clone(),
116                    });
117                }
118                other => {
119                    return Err(CredentialsConfigError::UnknownKind {
120                        name: name.clone(),
121                        kind: other.to_string(),
122                    });
123                }
124            }
125        }
126        Ok(CredentialsChainConfig { chain })
127    }
128}

130#[cfg(test)]
131mod tests {
132    use super::*;
133
134    #[test]
135    fn empty_defaults_to_env_only() {
136        let cfg = UncheckedCredentialsConfig::default().validate().unwrap();
137        assert_eq!(cfg.chain, vec![ProviderConfig::Env]);
138    }
139
140    #[test]
141    fn explicit_env_chain_without_provider_def_short_circuits() {
142        let toml = r#"chain = ["env"]"#;
143        let unchecked: UncheckedCredentialsConfig = toml::from_str(toml).unwrap();
144        let cfg = unchecked.validate().unwrap();
145        assert_eq!(cfg.chain, vec![ProviderConfig::Env]);
146    }
147
148    #[test]
149    fn env_then_file_validates() {
150        let toml = r#"
151chain = ["env", "file"]
152[providers.file]
153type = "file"
154dir = "/var/run/secrets"
155key_map = { anthropic_api_key = "anthropic-key" }
156"#;
157        let unchecked: UncheckedCredentialsConfig = toml::from_str(toml).unwrap();
158        let cfg = unchecked.validate().unwrap();
159        assert_eq!(cfg.chain.len(), 2);
160        assert_eq!(cfg.chain[0], ProviderConfig::Env);
161        match &cfg.chain[1] {
162            ProviderConfig::File { dir, key_map } => {
163                assert_eq!(dir, "/var/run/secrets");
164                assert_eq!(key_map.get("anthropic_api_key").unwrap(), "anthropic-key");
165            }
166            _ => panic!("expected file provider"),
167        }
168    }
169
170    #[test]
171    fn undefined_provider_rejected() {
172        let toml = r#"chain = ["vault"]"#;
173        let unchecked: UncheckedCredentialsConfig = toml::from_str(toml).unwrap();
174        assert_eq!(
175            unchecked.validate().unwrap_err(),
176            CredentialsConfigError::UndefinedProvider("vault".to_string())
177        );
178    }
179
180    #[test]
181    fn file_without_dir_rejected() {
182        let toml = r#"
183chain = ["file"]
184[providers.file]
185type = "file"
186"#;
187        let unchecked: UncheckedCredentialsConfig = toml::from_str(toml).unwrap();
188        assert_eq!(
189            unchecked.validate().unwrap_err(),
190            CredentialsConfigError::FileMissingDir("file".to_string())
191        );
192    }
193
194    #[test]
195    fn unknown_kind_rejected() {
196        let toml = r#"
197chain = ["weird"]
198[providers.weird]
199type = "smoke-signal"
200"#;
201        let unchecked: UncheckedCredentialsConfig = toml::from_str(toml).unwrap();
202        assert!(matches!(
203            unchecked.validate().unwrap_err(),
204            CredentialsConfigError::UnknownKind { .. }
205        ));
206    }
