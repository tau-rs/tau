1//! Install lifecycle for tau-pkg.
2//!
3//! [`install`] performs the 10-step pipeline:
4//!
5//! 1. Pre-flight: `git --version` check + acquire scope-level advisory file lock.
6//! 2. Clone source into `<scope>/.tau/packages/.staging/<random>/`.
7//! 3. Parse `tau.toml` from the staging directory.
8//! 4. Validate the manifest structurally (handled by `read_manifest`).
9//! 5. Verify the user-supplied source matches the manifest's declared source.
10//! 6. Capability validation — warn (don't error) on unknown kinds and
11//!    non-namespaced `Capability::Custom` names.
12//! 7. Resolve the cloned repo's HEAD to a 40-char SHA.
13//! 8. Materialize: atomically `fs::rename` staging → `<scope>/.tau/packages/<name>/<version>/`.
14//! 9. Update the lockfile (atomic write).
15//! 10. Release the advisory file lock.
16//!
17//! Failure at any step before (8) leaves no on-disk state (the staging
18//! TempDir is auto-removed). At step (8), the staging directory's
19//! auto-cleanup is disabled before the rename; on rename failure the
20//! orphaned staging dir is best-effort removed. Failure at (8) post-
21//! rename or at (9) is unlikely (both are atomic operations).
22//!
23//! Note: idempotency cannot be checked before the clone because the
24//! package name (lockfile key) is not known until the manifest is
25//! parsed. The clone is therefore always performed; the idempotency
26//! short-circuit happens at step (8) before the rename, avoiding only
27//! the disk-write phase.
28//!
29//! # Trust boundary (security)
30//!
31//! `tau install <source>` clones an arbitrary repository and, for a
32//! buildable package, compiles it (`cargo build`, which executes the
33//! package's `build.rs` and proc macros) and may spawn the freshly built
34//! binary for a capability cross-check — all **before** any Layer-4
35//! sandbox or capability enforcement applies. **Installing a package
36//! therefore executes its author's code on your machine: it is equivalent
37//! to trusting that author.** Only install sources you trust. Running the
38//! install-time build under a sandbox tier is tracked as a follow-up
39//! (finding S2); the bundle-vs-source authenticity check (finding S3) is
40//! implemented in `bundle::verify`.

60/// Run `make_fut()` to completion from a synchronous function that may
61/// itself be called from inside a Tokio runtime.
62///
63/// The install pipeline is synchronous but must `await` the Layer-2
64/// cross-check exactly once. Building a current-thread runtime *inline*
65/// and calling `block_on` panics ("Cannot start a runtime from within a
66/// runtime") when `install_with_options` runs inside an existing async
67/// context — which it does under `tau run` / `tau install` (both
68/// `#[tokio::main]`). Offloading the runtime onto a dedicated OS thread
69/// sidesteps the nesting entirely: the new thread has no ambient
70/// runtime, so `block_on` is always legal there, regardless of whether
71/// the caller's runtime is multi-thread or current-thread (D5).
72///
73/// The future is built *inside* the worker thread, so it need not be
74/// `Send` — only the builder closure and the output value cross the
75/// thread boundary. A fully `async` install pipeline is the cleaner
76/// long-term shape but ripples through every sync caller and their test
77/// sites; this contained seam fixes the panic without that churn.
78fn block_on_in_fresh_thread<T, B, Fut>(make_fut: B) -> Result<T, InstallError>
79where
80    B: FnOnce() -> Fut + Send + 'static,
81    Fut: std::future::Future<Output = T>,
82    T: Send + 'static,
83{
84    std::thread::spawn(move || {
85        let rt = tokio::runtime::Builder::new_current_thread()
86            .enable_all()
87            .build()
88            .map_err(|e| InstallError::Internal {
89                message: format!("build tokio runtime for cross-check: {e}"),
90            })?;
91        Ok(rt.block_on(make_fut()))
92    })
93    .join()
94    .map_err(|_| InstallError::Internal {
95        message: "cross-check worker thread panicked".into(),
96    })?
97}

99/// Options governing plugin builds during install.
100///
101/// Used by [`InstallOptions::build`] to control the `cargo build` step
102/// for plugin packages (those with a `[plugin]` table in `tau.toml`).
103/// For data-only packages, these options are ignored.
104///
105/// `#[non_exhaustive]`: future fields are non-breaking. External
106/// callers construct via [`BuildOptions::new`] / [`BuildOptions::default`]
107/// and mutate fields by name.
108///
109/// # Example
110///
111/// ```
112/// use tau_pkg::BuildOptions;
113///
114/// // `BuildOptions` is `#[non_exhaustive]`; struct-literal construction
115/// // is blocked across crate boundaries. Default + field mutation is
116/// // the canonical pattern.
117/// let mut opts = BuildOptions::default();
118/// opts.skip_build = true;
119/// assert!(opts.skip_build);
120/// ```
121#[non_exhaustive]
122#[derive(Debug, Clone, Default)]
123pub struct BuildOptions {
124    /// Skip the build step entirely. Used by tests that synthesize
125    /// lockfiles against pre-built binaries, and by
126    /// `tau install --no-build`.
127    pub skip_build: bool,
128    /// Override the cargo binary path. Defaults to the `cargo` on PATH.
129    pub cargo_path: Option<PathBuf>,
130    /// Extra arguments passed through to `cargo build`
131    /// (e.g., `--features foo`).
132    pub extra_args: Vec<String>,
133}
134
135impl BuildOptions {
136    /// Construct a fresh `BuildOptions` with defaults: build enabled,
137    /// `cargo` discovered on PATH, no extra args.
138    ///
139    /// `BuildOptions` is `#[non_exhaustive]`; external callers use this
140    /// constructor (or [`BuildOptions::default`]) and mutate fields by
141    /// name.
142    ///
143    /// # Example
144    ///
145    /// ```
146    /// use tau_pkg::BuildOptions;
147    ///
148    /// let opts = BuildOptions::new();
149    /// assert!(!opts.skip_build);
150    /// assert!(opts.cargo_path.is_none());
151    /// assert!(opts.extra_args.is_empty());
152    /// ```
153    pub fn new() -> Self {
154        Self::default()
155    }
156}

158/// Options for [`install_with_options`].
159///
160/// `Debug` is hand-written rather than derived because [`InstallOptions::sandbox`]
161/// is a trait object that does not implement `Debug`.
162#[non_exhaustive]
163#[derive(Clone)]
164pub struct InstallOptions {
165    /// If `true` (default), wait indefinitely for a concurrent install
166    /// to release the advisory file lock. If `false`, error
167    /// immediately with [`InstallError::Locked`].
168    pub block_on_lock: bool,
169    /// If `true`, force re-clone even if the target version directory
170    /// already exists. Default: `false` (idempotent — skip clone if
171    /// the dir exists and the lockfile already records this version).
172    pub force: bool,
173    /// Plugin-build options, applied during the build step for
174    /// packages whose `tau.toml` contains a `[plugin]` table.
175    ///
176    /// Defaults: build enabled, `cargo` from PATH, no extra args.
177    /// Ignored for data-only packages (no `[plugin]` table).
178    pub build: BuildOptions,
179    /// If `true`, skip the Layer 2 cross-check at step 8.7 (sub-project B).
180    ///
181    /// Default: `false` — production installs always run the cross-check.
182    /// Tests that build stub plugin binaries which don't implement the
183    /// `meta.handshake` protocol set this to `true` to bypass the
184    /// cross-check.
185    pub skip_cross_check: bool,
186    /// Install-time sandbox gate, injected by the caller (`tau-cli`). When an
187    /// enforcing gate is present, the build + cross-check run sandboxed. When
188    /// absent (or non-enforcing), the install fails closed unless
189    /// `allow_unsandboxed_build` is set. Default: `None` (audit S2).
190    pub sandbox: Option<Arc<dyn InstallSandbox>>,
191    /// Explicit escape hatch: permit an unsandboxed build/cross-check when no
192    /// enforcing gate is available. Maps to `tau install
193    /// --allow-unsandboxed-build`. Default: `false` (fail-closed).
194    pub allow_unsandboxed_build: bool,
195}

197impl std::fmt::Debug for InstallOptions {
198    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
199        f.debug_struct("InstallOptions")
200            .field("block_on_lock", &self.block_on_lock)
201            .field("force", &self.force)
202            .field("build", &self.build)
203            .field("skip_cross_check", &self.skip_cross_check)
204            .field("sandbox", &self.sandbox.as_ref().map(|_| "<gate>"))
205            .field("allow_unsandboxed_build", &self.allow_unsandboxed_build)
206            .finish()
207    }
208}

260/// Install a package from `source` into `scope` with explicit `options`.
261/// See the module-level documentation for the 10-step lifecycle.
262pub fn install_with_options(
263    source: &PackageSource,
264    scope: &Scope,
265    options: InstallOptions,
266) -> Result<InstalledPackage, InstallError> {
267    // Step 1: pre-flight.
268    let _git_version = Git::version_check()?;
269
270    let lock_path = scope.install_lock_path();
271    if let Some(parent) = lock_path.parent() {
272        fs::create_dir_all(parent).map_err(|e| InstallError::Internal {
273            message: format!("creating lock directory {}: {e}", parent.display()),
274        })?;
275    }
276
277    let lock_file = fs::OpenOptions::new()
278        .read(true)
279        .write(true)
280        .create(true)
281        .truncate(false)
282        .open(&lock_path)
283        .map_err(|e| InstallError::Internal {
284            message: format!("opening install lock {}: {e}", lock_path.display()),
285        })?;
286
287    if options.block_on_lock {
288        lock_file
289            .lock_exclusive()
290            .map_err(|e| InstallError::Internal {
291                message: format!("acquiring install lock: {e}"),
292            })?;
293    } else {
294        match lock_file.try_lock_exclusive() {
295            Ok(()) => {}
296            Err(e) if e.kind() == fs4::lock_contended_error().kind() => {
297                return Err(InstallError::Locked {
298                    scope: scope.state_path().display().to_string(),
299                });
300            }
301            Err(e) => {
302                return Err(InstallError::Internal {
303                    message: format!("trying install lock: {e}"),
304                });
305            }
306        }
307    }
308
309    // From here on, ensure the lock is released even on early returns.
310    // `lock_file` stays in scope until the end; we explicitly unlock at step 10.
311
312    let result = (|| -> Result<InstalledPackage, InstallError> {
313        // Step 2: clone into staging.
314        let staging_root = scope.packages_dir().join(".staging");
315        fs::create_dir_all(&staging_root).map_err(|e| InstallError::Internal {
316            message: format!("creating staging dir {}: {e}", staging_root.display()),
317        })?;
318
319        let staging_dir = tempfile::Builder::new()
320            .prefix("staging-")
321            .tempdir_in(&staging_root)
322            .map_err(|e| InstallError::Internal {
323                message: format!("creating staging tempdir: {e}"),
324            })?;
325
326        Git::clone(source, staging_dir.path())?;
327
328        // Step 3 + 4: parse + validate manifest (validate is part of read_manifest).
329        // Skills-5: auto-detect format. Tau-format workspaces read tau.toml (existing
330        // path). Anthropic-format workspaces (SKILL.md only) synthesize a manifest
331        // in-memory. Neither file → NotASkillPackage.
332        let workspace_format = tau_domain::detect_format(staging_dir.path());
333        let (manifest, synthesized_from) = match workspace_format {
334            tau_domain::SkillFormat::Tau => {
335                let m = read_manifest(&staging_dir.path().join("tau.toml"))?;
336                (m, None)
337            }
338            tau_domain::SkillFormat::Anthropic => {
339                let m = crate::synthesize::synthesize_anthropic_skill(
340                    staging_dir.path(),
341                    source.clone(),
342                )
343                .map_err(crate::error::InstallError::SynthesizeFailed)?;
344
345                // Write the synthesized tau.toml to the staging workspace so
346                // that it is present in the install directory after the rename
347                // at step 8. This allows `find_installed_skill` (used by
348                // `tau skill show` and `tau skill export`) to locate the
349                // manifest without needing the lockfile.
350                let toml_text = toml::to_string_pretty(&m).map_err(|e| InstallError::Internal {
351                    message: format!("serializing synthesized tau.toml: {e}"),
352                })?;
353                fs::write(staging_dir.path().join("tau.toml"), &toml_text).map_err(|e| {
354                    InstallError::Internal {
355                        message: format!("writing synthesized tau.toml to staging dir: {e}"),
356                    }
357                })?;
358
359                (m, Some(crate::lockfile::SynthesizedSource::Anthropic))
360            }
361            tau_domain::SkillFormat::Invalid => {
362                return Err(crate::error::InstallError::NotASkillPackage {
363                    path: staging_dir.path().to_owned(),
364                    detail: "directory has neither tau.toml nor SKILL.md".into(),
365                });
366            }
367            _ => {
368                // Forward-compat: treat unknown variants as invalid.
369                return Err(crate::error::InstallError::NotASkillPackage {
370                    path: staging_dir.path().to_owned(),
371                    detail: "unrecognized workspace format".into(),
372                });
373            }
374        };
375
376        // Step 5: source / manifest match.
377        if *manifest.source() != *source {
378            return Err(InstallError::SourceManifestMismatch {
379                expected: source.to_string(),
380                found: manifest.source().to_string(),
381            });
382        }
383
384        // Step 6: capability validation (warnings only — NG12).
385        warn_unknown_kind(&manifest);
386        // (D7-B PR2: the former non-namespaced-Custom warning is obsolete —
387        // a `Capability::Custom` now requires an explicit `custom.` prefix at
388        // parse time, so a non-namespaced custom capability can no longer be
389        // constructed.)
390
391        // Step 7: resolve commit.
392        let resolved_commit = Git::resolve_head(staging_dir.path())?;
393
394        // Step 8: materialize.
395        let target = scope.package_dir(manifest.name(), manifest.version());
396        let lockfile_path = scope.lockfile_path();
397        let mut lf = LockFile::load(&lockfile_path)?;
398
399        let already_recorded = lf
400            .find(manifest.name())
401            .map(|p| {
402                p.installed_versions
403                    .iter()
404                    .any(|v| v.version == *manifest.version())
405            })
406            .unwrap_or(false);
407
408        if target.exists() {
409            if !options.force && already_recorded {
410                // Idempotent: already installed at this version; return the
411                // existing entry's metadata.
412                let installed_at = lf
413                    .find(manifest.name())
414                    .and_then(|p| {
415                        p.installed_versions
416                            .iter()
417                            .find(|v| v.version == *manifest.version())
418                            .map(|v| v.installed_at)
419                    })
420                    .ok_or_else(|| InstallError::Internal {
421                        message: format!(
422                            "lockfile recorded {}@{} but installed_at lookup returned None — \
423                             inconsistent state",
424                            manifest.name(),
425                            manifest.version(),
426                        ),
427                    })?;
428                return Ok(InstalledPackage {
429                    name: manifest.name().clone(),
430                    version: manifest.version().clone(),
431                    source: source.clone(),
432                    installed_path: target,
433                    installed_at,
434                });
435            }
436            // Otherwise: force OR orphaned directory — overwrite.
437            fs::remove_dir_all(&target).map_err(|e| InstallError::Internal {
438                message: format!("removing existing target {}: {e}", target.display()),
439            })?;
440        }
441
442        if let Some(parent) = target.parent() {
443            fs::create_dir_all(parent).map_err(|e| InstallError::Internal {
444                message: format!("creating target parent {}: {e}", parent.display()),
445            })?;
446        }
447
448        // Take ownership of the staging path before rename (disables auto-cleanup).
449        let staged_path = staging_dir.keep();
450        if let Err(e) = fs::rename(&staged_path, &target) {
451            // Best-effort cleanup — the rename failed, so the staging dir is now
452            // orphaned (tempfile no longer manages it). Ignore secondary failure.
453            let _ = fs::remove_dir_all(&staged_path);
454            return Err(InstallError::Internal {
455                message: format!(
456                    "renaming {} -> {}: {e}",
457                    staged_path.display(),
458                    target.display(),
459                ),
460            });
461        }
462
463        // Step 8.6: compute source tree SHA-256 (post-rename, the install
464        // dir is in its final location; `target/` is excluded by tree_hash).
465        let source_sha256 =
466            crate::tree_hash::tree_hash(&target).map_err(|e| InstallError::Internal {
467                message: format!("computing source tree hash: {e}"),
468            })?;
469
470        // Step 8.5: build (only for kind = "rust-cargo" plugin packages).
471        //
472        // Per spec §6.3, the build is invoked between materialization
473        // (rename) and lockfile write. On failure the lockfile is NOT
474        // updated, but the cloned source is left in place so users can
475        // diagnose without re-cloning. Users retry with `tau install
476        // --force` or inspect the failure under the package dir.
477        let mut locked_plugin = build_plugin_if_needed(
478            &manifest,
479            &target,
480            &options.build,
481            options.sandbox.as_deref(),
482            options.allow_unsandboxed_build,
483        )?;
484
485        // Step 8.7: Layer 2 cross-check — spawn the freshly-built binary,
486        // perform handshake + per-method tool.describe_capabilities, compare
487        // against the manifest's [[capabilities]]. On error, abort install
488        // (binary stays on disk; user retries via `tau install --force`
489        // after fixing the manifest).
490        //
491        // Skipped when:
492        //   - Data-only package (no [plugin] table → locked_plugin is None).
493        //   - Build was skipped via skip_build = true (locked_plugin is None
494        //     even for plugin packages — test / --no-build path).
495        //
496        // cross_check_plugin_capabilities is async; install_with_options is
497        // synchronous. Bridge via `block_on_in_fresh_thread`, which runs the
498        // future on a dedicated thread with its own runtime — never nesting a
499        // `block_on` inside the caller's async context (D5).
500        if !options.skip_cross_check {
501            if let Some(ref mut lp) = locked_plugin {
502                let binary_path = lp.binary_path.clone();
503                let manifest_for_check = manifest.clone();
504                let gate_for_check = options.sandbox.clone();
505                let allow_for_check = options.allow_unsandboxed_build;
506                let shapes = block_on_in_fresh_thread(move || async move {
507                    crate::sandbox_check::cross_check_plugin_capabilities_gated(
508                        &binary_path,
509                        &manifest_for_check,
510                        gate_for_check.as_deref(),
511                        allow_for_check,
512                    )
513                    .await
514                })?
515                .map_err(|e| InstallError::CrossCheck {
516                    message: e.to_string(),
517                })?;
518                lp.required_shapes = shapes;
519            }
520
521            // Skills-2 cross-check: validates SKILL.md content + frontmatter +
522            // reference lint. Hard-fails on missing capabilities.
523            // Skills have no plugin process to spawn; no tokio runtime needed.
524            if manifest.skill().is_some() {
525                crate::skill_check::cross_check_skill_package(&target, &manifest)?;
526            }
527        }
528
529        // Step 8.8: for skill packages, compute SHA-256 of the SKILL.md bytes
530        // and snapshot the frontmatter into a LockedSkill for the lockfile.
531        let locked_skill: Option<LockedSkill> = if manifest.skill().is_some() {
532            let content_path = target.join(&manifest.skill().unwrap().content);
533            let bytes = std::fs::read(&content_path).map_err(|e| InstallError::Internal {
534                message: format!("reading SKILL.md for sha256 at {content_path:?}: {e}"),
535            })?;
536            let content_sha256 = {
537                use sha2::{Digest, Sha256};
538                let mut hasher = Sha256::new();
539                hasher.update(&bytes);
540                crate::tree_hash::to_hex_lower(&hasher.finalize())
541            };
542
543            // Re-parse for the frontmatter snapshot. cross_check_skill_package
544            // already validated; re-parsing here is a small cost to keep the
545            // two concerns (validation vs snapshot) cleanly separated.
546            let body_text = std::str::from_utf8(&bytes).map_err(|e| InstallError::Internal {
547                message: format!("SKILL.md is not UTF-8 at {content_path:?}: {e}"),
548            })?;
549            let parsed = tau_domain::parse_skill_md(body_text).map_err(|e| {
550                // Should be unreachable — cross_check just validated — but
551                // surface as Internal if it fires.
552                InstallError::Internal {
553                    message: format!("post-validation SKILL.md re-parse failed: {e}"),
554                }
555            })?;
556
557            Some(LockedSkill::new(
558                content_sha256,
559                SkillFrontmatterSnapshot {
560                    name: parsed.frontmatter.name,
561                    description: parsed.frontmatter.description,
562                },
563            ))
564        } else {
565            None
566        };
567
568        // Step 9: update lockfile.
569        let now = SystemTime::now();
570        let new_locked_version = LockedVersion {
571            version: manifest.version().clone(),
572            rev: rev_from_source(source),
573            resolved_commit,
574            sha256: source_sha256,
575            installed_at: now,
576        };
577
578        let updated_package = match lf.find(manifest.name()).cloned() {
579            Some(mut existing) => {
580                // Replace the matching version (if present) or append.
581                let mut found = false;
582                for v in existing.installed_versions.iter_mut() {
583                    if v.version == *manifest.version() {
584                        *v = new_locked_version.clone();
585                        found = true;
586                        break;
587                    }
588                }
589                if !found {
590                    existing.installed_versions.push(new_locked_version.clone());
591                }
592                existing.active_version = manifest.version().clone();
593                existing.source = source.clone();
594                existing.plugin = locked_plugin.clone();
595                existing.skill = locked_skill.clone();
596                existing.synthesized_from = synthesized_from;
597                // Sort installed_versions for deterministic lockfile diffs.
598                existing
599                    .installed_versions
600                    .sort_by(|a, b| a.version.cmp(&b.version));
601                existing
602            }
603            None => LockedPackage {
604                name: manifest.name().clone(),
605                active_version: manifest.version().clone(),
606                source: source.clone(),
607                installed_versions: vec![new_locked_version.clone()],
608                plugin: locked_plugin.clone(),
609                skill: locked_skill,
610                synthesized_from,
611            },
612        };
613
614        lf.upsert(updated_package);
615        lf.packages
616            .sort_by(|a, b| a.name.as_str().cmp(b.name.as_str()));
617        lf.generated_at = now;
618        lf.save(&lockfile_path)?;
619
620        Ok(InstalledPackage {
621            name: manifest.name().clone(),
622            version: manifest.version().clone(),
623            source: source.clone(),
624            installed_path: target,
625            installed_at: now,
626        })
627    })();
628
629    // Step 10: release the lock (explicit via FileExt::unlock; also released on drop).
630    let _ = lock_file.unlock();
631
632    result
633}

659/// `binary_path` canonicalized.
660///
661/// Returns `Err(InstallError::CargoNotFound)` if the configured cargo
662/// binary cannot be invoked, and `Err(InstallError::BuildFailed { .. })`
663/// for non-zero cargo exits.
664fn build_plugin_if_needed(
665    manifest: &tau_domain::PackageManifest,
666    package_dir: &Path,
667    options: &BuildOptions,
668    gate: Option<&dyn InstallSandbox>,
669    allow_unsandboxed_build: bool,
670) -> Result<Option<LockedPlugin>, InstallError> {
671    let Some(plugin_manifest) = manifest.plugin() else {
672        return Ok(None);
673    };
674    if options.skip_build {
675        return Ok(None);
676    }
677
678    match plugin_manifest.kind {
679        PluginKind::RustCargo => build_rust_cargo_plugin(
680            plugin_manifest,
681            manifest.name(),
682            manifest.version(),

684            options,
685            gate,
686            allow_unsandboxed_build,
687        ),
688        // `PluginKind` is `#[non_exhaustive]`. Future variants (e.g.
689        // `PythonPip`, `NodeNpm`, `Prebuilt`) get their own build paths;
690        // until then, surface unknown kinds via `Internal` so the user
691        // gets a typed error rather than a silent no-op.
692        other => Err(InstallError::Internal {
693            message: format!("plugin kind {other} not yet supported by this tau-pkg version"),
694        }),
695    }

698/// Construct the `cargo build` command, apply the fail-closed sandbox
699/// decision, wrap the command under the gate when enforcing, and spawn it.
700///
701/// Fail-closed (audit S2): with no enforcing gate and no
702/// `--allow-unsandboxed-build`, this refuses with
703/// [`InstallError::UnsandboxedBuildRefused`] *before* spawning anything.
704fn run_cargo_build_gated(
705    package_dir: &Path,
706    target_dir: &Path,
707    bin: &str,
708    options: &BuildOptions,
709    gate: Option<&dyn InstallSandbox>,
710    allow_unsandboxed_build: bool,
711) -> Result<std::process::Output, InstallError> {
712    let cargo = options
713        .cargo_path
714        .clone()
715        .unwrap_or_else(|| PathBuf::from("cargo"));
716
717    let mut cmd = Command::new(&cargo);
718    cmd.arg("build")
719        .arg("--release")
720        .arg("--bin")
721        .arg(bin)
722        .current_dir(package_dir)
723        .env("CARGO_TARGET_DIR", target_dir);
724    for arg in &options.extra_args {
725        cmd.arg(arg);
726    }
727
728    // Decide-then-wrap-then-spawn. The guard holds any sandbox resources
729    // (egress proxy, namespace fds) and must outlive `cmd.output()`.
730    let _guard = match sandbox_decision(gate, allow_unsandboxed_build) {
731        SandboxDecision::Sandbox => {
732            let plan = build_envelope(package_dir);
733            let g = gate.expect("Sandbox decision implies a gate is present");
734            Some(
735                g.wrap(&plan, &mut cmd)
736                    .map_err(|e| InstallError::Internal {
737                        message: format!("wrapping cargo build in sandbox: {e}"),
738                    })?,
739            )
740        }
741        SandboxDecision::Unsandboxed => {
742            tracing::warn!(
743                target: "tau_pkg::install",
744                "building untrusted package code WITHOUT a sandbox \
745                 (--allow-unsandboxed-build); build.rs runs with full host access",
746            );
747            None
748        }
749        SandboxDecision::Refuse => return Err(InstallError::UnsandboxedBuildRefused),
750    };
751
752    cmd.output().map_err(|e| {
753        if e.kind() == std::io::ErrorKind::NotFound {
754            InstallError::CargoNotFound
755        } else {
756            InstallError::Internal {
757                message: format!("spawning cargo at {}: {e}", cargo.display()),
758            }
759        }
760    })
761}

763/// Run `cargo build --release --bin <plugin.bin>` in `package_dir` and
764/// return the [`LockedPlugin`] for the lockfile.
765fn build_rust_cargo_plugin(
766    plugin_manifest: &tau_domain::PluginManifest,
767    package_name: &PackageName,
768    package_version: &Version,
769    package_dir: &Path,
770    options: &BuildOptions,
771    gate: Option<&dyn InstallSandbox>,
772    allow_unsandboxed_build: bool,
773) -> Result<Option<LockedPlugin>, InstallError> {
774    // Single source of truth for where this plugin's build artifacts
775    // land. Both the spawned build (via `CARGO_TARGET_DIR`) and the
776    // binary-path lookup below derive from `target_dir`, so they can
777    // never drift.
778    //
779    // Pinning `CARGO_TARGET_DIR` explicitly — rather than relying on
780    // cargo's default of `<cwd>/target` — makes the build robust to an
781    // inherited `CARGO_TARGET_DIR` in the parent environment. This
782    // workspace's CLAUDE.md requires every cargo invocation to set a
783    // per-agent target dir; that variable is inherited by child
784    // processes, so without an explicit override the inner cargo would
785    // build into the *outer* target dir and this lookup would fail with
786    // "binary not found after build" (audit finding D6). An explicit
787    // `env` set wins over any inherited value, so no defensive
788    // `env_remove` is needed.
789    //
790    // `package_dir` is scope-derived and therefore absolute, so this
791    // `CARGO_TARGET_DIR` is an absolute path; cargo doesn't re-resolve
792    // it against the spawned build's cwd (also `package_dir`).
793    let target_dir = package_dir.join("target");
794
795    log_build_start(package_name, package_version, plugin_manifest, package_dir);
796
797    let output = run_cargo_build_gated(
798        package_dir,
799        &target_dir,
800        &plugin_manifest.bin,
801        options,
802        gate,
803        allow_unsandboxed_build,
804    )?;
805
806    // Stream cargo's captured output to the host's stderr so users see
807    // build progress. (We capture rather than inherit so we can record
808    // the stderr tail on failure.)
809    if !output.stdout.is_empty() {
810        let s = String::from_utf8_lossy(&output.stdout);
811        eprint!("{s}");
812    }
813    if !output.stderr.is_empty() {
814        let s = String::from_utf8_lossy(&output.stderr);
815        eprint!("{s}");
816    }
817
818    if !output.status.success() {
819        return Err(InstallError::BuildFailed {
820            exit_status: output.status,
821            stderr_tail: tail_lossy_utf8(&output.stderr, STDERR_TAIL_BYTES),
822        });
823    }
824
825    // Cargo appends the platform-specific executable extension
826    // (`.exe` on Windows, empty elsewhere). Match it via
827    // `std::env::consts::EXE_SUFFIX` so Windows installs find their
828    // binaries.
829    let bin_filename = format!("{}{}", plugin_manifest.bin, std::env::consts::EXE_SUFFIX);
830    let binary_path = target_dir.join("release").join(&bin_filename);
831    let canonical = binary_path
832        .canonicalize()
833        .map_err(|e| InstallError::BuildFailed {
834            exit_status: output.status,
835            stderr_tail: format!(
836                "cargo succeeded but binary not found at {}: {e}",
837                binary_path.display(),
838            ),
839        })?;
840
841    let binary_sha256 =
842        crate::tree_hash::sha256_of_file(&canonical).map_err(|e| InstallError::Internal {
843            message: format!("computing plugin binary hash: {e}"),
844        })?;
845
846    Ok(Some(LockedPlugin::new(
847        plugin_manifest.clone(),
848        canonical,
849        SystemTime::now(),
850        binary_sha256,
851    )))
852}

920/// Uninstall a package from `scope`.
921///
922/// If `version` is `None`, removes ALL installed versions and the
923/// lockfile entry. If `Some`, removes just that version directory
924/// and updates the lockfile. If the removed version was the active
925/// version, promotes the highest remaining (semver-sorted) version
926/// as active; if no versions remain, the package entry is removed
927/// entirely.
928///
929/// # Errors
930///
931/// - [`UninstallError::NotInstalled`] — the package isn't recorded
932///   in the lockfile at all.
933/// - [`UninstallError::VersionNotInstalled`] — the package is
934///   installed but not at the requested version.
935/// - [`UninstallError::Io`] — file-system removal failed.
936/// - [`UninstallError::Locked`] — the per-scope advisory file lock
937///   is held by another process (we currently always block on the
938///   lock; this is reserved for future non-blocking flavors).
939/// - [`UninstallError::Registry`] — lockfile load/save failed.
940/// - [`UninstallError::Scope`] — scope state directory issue.
941///
942/// # Example
943///
944/// ```no_run
945/// // `uninstall` acquires a file lock and modifies on-disk state —
946/// // cannot run in a doctest without a pre-populated install scope.
947/// use tau_pkg::{uninstall, Scope};
948/// use std::str::FromStr;
949///
950/// let scope = Scope::global().unwrap();
951/// let name: tau_domain::PackageName = "acme-tool".parse().unwrap();
952/// uninstall(&name, None, &scope).unwrap();
953/// ```
954pub fn uninstall(
955    name: &PackageName,
956    version: Option<&Version>,
957    scope: &Scope,
958) -> Result<(), UninstallError> {
959    let lock_path = scope.install_lock_path();
960    if let Some(parent) = lock_path.parent() {
961        fs::create_dir_all(parent).map_err(|e| UninstallError::Io {
962            message: format!("creating lock directory {}: {e}", parent.display()),
963        })?;
964    }
965
966    let lock_file = fs::OpenOptions::new()
967        .read(true)
968        .write(true)
969        .create(true)
970        .truncate(false)
971        .open(&lock_path)
972        .map_err(|e| UninstallError::Io {
973            message: format!("opening uninstall lock {}: {e}", lock_path.display()),
974        })?;
975
976    FileExt::lock_exclusive(&lock_file).map_err(|e| UninstallError::Io {
977        message: format!("acquiring uninstall lock: {e}"),
978    })?;
979
980    let result = (|| -> Result<(), UninstallError> {
981        let lockfile_path = scope.lockfile_path();
982        let mut lf = LockFile::load(&lockfile_path)?;
983
984        if lf.find(name).is_none() {
985            return Err(UninstallError::NotInstalled {
986                name: name.as_str().to_owned(),
987            });
988        }
989
990        match version {
991            None => {
992                // Remove all versions of the package.
993                let pkg_dir = scope.packages_dir().join(name.as_str());
994                if pkg_dir.exists() {
995                    fs::remove_dir_all(&pkg_dir).map_err(|e| UninstallError::Io {
996                        message: format!("removing {}: {e}", pkg_dir.display()),
997                    })?;
998                }
999                lf.remove(name);
1000            }
1001            Some(v) => {
1002                // Remove just one version.
1003                let pkg = lf.find(name).expect("verified above");
1004                let has_version = pkg.installed_versions.iter().any(|lv| lv.version == *v);
1005                if !has_version {
1006                    return Err(UninstallError::VersionNotInstalled {
1007                        name: name.as_str().to_owned(),
1008                        version: v.to_string(),
1009                    });
1010                }
1011
1012                let version_dir = scope.package_dir(name, v);
1013                if version_dir.exists() {
1014                    fs::remove_dir_all(&version_dir).map_err(|e| UninstallError::Io {
1015                        message: format!("removing {}: {e}", version_dir.display()),
1016                    })?;
1017                }
1018
1019                // Mutate the lockfile entry: drop the LockedVersion, possibly
1020                // promote a new active_version, or remove the package entirely.
1021                let mut pkg_clone = lf.find(name).expect("verified above").clone();
1022                pkg_clone.installed_versions.retain(|lv| lv.version != *v);
1023
1024                if pkg_clone.installed_versions.is_empty() {
1025                    lf.remove(name);
1026                } else {
1027                    if pkg_clone.active_version == *v {
1028                        // Promote the highest remaining version (semver-sorted).
1029                        let highest = pkg_clone
1030                            .installed_versions
1031                            .iter()
1032                            .map(|lv| lv.version.clone())
1033                            .max()
1034                            .expect("non-empty after retain check");
1035                        pkg_clone.active_version = highest;
1036                    }
1037                    // Keep installed_versions sorted by semver for stable diffs.
1038                    pkg_clone
1039                        .installed_versions
1040                        .sort_by(|a, b| a.version.cmp(&b.version));
1041                    lf.upsert(pkg_clone);
1042                }
1043            }
1044        }
1045
1046        // Top-level packages list also kept sorted for stable diffs.
1047        lf.packages
1048            .sort_by(|a, b| a.name.as_str().cmp(b.name.as_str()));
1049        lf.generated_at = SystemTime::now();
1050        lf.save(&lockfile_path)?;
1051        Ok(())
1052    })();
1053
1054    let _ = FileExt::unlock(&lock_file);
1055    result
1056}

1426    /// One captured `tracing` event: its target, level, and the values of
1427    /// the `package` / `version` fields (if present).
1428    #[derive(Clone, Debug)]
1429    struct Captured {
1430        target: String,
1431        level: Level,
1432        package: Option<String>,
1433        version: Option<String>,
1434        #[allow(dead_code)]
1435        message: Option<String>,
1436    }
1437
1438    #[derive(Default, Clone)]
1439    struct CaptureLayer(Arc<Mutex<Vec<Captured>>>);
1440
1441    impl<S: Subscriber> Layer<S> for CaptureLayer {
1442        fn on_event(&self, event: &Event<'_>, _ctx: Context<'_, S>) {
1443            let mut v = FieldVisitor::default();
1444            event.record(&mut v);
1445            self.0
1446                .lock()
1447                .expect("capture mutex poisoned")
1448                .push(Captured {
1449                    target: event.metadata().target().to_string(),
1450                    level: *event.metadata().level(),
1451                    package: v.package,
1452                    version: v.version,
1453                    message: v.message,
1454                });
1455        }
1456    }
1457
1458    /// Extracts the `package` field and the event `message` (recorded by
1459    /// `tracing` under the reserved `message` field name). Both arrive via
1460    /// `record_str` or `record_debug` depending on the recorder; accept
1461    /// both and strip the `Debug` quotes.
1462    #[derive(Default)]
1463    struct FieldVisitor {
1464        package: Option<String>,
1465        version: Option<String>,
1466        message: Option<String>,
1467    }
1468
1469    impl FieldVisitor {
1470        fn set(&mut self, name: &str, value: String) {
1471            match name {
1472                "package" => self.package = Some(value),
1473                "version" => self.version = Some(value),
1474                "message" => self.message = Some(value),
1475                _ => {}
1476            }
1477        }
1478    }
1479
1480    impl Visit for FieldVisitor {
1481        fn record_str(&mut self, field: &Field, value: &str) {
1482            self.set(field.name(), value.to_string());
1483        }
1484        fn record_debug(&mut self, field: &Field, value: &dyn std::fmt::Debug) {
1485            let raw = format!("{value:?}");
1486            self.set(field.name(), raw.trim_matches('"').to_string());
1487        }
1488    }
