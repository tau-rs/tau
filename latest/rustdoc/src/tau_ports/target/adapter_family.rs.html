3use alloc::string::ToString;
4use core::fmt;
5use core::str::FromStr;
6
7use crate::target::parse::ParseError;

9/// Sandbox adapter family identified in a `TargetTriple`.
10///
11/// Mirrors `tau_runtime_tokio::process_gate::registry::RegistryKind` plus a `Wasi`
12/// variant reserved for future WASI sandbox adapters (no impl in v1).
13#[non_exhaustive]
14#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
15pub enum AdapterFamily {
16    /// OS-native sandbox adapter (landlock/seccomp on Linux, sandbox-exec on Darwin).
17    Native,
18    /// Container-based sandbox adapter (Podman/Docker).
19    Container,
20    /// Remote sandbox adapter (delegated execution).
21    Remote,
22    /// WASI sandbox adapter (reserved for future use).
23    Wasi,
24    /// Passthrough (no sandboxing); used in the single-segment `passthrough` special.
25    Passthrough,
26}
27
28impl AdapterFamily {
29    /// Returns the canonical lowercase string for this adapter family.
30    pub fn as_str(&self) -> &'static str {
31        match self {
32            AdapterFamily::Native => "native",
33            AdapterFamily::Container => "container",
34            AdapterFamily::Remote => "remote",
35            AdapterFamily::Wasi => "wasi",
36            AdapterFamily::Passthrough => "passthrough",
37        }
38    }
39}
40
41impl fmt::Display for AdapterFamily {
42    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
43        f.write_str(self.as_str())
44    }
45}
46
47impl FromStr for AdapterFamily {
48    type Err = ParseError;
49    fn from_str(s: &str) -> Result<Self, Self::Err> {
50        match s {
51            "native" => Ok(AdapterFamily::Native),
52            "container" => Ok(AdapterFamily::Container),
53            "remote" => Ok(AdapterFamily::Remote),
54            "wasi" => Ok(AdapterFamily::Wasi),
55            "passthrough" => Ok(AdapterFamily::Passthrough),
56            other => Err(ParseError::UnknownAdapterFamily(other.to_string())),
57        }
58    }
59}
