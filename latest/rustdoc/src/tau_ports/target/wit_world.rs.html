1//! WIT-world generation for the wasm target (EPIC 3.2).
2//!
3//! [`generate_world`] folds a capability set through the 3.1
4//! [`map_capability`](super::wasi_map::map_capability) table, unions the
5//! `Disposition::Wasi` WIT imports, expands their hardcoded transitive
6//! closure, and renders a deterministic WIT `world`. The world is the frozen
7//! `tau:host` `runner` world's superset with the cap-derived WASI imports
8//! added. An `Unsupported` capability (fs.exec, process.spawn) is a hard error.
9//!
10//! Output is a deterministic ABI *manifest*: without vendored WASI `.wit`
11//! packages it is not standalone-resolvable WIT (that is 3.3+). Determinism is
12//! the contract 3.5's `verify --bundle` byte-compare relies on.
13//!
14//! See `docs/superpowers/specs/2026-07-24-epic-3-2-wit-world-gen-design.md`.
15

24/// Error raised when a capability cannot be realized on the wasm target.
25#[derive(Debug, thiserror::Error, PartialEq, Eq)]
26pub enum WitWorldError {
27    /// A capability maps to `Disposition::Unsupported` on wasm (fs.exec,
28    /// process.spawn) — it has no WASI ABI realization.
29    #[error("capability `{cap}` cannot target wasm: {reason}")]
30    UnsupportedOnWasm {
31        /// Debug rendering of the offending capability.
32        cap: String,
33        /// The reason carried by `Disposition::Unsupported`.
34        reason: &'static str,
35    },
36}
37
38/// Transitive WASI interfaces (as fully-qualified package-ids at
39/// `WASI_VERSION`) that a direct [`WitInterface`] pulls in. These interfaces
40/// are NOT in 3.1's `WitInterface` enum — they are the closure 3.2 owns.
41///
42/// Edges (WASI 0.2.3): `http/types` → io/{streams,poll,error} +
43/// clocks/monotonic-clock; `filesystem/types` → io/{streams,poll,error} +
44/// clocks/wall-clock; `io/streams` → io/{error,poll};
45/// `clocks/monotonic-clock` → io/poll (all folded into the sets below).
46fn transitive_closure(iface: WitInterface) -> &'static [&'static str] {
47    match iface {
48        WitInterface::WasiHttpTypes | WitInterface::WasiHttpOutgoingHandler => &[
49            "wasi:io/streams@0.2.3",
50            "wasi:io/poll@0.2.3",
51            "wasi:io/error@0.2.3",
52            "wasi:clocks/monotonic-clock@0.2.3",
53        ],
54        WitInterface::WasiFilesystemTypes | WitInterface::WasiFilesystemPreopens => &[
55            "wasi:io/streams@0.2.3",
56            "wasi:io/poll@0.2.3",
57            "wasi:io/error@0.2.3",
58            "wasi:clocks/wall-clock@0.2.3",
59        ],
60        // NOTE: no wildcard arm. `WitInterface` is `#[non_exhaustive]` for
61        // *other* crates, but this match is in the crate that defines it, so
62        // rustc checks it exhaustively today (an unreachable-arm wildcard
63        // here would be a hard `-D warnings` error). If 3.1 ever adds a
64        // variant, this match fails to compile until extended — fail-closed
65        // by construction rather than silently contributing no closure.
66    }
67}

69/// Generate the guest component's WIT `world` from a capability set.
70///
71/// Folds each capability through [`map_capability`], keeps the
72/// `Disposition::Wasi` imports, unions them, expands the transitive closure,
73/// and renders a deterministic `world runner` importing `tau:host` + the
74/// resulting WASI interfaces and exporting `run`. `InGuest`/`HostMediated`
75/// capabilities contribute no import; an `Unsupported` capability is a hard
76/// error ([`WitWorldError::UnsupportedOnWasm`]).
77///
78/// # Example
79///
80/// ```
81/// use tau_ports::target::wit_world::generate_world;
82/// use tau_domain::fixtures::cap_fs_read;
83///
84/// let wit = generate_world(&[cap_fs_read(&["/d"])]).unwrap();
85/// assert!(wit.contains("import wasi:filesystem/types@0.2.3;"));
86/// assert!(wit.contains("export run: func(prompt: string)"));
87/// ```
88pub fn generate_world(caps: &[Capability]) -> Result<String, WitWorldError> {
89    // 1. Union the direct WASI interfaces the granted caps require.
90    let mut ifaces: BTreeSet<WitInterface> = BTreeSet::new();
91    for cap in caps {
92        let mapping = map_capability(cap);
93        match mapping.disposition {
94            Disposition::Wasi => ifaces.extend(mapping.imports),
95            Disposition::Unsupported { reason } => {
96                return Err(WitWorldError::UnsupportedOnWasm {
97                    cap: format!("{cap:?}"),
98                    reason,
99                });
100            }
101            // No WASI surface — contributes nothing to the world.
102            //
103            // NOTE: no wildcard arm here either, for the same reason as
104            // `transitive_closure` above: `Disposition` is `#[non_exhaustive]`
105            // only for external crates, and this match is exhaustive today
106            // within tau-ports, so a wildcard would be an unreachable-pattern
107            // `-D warnings` error. A future `Disposition` variant fails this
108            // match at compile time until handled explicitly.
109            Disposition::InGuest | Disposition::HostMediated => {}
110        }
111    }
112
113    // 2. Expand to fully-qualified package-ids (direct + transitive closure),
114    //    deduped and sorted via BTreeSet → deterministic output.
115    let mut imports: BTreeSet<&'static str> = BTreeSet::new();
116    for iface in &ifaces {
117        imports.insert(iface.package_id());
118        for id in transitive_closure(*iface) {
119            imports.insert(id);
120        }
121    }
122
123    // 3. Render. `import host;` mirrors the frozen `wit/tau-host.wit` style.
124    let mut out = String::new();
125    out.push_str("package tau:generated@0.1.0;\n\nworld runner {\n    import host;\n");
126    for id in &imports {
127        out.push_str("    import ");
128        out.push_str(id);
129        out.push_str(";\n");
130    }
131    out.push_str("\n    export run: func(prompt: string) -> result<string, string>;\n}\n");
132    Ok(out)
133}

135#[cfg(test)]
136mod tests {
137    use super::*;
138    use tau_domain::fixtures::{cap_agent_spawn, cap_fs_read, cap_net_http, cap_process_spawn};
139
140    #[test]
141    fn empty_cap_set_yields_host_only_world() {
142        let world = generate_world(&[]).expect("empty is ok");
143        assert_eq!(
144            world,
145            "package tau:generated@0.1.0;\n\
146             \n\
147             world runner {\n\
148             \x20   import host;\n\
149             \n\
150             \x20   export run: func(prompt: string) -> result<string, string>;\n\
151             }\n"
152        );
153    }
154
155    #[test]
156    fn net_only_imports_http_plus_transitive() {
157        let world = generate_world(&[cap_net_http(&["api.anthropic.com"], &["POST"])]).unwrap();
158        for want in [
159            "import host;",
160            "import wasi:http/types@0.2.3;",
161            "import wasi:http/outgoing-handler@0.2.3;",
162            "import wasi:io/streams@0.2.3;",
163            "import wasi:io/poll@0.2.3;",
164            "import wasi:io/error@0.2.3;",
165            "import wasi:clocks/monotonic-clock@0.2.3;",
166        ] {
167            assert!(world.contains(want), "missing `{want}` in:\n{world}");
168        }
169        // fs / wall-clock interfaces must NOT appear for a net-only cap set.
170        assert!(
171            !world.contains("wasi:filesystem"),
172            "net-only leaked fs:\n{world}"
173        );
174        assert!(
175            !world.contains("wall-clock"),
176            "net-only leaked wall-clock:\n{world}"
177        );
178    }
179
180    #[test]
181    fn fs_only_imports_filesystem_plus_transitive() {
182        let world = generate_world(&[cap_fs_read(&["/data/**"])]).unwrap();
183        for want in [
184            "import wasi:filesystem/types@0.2.3;",
185            "import wasi:filesystem/preopens@0.2.3;",
186            "import wasi:io/streams@0.2.3;",
187            "import wasi:clocks/wall-clock@0.2.3;",
188        ] {
189            assert!(world.contains(want), "missing `{want}` in:\n{world}");
190        }
191        assert!(
192            !world.contains("wasi:http"),
193            "fs-only leaked http:\n{world}"
194        );
195        assert!(
196            !world.contains("monotonic-clock"),
197            "fs-only leaked monotonic:\n{world}"
198        );
199    }
200
201    #[test]
202    fn mixed_unions_and_dedupes() {
203        let caps = [cap_net_http(&["h"], &[]), cap_fs_read(&["/d"])];
204        let world = generate_world(&caps).unwrap();
205        assert!(world.contains("wasi:http/types@0.2.3;"));
206        assert!(world.contains("wasi:filesystem/types@0.2.3;"));
207        // io/streams is shared by both families — appears exactly once.
208        assert_eq!(world.matches("import wasi:io/streams@0.2.3;").count(), 1);
209    }
210
211    #[test]
212    fn in_guest_caps_contribute_no_import() {
213        let world = generate_world(&[cap_agent_spawn(&["worker"])]).unwrap();
214        assert!(
215            !world.contains("wasi:"),
216            "in-guest cap leaked a wasi import:\n{world}"
217        );
218        assert!(world.contains("import host;"));
219    }
220
221    #[test]
222    fn unsupported_cap_is_a_hard_error() {
223        let err = generate_world(&[cap_process_spawn(&["ls"])]).unwrap_err();
224        match err {
225            WitWorldError::UnsupportedOnWasm { reason, .. } => assert!(!reason.is_empty()),
226        }
227    }

229    #[test]
230    fn output_is_deterministic_regardless_of_cap_order() {
231        let a = [cap_net_http(&["h"], &[]), cap_fs_read(&["/d"])];
232        let b = [cap_fs_read(&["/d"]), cap_net_http(&["h"], &[])];
233        assert_eq!(generate_world(&a).unwrap(), generate_world(&b).unwrap());
234    }

236    #[test]
237    fn every_transitive_id_is_version_pinned() {
238        for iface in [
239            WitInterface::WasiHttpTypes,
240            WitInterface::WasiFilesystemTypes,
241        ] {
242            for id in transitive_closure(iface) {
243                assert!(id.starts_with("wasi:"), "not qualified: {id}");
244                assert!(id.ends_with("@0.2.3"), "version drift: {id}");
245            }
246        }
247    }
