1//! Capability → WASI/WIT mapping table for the wasm target (EPIC 3.1).
2//!
3//! `map_capability` lowers one [`tau_domain::Capability`] to its WASI/WIT
4//! realization: the WIT interface [`WitInterface`] imports the generated world
5//! must declare (3.2), the `WasiConfig` fragment the host `WasiCtx` consumes
6//! (3.3), and the `Disposition` that says how the capability is satisfied on
7//! wasm (3.4). Pure, total, and read-only over `tau_domain`.
8//!
9//! See `docs/superpowers/specs/2026-07-23-epic-3-1-cap-wit-table-design.md`.

26/// The WASI interfaces this table references. [`WitInterface::package_id`]
27/// returns the fully-qualified WIT package id, e.g.
28/// `"wasi:http/outgoing-handler@0.2.3"`.
29#[non_exhaustive]
30#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
31pub enum WitInterface {
32    /// `wasi:http/types` — HTTP request/response value types.
33    WasiHttpTypes,
34    /// `wasi:http/outgoing-handler` — outbound HTTP; carries the host allow-list.
35    WasiHttpOutgoingHandler,
36    /// `wasi:filesystem/types` — filesystem descriptors and operations.
37    WasiFilesystemTypes,
38    /// `wasi:filesystem/preopens` — the set of preopened directories.
39    WasiFilesystemPreopens,
40}
41
42impl WitInterface {
43    /// Fully-qualified WIT package id (interface path + `@` + [`WASI_VERSION`]).
44    pub fn package_id(&self) -> &'static str {
45        match self {
46            WitInterface::WasiHttpTypes => "wasi:http/types@0.2.3",
47            WitInterface::WasiHttpOutgoingHandler => "wasi:http/outgoing-handler@0.2.3",
48            WitInterface::WasiFilesystemTypes => "wasi:filesystem/types@0.2.3",
49            WitInterface::WasiFilesystemPreopens => "wasi:filesystem/preopens@0.2.3",
50        }
51    }
52}

54/// How a capability is satisfied on the wasm target.
55#[non_exhaustive]
56#[derive(Debug, Clone, PartialEq, Eq)]
57pub enum Disposition {
58    /// Bounded by a WASI import + config: network, fs.read, fs.write.
59    Wasi,
60    /// Enforced in-guest by the tau runtime; no WASI surface
61    /// (taskllist, plan, agent.spawn, skill.spawn).
62    InGuest,
63    /// Requires host mediation outside the WASI ABI; out of scope for wasm
64    /// capability gating (hardware / generic `Custom`).
65    HostMediated,
66    /// Cannot be expressed on the wasm target (fs.exec, process.spawn).
67    Unsupported {
68        /// Human-readable reason, surfaced by 3.2/3.4 diagnostics.
69        reason: &'static str,
70    },
71}

74#[derive(Debug, Clone, PartialEq, Eq)]
75pub enum PreopenAccess {
76    /// fs.read → read-only preopen.
77    ReadOnly,
78    /// fs.write → read-write preopen.
79    ReadWrite,

82/// A single preopen derived from an fs capability. Glob → directory
83/// resolution is deferred to story 3.3.
84#[derive(Debug, Clone, PartialEq, Eq)]
85pub struct Preopen {
86    /// Glob patterns copied verbatim from the fs capability.
87    pub paths: Vec<String>,
88    /// Read-only (fs.read) or read-write (fs.write).

92/// Runtime configuration a capability contributes to the host `WasiCtx` (3.3).
93#[non_exhaustive]
94#[derive(Debug, Clone, PartialEq, Eq)]
95pub enum WasiConfig {
96    /// No runtime config (all non-`Wasi` dispositions).
97    None,
98    /// Network egress filter. `hosts` reuses D4-B [`HostSet`] semantics
99    /// (exact | typed `Any`); `methods == None` means all methods.
100    AllowedHosts {
101        /// Allowed hostnames, copied verbatim from the capability.
102        hosts: HostSet,
103        /// Allowed HTTP methods; `None` = all.
104        methods: Option<BTreeSet<HttpMethod>>,
105    },
106    /// Filesystem preopens derived from the capability's glob paths.
107    Preopens(Vec<Preopen>),
108}

110/// The WASI/WIT realization of a single capability on the wasm target.
111#[derive(Debug, Clone, PartialEq, Eq)]
112pub struct WasiMapping {
113    /// WIT interface imports the generated world must declare (3.2). Empty
114    /// unless `disposition == Disposition::Wasi`.
115    pub imports: Vec<WitInterface>,
116    /// Runtime config fragment this capability contributes to `WasiCtx` (3.3).
117    pub config: WasiConfig,
118    /// How this capability is satisfied on the wasm target.
119    pub disposition: Disposition,
120}

122/// Lower one tau [`Capability`] to its WASI/WIT realization on the wasm target.
123///
124/// Total and pure: every capability yields a [`WasiMapping`]. Capabilities
125/// that bind to a WASI import return `Disposition::Wasi` with non-empty
126/// `imports`; all others carry empty `imports` and `WasiConfig::None`.
127///
128/// # Example
129///
130/// ```
131/// use tau_ports::target::wasi_map::{map_capability, Disposition};
132/// use tau_domain::fixtures::cap_fs_read;
133///
134/// let cap = cap_fs_read(&["/d"]);
135/// assert!(matches!(map_capability(&cap).disposition, Disposition::Wasi));
136/// ```
137pub fn map_capability(cap: &Capability) -> WasiMapping {
138    match cap {
139        Capability::Network(NetCapability::Http { hosts, methods, .. }) => WasiMapping {
140            imports: vec![
141                WitInterface::WasiHttpTypes,
142                WitInterface::WasiHttpOutgoingHandler,
143            ],
144            config: WasiConfig::AllowedHosts {
145                hosts: hosts.clone(),
146                methods: methods.clone(),
147            },
148            disposition: Disposition::Wasi,
149        },
150        Capability::Filesystem(FsCapability::Read { paths, .. }) => {
151            fs_preopen(paths.clone(), PreopenAccess::ReadOnly)
152        }
153        Capability::Filesystem(FsCapability::Write { paths, .. }) => {
154            fs_preopen(paths.clone(), PreopenAccess::ReadWrite)
155        }
156        Capability::Filesystem(FsCapability::Exec { .. }) => WasiMapping {
157            imports: Vec::new(),
158            config: WasiConfig::None,
159            disposition: Disposition::Unsupported {
160                reason: "wasm target has no exec surface",
161            },
162        },
163        Capability::Process(ProcessCapability::Spawn { .. }) => WasiMapping {
164            imports: Vec::new(),
165            config: WasiConfig::None,
166            disposition: Disposition::Unsupported {
167                reason: "wasm target cannot spawn OS processes",
168            },
169        },
170        Capability::Agent(AgentCapability::Spawn { .. })
171        | Capability::Skill(SkillCapability::Spawn { .. })
172        | Capability::TaskList { .. }
173        | Capability::Plan { .. } => WasiMapping {
174            imports: Vec::new(),
175            config: WasiConfig::None,
176            disposition: Disposition::InGuest,
177        },
178        Capability::Custom { .. } => WasiMapping {
179            imports: Vec::new(),
180            config: WasiConfig::None,
181            disposition: Disposition::HostMediated,
182        },
183        // Fail-closed: an unknown future capability (or future FsCapability /
184        // NetCapability / … variant, all `#[non_exhaustive]`) is NOT granted a
185        // WASI import. It maps to HostMediated so it can never silently reach
186        // the guest's WASI ABI.
187        _ => WasiMapping {
188            imports: Vec::new(),
189            config: WasiConfig::None,
190            disposition: Disposition::HostMediated,
191        },
192    }
193}

195/// Build a filesystem `Wasi` mapping (shared by fs.read / fs.write).
196fn fs_preopen(paths: Vec<String>, access: PreopenAccess) -> WasiMapping {
197    WasiMapping {
198        imports: vec![
199            WitInterface::WasiFilesystemTypes,
200            WitInterface::WasiFilesystemPreopens,
201        ],
202        config: WasiConfig::Preopens(vec![Preopen { paths, access }]),
203        disposition: Disposition::Wasi,
204    }
205}

207/// Whether a resolved preopen equals its capability, or is broader because
208/// WASI preopens are directory-granular. See the design doc's
209/// "fs-granularity divergence".
210#[non_exhaustive]
211#[derive(Debug, Clone, Copy, PartialEq, Eq)]
212pub enum PreopenGranularity {
213    /// Preopen == cap: an all-literal path (preopened as-is), or a literal
214    /// prefix + trailing `**` (the preopen equals the cap subtree).
215    Exact,
216    /// Preopen is broader than the cap: a whole-segment `*` glob whose
217    /// preopened literal-prefix directory admits more than the `*`-matched
218    /// set. The build gate (3.2/3.4) rejects this by default. A literal
219    /// single file is NOT widened — it is preopened as-is and fails at
220    /// apply-time if it is really a file.
221    WidenedToDir,
222}
223
224/// One preopen after glob→directory resolution: a host directory to hand the
225/// guest, its access mode, whether resolution widened the cap, and the
226/// originating cap glob(s) for diagnostics.
227#[derive(Debug, Clone, PartialEq, Eq)]
228pub struct ResolvedPreopen {
229    /// The absolute host directory the embedder will preopen.
230    pub host_dir: String,
231    /// Read-only (fs.read) or read-write (fs.write).
232    pub access: PreopenAccess,
233    /// Whether this preopen exactly matches the cap or widened it.
234    pub granularity: PreopenGranularity,
235    /// Original capability glob(s) that produced this preopen.
236    pub from: Vec<String>,
237}
238
239/// The `(host_dir, granularity)` a single normalized G2 pattern resolves to.
240///
241/// `host_dir` is the leading `Literal` prefix joined under `/` (empty prefix
242/// → `"/"`). Granularity is `WidenedToDir` iff the pattern contains any
243/// whole-segment `Star`; `StarStar` and all-literal are `Exact`.
244fn resolve_pattern(pat: &Pattern) -> (String, PreopenGranularity) {
245    let mut dir = String::new();
246    let mut widened = false;
247    let mut stopped = false; // true once we pass the first wildcard segment
248    for seg in &pat.0 {
249        match seg {
250            Segment::Literal(s) if !stopped => {
251                dir.push('/');
252                dir.push_str(s);
253            }
254            Segment::Star => {
255                widened = true;
256                stopped = true;
257            }
258            Segment::StarStar => {
259                stopped = true;
260            }
261            // A literal after the first wildcard does not extend host_dir.
262            Segment::Literal(_) => {}
263        }
264    }
265    if dir.is_empty() {
266        dir.push('/');
267    }
268    let granularity = if widened {
269        PreopenGranularity::WidenedToDir
270    } else {
271        PreopenGranularity::Exact
272    };
273    (dir, granularity)
274}

276/// The `ResolvedPreopen`s one raw fs-cap path yields. Non-G2 / malformed
277/// (`expand` → `None`) → empty (fail-closed drop). A brace glob yields one
278/// per expanded arm; each records the raw `path` in `from`.
279fn preopens_for_path(path: &str, access: PreopenAccess) -> Vec<ResolvedPreopen> {
280    match expand(path) {
281        None => Vec::new(),
282        Some(patterns) => patterns
283            .iter()
284            .map(|p| {
285                let (host_dir, granularity) = resolve_pattern(p);
286                ResolvedPreopen {
287                    host_dir,
288                    access: access.clone(),
289                    granularity,
290                    from: vec![path.to_string()],
291                }
292            })
293            .collect(),
294    }
295}

297/// The whole host WASI configuration folded from a capability set. Consumed
298/// by the wasm host embedder (3.2-paired work) to build a
299/// `wasmtime_wasi::WasiCtx`. All non-`Disposition::Wasi` caps contribute
300/// nothing.
301#[derive(Debug, Clone, PartialEq, Eq)]
302pub struct WasiConfiguration {
303    /// Egress allow-list: `host_union` of every `net.http` `hosts`. Absent
304    /// net.http caps yield `HostSet::Exact({})` (deny-all egress).
305    pub allowed_hosts: HostSet,
306    /// Allowed HTTP methods across all net.http caps. `None` = all methods
307    /// (absorbing); else the union of the per-cap sets. No net.http cap →
308    /// `None`.
309    pub methods: Option<BTreeSet<HttpMethod>>,
310    /// Deduplicated, glob-resolved preopens, sorted by `host_dir`. Empty
311    /// absent fs caps.
312    pub preopens: Vec<ResolvedPreopen>,
313}
314
315/// Fold a capability set into one host WASI configuration.
316///
317/// Total and pure: every set yields a [`WasiConfiguration`]. Calls
318/// [`map_capability`] per cap and folds the `Disposition::Wasi` fragments —
319/// host union, `None`-absorbing method union, and preopen dedup + glob→dir
320/// resolution. Fail-closed: no net.http cap → deny-all egress; no fs cap →
321/// no preopens; a non-G2 fs path is dropped.
322///
323/// # Example
324///
325/// ```
326/// use tau_ports::target::wasi_map::{resolve_wasi_config, PreopenAccess, PreopenGranularity};
327/// use tau_domain::fixtures::{cap_fs_read, cap_net_http};
328///
329/// let caps = [cap_net_http(&["api.example.com"], &["POST"]), cap_fs_read(&["/data/**"])];
330/// let cfg = resolve_wasi_config(&caps);
331/// assert_eq!(cfg.preopens.len(), 1);
332/// assert_eq!(cfg.preopens[0].host_dir, "/data");
333/// assert_eq!(cfg.preopens[0].access, PreopenAccess::ReadOnly);
334/// assert_eq!(cfg.preopens[0].granularity, PreopenGranularity::Exact);
335/// ```
336pub fn resolve_wasi_config<'a>(
337    caps: impl IntoIterator<Item = &'a Capability>,
338) -> WasiConfiguration {
339    let mut host_sets: Vec<HostSet> = Vec::new();
340    let mut any_net = false;
341    let mut methods_all = false;
342    let mut method_union: BTreeSet<HttpMethod> = BTreeSet::new();
343    let mut raw_preopens: Vec<ResolvedPreopen> = Vec::new();
344
345    for cap in caps {
346        match map_capability(cap).config {
347            WasiConfig::AllowedHosts { hosts, methods } => {
348                any_net = true;
349                host_sets.push(hosts);
350                match methods {
351                    None => methods_all = true,
352                    Some(s) => method_union.extend(s),
353                }
354            }
355            WasiConfig::Preopens(preopens) => {
356                for p in preopens {
357                    for path in &p.paths {
358                        raw_preopens.extend(preopens_for_path(path, p.access.clone()));
359                    }
360                }
361            }
362            WasiConfig::None => {}
363        }
364    }
365
366    // host_union over an empty iterator returns Exact(∅) = deny-all (host.rs:21).
367    let allowed_hosts = host_union(host_sets.iter());
368    let methods = if !any_net || methods_all {
369        None
370    } else {
371        Some(method_union)
372    };
373
374    WasiConfiguration {
375        allowed_hosts,
376        methods,
377        preopens: dedup_preopens(raw_preopens),
378    }
379}
380
381/// Merge preopens by host directory: RW absorbs RO, `WidenedToDir` absorbs
382/// `Exact`, `from` concatenated. `BTreeMap` yields a deterministic
383/// host_dir-sorted result. Nested directories stay separate (distinct keys).
384fn dedup_preopens(raw: Vec<ResolvedPreopen>) -> Vec<ResolvedPreopen> {
385    use alloc::collections::btree_map::Entry;
386    let mut by_dir: BTreeMap<String, ResolvedPreopen> = BTreeMap::new();
387    for p in raw {
388        match by_dir.entry(p.host_dir.clone()) {
389            Entry::Occupied(mut o) => {
390                let e = o.get_mut();
391                if p.access == PreopenAccess::ReadWrite {
392                    e.access = PreopenAccess::ReadWrite;
393                }
394                if p.granularity == PreopenGranularity::WidenedToDir {
395                    e.granularity = PreopenGranularity::WidenedToDir;
396                }
397                e.from.extend(p.from);
398            }
399            Entry::Vacant(v) => {
400                v.insert(p);
401            }
402        }
403    }
404    by_dir.into_values().collect()
405}

602            ("/**", "/", PreopenGranularity::Exact),         // whole FS, exact
603            ("/*", "/", PreopenGranularity::WidenedToDir),   // whole-segment star at root
604        ];
605        for (glob, dir, gran) in cases {
606            let pats = expand(glob).expect("G2 valid");
607            assert_eq!(pats.len(), 1, "{glob} brace-expands to one pattern");
608            let (host_dir, granularity) = resolve_pattern(&pats[0]);
609            assert_eq!(host_dir, dir, "host_dir for {glob}");
610            assert_eq!(granularity, gran, "granularity for {glob}");
611        }
612    }
