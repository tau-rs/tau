1//! Skill resolver port — looks up an installed skill by name and
2//! returns its install path, declared capabilities, and SKILL.md body.
3//!
4//! The kernel doesn't know how skills are stored on disk — that's a
5//! host-shell concern (tau-runtime-tokio ships a `tau_pkg`-backed impl
6//! that reads the scope lockfile + the skill's `tau.toml` + SKILL.md).
7//! Embassy/wasm guest shells with no on-disk package store can ship the
8//! [`NoSkillResolver`] (always `NotFound`).
9//!
10//! Routing skill resolution through a port is what lets
11//! `tau-runtime-core::Runtime` drive the `skill.<name>.spawn` virtual
12//! tool without linking `tau-pkg` (which pulls tokio/rustix and does not
13//! cross-compile to `wasm32-wasip2`).
14
15use alloc::string::String;
16use alloc::vec::Vec;
17
18use tau_domain::Capability;
19
20/// A resolved installed skill, ready for the kernel's
21/// `skill.<name>.spawn` dispatch.
22///
23/// Produced by [`SkillResolver::resolve`]; the kernel applies
24/// `${SKILL_DIR}` substitution, scope narrowing, and the capability
25/// subset law to these fields before spawning the child agent.
26///
27/// `install_path` is a `String` (not `PathBuf`) so the type stays usable
28/// in `no_std + alloc` guest shells; host adapters pass
29/// `path.display().to_string()`.
30#[derive(Debug, Clone)]
31pub struct ResolvedSkill {
32    /// Absolute path to the installed skill directory, as a display
33    /// string. Used as the `${SKILL_DIR}` substitution value.
34    pub install_path: String,
35    /// Declared capabilities from the skill's manifest (pre-substitution).
36    pub capabilities: Vec<Capability>,
37    /// The skill's default system prompt — the SKILL.md body, already
38    /// read and parsed by the adapter. The kernel uses this unless the
39    /// caller supplies a `system_prompt` override.
40    pub system_prompt: String,
41}
42
43/// Error returned by [`SkillResolver::resolve`].
44///
45/// Variants map onto the kernel's `OrchestrationError` skill variants so
46/// the kernel can surface a typed error without depending on the host's
47/// concrete error type (`tau_pkg::FindSkillError`).
48#[derive(Debug, Clone)]
49pub enum SkillResolveError {
50    /// No installed skill matches the requested name.
51    NotFound,
52    /// A lockfile entry exists but the install path is missing on disk.
53    InstallPathMissing {
54        /// The expected install path, as a display string.
55        expected_path: String,
56    },
57    /// The skill's manifest or SKILL.md could not be read/parsed, or the
58    /// scope itself could not be resolved.
59    Invalid {
60        /// Human-readable reason.
61        detail: String,
62    },
63}
64
65/// Resolve an installed skill by name.
66///
67/// Host shells implement this against their on-disk package store
68/// (tau-runtime-tokio ships `TauPkgSkillResolver`). Guest shells with no
69/// store ship [`NoSkillResolver`].
70pub trait SkillResolver: Send + Sync {
71    /// Look up `name` and return the resolved skill, or a typed error.
72    fn resolve(&self, name: &str) -> Result<ResolvedSkill, SkillResolveError>;
73}
74
75/// A [`SkillResolver`] that always reports [`SkillResolveError::NotFound`].
76///
77/// Ships for guest shells (wasm/embassy) that have no on-disk skill store
78/// but still link the kernel. A `skill.<name>.spawn` call then fails
79/// gracefully with a skill-not-installed error instead of panicking.
80#[derive(Debug, Clone, Copy, Default)]
81pub struct NoSkillResolver;
82
83impl SkillResolver for NoSkillResolver {
84    fn resolve(&self, _name: &str) -> Result<ResolvedSkill, SkillResolveError> {
85        Err(SkillResolveError::NotFound)
86    }
87}
