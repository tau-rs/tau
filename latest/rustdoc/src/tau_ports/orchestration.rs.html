1//! Entity types shared across the multi-agent orchestration layer.
2//!
3//! Lives in `tau-ports` so consumers (`tau-cli`, future serve-mode) can
4//! import the types without depending on the runtime kernel. Behavior
5//! (state transitions, locking, dispatch) lives in
6//! `tau-runtime::orchestration`.
7//!
8//! See `docs/superpowers/specs/2026-05-12-multi-agent-orchestration-design.md`
9//! for the design and `docs/decisions/0023-multi-agent-orchestration.md`
10//! for the ADR.

27/// Task lifecycle status.
28#[derive(Debug, Clone, Copy, PartialEq, Eq)]
29#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
30#[cfg_attr(feature = "serde", serde(rename_all = "snake_case"))]
31pub enum TaskStatus {
32    /// Created; no agent has claimed ownership yet.
33    Pending,
34    /// Claimed by an owner; lease is active; no work yet started.
35    Claimed,
36    /// Owner is actively executing.
37    InProgress,
38    /// Completed successfully.
39    Done,
40    /// Failed; owner reported an error.
41    Failed,
42    /// Explicitly accepted as orphan by the orchestrator (won't fail the run).
43    Discarded,
44}

206        /// Which budget.
207        budget: String,

209        final_value: u64,
210        /// Limit value.
211        limit: u64,
212    },
213    /// An agent completed normally.
214    Completion {
215        /// The agent.
216        agent_id: AgentId,
217        /// `"completed"` or `"failed"`.
218        status: String,
219    },
220    /// Run aborted by host (budget, watchdog, SIGINT).

222        /// Human-readable reason.
223        reason: String,
224    },
225    /// Orphan tasks present at root completion.
226    OrphanedTasksAtTermination {
227        /// Their ids.
228        task_ids: Vec<TaskId>,
229    },

291/// A resumable snapshot of an agent run taken at a completed turn boundary.
292///
293/// The full message history is carried so resume is "feed the history back
294/// in" — β.4's context pipeline is stateless and re-derives deterministically
295/// from the history, so no separate context-manager state is needed (ADR-0053
296/// D4). Token counts are plain `u64` (not [`crate::llm::TokenUsage`], which is
297/// `u32`-based) so the runtime's `u64` accounting round-trips without
298/// narrowing; the runtime reconstructs its own token type on resume.
299#[derive(Debug, Clone, PartialEq)]
300#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
301pub struct TurnCheckpoint {
302    /// Run this checkpoint belongs to (the `--resume` key).
303    pub run_id: RunId,
304    /// The turn number that just completed (1-based). Resume re-enters at
305    /// `turn + 1`.
306    pub turn: u32,
307    /// Full message history as of the end of `turn`.
308    pub history: Vec<tau_domain::Message>,
309    /// Cumulative input (prompt) tokens through `turn`.
310    pub input_tokens: u64,
311    /// Cumulative output (completion) tokens through `turn`.
312    pub output_tokens: u64,
313    /// Tools the model requested in `turn` that had **not** completed when
314    /// this snapshot was taken (`PerToolCall` mid-turn checkpoints only).
315    /// Empty for a `PerTurn` / turn-boundary checkpoint — serde-skipped so
316    /// those snapshots stay byte-identical to the A-minimal wire form.
317    /// On resume the runtime re-dispatches exactly these before the next
318    /// LLM call; carried explicitly because they are not derivable from
319    /// `history` (see ADR-0053 follow-up).
320    #[cfg_attr(
321        feature = "serde",
322        serde(default, skip_serializing_if = "Vec::is_empty")
323    )]
324    pub pending_tool_uses: Vec<crate::llm::ToolUse>,
325}

327/// Errors surfaced by a [`CheckpointStore`].
328///
329/// Messages are `String` (not `std::io::Error`) so the type stays
330/// no_std-clean; the host `FileCheckpointStore` maps I/O failures into
331/// [`CheckpointError::Io`].
332#[non_exhaustive]
333#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
334pub enum CheckpointError {
335    /// Reading or writing the durable store failed.
336    #[error("checkpoint store I/O failed: {0}")]
337    Io(String),
338    /// Encoding or decoding a checkpoint failed.
339    #[error("checkpoint (de)serialization failed: {0}")]
340    Serialization(String),
341}

343/// Port: persists and loads [`TurnCheckpoint`]s for durable agents (ADR-0053).
344///
345/// Injected into the runtime via `RunOptions`; the agent loop calls
346/// [`CheckpointStore::persist`] after each `TurnCompleted` when the agent is
347/// durable, and `tau run --resume` calls [`CheckpointStore::load_latest`]
348/// before the loop. Keeping this behind a port (rather than feature-gating
349/// file I/O inside the no_std kernel) lets the kill-and-resume test run
350/// in-core against an in-memory mock — see `crate::fixtures` under the
351/// `test-fixtures` feature.
352pub trait CheckpointStore: Send + Sync {
353    /// Durably commit a turn boundary. At-least-once: a side effect in the
354    /// turn may have already happened, so callers must be idempotent.
355    fn persist(&self, ckpt: &TurnCheckpoint) -> Result<(), CheckpointError>;
356
357    /// Load the latest (highest-`turn`) checkpoint for `run_id`, or `None`
358    /// if the run has no committed checkpoint.
359    fn load_latest(&self, run_id: &RunId) -> Result<Option<TurnCheckpoint>, CheckpointError>;
360}

362/// A resumable snapshot of a pipeline paused at a top-level `Suspend` step.
363///
364/// Distinct from [`TurnCheckpoint`] (agent-turn durability): this carries the
365/// pipeline `OutputStore` snapshot + the step cursor, not message history. Both
366/// share the `run_id` handle and the `.tau/runs/<run_id>/` directory. Resume is
367/// restore-and-continue: rehydrate `outputs`, jump to `step_cursor + 1`.
368#[derive(Debug, Clone, PartialEq)]
369#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
370pub struct PipelineSuspension {
371    /// Run this suspension belongs to (the `--resume` key).
372    pub run_id: RunId,
373    /// Signal the resumer must match to continue (`--signal`).
374    pub resume_signal: String,
375    /// Index of the `Suspend` step in the top-level pipeline slice. Resume
376    /// re-enters at `step_cursor + 1`.
377    pub step_cursor: usize,
378    /// The `Suspend` step's id (for the "paused at <id>" human message).
379    pub step_id: String,
380    /// Canonical-IR SHA-256 of the module at pause time (`"sha256:" + hex`).
381    /// Resume rejects a project that changed since the pause.
382    pub ir_digest: String,
383    /// The `OutputStore` snapshot as of the pause (step id -> output value).
384    pub outputs: BTreeMap<String, serde_json::Value>,
385    /// Per-check retry-attempt counts accumulated up to the pause (check id ->
386    /// count). Carried across resume so a `Check` whose `retry.gate` sits
387    /// *before* the `Suspend` step cannot reset its attempt budget on every
388    /// resume (which would let it rewind→re-suspend forever without
389    /// `max_attempts` ever tripping). Absent in pre-followup snapshots, so it
390    /// defaults to empty on deserialize.
391    #[cfg_attr(feature = "serde", serde(default))]
392    pub attempts: BTreeMap<String, u32>,
393}

395/// Port: persists and loads a pipeline [`PipelineSuspension`] for HITL resume.
396///
397/// One live suspension per run (a second `Suspend` on resume overwrites it; a
398/// completed run removes it). Keyed by the same `RunId` as [`CheckpointStore`]
399/// and stored in the same run directory, so one `--resume <run_id>` handle
400/// covers both agent-turn and pipeline-step resume.
401pub trait SuspensionStore: Send + Sync {
402    /// Durably record the pause point. Overwrites any prior suspension for the
403    /// same `run_id`.
404    fn persist_suspension(&self, s: &PipelineSuspension) -> Result<(), CheckpointError>;
405
406    /// Load the current suspension for `run_id`, or `None` if the run is not
407    /// paused.
408    fn load_suspension(
409        &self,
410        run_id: &RunId,
411    ) -> Result<Option<PipelineSuspension>, CheckpointError>;
412
413    /// Remove the suspension for `run_id`, if any. Idempotent: absent = `Ok(())`.
414    /// Called after a resumed run completes so a stale snapshot cannot be
415    /// re-resumed (which would re-run the post-suspend steps). Default no-op for
416    /// stores that do not need cleanup.
417    fn delete_suspension(&self, run_id: &RunId) -> Result<(), CheckpointError> {
418        let _ = run_id;
419        Ok(())
420    }
421}
