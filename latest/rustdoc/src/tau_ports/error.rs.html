1//! Per-concern error enums for `tau-ports`.
2//!
3//! Each error type is `#[non_exhaustive]` so additive variants are non-breaking.
4//! All errors derive `Debug + Error + Clone + PartialEq + Eq`; tests with
5//! free-form `String` fields use `matches!()` to avoid brittle wording
6//! comparisons.
7//!
8//! `LlmError`, `ToolError`, `StorageError`, and `CapabilityError` are the
9//! per-trait error types. `NamespaceError` and `KeyError` are the validation
10//! errors for the `Namespace` and `Key` newtypes.

21/// Errors returned by `crate::llm::LlmBackend` implementations.
22///
23/// Variants partition transport, protocol, and provider failures so adapters
24/// can map provider-specific errors to a uniform surface. Use
25/// [`LlmError::is_retryable`] for the default retry-policy hint.
26#[non_exhaustive]
27#[derive(Debug, Clone, PartialEq, Eq, Error)]
28#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
29pub enum LlmError {
30    /// The request was malformed or violated provider constraints.
31    #[error("invalid request: {reason}")]
32    InvalidRequest {
33        /// Human-readable reason the request was rejected.
34        reason: String,
35    },
36    /// The provider rate-limited the caller.
37    #[error("rate limited: retry after {retry_after_seconds:?}s")]
38    RateLimited {
39        /// Suggested wait, in seconds, before retrying. `None` if the provider
40        /// did not signal one.
41        retry_after_seconds: Option<u32>,
42    },
43    /// Authentication failed (bad credentials, missing API key, etc.).
44    #[error("authentication failed: {message}")]
45    Auth {
46        /// Provider-supplied or adapter-synthesized auth-failure detail.
47        message: String,
48    },
49    /// Network/transport failure reaching the provider.
50    #[error("transport: {message}")]
51    Transport {
52        /// Description of the underlying transport failure.
53        message: String,
54    },
55    /// Mid-stream error (only emitted from `CompletionStream` items, never
56    /// from the return of `complete()`).
57    #[error("stream error: {message}")]
58    Stream {
59        /// Description of the stream-time failure.
60        message: String,
61    },
62    /// The provider returned a server-side error not covered by other variants.
63    #[error("provider error: {message}")]
64    Provider {
65        /// Provider-supplied error detail.
66        message: String,
67    },
68    /// The requested feature is not supported by this backend.
69    #[error("unsupported: {what}")]
70    Unsupported {
71        /// Description of the unsupported feature or option.
72        what: String,
73    },
74    /// Plugin internal error.
75    ///
76    /// See: [escape-hatches.md#llmerror-internal](../docs/explanation/escape-hatches.md#llmerror-internal).
77    #[error("internal: {message}")]
78    Internal {
79        /// Free-form internal-error message; not part of the stable API surface.
80        message: String,
81    },
82}
83
84impl LlmError {
85    /// Heuristic: is the error likely transient? Default-policy hint;
86    /// nuanced policies should match on variants directly.
87    pub fn is_retryable(&self) -> bool {
88        matches!(
89            self,
90            LlmError::RateLimited { .. }
91                | LlmError::Transport { .. }
92                | LlmError::Stream { .. }
93                | LlmError::Provider { .. },
94        )
95    }
96}

98/// Errors returned by [`crate::storage::Storage`] implementations.
99///
100/// Backend-time rejection of a `Namespace`/`Key` is distinct from
101/// construction-time validation: see [`NamespaceError`] / [`KeyError`] for the
102/// latter. Use [`StorageError::is_retryable`] for the default retry-policy
103/// hint.
104#[non_exhaustive]
105#[derive(Debug, Clone, PartialEq, Eq, Error)]
106#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
107pub enum StorageError {
108    /// Backend rejected the namespace (length cap, reserved prefix, charset).
109    #[error("invalid namespace: {reason}")]
110    InvalidNamespace {
111        /// Backend-supplied reason the namespace was rejected.
112        reason: String,
113    },
114    /// Backend rejected the key (length, charset, reserved prefix).
115    #[error("invalid key: {reason}")]
116    InvalidKey {
117        /// Backend-supplied reason the key was rejected.
118        reason: String,
119    },
120    /// Backend is currently unavailable (connection refused, degraded mode).
121    #[error("unavailable: {message}")]
122    Unavailable {
123        /// Description of the availability failure.
124        message: String,
125    },
126    /// Operation timed out.
127    #[error("timeout")]
128    Timeout,
129    /// The requested operation is not supported by this backend.
130    #[error("unsupported: {what}")]
131    Unsupported {
132        /// Description of the unsupported operation.
133        what: String,
134    },
135    /// Plugin internal error.
136    ///
137    /// See: [escape-hatches.md#storageerror-internal](../docs/explanation/escape-hatches.md#storageerror-internal).
138    #[error("internal: {message}")]
139    Internal {
140        /// Free-form internal-error message; not part of the stable API surface.
141        message: String,
142    },
143}
144
145impl StorageError {
146    /// Heuristic: is the error likely transient? Default-policy hint;
147    /// nuanced policies should match on variants directly.
148    pub fn is_retryable(&self) -> bool {
149        matches!(
150            self,
151            StorageError::Unavailable { .. } | StorageError::Timeout,
152        )
153    }
154}

156/// Errors returned by [`crate::capability_gate::CapabilityGate`] implementations.
157///
158/// Stable as of v0.1 of the sandboxing sub-project. Variant evolution
159/// is handled by `#[non_exhaustive]` at the enum level and on each
160/// struct-style variant.
161#[non_exhaustive]
162#[derive(Debug, Clone, PartialEq, Eq, Error)]
163#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
164pub enum CapabilityError {
165    /// The adapter is not usable on this host (probe returned Unavailable).
166    #[error("sandbox unavailable: {reason}")]
167    Unavailable {
168        /// Reason from [`crate::capability_gate::CapabilityProbe::Unavailable`].
169        reason: String,
170    },
171    /// The plan requires a capability shape this adapter does not support.
172    #[error("sandbox: unsupported shape {shape:?}")]
173    ShapeUnsupported {
174        /// The shape that was rejected.
175        shape: tau_domain::CapabilityShape,
176    },
177    /// The adapter could not apply sandbox enforcement to the spawn.
178    /// (Examples: landlock syscall failed, seccomp filter compile failed,
179    /// `docker run` returned non-zero.)
180    #[error("sandbox wrap-spawn failed: {message}")]
181    WrapFailed {
182        /// Free-form diagnostic; not part of the stable API surface.
183        message: String,
184    },
185    /// Runtime sandbox violation reported by the kernel
186    /// (SIGSYS from seccomp, EACCES from landlock, etc).
187    #[error("sandbox violation: {detail}")]
188    Violation {
189        /// Detail about the violating syscall / path / host.
190        detail: String,
191    },
192    /// The requested feature is not supported by this sandbox.
193    #[error("sandbox unsupported: {what}")]
194    Unsupported {
195        /// Description of the unsupported feature.
196        what: String,
197    },
198    /// A configured resource limit was exceeded.
199    #[error("sandbox limit exceeded: {limit}")]
200    LimitExceeded {
201        /// Identifier of the limit that was exceeded.
202        limit: String,
203    },
204    /// Sandbox proxy failed to set up or relay a connection.
205    /// Includes proxy task spawn errors, allow-list violations surfaced
206    /// from the bridge, etc.
207    #[error("sandbox proxy: {message}")]
208    Proxy {
209        /// Free-form message including the failure context.
210        message: String,
211    },
212    /// Plugin internal error.
213    ///
214    /// See: [escape-hatches.md#sandboxerror-internal](../docs/explanation/escape-hatches.md#sandboxerror-internal).
215    #[error("sandbox internal: {message}")]
216    Internal {
217        /// Free-form internal-error message; not part of the stable API surface.
218        message: String,
219    },
220}

222/// Errors returned by [`crate::tool::Tool`] implementations.
223///
224/// `ToolError` composes [`LlmError`] and [`StorageError`] via `#[from]` so
225/// tools that internally drive an LLM or storage backend can propagate via
226/// `?`. Direction is unidirectional (Tool → LlmError/StorageError, not vice
227/// versa) to avoid cycles and preserve layering. Use
228/// [`ToolError::is_retryable`] for the default retry-policy hint; it
229/// delegates to the inner predicate for `Llm`/`Storage` and returns `false`
230/// otherwise.
231#[non_exhaustive]
232#[derive(Debug, Clone, PartialEq, Eq, Error)]
233#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
234pub enum ToolError {
235    /// Caller passed arguments the tool rejected.
236    #[error("bad args: {reason}")]
237    BadArgs {
238        /// Human-readable reason the arguments were rejected.
239        reason: String,
240    },
241    /// The tool's session is no longer usable; the caller should reopen it.
242    #[error("session unusable: {reason}")]
243    SessionDead {
244        /// Reason the session was declared dead.
245        reason: String,
246    },
247    /// The tool exceeded its deadline.
248    #[error("deadline exceeded")]
249    DeadlineExceeded,
250    /// The tool requires a capability that the caller is not granted.
251    #[error("capability denied: {capability}")]
252    CapabilityDenied {
253        /// Identifier of the denied capability.
254        capability: String,
255    },
256    /// Underlying LLM call failed (for tools that internally use an LLM).
257    #[error("llm: {0}")]
258    Llm(#[from] LlmError),
259    /// Underlying storage operation failed.
260    #[error("storage: {0}")]
261    Storage(#[from] StorageError),
262    /// Plugin internal error.
263    ///
264    /// See: [escape-hatches.md#toolerror-internal](../docs/explanation/escape-hatches.md#toolerror-internal).
265    #[error("internal: {message}")]
266    Internal {
267        /// Free-form internal-error message; not part of the stable API surface.
268        message: String,
269    },
270}
271
272impl ToolError {
273    /// Heuristic: is the error likely transient? Most `ToolError` variants
274    /// are NOT retryable — `SessionDead` means reopen the session, `BadArgs`
275    /// is permanent, etc. Composed `Llm`/`Storage` errors delegate to their
276    /// inner predicate.
277    pub fn is_retryable(&self) -> bool {
278        match self {
279            ToolError::Llm(e) => e.is_retryable(),
280            ToolError::Storage(e) => e.is_retryable(),
281            _ => false,
282        }
283    }
284}

286/// Validation errors for [`crate::storage::Namespace`].
287///
288/// # Example
289///
290/// ```
291/// use tau_ports::{NamespaceError, storage::Namespace};
292///
293/// let err = Namespace::try_new("").unwrap_err();
294/// assert_eq!(err, NamespaceError::Empty);
295/// ```
296#[non_exhaustive]
297#[derive(Debug, Clone, PartialEq, Eq, Error)]
298#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
299pub enum NamespaceError {
300    /// The input was empty.
301    #[error("namespace is empty")]
302    Empty,
303    /// The input exceeded the byte cap.
304    #[error("namespace exceeds {max} bytes: got {got}")]
305    TooLong {
306        /// Maximum permitted length, in bytes.
307        max: usize,
308        /// Actual length of the input, in bytes.
309        got: usize,
310    },
311    /// The input contained a NUL byte or control character.
312    #[error("namespace contains invalid byte (NUL or control char) at position {pos}")]
313    InvalidByte {
314        /// Byte position in the input.
315        pos: usize,
316    },
317}

482/// Errors raised while resolving a credential through a provider or chain.
483///
484/// Chain walk semantics distinguish `Ok(None)` ("not here, try the next
485/// provider") from `Err` ("this provider owns the request but failed").
486/// The chain fails fast on any `Err`.
487#[non_exhaustive]
488#[derive(Debug, Clone, PartialEq, Eq, Error)]
489#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
490pub enum CredentialError {
491    /// The whole chain was walked and no provider held the credential.
492    #[error("credential not found: {id}")]
493    NotFound {
494        /// The logical credential id that was not found.
495        id: String,
496    },
497    /// A configured provider could not be reached (e.g. Vault down).
498    #[error("credential provider {provider} unavailable: {reason}")]
499    ProviderUnavailable {
500        /// Provider name.
501        provider: String,
502        /// Human-readable reason.
503        reason: String,
504    },
505    /// A provider found the credential but its content was malformed.
506    #[error("credential {id} malformed: {reason}")]
507    Malformed {
508        /// The logical credential id.
509        id: String,
510        /// Human-readable reason.
511        reason: String,
512    },
513    /// An I/O error occurred while resolving (e.g. unreadable secret file).
514    #[error("credential I/O error: {reason}")]
515    Io {
516        /// Human-readable reason.
517        reason: String,
518    },
519    /// An unexpected internal error.
520    ///
521    /// See: [escape-hatches.md#credentialerror-internal](../docs/explanation/escape-hatches.md#credentialerror-internal).
522    #[error("internal credential error: {reason}")]
523    Internal {
524        /// Human-readable reason.
525        reason: String,
526    },
