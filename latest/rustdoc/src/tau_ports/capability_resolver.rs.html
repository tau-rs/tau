1//! Capability resolver port — turns a package manifest's declared
2//! capabilities into the post-override grant set that flows into the
3//! agent loop.
4//!
5//! The kernel doesn't know about project tau.toml overrides — that's
6//! a host-shell concern (tau-runtime ships a `tau_pkg`-backed impl).
7//! Embassy/wasm shells with no override system can either supply a
8//! passthrough impl, or omit the resolver entirely (the kernel falls
9//! back to the manifest's declared capabilities unchanged).
10//!
11//! Routing capability resolution through a port is what lets
12//! `tau-runtime-core::Runtime` drive the agent loop without linking
13//! `tau-pkg` (which is std-only).
14

22/// Resolved capability grant set after applying project-level overrides
23/// to a package manifest's declared capabilities.
24///
25/// Produced by [`CapabilityResolver::resolve`] and stuffed into
26/// `RunOptions` by the host shell; the kernel reads it directly when
27/// dispatching tool calls.
28///
29/// Three views, one per consumer:
30/// - `for_kernel` — the un-narrowed source capabilities; the kernel's
31///   structural capability check (`fs.read ⊇ fs.read`) reads these.
32/// - `for_session` — the narrowed views handed to plugins via
33///   `SessionContext.granted_capabilities`. Same shape as `for_kernel`
34///   but with allow-lists narrowed to the project override.
35/// - `deny_entries` — explicit deny carve-outs, one entry per
36///   capability kind with a non-empty deny list.
37#[derive(Debug, Clone)]
38pub struct ResolvedGrants {
39    /// Un-narrowed source capabilities — fed into the kernel's
40    /// structural capability check.
41    pub for_kernel: Vec<Capability>,
42    /// Narrowed views handed to plugins via session context.
43    pub for_session: Vec<Capability>,
44    /// Per-kind deny carve-outs.
45    pub deny_entries: Vec<DenyEntry>,
46}
47
48/// Resolve a package manifest's declared capabilities into the
49/// effective grant set for an agent run.
50///
51/// Host shells implement this against their override system:
52/// - tau-runtime ships `TauPkgCapabilityResolver` wrapping
53///   `tau_pkg::capability_override::compute_effective`.
54/// - Embassy / wasm shells with no override layer can ship a noop impl
55///   that maps the input through to `for_kernel = for_session =
56///   manifest_capabilities; deny_entries = []`.
57///
58/// Errors are reported as [`CapabilityResolverError`] (kind + reason)
59/// so the kernel can surface them as `CapabilityOverrideExpands` without
60/// depending on the host's concrete error type.
61pub trait CapabilityResolver: Send + Sync {
62    /// Compute the post-override grant set for a run.
63    fn resolve(
64        &self,
65        manifest_capabilities: &[Capability],
66    ) -> Result<ResolvedGrants, CapabilityResolverError>;
67}
68
69/// Error returned by [`CapabilityResolver::resolve`] when an override
70/// would expand the package manifest's grant set (forbidden by the
71/// subset law).
72///
73/// `kind` is the capability discriminator (e.g. `"fs.read"`); `reason`
74/// is a human-readable explanation suitable for the runtime error's
75/// `Display` impl.
76#[derive(Debug, Clone)]
77pub struct CapabilityResolverError {
78    /// Capability kind that triggered the rejection (`fs.read`,
79    /// `fs.write`, `fs.exec`, `net.http`, `process.spawn`, ...).
80    pub kind: String,
81    /// Human-readable reason — e.g. the offending glob and the
82    /// override path.
83    pub reason: String,
84}

86#[cfg(test)]
87mod tests {
88    use super::*;
89    use alloc::sync::Arc;
90    use alloc::vec;
91
92    struct PassthroughResolver;
93
94    impl CapabilityResolver for PassthroughResolver {
95        fn resolve(
96            &self,
97            manifest_capabilities: &[Capability],
98        ) -> Result<ResolvedGrants, CapabilityResolverError> {
99            Ok(ResolvedGrants {
100                for_kernel: manifest_capabilities.to_vec(),
101                for_session: manifest_capabilities.to_vec(),
102                deny_entries: vec![],
103            })
104        }
105    }
106
107    #[test]
108    fn passthrough_resolver_round_trips() {
109        let r: Arc<dyn CapabilityResolver> = Arc::new(PassthroughResolver);
110        let resolved = r.resolve(&[]).expect("resolve");
111        assert!(resolved.for_kernel.is_empty());
112        assert!(resolved.for_session.is_empty());
113        assert!(resolved.deny_entries.is_empty());
114    }
