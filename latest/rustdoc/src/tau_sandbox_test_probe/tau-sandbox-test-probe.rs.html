1//! In-container test probe for tau-sandbox-windows integration tests.
2//!
3//! Pure std. Modes (argv[1]):
4//! - `http-get <url>` — proxy-form (absolute-URI) GET through the proxy
5//!   named by the HTTP_PROXY env var, exactly as `tau-net-bridge-win`
6//!   sets it for the wrapped plugin. Prints the response's status line;
7//!   exit 0 iff the response is 200.
8//! - `read-file <path>` — read a file; exit 0 on success. Regression
9//!   guard for leaf-only ACL reachability (spike #626 H3:
10//!   AppContainers retain bypass-traverse-checking).
11//! - `pipe-open <name>` — open `\\.\pipe\<name>` read+write; exit 0 on
12//!   success. Used by the foreign-container pipe-access control test.
13//! - `spawn <prog> [args...]` — spawn `prog` as a CHILD of this
14//!   already-in-container process (stdio inherited) and report whether
15//!   image activation succeeded. See [`spawn_child`].
16//!
17//! An 8s watchdog hard-exits with code 9 so a broken chain never hangs
18//! the CI job.
19
20use std::io::{Read, Write};
21
22fn main() {
23    // stderr canary. Result markers go to STDOUT, so a run where the
24    // stdout markers arrive but this line does not proves the launcher
25    // -> bridge -> probe chain loses stderr — which would also mean the
26    // bridge's own `BRIDGE ` diagnostics never reach CI. Distinguishing
27    // "the bridge printed nothing" from "nothing the bridge prints can
28    // be seen" is otherwise a whole CI round.
29    eprintln!("PROBE stderr-alive");
30    std::thread::spawn(|| {
31        std::thread::sleep(std::time::Duration::from_secs(8));
32        println!("PROBE result=watchdog-timeout");
33        std::process::exit(9);
34    });
35    let args: Vec<String> = std::env::args().skip(1).collect();
36    match (args.first().map(String::as_str), args.get(1)) {
37        (Some("http-get"), Some(url)) => http_get(url),
38        (Some("read-file"), Some(path)) => read_file(path),
39        (Some("pipe-open"), Some(name)) => pipe_open(name),
40        (Some("spawn"), Some(prog)) => spawn_child(prog, &args[2..]),
41        _ => {
42            eprintln!(
43                "usage: tau-sandbox-test-probe http-get <url> | read-file <path> | pipe-open <name> | spawn <prog> [args...]"
44            );
45            std::process::exit(2);
46        }
47    }
48}

50/// Open `\\.\pipe\<name>` read+write; exit 0 on success, 1 on error.
51fn pipe_open(name: &str) -> ! {
52    let path = format!(r"\\.\pipe\{name}");
53    match std::fs::OpenOptions::new()
54        .read(true)
55        .write(true)
56        .open(&path)
57    {
58        Ok(_) => {
59            println!("PROBE result=ok detail=opened {path}");
60            std::process::exit(0);
61        }
62        Err(e) => {
63            println!("PROBE result=err detail=open {path}: {e}");
64            std::process::exit(1);
65        }
66    }
67}

69/// Spawn `prog args...` as a child of *this* process, inheriting stdio,
70/// and report whether the OS could activate it.
71///
72/// # Why this mode exists (#622 FAILURE B)
73///
74/// `install_rust_cargo_acceptance` fails with `CreateProcess` of
75/// `rustc.exe` returning `Access is denied (os error 5)`. That
76/// activation is performed by a process **already inside** the
77/// AppContainer (cargo), so the image-file access check runs against the
78/// AppContainer token. Every other execution in this suite is started by
79/// `tau-appcontainer-launcher`, which runs as the *host user* — for
80/// those, the image access check need not involve the container's SID at
81/// all. So "the launcher could run this exe" does not imply "a process
82/// inside the container can run this exe", and only this mode exercises
83/// the latter.
84///
85/// Markers: `PROBE result=spawn-exit code=Some(N)` on successful
86/// activation (the child's own markers appear on the inherited stdout
87/// just before it), `PROBE result=err detail=spawn <prog>: <e>` if
88/// creation itself failed — which is where an `Access is denied
89/// (os error 5)` would surface.
90fn spawn_child(prog: &str, args: &[String]) -> ! {
91    match std::process::Command::new(prog).args(args).status() {
92        Ok(st) => {
93            println!("PROBE result=spawn-exit code={:?}", st.code());
94            std::process::exit(if st.success() { 0 } else { 1 });
95        }
96        Err(e) => {
97            println!("PROBE result=err detail=spawn {prog}: {e}");
98            std::process::exit(1);
99        }
100    }
101}

103/// GET `url` (plain http) through the HTTP_PROXY proxy, absolute-URI
104/// form — exactly the request shape tau-sandbox-proxy's handle_http
105/// expects (and what `tau-net-bridge-win` gives the wrapped plugin via
106/// HTTP_PROXY).
107fn http_get(url: &str) -> ! {
108    let proxy = std::env::var("HTTP_PROXY").unwrap_or_default();
109    let Some(hostport) = proxy.strip_prefix("http://") else {
110        println!("PROBE result=err detail=no-http-proxy-env");
111        std::process::exit(1);
112    };
113    eprintln!("PROBE proxy={proxy}");
114    let host = url
115        .strip_prefix("http://")
116        .and_then(|r| r.split('/').next())
117        .unwrap_or_default();
118    let mut conn = match std::net::TcpStream::connect(hostport) {
119        Ok(c) => c,
120        Err(e) => {
121            println!("PROBE result=err detail=proxy-connect: {e}");
122            std::process::exit(1);
123        }
124    };
125    conn.set_read_timeout(Some(std::time::Duration::from_secs(5)))
126        .ok();
127    let req = format!("GET {url} HTTP/1.1\r\nHost: {host}\r\nConnection: close\r\n\r\n");
128    if let Err(e) = conn.write_all(req.as_bytes()) {
129        println!("PROBE result=err detail=write: {e}");
130        std::process::exit(1);
131    }
132    // Read incrementally rather than `read_to_end` so the *reason* the
133    // read ended is reportable. `read_to_end` returns `Err` on the
134    // socket's read timeout and drops whatever the failing iteration had
135    // buffered, which made "the proxy answered nothing" and "the proxy
136    // answered but never closed" produce the identical, useless
137    // `PROBE result=status detail=` (empty) seen in #622 CI round 2.
138    let mut resp = Vec::new();
139    let mut chunk = [0u8; 4096];
140    let read_end;
141    loop {
142        match conn.read(&mut chunk) {
143            Ok(0) => {
144                read_end = "eof".to_string();
145                break;
146            }
147            Ok(n) => resp.extend_from_slice(&chunk[..n]),
148            Err(e) => {
149                read_end = format!("err:{e}");
150                break;
151            }
152        }
153    }
154    let head = String::from_utf8_lossy(&resp);
155    let status = head.lines().next().unwrap_or("");
156    // Diagnostics on stderr; the stdout marker below stays byte-stable
157    // because the integration tests match it literally.
158    eprintln!(
159        "PROBE http bytes={} read-end={read_end} raw={:?}",
160        resp.len(),
161        head.chars().take(200).collect::<String>()
162    );
163    println!("PROBE result=status detail={status}");
164    if status.contains(" 200 ") {
165        std::process::exit(0);
166    }
167    std::process::exit(1);
168}

170/// Read `path` to a string; exit 0 on success, 1 on error.
171fn read_file(path: &str) -> ! {
172    match std::fs::read_to_string(path) {
173        Ok(s) => {
174            println!("PROBE result=ok detail=read {} bytes", s.len());
175            std::process::exit(0);
176        }
177        Err(e) => {
178            println!("PROBE result=err detail=read {path}: {e}");
179            std::process::exit(1);
180        }
181    }
