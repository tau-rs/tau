1//! Toy `LlmBackend` plugin that replays canned responses from config.
2//!
3//! Used by tau-cli integration tests to exercise the plugin loading
4//! mechanism end-to-end without depending on a real LLM provider.
5//!
6//! # Configuration
7//!
8//! Configurable via the handshake `config` field (set in
9//! `[agents.<id>.config]` of the project tau.toml):
10//!
11//! - `canned_text: String` — single canned text returned by every
12//!   `llm.complete` call. Default: empty string.
13//! - `script: Vec<String>` — multi-turn script. Indexed by an internal
14//!   atomic counter that increments on each `complete` call. If the
15//!   counter exceeds the script length, falls back to `canned_text`.
16//! - `crash_after_handshake: bool` — if `true`, panic at the start of
17//!   any `complete`/`stream` call. The handshake itself completes; the
18//!   panic surfaces at first dispatch. Used by failure-path tests.
19//! - `delay_response_ms: Option<u64>` — sleep this many milliseconds
20//!   before responding. Used by handshake/timeout tests.
21//! - `error_on_method: Option<String>` — return `Err(LlmError::Internal)`
22//!   when this method is called (e.g. `"llm.complete"` or
23//!   `"llm.stream"`).
24//! - `tool_calls: Vec<Vec<ScriptedToolCall>>` — per-turn scripted tool
25//!   calls, indexed by the SAME atomic turn counter as `script`. When
26//!   turn `i` has a non-empty entry, that turn returns
27//!   `StopReason::ToolUse` with those calls (and empty text) instead of
28//!   the usual `script`/`canned_text` + `StopReason::EndTurn`. Turns
29//!   with no scripted calls (including every turn of a config that
30//!   never sets this field — the default is an empty `Vec`) are exactly
31//!   today's text-only behavior. Each entry is `{ name, args }`; `args`
32//!   defaults to `null`.
33//! - `canned_tool_call: Option<{ name, input }>` — on the FIRST
34//!   `llm.complete` call only, emit a tool-use block for `name` with
35//!   `input` and `StopReason::ToolUse`. The agent loop dispatches the
36//!   tool and calls back; turn 2 returns text as usual, ending the run.
37//!   Lets a test drive a real tool round trip without a live provider.
38//!   Checked only when `tool_calls` has no scripted entry for that turn
39//!   (`tool_calls` takes precedence when both are configured).

56/// One scripted tool call for a turn (config-driven).
57#[derive(Debug, Clone, Default, Deserialize)]
58struct ScriptedToolCall {
59    /// Tool name the mock backend claims to call (e.g.
60    /// `"weather.get_forecast"`).
61    #[serde(default)]
62    name: String,
63    /// Tool arguments, passed through as-is. Defaults to `null`.
64    #[serde(default)]
65    args: serde_json::Value,
66}
67
68/// Static configuration consumed from the handshake `config` field.
69#[derive(Debug, Default, Deserialize)]
70struct EchoConfig {
71    /// Single canned text returned by every `llm.complete` call.
72    #[serde(default)]
73    canned_text: String,
74    /// Multi-turn script; indexed by an atomic turn counter.
75    #[serde(default)]
76    script: Vec<String>,
77    /// If `true`, panic at the start of any `complete`/`stream` call.
78    #[serde(default)]
79    crash_after_handshake: bool,
80    /// Sleep this many milliseconds before responding. Used by
81    /// handshake/timeout tests.
82    #[serde(default)]
83    delay_response_ms: Option<u64>,
84    /// Return `Err(LlmError::Internal)` when this method is called.
85    #[serde(default)]
86    error_on_method: Option<String>,
87    /// Per-turn scripted tool calls; see the module docs. Empty (the
88    /// default) preserves exactly today's text-only output for every
89    /// existing config.
90    #[serde(default)]
91    tool_calls: Vec<Vec<ScriptedToolCall>>,
92    /// Emit this tool call on the first `llm.complete` turn (checked only
93    /// when `tool_calls` has no scripted entry for that turn).
94    #[serde(default)]
95    canned_tool_call: Option<CannedToolCall>,
96}
97
98/// A tool-use block for the plugin to emit on its first completion turn.
99#[derive(Debug, Deserialize)]
100struct CannedToolCall {
101    /// Tool name — must match a `ToolSpec::name` the agent was given.
102    name: String,
103    /// Arguments passed to the tool. Defaults to `null`.
104    #[serde(default = "null_value")]
105    input: tau_domain::Value,
106}
107
108/// `tau_domain::Value` has no `Default` impl; serde needs one for `input`.
109fn null_value() -> tau_domain::Value {
110    tau_domain::Value::Null
111}

113/// Toy `LlmBackend` plugin.
114struct EchoLlm {

117}

130impl EchoLlm {
131    /// Apply the test-only side effects (`crash_after_handshake`,
132    /// `error_on_method`, `delay_response_ms`) and produce the next
133    /// turn's response shape: `(turn_index, text, tool_uses, stop_reason)`.
134    /// Shared by `complete` and `stream` via [`Self::next_response`].
135    ///
136    /// Reads the turn counter exactly ONCE (`fetch_add`) so the
137    /// `tool_calls` and `script` lookups for this turn stay
138    /// index-synchronized — a second `fetch_add` here would silently
139    /// desync the two scripts.
140    ///
141    /// When `tool_calls[i]` is non-empty, this turn returns those calls
142    /// with `StopReason::ToolUse` and empty text — mirroring how a real
143    /// provider replies to a tool-use turn. Otherwise (including every
144    /// turn of a config that never sets `tool_calls`) it falls back to
145    /// exactly today's behavior: `script[i]` (or `canned_text`) with
146    /// `StopReason::EndTurn`.
147    async fn next_turn(
148        &self,
149        method: &str,
150    ) -> Result<(usize, String, Vec<ToolUse>, StopReason), LlmError> {
151        if self.config.crash_after_handshake {
152            panic!("echo-llm crash_after_handshake = true (test-only mode)");
153        }
154        if self.config.error_on_method.as_deref() == Some(method) {
155            return Err(LlmError::Internal {
156                message: format!("echo-llm error_on_method test mode tripped on {method}"),
157            });
158        }
159        if let Some(ms) = self.config.delay_response_ms {
160            tokio::time::sleep(Duration::from_millis(ms)).await;
161        }
162        let i = self.turn.fetch_add(1, Ordering::Relaxed);
163
164        let scripted_calls = self.config.tool_calls.get(i).cloned().unwrap_or_default();
165        let text = self
166            .config
167            .script
168            .get(i)
169            .cloned()
170            .unwrap_or_else(|| self.config.canned_text.clone());
171        if scripted_calls.is_empty() {
172            Ok((i, text, Vec::new(), StopReason::EndTurn))
173        } else {
174            let tool_uses = scripted_calls
175                .into_iter()
176                .enumerate()
177                .map(|(j, call)| {
178                    let tool_name = call.name.clone();
179                    let input = serde_json::from_value::<tau_domain::Value>(call.args)
180                        .unwrap_or_else(|e| {
181                            panic!(
182                                "echo-llm: tool_calls[{i}][{j}] for tool {tool_name:?} has \
183                                 args that failed to deserialize into tau_domain::Value: {e}"
184                            )
185                        });
186                    ToolUse::new(format!("echo_tool_{i}_{j}"), call.name, input)
187                })
188                .collect();
189            Ok((i, String::new(), tool_uses, StopReason::ToolUse))
190        }
191    }
192
193    /// The full canned response for one turn. `complete` and `stream`
194    /// MUST share this: the runtime drives `stream`
195    /// (`tau-runtime-core/src/stream.rs`), so a tool call wired only into
196    /// `complete` would never reach an agent loop.
197    ///
198    /// `canned_tool_call` fires on the first turn only — the agent loop
199    /// dispatches the tool and calls back, and turn 2 returns plain text
200    /// to end the run. Emitting it every turn would spin to the turn cap.
201    /// Checked only when `tool_calls` produced no scripted call for that
202    /// turn — `tool_calls` takes precedence when both are configured.
203    async fn next_response(&self, method: &str) -> Result<CompletionResponse, LlmError> {
204        let (turn, text, tool_uses, stop_reason) = self.next_turn(method).await?;
205        if !tool_uses.is_empty() {
206            return Ok(make_completion_response(text, tool_uses, stop_reason, None));
207        }
208        match (turn, self.config.canned_tool_call.as_ref()) {
209            (0, Some(call)) => Ok(make_completion_response(
210                text,
211                vec![make_tool_use(
212                    "echo-llm-tool-use-0".to_string(),
213                    call.name.clone(),
214                    call.input.clone(),
215                )],
216                StopReason::ToolUse,
217                None,
218            )),
219            _ => Ok(make_completion_response(
220                text,
221                Vec::new(),
222                stop_reason,
223                None,
224            )),
225        }
226    }
227}
228
229impl LlmBackend for EchoLlm {
230    fn name(&self) -> &str {
231        "echo-llm"
232    }
233
234    async fn complete(&self, _req: CompletionRequest) -> Result<CompletionResponse, LlmError> {
235        self.next_response("llm.complete").await
236    }
237
238    async fn stream(&self, _req: CompletionRequest) -> Result<CompletionStream, LlmError> {
239        let resp = self.next_response("llm.stream").await?;
240        Ok(batch_to_stream(resp))
241    }
242}

249#[cfg(test)]
250mod tests {
251    use super::*;
252
253    /// (a) — additivity guarantee (#631 I4): a config that sets neither
254    /// `tool_calls` nor `canned_tool_call` — every config written before
255    /// this branch — must still produce exactly today's shape: the
256    /// scripted/canned text, no tool calls, `StopReason::EndTurn`. This
257    /// branch's whole premise is that existing configs behave
258    /// byte-identically; this test is what actually checks that premise
259    /// rather than asserting it in a doc comment.
260    #[tokio::test]
261    async fn empty_config_yields_canned_text_no_tools_end_turn() {
262        let echo = EchoLlm::from_config(EchoConfig {
263            canned_text: "hello".to_string(),
264            ..Default::default()
265        })
266        .unwrap();
267
268        let (turn, text, tool_uses, stop_reason) = echo.next_turn("llm.complete").await.unwrap();
269        assert_eq!(turn, 0);
270        assert_eq!(text, "hello");
271        assert!(tool_uses.is_empty());
272        assert_eq!(stop_reason, StopReason::EndTurn);
273    }
274
275    /// (b) — a scripted turn 0 yields a `ToolUse` with the right
276    /// name/args and `StopReason::ToolUse`.
277    #[tokio::test]
278    async fn scripted_turn_zero_yields_a_tool_use() {
279        let echo = EchoLlm::from_config(EchoConfig {
280            tool_calls: vec![vec![ScriptedToolCall {
281                name: "weather.get_forecast".to_string(),
282                args: serde_json::json!({"location": "Paris"}),
283            }]],
284            ..Default::default()
285        })
286        .unwrap();
287
288        let (turn, text, tool_uses, stop_reason) = echo.next_turn("llm.complete").await.unwrap();
289        assert_eq!(turn, 0);
290        assert_eq!(text, "");
291        assert_eq!(stop_reason, StopReason::ToolUse);
292        assert_eq!(tool_uses.len(), 1);
293        assert_eq!(tool_uses[0].name, "weather.get_forecast");
294        let expected_input: tau_domain::Value =
295            serde_json::from_value(serde_json::json!({"location": "Paris"})).unwrap();
296        assert_eq!(tool_uses[0].input, expected_input);
297    }
298
299    /// (c) — turn 1 falls back to text/`EndTurn`, which is what
300    /// terminates the agent loop (only turn 0 is scripted above).
301    #[tokio::test]
302    async fn turn_after_scripted_call_falls_back_to_text_end_turn() {
303        let echo = EchoLlm::from_config(EchoConfig {
304            tool_calls: vec![vec![ScriptedToolCall {
305                name: "weather.get_forecast".to_string(),
306                args: serde_json::Value::Null,
307            }]],
308            canned_text: "done".to_string(),
309            ..Default::default()
310        })
311        .unwrap();
312
313        // Turn 0: consumes the scripted call.
314        let _ = echo.next_turn("llm.complete").await.unwrap();
315        // Turn 1: tool_calls has no entry for this index -> falls back
316        // exactly like a config that never sets tool_calls at all.
317        let (turn, text, tool_uses, stop_reason) = echo.next_turn("llm.complete").await.unwrap();
318        assert_eq!(turn, 1);
319        assert_eq!(text, "done");
320        assert!(tool_uses.is_empty());
321        assert_eq!(stop_reason, StopReason::EndTurn);
322    }
