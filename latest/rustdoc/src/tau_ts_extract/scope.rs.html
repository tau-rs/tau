1//! Top-level constant walker → name → Expr map.
2
3use std::collections::BTreeMap;
4
5use swc_ecma_ast::{Decl, Expr, Module, ModuleDecl, ModuleItem, Pat, Stmt, VarDecl, VarDeclarator};
6
7/// Map from top-level constant name to its initializer expression.
8pub type NameMap<'a> = BTreeMap<String, &'a Expr>;
9
10/// Walk a parsed Module and collect every top-level `const NAME = EXPR;`
11/// (including `export const`).
12pub fn collect_top_level(module: &Module) -> NameMap<'_> {
13    let mut names = BTreeMap::new();
14    for item in &module.body {
15        match item {
16            ModuleItem::Stmt(Stmt::Decl(Decl::Var(var))) => {
17                if var.kind == swc_ecma_ast::VarDeclKind::Const {
18                    collect_from_var(var, &mut names);
19                }
20            }
21            ModuleItem::ModuleDecl(ModuleDecl::ExportDecl(exp)) => {
22                if let Decl::Var(var) = &exp.decl {
23                    if var.kind == swc_ecma_ast::VarDeclKind::Const {
24                        collect_from_var(var, &mut names);
25                    }
26                }
27            }
28            _ => {}
29        }
30    }
31    names
32}
33
34fn collect_from_var<'a>(var: &'a VarDecl, names: &mut NameMap<'a>) {
35    for decl in &var.decls {
36        if let (Some(name), Some(init)) = (extract_ident_name(decl), decl.init.as_deref()) {
37            names.insert(name, init);
38        }
39    }
40}
41
42fn extract_ident_name(decl: &VarDeclarator) -> Option<String> {
43    match &decl.name {
44        Pat::Ident(binding) => Some(binding.id.sym.to_string()),
45        _ => None,
46    }
47}

49#[cfg(test)]
50mod tests {
51    use super::*;
52    use crate::parse::parse_module;
53    use std::path::Path;
54
55    #[test]
56    fn collects_single_top_level_const() {
57        let src = r#"const foo = 42;"#;
58        let (module, sm) = parse_module(src, Path::new("/tmp/t.ts")).unwrap();
59        let names = collect_top_level(&module);
60        assert!(
61            names.contains_key("foo"),
62            "expected `foo`, got: {:?}",
63            names.keys().collect::<Vec<_>>()
64        );
65        let _ = sm; // keep SourceMap alive
66    }
67
68    #[test]
69    fn collects_exported_const() {
70        let src = r#"export const bar = "hello";"#;
71        let (module, _sm) = parse_module(src, Path::new("/tmp/t.ts")).unwrap();
72        let names = collect_top_level(&module);
73        assert!(names.contains_key("bar"));
74    }
75
76    #[test]
77    fn collects_multiple_declarations() {
78        let src = r#"
79            const a = 1;
80            const b = "x";
81            export const c = { foo: "bar" };
82        "#;
83        let (module, _sm) = parse_module(src, Path::new("/tmp/t.ts")).unwrap();
84        let names = collect_top_level(&module);
85        assert_eq!(names.len(), 3);
86        assert!(names.contains_key("a"));
87        assert!(names.contains_key("b"));
88        assert!(names.contains_key("c"));
89    }
90}
