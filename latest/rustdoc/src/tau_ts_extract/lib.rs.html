5//! and ADR-0041 (forthcoming) for the design.

28/// Extract a `ProjectConfig` from a TypeScript source string.
29///
30/// `source_path` is used only for error positioning (file:line:col).
31/// The function does NOT read from disk — caller is responsible for
32/// reading + UTF-8 validation.
33pub fn extract_project(source: &str, source_path: &Path) -> Result<ProjectConfig, TsExtractError> {
34    let (module, sm) = parse::parse_module(source, source_path)?;
35    let names = scope::collect_top_level(&module);
36    lower::build_project_config(&module, &names, source_path, &sm)
37}
