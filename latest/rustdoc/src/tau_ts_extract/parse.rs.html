1//! swc parser setup + module-level AST acquisition.
2
3use std::path::Path;
4
5use swc_common::{sync::Lrc, FileName, SourceMap, Spanned};
6use swc_ecma_ast::Module;
7use swc_ecma_parser::{lexer::Lexer, Parser, StringInput, Syntax, TsSyntax};
8
9use crate::error::TsExtractError;
10
11/// Parse a TS source string into an `swc_ecma_ast::Module`.
12///
13/// Returns the parsed module AND the `SourceMap` used (caller keeps it
14/// alive for span-to-position resolution during error reporting).
15pub fn parse_module(
16    source: &str,
17    source_path: &Path,
18) -> Result<(Module, Lrc<SourceMap>), TsExtractError> {
19    let cm: Lrc<SourceMap> = Default::default();
20
21    let fm = cm.new_source_file(
22        Lrc::new(FileName::Real(source_path.to_path_buf())),
23        source.to_string(),
24    );
25
26    let lexer = Lexer::new(
27        Syntax::Typescript(TsSyntax::default()),
28        Default::default(),
29        StringInput::from(&*fm),
30        None,
31    );
32
33    let mut parser = Parser::new_from(lexer);
34
35    let module = parser.parse_module().map_err(|e| {
36        let span = e.span();
37        let pos = TsExtractError::position_from_span(&cm, span, source_path.to_path_buf());
38        TsExtractError::ParseError {
39            pos,
40            message: format!("{:?}", e.kind()),
41        }
42    })?;
43
44    Ok((module, cm))
45}
