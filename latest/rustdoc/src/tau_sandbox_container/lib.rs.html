1//! Container sandbox adapter for tau.
2//!
3//! Shells out to `docker` or `podman` to wrap plugin processes in a container
4//! with capability-derived isolation. Works on any host with one of the
5//! runtimes installed (Linux / macOS / Windows with Docker Desktop).
6//!
7//! Compared to [`tau_sandbox_native`](https://docs.rs/tau-sandbox-native):
8//! - **Pros**: cross-platform; stronger isolation (full container); no kernel
9//!   feature requirements (works on macOS/Windows hosts).
10//! - **Cons**: requires docker/podman on PATH; container-start latency
11//!   (~50-200 ms cold); image management (pull / authentication).
12

28/// Container runtime selection passed to [`ContainerSandbox::new`].
29#[non_exhaustive]
30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31pub enum ContainerRuntime {
32    /// Use the `docker` binary on PATH.
33    Docker,
34    /// Use the `podman` binary on PATH.
35    Podman,
36    /// Probe `podman` first, then `docker`; first found wins.
37    ///
38    /// Podman is preferred because it is daemonless, rootless-by-default,
39    /// and Apache-2.0-licensed.
40    Auto,
41}
42
43/// Container sandbox adapter.
44///
45/// Construct via [`ContainerSandbox::new`]. The probe result and resolved
46/// runtime are cached together in a [`tokio::sync::OnceCell`] so the runtime
47/// binary is shelled out to at most once per adapter instance — even for
48/// [`ContainerRuntime::Auto`], which would otherwise re-probe on every spawn.
49pub struct ContainerSandbox {
50    name: String,
51    runtime: ContainerRuntime,
52    /// Cached `(probe result, resolved runtime)`. Populated lazily on the
53    /// first call to [`CapabilityGate::probe`] or [`ProcessCapabilityGate::wrap_spawn`].
54    probe_cache: Arc<OnceCell<(CapabilityProbe, probe::ResolvedRuntime)>>,
55}
56
57impl ContainerSandbox {
58    /// Construct a container adapter using the given runtime selection.
59    ///
60    /// The probe is deferred until the first call to [`CapabilityGate::probe`].
61    pub fn new(name: impl Into<String>, runtime: ContainerRuntime) -> Self {
62        Self {
63            name: name.into(),
64            runtime,
65            probe_cache: Arc::new(OnceCell::new()),
66        }
67    }
68}
69
70impl CapabilityGate for ContainerSandbox {
71    fn name(&self) -> &str {
72        &self.name
73    }
74
75    async fn probe(&self) -> CapabilityProbe {
76        self.probe_cache
77            .get_or_init(|| async { probe::run_probe(self.runtime).await })
78            .await
79            .0
80            .clone()
81    }
82
83    fn supported_shapes(&self) -> CapabilityShapeSet {
84        let mut set = CapabilityShapeSet::new();
85        set.insert(tau_domain::CapabilityShape::FilesystemRead);
86        set.insert(tau_domain::CapabilityShape::FilesystemWrite);
87        set.insert(tau_domain::CapabilityShape::ProcessExec);
88        set.insert(tau_domain::CapabilityShape::NetworkHttp);
89        set
90    }
91
92    fn validate_plan(&self, plan: &CapabilityPlan) -> Result<(), CapabilityError> {
93        let supported = self.supported_shapes();
94        for cap in &plan.capabilities {
95            let shape = cap.required_shape();
96            if !supported.contains(&shape) {
97                return Err(CapabilityError::ShapeUnsupported { shape });
98            }
99        }
100        Ok(())
101    }
102}
103
104impl ProcessCapabilityGate for ContainerSandbox {
105    async fn wrap_spawn(
106        &self,
107        plan: &CapabilityPlan,
108        cmd: &mut Command,
109    ) -> Result<CapabilityHandle, CapabilityError> {
110        self.validate_plan(plan)?;
111
112        let (probe_result, runtime_kind) = self
113            .probe_cache
114            .get_or_init(|| async { probe::run_probe(self.runtime).await })
115            .await;
116        match probe_result {
117            CapabilityProbe::Available { .. } => runner::wrap_command(plan, cmd, *runtime_kind),
118            CapabilityProbe::Unavailable { reason } => Err(CapabilityError::Unavailable {
119                reason: reason.clone(),
120            }),
121            // Non-exhaustive catch-all: treat any unknown future variant as an
122            // internal error rather than silently proceeding.
123            other => Err(CapabilityError::WrapFailed {
124                message: format!("unexpected probe result: {other:?}"),
125            }),
126        }
127    }
128}

130#[cfg(test)]
131mod tests {
132    use super::*;
133
134    #[test]
135    fn name_round_trip() {
136        let s = ContainerSandbox::new("ctr", ContainerRuntime::Docker);
137        assert_eq!(s.name(), "ctr");
138    }
139
140    #[test]
141    fn supported_shapes_includes_all() {
142        let s = ContainerSandbox::new("ctr", ContainerRuntime::Auto);
143        let supported = s.supported_shapes();
144        assert!(supported.contains(&tau_domain::CapabilityShape::FilesystemRead));
145        assert!(supported.contains(&tau_domain::CapabilityShape::FilesystemWrite));
146        assert!(supported.contains(&tau_domain::CapabilityShape::ProcessExec));
147        assert!(supported.contains(&tau_domain::CapabilityShape::NetworkHttp));
148    }
149
150    #[test]
151    fn validate_plan_rejects_custom() {
152        let s = ContainerSandbox::new("ctr", ContainerRuntime::Auto);
153        let plan_json = serde_json::json!({
154            "capabilities": [{ "kind": "custom.weird" }],
155            "context": null,
156            "limits": null,
157        });
158        let plan: CapabilityPlan = serde_json::from_value(plan_json).expect("decode");
159        let err = s.validate_plan(&plan).expect_err("must reject custom");
160        assert!(
161            matches!(err, CapabilityError::ShapeUnsupported { .. }),
162            "expected ShapeUnsupported, got {err:?}"
163        );
164    }
165
166    #[test]
167    fn validate_plan_accepts_known_shapes() {
168        let s = ContainerSandbox::new("ctr", ContainerRuntime::Docker);
169        let plan_json = serde_json::json!({
170            "capabilities": [
171                { "kind": "fs.read",  "paths": ["/etc"] },
172                { "kind": "fs.write", "paths": ["/tmp"] },
173                { "kind": "net.http", "hosts": ["example.com"], "methods": ["GET"] }
174            ],
175            "context": null,
176            "limits": null,
177        });
178        let plan: CapabilityPlan = serde_json::from_value(plan_json).expect("decode");
179        s.validate_plan(&plan)
180            .expect("known shapes must be accepted");
181    }
182
183    #[test]
184    fn runtime_variants_are_copy() {
185        // ContainerRuntime derives Copy; verify it can be used by value.
186        let r = ContainerRuntime::Docker;
187        let _r2 = r;
188        let _r3 = r; // would fail to compile if not Copy
189    }
