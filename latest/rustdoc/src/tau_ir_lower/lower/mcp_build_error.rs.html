1//! Build-time errors specific to MCP contract resolution + expansion
2//! (per β.3 design doc §5 build-time invariants table).
3//!
4//! All variants surface through `LowerError::McpBuild(...)`; the `tau check`
5//! aggregator renders them with exit code 64 (validation).
6
7use alloc::string::String;
8use alloc::vec::Vec;
9use thiserror::Error;
10
11/// Per-spec §5 build-time invariant violations.
12#[derive(Debug, Clone, PartialEq, Eq, Error)]
13#[non_exhaustive]
14pub enum McpBuildError {
15    /// Live resolver couldn't reach the MCP server (network / spawn /
16    /// handshake failure). Re-raised from the resolver's typed error.
17    #[error("MCP contract unreachable for entry {entry:?}: {reason}")]
18    ContractUnreachable {
19        /// `[tools.<entry>]` name from tau.toml.
20        entry: String,
21        /// Resolver-side reason (e.g. "handshake timeout after 30000ms").
22        reason: String,
23    },
24    /// One or more server-tool capability requirements aren't covered by
25    /// the author's envelope.
26    #[error(
27        "envelope does not cover server-tool {tool:?} capabilities for entry {entry:?}: \
28         missing {missing:?}"
29    )]
30    EnvelopeCoversContract {
31        /// `[tools.<entry>]` name.
32        entry: String,
33        /// Server-side tool name.
34        tool: String,
35        /// Capabilities the contract declared that the envelope omits
36        /// (rendered as `kind`/`host`/`path` keys for diagnostics).
37        missing: Vec<String>,
38    },
39    /// `roots = [...]` declared in tau.toml are not all covered by the
40    /// envelope's `fs.read` capabilities.
41    #[error("roots {roots:?} for entry {entry:?} not covered by fs.read caps")]
42    RootsExceedFsCaps {
43        /// `[tools.<entry>]` name.
44        entry: String,
45        /// Offending root paths.
46        roots: Vec<String>,
47    },
48    /// The MCP server's contract requires sampling (any tool's caps
49    /// include `sampling.*`) but the author left `sampling.models = []`.
50    #[error("entry {entry:?} server contract requires sampling but sampling.models is empty")]
51    SamplingRequiredByContract {
52        /// `[tools.<entry>]` name.
53        entry: String,
54    },
55    /// `--offline` was passed but `.tau/mcp/<entry>.contract.json` is missing.
56    #[error(
57        "entry {entry:?}: --offline requested but pinned contract file is missing at {path:?}"
58    )]
59    PinnedContractMissing {
60        /// `[tools.<entry>]` name.
61        entry: String,
62        /// Expected path on disk.
63        path: String,
64    },
65    /// A server-tool name contains `.`, which would collide with the
66    /// `<entry>.<server-tool>` ToolId convention.
67    #[error("server-tool name {name:?} for entry {entry:?} contains '.', which is reserved as the ToolId separator")]
68    ServerToolNameContainsDot {
69        /// `[tools.<entry>]` name.
70        entry: String,
71        /// Server-side tool name that contains `.`.
72        name: String,
73    },
74}
