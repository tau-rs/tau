1//! D-3b: strict build-time capability-fit check.
2//!
3//! Refuses the build (returns `LowerError::CapabilityFitFailed`) if any
4//! tool's declared capabilities require a shape that the target's
5//! `required_shapes` does not include. **No override flag** — the
6//! caller's user-facing diagnostic must say so explicitly. Matches the
7//! Rust-like build-time enforcement principle per
8//! `feedback_tau_rust_like_build_enforcement`.
9//!
10//! # Field-name correction
11//!
12//! The plan stub references `entry.profile().supported_shapes`. The
13//! actual field name in `tau_ports::target::profile::TargetCapabilityProfile`
14//! is `required_shapes` (the set of shapes the adapter declares it can
15//! enforce). This module uses `required_shapes` throughout.
16
17use alloc::vec::Vec;
18use tau_domain::CapabilityShape;
19use tau_ports::target::{registry, TargetTriple};
20
21use crate::error::LowerError;
22use tau_ir::ids::ToolId;
23
24use super::parse::Parsed;
25
26/// Run the capability-fit check on a `Parsed` workflow against a target.
27///
28/// Returns `Ok(())` if every shape required by any tool in the workflow
29/// is present in the target's `required_shapes` set. Returns
30/// `Err(LowerError::CapabilityFitFailed)` on the first miss (with the
31/// full list of missing shapes and the tools that caused them).
32///
33/// If the target triple is not found in the registry, the check also
34/// fails with an empty `missing` list — the caller's diagnostic should
35/// handle this as "unknown target".
36pub(super) fn check(parsed: &Parsed, target: &TargetTriple) -> Result<(), LowerError> {
37    let entry = registry::lookup(target).ok_or_else(|| LowerError::CapabilityFitFailed {
38        missing: Vec::new(),
39        tools: Vec::new(),
40    })?;
41    let supported = entry.profile().required_shapes;
42
43    let mut missing: Vec<CapabilityShape> = Vec::new();
44    let mut blamed_tools: Vec<ToolId> = Vec::new();
45
46    for (tool_id, tool) in parsed.workflow.tools.iter() {
47        for cap in tool.capabilities.declared.iter() {
48            let shape = cap.required_shape();
49            if !supported.contains(&shape) {
50                if !missing.contains(&shape) {
51                    missing.push(shape);
52                }
53                if !blamed_tools.contains(tool_id) {
54                    blamed_tools.push(tool_id.clone());
55                }
56            }
57        }
58    }
59
60    if missing.is_empty() {
61        Ok(())
62    } else {
63        Err(LowerError::CapabilityFitFailed {
64            missing,
65            tools: blamed_tools,
66        })
67    }
68}
