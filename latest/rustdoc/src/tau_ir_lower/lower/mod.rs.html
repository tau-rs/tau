1//! Lowering pass: `tau_pkg::ProjectConfig` → `IrModule`.
2//!
3//! The lowering pass is pure: it consumes an already-parsed
4//! `ProjectConfig`, resolves external references against caches the
5//! caller supplies (native tool registry, MCP contract cache, skill
6//! content-hash table), runs the capability-fit check against the
7//! target triple, and produces a typed `IrModule`. Any error short-
8//! circuits with `LowerError`.
9//!
10//! `tau build` is the caller; `tau dev` is also a caller (it lowers
11//! once per source change to drive the interpreter against a fresh IR).
12

31/// Per-server-tool slice of a resolved MCP contract.
32///
33/// The resolve stage expands one `ToolImpl::Mcp` author entry into N IR
34/// nodes (one per server-tool). Each node uses one of these structs to
35/// fill in its `capability_subset` + `server_tool_name` + (separately)
36/// the agent's `tool_refs` rewrite.
37#[derive(Debug, Clone, PartialEq)]
38pub struct ResolvedServerTool {
39    /// Server-side tool name (what tau-mcp-tokio sends on `tools/call`).
40    pub name: alloc::string::String,
41    /// Capability requirements the server declares for this tool.
42    pub caps: CapabilityRequirements,
43    /// JSON schema for the tool's input (passed through to IR; opaque to
44    /// the resolver).
45    pub input_schema: serde_json::Value,
46}

48/// Error returned by the injected `prompt_file` reader (see [`Caches`]).
49///
50/// Carries a human-readable reason (typically an `std::io::Error` rendered by
51/// the caller); lowering wraps it into [`LowerError::PromptFileUnreadable`].
52#[derive(Debug, Clone)]
53pub struct PromptFileError(pub alloc::string::String);

55/// Output of [`lower_project`]: the lowered module plus the content-addressed
56/// asset blobs (currently agent prompts) the bundle must carry (D6-B).
57///
58/// Assets are keyed by their hash (`"sha256:" + 64 lowercase hex`), matching
59/// the [`tau_ir::prompt::PromptSource::Asset`] references inside `module`.
60/// Identical content across agents dedupes to one blob by construction.
61#[derive(Debug)]
62pub struct LowerOutput {
63    /// The lowered IR module.
64    pub module: IrModule,
65    /// Content-addressed assets referenced by `module`, keyed by hash.
66    pub assets: alloc::collections::BTreeMap<alloc::string::String, tau_ir::asset::AssetBlob>,
67}

69/// Resolved MCP contract for one author-side `[tools.<entry>]` entry.
70#[derive(Debug, Clone, PartialEq)]
71pub struct ResolvedMcpContract {
72    /// SHA-256 of the canonical contract — participates in the IR
73    /// module hash so contract drift invalidates the bundle.
74    pub hash: [u8; 32],
75    /// All server-side tools the contract advertises (one ToolImpl::Mcp
76    /// IR node will be emitted per entry).
77    pub expanded_tools: alloc::vec::Vec<ResolvedServerTool>,
78    /// Whether the contract advertises `sampling/*` capabilities. Used
79    /// by resolve to enforce the `SamplingRequiredByContract` invariant.
80    pub requires_sampling: bool,
81}

83/// Lower a parsed `ProjectConfig` into an `IrModule` for the given target.
84///
85/// Pipeline:
86/// 1. `parse` — extract per-agent and per-tool declarations.
87/// 2. `resolve` — resolve native tool content-hashes, MCP contract
88///    hashes, skill content-hashes (caller-supplied caches).
89/// 3. `typecheck` — agents' tool_refs exist, subflow targets exist,
90///    cap_subset is a subset of parent grant.
91/// 4. `capability_fit` — every required shape supported by `target`.
92/// 5. `feature_fit` — every IR feature used is executable on `target`.
93/// 6. `predicate_fit` — every goal predicate/deterministic fn used has a
94///    wasm guest execution path (wasm-only; no-op for native targets).
95///
96/// # Example
97///
98/// ```
99/// use tau_ir_lower::{lower_project, Caches};
100/// use tau_pkg::project::ProjectConfig;
101/// use tau_ports::target::registry;
102///
103/// let toml = r#"
104///     [project]
105///     name = "demo"
106/// "#;
107/// let config = ProjectConfig::parse_str(toml).unwrap();
108/// let target = registry::list_available().next().unwrap().triple;
109/// let caches = Caches {
110///     native_tool: &|_| None,
111///     mcp_contract: &|_| None,
112///     skill: &|_| None,
113///     prompt_file: &|_| Ok(Vec::new()),
114/// };
115/// let out = lower_project(&config, &target, &caches).unwrap();
116/// assert_eq!(out.module.ir_format.0, tau_ir::IrFormatVersion::CURRENT);
117/// assert!(out.assets.is_empty());
118/// ```
119pub fn lower_project(
120    config: &ProjectConfig,
121    target: &TargetTriple,
122    caches: &Caches,
123) -> Result<LowerOutput, LowerError> {
124    let parsed = parse::parse(config, caches.prompt_file)?;
125    let resolved = resolve::resolve(parsed, caches)?;
126    typecheck::typecheck(&resolved)?;
127    capability_fit::check(&resolved, target)?;
128    feature_fit::check(&resolved, target)?;
129    predicate_fit::check(&resolved, target)?;
130    tool_impl_fit::check(&resolved, target)?;
131    Ok(build_output(resolved, target))
132}

134/// Caches the caller supplies for resolution. Each is a closure over an
135/// existing tau-pkg / tau-cli registry so the lowering pass stays pure.
136pub struct Caches<'a> {
137    /// Resolves a native tool symbolic name to its content hash.
138    pub native_tool: &'a dyn Fn(&str) -> Option<[u8; 32]>,
139    /// Resolves an MCP URL to its fully-expanded contract (per
140    /// β.3 design doc §5). Returns `None` only if `tau build` did not
141    /// pre-fetch this URL — the resolver typically errors instead.
142    pub mcp_contract: &'a dyn Fn(&str) -> Option<ResolvedMcpContract>,
143    /// Resolves a skill name to its content hash (from Skills-2 lockfile).
144    ///
145    /// **Reserved for a future `ToolImpl::Skill` variant.** No code path
146    /// in v0 calls this closure; callers may supply a `|_| None` stub.
147    /// The field stays in `Caches` to lock the resolver signature so
148    /// adding `ToolImpl::Skill` later is non-breaking.
149    pub skill: &'a dyn Fn(&str) -> Option<[u8; 32]>,
150    /// Reads an agent `system_file` prompt's bytes at build time (D6-B).
151    ///
152    /// Lowering hashes the returned bytes and emits a
153    /// [`tau_ir::prompt::PromptSource::Asset`] reference plus an asset blob
154    /// (collected into [`LowerOutput::assets`]). The closure resolves the
155    /// path (relative paths against the project root) and returns the
156    /// content; a read failure becomes a
157    /// [`LowerError::PromptFileUnreadable`] build error. Callers with no
158    /// file prompts may supply a `|_| Ok(Vec::new())` stub.
159    pub prompt_file: &'a dyn Fn(&std::path::Path) -> Result<alloc::vec::Vec<u8>, PromptFileError>,
160}

162fn build_output(parsed: crate::lower::parse::Parsed, target: &TargetTriple) -> LowerOutput {
163    let crate::lower::parse::Parsed {
164        workflow,
165        triggers,
166        assets,
167    } = parsed;
168    // Option B (ADR-0044 §D1): triggers do NOT bump ir_format. ir_format is set
169    // to `IrFormatVersion::current()` (advanced since ADR-0044 for unrelated IR
170    // changes); the point here is that adding the `triggers` field did not force
171    // a bump. The `triggers` field's skip-empty serialization preserves
172    // trigger-less hashes; the appended array differentiates trigger-bearing
173    // hashes on its own.
174    let module = IrModule {
175        ir_format: tau_ir::IrFormatVersion::current(),
176        tau_version: env!("CARGO_PKG_VERSION").into(),
177        target: *target,
178        workflow,
179        triggers,
180    };
181    LowerOutput { module, assets }
182}
