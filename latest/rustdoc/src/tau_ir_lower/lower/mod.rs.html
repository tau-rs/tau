1//! Lowering pass: `tau_pkg::ProjectConfig` → `IrModule`.
2//!
3//! The lowering pass is pure: it consumes an already-parsed
4//! `ProjectConfig`, resolves external references against caches the
5//! caller supplies (native tool registry, MCP contract cache, skill
6//! content-hash table), runs the capability-fit check against the
7//! target triple, and produces a typed `IrModule`. Any error short-
8//! circuits with `LowerError`.
9//!
10//! `tau build` is the caller; `tau dev` is also a caller (it lowers
11//! once per source change to drive the interpreter against a fresh IR).
12

14pub mod feature_fit;
15pub mod mcp_build_error;

29/// Per-server-tool slice of a resolved MCP contract.
30///
31/// The resolve stage expands one `ToolImpl::Mcp` author entry into N IR
32/// nodes (one per server-tool). Each node uses one of these structs to
33/// fill in its `capability_subset` + `server_tool_name` + (separately)
34/// the agent's `tool_refs` rewrite.
35#[derive(Debug, Clone, PartialEq)]
36pub struct ResolvedServerTool {
37    /// Server-side tool name (what tau-mcp-tokio sends on `tools/call`).
38    pub name: alloc::string::String,
39    /// Capability requirements the server declares for this tool.
40    pub caps: CapabilityRequirements,
41    /// JSON schema for the tool's input (passed through to IR; opaque to
42    /// the resolver).
43    pub input_schema: serde_json::Value,
44}
45
46/// Error returned by the injected `prompt_file` reader (see [`Caches`]).
47///
48/// Carries a human-readable reason (typically an `std::io::Error` rendered by
49/// the caller); lowering wraps it into [`LowerError::PromptFileUnreadable`].
50#[derive(Debug, Clone)]
51pub struct PromptFileError(pub alloc::string::String);

53/// Output of [`lower_project`]: the lowered module plus the content-addressed
54/// asset blobs (currently agent prompts) the bundle must carry (D6-B).
55///
56/// Assets are keyed by their hash (`"sha256:" + 64 lowercase hex`), matching
57/// the [`tau_ir::prompt::PromptSource::Asset`] references inside `module`.
58/// Identical content across agents dedupes to one blob by construction.
59#[derive(Debug)]
60pub struct LowerOutput {
61    /// The lowered IR module.
62    pub module: IrModule,
63    /// Content-addressed assets referenced by `module`, keyed by hash.
64    pub assets: alloc::collections::BTreeMap<alloc::string::String, tau_ir::asset::AssetBlob>,
65}

67/// Resolved MCP contract for one author-side `[tools.<entry>]` entry.
68#[derive(Debug, Clone, PartialEq)]
69pub struct ResolvedMcpContract {
70    /// SHA-256 of the canonical contract — participates in the IR
71    /// module hash so contract drift invalidates the bundle.
72    pub hash: [u8; 32],
73    /// All server-side tools the contract advertises (one ToolImpl::Mcp
74    /// IR node will be emitted per entry).
75    pub expanded_tools: alloc::vec::Vec<ResolvedServerTool>,
76    /// Whether the contract advertises `sampling/*` capabilities. Used
77    /// by resolve to enforce the `SamplingRequiredByContract` invariant.
78    pub requires_sampling: bool,
79}

81/// Lower a parsed `ProjectConfig` into an `IrModule` for the given target.
82///
83/// Pipeline:
84/// 1. `parse` — extract per-agent and per-tool declarations.
85/// 2. `resolve` — resolve native tool content-hashes, MCP contract
86///    hashes, skill content-hashes (caller-supplied caches).
87/// 3. `typecheck` — agents' tool_refs exist, subflow targets exist,
88///    cap_subset is a subset of parent grant.
89/// 4. `capability_fit` — every required shape supported by `target`.
90/// 5. `feature_fit` — every IR feature used is executable on `target`.
91///
92/// # Example
93///
94/// ```
95/// use tau_ir_lower::{lower_project, Caches};
96/// use tau_pkg::project::ProjectConfig;
97/// use tau_ports::target::registry;
98///
99/// let toml = r#"
100///     [project]
101///     name = "demo"
102/// "#;
103/// let config = ProjectConfig::parse_str(toml).unwrap();
104/// let target = registry::list_available().next().unwrap().triple;
105/// let caches = Caches {
106///     native_tool: &|_| None,
107///     mcp_contract: &|_| None,
108///     skill: &|_| None,
109///     prompt_file: &|_| Ok(Vec::new()),
110/// };
111/// let out = lower_project(&config, &target, &caches).unwrap();
112/// assert_eq!(out.module.ir_format.0, tau_ir::IrFormatVersion::CURRENT);
113/// assert!(out.assets.is_empty());
114/// ```
115pub fn lower_project(
116    config: &ProjectConfig,
117    target: &TargetTriple,
118    caches: &Caches,
119) -> Result<LowerOutput, LowerError> {
120    let parsed = parse::parse(config, caches.prompt_file)?;
121    let resolved = resolve::resolve(parsed, caches)?;
122    typecheck::typecheck(&resolved)?;
123    capability_fit::check(&resolved, target)?;
124    feature_fit::check(&resolved, target)?;
125    Ok(build_output(resolved, target))
126}

128/// Caches the caller supplies for resolution. Each is a closure over an
129/// existing tau-pkg / tau-cli registry so the lowering pass stays pure.
130pub struct Caches<'a> {
131    /// Resolves a native tool symbolic name to its content hash.
132    pub native_tool: &'a dyn Fn(&str) -> Option<[u8; 32]>,
133    /// Resolves an MCP URL to its fully-expanded contract (per
134    /// β.3 design doc §5). Returns `None` only if `tau build` did not
135    /// pre-fetch this URL — the resolver typically errors instead.
136    pub mcp_contract: &'a dyn Fn(&str) -> Option<ResolvedMcpContract>,
137    /// Resolves a skill name to its content hash (from Skills-2 lockfile).
138    ///
139    /// **Reserved for a future `ToolImpl::Skill` variant.** No code path
140    /// in v0 calls this closure; callers may supply a `|_| None` stub.
141    /// The field stays in `Caches` to lock the resolver signature so
142    /// adding `ToolImpl::Skill` later is non-breaking.
143    pub skill: &'a dyn Fn(&str) -> Option<[u8; 32]>,
144    /// Reads an agent `system_file` prompt's bytes at build time (D6-B).
145    ///
146    /// Lowering hashes the returned bytes and emits a
147    /// [`tau_ir::prompt::PromptSource::Asset`] reference plus an asset blob
148    /// (collected into [`LowerOutput::assets`]). The closure resolves the
149    /// path (relative paths against the project root) and returns the
150    /// content; a read failure becomes a
151    /// [`LowerError::PromptFileUnreadable`] build error. Callers with no
152    /// file prompts may supply a `|_| Ok(Vec::new())` stub.
153    pub prompt_file: &'a dyn Fn(&std::path::Path) -> Result<alloc::vec::Vec<u8>, PromptFileError>,
154}

156fn build_output(parsed: crate::lower::parse::Parsed, target: &TargetTriple) -> LowerOutput {
157    let crate::lower::parse::Parsed {
158        workflow,
159        triggers,
160        assets,
161    } = parsed;
162    // Option B (ADR-0044 §D1): triggers do NOT bump ir_format. ir_format is set
163    // to `IrFormatVersion::current()` (advanced since ADR-0044 for unrelated IR
164    // changes); the point here is that adding the `triggers` field did not force
165    // a bump. The `triggers` field's skip-empty serialization preserves
166    // trigger-less hashes; the appended array differentiates trigger-bearing
167    // hashes on its own.
168    let module = IrModule {
169        ir_format: tau_ir::IrFormatVersion::current(),
170        tau_version: env!("CARGO_PKG_VERSION").into(),
171        target: *target,
172        workflow,
173        triggers,
174    };
175    LowerOutput { module, assets }
176}
