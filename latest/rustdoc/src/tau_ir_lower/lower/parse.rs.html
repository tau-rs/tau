1//! First lowering stage: pull typed declarations out of `ProjectConfig`.
2//!
3//! This stage does no I/O and no resolution. It walks the
4//! `ProjectConfig`'s agent/tool/step tables and produces an
5//! in-memory `Parsed` value: a partially-populated `Workflow` whose
6//! `ToolImpl::Native::content_hash` and similar resolution slots are
7//! filled with zero bytes (the `resolve` stage fills them).
8
9use alloc::collections::BTreeMap;
10use tau_pkg::project::{PipelineRunRef, ProjectConfig};
11use tau_pkg::{PromptEntry, ToolBody};
12
13use crate::error::LowerError;
14use tau_ir::capability::{CapabilityRequirements, CapabilityTable};
15use tau_ir::check::{Check, CheckVerify, GoalPredicate, JudgeRef, Locus, OnFail, RetryPolicy};
16use tau_ir::ids::{AgentId, CheckId, PipelineStepId, StepId, ToolId};
17use tau_ir::module::Workflow;
18use tau_ir::node::{Agent, Deterministic, Tool, ToolSpec};
19use tau_ir::pipeline::{Pipeline, PipelineStep, StepRun};
20use tau_ir::prompt::PromptSource;
21use tau_ir::subflow::SubflowEdge;
22use tau_ir::tool_impl::{Hash256, NativeFnRef, ToolImpl};
23use tau_ir::trigger::{
24    Backoff, BackoffStrategy, RetryPolicy as TriggerRetryPolicy, TriggerBinding, TriggerKind,
25};
26use tau_ir::AgentBudget;

28/// Output of the parse stage.
29#[derive(Debug)]
30pub(super) struct Parsed {
31    /// Partially-populated workflow (content hashes are zero pending `resolve`).
32    pub(super) workflow: Workflow,
33    /// Trigger bindings, canonically ordered by name (BTreeMap iteration).
34    pub(super) triggers: alloc::vec::Vec<TriggerBinding>,
35    /// Content-addressed asset blobs (currently `system_file` prompts) read
36    /// at build time, keyed by hash (`"sha256:" + 64 hex`). Deduped by hash.
37    pub(super) assets: BTreeMap<alloc::string::String, tau_ir::asset::AssetBlob>,
38}

40/// Run the parse stage on a `ProjectConfig`.
41///
42/// `prompt_file` reads an agent `system_file` prompt's bytes (D6-B); see
43/// [`crate::Caches::prompt_file`].
44pub(super) fn parse(
45    config: &ProjectConfig,
46    prompt_file: &dyn Fn(&std::path::Path) -> Result<alloc::vec::Vec<u8>, crate::PromptFileError>,
47) -> Result<Parsed, LowerError> {
48    let mut agents: BTreeMap<AgentId, Agent> = BTreeMap::new();
49    let mut tools: BTreeMap<ToolId, Tool> = BTreeMap::new();
50    let mut steps: BTreeMap<StepId, Deterministic> = BTreeMap::new();
51    let mut assets: BTreeMap<alloc::string::String, tau_ir::asset::AssetBlob> = BTreeMap::new();
52    let edges: alloc::vec::Vec<SubflowEdge> = alloc::vec::Vec::new();
53    let mut capability_table: BTreeMap<ToolId, CapabilityRequirements> = BTreeMap::new();
54
55    // --- Tools ---------------------------------------------------------
56    //
57    // `ProjectConfig::tools` is a `BTreeMap<String, ToolEntry>` produced by
58    // tau-pkg::config. Each `ToolEntry` discriminates Native vs Mcp vs Subflow.
59    for (name, entry) in config.tools.iter() {
60        let tool_id = ToolId(name.clone());
61        let caps = CapabilityRequirements {
62            declared: entry.capabilities.clone(),
63        };
64        let impl_ = match &entry.body {
65            ToolBody::Native(fn_name) => ToolImpl::Native {
66                fn_ref: NativeFnRef {
67                    name: fn_name.clone(),
68                },
69                // resolved by `resolve` stage; zero is a sentinel
70                content_hash: Hash256::default(),
71            },
72            ToolBody::Mcp(url) => ToolImpl::Mcp {
73                url: url.clone(),
74                contract_hash: Hash256::default(),
75                capability_subset: caps.clone(),
76                // server_tool_name is empty at parse time; the resolve stage
77                // expands this single entry into N entries (one per server-tool)
78                // with real names filled in.
79                server_tool_name: alloc::string::String::new(),
80            },
81            ToolBody::Subflow(target) => ToolImpl::Subflow {
82                target: AgentId(target.clone()),
83            },
84            // `ToolBody` is `#[non_exhaustive]`; future variants are
85            // forwarded as a parse error so existing callers aren't
86            // silently broken when a new variant lands.
87            _ => {
88                return Err(LowerError::Parse("unsupported tool body variant".into()));
89            }
90        };
91        let spec = ToolSpec {
92            name: name.clone(),
93            description: entry.description.clone(),
94            input_schema: entry.input_schema.clone(),
95        };
96        capability_table.insert(tool_id.clone(), caps.clone());
97        tools.insert(
98            tool_id.clone(),
99            Tool {
100                id: tool_id,
101                impl_,
102                capabilities: caps,
103                spec,
104            },
105        );
106    }
107
108    // --- Agents --------------------------------------------------------
109    for (name, entry) in config.agents.iter() {
110        let agent_id = AgentId(name.clone());
111        let tool_refs = entry.tool_refs.iter().cloned().map(ToolId).collect();
112        // Lower the prompt source (D6-B). `Inline` maps straight through.
113        // `File` is read at build time via the injected `prompt_file` reader,
114        // hashed, and emitted as a content-addressed `PromptSource::Asset`;
115        // the bytes are collected into `assets` (deduped by hash). Paths never
116        // enter the IR, and a missing/unreadable file is a hard build error —
117        // this fixes the non-hermetic `system_file` bug (the IR used to carry
118        // the path string as the prompt).
119        let prompt = match &entry.prompt {
120            PromptEntry::Inline(s) => PromptSource::inline(s.clone()),
121            PromptEntry::File(p) => {
122                let bytes = prompt_file(p).map_err(|e| LowerError::PromptFileUnreadable {
123                    agent: agent_id.clone(),
124                    path: p.to_string_lossy().into_owned(),
125                    reason: e.0,
126                })?;
127                let hash = tau_ir::asset::asset_hash(&bytes);
128                // Identical content across agents dedupes to one blob.
129                assets
130                    .entry(hash.clone())
131                    .or_insert_with(|| tau_ir::asset::AssetBlob::prompt(bytes));
132                PromptSource::asset(hash)
133            }
134            PromptEntry::None => PromptSource::inline(""),
135            // `PromptEntry` is `#[non_exhaustive]`, so this wildcard is
136            // required. A future variant reaching here is fail-closed (D7-B /
137            // ADR-0065): it used to silently become an empty prompt.
138            other => {
139                return Err(LowerError::UnsupportedPromptKind {
140                    agent: agent_id.clone(),
141                    detail: alloc::format!("{other:?}"),
142                });
143            }
144        };
145        agents.insert(
146            agent_id.clone(),
147            Agent {
148                id: agent_id,
149                prompt,
150                model_ref: resolve_model_ref(config, &entry.model)?,
151                tool_refs,
152                context: lower_context(name, entry)?,
153                budget: AgentBudget {
154                    max_turns: entry.max_turns,
155                    max_tokens: entry.max_tokens,
156                },
157                produces: entry.produces.clone(),
158                output_schema: entry.output_schema.clone(),
159                durable: lower_durable(entry),
160            },
161        );
162    }
163
164    // --- Deterministic steps ------------------------------------------
165    for (name, entry) in config.steps.iter() {
166        let step_id = StepId(name.clone());
167        steps.insert(
168            step_id.clone(),
169            Deterministic {
170                id: step_id,
171                fn_ref: NativeFnRef {
172                    name: entry.fn_name.clone(),
173                },
174                input_schema: entry.input_schema.clone(),
175                output_schema: entry.output_schema.clone(),
176            },
177        );
178    }
179
180    // --- Deterministic steps as tools ----------------------------------
181    //
182    // Each [steps.<name>] block registers BOTH a `Deterministic` node
183    // (above) and a `Tool` with `ToolImpl::Step { id }`. The Tool
184    // registration is what lets an agent reference the step in its
185    // `tool_refs`; the Deterministic node is what the runtime registry
186    // dispatches against at invoke time.
187    for (name, entry) in config.steps.iter() {
188        let step_id = StepId(name.clone());
189        let tool_id = ToolId(name.clone());
190
191        // Guard: a [tools.<name>] entry with the same key would be
192        // silently overwritten, causing the wrong implementation to be
193        // dispatched at runtime. Reject early with a clear error.
194        if tools.contains_key(&tool_id) {
195            return Err(LowerError::Parse(alloc::format!(
196                "step name {name:?} collides with an existing tool name; \
197                 rename the step or the tool"
198            )));
199        }
200
201        let caps = CapabilityRequirements {
202            declared: alloc::vec::Vec::new(),
203        };
204        let spec = ToolSpec {
205            name: name.clone(),
206            description: alloc::string::String::new(),
207            input_schema: entry.input_schema.clone(),
208        };
209        capability_table.insert(tool_id.clone(), caps.clone());
210        tools.insert(
211            tool_id.clone(),
212            Tool {
213                id: tool_id,
214                impl_: ToolImpl::Step { id: step_id },
215                capabilities: caps,
216                spec,
217            },
218        );
219    }
220
221    // --- Triggers (slice 1) --------------------------------------------
222    let mut triggers: alloc::vec::Vec<TriggerBinding> = alloc::vec::Vec::new();
223    for (name, entry) in config.triggers.iter() {
224        let kind = match entry.kind.as_str() {
225            "cron" => TriggerKind::Cron,
226            "manual" => TriggerKind::Manual,
227            // validate_trigger already rejected anything else; defensive.
228            other => {
229                return Err(LowerError::Parse(alloc::format!(
230                    "trigger {name:?}: unsupported kind {other:?} reached lowering"
231                )));
232            }
233        };
234        let retry = match entry.retry.as_ref() {
235            None => None,
236            Some(r) => {
237                let strategy = match r.backoff_strategy.as_str() {
238                    "fixed" => BackoffStrategy::Fixed,
239                    "exponential" => BackoffStrategy::Exponential,
240                    // validate_trigger already rejected anything else; defensive.
241                    other => {
242                        return Err(LowerError::Parse(alloc::format!(
243                            "trigger {name:?}: unsupported backoff strategy {other:?} reached lowering"
244                        )));
245                    }
246                };
247                Some(TriggerRetryPolicy {
248                    max_attempts: r.max_attempts,
249                    backoff: Backoff {
250                        strategy,
251                        base: r.backoff_base.clone(),
252                        max: r.backoff_max.clone(),
253                    },
254                    dead_letter: r.dead_letter.clone(),
255                })
256            }
257        };
258        triggers.push(TriggerBinding {
259            name: name.clone(),
260            kind,
261            agent: AgentId(entry.agent.clone()),
262            schedule: entry.schedule.clone(),
263            timezone: if entry.timezone.is_empty() {
264                None
265            } else {
266                Some(entry.timezone.clone())
267            },
268            retry,
269        });
270    }
271
272    // --- Pipeline ---------------------------------------------------------
273    let kinds: BTreeMap<alloc::string::String, alloc::vec::Vec<tau_domain::Capability>> = config
274        .agent_kinds
275        .iter()
276        .map(|(name, kind)| (name.clone(), kind.capabilities.clone()))
277        .collect();
278    let mut pipeline = match &config.pipeline {
279        None => None,
280        Some(p) => Some(Pipeline {
281            steps: p
282                .steps
283                .iter()
284                .map(|s| lower_step(s, &kinds))
285                .collect::<Result<_, _>>()?,
286        }),
287    };
288
289    // --- Checks (goals + deliverables) -----------------------------------
290    let checks = lower_checks(config, &mut pipeline)?;
291
292    Ok(Parsed {
293        workflow: Workflow {
294            agents,
295            tools,
296            steps,
297            edges,
298            capability_table: CapabilityTable(capability_table),
299            pipeline,
300            checks,
301        },
302        triggers,
303        assets,
304    })
305}

316/// Lower `[goals.*]`/`[deliverables.*]` into IR [`Check`]s and position a
317/// `StepRun::Check` step for each.
318///
319/// Checks already placed explicitly via a `check:<id>` pipeline step keep
320/// that position; the rest are appended at the pipeline tail in
321/// deterministic order (all goals by id, then all deliverables by id). If
322/// the config has no pipeline but does declare checks, a fresh pipeline is
323/// synthesized to hold the appended check steps.
324///
325/// Producer binding and gate resolution were performed by tau-pkg's
326/// validator (`DeliverableEntry::producer` / `::gate`), so this is a pure
327/// structural copy — no re-derivation here.
328/// Resolve a model alias to a concrete [`ModelRef`] via `[models]`, or via
329/// `[allow.models]` when the project declares an `[allow]` ceiling (a governed
330/// project moves its alias map under `[allow.models]`, per ADR-0057 / EPIC 1.2).
331///
332/// Infallible in practice — `validate_models` (tau-pkg) guarantees the alias
333/// exists in one of the two tables before lowering runs; the error arm is
334/// defense-in-depth.
335fn resolve_model_ref(
336    config: &ProjectConfig,
337    alias: &str,
338) -> Result<tau_ir::model_ref::ModelRef, LowerError> {
339    let m = config
340        .models
341        .get(alias)
342        .or_else(|| config.allow.as_ref().and_then(|a| a.models.get(alias)))
343        .ok_or_else(|| {
344            LowerError::Parse(alloc::format!(
345                "model alias `{alias}` not in [models] or [allow.models]"
346            ))
347        })?;
348    Ok(tau_ir::model_ref::ModelRef {
349        backend: m.backend.clone(),
350        model_id: m.model.clone(),
351    })
352}

354fn lower_checks(
355    config: &ProjectConfig,
356    pipeline: &mut Option<Pipeline>,
357) -> Result<BTreeMap<CheckId, Check>, LowerError> {
358    let mut checks: BTreeMap<CheckId, Check> = BTreeMap::new();
359
360    // Build the checks map, goals first then deliverables (BTreeMap order).
361    for (id, goal) in config.goals.iter() {
362        checks.insert(
363            CheckId(id.clone()),
364            Check {
365                id: CheckId(id.clone()),
366                verify: CheckVerify::Goal {
367                    evaluates: lower_locus(&goal.evaluates),
368                    predicate: lower_predicate(&goal.predicate),
369                },
370                // Goals always abort on failure; the gate is unused but
371                // must be a valid step id — set it to the check's own id.
372                retry: RetryPolicy {
373                    on_fail: OnFail::Abort,
374                    max_attempts: 1,
375                    gate: PipelineStepId(id.clone()),
376                },
377            },
378        );
379    }
380    for (id, d) in config.deliverables.iter() {
381        // `gate` was resolved by tau-pkg validation (empty for abort);
382        // fall back to the check's own id so the field is always valid.
383        let gate = if d.gate.is_empty() {
384            id.clone()
385        } else {
386            d.gate.clone()
387        };
388        checks.insert(
389            CheckId(id.clone()),
390            Check {
391                id: CheckId(id.clone()),
392                verify: CheckVerify::Deliverable {
393                    locus: lower_locus(&d.locus),
394                    must_satisfy: d.must_satisfy.clone(),
395                    judge: lower_judge(config, d)?,
396                },
397                retry: RetryPolicy {
398                    on_fail: lower_on_fail(&d.on_fail),
399                    max_attempts: d.max_attempts,
400                    gate: PipelineStepId(gate),
401                },
402            },
403        );
404    }
405
406    if checks.is_empty() {
407        return Ok(checks);
408    }
409
410    // Collect the check ids that are already explicitly positioned via a
411    // `check:<id>` pipeline step — those are not auto-appended.
412    let mut placed: alloc::collections::BTreeSet<alloc::string::String> =
413        alloc::collections::BTreeSet::new();
414    if let Some(p) = pipeline.as_ref() {
415        for s in p.steps.iter() {
416            if let StepRun::Check(CheckId(id)) = &s.run {
417                placed.insert(id.clone());
418            }
419        }
420    }
421
422    // Append a synthetic check step for every check not already placed, in
423    // deterministic order: goals by id, then deliverables by id.
424    let mut appended: alloc::vec::Vec<PipelineStep> = alloc::vec::Vec::new();
425    for id in config.goals.keys().chain(config.deliverables.keys()) {
426        if placed.contains(id) {
427            continue;
428        }
429        appended.push(PipelineStep {
430            id: PipelineStepId(id.clone()),
431            run: StepRun::Check(CheckId(id.clone())),
432            input: "${input}".into(),
433        });
434    }
435
436    match pipeline {
437        Some(p) => p.steps.extend(appended),
438        None => *pipeline = Some(Pipeline { steps: appended }),
439    }
440
441    Ok(checks)
442}

444/// Lower a tau-pkg [`AgentEntry`]'s `[agents.<id>.context]` pipeline into an
445/// IR [`ContextConfig`]. Returns `None` when no context pipeline is declared
446/// (the default behaviour — the agent loop applies no context transforms).
447///
448/// tau-pkg's validator shape-checks the custom-node `(source, package)` pairs
449/// but does **not** constrain the determinism string (it passes any value
450/// through via `unwrap_or_else(|| "pure")`). D7-B / ADR-0065 makes an
451/// unrecognized determinism string a hard `LowerError::UnknownDeterminism`
452/// here — the only build-time gate — instead of the former silent
453/// `_ => DeterminismClass::Pure` downgrade.
454fn lower_context(
455    agent: &str,
456    entry: &tau_pkg::project::AgentEntry,
457) -> Result<Option<tau_ir::context::ContextConfig>, LowerError> {
458    use tau_ir::context::{ContextConfig, ContextNodeKind, ContextStep, DeterminismClass};
459    if entry.context.is_empty() {
460        return Ok(None);
461    }
462    let mut pipeline = alloc::vec::Vec::with_capacity(entry.context.len());
463    for s in entry.context.iter() {
464        let determinism = match s.determinism.as_str() {
465            "pure" => DeterminismClass::Pure,
466            "llm_backed" => DeterminismClass::LlmBacked,
467            "stateful" => DeterminismClass::Stateful,
468            _ => {
469                return Err(LowerError::UnknownDeterminism {
470                    agent: agent.into(),
471                    transformer: s.transformer.clone(),
472                    determinism: s.determinism.clone(),
473                });
474            }
475        };
476        pipeline.push(ContextStep {
477            transformer: s.transformer.clone(),
478            determinism,
479            kind: match &s.custom {
480                Some((source, package)) => ContextNodeKind::Custom {
481                    source: source.clone(),
482                    package: package.clone(),
483                },
484                None => ContextNodeKind::Builtin,
485            },
486            config: s.config.clone(),
487        });
488    }
489    // `ContextConfig` is `#[non_exhaustive]`; it cannot be built with a
490    // struct literal from outside `tau-ir`. Build via `Default` + the
491    // public `pipeline` field instead (behavior-identical).
492    let mut cfg = ContextConfig::default();
493    cfg.pipeline = pipeline;
494    Ok(Some(cfg))
495}

497/// Convert a validated tau-pkg [`DurableEntry`] into the IR
498/// [`tau_ir::durable::Durability`] (ADR-0053).
499///
500/// tau-pkg's validator guarantees the strings are known values, so the
501/// mapping is total; the wildcard arms are defense-in-depth and resolve
502/// to the A-minimal defaults.
503pub fn durable_entry_to_ir(
504    entry: &tau_pkg::project::project::DurableEntry,
505) -> tau_ir::durable::Durability {
506    use tau_ir::durable::{CheckpointGranularity, Durability, DurabilityIntent, DurableStore};
507    use tau_pkg::project::project::DurableEntry;
508    match entry {
509        DurableEntry::Intent(s) => {
510            // tau-pkg validated the string; the mapping is total.
511            debug_assert_eq!(s, "survive-restarts");
512            Durability::Intent(DurabilityIntent::SurviveRestarts)
513        }
514        DurableEntry::Explicit {
515            checkpoint,
516            store: _,
517        } => {
518            // tau-pkg validated both strings; wildcard arms are defence-in-depth.
519            let checkpoint = match checkpoint.as_str() {
520                "per_tool_call" => CheckpointGranularity::PerToolCall,
521                _ => CheckpointGranularity::PerTurn,
522            };
523            // TODO: map store when DurableStore::Kv lands (A-full)
524            let store = DurableStore::File;
525            Durability::Explicit { checkpoint, store }
526        }
527    }
528}

537/// Lower one validated pipeline step to an IR [`PipelineStep`], recursing
538/// into `Branch`/`Parallel`/`Loop` arms. Reference/scope integrity is
539/// validated separately by the IR typecheck (`check_pipeline`). `Dynamic`
540/// regions resolve their `spawns` kind names against `kinds` here (EPIC
541/// 4.4), which is the one way this mapping can fail — an unknown kind
542/// yields `LowerError::UnknownAgentKind`.
543fn lower_step(
544    s: &tau_pkg::project::PipelineStepConfig,
545    kinds: &BTreeMap<alloc::string::String, alloc::vec::Vec<tau_domain::Capability>>,
546) -> Result<PipelineStep, LowerError> {
547    let run = match &s.run {
548        PipelineRunRef::Agent(id) => StepRun::Agent(AgentId(id.clone())),
549        PipelineRunRef::Tool(id) => StepRun::Tool(ToolId(id.clone())),
550        PipelineRunRef::Deterministic(id) => StepRun::Deterministic(StepId(id.clone())),
551        PipelineRunRef::Check(id) => StepRun::Check(CheckId(id.clone())),
552        PipelineRunRef::Suspend { resume_signal } => StepRun::Suspend {
553            resume_signal: resume_signal.clone(),
554        },
555        PipelineRunRef::Branch {
556            on,
557            then,
558            otherwise,
559        } => StepRun::Branch {
560            on: lower_condition(on),
561            then: then
562                .iter()
563                .map(|st| lower_step(st, kinds))
564                .collect::<Result<_, _>>()?,
565            otherwise: otherwise
566                .iter()
567                .map(|st| lower_step(st, kinds))
568                .collect::<Result<_, _>>()?,
569        },
570        PipelineRunRef::Parallel { branches } => StepRun::Parallel {
571            branches: branches
572                .iter()
573                .map(|b| b.iter().map(|st| lower_step(st, kinds)).collect())
574                .collect::<Result<_, _>>()?,
575        },
576        PipelineRunRef::Loop {
577            body,
578            until,
579            max_iters,
580        } => StepRun::Loop {
581            body: body
582                .iter()
583                .map(|st| lower_step(st, kinds))
584                .collect::<Result<_, _>>()?,
585            until: lower_condition(until),
586            max_iters: *max_iters,
587        },
588        PipelineRunRef::Dynamic {
589            spawns,
590            ceiling,
591            max_spawns,
592            max_concurrency,
593            agent: _,
594        } => {
595            let mut resolved = alloc::vec::Vec::with_capacity(spawns.len());
596            for kind in spawns {
597                let caps = kinds
598                    .get(kind)
599                    .ok_or_else(|| LowerError::UnknownAgentKind {
600                        kind: kind.clone(),
601                        step: s.id.clone(),
602                    })?;
603                resolved.push(tau_ir::pipeline::DynamicSpawn {
604                    kind: kind.clone(),
605                    capabilities: CapabilityRequirements {
606                        declared: caps.clone(),
607                    },
608                });
609            }
610            StepRun::Dynamic {
611                envelope: CapabilityRequirements {
612                    declared: ceiling.clone(),
613                },
614                spawns: resolved,
615                max_spawns: *max_spawns,
616                max_concurrency: *max_concurrency,
617            }
618        }
619    };
620    Ok(PipelineStep {
621        id: PipelineStepId(s.id.clone()),
622        run,
623        input: s.input.clone(),
624    })
625}

627/// Lower a tau-pkg [`ConditionConfig`](tau_pkg::project::ConditionConfig) to an
628/// IR [`Condition`](tau_ir::check::Condition), reusing the same locus/predicate
629/// mappings as `[goals.*]`.
630fn lower_condition(c: &tau_pkg::project::ConditionConfig) -> tau_ir::check::Condition {
631    tau_ir::check::Condition {
632        evaluates: lower_locus(&c.evaluates),
633        predicate: lower_predicate(&c.predicate),
634    }

637/// Map a tau-pkg [`LocusConfig`] to an IR [`Locus`].
638fn lower_locus(locus: &tau_pkg::project::LocusConfig) -> Locus {
639    use tau_pkg::project::LocusConfig;
640    match locus {
641        LocusConfig::Path(p) => Locus::Path(p.clone()),
642        LocusConfig::Output(s) => Locus::Output(PipelineStepId(s.clone())),
643    }
644}
645
646/// Map a tau-pkg [`GoalPredicateConfig`] to an IR [`GoalPredicate`].
647fn lower_predicate(p: &tau_pkg::project::GoalPredicateConfig) -> GoalPredicate {
648    use tau_pkg::project::GoalPredicateConfig;
649    match p {
650        GoalPredicateConfig::Exists => GoalPredicate::Exists,
651        GoalPredicateConfig::NonEmpty => GoalPredicate::NonEmpty,
652        GoalPredicateConfig::Equals(s) => GoalPredicate::Equals(s.clone()),
653        GoalPredicateConfig::Matches(s) => GoalPredicate::Matches(s.clone()),
654        GoalPredicateConfig::MinCount(n) => GoalPredicate::MinCount(*n),
655        GoalPredicateConfig::SchemaValid(v) => GoalPredicate::SchemaValid(v.clone()),
656        // Mirror how `Deterministic.fn_ref` is built above:
657        // `NativeFnRef { name }`.
658        GoalPredicateConfig::NativeFn(name) => {
659            GoalPredicate::NativeFn(NativeFnRef { name: name.clone() })
660        }
661    }
662}

664/// Map a tau-pkg [`OnFailConfig`] to an IR [`OnFail`].
665fn lower_on_fail(o: &tau_pkg::project::OnFailConfig) -> OnFail {
666    use tau_pkg::project::OnFailConfig;
667    match o {
668        OnFailConfig::Abort => OnFail::Abort,
669        OnFailConfig::Retry => OnFail::Retry,
670    }
671}
672
673/// Map a tau-pkg [`JudgeConfig`] to an IR [`JudgeRef`], resolving the model.
674///
675/// The canonical judge (`Default`) resolves its model from `[models]`: an
676/// explicit `judge_model` alias if present, otherwise it inherits the
677/// producing agent's resolved model (Q4). The producer is resolved by
678/// tau-pkg validation into `DeliverableEntry.producer`.
679fn lower_judge(
680    config: &ProjectConfig,
681    d: &tau_pkg::project::DeliverableEntry,
682) -> Result<JudgeRef, LowerError> {
683    use tau_pkg::project::JudgeConfig;
684    match &d.judge {
685        JudgeConfig::Default { model } => {
686            let model_ref = match model {
687                Some(alias) => resolve_model_ref(config, alias)?,
688                None => {
689                    let producer = config.agents.get(&d.producer).ok_or_else(|| {
690                        LowerError::Parse(alloc::format!(
691                            "deliverable `{}` producer `{}` not found",
692                            d.id,
693                            d.producer
694                        ))
695                    })?;
696                    resolve_model_ref(config, &producer.model)?
697                }
698            };
699            Ok(JudgeRef::Default { model_ref })
700        }
701        JudgeConfig::Agent(n) => Ok(JudgeRef::Agent(AgentId(n.clone()))),
702    }
703}
