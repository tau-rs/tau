1//! First lowering stage: pull typed declarations out of `ProjectConfig`.
2//!
3//! This stage does no I/O and no resolution. It walks the
4//! `ProjectConfig`'s agent/tool/step tables and produces an
5//! in-memory `Parsed` value: a partially-populated `Workflow` whose
6//! `ToolImpl::Native::content_hash` and similar resolution slots are
7//! filled with zero bytes (the `resolve` stage fills them).
8
9use alloc::collections::BTreeMap;
10use tau_pkg::project::{PipelineRunRef, ProjectConfig};
11use tau_pkg::{PromptEntry, ToolBody};
12
13use crate::error::LowerError;
14use tau_ir::capability::{CapabilityRequirements, CapabilityTable};
15use tau_ir::check::{Check, CheckVerify, GoalPredicate, JudgeRef, Locus, OnFail, RetryPolicy};
16use tau_ir::ids::{AgentId, CheckId, PipelineStepId, StepId, ToolId};
17use tau_ir::module::Workflow;
18use tau_ir::node::{Agent, Deterministic, Tool, ToolSpec};
19use tau_ir::pipeline::{Pipeline, PipelineStep, StepRun};
20use tau_ir::prompt::PromptSource;
21use tau_ir::subflow::SubflowEdge;
22use tau_ir::tool_impl::{Hash256, NativeFnRef, ToolImpl};
23use tau_ir::trigger::{
24    Backoff, BackoffStrategy, RetryPolicy as TriggerRetryPolicy, TriggerBinding, TriggerKind,
25};
26use tau_ir::AgentBudget;

28/// Output of the parse stage.
29#[derive(Debug)]
30pub(super) struct Parsed {
31    /// Partially-populated workflow (content hashes are zero pending `resolve`).
32    pub(super) workflow: Workflow,
33    /// Trigger bindings, canonically ordered by name (BTreeMap iteration).
34    pub(super) triggers: alloc::vec::Vec<TriggerBinding>,
35    /// Content-addressed asset blobs (currently `system_file` prompts) read
36    /// at build time, keyed by hash (`"sha256:" + 64 hex`). Deduped by hash.
37    pub(super) assets: BTreeMap<alloc::string::String, tau_ir::asset::AssetBlob>,
38}

40/// Run the parse stage on a `ProjectConfig`.
41///
42/// `prompt_file` reads an agent `system_file` prompt's bytes (D6-B); see
43/// [`crate::Caches::prompt_file`].
44pub(super) fn parse(
45    config: &ProjectConfig,
46    prompt_file: &dyn Fn(&std::path::Path) -> Result<alloc::vec::Vec<u8>, crate::PromptFileError>,
47) -> Result<Parsed, LowerError> {
48    let mut agents: BTreeMap<AgentId, Agent> = BTreeMap::new();
49    let mut tools: BTreeMap<ToolId, Tool> = BTreeMap::new();
50    let mut steps: BTreeMap<StepId, Deterministic> = BTreeMap::new();
51    let mut assets: BTreeMap<alloc::string::String, tau_ir::asset::AssetBlob> = BTreeMap::new();
52    let edges: alloc::vec::Vec<SubflowEdge> = alloc::vec::Vec::new();
53    let mut capability_table: BTreeMap<ToolId, CapabilityRequirements> = BTreeMap::new();
54
55    // --- Tools ---------------------------------------------------------
56    //
57    // `ProjectConfig::tools` is a `BTreeMap<String, ToolEntry>` produced by
58    // tau-pkg::config. Each `ToolEntry` discriminates Native vs Mcp vs Subflow.
59    for (name, entry) in config.tools.iter() {
60        let tool_id = ToolId(name.clone());
61        let caps = CapabilityRequirements {
62            declared: entry.capabilities.clone(),
63        };
64        let impl_ = match &entry.body {
65            ToolBody::Native(fn_name) => ToolImpl::Native {
66                fn_ref: NativeFnRef {
67                    name: fn_name.clone(),
68                },
69                // resolved by `resolve` stage; zero is a sentinel
70                content_hash: Hash256::default(),
71            },
72            ToolBody::Mcp(url) => ToolImpl::Mcp {
73                url: url.clone(),
74                contract_hash: Hash256::default(),
75                capability_subset: caps.clone(),
76                // server_tool_name is empty at parse time; the resolve stage
77                // expands this single entry into N entries (one per server-tool)
78                // with real names filled in.
79                server_tool_name: alloc::string::String::new(),
80            },
81            ToolBody::Subflow(target) => ToolImpl::Subflow {
82                target: AgentId(target.clone()),
83            },
84            // `ToolBody` is `#[non_exhaustive]`; future variants are
85            // forwarded as a parse error so existing callers aren't
86            // silently broken when a new variant lands.
87            _ => {
88                return Err(LowerError::Parse("unsupported tool body variant".into()));
89            }
90        };
91        let spec = ToolSpec {
92            name: name.clone(),
93            description: entry.description.clone(),
94            input_schema: entry.input_schema.clone(),
95        };
96        capability_table.insert(tool_id.clone(), caps.clone());
97        tools.insert(
98            tool_id.clone(),
99            Tool {
100                id: tool_id,
101                impl_,
102                capabilities: caps,
103                spec,
104            },
105        );
106    }
107
108    // --- Agents --------------------------------------------------------
109    for (name, entry) in config.agents.iter() {
110        let agent_id = AgentId(name.clone());
111        let tool_refs = entry.tool_refs.iter().cloned().map(ToolId).collect();
112        // Lower the prompt source (D6-B). `Inline` maps straight through.
113        // `File` is read at build time via the injected `prompt_file` reader,
114        // hashed, and emitted as a content-addressed `PromptSource::Asset`;
115        // the bytes are collected into `assets` (deduped by hash). Paths never
116        // enter the IR, and a missing/unreadable file is a hard build error —
117        // this fixes the non-hermetic `system_file` bug (the IR used to carry
118        // the path string as the prompt).
119        let prompt = match &entry.prompt {
120            PromptEntry::Inline(s) => PromptSource::inline(s.clone()),
121            PromptEntry::File(p) => {
122                let bytes = prompt_file(p).map_err(|e| LowerError::PromptFileUnreadable {
123                    agent: agent_id.clone(),
124                    path: p.to_string_lossy().into_owned(),
125                    reason: e.0,
126                })?;
127                let hash = tau_ir::asset::asset_hash(&bytes);
128                // Identical content across agents dedupes to one blob.
129                assets
130                    .entry(hash.clone())
131                    .or_insert_with(|| tau_ir::asset::AssetBlob::prompt(bytes));
132                PromptSource::asset(hash)
133            }
134            PromptEntry::None => PromptSource::inline(""),
135            // `PromptEntry` is `#[non_exhaustive]`, so this wildcard is
136            // required. A future variant reaching here is fail-closed (D7-B /
137            // ADR-0065): it used to silently become an empty prompt.
138            other => {
139                return Err(LowerError::UnsupportedPromptKind {
140                    agent: agent_id.clone(),
141                    detail: alloc::format!("{other:?}"),
142                });
143            }
144        };
145        agents.insert(
146            agent_id.clone(),
147            Agent {
148                id: agent_id,
149                prompt,
150                model_ref: resolve_model_ref(config, &entry.model)?,
151                tool_refs,
152                context: lower_context(name, entry)?,
153                budget: AgentBudget {
154                    max_turns: entry.max_turns,
155                    max_tokens: entry.max_tokens,
156                },
157                produces: entry.produces.clone(),
158                output_schema: entry.output_schema.clone(),
159                durable: lower_durable(entry),
160            },
161        );
162    }
163
164    // --- Deterministic steps ------------------------------------------
165    for (name, entry) in config.steps.iter() {
166        let step_id = StepId(name.clone());
167        steps.insert(
168            step_id.clone(),
169            Deterministic {
170                id: step_id,
171                fn_ref: NativeFnRef {
172                    name: entry.fn_name.clone(),
173                },
174                input_schema: entry.input_schema.clone(),
175                output_schema: entry.output_schema.clone(),
176            },
177        );
178    }
179
180    // --- Deterministic steps as tools ----------------------------------
181    //
182    // Each [steps.<name>] block registers BOTH a `Deterministic` node
183    // (above) and a `Tool` with `ToolImpl::Step { id }`. The Tool
184    // registration is what lets an agent reference the step in its
185    // `tool_refs`; the Deterministic node is what the runtime registry
186    // dispatches against at invoke time.
187    for (name, entry) in config.steps.iter() {
188        let step_id = StepId(name.clone());
189        let tool_id = ToolId(name.clone());
190
191        // Guard: a [tools.<name>] entry with the same key would be
192        // silently overwritten, causing the wrong implementation to be
193        // dispatched at runtime. Reject early with a clear error.
194        if tools.contains_key(&tool_id) {
195            return Err(LowerError::Parse(alloc::format!(
196                "step name {name:?} collides with an existing tool name; \
197                 rename the step or the tool"
198            )));
199        }
200
201        let caps = CapabilityRequirements {
202            declared: alloc::vec::Vec::new(),
203        };
204        let spec = ToolSpec {
205            name: name.clone(),
206            description: alloc::string::String::new(),
207            input_schema: entry.input_schema.clone(),
208        };
209        capability_table.insert(tool_id.clone(), caps.clone());
210        tools.insert(
211            tool_id.clone(),
212            Tool {
213                id: tool_id,
214                impl_: ToolImpl::Step { id: step_id },
215                capabilities: caps,
216                spec,
217            },
218        );
219    }
220
221    // --- Triggers (slice 1) --------------------------------------------
222    let mut triggers: alloc::vec::Vec<TriggerBinding> = alloc::vec::Vec::new();
223    for (name, entry) in config.triggers.iter() {
224        let kind = match entry.kind.as_str() {
225            "cron" => TriggerKind::Cron,
226            "manual" => TriggerKind::Manual,
227            // validate_trigger already rejected anything else; defensive.
228            other => {
229                return Err(LowerError::Parse(alloc::format!(
230                    "trigger {name:?}: unsupported kind {other:?} reached lowering"
231                )));
232            }
233        };
234        let retry = match entry.retry.as_ref() {
235            None => None,
236            Some(r) => {
237                let strategy = match r.backoff_strategy.as_str() {
238                    "fixed" => BackoffStrategy::Fixed,
239                    "exponential" => BackoffStrategy::Exponential,
240                    // validate_trigger already rejected anything else; defensive.
241                    other => {
242                        return Err(LowerError::Parse(alloc::format!(
243                            "trigger {name:?}: unsupported backoff strategy {other:?} reached lowering"
244                        )));
245                    }
246                };
247                Some(TriggerRetryPolicy {
248                    max_attempts: r.max_attempts,
249                    backoff: Backoff {
250                        strategy,
251                        base: r.backoff_base.clone(),
252                        max: r.backoff_max.clone(),
253                    },
254                    dead_letter: r.dead_letter.clone(),
255                })
256            }
257        };
258        triggers.push(TriggerBinding {
259            name: name.clone(),
260            kind,
261            agent: AgentId(entry.agent.clone()),
262            schedule: entry.schedule.clone(),
263            timezone: if entry.timezone.is_empty() {
264                None
265            } else {
266                Some(entry.timezone.clone())
267            },
268            retry,
269        });
270    }
271
272    // --- Pipeline ---------------------------------------------------------
273    let mut pipeline = config.pipeline.as_ref().map(|p| Pipeline {
274        steps: p.steps.iter().map(lower_step).collect(),
275    });
276
277    // --- Checks (goals + deliverables) -----------------------------------
278    let checks = lower_checks(config, &mut pipeline)?;
279
280    Ok(Parsed {
281        workflow: Workflow {
282            agents,
283            tools,
284            steps,
285            edges,
286            capability_table: CapabilityTable(capability_table),
287            pipeline,
288            checks,
289        },
290        triggers,
291        assets,
292    })
293}

304/// Lower `[goals.*]`/`[deliverables.*]` into IR [`Check`]s and position a
305/// `StepRun::Check` step for each.
306///
307/// Checks already placed explicitly via a `check:<id>` pipeline step keep
308/// that position; the rest are appended at the pipeline tail in
309/// deterministic order (all goals by id, then all deliverables by id). If
310/// the config has no pipeline but does declare checks, a fresh pipeline is
311/// synthesized to hold the appended check steps.
312///
313/// Producer binding and gate resolution were performed by tau-pkg's
314/// validator (`DeliverableEntry::producer` / `::gate`), so this is a pure
315/// structural copy — no re-derivation here.
316/// Resolve a model alias to a concrete [`ModelRef`] via `[models]`, or via
317/// `[allow.models]` when the project declares an `[allow]` ceiling (a governed
318/// project moves its alias map under `[allow.models]`, per ADR-0057 / EPIC 1.2).
319///
320/// Infallible in practice — `validate_models` (tau-pkg) guarantees the alias
321/// exists in one of the two tables before lowering runs; the error arm is
322/// defense-in-depth.
323fn resolve_model_ref(
324    config: &ProjectConfig,
325    alias: &str,
326) -> Result<tau_ir::model_ref::ModelRef, LowerError> {
327    let m = config
328        .models
329        .get(alias)
330        .or_else(|| config.allow.as_ref().and_then(|a| a.models.get(alias)))
331        .ok_or_else(|| {
332            LowerError::Parse(alloc::format!(
333                "model alias `{alias}` not in [models] or [allow.models]"
334            ))
335        })?;
336    Ok(tau_ir::model_ref::ModelRef {
337        backend: m.backend.clone(),
338        model_id: m.model.clone(),
339    })
340}

342fn lower_checks(
343    config: &ProjectConfig,
344    pipeline: &mut Option<Pipeline>,
345) -> Result<BTreeMap<CheckId, Check>, LowerError> {
346    let mut checks: BTreeMap<CheckId, Check> = BTreeMap::new();
347
348    // Build the checks map, goals first then deliverables (BTreeMap order).
349    for (id, goal) in config.goals.iter() {
350        checks.insert(
351            CheckId(id.clone()),
352            Check {
353                id: CheckId(id.clone()),
354                verify: CheckVerify::Goal {
355                    evaluates: lower_locus(&goal.evaluates),
356                    predicate: lower_predicate(&goal.predicate),
357                },
358                // Goals always abort on failure; the gate is unused but
359                // must be a valid step id — set it to the check's own id.
360                retry: RetryPolicy {
361                    on_fail: OnFail::Abort,
362                    max_attempts: 1,
363                    gate: PipelineStepId(id.clone()),
364                },
365            },
366        );
367    }
368    for (id, d) in config.deliverables.iter() {
369        // `gate` was resolved by tau-pkg validation (empty for abort);
370        // fall back to the check's own id so the field is always valid.
371        let gate = if d.gate.is_empty() {
372            id.clone()
373        } else {
374            d.gate.clone()
375        };
376        checks.insert(
377            CheckId(id.clone()),
378            Check {
379                id: CheckId(id.clone()),
380                verify: CheckVerify::Deliverable {
381                    locus: lower_locus(&d.locus),
382                    must_satisfy: d.must_satisfy.clone(),
383                    judge: lower_judge(config, d)?,
384                },
385                retry: RetryPolicy {
386                    on_fail: lower_on_fail(&d.on_fail),
387                    max_attempts: d.max_attempts,
388                    gate: PipelineStepId(gate),
389                },
390            },
391        );
392    }
393
394    if checks.is_empty() {
395        return Ok(checks);
396    }
397
398    // Collect the check ids that are already explicitly positioned via a
399    // `check:<id>` pipeline step — those are not auto-appended.
400    let mut placed: alloc::collections::BTreeSet<alloc::string::String> =
401        alloc::collections::BTreeSet::new();
402    if let Some(p) = pipeline.as_ref() {
403        for s in p.steps.iter() {
404            if let StepRun::Check(CheckId(id)) = &s.run {
405                placed.insert(id.clone());
406            }
407        }
408    }
409
410    // Append a synthetic check step for every check not already placed, in
411    // deterministic order: goals by id, then deliverables by id.
412    let mut appended: alloc::vec::Vec<PipelineStep> = alloc::vec::Vec::new();
413    for id in config.goals.keys().chain(config.deliverables.keys()) {
414        if placed.contains(id) {
415            continue;
416        }
417        appended.push(PipelineStep {
418            id: PipelineStepId(id.clone()),
419            run: StepRun::Check(CheckId(id.clone())),
420            input: "${input}".into(),
421        });
422    }
423
424    match pipeline {
425        Some(p) => p.steps.extend(appended),
426        None => *pipeline = Some(Pipeline { steps: appended }),
427    }
428
429    Ok(checks)
430}

432/// Lower a tau-pkg [`AgentEntry`]'s `[agents.<id>.context]` pipeline into an
433/// IR [`ContextConfig`]. Returns `None` when no context pipeline is declared
434/// (the default behaviour — the agent loop applies no context transforms).
435///
436/// tau-pkg's validator shape-checks the custom-node `(source, package)` pairs
437/// but does **not** constrain the determinism string (it passes any value
438/// through via `unwrap_or_else(|| "pure")`). D7-B / ADR-0065 makes an
439/// unrecognized determinism string a hard `LowerError::UnknownDeterminism`
440/// here — the only build-time gate — instead of the former silent
441/// `_ => DeterminismClass::Pure` downgrade.
442fn lower_context(
443    agent: &str,
444    entry: &tau_pkg::project::AgentEntry,
445) -> Result<Option<tau_ir::context::ContextConfig>, LowerError> {
446    use tau_ir::context::{ContextConfig, ContextNodeKind, ContextStep, DeterminismClass};
447    if entry.context.is_empty() {
448        return Ok(None);
449    }
450    let mut pipeline = alloc::vec::Vec::with_capacity(entry.context.len());
451    for s in entry.context.iter() {
452        let determinism = match s.determinism.as_str() {
453            "pure" => DeterminismClass::Pure,
454            "llm_backed" => DeterminismClass::LlmBacked,
455            "stateful" => DeterminismClass::Stateful,
456            _ => {
457                return Err(LowerError::UnknownDeterminism {
458                    agent: agent.into(),
459                    transformer: s.transformer.clone(),
460                    determinism: s.determinism.clone(),
461                });
462            }
463        };
464        pipeline.push(ContextStep {
465            transformer: s.transformer.clone(),
466            determinism,
467            kind: match &s.custom {
468                Some((source, package)) => ContextNodeKind::Custom {
469                    source: source.clone(),
470                    package: package.clone(),
471                },
472                None => ContextNodeKind::Builtin,
473            },
474            config: s.config.clone(),
475        });
476    }
477    // `ContextConfig` is `#[non_exhaustive]`; it cannot be built with a
478    // struct literal from outside `tau-ir`. Build via `Default` + the
479    // public `pipeline` field instead (behavior-identical).
480    let mut cfg = ContextConfig::default();
481    cfg.pipeline = pipeline;
482    Ok(Some(cfg))
483}

485/// Convert a validated tau-pkg [`DurableEntry`] into the IR
486/// [`tau_ir::durable::Durability`] (ADR-0053).
487///
488/// tau-pkg's validator guarantees the strings are known values, so the
489/// mapping is total; the wildcard arms are defense-in-depth and resolve
490/// to the A-minimal defaults.
491pub fn durable_entry_to_ir(
492    entry: &tau_pkg::project::project::DurableEntry,
493) -> tau_ir::durable::Durability {
494    use tau_ir::durable::{CheckpointGranularity, Durability, DurabilityIntent, DurableStore};
495    use tau_pkg::project::project::DurableEntry;
496    match entry {
497        DurableEntry::Intent(s) => {
498            // tau-pkg validated the string; the mapping is total.
499            debug_assert_eq!(s, "survive-restarts");
500            Durability::Intent(DurabilityIntent::SurviveRestarts)
501        }
502        DurableEntry::Explicit {
503            checkpoint,
504            store: _,
505        } => {
506            // tau-pkg validated both strings; wildcard arms are defence-in-depth.
507            let checkpoint = match checkpoint.as_str() {
508                "per_tool_call" => CheckpointGranularity::PerToolCall,
509                _ => CheckpointGranularity::PerTurn,
510            };
511            // TODO: map store when DurableStore::Kv lands (A-full)
512            let store = DurableStore::File;
513            Durability::Explicit { checkpoint, store }
514        }
515    }
516}

525/// Lower one validated pipeline step to an IR [`PipelineStep`], recursing
526/// into `Branch` arms. Reference/scope integrity is validated separately by
527/// the IR typecheck (`check_pipeline`), so this mapping stays infallible.
528fn lower_step(s: &tau_pkg::project::PipelineStepConfig) -> PipelineStep {
529    let run = match &s.run {
530        PipelineRunRef::Agent(id) => StepRun::Agent(AgentId(id.clone())),
531        PipelineRunRef::Tool(id) => StepRun::Tool(ToolId(id.clone())),
532        PipelineRunRef::Deterministic(id) => StepRun::Deterministic(StepId(id.clone())),
533        PipelineRunRef::Check(id) => StepRun::Check(CheckId(id.clone())),
534        PipelineRunRef::Suspend { resume_signal } => StepRun::Suspend {
535            resume_signal: resume_signal.clone(),
536        },
537        PipelineRunRef::Branch {
538            on,
539            then,
540            otherwise,
541        } => StepRun::Branch {
542            on: lower_condition(on),
543            then: then.iter().map(lower_step).collect(),
544            otherwise: otherwise.iter().map(lower_step).collect(),
545        },
546        PipelineRunRef::Parallel { branches } => StepRun::Parallel {
547            branches: branches
548                .iter()
549                .map(|b| b.iter().map(lower_step).collect())
550                .collect(),
551        },
552        PipelineRunRef::Loop {
553            body,
554            until,
555            max_iters,
556        } => StepRun::Loop {
557            body: body.iter().map(lower_step).collect(),
558            until: lower_condition(until),
559            max_iters: *max_iters,
560        },
561    };
562    PipelineStep {
563        id: PipelineStepId(s.id.clone()),
564        run,
565        input: s.input.clone(),
566    }
567}

588/// Map a tau-pkg [`GoalPredicateConfig`] to an IR [`GoalPredicate`].
589fn lower_predicate(p: &tau_pkg::project::GoalPredicateConfig) -> GoalPredicate {
590    use tau_pkg::project::GoalPredicateConfig;
591    match p {
592        GoalPredicateConfig::Exists => GoalPredicate::Exists,
593        GoalPredicateConfig::NonEmpty => GoalPredicate::NonEmpty,
594        GoalPredicateConfig::Equals(s) => GoalPredicate::Equals(s.clone()),
595        GoalPredicateConfig::Matches(s) => GoalPredicate::Matches(s.clone()),
596        GoalPredicateConfig::MinCount(n) => GoalPredicate::MinCount(*n),
597        GoalPredicateConfig::SchemaValid(v) => GoalPredicate::SchemaValid(v.clone()),
598        // Mirror how `Deterministic.fn_ref` is built above:
599        // `NativeFnRef { name }`.
600        GoalPredicateConfig::NativeFn(name) => {
601            GoalPredicate::NativeFn(NativeFnRef { name: name.clone() })
602        }
603    }
604}

615/// Map a tau-pkg [`JudgeConfig`] to an IR [`JudgeRef`], resolving the model.
616///
617/// The canonical judge (`Default`) resolves its model from `[models]`: an
618/// explicit `judge_model` alias if present, otherwise it inherits the
619/// producing agent's resolved model (Q4). The producer is resolved by
620/// tau-pkg validation into `DeliverableEntry.producer`.
621fn lower_judge(
622    config: &ProjectConfig,
623    d: &tau_pkg::project::DeliverableEntry,
624) -> Result<JudgeRef, LowerError> {
625    use tau_pkg::project::JudgeConfig;
626    match &d.judge {
627        JudgeConfig::Default { model } => {
628            let model_ref = match model {
629                Some(alias) => resolve_model_ref(config, alias)?,
630                None => {
631                    let producer = config.agents.get(&d.producer).ok_or_else(|| {
632                        LowerError::Parse(alloc::format!(
633                            "deliverable `{}` producer `{}` not found",
634                            d.id,
635                            d.producer
636                        ))
637                    })?;
638                    resolve_model_ref(config, &producer.model)?
639                }
640            };
641            Ok(JudgeRef::Default { model_ref })
642        }
643        JudgeConfig::Agent(n) => Ok(JudgeRef::Agent(AgentId(n.clone()))),
644    }
645}
