1//! #735: wasm-only build-time tool-dispatch gate (`tool_impl_fit`).
2//!
3//! Third sibling of [`feature_fit`](super::feature_fit) and
4//! [`predicate_fit`](super::predicate_fit). Where feature-fit asks "can the
5//! target *execute* this IR shape?" and predicate-fit asks "can the guest
6//! *answer* this fn?", tool-impl-fit asks "can the guest *dispatch* this
7//! tool at all?".
8//!
9//! `crates/tau-runtime-core/src/interpreter/agent_loop.rs` splits the four
10//! [`ToolImpl`] arms two ways: `Subflow` and `Step` are serviced by the
11//! interpreter itself (recursive spawn, and the dispatcher's
12//! `deterministic_registry()` — already gated by predicate-fit), while
13//! `Native` and `Mcp` are forwarded to the [`ToolDispatcher`]. The wasm
14//! guest's dispatcher (`crates/tau-wasm-guest/src/dispatcher.rs`) resolves
15//! only `Native`; an `Mcp` tool falls through to
16//! `tau_native_tools::invoke`, which does not know it, and the run dies
17//! mid-turn with `RuntimeError::Internal`.
18//!
19//! MCP needs a live transport (a spawned stdio server or an HTTP session)
20//! that a wasm guest cannot own, so this is not an oversight the guest can
21//! close on its own — it needs a host import. Until that exists, refusing
22//! at build time is strictly better than emitting a component that traps
23//! the first time the model reaches for the tool. **No override flag**,
24//! matching the Rust-like build-time enforcement principle its two
25//! siblings follow.
26

36/// Run the tool-impl-fit check on a `Parsed` workflow against a target.
37///
38/// Returns `Ok(())` for every non-Wasi target (this gate is wasm-only) and
39/// for a Wasi target whose tools are all guest-dispatchable. Returns
40/// `Err(LowerError::WasmToolImplUnsupported)` carrying the offending tool
41/// ids otherwise. Ids come out in `parsed.workflow.tools` iteration order,
42/// which is a `BTreeMap`, so the list is sorted and stable across runs.
43pub(super) fn check(parsed: &Parsed, target: &TargetTriple) -> Result<(), LowerError> {
44    if target.adapter_family != AdapterFamily::Wasi {
45        return Ok(());
46    }
47
48    let offending: Vec<ToolId> = parsed
49        .workflow
50        .tools
51        .iter()
52        .filter(|(_, tool)| matches!(tool.impl_, ToolImpl::Mcp { .. }))
53        .map(|(tool_id, _)| tool_id.clone())
54        .collect();
55
56    if offending.is_empty() {
57        Ok(())
58    } else {
59        Err(LowerError::WasmToolImplUnsupported {
60            tools: offending,
61            target: *target,
62        })
63    }
64}

134    fn parsed(toml: &str) -> Parsed {
135        let config = ProjectConfig::parse_str(toml).expect("toml parses");
136        parse::parse(&config, &no_prompt_files).expect("parse stage")
137    }
138
139    fn wasm() -> TargetTriple {
140        "any-wasi-strict".parse().unwrap()
141    }
142
143    fn native() -> TargetTriple {
144        "linux-native-strict".parse().unwrap()
145    }
146
147    #[test]
148    fn wasm_rejects_mcp_tool() {
149        let t = wasm();
150        let err = check(&parsed(MCP_TOOL_TOML), &t).expect_err("wasm must refuse an MCP tool");
151        match err {
152            LowerError::WasmToolImplUnsupported { tools, target } => {
153                assert_eq!(tools, alloc::vec![ToolId("weather".to_string())]);
154                assert_eq!(target, t);
155            }
156            other => panic!("expected WasmToolImplUnsupported, got {other:?}"),
157        }
158    }
159
160    #[test]
161    fn wasm_accepts_native_tools() {
162        assert!(check(&parsed(NATIVE_TOOLS_TOML), &wasm()).is_ok());
163    }
164
165    #[test]
166    fn wasm_blames_only_the_mcp_tool() {
167        let err = check(&parsed(MIXED_TOOLS_TOML), &wasm()).expect_err("refused");
168        match err {
169            LowerError::WasmToolImplUnsupported { tools, .. } => {
170                assert_eq!(tools, alloc::vec![ToolId("weather".to_string())]);
171            }
172            other => panic!("expected WasmToolImplUnsupported, got {other:?}"),
173        }
174    }
175
176    #[test]
177    fn wasm_reports_mcp_tools_sorted() {
178        let err = check(&parsed(TWO_MCP_TOOLS_TOML), &wasm()).expect_err("refused");
179        match err {
180            LowerError::WasmToolImplUnsupported { tools, .. } => {
181                assert_eq!(
182                    tools,
183                    alloc::vec![ToolId("alpha".to_string()), ToolId("zulu".to_string()),],
184                    "ids must come out in sorted BTreeMap order, not TOML order"
185                );
186            }
187            other => panic!("expected WasmToolImplUnsupported, got {other:?}"),
188        }
189    }
190
191    /// The gate is wasm-only: a native target dispatches MCP normally.
192    #[test]
193    fn native_target_accepts_mcp_tool() {
194        assert!(check(&parsed(MCP_TOOL_TOML), &native()).is_ok());
195    }
