2//! For MCP tools, also EXPANDS one author entry into N IR nodes
3//! (one per server-tool). See β.3 design doc §5.

17/// Run the resolve stage on a `Parsed` value.
18pub(super) fn resolve(mut parsed: Parsed, caches: &Caches<'_>) -> Result<Parsed, LowerError> {
19    // First pass: resolve native tool content_hashes (unchanged from β.2).
20    // Collect MCP entries that need expansion.
21    let mut mcp_entries: Vec<(ToolId, String)> = Vec::new(); // (old id, url)
22    for (id, tool) in parsed.workflow.tools.iter_mut() {
23        match &mut tool.impl_ {
24            ToolImpl::Native {
25                fn_ref,
26                content_hash,
27            } => {
28                if let Some(h) = (caches.native_tool)(&fn_ref.name) {
29                    *content_hash = h;
30                }
31                // If the cache returns None we KEEP the zero sentinel and
32                // let typecheck decide whether that's an error.
33            }
34            ToolImpl::Mcp { url, .. } => {
35                mcp_entries.push((id.clone(), url.clone()));
36            }
37            ToolImpl::Subflow { .. } | ToolImpl::Step { .. } => {}
38        }
39    }
40
41    // Second pass: per MCP entry, expand into N server-tool IR nodes.
42    //
43    // Cache-miss policy mirrors the native_tool path above: leave the
44    // original ToolImpl::Mcp untouched and let downstream stages (or
45    // a caller that wires an empty cache deliberately, e.g. the
46    // tau-ir-conformance tests that exercise the IR-level Mcp variant
47    // directly) decide whether that's an error. tau-cli's build path
48    // always populates the cache via PinnedResolver or
49    // LiveMcpContractResolver, so the strict "missing entry =>
50    // ContractUnreachable" behavior is enforced there via the
51    // resolver's own error path; resolve itself stays permissive.
52    for (old_id, url) in mcp_entries {
53        let Some(resolved) = (caches.mcp_contract)(&url) else {
54            continue;
55        };
56        expand_mcp_entry(&mut parsed, &old_id, &url, &resolved)?;
57    }
58
59    Ok(parsed)
60}
61
62fn expand_mcp_entry(
63    parsed: &mut Parsed,
64    old_id: &ToolId,
65    url: &str,
66    resolved: &ResolvedMcpContract,
67) -> Result<(), LowerError> {
68    // Capture the original tool (envelope + spec) before removing it.
69    let original_tool = parsed
70        .workflow
71        .tools
72        .get(old_id)
73        .cloned()
74        .expect("old_id was just collected from this map");
75    let envelope = match &original_tool.impl_ {
76        ToolImpl::Mcp {
77            capability_subset, ..
78        } => capability_subset.clone(),
79        _ => unreachable!("expand_mcp_entry called on non-Mcp tool"),
80    };
81
82    // Enforce SamplingRequiredByContract.
83    // TODO(PR-4-Phase-3): once Tool::sampling_models is wired from
84    // tau-pkg, check that the slice is non-empty here.
85    if resolved.requires_sampling {
86        // Dormant until Phase 3 adds Tool::sampling_models.
87    }
88
89    // Enforce no dot in server-tool names + caps subset + emit new entries.
90    let mut new_entries: Vec<(ToolId, Tool)> = Vec::new();
91    for st in &resolved.expanded_tools {
92        if st.name.contains('.') {
93            return Err(LowerError::McpBuild(
94                McpBuildError::ServerToolNameContainsDot {
95                    entry: old_id.0.clone(),
96                    name: st.name.clone(),
97                },
98            ));
99        }
100        let missing = caps_missing_from_envelope(&envelope, &st.caps);
101        if !missing.is_empty() {
102            return Err(LowerError::McpBuild(
103                McpBuildError::EnvelopeCoversContract {
104                    entry: old_id.0.clone(),
105                    tool: st.name.clone(),
106                    missing,
107                },
108            ));
109        }
110        let intersection = intersect_caps(&envelope, &st.caps);
111        let new_id = ToolId(alloc::format!("{}.{}", old_id.0, st.name));
112        let new_tool = Tool {
113            id: new_id.clone(),
114            spec: original_tool.spec.clone(),
115            impl_: ToolImpl::Mcp {
116                url: url.into(),
117                contract_hash: resolved.hash,
118                capability_subset: intersection,
119                server_tool_name: st.name.clone(),
120            },
121            capabilities: st.caps.clone(),
122        };
123        new_entries.push((new_id, new_tool));
124    }
125
126    // Remove old entry, insert N new ones.
127    parsed.workflow.tools.remove(old_id);
128    for (id, tool) in new_entries.iter() {
129        parsed.workflow.tools.insert(id.clone(), tool.clone());
130    }
131
132    // Rewrite each agent's tool_refs: replace old_id with all new_ids.
133    let new_ids: Vec<ToolId> = new_entries.iter().map(|(id, _)| id.clone()).collect();
134    for agent in parsed.workflow.agents.values_mut() {
135        let mut rewritten: Vec<ToolId> = Vec::with_capacity(agent.tool_refs.len());
136        for r in agent.tool_refs.iter() {
137            if r == old_id {
138                rewritten.extend(new_ids.iter().cloned());
139            } else {
140                rewritten.push(r.clone());
141            }
142        }
143        agent.tool_refs = rewritten;
144    }
145
146    Ok(())
147}

149/// Return the capability requirements for the expanded server-tool, bounded
150/// by the author's envelope.
151///
152/// v0: contract caps WIN when envelope ⊇ contract (enforced by
153/// `caps_missing_from_envelope` above). So the resolved entry holds
154/// exactly the contract caps (already a subset of the envelope).
155fn intersect_caps(
156    _envelope: &CapabilityRequirements,
157    contract: &CapabilityRequirements,
158) -> CapabilityRequirements {
159    // Envelope ⊇ contract is enforced by `caps_missing_from_envelope` before
160    // this function is called, so we can safely return the contract's caps.
161    contract.clone()
162}
163
164/// Return human-readable names for capabilities the contract declares
165/// that the envelope omits.
166///
167/// `CapabilityRequirements::declared` is a `Vec<tau_domain::Capability>`.
168/// `Capability` implements `PartialEq`, so we do a simple set-difference:
169/// for each item in `contract.declared`, check it appears in
170/// `envelope.declared`.
171fn caps_missing_from_envelope(
172    envelope: &CapabilityRequirements,
173    contract: &CapabilityRequirements,
174) -> Vec<String> {
175    contract
176        .declared
177        .iter()
178        .filter(|cap| !envelope.declared.contains(cap))
179        .map(|cap| alloc::format!("{cap:?}"))
180        .collect()
181}

190    fn rsc_minimal(name: &str) -> ResolvedServerTool {
191        ResolvedServerTool {

194            input_schema: serde_json::Value::Null,

198    #[test]
199    fn dot_in_server_tool_name_rejected() {
200        // Build a minimal tau.toml with one MCP entry, then supply a
201        // ResolvedMcpContract whose server-tool name contains a dot.
202        // resolve() should return McpBuildError::ServerToolNameContainsDot.
203        let toml = r#"
204packages = ["mock-llm"]
205
206[project]
207name = "p"
208
209[models]
210default = { backend = "mock-llm", model = "mock-model" }
211
212[agents.main]
213display_name = "Main"
214package      = "p@^0.1"
215model        = "default"
216tool_refs    = ["weather"]
217
218[tools.weather]
219mcp = "https://mcp.example.com"
220capabilities = []
221"#;
222        let config = ProjectConfig::parse_str(toml).expect("parse");
223        let parsed = super::super::parse::parse(&config, &super::super::parse::no_prompt_files)
224            .expect("parse stage");
225
226        let bad_contract = ResolvedMcpContract {
227            hash: [1u8; 32],
228            expanded_tools: alloc::vec![rsc_minimal("bad.name")],
229            requires_sampling: false,
230        };
231
232        let caches = crate::lower::Caches {
233            native_tool: &|_| None,
234            mcp_contract: &|_url| Some(bad_contract.clone()),
235            skill: &|_| None,
236            prompt_file: &|_| Ok(alloc::vec::Vec::new()),
237        };
238
239        let err = resolve(parsed, &caches).unwrap_err();
240        match &err {
241            LowerError::McpBuild(McpBuildError::ServerToolNameContainsDot { entry, name }) => {
242                assert_eq!(entry, "weather");
243                assert_eq!(name, "bad.name");
244            }
245            other => panic!("expected ServerToolNameContainsDot; got {other:?}"),
246        }
247    }

249    #[test]
250    fn mcp_entry_expands_to_n_tools_and_rewrites_agent_refs() {
251        // One MCP entry with two server-tools should become two IR tools,
252        // and the agent's tool_refs should be rewritten accordingly.
253        let toml = r#"
254packages = ["mock-llm"]
255
256[project]
257name = "p"
258
259[models]
260default = { backend = "mock-llm", model = "mock-model" }
261
262[agents.main]
263display_name = "Main"
264package      = "p@^0.1"
265model        = "default"
266tool_refs    = ["weather"]
267
268[tools.weather]
269mcp = "https://mcp.example.com"
270capabilities = []
271"#;
272        let config = ProjectConfig::parse_str(toml).expect("parse");
273        let parsed = super::super::parse::parse(&config, &super::super::parse::no_prompt_files)
274            .expect("parse stage");
275
276        let contract = ResolvedMcpContract {
277            hash: [2u8; 32],
278            expanded_tools: alloc::vec![rsc_minimal("get_forecast"), rsc_minimal("get_current"),],
279            requires_sampling: false,
280        };
281
282        let caches = crate::lower::Caches {
283            native_tool: &|_| None,
284            mcp_contract: &|_url| Some(contract.clone()),
285            skill: &|_| None,
286            prompt_file: &|_| Ok(alloc::vec::Vec::new()),
287        };
288
289        let result = resolve(parsed, &caches).expect("resolve");
290
291        // Old entry removed.
292        assert!(!result
293            .workflow
294            .tools
295            .contains_key(&ToolId("weather".into())));
296        // New entries present.
297        assert!(result
298            .workflow
299            .tools
300            .contains_key(&ToolId("weather.get_forecast".into())));
301        assert!(result
302            .workflow
303            .tools
304            .contains_key(&ToolId("weather.get_current".into())));
305
306        // Agent refs rewritten.
307        let agent = result
308            .workflow
309            .agents
310            .get(&AgentId("main".into()))
311            .expect("agent present");
312        assert!(agent
313            .tool_refs
314            .contains(&ToolId("weather.get_forecast".into())));
315        assert!(agent
316            .tool_refs
317            .contains(&ToolId("weather.get_current".into())));
318        assert!(!agent.tool_refs.contains(&ToolId("weather".into())));
319    }
