9/// Run the typecheck stage on a `Parsed` value.
10pub(super) fn typecheck(parsed: &Parsed) -> Result<(), LowerError> {
11    // 1. Each Agent::tool_refs entry must exist in `tools`.
12    for (agent_id, agent) in parsed.workflow.agents.iter() {
13        for tool_ref in agent.tool_refs.iter() {
14            if !parsed.workflow.tools.contains_key(tool_ref) {
15                return Err(LowerError::UnknownToolRef {
16                    agent: agent_id.clone(),
17                    tool: tool_ref.clone(),
18                });
19            }
20        }
21    }
22
23    // 2. Each Subflow::Spawn must reference an existing agent.
24    for edge in parsed.workflow.edges.iter() {
25        match &edge.kind {
26            SubflowKind::Spawn {
27                target_agent,
28                cap_subset: _,
29            } => {
30                if !parsed.workflow.agents.contains_key(target_agent) {
31                    return Err(LowerError::UnknownSubflowTarget {
32                        subflow: edge.id.clone(),
33                        agent: target_agent.clone(),
34                    });
35                }
36                // cap_subset's subset-of-parent check is deferred: the
37                // PARENT agent (the one that contains this edge) is the
38                // one whose grant we'd narrow. v0's tau.toml does not yet
39                // express the parent linkage; the lowering pass treats
40                // every edge as adjacent-to-every-agent. β.2.4
41                // (interpreter) will enforce cap_subset ⊆ caller's grant
42                // dynamically.
43            }
44            SubflowKind::Compose { .. } => {
45                return Err(LowerError::UnsupportedComposeSubflow {
46                    subflow: edge.id.clone(),
47                });
48            }
49        }
50    }
51
52    // 3. Sanity: every Native tool's content_hash must be non-zero.
53    //    (If it's still the resolve-stage sentinel, the native tool
54    //    cache didn't know about it — this is the place to refuse.)
55    for (tool_id, tool) in parsed.workflow.tools.iter() {
56        if let ToolImpl::Native {
57            fn_ref,
58            content_hash,
59        } = &tool.impl_
60        {
61            if content_hash == &[0u8; 32] {
62                return Err(LowerError::UnknownNativeTool {
63                    tool: tool_id.clone(),
64                    fn_name: fn_ref.name.clone(),
65                });
66            }
67        }
68    }
69
70    // 4+5. Every ToolImpl::Subflow's target must exist in `agents`;
71    //      every ToolImpl::Step's id must exist in `steps`.
72    //      Both checks iterate the same map — fold them into one pass.
73    for (tool_id, tool) in parsed.workflow.tools.iter() {
74        if let ToolImpl::Subflow { target } = &tool.impl_ {
75            if !parsed.workflow.agents.contains_key(target) {
76                return Err(LowerError::UnknownSubflowToolTarget {
77                    tool: tool_id.clone(),
78                    agent: target.clone(),
79                });
80            }
81        }
82        if let ToolImpl::Step { id } = &tool.impl_ {
83            if !parsed.workflow.steps.contains_key(id) {
84                return Err(LowerError::UnknownStepToolTarget {
85                    tool: tool_id.clone(),
86                    step: id.clone(),
87                });
88            }
89        }
90    }
91
92    // 6. Each trigger's entrypoint agent must exist.
93    for trigger in parsed.triggers.iter() {
94        if !parsed.workflow.agents.contains_key(&trigger.agent) {
95            return Err(LowerError::UnknownTriggerAgent {
96                trigger: trigger.name.clone(),
97                agent: trigger.agent.clone(),
98            });
99        }
100    }
101
102    // 7. Pipeline checks: run targets exist, no dup ids, no forward refs.
103    check_pipeline(&parsed.workflow)?;
104
105    // 8. Context-pipeline structural checks.
106    check_context(&parsed.workflow)?;
107
108    Ok(())
109}

111/// Structural validation of every agent's context pipeline.
112///
113/// Builtins: `trim_old`, `compact_tool_outputs`, `fit_budget`. Custom nodes
114/// (`ContextNodeKind::Custom`) are accepted structurally here; their
115/// capability grants are checked in tau-pkg. Rules:
116/// - the last step must be the builtin `fit_budget` (guarantees a ceiling);
117/// - no transformer name repeats;
118/// - a `Builtin`-kind step must be a known builtin name.
119fn check_context(wf: &tau_ir::module::Workflow) -> Result<(), LowerError> {
120    use alloc::collections::BTreeSet;
121    use tau_ir::context::ContextNodeKind;
122
123    const BUILTINS: [&str; 3] = ["trim_old", "compact_tool_outputs", "fit_budget"];
124
125    for (id, agent) in wf.agents.iter() {
126        let Some(ctx) = &agent.context else { continue };
127        if ctx.pipeline.is_empty() {
128            continue;
129        }
130        let mut seen: BTreeSet<&str> = BTreeSet::new();
131        for step in &ctx.pipeline {
132            if !seen.insert(step.transformer.as_str()) {
133                return Err(crate::error::LowerError::DuplicateContextTransformer {
134                    agent: id.0.clone(),
135                    transformer: step.transformer.clone(),
136                });
137            }
138            if matches!(step.kind, ContextNodeKind::Builtin)
139                && !BUILTINS.contains(&step.transformer.as_str())
140            {
141                return Err(crate::error::LowerError::UnknownContextTransformer {
142                    agent: id.0.clone(),
143                    transformer: step.transformer.clone(),
144                });
145            }
146        }
147        let last = ctx.pipeline.last().expect("non-empty checked above");
148        if last.transformer != "fit_budget" {
149            return Err(crate::error::LowerError::ContextFitBudgetNotLast {
150                agent: id.0.clone(),
151                last: last.transformer.clone(),
152            });
153        }
154    }
155    Ok(())
156}

158fn check_pipeline(wf: &tau_ir::module::Workflow) -> Result<(), LowerError> {
159    use alloc::collections::BTreeSet;
160    use tau_ir::pipeline::StepRun;
161    use tau_ir::template::{extract_refs, TemplateRef};
162
163    let Some(pipeline) = &wf.pipeline else {
164        return Ok(());
165    };
166
167    // Tree-wide id uniqueness: nested Branch/Parallel/Loop steps share the
168    // same flat global namespace as top-level steps (EPIC 4.2 Decision 1),
169    // so uniqueness must be checked across the whole tree up front.
170    let mut all_ids_preorder = alloc::vec::Vec::new();
171    collect_all_ids(&pipeline.steps, &mut all_ids_preorder);
172    let mut all_ids: BTreeSet<&str> = BTreeSet::new();
173    for id in &all_ids_preorder {
174        if !all_ids.insert(*id) {
175            return Err(LowerError::DuplicatePipelineStepId { id: (*id).into() });
176        }
177    }
178
179    let mut seen_ids: BTreeSet<&str> = BTreeSet::new();
180    // Nested descendants of PRIOR top-level steps are also visible to a
181    // later top-level step's template refs (EPIC 4.2 Decision 1: the store
182    // is flat and block-transparent, so a downstream step may read a
183    // Branch-arm or Loop-body output by bare id). Populated at the END of
184    // each iteration below, so at step `i` this holds exactly the subtree
185    // ids of steps `[0..i)` — never the current step's or a later step's
186    // nested descendants.
187    let mut visible_from_prior: BTreeSet<&str> = BTreeSet::new();
188    for step in &pipeline.steps {
189        let sid = step.id.0.as_str();
190        // Global uniqueness was already enforced above; this insert only
191        // maintains the execution-order-visibility scope for output refs.
192        seen_ids.insert(sid);
193
194        // For Check steps, emit the more precise UnknownCheckRef error when
195        // the check id is absent from workflow.checks.
196        if let StepRun::Check(check_id) = &step.run {
197            if !wf.checks.contains_key(check_id) {
198                return Err(LowerError::UnknownCheckRef {
199                    step: sid.into(),
200                    check: check_id.0.clone(),
201                });
202            }
203        }
204
205        // Validate this step's run target and recurse into any nested steps.
206        validate_step_run(&step.run, wf, sid, &all_ids, &seen_ids)?;
207
208        // For Check steps, validate the check's locus integrity:
209        // if the check has a Locus::Output referencing a step, that step must
210        // appear strictly BEFORE the current check step in the pipeline.
211        if let StepRun::Check(check_id) = &step.run {
212            if let Some(check) = wf.checks.get(check_id) {
213                use tau_ir::check::{CheckVerify, Locus};
214                let locus = match &check.verify {
215                    CheckVerify::Goal { evaluates, .. } => Some(evaluates),
216                    CheckVerify::Deliverable { locus, .. } => Some(locus),
217                };
218                if let Some(Locus::Output(ref_step_id)) = locus {
219                    // seen_ids holds exactly the steps strictly BEFORE current
220                    // (current step's id was just inserted above, so we check
221                    // using the pre-insertion state — but since we check
222                    // whether the referenced id is in seen_ids BEFORE we
223                    // process the current step's template refs, we need to
224                    // check whether ref_step_id was inserted strictly before
225                    // sid). seen_ids already has sid, so we check ≠ sid AND
226                    // seen_ids contains the ref (modulo the current step).
227                    let is_earlier =
228                        ref_step_id.0 != sid && seen_ids.contains(ref_step_id.0.as_str());
229                    if !is_earlier {
230                        return Err(LowerError::UnknownCheckLocus {
231                            check: check_id.0.clone(),
232                            output: ref_step_id.0.clone(),
233                        });
234                    }
235                }
236            }
237        }
238
239        let refs = extract_refs(&step.input).map_err(|e| LowerError::BadPipelineTemplate {
240            step: sid.into(),
241            detail: alloc::format!("{e}"),
242        })?;
243        for r in refs {
244            if let TemplateRef::StepOutput(ref_id) = r {
245                // Existence is checked tree-wide (EPIC 4.2 Decision 1: nested
246                // Branch/Loop/Parallel steps share the same flat global
247                // namespace as top-level steps), not just among top-level ids.
248                if !all_ids.contains(ref_id.as_str()) {
249                    return Err(LowerError::UnknownOutputRef {
250                        step: sid.into(),
251                        referenced: ref_id,
252                    });
253                }
254                // seen_ids currently holds exactly the top-level steps at-or-
255                // before this one (the current step's id was just inserted
256                // above). visible_from_prior holds the subtree ids of every
257                // STRICTLY earlier top-level step (Branch arms, Loop bodies,
258                // Parallel branches all included — a downstream step may
259                // read a block's result by bare id per Decision 1). A
260                // reference is valid iff it is a strictly earlier top-level
261                // step OR a nested descendant of one. Guard self-reference
262                // explicitly; the two `!contains` catches forward references
263                // (later top-level steps, or nested descendants of the
264                // current/later top-level steps).
265                if ref_id == sid
266                    || (!seen_ids.contains(ref_id.as_str())
267                        && !visible_from_prior.contains(ref_id.as_str()))
268                {
269                    return Err(LowerError::ForwardOutputRef {
270                        step: sid.into(),
271                        referenced: ref_id,
272                    });
273                }
274            }
275        }
276
277        // Make this step's whole subtree (including its own id, which
278        // harmlessly overlaps seen_ids) visible to later top-level steps.
279        let mut sub = alloc::vec::Vec::new();
280        collect_all_ids(core::slice::from_ref(step), &mut sub);
281        visible_from_prior.extend(sub);
282    }
283    Ok(())
284}

286/// Collect every `PipelineStepId` in the tree (top-level + nested inside
287/// Branch/Parallel/Loop), in pre-order, so uniqueness spans the whole
288/// pipeline (EPIC 4.2's flat global namespace, Decision 1).
289fn collect_all_ids<'a>(
290    steps: &'a [tau_ir::pipeline::PipelineStep],
291    out: &mut alloc::vec::Vec<&'a str>,
292) {
293    use tau_ir::pipeline::StepRun;
294    for s in steps {
295        out.push(s.id.0.as_str());
296        match &s.run {
297            StepRun::Branch {
298                then, otherwise, ..
299            } => {
300                collect_all_ids(then, out);
301                collect_all_ids(otherwise, out);
302            }
303            StepRun::Parallel { branches } => {
304                for b in branches {
305                    collect_all_ids(b, out);
306                }
307            }
308            StepRun::Loop { body, .. } => collect_all_ids(body, out),
309            _ => {}
310        }
311    }
312}

314/// Ref-check one nested step's `input` template against the scope in force
315/// at its execution site. Mirrors the top-level template loop in
316/// `check_pipeline`: `StepOutput(ref_id)` must exist somewhere in the tree
317/// (`all_ids`) — else `UnknownOutputRef` — and must be visible in the
318/// caller-supplied `scope` (an earlier sibling or an outer step, never the
319/// step itself) — else `ForwardOutputRef`.
320fn check_nested_template_refs(
321    sid: &str,
322    input: &str,
323    all_ids: &alloc::collections::BTreeSet<&str>,
324    scope: &alloc::collections::BTreeSet<&str>,
325) -> Result<(), LowerError> {
326    use tau_ir::template::{extract_refs, TemplateRef};
327
328    let refs = extract_refs(input).map_err(|e| LowerError::BadPipelineTemplate {
329        step: sid.into(),
330        detail: alloc::format!("{e}"),
331    })?;
332    for r in refs {
333        if let TemplateRef::StepOutput(ref_id) = r {
334            if !all_ids.contains(ref_id.as_str()) {
335                return Err(LowerError::UnknownOutputRef {
336                    step: sid.into(),
337                    referenced: ref_id,
338                });
339            }
340            if ref_id == sid || !scope.contains(ref_id.as_str()) {
341                return Err(LowerError::ForwardOutputRef {
342                    step: sid.into(),
343                    referenced: ref_id,
344                });
345            }
346        }
347    }
348    Ok(())
349}

351/// Validate a [`StepRun`]`s referenced nodes and recurse into nested steps.
352///
353/// `outer_step_id` is the id of the top-level pipeline step that owns this
354/// run (used in error messages). `all_ids` is the flat global namespace of
355/// every `PipelineStepId` in the tree (top-level + nested), used to
356/// distinguish "unknown anywhere" from "known but out of scope" when
357/// ref-checking nested `input` templates. `seen_ids` is the set of step ids
358/// that are in scope at the point this run executes (used for
359/// `Locus::Output` checks inside conditions, and as the starting scope for
360/// nested control-flow blocks).
361///
362/// Scope rules (EPIC 4.2 Decision 1 — flat global namespace + execution-order
363/// visibility):
364/// - `Branch.on` sees only the outer scope (it runs *before* either arm).
365/// - Each `Branch` arm (`then`/`otherwise`) sees the outer scope plus its own
366///   prior siblings — never the other arm's ids (only one arm executes).
367/// - `Loop.until` sees the outer scope *plus every body id* (it runs *after*
368///   each body pass).
369/// - `Loop` body steps see the outer scope plus their own prior siblings.
370/// - Each `Parallel` branch sees the outer scope plus its own prior steps
371///   only — never a sibling branch's ids (no ordering between branches).
372fn validate_step_run(
373    run: &tau_ir::pipeline::StepRun,
374    wf: &tau_ir::module::Workflow,
375    outer_step_id: &str,
376    all_ids: &alloc::collections::BTreeSet<&str>,
377    seen_ids: &alloc::collections::BTreeSet<&str>,
378) -> Result<(), LowerError> {
379    use tau_ir::check::Locus;
380    use tau_ir::pipeline::StepRun;
381
382    match run {
383        StepRun::Agent(a) => {
384            if !wf.agents.contains_key(a) {
385                return Err(LowerError::UnknownPipelineRun {
386                    step: outer_step_id.into(),
387                    target: alloc::format!("agent:{}", a.0),
388                });
389            }
390        }
391        StepRun::Tool(t) => {
392            if !wf.tools.contains_key(t) {
393                return Err(LowerError::UnknownPipelineRun {
394                    step: outer_step_id.into(),
395                    target: alloc::format!("tool:{}", t.0),
396                });
397            }
398        }
399        StepRun::Deterministic(s) => {
400            if !wf.steps.contains_key(s) {
401                return Err(LowerError::UnknownPipelineRun {
402                    step: outer_step_id.into(),
403                    target: alloc::format!("deterministic:{}", s.0),
404                });
405            }
406        }
407        StepRun::Check(check_id) => {
408            // For nested Check refs (inside Branch/Parallel/Loop bodies) the
409            // outer check_pipeline guard never fires — validate here so that
410            // a ghost check ref inside a control-flow block is caught.
411            if !wf.checks.contains_key(check_id) {
412                return Err(LowerError::UnknownCheckRef {
413                    step: outer_step_id.into(),
414                    check: check_id.0.clone(),
415                });
416            }
417        }
418        StepRun::Branch {
419            on,
420            then,
421            otherwise,
422        } => {
423            // Validate the condition's locus if it references a step output.
424            if let Locus::Output(ref_step_id) = &on.evaluates {
425                let is_in_scope =
426                    ref_step_id.0 != outer_step_id && seen_ids.contains(ref_step_id.0.as_str());
427                if !is_in_scope {
428                    return Err(LowerError::ConditionUnknownOutput {
429                        step: outer_step_id.into(),
430                        output: ref_step_id.0.clone(),
431                    });
432                }
433            }
434            // Each arm sees the outer scope plus its own prior siblings.
435            // Arms are mutually exclusive (only one executes), so `then`
436            // and `otherwise` never see each other's ids.
437            for arm in [then, otherwise] {
438                let mut arm_scope = seen_ids.clone();
439                for nested in arm {
440                    validate_step_run(&nested.run, wf, outer_step_id, all_ids, &arm_scope)?;
441                    check_nested_template_refs(
442                        nested.id.0.as_str(),
443                        &nested.input,
444                        all_ids,
445                        &arm_scope,
446                    )?;
447                    arm_scope.insert(nested.id.0.as_str());
448                }
449            }
450        }
451        StepRun::Parallel { branches } => {
452            // Each branch sees the outer scope plus its own prior steps —
453            // never a sibling branch's ids (no ordering between branches).
454            for branch in branches {
455                let mut branch_scope = seen_ids.clone();
456                for nested in branch {
457                    validate_step_run(&nested.run, wf, outer_step_id, all_ids, &branch_scope)?;
458                    check_nested_template_refs(
459                        nested.id.0.as_str(),
460                        &nested.input,
461                        all_ids,
462                        &branch_scope,
463                    )?;
464                    branch_scope.insert(nested.id.0.as_str());
465                }
466            }
467        }
468        StepRun::Loop {
469            body,
470            until,
471            max_iters,
472        } => {
473            // Enforce the mandatory iteration bound.
474            if *max_iters == 0 {
475                return Err(LowerError::LoopMaxItersZero {
476                    step: outer_step_id.into(),
477                });
478            }
479            // Loop.until runs AFTER each body pass, so it may read body
480            // outputs (the EPIC 4.1 deferred hole, closed here).
481            let mut loop_scope: alloc::collections::BTreeSet<&str> = seen_ids.clone();
482            for nested in body {
483                loop_scope.insert(nested.id.0.as_str());
484            }
485            if let Locus::Output(ref_step_id) = &until.evaluates {
486                let is_in_scope =
487                    ref_step_id.0 != outer_step_id && loop_scope.contains(ref_step_id.0.as_str());
488                if !is_in_scope {
489                    return Err(LowerError::ConditionUnknownOutput {
490                        step: outer_step_id.into(),
491                        output: ref_step_id.0.clone(),
492                    });
493                }
494            }
495            // Body steps see the outer scope plus their own prior siblings.
496            let mut body_scope = seen_ids.clone();
497            for nested in body {
498                validate_step_run(&nested.run, wf, outer_step_id, all_ids, &body_scope)?;
499                check_nested_template_refs(
500                    nested.id.0.as_str(),
501                    &nested.input,
502                    all_ids,
503                    &body_scope,
504                )?;
505                body_scope.insert(nested.id.0.as_str());
506            }
507        }
508        StepRun::Suspend { .. } => {
509            // No refs to validate.
510        }
511    }
512    Ok(())
513}
