1//! IR-level errors raised during parsing, lowering, capability-fit
2//! checking, canonicalization, and hashing.
3
4use alloc::string::String;
5use alloc::vec::Vec;
6use tau_domain::{CapabilityShape, IrFeature};
7use thiserror::Error;
8
9use tau_ir::ids::{AgentId, StepId, SubflowId, ToolId};
10use tau_ports::target::TargetTriple;
11
12/// IR-level error type.
13#[derive(Debug, Error)]
14pub enum LowerError {
15    /// Workflow-shape error: an Agent references a Tool that doesn't
16    /// exist in the workflow.
17    #[error("agent {agent:?} references unknown tool {tool:?}")]
18    UnknownToolRef {
19        /// Agent that contains the bad reference.
20        agent: AgentId,
21        /// The unknown tool id.
22        tool: ToolId,
23    },
24
25    /// Workflow-shape error: a SubflowEdge::Spawn targets an Agent that
26    /// doesn't exist.
27    #[error("subflow {subflow:?} targets unknown agent {agent:?}")]
28    UnknownSubflowTarget {
29        /// The subflow.
30        subflow: SubflowId,
31        /// The unknown target.
32        agent: AgentId,
33    },
34
35    /// Workflow-shape error: a SubflowEdge::Spawn's `cap_subset` is
36    /// not a subset of the parent agent's grant.
37    #[error("subflow {subflow:?}: cap_subset is not a subset of parent agent grant")]
38    SubflowCapNotSubset {
39        /// The offending subflow.
40        subflow: SubflowId,
41    },
42
43    /// Capability-fit failure (D-3b). One or more required capability
44    /// shapes are not supported by the build target.
45    #[error("workflow needs unsupported capability shape(s) on target: {missing:?}")]
46    CapabilityFitFailed {
47        /// The shapes that the target does not support.
48        missing: Vec<CapabilityShape>,
49        /// Diagnostic: which tools required them.
50        tools: Vec<ToolId>,
51    },
52
53    /// Feature-fit failure (EPIC 4.2). The workflow uses one or more IR
54    /// execution features the build target cannot run — e.g. any control-flow
55    /// (`Branch`/`Parallel`/`Loop`) for a wasm target, whose guest drives
56    /// `run_ir_streaming` and has no `run_pipeline` control-flow path.
57    #[error("workflow uses IR feature(s) unsupported by target {target}: {missing:?}")]
58    FeatureUnsupported {
59        /// The features the target does not support.
60        missing: Vec<IrFeature>,
61        /// The target that lacks them.
62        target: TargetTriple,
63    },
64
65    /// A Deterministic step references a function name that the lowering
66    /// registry doesn't know.
67    #[error("deterministic step {step:?} references unknown fn `{fn_name}`")]
68    UnknownDeterministicFn {
69        /// The step id.
70        step: StepId,
71        /// The unresolved name.
72        fn_name: String,
73    },
74
75    /// A `ToolImpl::Native` reference's content hash could not be resolved
76    /// (the native tool registry did not know the symbolic name).
77    #[error("native tool {tool:?} references unknown fn `{fn_name}`")]
78    UnknownNativeTool {
79        /// The tool id that contains the unresolved native ref.
80        tool: ToolId,
81        /// The native fn name that was not resolved.
82        fn_name: String,
83    },
84
85    /// A `ToolImpl::Subflow` tool targets an agent that is not present in
86    /// the workflow.
87    #[error("subflow tool {tool:?} targets unknown agent {agent:?}")]
88    UnknownSubflowToolTarget {
89        /// The tool id whose `Subflow` variant points at a missing agent.
90        tool: ToolId,
91        /// The unresolved target agent id.
92        agent: AgentId,
93    },
94
95    /// A `ToolImpl::Step` tool references a step id that is not present in
96    /// the workflow's `steps` table.
97    #[error("step tool {tool:?} references unknown step {step:?}")]
98    UnknownStepToolTarget {
99        /// The tool id whose `Step` variant points at a missing step.
100        tool: ToolId,
101        /// The unresolved step id.
102        step: StepId,
103    },
104
105    /// An agent's `system_file` prompt could not be read at build time.
106    /// This deliberately moves prompt-file existence from run time to build
107    /// time (D6-B): a missing or unreadable prompt file is a hard build error.
108    #[error("agent {agent:?}: cannot read prompt file {path:?}: {reason}")]
109    PromptFileUnreadable {
110        /// The agent whose `system_file` prompt failed to load.
111        agent: AgentId,
112        /// The prompt file path, as written in the config.
113        path: String,
114        /// Why the read failed (from the injected `prompt_file` reader).
115        reason: String,
116    },
117
118    /// Generic parse failure surfacing from the upstream TOML parser.
119    #[error("tau.toml parse error: {0}")]
120    Parse(String),
121
122    /// MCP-specific build error (per β.3 design doc §5).
123    #[error("MCP build: {0}")]
124    McpBuild(#[from] crate::lower::McpBuildError),
125
126    /// SubflowEdge::Compose is not yet implemented (v0 reserves the variant).
127    #[error("subflow {subflow:?}: Compose variant is not supported in v0")]
128    UnsupportedComposeSubflow {
129        /// The offending subflow.
130        subflow: SubflowId,
131    },
132
133    /// A trigger binding names an entrypoint agent that is not present in
134    /// the workflow.
135    #[error("trigger {trigger:?} references unknown agent {agent:?}")]
136    UnknownTriggerAgent {
137        /// The trigger name.
138        trigger: String,
139        /// The unresolved entrypoint agent id.
140        agent: AgentId,
141    },
142
143    /// A pipeline step's `run` target does not exist in the workflow.
144    #[error("pipeline step {step:?}: run target {target} not found")]
145    UnknownPipelineRun {
146        /// The pipeline step id.
147        step: String,
148        /// The unresolved `kind:id` target, e.g. `agent:writer`.
149        target: String,
150    },
151
152    /// Two pipeline steps share an id.
153    #[error("pipeline step id {id:?} is declared more than once")]
154    DuplicatePipelineStepId {
155        /// The duplicated id.
156        id: String,
157    },
158
159    /// `${steps.x.output}` references a step that runs at or after this one.
160    #[error(
161        "pipeline step {step:?} references output of {referenced:?}, which is not an earlier step"
162    )]
163    ForwardOutputRef {
164        /// The referencing step.
165        step: String,
166        /// The referenced (later/self) step id.
167        referenced: String,
168    },
169
170    /// `${steps.x.output}` references a step id not in the pipeline.
171    #[error("pipeline step {step:?} references unknown step output {referenced:?}")]
172    UnknownOutputRef {
173        /// The referencing step.
174        step: String,
175        /// The unknown referenced id.
176        referenced: String,
177    },
178
179    /// A pipeline input template was malformed (unterminated/unrecognized).
180    #[error("pipeline step {step:?}: bad input template: {detail}")]
181    BadPipelineTemplate {
182        /// The step id.
183        step: String,
184        /// Human-readable template error.
185        detail: String,
186    },
187
188    /// A `StepRun::Check` references a check id absent from `workflow.checks`.
189    #[error("pipeline step '{step}' runs check '{check}' but no such check is defined")]
190    UnknownCheckRef {
191        /// The pipeline step id that contains the bad reference.
192        step: String,
193        /// The check id that was not found in `workflow.checks`.
194        check: String,
195    },
196
197    /// A context pipeline names a transformer that is neither a known
198    /// builtin nor a declared custom node.
199    #[error("agent '{agent}': context transformer '{transformer}' is not a known builtin or custom node")]
200    UnknownContextTransformer {
201        /// The agent id whose context pipeline is invalid.
202        agent: String,
203        /// The offending transformer name.
204        transformer: String,
205    },
206
207    /// A context pipeline's last step is not the builtin `fit_budget`.
208    #[error("agent '{agent}': the last context step must be `fit_budget` (found '{last}')")]
209    ContextFitBudgetNotLast {
210        /// The agent id whose context pipeline is invalid.
211        agent: String,
212        /// The actual last transformer name.
213        last: String,
214    },
215
216    /// A context pipeline repeats a transformer name.
217    #[error("agent '{agent}': duplicate context transformer '{transformer}'")]
218    DuplicateContextTransformer {
219        /// The agent id whose context pipeline is invalid.
220        agent: String,
221        /// The repeated transformer name.
222        transformer: String,
223    },
224
225    /// A check's `Locus::Output` references an unknown or later pipeline step.
226    #[error(
227        "check '{check}' evaluates output of '{output}' which is not an earlier pipeline step"
228    )]
229    UnknownCheckLocus {
230        /// The check id whose locus is invalid.
231        check: String,
232        /// The referenced step id that is missing or comes at/after the check step.
233        output: String,
234    },
235
236    /// A `StepRun::Loop` has `max_iters == 0`, which would never execute a
237    /// body iteration. The bound must be at least 1.
238    #[error("pipeline step '{step}': Loop.max_iters must be > 0")]
239    LoopMaxItersZero {
240        /// The pipeline step id containing the offending Loop.
241        step: String,
242    },
243
244    /// A dynamic region references an agent kind with no `[agent.kinds.<name>]`
245    /// definition (EPIC 4.4).
246    #[error("dynamic region in step '{step}' spawns unknown agent kind '{kind}' — declare [agent.kinds.{kind}]")]
247    UnknownAgentKind {
248        /// The undefined kind name.
249        kind: String,
250        /// The pipeline-step id of the region.
251        step: String,
252    },
253
254    /// A `StepRun::Dynamic` has `max_spawns == 0`, which would never spawn a
255    /// region member. The bound must be at least 1. Defense-in-depth: mirrors
256    /// `LoopMaxItersZero` — tau-pkg validates this at author time, but
257    /// hand-constructed IR that bypasses tau-pkg must still be rejected.
258    #[error("pipeline step '{step}': Dynamic.max_spawns must be > 0")]
259    DynamicMaxSpawnsZero {
260        /// The pipeline step id containing the offending Dynamic region.
261        step: String,
262    },
263
264    /// A `StepRun::Dynamic` has `max_concurrency` outside `1..=max_spawns`
265    /// (either zero, or greater than the total-spawn cap). Defense-in-depth:
266    /// mirrors `LoopMaxItersZero` for the Dynamic region's own bounds.
267    #[error("pipeline step '{step}': Dynamic.max_concurrency must be in 1..={max_spawns} (got {max_concurrency})")]
268    DynamicConcurrencyInvalid {
269        /// The pipeline step id containing the offending Dynamic region.
270        step: String,
271        /// The region's configured total-spawn cap.
272        max_spawns: u64,
273        /// The offending concurrency value.
274        max_concurrency: u64,
275    },
276
277    /// A context step declares a determinism class string that lowering does
278    /// not recognize (D7-B / ADR-0065). Replaces the former silent
279    /// `_ => DeterminismClass::Pure` default, which downgraded an unknown
280    /// string to the most permissive class.
281    #[error("agent '{agent}': context transformer '{transformer}' declares unknown determinism '{determinism}' (want one of: pure, llm_backed, stateful)")]
282    UnknownDeterminism {
283        /// The agent id whose context pipeline is invalid.
284        agent: String,
285        /// The offending transformer name.
286        transformer: String,
287        /// The unrecognized determinism string as authored.
288        determinism: String,
289    },
290
291    /// Lowering encountered a `PromptEntry` variant it does not know how to
292    /// lower (D7-B / ADR-0065). Replaces the former silent wildcard that
293    /// mapped an unknown prompt kind to an empty prompt. Only reachable if a
294    /// future `#[non_exhaustive]` `PromptEntry` variant is added upstream
295    /// without a corresponding lowering arm — a fail-closed guard, not a
296    /// user-authoring error.
297    #[error("agent {agent:?}: unsupported prompt kind (this tau cannot lower {detail}); rebuild with a matching tau")]
298    UnsupportedPromptKind {
299        /// The agent whose prompt could not be lowered.
300        agent: AgentId,
301        /// Debug rendering of the offending `PromptEntry`.
302        detail: String,
303    },
304
305    /// A `Condition::evaluates` is a `Locus::Output(id)` that names a step
306    /// not yet visible at this point in the pipeline.
307    #[error(
308        "condition in step '{step}' references unknown or out-of-scope output step '{output}'"
309    )]
310    ConditionUnknownOutput {
311        /// The pipeline step id containing the condition.
312        step: String,
313        /// The referenced step id that is missing or out of scope.
314        output: String,
315    },
316
317    /// A `Suspend` step appeared below the top-level pipeline slice (inside a
318    /// Branch arm, Loop body, or Parallel branch). Suspend is top-level only.
319    #[error("suspend step {step:?} is nested inside a control-flow block; suspend is only allowed at the top level of the pipeline (EPIC 4.3)")]
320    SuspendNotTopLevel {
321        /// The nested suspend step id.
322        step: String,
323    },
324
325    /// A step template references `${steps.<id>.output}` where `<id>` is a
326    /// `Suspend` step, which produces no output.
327    #[error("step {step:?} references {referenced:?}.output, but {referenced:?} is a suspend step and produces no output")]
328    SuspendHasNoOutput {
329        /// The referencing step.
330        step: String,
331        /// The suspend step id whose (nonexistent) output was referenced.
332        referenced: String,
333    },
334}

336#[cfg(test)]
337mod tests {
338    use super::*;
339    use alloc::string::ToString;
340
341    #[test]
342    fn unknown_determinism_renders_the_offending_values() {
343        let e = LowerError::UnknownDeterminism {
344            agent: "reviewer".into(),
345            transformer: "trim_old".into(),
346            determinism: "sometimes".into(),
347        };
348        let msg = e.to_string();
349        assert!(msg.contains("reviewer"), "got: {msg}");
350        assert!(msg.contains("trim_old"), "got: {msg}");
351        assert!(msg.contains("sometimes"), "got: {msg}");
352        assert!(msg.contains("pure"), "should name the valid set: {msg}");
353    }
354
355    // `UnsupportedPromptKind` is only reachable if a future
356    // `#[non_exhaustive]` `PromptEntry` variant is added upstream without a
357    // lowering arm; it cannot be triggered through the public lowering API
358    // (the three current variants are all handled). This asserts its Display
359    // is actionable so the fail-closed message is not vacuous.
360    #[test]
361    fn unsupported_prompt_kind_renders_actionably() {
362        let e = LowerError::UnsupportedPromptKind {
363            agent: AgentId("writer".into()),
364            detail: "SomeFutureVariant".into(),
365        };
366        let msg = e.to_string();
367        assert!(msg.contains("writer"), "got: {msg}");
368        assert!(msg.contains("SomeFutureVariant"), "got: {msg}");
369        assert!(msg.contains("rebuild"), "got: {msg}");
370    }
