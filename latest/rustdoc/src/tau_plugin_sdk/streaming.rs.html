1//! Helper: turn a `Stream<Item = Result<CompletionChunk, LlmError>>`
2//! into a series of `stream.chunk` notifications carrying the
3//! originating msgid, plus a final [`Frame::Response`] carrying
4//! `{ stop_reason, usage }` or an error envelope.
5//!
6//! See `docs/superpowers/specs/2026-04-28-plugin-loading-design.md`
7//! §4.6 for the wire-level streaming protocol.
8
9use futures_util::StreamExt;
10use serde::Serialize;
11use tau_plugin_protocol::{
12    error::{RpcErrorEnvelope, INTERNAL_ERROR},
13    Frame, FramedWriter,
14};
15use tokio::io::AsyncWrite;
16use tokio::pin;
17
18use crate::error::SdkError;
19
20/// Final summary serialized into the terminal `Frame::Response::result`
21/// at the end of a streaming dispatch.
22#[derive(Debug, Clone, Serialize)]
23struct StreamSummary {
24    stop_reason: Option<tau_ports::StopReason>,
25    usage: Option<tau_ports::TokenUsage>,
26}
27
28/// Pump an LLM completion stream out as `stream.chunk` notifications,
29/// terminating with a [`Frame::Response`] carrying the final summary
30/// or an error envelope.
31///
32/// The `msgid` is the originating `llm.stream` request id.
33/// Each [`tau_ports::CompletionChunk`] yielded by the stream is encoded
34/// as the second element of a `(msgid, chunk)` `stream.chunk`
35/// notification. When the stream ends cleanly, a single
36/// [`Frame::Response`] is sent carrying a JSON-shaped summary payload
37/// (`{ stop_reason, usage }`); both fields default to `None` if no
38/// terminal [`tau_ports::CompletionChunk::Finish`] was observed. If
39/// the stream yields an error, that final response carries an error
40/// envelope with code [`INTERNAL_ERROR`] and the dispatch terminates
41/// without exhausting any further items.
42///
43/// # Example
44///
45/// ```no_run
46/// # use futures_util::stream;
47/// # use tau_ports::{CompletionChunk, CompletionStream, LlmError, StopReason, TokenUsage};
48/// # use tau_plugin_protocol::{FramedWriter};
49/// #[tokio::main]
50/// async fn main() -> Result<(), Box<dyn std::error::Error>> {
51///     let stdout = tokio::io::stdout();
52///     let mut writer = FramedWriter::new(stdout);
53///     // Build a minimal two-chunk stream: one text chunk + one finish.
54///     let chunks: CompletionStream = Box::pin(stream::iter(vec![
55///         Ok(CompletionChunk::Text { delta: "hello".to_string() }),
56///         Ok(CompletionChunk::Finish {
57///             stop_reason: StopReason::EndTurn,
58///             usage: None,
59///         }),
60///     ]));
61///     tau_plugin_sdk::stream_completion(&mut writer, 1, chunks).await?;
62///     Ok(())
63/// }
64/// ```
65pub async fn stream_completion<W>(
66    writer: &mut FramedWriter<W>,
67    msgid: u32,
68    stream: tau_ports::CompletionStream,
69) -> Result<(), SdkError>
70where
71    W: AsyncWrite + Unpin,
72{
73    pin!(stream);
74    let mut final_stop_reason: Option<tau_ports::StopReason> = None;
75    let mut final_usage: Option<tau_ports::TokenUsage> = None;
76
77    while let Some(item) = stream.next().await {
78        match item {
79            Ok(chunk) => {
80                // Capture the terminal Finish marker for the final summary.
81                if let tau_ports::CompletionChunk::Finish { stop_reason, usage } = &chunk {
82                    final_stop_reason = Some(*stop_reason);
83                    final_usage = *usage;
84                }
85                // Forward the chunk as a `stream.chunk` notification carrying
86                // the originating msgid + the chunk itself.
87                let params_bytes = rmp_serde::to_vec(&(msgid, &chunk))?;
88                let frame = Frame::Notification {
89                    method: "stream.chunk".to_string(),
90                    params: params_bytes,
91                };
92                writer.write_frame(&frame.encode()?).await?;
93            }
94            Err(llm_err) => {
95                // Stream errored mid-flight: terminate with an error response.
96                let envelope = RpcErrorEnvelope::new(
97                    INTERNAL_ERROR,
98                    format!("llm stream error: {llm_err}"),
99                    None,
100                );
101                let frame = Frame::Response {
102                    id: msgid,
103                    error: Some(envelope),
104                    result: None,
105                };
106                writer.write_frame(&frame.encode()?).await?;
107                return Ok(());
108            }
109        }
110    }
111
112    // Stream ended cleanly: send the final summary response.
113    let summary = StreamSummary {
114        stop_reason: final_stop_reason,
115        usage: final_usage,
116    };
117    let result_bytes = rmp_serde::to_vec(&summary)?;
118    let frame = Frame::Response {
119        id: msgid,
120        error: None,
121        result: Some(result_bytes),
122    };
123    writer.write_frame(&frame.encode()?).await?;
124    Ok(())
125}
