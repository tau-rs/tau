1//! tracing-subscriber JSON layer that writes structured events to
2//! stderr. The host (in `tau-runtime::plugin_host`) reads each line,
3//! decodes the JSON, and re-emits as a `tracing::Event` on
4//! `target = "plugin::<plugin_name>"`.
5//!
6//! Internals delegate to [`tau_observe::install`] so all tau crates
7//! share one subscriber-init code path.
8
9/// Install the SDK's stderr-JSON tracing layer.
10///
11/// Idempotent: subsequent calls are no-ops. Plugin authors should
12/// call this once at the start of `main()`, OR call one of the
13/// `run_*` runners (which install it internally).
14///
15/// The default filter level is `info`; override via `RUST_LOG` env var
16/// (e.g., `RUST_LOG=tau_plugin_sdk=debug,my_plugin=trace`).
17///
18/// # Example
19///
20/// ```
21/// // Calling install() more than once is safe — subsequent calls are no-ops.
22/// tau_plugin_sdk::tracing_layer::install();
23/// tau_plugin_sdk::tracing_layer::install();
24/// // No panic means install is idempotent.
25/// assert!(true);
26/// ```
27pub fn install() {
28    let _guard = tau_observe::install::install(tau_observe::install::InstallOptions::plugin_sdk())
29        .expect("tau_observe::install never returns Err in current impl");
30}
