1//! Sequential pipeline: an ordered list of steps the engine executes
2//! top-to-bottom, threading each step's output to later steps via
3//! `${steps.<id>.output}` templating.
4
5use alloc::string::String;
6use alloc::vec::Vec;
7use serde::{Deserialize, Serialize};
8
9use crate::capability::CapabilityRequirements;
10use crate::check::Condition;
11use crate::ids::{AgentId, PipelineStepId, StepId, ToolId};
12
13/// An ordered, engine-sequenced pipeline of steps.
14#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
15#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
16pub struct Pipeline {
17    /// Steps, executed top-to-bottom in this order.
18    pub steps: Vec<PipelineStep>,
19}
20
21/// One step in a [`Pipeline`].
22#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
23#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
24pub struct PipelineStep {
25    /// Handle for this step; its output is addressable as
26    /// `steps.<id>.output` by later steps.
27    pub id: PipelineStepId,
28    /// What this step runs.
29    pub run: StepRun,
30    /// Input template (`${input}`, `${steps.<id>.output}`).
31    pub input: String,
32}
33
34/// One spawnable per-kind agent definition inside a [`StepRun::Dynamic`]
35/// region, resolved with its capability grant so the runtime gate
36/// (EPIC 4.5) is self-contained (no re-resolution against `tau.toml`).
37#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
38#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
39pub struct DynamicSpawn {
40    /// The agent-kind name (`[agent.kinds.<kind>]`).
41    pub kind: String,
42    /// The kind's resolved capability grant.
43    pub capabilities: CapabilityRequirements,
44}
45
46/// What a [`PipelineStep`] executes — a reference to an existing node.
47#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
48#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
49pub enum StepRun {
50    /// Run an agent node by id.
51    Agent(AgentId),
52    /// Invoke a tool node by id.
53    Tool(ToolId),
54    /// Run a deterministic step node by id.
55    Deterministic(StepId),
56    /// Evaluate a postcondition check by id.
57    ///
58    /// The referenced [`CheckId`](crate::ids::CheckId) must exist in
59    /// `workflow.checks`; this is enforced by the typecheck stage.
60    /// Runtime evaluation is implemented in `tau-runtime-core`'s
61    /// pipeline executor (Task 19).
62    Check(crate::ids::CheckId),
63    /// EPIC 4.1: run `then` if `on` holds, else `otherwise` (may be empty).
64    /// Executed by the interpreter in 4.2.
65    Branch {
66        /// The branch condition.
67        on: Condition,
68        /// Steps run when `on` holds.
69        then: Vec<PipelineStep>,
70        /// Steps run when `on` does not hold (may be empty).
71        otherwise: Vec<PipelineStep>,
72    },
73    /// EPIC 4.1: fork `branches`, run concurrently, join. Bounded fork-join
74    /// (concurrency capped by the interpreter, 4.2).
75    Parallel {
76        /// Independent step sequences run in parallel.
77        branches: Vec<Vec<PipelineStep>>,
78    },
79    /// EPIC 4.1: run `body` until `until` holds or `max_iters` is hit
80    /// (mandatory bound — no unbounded loops). Reuses `OnFail::Retry` rewind
81    /// in the interpreter (4.2).
82    Loop {
83        /// The loop body.
84        body: Vec<PipelineStep>,
85        /// Exit condition, checked each iteration.
86        until: Condition,
87        /// Hard iteration cap (`> 0`, enforced by typecheck).
88        max_iters: u64,
89    },
90    /// EPIC 4.1: human-in-the-loop pause — checkpoint and wait for
91    /// `resume_signal`, then seed-and-skip resume (ADR-0053 `per_tool_call`
92    /// checkpoint; round-trip in 4.3).
93    Suspend {
94        /// Signal name that resumes the run.
95        resume_signal: String,
96    },
97    /// EPIC 4.4: a bounded dynamic region. Spawns up to `max_spawns` children
98    /// (concurrency-capped by `max_concurrency`) drawn from `spawns`, each
99    /// attenuated to `envelope`. Build-time verified: every spawn ⊆ envelope ⊆
100    /// owner ⊆ root `[allow]` (see `tau check governance`). Runtime execution +
101    /// membership/attenuation/bounds counters land in EPIC 4.5; the interpreter
102    /// meets it with `RuntimeError::DynamicRegionRequiresRuntimeGate` until then.
103    Dynamic {
104        /// Region capability envelope (ceiling); every spawn ⊆ this.
105        envelope: CapabilityRequirements,
106        /// Spawnable per-kind agent definitions this region may launch.
107        spawns: Vec<DynamicSpawn>,
108        /// Hard cap on total spawns (`> 0`, enforced at author time).
109        max_spawns: u64,
110        /// Hard cap on concurrent spawns (`0 < n <= max_spawns`).
111        max_concurrency: u64,
112    },
113}

115#[cfg(test)]
116mod tests {
117    use super::*;
118    use crate::ids::{AgentId, CheckId};
119
120    #[test]
121    fn pipeline_serde_round_trips() {
122        let p = Pipeline {
123            steps: alloc::vec![PipelineStep {
124                id: PipelineStepId("writer".into()),
125                run: StepRun::Agent(AgentId("writer".into())),
126                input: "${steps.gather.output}".into(),
127            }],
128        };
129        let bytes = serde_json::to_vec(&p).expect("serializes");
130        let back: Pipeline = serde_json::from_slice(&bytes).expect("deserializes");
131        assert_eq!(p, back);
132    }
133
134    #[test]
135    fn check_step_serde_round_trips() {
136        let p = Pipeline {
137            steps: alloc::vec![PipelineStep {
138                id: PipelineStepId("check-report".into()),
139                run: StepRun::Check(CheckId("c".into())),
140                input: "${input}".into(),
141            }],
142        };
143        let bytes = serde_json::to_vec(&p).expect("serializes");
144        let back: Pipeline = serde_json::from_slice(&bytes).expect("deserializes");
145        assert_eq!(p, back);
