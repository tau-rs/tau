1//! Sequential pipeline: an ordered list of steps the engine executes
2//! top-to-bottom, threading each step's output to later steps via
3//! `${steps.<id>.output}` templating.
4
5use alloc::string::String;
6use alloc::vec::Vec;
7use serde::{Deserialize, Serialize};
8
9use crate::check::Condition;
10use crate::ids::{AgentId, PipelineStepId, StepId, ToolId};
11
12/// An ordered, engine-sequenced pipeline of steps.
13#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
14#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
15pub struct Pipeline {
16    /// Steps, executed top-to-bottom in this order.
17    pub steps: Vec<PipelineStep>,
18}
19
20/// One step in a [`Pipeline`].
21#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
22#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
23pub struct PipelineStep {
24    /// Handle for this step; its output is addressable as
25    /// `steps.<id>.output` by later steps.
26    pub id: PipelineStepId,
27    /// What this step runs.
28    pub run: StepRun,
29    /// Input template (`${input}`, `${steps.<id>.output}`).
30    pub input: String,
31}
32
33/// What a [`PipelineStep`] executes — a reference to an existing node.
34#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
35#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
36pub enum StepRun {
37    /// Run an agent node by id.
38    Agent(AgentId),
39    /// Invoke a tool node by id.
40    Tool(ToolId),
41    /// Run a deterministic step node by id.
42    Deterministic(StepId),
43    /// Evaluate a postcondition check by id.
44    ///
45    /// The referenced [`CheckId`](crate::ids::CheckId) must exist in
46    /// `workflow.checks`; this is enforced by the typecheck stage.
47    /// Runtime evaluation is implemented in `tau-runtime-core`'s
48    /// pipeline executor (Task 19).
49    Check(crate::ids::CheckId),
50    /// EPIC 4.1: run `then` if `on` holds, else `otherwise` (may be empty).
51    /// Executed by the interpreter in 4.2.
52    Branch {
53        /// The branch condition.
54        on: Condition,
55        /// Steps run when `on` holds.
56        then: Vec<PipelineStep>,
57        /// Steps run when `on` does not hold (may be empty).
58        otherwise: Vec<PipelineStep>,
59    },
60    /// EPIC 4.1: fork `branches`, run concurrently, join. Bounded fork-join
61    /// (concurrency capped by the interpreter, 4.2).
62    Parallel {
63        /// Independent step sequences run in parallel.
64        branches: Vec<Vec<PipelineStep>>,
65    },
66    /// EPIC 4.1: run `body` until `until` holds or `max_iters` is hit
67    /// (mandatory bound — no unbounded loops). Reuses `OnFail::Retry` rewind
68    /// in the interpreter (4.2).
69    Loop {
70        /// The loop body.
71        body: Vec<PipelineStep>,
72        /// Exit condition, checked each iteration.
73        until: Condition,
74        /// Hard iteration cap (`> 0`, enforced by typecheck).
75        max_iters: u64,
76    },
77    /// EPIC 4.1: human-in-the-loop pause — checkpoint and wait for
78    /// `resume_signal`, then seed-and-skip resume (ADR-0053 `per_tool_call`
79    /// checkpoint; round-trip in 4.3).
80    Suspend {
81        /// Signal name that resumes the run.
82        resume_signal: String,
83    },
84}

86#[cfg(test)]
87mod tests {
88    use super::*;
89    use crate::ids::{AgentId, CheckId};
90
91    #[test]
92    fn pipeline_serde_round_trips() {
93        let p = Pipeline {
94            steps: alloc::vec![PipelineStep {
95                id: PipelineStepId("writer".into()),
96                run: StepRun::Agent(AgentId("writer".into())),
97                input: "${steps.gather.output}".into(),
98            }],
99        };
100        let bytes = serde_json::to_vec(&p).expect("serializes");
101        let back: Pipeline = serde_json::from_slice(&bytes).expect("deserializes");
102        assert_eq!(p, back);
103    }
104
105    #[test]
106    fn check_step_serde_round_trips() {
