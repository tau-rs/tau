1//! Sequential pipeline: an ordered list of steps the engine executes
2//! top-to-bottom, threading each step's output to later steps via
3//! `${steps.<id>.output}` templating.
4
5use alloc::string::String;
6use alloc::vec::Vec;
7use serde::{Deserialize, Serialize};
8
9use crate::capability::CapabilityRequirements;
10use crate::check::Condition;
11use crate::ids::{AgentId, PipelineStepId, StepId, ToolId};
12
13/// An ordered, engine-sequenced pipeline of steps.
14#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
15#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
16pub struct Pipeline {
17    /// Steps, executed top-to-bottom in this order.
18    pub steps: Vec<PipelineStep>,
19}
20
21impl Pipeline {
22    /// Id of the last top-level step that records an output — skips
23    /// trailing `Check` and `Suspend` steps. `None` if no step qualifies.
24    pub fn final_leaf_step_id(&self) -> Option<&PipelineStepId> {
25        self.steps
26            .iter()
27            .rev()
28            .find(|s| !matches!(s.run, StepRun::Check(_) | StepRun::Suspend { .. }))
29            .map(|s| &s.id)
30    }
31}
32
33/// One step in a [`Pipeline`].
34#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
35#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
36pub struct PipelineStep {
37    /// Handle for this step; its output is addressable as
38    /// `steps.<id>.output` by later steps.
39    pub id: PipelineStepId,
40    /// What this step runs.
41    pub run: StepRun,
42    /// Input template (`${input}`, `${steps.<id>.output}`).
43    pub input: String,
44}
45
46/// One spawnable per-kind agent definition inside a [`StepRun::Dynamic`]
47/// region, resolved with its capability grant so the runtime gate
48/// (EPIC 4.5) is self-contained (no re-resolution against `tau.toml`).
49#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
50#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
51pub struct DynamicSpawn {
52    /// The agent-kind name (`[agent.kinds.<kind>]`).
53    pub kind: String,
54    /// The kind's resolved capability grant.
55    pub capabilities: CapabilityRequirements,
56}
57
58/// What a [`PipelineStep`] executes — a reference to an existing node.
59#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
60#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
61pub enum StepRun {
62    /// Run an agent node by id.
63    Agent(AgentId),
64    /// Invoke a tool node by id.
65    Tool(ToolId),
66    /// Run a deterministic step node by id.
67    Deterministic(StepId),
68    /// Evaluate a postcondition check by id.
69    ///
70    /// The referenced [`CheckId`](crate::ids::CheckId) must exist in
71    /// `workflow.checks`; this is enforced by the typecheck stage.
72    /// Runtime evaluation is implemented in `tau-runtime-core`'s
73    /// pipeline executor (Task 19).
74    Check(crate::ids::CheckId),
75    /// EPIC 4.1: run `then` if `on` holds, else `otherwise` (may be empty).
76    /// Executed by the interpreter in 4.2.
77    Branch {
78        /// The branch condition.
79        on: Condition,
80        /// Steps run when `on` holds.
81        then: Vec<PipelineStep>,
82        /// Steps run when `on` does not hold (may be empty).
83        otherwise: Vec<PipelineStep>,
84    },
85    /// EPIC 4.1: fork `branches`, run concurrently, join. Bounded fork-join
86    /// (concurrency capped by the interpreter, 4.2).
87    Parallel {
88        /// Independent step sequences run in parallel.
89        branches: Vec<Vec<PipelineStep>>,
90    },
91    /// EPIC 4.1: run `body` until `until` holds or `max_iters` is hit
92    /// (mandatory bound — no unbounded loops). Reuses `OnFail::Retry` rewind
93    /// in the interpreter (4.2).
94    Loop {
95        /// The loop body.
96        body: Vec<PipelineStep>,
97        /// Exit condition, checked each iteration.
98        until: Condition,
99        /// Hard iteration cap (`> 0`, enforced by typecheck).
100        max_iters: u64,
101    },
102    /// EPIC 4.1: human-in-the-loop pause — checkpoint and wait for
103    /// `resume_signal`, then seed-and-skip resume (ADR-0053 `per_tool_call`
104    /// checkpoint; round-trip in 4.3).
105    Suspend {
106        /// Signal name that resumes the run.
107        resume_signal: String,
108    },
109    /// EPIC 4.4: a bounded dynamic region. Spawns up to `max_spawns` children
110    /// (concurrency-capped by `max_concurrency`) drawn from `spawns`, each
111    /// attenuated to `envelope`. Build-time verified: every spawn ⊆ envelope ⊆
112    /// owner ⊆ root `[allow]` (see `tau check governance`). Runtime execution +
113    /// membership/attenuation/bounds counters land in EPIC 4.5; the interpreter
114    /// meets it with `RuntimeError::DynamicRegionRequiresRuntimeGate` until then.
115    Dynamic {
116        /// Region capability envelope (ceiling); every spawn ⊆ this.
117        envelope: CapabilityRequirements,
118        /// Spawnable per-kind agent definitions this region may launch.
119        spawns: Vec<DynamicSpawn>,
120        /// Hard cap on total spawns (`> 0`, enforced at author time).
121        max_spawns: u64,
122        /// Hard cap on concurrent spawns (`0 < n <= max_spawns`).
123        max_concurrency: u64,
124    },
125}

127#[cfg(test)]
128mod tests {
129    use super::*;
130    use crate::ids::{AgentId, CheckId};
131
132    #[test]
133    fn pipeline_serde_round_trips() {
134        let p = Pipeline {
135            steps: alloc::vec![PipelineStep {
136                id: PipelineStepId("writer".into()),
137                run: StepRun::Agent(AgentId("writer".into())),
138                input: "${steps.gather.output}".into(),
139            }],
140        };
141        let bytes = serde_json::to_vec(&p).expect("serializes");
142        let back: Pipeline = serde_json::from_slice(&bytes).expect("deserializes");
143        assert_eq!(p, back);
144    }
145
146    #[test]
147    fn check_step_serde_round_trips() {
148        let p = Pipeline {
149            steps: alloc::vec![PipelineStep {
150                id: PipelineStepId("check-report".into()),
151                run: StepRun::Check(CheckId("c".into())),
152                input: "${input}".into(),
153            }],
154        };
155        let bytes = serde_json::to_vec(&p).expect("serializes");
156        let back: Pipeline = serde_json::from_slice(&bytes).expect("deserializes");
157        assert_eq!(p, back);
158    }
159
160    #[test]
161    fn final_leaf_skips_trailing_check_and_suspend() {
162        let p = Pipeline {
163            steps: alloc::vec![
164                PipelineStep {
165                    id: PipelineStepId("a".into()),
166                    run: StepRun::Agent(AgentId("x".into())),
167                    input: "${input}".into(),
168                },
169                PipelineStep {
170                    id: PipelineStepId("b".into()),
171                    run: StepRun::Agent(AgentId("y".into())),
172                    input: "${steps.a.output}".into(),
173                },
174                PipelineStep {
175                    id: PipelineStepId("c".into()),
176                    run: StepRun::Check(CheckId("c".into())),
177                    input: "${steps.b.output}".into(),
178                },
179                PipelineStep {
180                    id: PipelineStepId("s".into()),
181                    run: StepRun::Suspend {
182                        resume_signal: "sig".into(),
183                    },
184                    input: "${steps.b.output}".into(),
185                },
186            ],
187        };
188        assert_eq!(p.final_leaf_step_id(), Some(&PipelineStepId("b".into())));
189
190        let checks_only = Pipeline {
191            steps: alloc::vec![PipelineStep {
192                id: PipelineStepId("c".into()),
193                run: StepRun::Check(CheckId("c".into())),
194                input: "${input}".into(),
195            }],
196        };
197        assert_eq!(checks_only.final_leaf_step_id(), None);
198    }
