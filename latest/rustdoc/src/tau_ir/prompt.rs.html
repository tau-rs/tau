1//! Where an agent's system prompt comes from (D6-B).
2//!
3//! Before v2.5.0 an `Agent.prompt` was a bare `String`. That conflated two
4//! very different things: a literal inline prompt, and — for `system_file`
5//! agents — the *path* the prompt was read from. Lowering stored the path
6//! string and the interpreter sent it verbatim as the system prompt, so a
7//! `system_file` agent ran with a file path as its prompt and the bundle hash
8//! pinned the path, not the content (non-hermetic).
9//!
10//! `PromptSource` replaces that conflation with an explicit source:
11//! - `Inline(String)` — a literal prompt. Serializes as a **bare string**, so
12//!   every pre-2.5 module (whose `prompt` was a bare string) reads back as
13//!   `Inline` unchanged and re-serializes byte-identically. This is the
14//!   additive-compat property that lets `ir_format` bump `v2.4.0 -> v2.5.0`
15//!   under the #442 policy.
16//! - `Asset { asset }` — a content-addressed reference (`"sha256:<64 hex>"`)
17//!   into the bundle's asset store. File-backed prompts lower to this: the
18//!   bytes are read at build time, hashed, and stored as an asset. Paths never
19//!   enter the IR.
20
21use alloc::string::String;
22// The schemars `pattern(...)` derive expands to a `.to_string()` call on the
23// pattern literal; bring `ToString` into scope for the `schema` build only.
24#[cfg(feature = "schema")]
25use alloc::string::ToString;
26use serde::{Deserialize, Serialize};
27
28/// Where an agent's system prompt comes from.
29///
30/// `#[serde(untagged)]` keeps `Inline` on the wire as a bare string —
31/// byte-identical to the pre-2.5 `prompt: String` field.
32#[derive(Debug, Clone, Eq, PartialEq, Serialize, Deserialize)]
33#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
34#[serde(untagged)]
35pub enum PromptSource {
36    /// A literal system prompt. On the wire: a bare JSON string.
37    Inline(String),
38    /// A content-addressed reference into the bundle's asset store.
39    /// On the wire: `{"asset":"sha256:<64 hex>"}`.
40    Asset(AssetRef),
41}
42
43/// A content-addressed reference to a bundle asset.
44///
45/// A distinct `deny_unknown_fields` struct (rather than an inline
46/// `Asset { asset }` variant) so the generated JSON Schema emits
47/// `additionalProperties: false` and rejects stray keys in the asset object.
48#[derive(Debug, Clone, Eq, PartialEq, Serialize, Deserialize)]
49#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
50#[serde(deny_unknown_fields)]
51pub struct AssetRef {
52    /// The asset hash, `"sha256:" + 64 lowercase hex`.
53    #[cfg_attr(feature = "schema", schemars(pattern(r"^sha256:[0-9a-f]{64}$")))]
54    pub asset: String,
55}
56
57impl PromptSource {
58    /// Construct an inline prompt.
59    pub fn inline(s: impl Into<String>) -> Self {
60        Self::Inline(s.into())
61    }
62
63    /// Construct an asset-backed prompt from a hash string.
64    ///
65    /// Does not validate the hash; use [`PromptSource::validate`] (or
66    /// [`is_well_formed_asset_hash`]) for that.
67    pub fn asset(hash: impl Into<String>) -> Self {
68        Self::Asset(AssetRef { asset: hash.into() })
69    }
70
71    /// The inline prompt text, if this is an `Inline` source.
72    pub fn as_inline(&self) -> Option<&str> {
73        match self {
74            Self::Inline(s) => Some(s),
75            Self::Asset(_) => None,
76        }
77    }
78
79    /// The asset hash, if this is an `Asset` source.
80    pub fn asset_hash(&self) -> Option<&str> {
81        match self {
82            Self::Asset(a) => Some(&a.asset),
83            Self::Inline(_) => None,
84        }
85    }
86
87    /// Validate that an `Asset` reference is well-formed. `Inline` always
88    /// passes. Returns the malformed hash string on failure so callers can
89    /// build a structured error (never `IrError::Parse`).
90    pub fn validate(&self) -> Result<(), &str> {
91        match self {
92            Self::Inline(_) => Ok(()),
93            Self::Asset(a) => {
94                if is_well_formed_asset_hash(&a.asset) {
95                    Ok(())
96                } else {
97                    Err(&a.asset)
98                }
99            }
100        }
101    }
102}
103
104impl From<String> for PromptSource {
105    fn from(s: String) -> Self {
106        Self::Inline(s)
107    }
108}
109
110impl From<&str> for PromptSource {
111    fn from(s: &str) -> Self {
112        Self::Inline(s.into())
113    }
114}
115
116/// True iff `s` is `"sha256:"` followed by exactly 64 lowercase hex digits.
117pub fn is_well_formed_asset_hash(s: &str) -> bool {
118    match s.strip_prefix("sha256:") {
119        Some(hex) => {
120            hex.len() == 64
121                && hex
122                    .bytes()
123                    .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
124        }
125        None => false,
126    }
127}

129#[cfg(test)]
130mod tests {
131    use super::*;
132
133    const GOOD_HASH: &str =
134        "sha256:e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855";
135
136    /// `Inline` serializes as a BARE STRING — byte-identical to the pre-2.5
137    /// `prompt: String` wire form.
138    #[test]
139    fn inline_serializes_as_bare_string() {
140        let p = PromptSource::inline("You are a careful assistant.");
141        let json = serde_json::to_string(&p).unwrap();
142        assert_eq!(json, "\"You are a careful assistant.\"");
143    }
144
145    /// A bare JSON string deserializes back into `Inline` (pre-2.5 modules).
146    #[test]
147    fn bare_string_deserializes_as_inline() {
148        let p: PromptSource = serde_json::from_str("\"hello\"").unwrap();
149        assert_eq!(p, PromptSource::inline("hello"));
150    }
151
152    /// `Asset` serializes as `{"asset":"sha256:..."}` and round-trips.
153    #[test]
154    fn asset_round_trips_as_object() {
155        let p = PromptSource::asset(GOOD_HASH);
156        let json = serde_json::to_string(&p).unwrap();
157        assert_eq!(json, alloc::format!("{{\"asset\":\"{GOOD_HASH}\"}}"));
158        let back: PromptSource = serde_json::from_str(&json).unwrap();
159        assert_eq!(back, p);
160    }
161
162    /// An asset object with a stray key is rejected (deny_unknown_fields).
163    #[test]
164    fn asset_with_unknown_key_is_rejected() {
165        let raw = alloc::format!("{{\"asset\":\"{GOOD_HASH}\",\"extra\":1}}");
166        let parsed: Result<PromptSource, _> = serde_json::from_str(&raw);
167        assert!(
168            parsed.is_err(),
169            "unknown key in asset object must be rejected"
170        );
171    }
172
173    #[test]
174    fn hash_wellformedness() {
175        assert!(is_well_formed_asset_hash(GOOD_HASH));
176        // wrong prefix
177        assert!(!is_well_formed_asset_hash(
178            "md5:e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"
179        ));
180        // too short
181        assert!(!is_well_formed_asset_hash("sha256:abcd"));
182        // uppercase hex not allowed
183        assert!(!is_well_formed_asset_hash(
184            "sha256:E3B0C44298FC1C149AFBF4C8996FB92427AE41E4649B934CA495991B7852B855"
185        ));
186        // non-hex char
187        assert!(!is_well_formed_asset_hash(
188            "sha256:g3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"
189        ));
190        // no prefix at all
191        assert!(!is_well_formed_asset_hash("deadbeef"));
192    }
193
194    #[test]
195    fn validate_reports_malformed_hash() {
196        assert!(PromptSource::inline("x").validate().is_ok());
197        assert!(PromptSource::asset(GOOD_HASH).validate().is_ok());
198        assert_eq!(
199            PromptSource::asset("sha256:nope").validate(),
200            Err("sha256:nope")
201        );
202    }
203
204    #[test]
205    fn accessors() {
206        assert_eq!(PromptSource::inline("hi").as_inline(), Some("hi"));
207        assert_eq!(PromptSource::inline("hi").asset_hash(), None);
208        assert_eq!(PromptSource::asset(GOOD_HASH).as_inline(), None);
209        assert_eq!(PromptSource::asset(GOOD_HASH).asset_hash(), Some(GOOD_HASH));
210    }
