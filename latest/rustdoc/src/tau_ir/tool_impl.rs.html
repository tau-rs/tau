1//! Tool implementation references.
2//!
3//! A [`Tool`](crate::Tool) node carries a [`ToolImpl`] that distinguishes native
4//! tools (statically linked Rust) from MCP-contracted tools (external
5//! servers reached via the MCP wire). The lowering pass resolves
6//! [`ToolImpl::Native::content_hash`] and [`ToolImpl::Mcp::contract_hash`]
7//! at build time so every IR module is fully hashable per D-6.
8
9use alloc::string::String;
10use serde::{Deserialize, Serialize};
11
12use crate::capability::CapabilityRequirements;
13
14/// 32-byte content hash (SHA-256 output) used to pin tool implementations
15/// and MCP contracts at build time.
16pub type Hash256 = [u8; 32];
17
18/// A reference to a statically linked native tool by symbolic name.
19///
20/// The symbolic name (e.g. `"ReadTemp"`) is the Rust identifier of the
21/// `impl Tool for X` type. The lowering pass resolves it against the
22/// project's native tool registry; AOT (β.7) lowers the call site
23/// directly. v0's interpreter dispatches by name through a
24/// `NativeFnRegistry` injected at runtime.
25#[derive(Debug, Clone, Eq, PartialEq, Ord, PartialOrd, Serialize, Deserialize)]
26#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
27pub struct NativeFnRef {
28    /// Symbolic name of the Rust `Tool` impl.
29    pub name: String,
30}
31
32/// How a [`crate::Tool`] node's behavior is provided at runtime.
33#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
34#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
35pub enum ToolImpl {
36    /// Statically linked native tool.
37    Native {
38        /// Reference to the native Rust impl by symbolic name.
39        fn_ref: NativeFnRef,
40        /// Hash of the impl's source bytes (Rust source, dependencies'
41        /// content hashes). Participates in the IR module hash.
42        content_hash: Hash256,
43    },
44    /// MCP-contracted external server.
45    Mcp {
46        /// MCP server URL (e.g. `"https://mcp.weather.com"`).
47        url: String,
48        /// Content hash of the MCP contract (the cached schema + capability
49        /// declaration the server advertises at handshake). Participates in
50        /// the IR module hash so a contract drift invalidates the bundle.
51        contract_hash: Hash256,
52        /// The subset of capabilities this MCP server is bounded to (a
53        /// subset of the contract's declared capabilities; narrowed by
54        /// `tau.toml` overrides).
55        capability_subset: CapabilityRequirements,
56        /// The name passed on the MCP wire (server-side tool name).
57        /// Differs from this `Tool` node's `ToolId` because lowering
58        /// expands one author-side entry (`weather`) into N IR nodes
59        /// (`weather.get_forecast`, `weather.get_current`, ...); each
60        /// expanded node carries the server-side name to forward on
61        /// `tools/call` requests.
62        server_tool_name: String,
63    },
64    /// Sub-workflow spawn: invoking this tool runs the named agent (in the
65    /// same `IrModule`) as a child loop with empty initial history. The
66    /// child's final assistant text (or empty string) becomes the tool
67    /// result body.
68    ///
69    /// v0 limitation: tool input args are NOT forwarded to the child as a
70    /// User message — the child runs with empty initial messages and its
71    /// own prompt + LLM script drive its behavior. β.7 (AOT codegen) is
72    /// the natural place to thread arg-forwarding.
73    Subflow {
74        /// Agent id (within this `IrModule`'s `workflow.agents`) to spawn.
75        target: crate::ids::AgentId,
76    },
77    /// Deterministic step: invoking this tool calls the pure Rust function
78    /// named by the step's `fn_ref` via the dispatcher's
79    /// `deterministic_registry()`. The function's return value becomes the
80    /// tool result body. No LLM, no I/O.
81    Step {
82        /// Step id (within this `IrModule`'s `workflow.steps`).
83        id: crate::ids::StepId,
84    },
85}

87#[cfg(test)]
88mod tests {
89    use super::*;
90    use crate::ids::{AgentId, StepId};
91    use alloc::string::ToString;
92
93    #[test]
94    fn tool_impl_subflow_round_trips_canonical_json() {
95        let original = ToolImpl::Subflow {
96            target: AgentId("child-agent".to_string()),
97        };
98        let bytes = serde_json::to_vec(&original).expect("serialize");
99        let decoded: ToolImpl = serde_json::from_slice(&bytes).expect("deserialize");
100        assert_eq!(original, decoded);
101    }
102
103    #[test]
104    fn tool_impl_step_round_trips_canonical_json() {
105        let original = ToolImpl::Step {
106            id: StepId("normalize".to_string()),
107        };
108        let bytes = serde_json::to_vec(&original).expect("serialize");
109        let decoded: ToolImpl = serde_json::from_slice(&bytes).expect("deserialize");
110        assert_eq!(original, decoded);
111    }
