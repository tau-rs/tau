16/// Semver-shaped IR format version (D-6).
17///
18/// Bumps follow semver rules:
19/// - MAJOR for breaking shape changes (removed node type, removed
20///   required field, changed lowering contract).
21/// - MINOR for additive changes (new optional field, new variant of a
22///   `#[non_exhaustive]` enum).
23/// - PATCH for spec-only edits with no IR-shape effect.
24#[derive(Debug, Clone, Eq, PartialEq, Ord, PartialOrd, Hash, Serialize, Deserialize)]
25#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
26pub struct IrFormatVersion(pub String);
27
28impl IrFormatVersion {
29    /// Current IR format version emitted by this `tau-ir` crate.
30    // Track 1 (per-agent model resolution) is a breaking change (Agent.model →
31    // model_ref, JudgeRef variant rename), so its MAJOR bump supersedes Track 2's
32    // MINOR v1.3.0 (output_schema, additive). Reconciled to v2.0.0.
33    // MINOR v2.1.0: Agent.durable additive optional field (ADR-0053).
34    // MINOR v2.2.0: CheckpointGranularity::PerToolCall variant +
35    // TurnCheckpoint.pending_tool_uses additive optional field (ADR-0053
36    // follow-up). Byte-stable when durable absent / PerTurn.
37    // MINOR v2.3.0: Durability gains the `Intent(survive-restarts)` variant
38    // (EPIC 6.1) alongside the explicit form. Optional field, additive shape.
39    // MINOR v2.4.0: StepRun gains Branch/Parallel/Loop/Suspend (additive; EPIC 4.1).
40    // MINOR v2.5.0: Agent.prompt String -> PromptSource (untagged: Inline | Asset)
41    // for the content-addressed asset store (D6-B). `Inline` serializes as a bare
42    // string, so pre-2.5 modules parse and re-serialize byte-identically.
43    pub const CURRENT: &'static str = "v2.5.0";
44
45    /// Major component of `CURRENT`. Kept as a literal (const string
46    /// parsing is awkward) and pinned to `CURRENT` by
47    /// `current_major_agrees_with_current_string`.
48    pub const CURRENT_MAJOR: u64 = 2;
49
50    /// Construct the version this crate emits.
51    pub fn current() -> Self {
52        Self(Self::CURRENT.into())
53    }
54
55    /// Parse the semver MAJOR out of the version string.
56    ///
57    /// Strips an optional leading `v`, then parses the integer up to the
58    /// first `.` (or the end of the string). Returns `Err(())` when that
59    /// leading segment is empty or not a `u64`, so `v2.4.0`, `2.4.0`, and a
60    /// dotless `v2` all yield the major, while `""`, `v.4`, and `garbage`
61    /// are rejected.
62    // `Result<u64, ()>` is the pinned Task-2 interface (D8-B brief); the
63    // caller only branches on Ok/Err and doesn't need a richer error type
64    // here — `FormatUnparseable` carries the diagnostic instead.
65    #[allow(clippy::result_unit_err)]
66    pub fn major(&self) -> Result<u64, ()> {
67        let s = self.0.strip_prefix('v').unwrap_or(&self.0);
68        let head = s.split('.').next().ok_or(())?;
69        if head.is_empty() {
70            return Err(());
71        }
72        head.parse::<u64>().map_err(|_| ())
73    }
74}

76/// The container for one workflow's IR.
77///
78/// `tau build` emits one `IrModule` per workflow (one per project for
79/// v0). `tau verify --bundle` re-builds and asserts byte-equality of
80/// the canonical form (D-6).
81#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
82#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
83pub struct IrModule {
84    /// IR language version (D-6 — separate from `tau_version`).
85    pub ir_format: IrFormatVersion,
86    /// Version of the `tau-ir-lower` crate that emitted this module (stamped
87    /// from its `CARGO_PKG_VERSION` at lowering). Workspace-versioned, so it
88    /// equals the `tau` CLI binary version today. Semver-shaped (e.g. `"0.X.Y"`).
89    pub tau_version: String,
90    /// Target triple this module was lowered for.
91    pub target: TargetTriple,
92    /// The workflow itself.
93    pub workflow: Workflow,
94    /// Trigger bindings — invocation metadata, a SIBLING of `workflow`
95    /// (triggers are about *how* tau is invoked, not the call graph).
96    /// `skip_serializing_if` + `default` means a trigger-less module emits
97    /// no `triggers` key and hashes identically to a pre-trigger module
98    /// (Option B / ADR-0044 §D1); older modules with no key read back as empty.
99    #[serde(default, skip_serializing_if = "Vec::is_empty")]
100    pub triggers: Vec<TriggerBinding>,
101}

103/// The set of nodes + edges that make up one workflow.
104#[derive(Debug, Clone, PartialEq, Default, Serialize, Deserialize)]
105#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
106pub struct Workflow {
107    /// Agent nodes by id.
108    pub agents: BTreeMap<AgentId, Agent>,
109    /// Tool nodes by id.
110    pub tools: BTreeMap<ToolId, Tool>,
111    /// Deterministic step nodes by id.
112    pub steps: BTreeMap<StepId, Deterministic>,
113    /// Subflow edges.
114    pub edges: Vec<SubflowEdge>,
115    /// Per-tool capability requirements. Derived from `tools` but
116    /// stored explicitly for the bundle's `tau.caps` custom section.
117    pub capability_table: CapabilityTable,
118    /// Optional engine-sequenced pipeline. `None` preserves single-entry
119    /// behavior (run the named entry agent). `Some` => `run_pipeline`.
120    pub pipeline: Option<Pipeline>,
121    /// Postcondition checks, keyed by id. Positioned in the pipeline via
122    /// `StepRun::Check`.
123    #[serde(
124        default,
125        skip_serializing_if = "alloc::collections::BTreeMap::is_empty"
126    )]
127    pub checks: alloc::collections::BTreeMap<crate::ids::CheckId, crate::check::Check>,
128}

143#[cfg(test)]
144mod tests {
145    use super::*;
146
147    #[test]
148    fn ir_format_version_is_v2_5_0() {
149        assert_eq!(IrFormatVersion::CURRENT, "v2.5.0");
150        assert_eq!(IrFormatVersion::current().0, "v2.5.0");
151    }
152
153    #[test]
154    fn major_parses_leading_v_and_bare() {
155        assert_eq!(IrFormatVersion("v2.4.0".into()).major(), Ok(2));
156        assert_eq!(IrFormatVersion("2.4.0".into()).major(), Ok(2));
157        assert_eq!(IrFormatVersion("v10.0.0".into()).major(), Ok(10));
158    }
159
160    #[test]
161    fn major_rejects_malformed() {
162        assert_eq!(IrFormatVersion("".into()).major(), Err(()));
163        assert_eq!(IrFormatVersion("garbage".into()).major(), Err(()));
164        assert_eq!(IrFormatVersion("v.4".into()).major(), Err(()));
165        assert_eq!(IrFormatVersion("v".into()).major(), Err(()));
166    }
167
168    #[test]
169    fn current_major_agrees_with_current_string() {
170        // A future CURRENT bump must not silently desync CURRENT_MAJOR.
171        assert_eq!(
172            IrFormatVersion::current().major(),
173            Ok(IrFormatVersion::CURRENT_MAJOR)
174        );
175    }
176
177    #[test]
178    fn default_workflow_has_empty_checks_and_serializes_without_checks_key() {
179        let wf = Workflow::default();
180        assert!(
181            wf.checks.is_empty(),
182            "default Workflow must have empty checks"
183        );
184
185        // Serialize and confirm "checks" key is absent (byte-stability).
186        let bytes = serde_json::to_vec(&wf).expect("serializes");
187        let obj: serde_json::Value = serde_json::from_slice(&bytes).expect("valid json");
188        assert!(
189            obj.get("checks").is_none(),
190            "empty checks must not appear in serialized Workflow; got: {}",
191            String::from_utf8_lossy(&bytes)
192        );
