1//! Capability requirements as carried in the IR.
2//!
3//! v0 wraps `tau_domain::Capability` (the existing source-of-truth shape) in
4//! a `CapabilityTable` newtype keyed by [`crate::ToolId`]. Per the D-3b
5//! decision, the lowering pass intersects this table against the target
6//! triple's `supported_shapes` at build time and refuses the build on any
7//! miss.
8
9use alloc::collections::BTreeMap;
10use alloc::vec::Vec;
11use serde::{Deserialize, Serialize};
12use tau_domain::Capability;
13
14use crate::ids::ToolId;
15
16/// The capability-requirement set for one tool.
17///
18/// Re-export shape over `Vec<tau_domain::Capability>` — the IR does not
19/// re-define what a capability *is*; it just carries the existing type
20/// across the boundary. Future evolution (capability narrowing in the IR
21/// pre-hash, etc.) lands here.
22#[derive(Debug, Clone, PartialEq, Default, Serialize, Deserialize)]
23#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
24pub struct CapabilityRequirements {
25    /// Declared capabilities; order is taken from the source TOML verbatim
26    /// (capability list order is taken from the source TOML verbatim; two
27    /// permuted capability arrays produce different canonical bytes. If
28    /// order-independent hashing becomes required, sort in the lowering
29    /// parse stage — `Capability` needs `Ord` first.)
30    pub declared: Vec<Capability>,
31}
32
33/// Per-tool capability table for a `Workflow`.
34///
35/// Built by the lowering pass from per-tool TOML declarations; consumed
36/// by the capability-fit check (D-3b) and embedded in the bundle's
37/// `tau.caps` custom section (D-3).
38#[derive(Debug, Clone, PartialEq, Default, Serialize, Deserialize)]
39#[cfg_attr(feature = "schema", derive(schemars::JsonSchema))]
40pub struct CapabilityTable(pub BTreeMap<ToolId, CapabilityRequirements>);
41
42#[cfg(test)]
43mod tests {
44    use super::*;
45    use crate::canonical::{from_canonical_bytes, to_canonical_bytes};
46    use crate::module::{IrFormatVersion, IrModule, Workflow};
47    use crate::node::{Tool, ToolSpec};
48    use crate::tool_impl::ToolImpl;
49    use alloc::string::ToString;
50    use serde_json::Value;
51    use tau_domain::Capability;
52    use tau_ports::target::registry;
53
54    /// D7-B / issue #481: a stray field on a KNOWN capability kind must be
55    /// rejected when the whole `IrModule` is decoded — not just when a bare
56    /// `Capability` is deserialized. This walks the real interchange path
57    /// (`from_canonical_bytes` → `IrModule` → `CapabilityRequirements` →
58    /// `Vec<Capability>` → `Capability`'s strict custom `Deserialize`) to
59    /// prove the strictness survives the IR wrapper types, which use derived
60    /// `Deserialize` with no `deny_unknown_fields` of their own.
61    #[test]
62    fn unknown_field_on_known_kind_rejected_through_ir_module_decode() {
63        let target = registry::list_available().next().unwrap().triple;
64        let tool = Tool {
65            id: ToolId("reader".to_string()),
66            impl_: ToolImpl::Step {
67                id: crate::ids::StepId("noop".to_string()),
68            },
69            capabilities: CapabilityRequirements {
70                // `FsCapability` is `#[non_exhaustive]`; build the fs.read cap
71                // through the same strict decode the interchange path uses.
72                declared: alloc::vec![serde_json::from_str::<Capability>(
73                    r#"{"kind":"fs.read","paths":["/x"]}"#
74                )
75                .unwrap()],
76            },
77            spec: ToolSpec {
78                name: "reader".to_string(),
79                description: "reads".to_string(),
80                input_schema: Value::Null,
81            },
82        };
83        let mut tools = BTreeMap::new();
84        tools.insert(ToolId("reader".to_string()), tool);
85        let m = IrModule {
86            ir_format: IrFormatVersion::current(),
87            tau_version: "0.0.0".to_string(),
88            target,
89            workflow: Workflow {
90                tools,
91                ..Workflow::default()
92            },
93            triggers: Vec::new(),
94        };
95
96        // Sanity: the clean module decodes.
97        let bytes = to_canonical_bytes(&m);
98        from_canonical_bytes(&bytes).expect("clean module round-trips");
99
100        // Inject a stray key onto the fs.read capability and re-encode.
101        let mut v: Value = serde_json::from_slice(&bytes).unwrap();
102        let cap = &mut v["workflow"]["tools"]["reader"]["capabilities"]["declared"][0];
103        assert_eq!(cap["kind"], "fs.read", "test setup: expected fs.read cap");
104        cap.as_object_mut()
105            .unwrap()
106            .insert("bogus".to_string(), Value::from(1));
107        let tampered = serde_json::to_vec(&v).unwrap();
108
109        let err = from_canonical_bytes(&tampered)
110            .expect_err("stray field on a known capability kind must fail decode");
111        let msg = alloc::format!("{err}");
112        assert!(
113            msg.contains("fs.read") && msg.contains("bogus"),
114            "decode error should name the kind and offending field; got: {msg}"
115        );
116    }
117}
