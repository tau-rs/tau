1//! Conformance case: rate-limited error path.
2//!
3//! Asserts:
4//! - The cassette returns 429s exhausting the plugin's retry budget.
5//! - The result is `Err(LlmError::RateLimited { .. })`.
6
7use std::path::Path;
8
9use tau_plugin_test_support::cassette;
10use tau_ports::{CompletionRequest, ContentBlock, LlmBackend, LlmError, LlmProviderMessage};
11
12pub(crate) async fn run<B, F>(build_plugin: &F, cassettes_dir: &Path)
13where
14    B: LlmBackend,
15    F: Fn(String) -> B + Send + Sync,
16{
17    let server = cassette::replay(cassettes_dir.join("error_rate_limited.yaml")).await;
18    let plugin = build_plugin(server.uri().into());
19
20    let mut req = CompletionRequest::new("conformance-model".into());
21    req.messages
22        .push(LlmProviderMessage::user(vec![ContentBlock::Text(
23            "hi".into(),
24        )]));
25    req.max_tokens = Some(20);
26
27    let err = match plugin.complete(req).await {
28        Ok(_) => panic!("[conformance error_rate_limited] expected Err, got Ok"),
29        Err(e) => e,
30    };
31    let LlmError::RateLimited { .. } = err else {
32        panic!("[conformance error_rate_limited] expected LlmError::RateLimited, got {err:?}",);
33    };
34}
